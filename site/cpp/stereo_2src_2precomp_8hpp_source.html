100template<typename T> inline int compressElems( T* ptr, const uchar* mask, int mstep, int count )
101{
102 int i, j;
103 for( i = j = 0; i < count; i++ )
104 if( mask[i*mstep] )
105 {
106 if( i > j )
107 ptr[j] = ptr[i];
108 j++;
109 }
110 return j;
111}

113static inline bool haveCollinearPoints( const Mat& m, int count )
114{
115 int j, k, i = count-1;
116 const Point2f* ptr = m.ptr<Point2f>();
117
118 // check that the i-th selected point does not belong
119 // to a line connecting some previously selected points
120 // also checks that points are not too close to each other
121 for( j = 0; j < i; j++ )
122 {
123 double dx1 = ptr[j].x - ptr[i].x;
124 double dy1 = ptr[j].y - ptr[i].y;
125 for( k = 0; k < j; k++ )
126 {
127 double dx2 = ptr[k].x - ptr[i].x;
128 double dy2 = ptr[k].y - ptr[i].y;
129 if( fabs(dx2*dy1 - dy2*dx1) <= FLT_EPSILON*(fabs(dx1) + fabs(dy1) + fabs(dx2) + fabs(dy2)))
130 return true;
131 }
132 }
133 return false;
134}
