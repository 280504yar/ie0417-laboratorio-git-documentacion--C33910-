46const char* depthToString_(int depth)
47{
48 static const char* depthNames[] = { "CV_8U", "CV_8S", "CV_16U", "CV_16S", "CV_32S", "CV_32F", "CV_64F", "CV_16F",
49 "CV_16BF", "CV_Bool", "CV_64U", "CV_64S", "CV_32U",
50 "CV_8F", "CV_8FNUZ" };
51 return (depth < CV_DEPTH_CURR_MAX && depth >= 0) ? depthNames[depth] : NULL;
52}

54cv::String typeToString_(int type)
55{
56 int depth = CV_MAT_DEPTH(type);
57 int cn = CV_MAT_CN(type);
58 if (depth >= 0 && depth < CV_DEPTH_CURR_MAX)
59 return cv::format("%sC%d", depthToString_(depth), cn);
60 return cv::String();
