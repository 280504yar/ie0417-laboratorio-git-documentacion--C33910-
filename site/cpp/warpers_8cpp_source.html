126void ProjectorBase::setCameraParams(InputArray _K, InputArray _R, InputArray _T)
127{
128 Mat K = _K.getMat(), R = _R.getMat(), T = _T.getMat();
129
130 CV_Assert(K.size() == Size(3, 3) && K.type() == CV_32F);
131 CV_Assert(R.size() == Size(3, 3) && R.type() == CV_32F);
132 CV_Assert((T.size() == Size(1, 3) || T.size() == Size(3, 1)) && T.type() == CV_32F);
133
134 Mat_<float> K_(K);
135 k[0] = K_(0,0); k[1] = K_(0,1); k[2] = K_(0,2);
136 k[3] = K_(1,0); k[4] = K_(1,1); k[5] = K_(1,2);
137 k[6] = K_(2,0); k[7] = K_(2,1); k[8] = K_(2,2);
138
139 Mat_<float> Rinv = R.t();
140 rinv[0] = Rinv(0,0); rinv[1] = Rinv(0,1); rinv[2] = Rinv(0,2);
141 rinv[3] = Rinv(1,0); rinv[4] = Rinv(1,1); rinv[5] = Rinv(1,2);
142 rinv[6] = Rinv(2,0); rinv[7] = Rinv(2,1); rinv[8] = Rinv(2,2);
143
144 Mat_<float> R_Kinv = R * K.inv();
145 r_kinv[0] = R_Kinv(0,0); r_kinv[1] = R_Kinv(0,1); r_kinv[2] = R_Kinv(0,2);
146 r_kinv[3] = R_Kinv(1,0); r_kinv[4] = R_Kinv(1,1); r_kinv[5] = R_Kinv(1,2);
147 r_kinv[6] = R_Kinv(2,0); r_kinv[7] = R_Kinv(2,1); r_kinv[8] = R_Kinv(2,2);
148
149 Mat_<float> K_Rinv = K * Rinv;
150 k_rinv[0] = K_Rinv(0,0); k_rinv[1] = K_Rinv(0,1); k_rinv[2] = K_Rinv(0,2);
151 k_rinv[3] = K_Rinv(1,0); k_rinv[4] = K_Rinv(1,1); k_rinv[5] = K_Rinv(1,2);
152 k_rinv[6] = K_Rinv(2,0); k_rinv[7] = K_Rinv(2,1); k_rinv[8] = K_Rinv(2,2);
153
154 Mat_<float> T_(T.reshape(0, 3));
155 t[0] = T_(0,0); t[1] = T_(1,0); t[2] = T_(2,0);
156}

193Rect PlaneWarper::buildMaps(Size src_size, InputArray K, InputArray R, InputArray T, OutputArray _xmap, OutputArray _ymap)
194{
195 projector_.setCameraParams(K, R, T);
196
197 Point dst_tl, dst_br;
198 detectResultRoi(src_size, dst_tl, dst_br);
199
200 Size dsize(dst_br.x - dst_tl.x + 1, dst_br.y - dst_tl.y + 1);
201 _xmap.create(dsize, CV_32FC1);
202 _ymap.create(dsize, CV_32FC1);
203
204#ifdef HAVE_OPENCL
205 if (ocl::isOpenCLActivated())
206 {
207 ocl::Kernel k("buildWarpPlaneMaps", ocl::stitching::warpers_oclsrc);
208 if (!k.empty())
209 {
210 int rowsPerWI = ocl::Device::getDefault().isIntel() ? 4 : 1;
211 Mat k_rinv(1, 9, CV_32FC1, projector_.k_rinv), t(1, 3, CV_32FC1, projector_.t);
212 UMat uxmap = _xmap.getUMat(), uymap = _ymap.getUMat(),
213 uk_rinv = k_rinv.getUMat(ACCESS_READ), ut = t.getUMat(ACCESS_READ);
214
215 k.args(ocl::KernelArg::WriteOnlyNoSize(uxmap), ocl::KernelArg::WriteOnly(uymap),
216 ocl::KernelArg::PtrReadOnly(uk_rinv), ocl::KernelArg::PtrReadOnly(ut),
217 dst_tl.x, dst_tl.y, 1/projector_.scale, rowsPerWI);
218
219 size_t globalsize[2] = { (size_t)dsize.width, ((size_t)dsize.height + rowsPerWI - 1) / rowsPerWI };
220 if (k.run(2, globalsize, NULL, true))
221 {
222 CV_IMPL_ADD(CV_IMPL_OCL);
223 return Rect(dst_tl, dst_br);
224 }
225 }
226 }
227#endif
228
229 Mat xmap = _xmap.getMat(), ymap = _ymap.getMat();
230
231 float x, y;
232 for (int v = dst_tl.y; v <= dst_br.y; ++v)
233 {
234 for (int u = dst_tl.x; u <= dst_br.x; ++u)
235 {
236 projector_.mapBackward(static_cast<float>(u), static_cast<float>(v), x, y);
237 xmap.at<float>(v - dst_tl.y, u - dst_tl.x) = x;
238 ymap.at<float>(v - dst_tl.y, u - dst_tl.x) = y;
239 }
240 }
241
242 return Rect(dst_tl, dst_br);
243}

283void PlaneWarper::detectResultRoi(Size src_size, Point &dst_tl, Point &dst_br)
284{
285 float tl_uf = std::numeric_limits<float>::max();
286 float tl_vf = std::numeric_limits<float>::max();
287 float br_uf = -std::numeric_limits<float>::max();
288 float br_vf = -std::numeric_limits<float>::max();
289
290 float u, v;
291
292 projector_.mapForward(0, 0, u, v);
293 tl_uf = std::min(tl_uf, u); tl_vf = std::min(tl_vf, v);
294 br_uf = std::max(br_uf, u); br_vf = std::max(br_vf, v);
295
296 projector_.mapForward(0, static_cast<float>(src_size.height - 1), u, v);
297 tl_uf = std::min(tl_uf, u); tl_vf = std::min(tl_vf, v);
298 br_uf = std::max(br_uf, u); br_vf = std::max(br_vf, v);
299
300 projector_.mapForward(static_cast<float>(src_size.width - 1), 0, u, v);
301 tl_uf = std::min(tl_uf, u); tl_vf = std::min(tl_vf, v);
302 br_uf = std::max(br_uf, u); br_vf = std::max(br_vf, v);
303
304 projector_.mapForward(static_cast<float>(src_size.width - 1), static_cast<float>(src_size.height - 1), u, v);
305 tl_uf = std::min(tl_uf, u); tl_vf = std::min(tl_vf, v);
306 br_uf = std::max(br_uf, u); br_vf = std::max(br_vf, v);
307
308 dst_tl.x = static_cast<int>(tl_uf);
309 dst_tl.y = static_cast<int>(tl_vf);
310 dst_br.x = static_cast<int>(br_uf);
311 dst_br.y = static_cast<int>(br_vf);
312}

354void AffineWarper::getRTfromHomogeneous(InputArray H_, Mat &R, Mat &T)
355{
356 Mat H = H_.getMat();
357 CV_Assert(H.size() == Size(3, 3) && H.type() == CV_32F);
358
359 T = Mat::zeros(3, 1, CV_32F);
360 R = H.clone();
361
362 T.at<float>(0,0) = R.at<float>(0,2);
363 T.at<float>(1,0) = R.at<float>(1,2);
364 R.at<float>(0,2) = 0.f;
365 R.at<float>(1,2) = 0.f;
366
367 // we want to compensate transform to fit into plane warper
368 R = R.t();
369 T = (R * T) * -1;
370}

373void SphericalWarper::detectResultRoi(Size src_size, Point &dst_tl, Point &dst_br)
374{
375 detectResultRoiByBorder(src_size, dst_tl, dst_br);
376
377 float tl_uf = static_cast<float>(dst_tl.x);
378 float tl_vf = static_cast<float>(dst_tl.y);
379 float br_uf = static_cast<float>(dst_br.x);
380 float br_vf = static_cast<float>(dst_br.y);
381
382 float x = projector_.rinv[1];
383 float y = projector_.rinv[4];
384 float z = projector_.rinv[7];
385 if (y > 0.f)
386 {
387 float x_ = (projector_.k[0] * x + projector_.k[1] * y) / z + projector_.k[2];
388 float y_ = projector_.k[4] * y / z + projector_.k[5];
389 if (x_ > 0.f && x_ < src_size.width && y_ > 0.f && y_ < src_size.height)
390 {
391 tl_uf = std::min(tl_uf, 0.f); tl_vf = std::min(tl_vf, static_cast<float>(CV_PI * projector_.scale));
392 br_uf = std::max(br_uf, 0.f); br_vf = std::max(br_vf, static_cast<float>(CV_PI * projector_.scale));
393 }
394 }
395
396 x = projector_.rinv[1];
397 y = -projector_.rinv[4];
398 z = projector_.rinv[7];
399 if (y > 0.f)
400 {
401 float x_ = (projector_.k[0] * x + projector_.k[1] * y) / z + projector_.k[2];
402 float y_ = projector_.k[4] * y / z + projector_.k[5];
403 if (x_ > 0.f && x_ < src_size.width && y_ > 0.f && y_ < src_size.height)
404 {
405 tl_uf = std::min(tl_uf, 0.f); tl_vf = std::min(tl_vf, static_cast<float>(0));
406 br_uf = std::max(br_uf, 0.f); br_vf = std::max(br_vf, static_cast<float>(0));
407 }
408 }
409
410 dst_tl.x = static_cast<int>(tl_uf);
411 dst_tl.y = static_cast<int>(tl_vf);
412 dst_br.x = static_cast<int>(br_uf);
413 dst_br.y = static_cast<int>(br_vf);
414}
