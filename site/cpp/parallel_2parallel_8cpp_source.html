42static
43std::shared_ptr<ParallelForAPI> createParallelForAPI()
44{
45 const std::string& name = getParallelBackendName();
46 bool isKnown = false;
47 const auto& backends = getParallelBackendsInfo();
48 if (!name.empty())
49 {
50 CV_LOG_INFO(NULL, "core(parallel): requested backend name: " << name);
51 }
52 for (size_t i = 0; i < backends.size(); i++)
53 {
54 const auto& info = backends[i];
55 if (!name.empty())
56 {
57 if (name != info.name)
58 {
59 continue;
60 }
61 isKnown = true;
62 }
63 try
64 {
65 CV_LOG_DEBUG(NULL, "core(parallel): trying backend: " << info.name << " (priority=" << info.priority << ")");
66 if (!info.backendFactory)
67 {
68 CV_LOG_DEBUG(NULL, "core(parallel): factory is not available (plugins require filesystem support): " << info.name);
69 continue;
70 }
71 std::shared_ptr<ParallelForAPI> backend = info.backendFactory->create();
72 if (!backend)
73 {
74 CV_LOG_DEBUG(NULL, "core(parallel): not available: " << info.name);
75 continue;
76 }
77 CV_LOG_INFO(NULL, "core(parallel): using backend: " << info.name << " (priority=" << info.priority << ")");
78 g_initializedParallelForAPI = true;
79 getParallelBackendName() = info.name;
80 return backend;
81 }
82 catch (const std::exception& e)
83 {
84 CV_LOG_WARNING(NULL, "core(parallel): can't initialize " << info.name << " backend: " << e.what());
85 }
86 catch (...)
87 {
88 CV_LOG_WARNING(NULL, "core(parallel): can't initialize " << info.name << " backend: Unknown C++ exception");
89 }
90 }
91 if (name.empty())
92 {
93 CV_LOG_DEBUG(NULL, "core(parallel): fallback on builtin code");
94 }
95 else
96 {
97 if (!isKnown)
98 {
99 CV_LOG_INFO(NULL, "core(parallel): unknown backend: " << name << ", fallback on builtin code");
100 }
101 else
102 {
103 CV_LOG_INFO(NULL, "core(parallel): backend=" << name << " is not available, fallback on builtin code");
104 }
105 }
106 g_initializedParallelForAPI = true;
107 getParallelBackendName() = std::string();
108 return std::shared_ptr<ParallelForAPI>();
109}

133bool setParallelForBackend(const std::string& backendName, bool propagateNumThreads)
134{
135 CV_TRACE_FUNCTION();
136
137 bool saved_initialized = g_initializedParallelForAPI;
138 std::string saved_backendName = getParallelBackendName();
139 std::shared_ptr<ParallelForAPI> saved_backend; // don't call getCurrentParallelForAPI() if g_initializedParallelForAPI = false to avoid unnecessary "default" initialization
140
141 std::string backendName_u = toUpperCase(backendName);
142 if (g_initializedParallelForAPI)
143 {
144 // ... already initialized
145 if (getParallelBackendName() == backendName_u)
146 {
147 CV_LOG_INFO(NULL, "core(parallel): backend is already activated: " << (backendName.empty() ? "builtin(legacy)" : backendName));
148 return true;
149 }
150 else
151 {
152 saved_backend = getCurrentParallelForAPI();
153 // ... re-create new
154 CV_LOG_DEBUG(NULL, "core(parallel): replacing parallel backend (old=" << saved_backendName << " new=" << backendName << ")...");
155 getParallelBackendName() = backendName_u;
156 getCurrentParallelForAPI() = createParallelForAPI();
157 }
158 }
159 else
160 {
161 // ... no backend exists, just specify the name (initialization is triggered by getCurrentParallelForAPI() call)
162 getParallelBackendName() = backendName_u;
163 }
164 std::shared_ptr<ParallelForAPI> api = getCurrentParallelForAPI();
165 if (!api)
166 {
167 if (!backendName.empty())
168 {
169 // restore previous backend or build default
170 getParallelBackendName() = saved_backendName;
171 if (saved_initialized)
172 {
173 CV_LOG_WARNING(NULL, "core(parallel): backend is not available: " << backendName << " (keep previous)");
174 getCurrentParallelForAPI() = saved_backend;
175 }
176 else
177 {
178 CV_LOG_WARNING(NULL, "core(parallel): backend is not available: " << backendName << " (use default)");
179 g_initializedParallelForAPI = false;
180 getCurrentParallelForAPI() = createDefaultParallelForAPI();
181 }
182 return false;
183 }
184 else
185 {
186 CV_LOG_WARNING(NULL, "core(parallel): switched to builtin code (legacy)");
187 }
188 }
189 if (!backendName_u.empty())
190 {
191 CV_Assert(backendName_u == getParallelBackendName()); // data race?
192 }
193
194 if (propagateNumThreads)
195 {
196 setNumThreads(numThreads);
197 }
198 return true;
199}
