29static const schar MASK8_RIGHT = '\x80'; // 1000 0000
30static const schar MASK8_NEW = '\x02'; // 0000 0010 (+2)
31static const schar MASK8_FLAGS = '\xFE'; // 1111 1110 (-2)
32static const schar MASK8_BLACK = '\x01'; // 0000 0001 - black pixel
33
34static const schar MASK8_LVAL = '\x7F'; // 0111 1111 (for table)

36template <>
37struct Trait<schar>
38{
39 static inline bool checkValue(const schar* elem, const schar*)
40 {
41 return *elem != 0;
42 }
43 static inline bool isVal(const schar* elem, const schar*)
44 {
45 return *elem == MASK8_BLACK;
46 }
47 static inline bool isRight(const schar* elem, const schar*)
48 {
49 return (*elem & MASK8_RIGHT) != 0;
50 }
51 static inline void setRightFlag(schar* elem, const schar*, schar nbd)
52 {
53 *elem = nbd | MASK8_RIGHT;
54 }
55 static inline void setNewFlag(schar* elem, const schar*, schar nbd)
56 {
57 *elem = nbd;
58 }
59};
60
61static const int MASK_RIGHT = 0x80000000; // 100..000
62static const int MASK_NEW = 0x40000000; // 010..000
63static const int MASK_FLAGS = 0xC0000000; // right + new
64static const int MASK_VAL = 0x3FFFFFFF; // ~flags - pixel label
65
66template <>
67struct Trait<int>
68{
69 static inline bool checkValue(const int* elem, const int* elem0)
70 {
71 return (*elem & MASK_VAL) == (*elem0 & MASK_VAL);
72 }
73 static inline bool isVal(const int* elem, const int* elem0)
74 {
75 return *elem == (*elem0 & MASK_VAL);
76 }
77 static inline bool isRight(const int* elem, const int* elem0)
78 {
79 return (*elem & MASK_RIGHT) == (*elem0 & MASK8_RIGHT);
80 }
81 static inline void setRightFlag(int* elem, const int* elem0, int)
82 {
83 *elem = (*elem0 & MASK_VAL) | MASK_NEW | MASK_RIGHT;
84 }
85 static inline void setNewFlag(int* elem, const int* elem0, int)
86 {
87 *elem = (*elem0 & MASK_VAL) | MASK_NEW;
88 }

99template <typename T>
100static bool icvTraceContour(Mat& image, const Point& start, const Point& end, bool isHole)
101{
102 const T* stop_ptr = image.ptr<T>(end.y, end.x);
103 const size_t step = image.step1();
104 const T *i0 = image.ptr<T>(start.y, start.x), *i1, *i3, *i4 = NULL;
105 const schar s_end = isHole ? 0 : 4;
106
107 schar s = s_end;
108 do
109 {
110 s = (s - 1) & 7;
111 i1 = i0 + getDelta(s, step);
112 }
113 while (!Trait<T>::checkValue(i1, i0) && s != s_end);
114
115 i3 = i0;
116
117 // check single pixel domain
118 if (s != s_end)
119 {
120 // follow border
121 for (;;)
122 {
123 CV_Assert(i3 != NULL);
124 s = clamp_direction(s);
125 while (s < MAX_SIZE - 1)
126 {
127 ++s;
128 i4 = i3 + getDelta(s, step);
129 CV_Assert(i4 != NULL);
130 if (Trait<T>::checkValue(i4, i0))
131 break;
132 }
133
134 if (i3 == stop_ptr)
135 {
136 if (!Trait<T>::isRight(i3, i0))
137 {
138 // it's the only contour
139 return true;
140 }
141
142 // check if this is the last contour
143 // encountered during a raster scan
144 const T* i5;
145 schar t = s;
146 while (true)
147 {
148 t = (t - 1) & 7;
149 i5 = i3 + getDelta(t, step);
150 if (*i5 != 0)
151 break;
152 if (t == 0)
153 return true;
154 }
155 }
156
157 if ((i4 == i0 && i3 == i1))
158 break;
159
160 i3 = i4;
161 s = (s + 4) & 7;
162 } // end of border following loop
163 }
164 else
165 {
166 return i3 == stop_ptr;
167 }
168
169 return false;
170}

172template <typename T>
173static void icvFetchContourEx(Mat& image,
174 const Point& start,
175 T nbd,
176 Contour& res_contour,
177 const bool isDirect)
178{
179 const size_t step = image.step1();
180 T *i0 = image.ptr<T>(start.y, start.x), *i1, *i3, *i4 = NULL;
181
182 Point pt = res_contour.origin;
183
184 cv::Rect rect(pt.x, pt.y, pt.x, pt.y);
185
186 schar s_end = res_contour.isHole ? 0 : 4;
187 schar s = s_end;
188 do
189 {
190 s = (s - 1) & 7;
191 i1 = i0 + getDelta(s, step);
192 }
193 while (!Trait<T>::checkValue(i1, i0) && s != s_end);
194
195 if (s == s_end)
196 {
197 Trait<T>::setRightFlag(i0, i0, nbd);
198 if (!res_contour.isChain)
199 {
200 res_contour.addPoint(pt);
201 }
202 }
203 else
204 {
205 i3 = i0;
206 schar prev_s = s ^ 4;
207
208 // follow border
209 for (;;)
210 {
211 s_end = s;
212 s = clamp_direction(s);
213 while (s < MAX_SIZE - 1)
214 {
215 ++s;
216 i4 = i3 + getDelta(s, step);
217 CV_Assert(i4 != NULL);
218 if (Trait<T>::checkValue(i4, i0))
219 break;
220 }
221 s &= 7;
222
223 // check "right" bound
224 if ((unsigned)(s - 1) < (unsigned)s_end)
225 {
226 Trait<T>::setRightFlag(i3, i0, nbd);
227 }
228 else if (Trait<T>::isVal(i3, i0))
229 {
230 Trait<T>::setNewFlag(i3, i0, nbd);
231 }
232
233 if (res_contour.isChain)
234 {
235 res_contour.codes.push_back(s);
236 }
237 else if (s != prev_s || isDirect)
238 {
239 res_contour.addPoint(pt);
240 }
241
242 if (s != prev_s)
243 {
244 // update bounds
245 if (pt.x < rect.x)
246 rect.x = pt.x;
247 else if (pt.x > rect.width)
248 rect.width = pt.x;
249
250 if (pt.y < rect.y)
251 rect.y = pt.y;
252 else if (pt.y > rect.height)
253 rect.height = pt.y;
254 }
255
256 prev_s = s;
257 pt += chainCodeDeltas[s];
258
259 if (i4 == i0 && i3 == i1)
260 break;
261
262 i3 = i4;
263 s = (s + 4) & 7;
264 }
265 }
266 rect.width -= rect.x - 1;
267 rect.height -= rect.y - 1;
268 res_contour.brect = rect;
269}

324shared_ptr<ContourScanner_> ContourScanner_::create(ContourPointsStorage::storage_t& pointsStorage, ContourCodesStorage::storage_t& codesStorage, Mat img, int mode, int method, Point offset)
325{
326 if (mode == RETR_CCOMP && img.type() == CV_32SC1)
327 mode = RETR_FLOODFILL;
328
329 if (mode == RETR_FLOODFILL)
330 CV_CheckTypeEQ(img.type(), CV_32SC1, "RETR_FLOODFILL mode supports only CV_32SC1 images");
331 else
332 CV_CheckTypeEQ(img.type(),
333 CV_8UC1,
334 "Modes other than RETR_FLOODFILL and RETR_CCOMP support only CV_8UC1 "
335 "images");
336
337 CV_Check(mode,
338 mode == RETR_EXTERNAL || mode == RETR_LIST || mode == RETR_CCOMP ||
339 mode == RETR_TREE || mode == RETR_FLOODFILL,
340 "Wrong extraction mode");
341
342 CV_Check(method,
343 method == 0 || method == CHAIN_APPROX_NONE || method == CHAIN_APPROX_SIMPLE ||
344 method == CHAIN_APPROX_TC89_L1 || method == CHAIN_APPROX_TC89_KCOS,
345 "Wrong approximation method");
346
347 Size size = img.size();
348 CV_Assert(size.height >= 1);
349
350 shared_ptr<ContourScanner_> scanner = make_shared<ContourScanner_>(pointsStorage, codesStorage);
351 scanner->image = img;
352 scanner->mode = mode;
353 scanner->offset = offset;
354 scanner->pt = Point(1, 1);
355 scanner->lnbd = Point(0, 1);
356 scanner->nbd = 2;
357 CNode& root = scanner->tree.newElem(Contour(&scanner->pointsStorage, &scanner->codesStorage));
358 CV_Assert(root.self() == 0);
359 root.body.isHole = true;
360 root.body.brect = Rect(Point(0, 0), size);
361 scanner->ctable.fill(-1);
362 scanner->approx_method2 = scanner->approx_method1 = method;
363 if (method == CHAIN_APPROX_TC89_L1 || method == CHAIN_APPROX_TC89_KCOS)
364 scanner->approx_method1 = CHAIN_CODE;
365 return scanner;
366}

368CNode& ContourScanner_::makeContour(schar& nbd_, const bool is_hole, const int x, const int y)
369{
370 const bool isChain = (this->approx_method1 == CHAIN_CODE); // TODO: get rid of old constant
371 const bool isDirect = (this->approx_method1 == CHAIN_APPROX_NONE);
372
373 const Point start_pt(x - (is_hole ? 1 : 0), y);
374
375 CNode& res = tree.newElem(Contour(&pointsStorage, &codesStorage));
376 res.body.isHole = is_hole;
377 res.body.isChain = isChain;
378 res.body.origin = start_pt + offset;
379 if (isSimple())
380 {
381 icvFetchContourEx<schar>(this->image, start_pt, MASK8_NEW, res.body, isDirect);
382 }
383 else
384 {
385 schar lval;
386 if (isInt())
387 {
388 const int start_val = this->image.at<int>(start_pt);
389 lval = start_val & MASK8_LVAL;
390 icvFetchContourEx<int>(this->image, start_pt, 0, res.body, isDirect);
391 }
392 else
393 {
394 lval = nbd_;
395 // change nbd
396 nbd_ = (nbd_ + 1) & MASK8_LVAL;
397 if (nbd_ == 0)
398 nbd_ = MASK8_BLACK | MASK8_NEW;
399 icvFetchContourEx<schar>(this->image, start_pt, lval, res.body, isDirect);
400 }
401 res.body.brect.x -= this->offset.x;
402 res.body.brect.y -= this->offset.y;
403 res.ctable_next = this->ctable[lval];
404 this->ctable[lval] = res.self();
405 }
406 const Point prev_origin = res.body.origin;
407 res.body.origin = start_pt;
408 if (this->approx_method1 != this->approx_method2)
409 {
410 CV_Assert(res.body.isChain);
411 approximateChainTC89(res.body.codes, prev_origin, this->approx_method2, res.body.pts);
412 res.body.isChain = false;
413 }
414 return res;
415}

417bool ContourScanner_::contourScan(const int prev, int& p, Point& last_pos, const int x, const int y)
418{
419 bool is_hole = false;
420
421 /* if not external contour */
422 if (isInt())
423 {
424 if (!(((prev & MASK_FLAGS) != 0 || prev == 0) && (p & MASK_FLAGS) == 0))
425 {
426 if ((prev & MASK_FLAGS) != 0 || ((p & MASK_FLAGS) != 0))
427 return false;
428
429 if (prev & MASK_FLAGS)
430 {
431 last_pos.x = x - 1;
432 }
433 is_hole = true;
434 }
435 }
436 else
437 {
438 if (!(prev == 0 && p == 1))
439 {
440 if (p != 0 || prev < 1)
441 return false;
442
443 if (prev & MASK8_FLAGS)
444 {
445 last_pos.x = x - 1;
446 }
447 is_hole = true;
448 }
449 }
450
451 if (mode == RETR_EXTERNAL && (is_hole || this->image.at<schar>(last_pos) > 0))
452 {
453 return false;
454 }
455
456 /* find contour parent */
457 int main_parent = -1;
458 if (isSimple() || (!is_hole && (mode == RETR_CCOMP || mode == RETR_FLOODFILL)) ||
459 last_pos.x <= 0)
460 {
461 main_parent = 0;
462 }
463 else
464 {
465 int lval;
466 if (isInt())
467 lval = this->image.at<int>(last_pos.y, last_pos.x) & MASK8_LVAL;
468 else
469 lval = this->image.at<schar>(last_pos.y, last_pos.x) & MASK8_LVAL;
470
471 main_parent = findFirstBoundingContour(last_pos, y, lval, main_parent);
472
473 // if current contour is a hole and previous contour is a hole or
474 // current contour is external and previous contour is external then
475 // the parent of the contour is the parent of the previous contour else
476 // the parent is the previous contour itself.
477 {
478 CNode& main_parent_elem = tree.elem(main_parent);
479 if (main_parent_elem.body.isHole == is_hole)
480 {
481 if (main_parent_elem.parent != -1)
482 {
483 main_parent = main_parent_elem.parent;
484 }
485 else
486 {
487 main_parent = 0;
488 }
489 }
490 }
491
492 // hole flag of the parent must differ from the flag of the contour
493 {
494 CNode& main_parent_elem = tree.elem(main_parent);
495 CV_Assert(main_parent_elem.body.isHole != is_hole);
496 }
497 }
498
499 last_pos.x = x - (is_hole ? 1 : 0);
500
501 schar nbd_ = this->nbd;
502 CNode& new_contour = makeContour(nbd_, is_hole, x, y);
503 if (new_contour.parent == -1)
504 {
505 tree.addChild(main_parent, new_contour.self());
506 }
507 this->pt.x = !isInt() ? (x + 1) : (x + 1 - (is_hole ? 1 : 0));
508 this->pt.y = y;
509 this->nbd = nbd_;
510 return true;
511}

513int ContourScanner_::findFirstBoundingContour(const Point& last_pos,
514 const int y,
515 const int lval,
516 int par)
517{
518 const Point end_point(last_pos.x, y);
519 int res = par;
520 int cur = ctable[lval];
521 while (cur != -1)
522 {
523 CNode& cur_elem = tree.elem(cur);
524 if (((last_pos.x - cur_elem.body.brect.x) < cur_elem.body.brect.width) &&
525 ((last_pos.y - cur_elem.body.brect.y) < cur_elem.body.brect.height))
526 {
527 if (res != -1)
528 {
529 CNode& res_elem = tree.elem(res);
530 const Point origin = res_elem.body.origin;
531 const bool isHole = res_elem.body.isHole;
532 if (isInt())
533 {
534 if (icvTraceContour<int>(this->image, origin, end_point, isHole))
535 break;
536 }
537 else
538 {
539 if (icvTraceContour<schar>(this->image, origin, end_point, isHole))
540 break;
541 }
542 }
543 res = cur;
544 }
545 cur = cur_elem.ctable_next;
546 }
547 return res;
548}

550int ContourScanner_::findNextX(int x, int y, int& prev, int& p)
551{
552 const int width = this->image.size().width - 1;
553 if (isInt())
554 {
555 for (; x < width &&
556 ((p = this->image.at<int>(y, x)) == prev || (p & MASK_VAL) == (prev & MASK_VAL));
557 x++)
558 prev = p;
559 }
560 else
561 {
562#if (CV_SIMD || CV_SIMD_SCALABLE)
563 if ((p = this->image.at<schar>(y, x)) != prev)
564 {
565 return x;
566 }
567 else
568 {
569 v_uint8 v_prev = vx_setall_u8((uchar)prev);
570 for (; x <= width - VTraits<v_uint8>::vlanes(); x += VTraits<v_uint8>::vlanes())
571 {
572 v_uint8 vmask = (v_ne(vx_load(this->image.ptr<uchar>(y, x)), v_prev));
573 if (v_check_any(vmask))
574 {
575 x += v_scan_forward(vmask);
576 p = this->image.at<schar>(y, x);
577 return x;
578 }
579 }
580 }
581#endif
582 for (; x < width && (p = this->image.at<schar>(y, x)) == prev; x++)
583 ;
584 }
585 return x;

588bool ContourScanner_::findNext()
589{
590 int x = this->pt.x;
591 int y = this->pt.y;
592 int width = this->image.size().width - 1;
593 int height = this->image.size().height - 1;
594 Point last_pos = this->lnbd;
595 int prev = isInt() ? this->image.at<int>(y, x - 1) : this->image.at<schar>(y, x - 1);
596
597 for (; y < height; y++)
598 {
599 int p = 0;
600 for (; x < width; x++)
601 {
602 x = findNextX(x, y, prev, p);
603 if (x >= width)
604 break;
605 if (contourScan(prev, p, last_pos, x, y))
606 {
607 this->lnbd = last_pos;
608 return true;
609 }
610 else
611 {
612 prev = p;
613 if ((isInt() && (prev & MASK_FLAGS)) || (!isInt() && (prev & MASK8_FLAGS)))
614 {
615 last_pos.x = x;
616 }
617 }
618 }
619 last_pos = Point(0, y + 1);
620 x = 1;
621 prev = 0;
622 }
623
624 return false;
625}

629void cv::findContours(InputArray _image,
630 OutputArrayOfArrays _contours,
631 OutputArray _hierarchy,
632 int mode,
633 int method,
634 Point offset)
635{
636 CV_INSTRUMENT_REGION();
637
638 // TODO: remove this block in future
639 if (method == 5 /*CV_LINK_RUNS*/)
640 {
641 CV_LOG_ONCE_WARNING(NULL,
642 "LINK_RUNS mode has been extracted to separate function: "
643 "cv::findContoursLinkRuns. "
644 "Calling through cv::findContours will be removed in future.");
645 CV_CheckTrue(!_hierarchy.needed() || mode == RETR_CCOMP,
646 "LINK_RUNS mode supports only simplified hierarchy output (mode=RETR_CCOMP)");
647 findContoursLinkRuns(_image, _contours, _hierarchy);
648 return;
649 }
650
651 // Fast path: RETR_LIST without hierarchy → findTRUContours (parallel contour extraction)
652 if (mode == RETR_LIST && !_hierarchy.needed() && _image.type() == CV_8UC1)
653 {
654 // findTRUContours requires FOREGROUND=255; binarize=true thresholds the padded
655 // image in-place, avoiding an extra allocation (findContours accepts any non-zero value)
656 findTRUContours(_image, _contours, 0, true,method);
657 if (offset != Point())
658 {
659 if (_contours.kind() == _InputArray::STD_VECTOR_VECTOR)
660 {
661 auto& vv = *reinterpret_cast<std::vector<std::vector<Point>>*>(_contours.getObj());
662 for (auto& c : vv)
663 for (auto& p : c)
664 p += offset;
665 }
666 else
667 {
668 const Scalar shift(offset.x, offset.y);
669 const int n = (int)_contours.size().height;
670 for (int i = 0; i < n; i++)
671 _contours.getMat(i) += shift;
672 }
673 }
674 return;
675 }
676
677 // TODO: need enum value, need way to return contour starting points with chain codes
678 if (method == 0 /*CV_CHAIN_CODE*/)
679 {
680 CV_LOG_ONCE_WARNING(NULL,
681 "Chain code output is an experimental feature and might change in "
682 "future!");
683 }
684
685 // Sanity check: output must be of type vector<vector<Point>>
686 CV_Assert((_contours.kind() == _InputArray::STD_VECTOR_VECTOR) ||
687 (_contours.kind() == _InputArray::STD_VECTOR_MAT) ||
688 (_contours.kind() == _InputArray::STD_VECTOR_UMAT));
689
690 const int res_type = (method == 0 /*CV_CHAIN_CODE*/) ? CV_8SC1 : CV_32SC2;
691 if (!_contours.empty())
692 {
693 CV_CheckTypeEQ(_contours.type(),
694 res_type,
695 "Contours must have type CV_8SC1 (chain code) or CV_32SC2 (other methods)");
696 }
697
698 if (_hierarchy.needed())
699 _hierarchy.clear();
700
701 // preprocess
702 Mat image;
703 copyMakeBorder(_image, image, 1, 1, 1, 1, BORDER_CONSTANT | BORDER_ISOLATED, Scalar(0));
704 if (image.type() != CV_32SC1)
705 threshold(image, image, 0, 1, THRESH_BINARY);
706
707 // find contours
708 ContourPointsStorage::storage_t pointsStorage;
709 ContourCodesStorage::storage_t codesStorage;
710 ContourScanner scanner = ContourScanner_::create(pointsStorage, codesStorage, image, mode, method, offset + Point(-1, -1));
711 while (scanner->findNext())
712 {
713 }
714
715 contourTreeToResults(scanner->tree, res_type, _contours, _hierarchy);
716}
