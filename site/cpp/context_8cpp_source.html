53static uint32_t findDeviceComputeQueue(const std::vector<VkQueueFamilyProperties>& queueFamilyProperties)
54{
55 // first try, compute only queue
56 for (uint32_t i = 0; i < queueFamilyProperties.size(); i++)
57 {
58 const VkQueueFamilyProperties& queueFamilyProperty = queueFamilyProperties[i];
59
60 if ((queueFamilyProperty.queueFlags & VK_QUEUE_COMPUTE_BIT)
61 && !(queueFamilyProperty.queueFlags & VK_QUEUE_GRAPHICS_BIT))
62 {
63 return i;
64 }
65 }
66
67 // second try, any queue with compute and graphics
68 for (uint32_t i = 0; i < queueFamilyProperties.size(); i++)
69 {
70 const VkQueueFamilyProperties& queueFamilyProperty = queueFamilyProperties[i];
71
72 if ((queueFamilyProperty.queueFlags & VK_QUEUE_COMPUTE_BIT)
73 && (queueFamilyProperty.queueFlags & VK_QUEUE_GRAPHICS_BIT))
74 {
75 return i;
76 }
77 }
78
79 // third try, any queue with compute
80 for (uint32_t i = 0; i < queueFamilyProperties.size(); i++)
81 {
82 const VkQueueFamilyProperties& queueFamilyProperty = queueFamilyProperties[i];
83
84 if (queueFamilyProperty.queueFlags & VK_QUEUE_COMPUTE_BIT)
85 {
86 return i;
87 }
88 }
89 return uint32_t(-1);
90}

92static uint32_t findDeviceGraphicsQueue(const std::vector<VkQueueFamilyProperties>& queueFamilyProperties)
93{
94 // first try, graphics only queue
95 for (uint32_t i = 0; i < queueFamilyProperties.size(); i++)
96 {
97 const VkQueueFamilyProperties& queueFamilyProperty = queueFamilyProperties[i];
98
99 if ((queueFamilyProperty.queueFlags & VK_QUEUE_GRAPHICS_BIT)
100 && !(queueFamilyProperty.queueFlags & VK_QUEUE_COMPUTE_BIT))
101 {
102 return i;
103 }
104 }
105
106 // second try, any queue with graphics and compute

132static uint32_t findDeviceTransferQueue(const std::vector<VkQueueFamilyProperties>& queueFamilyProperties)
133{
134 // first try, transfer only queue
135 for (uint32_t i = 0; i < queueFamilyProperties.size(); i++)
136 {
137 const VkQueueFamilyProperties& queueFamilyProperty = queueFamilyProperties[i];
138
139 if ((queueFamilyProperty.queueFlags & VK_QUEUE_TRANSFER_BIT)
140 && !(queueFamilyProperty.queueFlags & VK_QUEUE_COMPUTE_BIT)
141 && !(queueFamilyProperty.queueFlags & VK_QUEUE_GRAPHICS_BIT))
142 {
143 return i;
144 }
145 }
146
147 // second try, any queue with transfer
148 for (uint32_t i = 0; i < queueFamilyProperties.size(); i++)
149 {
150 const VkQueueFamilyProperties& queueFamilyProperty = queueFamilyProperties[i];
151
152 if (queueFamilyProperty.queueFlags & VK_QUEUE_TRANSFER_BIT)
153 {
154 return i;
155 }
156 }
157
158 // third try, use compute queue
159 uint32_t compute_queue_index = findDeviceComputeQueue(queueFamilyProperties);
160 if (compute_queue_index != (uint32_t)-1)
161 {
162 return compute_queue_index;
163 }
164
165 // fourth try, use graphics queue
166 uint32_t graphics_queue_index = findDeviceGraphicsQueue(queueFamilyProperties);
167 if (graphics_queue_index != (uint32_t)-1)
168 {
169 return graphics_queue_index;
170 }
171
172 return uint32_t(-1);
173}

207void Context::createInstance()
208{
209 if (kInstance != VK_NULL_HANDLE)
210 return;
211
212 VkResult result;
213
214 if (enableValidationLayers)
215 {
216 uint32_t instanceLayerPropertyCount;
217 result = vkEnumerateInstanceLayerProperties(&instanceLayerPropertyCount, NULL);
218
219 if (result != VK_SUCCESS)
220 {
221 CV_Error(cv::Error::StsError, "Vulkan: vkEnumerateInstanceLayerProperties failed!");
222 return;
223 }
224
225 std::vector<VkLayerProperties> instanceLayerProperties(instanceLayerPropertyCount);
226 result = vkEnumerateInstanceLayerProperties(&instanceLayerPropertyCount, instanceLayerProperties.data());
227
228 if (result != VK_SUCCESS)
229 {
230 CV_Error(cv::Error::StsError, "Vulkan: vkEnumerateInstanceLayerProperties failed!");
231 return;
232 }
233
234 for (uint32_t i = 0; i < instanceLayerPropertyCount; i++)
235 {
236 const VkLayerProperties& lp = instanceLayerProperties[i];
237 CV_LOG_INFO(NULL, "instance layer "<< lp.layerName << lp.implementationVersion);
238
239 if (strcmp(lp.layerName, "VK_LAYER_LUNARG_standard_validation") == 0)
240 {
241 kEnabledLayers.push_back("VK_LAYER_LUNARG_standard_validation");
242 }
243 if (strcmp(lp.layerName, "VK_LAYER_LUNARG_parameter_validation") == 0)
244 {
245 kEnabledLayers.push_back("VK_LAYER_LUNARG_parameter_validation");
246 }
247 if (strcmp(lp.layerName, "VK_LAYER_KHRONOS_validation") == 0)
248 {
249 kEnabledLayers.push_back("VK_LAYER_KHRONOS_validation");
250 }
251 }
252 }
253
254 for (uint32_t j = 0; j < instanceExtensionPropertyCount; j++)
255 {
256 const VkExtensionProperties& exp = instanceExtensionProperties[j];
257
258 if (strcmp(exp.extensionName, "VK_KHR_external_memory_capabilities") == 0)
259 support_VK_KHR_external_memory_capabilities = exp.specVersion;
260 else if (strcmp(exp.extensionName, "VK_KHR_get_physical_device_properties2") == 0)
261 support_VK_KHR_get_physical_device_properties2 = exp.specVersion;
262 else if (strcmp(exp.extensionName, "VK_KHR_get_surface_capabilities2") == 0)
263 support_VK_KHR_get_surface_capabilities2 = exp.specVersion;
264 else if (strcmp(exp.extensionName, "VK_KHR_portability_enumeration") == 0)
265 support_VK_KHR_portability_enumeration = exp.specVersion;
266 else if (strcmp(exp.extensionName, "VK_KHR_surface") == 0)
267 support_VK_KHR_surface = exp.specVersion;
268 else if (strcmp(exp.extensionName, "VK_EXT_debug_report") == 0)
269 support_VK_EXT_debug_report = exp.specVersion;
270#if defined(__ANDROID_API__) && __ANDROID_API__ >= 26
271 else if (strcmp(exp.extensionName, "VK_KHR_android_surface") == 0)
272 support_VK_KHR_android_surface = exp.specVersion;
273#endif // __ANDROID_API__ >= 26
274 }
275
276 if (support_VK_KHR_external_memory_capabilities)
277 enabledExtensions.push_back("VK_KHR_external_memory_capabilities");
278 if (support_VK_KHR_get_physical_device_properties2)
279 enabledExtensions.push_back("VK_KHR_get_physical_device_properties2");
280 if (support_VK_KHR_get_surface_capabilities2)
281 enabledExtensions.push_back("VK_KHR_get_surface_capabilities2");
282 if (support_VK_KHR_portability_enumeration)
283 enabledExtensions.push_back("VK_KHR_portability_enumeration");
284 if (support_VK_KHR_surface)
285 enabledExtensions.push_back("VK_KHR_surface");
286 if (enableValidationLayers && support_VK_EXT_debug_report)
287 enabledExtensions.push_back("VK_EXT_debug_report");
288#if defined(__ANDROID_API__) && __ANDROID_API__ >= 26
289 if (support_VK_KHR_android_surface)
290 enabledExtensions.push_back("VK_KHR_android_surface");
291#endif // __ANDROID_API__ >= 26
292
293 instanceApiVersion = VK_MAKE_VERSION(1, 0, 0);
294
295 if (vkEnumerateInstanceVersion)
296 {
297 VK_CHECK_RESULT(vkEnumerateInstanceVersion(&instanceApiVersion));
298 }
299
300 CV_LOG_INFO(NULL, "instance apiVersion = "
301 <<VK_VERSION_MAJOR(instanceApiVersion)<<"."
302 <<VK_VERSION_MINOR(instanceApiVersion)<<"."
303 <<VK_VERSION_PATCH(instanceApiVersion));
304
305 VkApplicationInfo applicationInfo = {};
306 applicationInfo.sType = VK_STRUCTURE_TYPE_APPLICATION_INFO;
307 applicationInfo.pApplicationName = "OpenCV DNN Vulkan";
308 applicationInfo.applicationVersion = 0;
309 applicationInfo.pEngineName = "vkcom";
310 applicationInfo.engineVersion = 0;
311 applicationInfo.apiVersion = instanceApiVersion;;
312
313 VkInstanceCreateInfo createInfo = {};
314 createInfo.sType = VK_STRUCTURE_TYPE_INSTANCE_CREATE_INFO;
315 createInfo.pNext = nullptr;
316 createInfo.flags = 0;
317 if (support_VK_KHR_portability_enumeration)
318 createInfo.flags |= VK_INSTANCE_CREATE_ENUMERATE_PORTABILITY_BIT_KHR;
319 createInfo.pApplicationInfo = &applicationInfo;
320
321 // Give our desired layers and extensions to vulkan.
322 createInfo.enabledLayerCount = kEnabledLayers.size();
323 createInfo.ppEnabledLayerNames = kEnabledLayers.data();
324 createInfo.enabledExtensionCount = enabledExtensions.size();
325 createInfo.ppEnabledExtensionNames = enabledExtensions.data();
326
327 VK_CHECK_RESULT(vkCreateInstance(&createInfo, NULL, &kInstance));
328
329 // Optional: Validation things.
330 if (enableValidationLayers && support_VK_EXT_debug_report)
331 {
332 VkDebugReportCallbackCreateInfoEXT createInfo = {};
333 createInfo.sType = VK_STRUCTURE_TYPE_DEBUG_REPORT_CALLBACK_CREATE_INFO_EXT;
334 createInfo.flags = VK_DEBUG_REPORT_ERROR_BIT_EXT |
335 VK_DEBUG_REPORT_WARNING_BIT_EXT |
336 VK_DEBUG_REPORT_PERFORMANCE_WARNING_BIT_EXT;
337 createInfo.pfnCallback = &debugReportCallbackFn;
338
339 // Create and register callback.
340 VK_CHECK_RESULT(vkCreateDebugReportCallbackEXT(kInstance, &createInfo,
341 NULL, &kDebugReportCallback));
342 }
343}

345Context::Context()
346{
347 if(!loadVulkanLibrary())
348 {
349 CV_Error(Error::StsError, "loadVulkanLibrary failed");
350 return;
351 }
352 else if (!loadVulkanEntry())
353 {
354 CV_Error(Error::StsError, "loadVulkanEntry failed");
355 return;
356 }
357 else if (!loadVulkanGlobalFunctions())
358 {
359 CV_Error(Error::StsError, "loadVulkanGlobalFunctions failed");
360 return;
361 }
362
363 // Step0: get the extension info from Vulkan library.
364 vkEnumerateInstanceExtensionProperties(nullptr, &instanceExtensionPropertyCount, NULL);
365 instanceExtensionProperties.resize(instanceExtensionPropertyCount);
366 vkEnumerateInstanceExtensionProperties(nullptr, &instanceExtensionPropertyCount, instanceExtensionProperties.data());
367
368 // Step1: create kInstance
369 createInstance();
370
371 init_instance_extension(kInstance);
372
373 if (!loadVulkanFunctions(kInstance))
374 {
375 CV_Error(Error::StsError, "loadVulkanFunctions failed");
376 return;
377 }
378
379 // Step2: Find the best suitable Physical Device.
380 uint32_t deviceCount = 0;
381 vkEnumeratePhysicalDevices(kInstance, &deviceCount, NULL);
382 if (deviceCount == 0)
383 {
384 CV_Error(cv::Error::StsError, "Vulkan Backend: could not find a device with vulkan support!");
385 }
386
387 std::vector<VkPhysicalDevice> devices(deviceCount);
388 vkEnumeratePhysicalDevices(kInstance, &deviceCount, devices.data());
389
390 // TODO: should add better device selection strategy.
391 for (VkPhysicalDevice device : devices)
392 {
393 gpuInfoList.emplace_back(parseGPUInfo(device));
394 }
395
396 bestGPUIndex = findBestPhysicalGPUIndex();
397 CV_Assert(bestGPUIndex >= 0 && bestGPUIndex < devices.size());
398 kPhysicalDevice = devices[bestGPUIndex];
399 physicalDeviceMemoryProperties = gpuInfoList[bestGPUIndex].physicalDeviceMemoryProperties;
400
401 // TODO: try to optimize the speed in discrete GPU.
402 if (gpuInfoList[bestGPUIndex].type == GPU_TYPE_DISCRETE)
403 CV_LOG_WARNING(NULL, "DNN Vulkan backend will work VERY SLOWLY! Because it currently not compatible with discrete graphics cards!");
404
405 // Step3: Create VkQueue
406 kQueueFamilyIndex = gpuInfoList[bestGPUIndex].computeQueueFamilyIndex;
407
408 // create device, queue, command pool
409 VkDeviceQueueCreateInfo queueCreateInfo = {};
410 queueCreateInfo.sType = VK_STRUCTURE_TYPE_DEVICE_QUEUE_CREATE_INFO;
411 queueCreateInfo.queueFamilyIndex = kQueueFamilyIndex;
412 queueCreateInfo.queueCount = 1; // create one queue in this family. We don't need more.
413 float queuePriorities = 1.0; // we only have one queue, so this is not that important.
414 queueCreateInfo.pQueuePriorities = &queuePriorities;
415
416 VkDeviceCreateInfo deviceCreateInfo = {};
417
418 // Step4: Create Logical Device
419 // Specify any desired device features here. We do not need any for this application, though.
420 VkPhysicalDeviceFeatures deviceFeatures = {};
421
422 deviceCreateInfo.sType = VK_STRUCTURE_TYPE_DEVICE_CREATE_INFO;
423 deviceCreateInfo.enabledLayerCount = kEnabledLayers.size();
424 deviceCreateInfo.ppEnabledLayerNames = kEnabledLayers.data();
425 deviceCreateInfo.pQueueCreateInfos = &queueCreateInfo;
426 deviceCreateInfo.queueCreateInfoCount = 1;
427 deviceCreateInfo.pEnabledFeatures = &deviceFeatures;
428
429 VK_CHECK_RESULT(vkCreateDevice(kPhysicalDevice, &deviceCreateInfo, NULL, &kDevice));
430
431 // Get a handle to the only member of the queue family.
432 vkGetDeviceQueue(kDevice, kQueueFamilyIndex, 0, &kQueue);
433
434 // Step4: Create CommandPool and PipelineFactory
435 if (!cmdPoolPtr)
436 cmdPoolPtr = CommandPool::create(kQueue, kQueueFamilyIndex);
437 else
438 CV_Error(cv::Error::StsError, "cmdPoolPtr has been created before!!");
439
440 pipelineFactoryPtr = PipelineFactory::create();
441}

443GPUInfo Context::parseGPUInfo(VkPhysicalDevice& kPhysicalDevice)
444{
445 GPUInfo info;
446
447 // device type
448 VkPhysicalDeviceProperties physicalDeviceProperties;
449 vkGetPhysicalDeviceProperties(kPhysicalDevice, &physicalDeviceProperties);
450
451 // GPU reference.
452 // mali
453 // t760 = 0x13b5 0x7500001 / 0x7501000
454 // t860 = 0x13b5 0x8602000
455 // t880 = 0x13b5 0x8800020
456 // g31 = 0x13b5 0x70930000
457 // g51 = 0x13b5 0x70901010
458 // g52 = 0x13b5 0x74021000 / 0x72120000
459 // g71 = 0x13b5 0x60a00002
460 // g72 = 0x13b5 0x62210001
461 // g76 = 0x13b5 0x72110000
462 // g77 = 0x13b5 0x90800011
463
464 // adreno
465 // 506 = 0x5143 0x5000600
466 // 510 = 0x5143 0x5010000
467 // 512 = 0x5143 0x5010200
468 // 530 = 0x5143 0x5030004
469 // 540 = 0x5143 0x5040001
470 // 616 = 0x5143 0x6010600
471 // 630 = 0x5143 0x6030001
472 // 640 = 0x5143 0x6040001
473 // 650 = 0x5143 0x6050002
474
475 CV_LOG_INFO(NULL, "Begin parse GPU Info "<<gpuInfoList.size()<<"...");
476 CV_LOG_INFO(NULL, "API version is "
477 <<VK_VERSION_MAJOR(physicalDeviceProperties.apiVersion)<<"."
478 <<VK_VERSION_MINOR(physicalDeviceProperties.apiVersion)<<"."
479 <<VK_VERSION_PATCH(physicalDeviceProperties.apiVersion)<<".");
480 CV_LOG_INFO(NULL, "Driver version is "
481 <<VK_VERSION_MAJOR(physicalDeviceProperties.driverVersion)<<"."
482 <<VK_VERSION_MINOR(physicalDeviceProperties.driverVersion)<<"."
483 <<VK_VERSION_PATCH(physicalDeviceProperties.driverVersion)<<".");
484 CV_LOG_INFO(NULL, "Vendor ID:"<<physicalDeviceProperties.vendorID<<".");
485 CV_LOG_INFO(NULL, "Device ID:"<<physicalDeviceProperties.deviceID<<".");
486 CV_LOG_INFO(NULL, "Device name:"<<physicalDeviceProperties.deviceName<<".");
487
488 // info
489 info.apiVersion = physicalDeviceProperties.apiVersion;
490 info.driverVersion = physicalDeviceProperties.driverVersion;
491 info.vendorId = physicalDeviceProperties.vendorID;
492 info.deviceId = physicalDeviceProperties.deviceID;
493 memcpy(info.deviceName, physicalDeviceProperties.deviceName, VK_MAX_PHYSICAL_DEVICE_NAME_SIZE);
494 memcpy(info.pipelineCacheUUID, physicalDeviceProperties.pipelineCacheUUID, VK_UUID_SIZE);
495
496 if (physicalDeviceProperties.deviceType == VK_PHYSICAL_DEVICE_TYPE_DISCRETE_GPU)
497 info.type = GPU_TYPE_DISCRETE;
498 else if (physicalDeviceProperties.deviceType == VK_PHYSICAL_DEVICE_TYPE_INTEGRATED_GPU)
499 info.type = GPU_TYPE_INTEGRATED;
500 else if (physicalDeviceProperties.deviceType == VK_PHYSICAL_DEVICE_TYPE_VIRTUAL_GPU)
501 info.type = GPU_TYPE_VIRTUAL;
502 else if (physicalDeviceProperties.deviceType == VK_PHYSICAL_DEVICE_TYPE_CPU)
503 info.type = GPU_TYPE_CPU_ONLY;
504 else
505 info.type = GPU_TYPE_NOFOUND;
506
507 // device capability
508 info.maxSharedMemorySize = physicalDeviceProperties.limits.maxComputeSharedMemorySize;
509
510 info.maxWorkgroupCount_x = physicalDeviceProperties.limits.maxComputeWorkGroupCount[0];
511 info.maxWorkgroupCount_y = physicalDeviceProperties.limits.maxComputeWorkGroupCount[1];
512 info.maxWorkgroupCount_z = physicalDeviceProperties.limits.maxComputeWorkGroupCount[2];
513
514 info.maxWorkgroup_invocations = physicalDeviceProperties.limits.maxComputeWorkGroupInvocations;
515
516 info.maxWorkgroupSize_x = physicalDeviceProperties.limits.maxComputeWorkGroupSize[0];
517 info.maxWorkgroupSize_y = physicalDeviceProperties.limits.maxComputeWorkGroupSize[1];
518 info.maxWorkgroupSize_z = physicalDeviceProperties.limits.maxComputeWorkGroupSize[2];
519
520 info.memoryMapAlignment = physicalDeviceProperties.limits.minMemoryMapAlignment;
521 info.bufferOffsetAlignment = physicalDeviceProperties.limits.minStorageBufferOffsetAlignment;
522 info.non_coherent_atom_size = physicalDeviceProperties.limits.nonCoherentAtomSize;
523 info.bufferImageGranularity = physicalDeviceProperties.limits.bufferImageGranularity;
524 info.maxImageDimension_1d = physicalDeviceProperties.limits.maxImageDimension1D;
525 info.maxImageDimension_1d = physicalDeviceProperties.limits.maxImageDimension2D;
526 info.maxImageDimension_1d = physicalDeviceProperties.limits.maxImageDimension3D;
527
528 info.timestampPeriod = physicalDeviceProperties.limits.timestampPeriod;
529
530 CV_LOG_INFO(NULL, "maxSharedMemorySize = "<<info.maxSharedMemorySize<<".");
531 CV_LOG_INFO(NULL, "maxWorkgroupCount ( "
532 <<info.maxWorkgroupSize_x<<", "<<info.maxWorkgroupSize_y<<", "<<info.maxWorkgroupSize_z<<").");
533 CV_LOG_INFO(NULL, "maxWorkgroup_invocations = "<<info.maxWorkgroup_invocations<<".");
534 CV_LOG_INFO(NULL, "maxWorkgroupSize ( "
535 <<info.maxWorkgroupSize_x<<", "<<info.maxWorkgroupSize_y<<", "<<info.maxWorkgroupSize_z<<").");
536 CV_LOG_INFO(NULL, "memoryMapAlignment = "<<info.memoryMapAlignment<<".");
537 CV_LOG_INFO(NULL, "bufferOffsetAlignment = "<<info.bufferOffsetAlignment<<".");
538
539 // find compute queue
540 uint32_t queueFamilyPropertiesCount;
541 vkGetPhysicalDeviceQueueFamilyProperties(kPhysicalDevice, &queueFamilyPropertiesCount, 0);
542
543 std::vector<VkQueueFamilyProperties> queueFamilyProperties(queueFamilyPropertiesCount);
544 vkGetPhysicalDeviceQueueFamilyProperties(kPhysicalDevice, &queueFamilyPropertiesCount, queueFamilyProperties.data());
545
546 info.computeQueueFamilyIndex = findDeviceComputeQueue(queueFamilyProperties);
547 info.graphicsQueueFamilyIndex = findDeviceGraphicsQueue(queueFamilyProperties);
548 info.transferQueueFamilyIndex = findDeviceTransferQueue(queueFamilyProperties);
549
550 info.computeQueueCount = queueFamilyProperties[info.computeQueueFamilyIndex].queueCount;
551 info.graphicsQueueCount = queueFamilyProperties[info.graphicsQueueFamilyIndex].queueCount;
552 info.transferQueueCount = queueFamilyProperties[info.transferQueueFamilyIndex].queueCount;
553
554 info.unifiedComputeTransferQueue = info.computeQueueFamilyIndex == info.transferQueueFamilyIndex;
555
556 // additional device properties
557 info.subgroupSize = 64;
558 info.supportSubgroupBasic = false;
559 info.supportSubgroupVote = false;
560 info.supportSubgroupBallot = false;
561 info.supportSubgroupShuffle = false;
562
563 if (support_VK_KHR_get_physical_device_properties2)
564 {
565 void* queryDeviceProperties = 0;
566
567 // query subgroup
568 VkPhysicalDeviceSubgroupProperties physicalDeviceSubgroupProperties;
569 physicalDeviceSubgroupProperties.sType = VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_SUBGROUP_PROPERTIES;
570 physicalDeviceSubgroupProperties.pNext = queryDeviceProperties;
571 if (VK_VERSION_MAJOR(instanceApiVersion) >= 1 && VK_VERSION_MINOR(instanceApiVersion) >= 1)
572 {
573 queryDeviceProperties = &physicalDeviceSubgroupProperties;
574 }
575
576 VkPhysicalDeviceProperties2KHR queryProperties;
577 queryProperties.sType = VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_PROPERTIES_2_KHR;
578 queryProperties.pNext = queryDeviceProperties;
579 vkGetPhysicalDeviceProperties2KHR = (PFN_vkGetPhysicalDeviceProperties2KHR)vkGetInstanceProcAddr(kInstance, "vkGetPhysicalDeviceProperties2KHR");
580 vkGetPhysicalDeviceProperties2KHR(kPhysicalDevice, &queryProperties);
581
582 if (VK_VERSION_MAJOR(instanceApiVersion) >= 1 && VK_VERSION_MINOR(instanceApiVersion) >= 1)
583 {
584 info.subgroupSize = physicalDeviceSubgroupProperties.subgroupSize;
585 if (physicalDeviceSubgroupProperties.supportedStages & VK_SHADER_STAGE_COMPUTE_BIT)
586 {
587 info.supportSubgroupBasic = physicalDeviceSubgroupProperties.supportedOperations & VK_SUBGROUP_FEATURE_BASIC_BIT;
588 info.supportSubgroupVote = physicalDeviceSubgroupProperties.supportedOperations & VK_SUBGROUP_FEATURE_VOTE_BIT;
589 info.supportSubgroupBallot = physicalDeviceSubgroupProperties.supportedOperations & VK_SUBGROUP_FEATURE_BALLOT_BIT;
590 info.supportSubgroupShuffle = physicalDeviceSubgroupProperties.supportedOperations & VK_SUBGROUP_FEATURE_SHUFFLE_BIT;
591 }
592 }
593 else
594 {
595 if (physicalDeviceProperties.vendorID == 0x5143) // qcom adreno prefer very large workgroup :P
596 info.subgroupSize = 128;

793int Context::findBestPhysicalGPUIndex()
794{
795 // first try, discrete gpu
796 for (int i = 0; i < gpuInfoList.size(); i++)
797 {
798 if (gpuInfoList[i].type == GPU_TYPE_DISCRETE)
799 return i;
800 }
801
802 // second try, integrated gpu
803 for (int i = 0; i < gpuInfoList.size(); i++)
804 {
805 if (gpuInfoList[i].type == GPU_TYPE_INTEGRATED)
806 return i;
807 }
808
809 // third try, any probed device
810 if (gpuInfoList.size() > 0)
811 return 0;
812
813 CV_LOG_ERROR(NULL, "no vulkan device");
814 return -1;
815}
