24uint32_t unicode_cpt_from_utf8(const std::string & utf8, size_t & offset) {
25 assert(offset < utf8.size());
26 if (!(utf8[offset + 0] & 0x80)) {
27 auto result = utf8[offset + 0];
28 offset += 1;
29 return result;
30 }
31 if (!(utf8[offset + 0] & 0x40)) {
32 throw std::invalid_argument("invalid character");
33 }
34 if (!(utf8[offset + 0] & 0x20)) {
35 if (offset + 1 >= utf8.size() || ! ((utf8[offset + 1] & 0xc0) == 0x80)) {
36 throw std::invalid_argument("invalid character");
37 }
38 auto result = ((utf8[offset + 0] & 0x1f) << 6) | (utf8[offset + 1] & 0x3f);
39 offset += 2;
40 return result;
41 }
42 if (!(utf8[offset + 0] & 0x10)) {
43 if (offset + 2 >= utf8.size() || ! ((utf8[offset + 1] & 0xc0) == 0x80) || ! ((utf8[offset + 2] & 0xc0) == 0x80)) {
44 throw std::invalid_argument("invalid character");
45 }
46 auto result = ((utf8[offset + 0] & 0x0f) << 12) | ((utf8[offset + 1] & 0x3f) << 6) | (utf8[offset + 2] & 0x3f);
47 offset += 3;
48 return result;
49 }
50 if (!(utf8[offset + 0] & 0x08)) {
51 if (offset + 3 >= utf8.size() || ! ((utf8[offset + 1] & 0xc0) == 0x80) || ! ((utf8[offset + 2] & 0xc0) == 0x80) || !((utf8[offset + 3] & 0xc0) == 0x80)) {
52 throw std::invalid_argument("invalid character");
53 }
54 auto result = ((utf8[offset + 0] & 0x07) << 18) | ((utf8[offset + 1] & 0x3f) << 12) | ((utf8[offset + 2] & 0x3f) << 6) | (utf8[offset + 3] & 0x3f);
55 offset += 4;
56 return result;
57 }
58 throw std::invalid_argument("failed to convert utf8 to codepoint");
59}

93static std::unordered_map<uint8_t, std::string> unicode_byte_to_utf8_map() {
94 std::unordered_map<uint8_t, std::string> map;
95 for (int ch = 0x21; ch <= 0x7E; ++ch) { // u'!' to u'~'
96 assert(0 <= ch && ch < 256);
97 map[ch] = unicode_cpt_to_utf8(ch);
98 }
99 for (int ch = 0xA1; ch <= 0xAC; ++ch) { // u'¡' to u'¬'
100 assert(0 <= ch && ch < 256);
101 map[ch] = unicode_cpt_to_utf8(ch);
102 }
103 for (int ch = 0xAE; ch <= 0xFF; ++ch) { // u'®' to u'ÿ'
104 assert(0 <= ch && ch < 256);
105 map[ch] = unicode_cpt_to_utf8(ch);
106 }
107 auto n = 0;
108 for (int ch = 0; ch < 256; ++ch) {
109 if (map.find(ch) == map.end()) {
110 map[ch] = unicode_cpt_to_utf8(256 + n);
111 ++n;
112 }
113 }
114 return map;
115}

180// GPT2 system regex: 's|'t|'re|'ve|'m|'ll|'d| ?\p{L}+| ?\p{N}+| ?[^\s\p{L}\p{N}]+|\s+(?!\S)|\s+
181static std::vector<size_t> unicode_regex_split_custom_gpt2(const std::string & text, const std::vector<size_t> & offsets) {
182 std::vector<size_t> bpe_offsets; // store the offset of each word
183 bpe_offsets.reserve(offsets.size()); // Reserve memory for the approximate size
184
185 const auto cpts = unicode_cpts_from_utf8(text);
186
187 size_t start = 0;
188 for (auto offset : offsets) {
189 const size_t offset_ini = start;
190 const size_t offset_end = start + offset;
191 assert(offset_end <= cpts.size());
192 start = offset_end;
193
194 static const uint32_t OUT_OF_RANGE = 0xFFFFFFFF;
195 auto _get_cpt = [&] (const size_t pos) -> uint32_t {
196 return (offset_ini <= pos && pos < offset_end) ? cpts[pos] : OUT_OF_RANGE;
197 };
198
199 auto _get_flags = [&] (const size_t pos) -> unicode_cpt_flags {
200 return (offset_ini <= pos && pos < offset_end) ? unicode_cpt_flags_from_cpt(cpts[pos]) : unicode_cpt_flags{};
201 };
202
203 size_t _prev_end = offset_ini;
204 auto _add_token = [&] (const size_t end) -> size_t {
205 assert(_prev_end <= end && end <= offset_end);
206 size_t len = end - _prev_end;
207 if (len > 0) {
208 bpe_offsets.push_back(len);
209 }
210 _prev_end = end;
211 //if (len > 0) {
212 // std::string s = "";
213 // for(size_t p = end-len; p < end; p++)
214 // s += unicode_cpt_to_utf8(cpts[p]);
215 // printf(">>> '%s'\n", s.c_str());
216 //}
217 return len;
218 };
219
220 for (size_t pos = offset_ini; pos < offset_end; /*pos++*/ ) {
221 const uint32_t cpt = _get_cpt(pos);
222 const auto flags = _get_flags(pos);
223
224 // regex: 's|'t|'re|'ve|'m|'ll|'d
225 if (cpt == '\'' && pos+1 < offset_end) {
226 uint32_t cpt_next = _get_cpt(pos+1);
227 if (cpt_next == 's' || cpt_next == 't' || cpt_next == 'm' || cpt_next == 'd') {
228 pos += _add_token(pos+2);
229 continue;
230 }
231 if (pos+2 < offset_end) {
232 uint32_t cpt_next_next = _get_cpt(pos+2);
233 if ((cpt_next == 'r' && cpt_next_next == 'e') ||
234 (cpt_next == 'v' && cpt_next_next == 'e') ||
235 (cpt_next == 'l' && cpt_next_next == 'l')) {
236 pos += _add_token(pos+3);
237 continue;
238 }
239 }
240 }
241
242 auto flags2 = (cpt == ' ' ? _get_flags(pos+1) : flags);
243 // regex: <space>?\p{L}+
244 if (flags2.is_letter) {
245 pos += (cpt == ' ');
246 while (flags2.is_letter) {
247 flags2 = _get_flags(++pos);
248 }
249 _add_token(pos);
250 continue;
251 }
252 // regex: <space>?\p{N}+
253 if (flags2.is_number) {
254 pos += (cpt == ' ');
255 while (flags2.is_number) {
256 flags2 = _get_flags(++pos);
257 }
258 _add_token(pos);
259 continue;
260 }
261 // regex: <space>?[^\s\p{L}\p{N}]+
262 if (!(flags2.is_whitespace | flags2.is_letter | flags2.is_number) && flags2.as_uint()) {
263 pos += (cpt == ' ');
264 while (!(flags2.is_whitespace | flags2.is_letter | flags2.is_number) && flags2.as_uint()) {
265 flags2 = _get_flags(++pos);
266 }
267 _add_token(pos);
268 continue;
269 }
270
271 size_t num_whitespaces = 0;
272 while (_get_flags(pos+num_whitespaces).is_whitespace) {
273 num_whitespaces++;
274 }
275
276 // regex: \s+(?!\S)
277 if (num_whitespaces > 1 && _get_cpt(pos+num_whitespaces) != OUT_OF_RANGE) {
278 pos += num_whitespaces - 1;
279 _add_token(pos);
280 continue;
281 }
282
283 // regex: \s+
284 if (num_whitespaces > 0) {
285 pos += num_whitespaces;
286 _add_token(pos);
287 continue;
288 }
289
290 // no matches
291 _add_token(++pos);
292 }
293 }
294
295 return bpe_offsets;
296}

298// LLAMA3 system regex: "(?i:'s|'t|'re|'ve|'m|'ll|'d)|[^\r\n\p{L}\p{N}]?\p{L}+|\p{N}{1,3}| ?[^\s\p{L}\p{N}]+[\r\n]*|\s*[\r\n]+|\s+(?!\S)|\s+"
299static std::vector<size_t> unicode_regex_split_custom_llama3(const std::string & text, const std::vector<size_t> & offsets) {
300 std::vector<size_t> bpe_offsets; // store the offset of each word
301 bpe_offsets.reserve(offsets.size()); // Reserve memory for the approximate size
302
303 const auto cpts = unicode_cpts_from_utf8(text);
304
305 size_t start = 0;
306 for (auto offset : offsets) {
307 const size_t offset_ini = start;
308 const size_t offset_end = start + offset;
309 assert(offset_end <= cpts.size());
310 start = offset_end;
311
312 static const uint32_t OUT_OF_RANGE = 0xFFFFFFFF;
313 auto _get_cpt = [&] (const size_t pos) -> uint32_t {
314 return (offset_ini <= pos && pos < offset_end) ? cpts[pos] : OUT_OF_RANGE;
315 };
316
317 auto _get_flags = [&] (const size_t pos) -> unicode_cpt_flags {
318 return (offset_ini <= pos && pos < offset_end) ? unicode_cpt_flags_from_cpt(cpts[pos]) : unicode_cpt_flags{};
319 };
320
321 size_t _prev_end = offset_ini;
322 auto _add_token = [&] (const size_t end) -> size_t {
323 assert(_prev_end <= end && end <= offset_end);
324 size_t len = end - _prev_end;
325 if (len > 0) {
326 bpe_offsets.push_back(len);
327 }
328 _prev_end = end;
329 //if (len > 0) {
330 // std::string s = "";
331 // for(size_t p = end-len; p < end; p++)
332 // s += unicode_cpt_to_utf8(cpts[p]);
333 // printf(">>> '%s'\n", s.c_str());
334 //}
335 return len;
336 };
337
338 for (size_t pos = offset_ini; pos < offset_end; /*pos++*/ ) {
339 const uint32_t cpt = _get_cpt(pos);
340 const auto flags = _get_flags(pos);
341
342 // regex: (?i:'s|'t|'re|'ve|'m|'ll|'d) // case insensitive
343 if (cpt == '\'' && pos+1 < offset_end) {
344 uint32_t cpt_next = unicode_tolower(_get_cpt(pos+1));
345 if (cpt_next == 's' || cpt_next == 't' || cpt_next == 'm' || cpt_next == 'd') {
346 pos += _add_token(pos+2);
347 continue;
348 }
349 if (pos+2 < offset_end) {
350 uint32_t cpt_next_next = unicode_tolower(_get_cpt(pos+2));
351 if ((cpt_next == 'r' && cpt_next_next == 'e') ||
352 (cpt_next == 'v' && cpt_next_next == 'e') ||
353 (cpt_next == 'l' && cpt_next_next == 'l')) {
354 pos += _add_token(pos+3);
355 continue;
356 }
357 }
358 }
359
360 // regex: [^\r\n\p{L}\p{N}]?\p{L}+
361 if (!(cpt == '\r' || cpt == '\n' || flags.is_number)) {
362 if (flags.is_letter || _get_flags(pos+1).is_letter) { // one or more letters
363 pos++;
364 while (_get_flags(pos).is_letter) {
365 pos++;
366 }
367 _add_token(pos);
368 continue;
369 }
370 }
371
372 // regex: \p{N}{1,3}
373 if (flags.is_number) {
374 size_t ini = pos;
375 while (_get_flags(pos).is_number) {
376 if (++pos - ini >= 3 ) {
377 _add_token(pos);
378 ini = pos;
379 }
380 }
381 _add_token(pos);
382 continue;
383 }
384
385 // regex: <space>?[^\s\p{L}\p{N}]+[\r\n]*
386 auto flags2 = (cpt == ' ' ? _get_flags(pos+1) : flags);
387 if (!(flags2.is_whitespace | flags2.is_letter | flags2.is_number) && flags.as_uint()) {
388 pos += (cpt == ' ');
389 while (!(flags2.is_whitespace | flags2.is_letter | flags2.is_number) && flags2.as_uint()) {
390 flags2 = _get_flags(++pos);
391 }
392 uint32_t cpt2 = _get_cpt(pos);
393 while (cpt2 == '\r' || cpt2 == '\n') {
394 cpt2 = _get_cpt(++pos);
395 }
396 _add_token(pos);
397 continue;
398 }
399
400 size_t num_whitespaces = 0;
401 size_t last_end_r_or_n = 0;
402 while (_get_flags(pos+num_whitespaces).is_whitespace) {
403 uint32_t cpt2 = _get_cpt(pos+num_whitespaces);
404 if (cpt2 == '\r' || cpt2 == '\n') {
405 last_end_r_or_n = pos + num_whitespaces + 1;
406 }
407 num_whitespaces++;
408 }
409
410 // regex: \s*[\r\n]+
411 if (last_end_r_or_n > 0) {
412 pos = last_end_r_or_n;
413 _add_token(pos);
414 continue;
415 }
416
417 // regex: \s+(?!\S)
418 if (num_whitespaces > 1 && _get_cpt(pos+num_whitespaces) != OUT_OF_RANGE) {
419 pos += num_whitespaces - 1;
420 _add_token(pos);
421 continue;
422 }
423
424 // regex: \s+
425 if (num_whitespaces > 0) {
426 pos += num_whitespaces;
427 _add_token(pos);
428 continue;
429 }
430
431 // no matches
432 _add_token(++pos);
433 }
434 }
435
436 return bpe_offsets;
437}

440static std::vector<size_t> unicode_regex_split_stl(const std::wstring & wtext, const std::wstring & regex_expr, const std::vector<size_t> & offsets) {
441 std::wregex expr(regex_expr);
442 std::vector<size_t> bpe_offsets; // store the offset of each word
443 bpe_offsets.reserve(offsets.size()); // Reserve memory for the approximate size
444 size_t start = 0;
445 for (auto offset : offsets) {
446 std::wcregex_iterator it(wtext.data() + start, wtext.data() + start + offset, expr);
447 std::wcregex_iterator end;
448
449 int64_t start_idx = 0;
450 while (it != end) {
451 std::wcmatch& match = const_cast<std::wcmatch&>(*it);
452 if (match.position() > start_idx) {
453 bpe_offsets.emplace_back(match.position() - start_idx);
454 }
455 bpe_offsets.emplace_back(match.length());
456 start_idx = match.position() + match.length();
457 ++it;
458 }
459
460 if (start_idx < (int64_t) offset) {
461 bpe_offsets.emplace_back(offset - start_idx);
462 }
463 start += offset;
464 }
465
466 return bpe_offsets;
467}

518std::string unicode_cpt_to_utf8(uint32_t cpt) {
519 std::string result;
520
521 if (/* 0x00 <= cpt && */ cpt <= 0x7f) {
522 result.push_back(cpt);
523 return result;
524 }
525 if (0x80 <= cpt && cpt <= 0x7ff) {
526 result.push_back(0xc0 | ((cpt >> 6) & 0x1f));
527 result.push_back(0x80 | (cpt & 0x3f));
528 return result;
529 }
530 if (0x800 <= cpt && cpt <= 0xffff) {
531 result.push_back(0xe0 | ((cpt >> 12) & 0x0f));
532 result.push_back(0x80 | ((cpt >> 6) & 0x3f));
533 result.push_back(0x80 | (cpt & 0x3f));
534 return result;
535 }
536 if (0x10000 <= cpt && cpt <= 0x10ffff) {
537 result.push_back(0xf0 | ((cpt >> 18) & 0x07));
538 result.push_back(0x80 | ((cpt >> 12) & 0x3f));
539 result.push_back(0x80 | ((cpt >> 6) & 0x3f));
540 result.push_back(0x80 | (cpt & 0x3f));
541 return result;
542 }
543
544 throw std::invalid_argument("invalid codepoint");
545}

547std::vector<uint32_t> unicode_cpts_from_utf8(const std::string & utf8) {
548 std::vector<uint32_t> result;
549 result.reserve(utf8.size());
550 size_t offset = 0;
551 while (offset < utf8.size()) {
552 try {
553 result.push_back(unicode_cpt_from_utf8(utf8, offset));
554 }
555 catch (const std::invalid_argument & /*ex*/) {
556 // Silently ignore invalid UTF-8 input to avoid leaking the exception beyond llama_tokenize
557 ++offset;
558 result.emplace_back(0xFFFD); // replacement character
559 }
560 }
561 return result;
562}

592std::vector<std::string> unicode_regex_split(const std::string & text, const std::vector<std::string> & regex_exprs) {
593 // unicode categories
594 static const std::map<std::string, int> k_ucat_enum = {
595 { "\\p{N}", unicode_cpt_flags::NUMBER },
596 { "\\p{L}", unicode_cpt_flags::LETTER },
597 { "\\p{P}", unicode_cpt_flags::PUNCTUATION },
598 { "\\p{M}", unicode_cpt_flags::ACCENT_MARK },
599 { "\\p{S}", unicode_cpt_flags::SYMBOL },
600 };
601
602 static const std::map<int, int> k_ucat_cpt = {
603 { unicode_cpt_flags::NUMBER, 0xD1 },
604 { unicode_cpt_flags::LETTER, 0xD2 },
605 { unicode_cpt_flags::PUNCTUATION, 0xD3 },
606 { unicode_cpt_flags::ACCENT_MARK, 0xD4 },
607 { unicode_cpt_flags::SYMBOL, 0xD5 },
608 };
609
610 static const std::map<int, std::string> k_ucat_map = {
611 { unicode_cpt_flags::NUMBER, "\x30-\x39" }, // 0-9
612 { unicode_cpt_flags::LETTER, "\x41-\x5A\x61-\x7A" }, // A-Za-z
613 { unicode_cpt_flags::PUNCTUATION, "\x21-\x23\x25-\x2A\x2C-\x2F\x3A-\x3B\x3F-\x40\\\x5B-\\\x5D\x5F\\\x7B\\\x7D" }, // !-#%-*,-/:-;?-@\‍[-\‍]_\{\}
614 { unicode_cpt_flags::ACCENT_MARK, "" }, // no sub-128 codepoints
615 { unicode_cpt_flags::SYMBOL, "\\\x24\\\x2B\x3C-\x3E\x5E\x60\\\x7C" }, // $+<=>^`|
616 };
617
618 // compute collapsed codepoints only if needed by at least one regex
619 bool need_collapse = false;
620 for (const auto & regex_expr : regex_exprs) {
621 // search for unicode categories
622 for (const auto & ucat : k_ucat_enum) {
623 if (std::string::npos != regex_expr.find(ucat.first)) {
624 need_collapse = true;
625 break;
626 }
627 }
628 }
629
630 const auto cpts = unicode_cpts_from_utf8(text);
631
632 // generate a "collapsed" representation of the text, where all codepoints are replaced by a single byte
633 // ref: https://github.com/ggml-org/llama.cpp/pull/6920#issuecomment-2081479935
634 std::string text_collapsed;
635 if (need_collapse) {
636 // collapse all unicode categories
637 text_collapsed.resize(cpts.size());
638
639 for (size_t i = 0; i < cpts.size(); ++i) {
640 // keep single-byte codepoints as is
641 if (cpts[i] < 128) {
642 text_collapsed[i] = cpts[i];
643 continue;
644 }
645
646 const auto flags = unicode_cpt_flags_from_cpt(cpts[i]);
647
648 if (flags.is_whitespace) {
649 //NOTE: C++ std::regex \s does not mach 0x85, Rust and Python regex does.
650 //text_collapsed[i] = (char) 0x85; // <Next Line> as whitespace fallback
651 text_collapsed[i] = (char) 0x0B; // <vertical tab> as whitespace fallback
652 } else if (k_ucat_cpt.find(flags.category_flag()) != k_ucat_cpt.end()) {
653 text_collapsed[i] = k_ucat_cpt.at(flags.category_flag());
654 } else {
655 text_collapsed[i] = (char) 0xD0; // fallback
656 }
657 }
658 }
659
660 std::vector<size_t> bpe_offsets = { cpts.size() };
661
662 for (const auto & regex_expr : regex_exprs) {
663 // first, see if we have an efficient custom regex implementation
664 auto tmp = unicode_regex_split_custom(text, regex_expr, bpe_offsets);
665
666 if (!tmp.empty()) {
667 bpe_offsets = std::move(tmp);
668 continue;
669 }
670
671 // fallback to general-purpose std::regex / std::wregex
672 try {
673 // if a unicode category is used in the regex, we use the collapsed text and replace the unicode category
674 // with the corresponding collapsed representation
675 bool use_collapsed = false;
676 for (const auto & ucat : k_ucat_enum) {
677 if (std::string::npos != regex_expr.find(ucat.first)) {
678 use_collapsed = true;
679 break;
680 }
681 }
682
683 if (use_collapsed) {
684 // sanity-check that the original regex does not contain any non-ASCII characters
685 const auto cpts_regex = unicode_cpts_from_utf8(regex_expr);
686 for (size_t i = 0; i < cpts_regex.size(); ++i) {
687 if (cpts_regex[i] >= 128) {
688 throw std::runtime_error("Regex includes both unicode categories and non-ASCII characters - not supported");
689 }
690 }
691
692 // generate a collapsed representation of the regex
693 std::string regex_expr_collapsed;
694
695 // track if we are inside [], because nested [] are not allowed
696 bool inside = false;
697 for (size_t i = 0; i < regex_expr.size(); ++i) {
698 if (regex_expr[i] == '[' && (i == 0 || regex_expr[i - 1] != '\\')) {
699 regex_expr_collapsed += '[';
700 inside = true;
701 continue;
702 }
703
704 if (inside && regex_expr[i] == ']' && regex_expr[i - 1] != '\\') {
705 regex_expr_collapsed += ']';
706 inside = false;
707 continue;
708 }
709
710 if (regex_expr[i + 0] == '\\' && i + 4 < regex_expr.size() &&
711 regex_expr[i + 1] == 'p' &&
712 regex_expr[i + 2] == '{' &&
713 regex_expr[i + 4] == '}') {
714 const std::string pat = regex_expr.substr(i, 5);
715 if (k_ucat_enum.find(pat) != k_ucat_enum.end()) {
716 if (!inside) {
717 regex_expr_collapsed += '[';
718 }
719 regex_expr_collapsed += k_ucat_cpt.at(k_ucat_enum.at(pat));
720 regex_expr_collapsed += k_ucat_map.at(k_ucat_enum.at(pat));
721 if (!inside) {
722 regex_expr_collapsed += ']';
723 }
724 i += 4;
725 continue;
726 }
727 }
728
729 regex_expr_collapsed += regex_expr[i];
730 }
731
732 //printf("text_collapsed: %s\n", text_collapsed.c_str());
733 //printf("regex_expr_collapsed: %s\n", regex_expr_collapsed.c_str());
734 bpe_offsets = unicode_regex_split_stl(text_collapsed, regex_expr_collapsed, bpe_offsets);
735 } else {
736 // no unicode category used, we can use std::wregex directly
737 const std::wstring wregex_expr = unicode_wstring_from_utf8(regex_expr);
738
739 // std::wregex \s does not mach non-ASCII whitespaces, using 0x0B as fallback
740 std::wstring wtext(cpts.begin(), cpts.end());
741 for (size_t i = 0; i < wtext.size(); ++i) {
742 if (wtext[i] > 0x7F && unicode_cpt_flags_from_cpt(wtext[i]).is_whitespace) {
743 wtext[i] = 0x0B;
744 }
745 }
746
747 //printf("text: %s\n", text.c_str());
748 //printf("regex_expr: %s\n", regex_expr.c_str());
749 bpe_offsets = unicode_regex_split_stl(wtext, wregex_expr, bpe_offsets);
750 }
751 } catch (std::regex_error & e) {
752 fprintf(stderr, "Failed to process regex: '%s'\n", regex_expr.c_str());
753 fprintf(stderr, "Regex error: %s\n", e.what());
754 throw std::runtime_error("Failed to process regex");
755 }
756 }
757
758 std::vector<std::string> bpe_words;
759 bpe_words.reserve(bpe_offsets.size()); // reserve memory for the approximate size
760
761 size_t start = 0;
762 for (size_t & offset : bpe_offsets) {
763 bpe_words.emplace_back();
764 for (size_t i = start; i < start + offset; ++i) {
765 bpe_words.back() += unicode_cpt_to_utf8(cpts[i]);
766 }
767 start += offset;
768 }
769
770 return bpe_words;
771
772 return unicode_byte_encoding_process(bpe_words);
773}
