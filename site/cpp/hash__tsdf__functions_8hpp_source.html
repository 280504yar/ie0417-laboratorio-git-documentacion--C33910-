23struct tsdf_hash
24{
25 size_t operator()(const Vec3i& x) const noexcept
26 {
27 size_t seed = 0;
28 constexpr uint32_t GOLDEN_RATIO = 0x9e3779b9;
29 for (uint16_t i = 0; i < 3; i++)
30 {
31 seed ^= std::hash<int>()(x[i]) + GOLDEN_RATIO + (seed << 6) + (seed >> 2);
32 }
33 return seed;
34 }
35};

46class CustomHashSet
47{
48public:
49 static const int hashDivisor = 32768;
50 static const int startCapacity = 2048;
51
52 std::vector<int> hashes;
53 // 0-3 for key, 4th for internal use
54 // don't keep keep value
55 std::vector<Vec4i> data;
56 int capacity;
57 int last;
58
59 CustomHashSet()
60 {
61 hashes.resize(hashDivisor);
62 for (int i = 0; i < hashDivisor; i++)
63 hashes[i] = -1;
64 capacity = startCapacity;
65
66 data.resize(capacity);
67 for (int i = 0; i < capacity; i++)
68 data[i] = { 0, 0, 0, -1 };
69
70 last = 0;
71 }
72
73 ~CustomHashSet() { }

75 inline size_t calc_hash(Vec3i x) const
76 {
77 uint32_t seed = 0;
78 constexpr uint32_t GOLDEN_RATIO = 0x9e3779b9;
79 for (int i = 0; i < 3; i++)
80 {
81 seed ^= x[i] + GOLDEN_RATIO + (seed << 6) + (seed >> 2);
82 }
83 return seed;
84 }
85
86 // should work on existing elements too
87 // 0 - need resize
88 // 1 - idx is inserted
89 // 2 - idx already exists
90 int insert(Vec3i idx)
91 {
92 if (last < capacity)
93 {
94 int hash = int(calc_hash(idx) % hashDivisor);
95 int place = hashes[hash];
96 if (place >= 0)
97 {
98 int oldPlace = place;
99 while (place >= 0)
100 {
101 if (data[place][0] == idx[0] &&
102 data[place][1] == idx[1] &&
103 data[place][2] == idx[2])
104 return 2;
105 else
106 {
107 oldPlace = place;
108 place = data[place][3];
109 //std::cout << "place=" << place << std::endl;
110 }
111 }
112
113 // found, create here
114 data[oldPlace][3] = last;
115 }
116 else
117 {
118 // insert at last
119 hashes[hash] = last;
120 }
121
122 data[last][0] = idx[0];
123 data[last][1] = idx[1];
124 data[last][2] = idx[2];
125 data[last][3] = -1;
126 last++;
127
128 return 1;
129 }
130 else
131 return 0;
132 }
133
134 int find(Vec3i idx) const
135 {
136 int hash = int(calc_hash(idx) % hashDivisor);
137 int place = hashes[hash];
138 // search a place
139 while (place >= 0)
140 {
141 if (data[place][0] == idx[0] &&
142 data[place][1] == idx[1] &&
143 data[place][2] == idx[2])
144 break;
145 else
146 {
147 place = data[place][3];
148 }
149 }
150
151 return place;
152 }
153};

172class VolumesTable
173{
174public:
175 const int hash_divisor = _hash_divisor;
176 const int list_size = _list_size;
177 const int32_t free_row = -1;
178 const int32_t free_isActive = 0;
179
180 const cv::Vec4i nan4 = cv::Vec4i(NAN_ELEMENT);
181
182 int bufferNums;
183 cv::Mat volumes;
184
185 VolumesTable() : bufferNums(1)
186 {
187 this->volumes = cv::Mat(hash_divisor * list_size, 1, rawType<Volume_NODE>());
188 for (int i = 0; i < volumes.size().height; i++)
189 {
190 Volume_NODE* v = volumes.ptr<Volume_NODE>(i);
191 v->idx = nan4;
192 v->row = -1;
193 v->nextVolumeRow = -1;
194 }
195 }
196 const VolumesTable& operator=(const VolumesTable& vt)
197 {
198 this->volumes = vt.volumes;
199 this->bufferNums = vt.bufferNums;
200 return *this;
201 }
202 ~VolumesTable() {}
203
204 bool insert(Vec3i idx, int row)
205 {
206 CV_Assert(row >= 0);
207
208 int bufferNum = 0;
209 int hash = int(calc_hash(idx) % hash_divisor);
210 int start = getPos(idx, bufferNum);
211 int i = start;
212
213 while (i >= 0)
214 {
215 Volume_NODE* v = volumes.ptr<Volume_NODE>(i);
216
217 if (v->idx[0] == NAN_ELEMENT)
218 {
219 Vec4i idx4(idx[0], idx[1], idx[2], 0);
220
221 bool extend = false;
222 if (i != start && i % list_size == 0)
223 {
224 if (bufferNum >= bufferNums - 1)
225 {
226 extend = true;
227 volumes.resize(hash_divisor * bufferNums);
228 bufferNums++;
229 }
230 bufferNum++;
231 v->nextVolumeRow = (bufferNum * hash_divisor + hash) * list_size;
232 }
233 else
234 {
235 v->nextVolumeRow = i + 1;
236 }
237
238 v->idx = idx4;
239 v->row = row;
240
241 return extend;
242 }
243
244 i = v->nextVolumeRow;
245 }
246 return false;
247 }
248 int findRow(Vec3i idx) const
249 {
250 int bufferNum = 0;
251 int i = getPos(idx, bufferNum);
252
253 while (i >= 0)
254 {
255 const Volume_NODE* v = volumes.ptr<Volume_NODE>(i);
256
257 if (v->idx == Vec4i(idx[0], idx[1], idx[2], 0))
258 return v->row;
259 else
260 i = v->nextVolumeRow;
261 }
262
263 return -1;
264 }
265
266 inline int getPos(Vec3i idx, int bufferNum) const
267 {
268 int hash = int(calc_hash(idx) % hash_divisor);
269 return (bufferNum * hash_divisor + hash) * list_size;
270 }
