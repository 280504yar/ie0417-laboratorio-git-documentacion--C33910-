22class PluginDNNBackend CV_FINAL: public std::enable_shared_from_this<PluginDNNBackend>
23{
24protected:
25 void initPluginAPI()
26 {
27 const char* init_name = "opencv_dnn_plugin_init_v0";
28 FN_opencv_dnn_plugin_init_t fn_init = reinterpret_cast<FN_opencv_dnn_plugin_init_t>(lib_->getSymbol(init_name));
29 if (fn_init)
30 {
31 CV_LOG_DEBUG(NULL, "Found entry: '" << init_name << "'");
32 for (int supported_api_version = API_VERSION; supported_api_version >= 0; supported_api_version--)
33 {
34 plugin_api_ = fn_init(ABI_VERSION, supported_api_version, NULL);
35 if (plugin_api_)
36 break;
37 }
38 if (!plugin_api_)
39 {
40 CV_LOG_INFO(NULL, "DNN: plugin is incompatible (can't be initialized): " << lib_->getName());
41 return;
42 }
43 // NB: force strict minor version check (ABI is not preserved for now)
44 if (!checkCompatibility(plugin_api_->api_header, ABI_VERSION, API_VERSION, true))
45 {
46 plugin_api_ = NULL;
47 return;
48 }
49 CV_LOG_INFO(NULL, "DNN: plugin is ready to use '" << plugin_api_->api_header.api_description << "'");
50 }
51 else
52 {
53 CV_LOG_INFO(NULL, "DNN: plugin is incompatible, missing init function: '" << init_name << "', file: " << lib_->getName());
54 }
55 }

58 bool checkCompatibility(const OpenCV_API_Header& api_header, unsigned int abi_version, unsigned int api_version, bool checkMinorOpenCVVersion)
59 {
60 if (api_header.opencv_version_major != CV_VERSION_MAJOR)
61 {
62 CV_LOG_ERROR(NULL, "DNN: wrong OpenCV major version used by plugin '" << api_header.api_description << "': " <<
63 cv::format("%d.%d, OpenCV version is '" CV_VERSION "'", api_header.opencv_version_major, api_header.opencv_version_minor))
64 return false;
65 }
66 if (!checkMinorOpenCVVersion)
67 {
68 // no checks for OpenCV minor version
69 }
70 else if (api_header.opencv_version_minor != CV_VERSION_MINOR)
71 {
72 CV_LOG_ERROR(NULL, "DNN: wrong OpenCV minor version used by plugin '" << api_header.api_description << "': " <<
73 cv::format("%d.%d, OpenCV version is '" CV_VERSION "'", api_header.opencv_version_major, api_header.opencv_version_minor))
74 return false;
75 }
76 CV_LOG_DEBUG(NULL, "DNN: initialized '" << api_header.api_description << "': built with "
77 << cv::format("OpenCV %d.%d (ABI/API = %d/%d)",
78 api_header.opencv_version_major, api_header.opencv_version_minor,
79 api_header.min_api_version, api_header.api_version)
80 << ", current OpenCV version is '" CV_VERSION "' (ABI/API = " << abi_version << "/" << api_version << ")"
81 );
82 if (api_header.min_api_version != abi_version) // future: range can be here
83 {
84 // actually this should never happen due to checks in plugin's init() function
85 CV_LOG_ERROR(NULL, "DNN: plugin is not supported due to incompatible ABI = " << api_header.min_api_version);
86 return false;
87 }
88 if (api_header.api_version != api_version)
89 {
90 CV_LOG_INFO(NULL, "DNN: NOTE: plugin is supported, but there is API version mismath: "
91 << cv::format("plugin API level (%d) != OpenCV API level (%d)", api_header.api_version, api_version));
92 if (api_header.api_version < api_version)
93 {
94 CV_LOG_INFO(NULL, "DNN: NOTE: some functionality may be unavailable due to lack of support by plugin implementation");
95 }
96 }
97 return true;
98 }

176std::vector<FileSystemPath_t> getPluginCandidates(const std::string& baseName)
177{
178 using namespace cv::utils;
179 using namespace cv::utils::fs;
180 const std::string baseName_l = toLowerCase(baseName);
181 const std::string baseName_u = toUpperCase(baseName);
182 const FileSystemPath_t baseName_l_fs = toFileSystemPath(baseName_l);
183 std::vector<FileSystemPath_t> paths;
184 // TODO OPENCV_PLUGIN_PATH
185 const std::vector<std::string> paths_ = getConfigurationParameterPaths("OPENCV_DNN_PLUGIN_PATH", std::vector<std::string>());
186 if (paths_.size() != 0)
187 {
188 for (size_t i = 0; i < paths_.size(); i++)
189 {
190 paths.push_back(toFileSystemPath(paths_[i]));
191 }
192 }
193 else
194 {
195 FileSystemPath_t binaryLocation;
196 if (getBinLocation(binaryLocation))
197 {
198 binaryLocation = getParent(binaryLocation);
199#ifndef CV_DNN_PLUGIN_SUBDIRECTORY
200 paths.push_back(binaryLocation);
201#else
202 paths.push_back(binaryLocation + toFileSystemPath("/") + toFileSystemPath(CV_DNN_PLUGIN_SUBDIRECTORY_STR));
203#endif
204 }
205 }
206 const std::string default_expr = libraryPrefix() + "opencv_dnn_" + baseName_l + "*" + librarySuffix();
207 const std::string plugin_expr = getConfigurationParameterString((std::string("OPENCV_DNN_PLUGIN_") + baseName_u).c_str(), default_expr.c_str());
208 std::vector<FileSystemPath_t> results;
209#ifdef _WIN32
210 FileSystemPath_t moduleName = toFileSystemPath(libraryPrefix() + "opencv_dnn_" + baseName_l + librarySuffix());
211 if (plugin_expr != default_expr)
212 {
213 moduleName = toFileSystemPath(plugin_expr);
214 results.push_back(moduleName);
215 }
216 for (const FileSystemPath_t& path : paths)
217 {
218 results.push_back(path + L"\\" + moduleName);
219 }
220 results.push_back(moduleName);
221#else
222 CV_LOG_DEBUG(NULL, "DNN: " << baseName << " plugin's glob is '" << plugin_expr << "', " << paths.size() << " location(s)");
223 for (const std::string& path : paths)
224 {
225 if (path.empty())
226 continue;
227 std::vector<std::string> candidates;
228 cv::glob(utils::fs::join(path, plugin_expr), candidates);
229 // Prefer candidates with higher versions
230 // TODO: implemented accurate versions-based comparator
231 std::sort(candidates.begin(), candidates.end(), std::greater<std::string>());
232 CV_LOG_DEBUG(NULL, " - " << path << ": " << candidates.size());
233 copy(candidates.begin(), candidates.end(), back_inserter(results));
234 }
235#endif
236 CV_LOG_DEBUG(NULL, "Found " << results.size() << " plugin(s) for " << baseName);
237 return results;
238}

240void PluginDNNBackendFactory::loadPlugin()
241{
242 for (const FileSystemPath_t& plugin : getPluginCandidates(baseName_))
243 {
244 auto lib = std::make_shared<cv::plugin::impl::DynamicLib>(plugin);
245 if (!lib->isLoaded())
246 {
247 continue;
248 }
249 try
250 {
251 auto pluginBackend = std::make_shared<PluginDNNBackend>(lib);
252 if (!pluginBackend)
253 {
254 continue;
255 }
256 if (pluginBackend->plugin_api_ == NULL)
257 {
258 CV_LOG_ERROR(NULL, "DNN: no compatible plugin API for backend: " << baseName_ << " in " << toPrintablePath(plugin));
259 continue;
260 }
261 // NB: we are going to use backend, so prevent automatic library unloading
262 lib->disableAutomaticLibraryUnloading();
263 backend = pluginBackend;
264 return;
265 }
266 catch (...)
267 {
268 CV_LOG_WARNING(NULL, "DNN: exception during plugin initialization: " << toPrintablePath(plugin) << ". SKIP");
269 }
270 }
271}

282std::shared_ptr<IDNNBackendFactory> createPluginDNNBackendFactory(const std::string& baseName)
283{
284#if OPENCV_HAVE_FILESYSTEM_SUPPORT && defined(ENABLE_PLUGINS)
285 const std::string baseName_u = toUpperCase(baseName);
286 AutoLock lock(getInitializationMutex());
287 static std::map<std::string, std::shared_ptr<IDNNBackendFactory>> g_plugins_cache;
288 auto it = g_plugins_cache.find(baseName_u);
289 if (it == g_plugins_cache.end())
290 {
291 auto factory = std::make_shared<impl::PluginDNNBackendFactory>(baseName);
292 g_plugins_cache.insert(std::pair<std::string, std::shared_ptr<IDNNBackendFactory>>(baseName_u, factory));
293 return factory;
294 }
295 return it->second;
296#else
297 CV_UNUSED(baseName);
298 return std::shared_ptr<IDNNBackendFactory>();
299#endif
300}
