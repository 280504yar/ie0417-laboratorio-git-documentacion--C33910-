7// Cleans up redundant Reshape/Transpose chains:
8// - Reshape(A) -> Reshape(B): drop the inner reshape (A's shape is irrelevant
9// once B reshapes to the final target).
10// - Transpose(p1) -> Transpose(p2): compose into a single Transpose with
11// perm[i] = p1[p2[i]].
12// - Identity Transpose (perm == [0,1,...,n-1]): bypass.
13// These run iteratively, so a Reshape+Transpose+Reshape that ends up with an
14// identity transpose collapses to a single Reshape, and adjacent transposes
15// in attention layouts merge.

32 for (int iter = 0; iter < 10; iter++) {
33 netimpl->useCounts(usecounts);

39 static bool isIdentityPerm(const vector<int>& perm)
40 {
41 if (perm.empty()) return false;
42 for (size_t i = 0; i < perm.size(); i++)
43 if (perm[i] != (int)i) return false;
44 return true;
45 }

47 bool fuseGraph(Ptr<Graph>& graph)
48 {
49 const vector<Ptr<LayerInfo>>& prog = graph->prog();
50 size_t nops = prog.size();
51 bool modified = false;
52
53 for (size_t i = 0; i < nops; i++) {
54 if (!prog[i]) continue;
55 vector<Ptr<Graph>>* subgraphs = prog[i]->subgraphs();
56 if (subgraphs) {
57 for (Ptr<Graph>& g : *subgraphs)
58 if (fuseGraph(g)) modified = true;
59 }
60 }
61
62 std::map<int, int> producer;
63 std::set<int> externalArgs;
64 for (Arg out : graph->outputs())
65 externalArgs.insert(out.idx);
66 for (size_t i = 0; i < nops; i++) {
67 if (!prog[i]) continue;
68 for (Arg out : prog[i]->outputs)
69 producer[out.idx] = (int)i;
70 }
71
72 vector<bool> dropped(nops, false);
73
74 for (size_t i = 0; i < nops; i++) {
75 const Ptr<LayerInfo>& layer = prog[i];
76 if (!layer || dropped[i]) continue;
77 if (layer->inputs.empty() || layer->outputs.empty()) continue;
78
79 // Identity Transpose: bypass.
80 TransposeLayer* tr = dynamic_cast<TransposeLayer*>(layer.get());
81 if (tr && layer->outputs.size() == 1 && isIdentityPerm(tr->perm)) {
82 Arg out = layer->outputs[0];
83 if (externalArgs.count(out.idx) == 0) {
84 redirectConsumers(prog, dropped, i + 1, out, layer->inputs[0]);
85 dropped[i] = true;
86 usecounts[out.idx] = 0;
87 modified = true;
88 continue;
89 }
90 }
91
92 // Transpose + Transpose -> Transpose (compose perms).
93 if (tr && layer->outputs.size() == 1) {
94 auto it = producer.find(layer->inputs[0].idx);
95 if (it != producer.end()) {
96 int prod_idx = it->second;
97 if (prod_idx >= 0 && !dropped[prod_idx]) {
98 const Ptr<LayerInfo>& pl = prog[prod_idx];
99 TransposeLayer* prevTr = dynamic_cast<TransposeLayer*>(pl.get());
100 Arg prevOut = layer->inputs[0];
101 bool single_consumer = usecounts[prevOut.idx] == 1
102 && externalArgs.count(prevOut.idx) == 0;
103 if (prevTr && pl->outputs.size() == 1 && single_consumer
104 && !prevTr->perm.empty()
105 && prevTr->perm.size() == tr->perm.size())
106 {
107 // composed[i] = prevTr->perm[tr->perm[i]]
108 vector<int> composed(tr->perm.size());
109 bool ok = true;
110 for (size_t k = 0; k < tr->perm.size(); k++) {
111 int idx = tr->perm[k];
112 if (idx < 0 || idx >= (int)prevTr->perm.size()) {
113 ok = false; break;
114 }
115 composed[k] = prevTr->perm[idx];
116 }
117 if (ok) {
118 tr->perm = composed;
119 layer->inputs[0] = pl->inputs[0];
120 dropped[prod_idx] = true;
121 usecounts[prevOut.idx] = 0;
122 producer[layer->outputs[0].idx] = (int)i;
123 modified = true;
124 continue;
125 }
126 }
127 }
128 }
129 }
130
131 //Reshape + Reshape -> Reshape (drop the inner reshape).
132 Reshape2Layer* rs = dynamic_cast<Reshape2Layer*>(layer.get());
133 if (rs && layer->outputs.size() == 1) {
134 auto it = producer.find(layer->inputs[0].idx);
135 if (it != producer.end()) {
136 int prod_idx = it->second;
137 if (prod_idx >= 0 && !dropped[prod_idx]) {
138 const Ptr<LayerInfo>& pl = prog[prod_idx];
139 Reshape2Layer* prevRs = dynamic_cast<Reshape2Layer*>(pl.get());
140 Arg prevOut = layer->inputs[0];
141 bool single_consumer = usecounts[prevOut.idx] == 1
142 && externalArgs.count(prevOut.idx) == 0;
143 if (prevRs && pl->outputs.size() == 1 && single_consumer)
144 {
145 layer->inputs[0] = pl->inputs[0];
146 dropped[prod_idx] = true;
147 usecounts[prevOut.idx] = 0;
148 modified = true;
149 continue;
150 }
151 }
152 }
153 }
154 }
155
156 if (modified) {
157 vector<Ptr<LayerInfo>> newprog;
158 newprog.reserve(nops);
159 for (size_t i = 0; i < nops; i++) {
160 if (!dropped[i] && prog[i])
161 newprog.push_back(prog[i]);
162 }
163 graph->setProg(newprog);
164 }
165
166 return modified;
167 }

169 void redirectConsumers(const vector<Ptr<LayerInfo>>& prog,
170 const vector<bool>& dropped,
171 size_t start_idx, Arg from, Arg to)
172 {
173 for (size_t j = start_idx; j < prog.size(); j++) {
174 if (!prog[j] || dropped[j]) continue;
175 for (Arg& in : prog[j]->inputs) {
176 if (in.idx == from.idx) {
177 in = to;
178 usecounts[to.idx]++;
179 }
180 }
181 }
182 }

184 Net::Impl* netimpl;
185 vector<int> usecounts;
186};
