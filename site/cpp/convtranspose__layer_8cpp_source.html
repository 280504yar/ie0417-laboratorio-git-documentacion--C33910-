84 virtual void setWeights(InputArray weights_arr, InputArray bias_arr,
85 int C0, int /*accuracy*/) CV_OVERRIDE
86 {
87 Mat rawWeights = weights_arr.getMat();
88 Mat rawBias = bias_arr.getMat();
89 CV_Assert(!rawWeights.empty());
90
91 wshape0 = rawWeights.shape();
92
93 Mat wfloat;
94 if (rawWeights.type() != CV_32F)
95 rawWeights.convertTo(wfloat, CV_32F);
96 else
97 wfloat = rawWeights;
98 repackDeconvWeights(wfloat, weights, CV_32F, ngroups, C0);
99
100 if (!rawBias.empty())
101 rawBias.convertTo(bias, CV_32F);
102 }

104 virtual bool fuseAddBias(InputArray arr) CV_OVERRIDE
105 {
106 if (inputs.size() > 1)
107 return false;
108 Mat new_bias = arr.getMat();
109 CV_Assert(new_bias.isContinuous() && new_bias.dims == 1);
110 if (new_bias.type() != CV_32F) {
111 Mat temp;
112 new_bias.convertTo(temp, CV_32F);
113 new_bias = temp;
114 }
115 if (!bias.empty()) {
116 CV_Assert(bias.shape() == new_bias.shape());
117 add(bias, new_bias, bias);
118 } else {
119 new_bias.copyTo(bias);
120 }
121 return true;
122 }

124 virtual int64_t getFLOPS(const std::vector<MatShape>& inputs,
125 const std::vector<MatShape>& outputs) const CV_OVERRIDE
126 {
127 CV_Assert(inputs.size() >= 1);
128 CV_Assert(outputs.size() == 1);
129 MatShape wshape = inputs.size() > 1 ? inputs[1] : wshape0;
130 int K = wshape[1] * ngroups;
131 size_t ksize = wshape.total() / (wshape[0] * wshape[1]);
132 int C = inputs[0][1] * inputs[0].back();
133 return (int64_t)((inputs[0].total() / C) * ksize * K);
134 }

182 void forward(InputArrayOfArrays input_arrs,
183 OutputArrayOfArrays output_arrs,
184 OutputArrayOfArrays) CV_OVERRIDE
185 {
186 auto* netimpl_ = getNetImpl(this);
187 int ninputs = (int)input_arrs.total();
188 CV_Assert(ninputs >= 1);
189 const Mat& inp = input_arrs.getMat(0);
190 int inptype = inp.type();
191 MatShape inpshape = inp.shape();
192 CV_Assert(inpshape.layout == DATA_LAYOUT_BLOCK);
193 CV_Assert(inp.isContinuous());
194
195 bool dynamicWeights = false;
196 for (int i = 1; i < ninputs; i++) {
197 if (!netimpl_->isConstArg(inputs[i]))
198 dynamicWeights = true;
199 }
200 if (dynamicWeights || weights.empty()) {
201 setWeights(input_arrs.getMat(1), ninputs > 2 ? input_arrs.getMat(2) : Mat(),
202 inpshape.back(), netimpl_->accuracy);
203 }
204
205 MatShape outshape = deconvInferShape(inpshape, wshape0, emptyKernelShape,
206 ngroups, strides, dilations,
207 pads, adjust_pads, auto_pad);
208
209 // compute actual pads for SAME/VALID auto-padding
210 int nsd = inpshape.dims - 3;
211 std::vector<int> pads_resolved = pads;
212 if (auto_pad != AUTO_PAD_NONE) {
213 pads_resolved.resize(nsd * 2, 0);
214 for (int i = 0; i < nsd; i++) {
215 int inpsz = inpshape[2 + i];
216 int outsz = outshape[2 + i];
217 int adj_i = adjust_pads.empty() ? 0 : adjust_pads[i];
218 int stride = strides.empty() ? 1 : strides[i];
219 int dil = dilations.empty() ? 1 : dilations[i];
220 int ki = wshape0[2 + i];
221 int total = (inpsz - 1) * stride + dil * (ki - 1) + 1 + adj_i - outsz;
222 int pb;
223 if (auto_pad == AUTO_PAD_SAME_UPPER && stride <= ki * dil) {
224 pb = std::max((total - (outsz - 1 + stride) % stride) / 2, 0);
225 } else {
226 pb = total / 2;
227 }
228 pads_resolved[i] = pb;
229 pads_resolved[nsd + i] = total - pb;
230 }
231 }
232
233 int outtype = inferType(inptype);
234
235 if (inpshape != prevInpshape) {
236 cs.initDeconv(inpshape, wshape0, outshape, ngroups,
237 strides, dilations, pads_resolved);
238 prevInpshape = inpshape;
239 }
240
241 int outkind = output_arrs.kind();
242 CV_Assert(outkind == _InputArray::STD_VECTOR_MAT ||
243 outkind == _InputArray::STD_VECTOR_UMAT);
244
245 std::vector<Mat>* outs = nullptr;
246 std::vector<UMat>* uouts = nullptr;
247 Mat out;
248
249 if (outkind == _InputArray::STD_VECTOR_MAT) {
250 outs = &output_arrs.getMatVecRef();
251 outs->resize(1);
252 outs->at(0).fit(outshape, outtype);
253 out = outs->at(0);
254 } else {
255 uouts = &output_arrs.getUMatVecRef();
256 uouts->resize(1);
257 uouts->at(0).fit(outshape, outtype);
258 out.fit(outshape, outtype);
259 }
260
261 DeconvFunc func = getDeconvFunc(inptype);
262 CV_Assert(func != nullptr);
263 const float* bias_data = bias.empty() ? nullptr : bias.ptr<float>();
264 func(inp.data, nullptr, out.data, cs, weights.data, nullptr, bias_data);
265
266 if (uouts) {
267 out.copyTo(uouts->at(0));
268 }
269
270 if (dynamicWeights) {
271 weights.release();
272 }
273 }
274
275 std::vector<int> emptyKernelShape;
276 Mat weights, bias;
277 MatShape wshape0, prevInpshape;
278 ConvState cs;
279};
