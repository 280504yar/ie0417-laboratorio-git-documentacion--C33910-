69template<typename T>
70static PyObject* to_dlpack(const T& src, PyObject* self, PyObject* py_args, PyObject* kw)
71{
72 int stream = 0;
73 PyObject* maxVersion = nullptr;
74 PyObject* dlDevice = nullptr;
75 bool copy = false;
76 const char* keywords[] = { "stream", "max_version", "dl_device", "copy", NULL };
77 if (!PyArg_ParseTupleAndKeywords(py_args, kw, "|iOOp:__dlpack__", (char**)keywords, &stream, &maxVersion, &dlDevice, &copy))
78 return nullptr;
79
80 DLDevice device = {(DLDeviceType)-1, 0};
81 if (dlDevice && dlDevice != Py_None && PyTuple_Check(dlDevice))
82 {
83 device.device_type = static_cast<DLDeviceType>(PyLong_AsLong(PyTuple_GetItem(dlDevice, 0)));
84 device.device_id = PyLong_AsLong(PyTuple_GetItem(dlDevice, 1));
85 }
86
87 int ndim = GetNumDims(src);
88 void* ptr = PyMem_Malloc(sizeof(DLManagedTensor) + sizeof(int64_t) * ndim * 2);
89 if (!ptr) {
90 PyErr_NoMemory();
91 return nullptr;
92 }
93 DLManagedTensor* tensor = reinterpret_cast<DLManagedTensor*>(ptr);
94 tensor->manager_ctx = self;
95 tensor->deleter = array_dlpack_deleter;
96 tensor->dl_tensor.ndim = ndim;
97 tensor->dl_tensor.shape = reinterpret_cast<int64_t*>(reinterpret_cast<char*>(ptr) + sizeof(DLManagedTensor));
98 tensor->dl_tensor.strides = tensor->dl_tensor.shape + ndim;
99 fillDLPackTensor(src, tensor, device);
100
101 PyObject* capsule = PyCapsule_New(ptr, CV_DLPACK_CAPSULE_NAME, dlpack_capsule_deleter);
102 if (!capsule) {
103 PyMem_Free(ptr);
104 return nullptr;
105 }
106
107 // the capsule holds a reference
108 Py_INCREF(self);
109
110 return capsule;
111}

113template<typename T>
114static PyObject* from_dlpack(PyObject* py_args, PyObject* kw)
115{
116 PyObject* arr = nullptr;
117 PyObject* device = nullptr;
118 bool copy = false;
119 const char* keywords[] = { "device", "copy", NULL };
120 if (!PyArg_ParseTupleAndKeywords(py_args, kw, "O|Op:from_dlpack", (char**)keywords, &arr, &device, &copy))
121 return nullptr;
122
123 PyObject* capsule = nullptr;
124 if (PyCapsule_CheckExact(arr))
125 {
126 capsule = arr;
127 }
128 else
129 {
130 PyGILState_STATE gstate;
131 gstate = PyGILState_Ensure();
132 capsule = PyObject_CallMethodObjArgs(arr, PyString_FromString("__dlpack__"), NULL);
133 PyGILState_Release(gstate);
134 }
135
136 DLManagedTensor* tensor = reinterpret_cast<DLManagedTensor*>(PyCapsule_GetPointer(capsule, CV_DLPACK_CAPSULE_NAME));
137 if (tensor == nullptr)
138 {
139 if (capsule != arr)
140 Py_DECREF(capsule);
141 return nullptr;
142 }
143
144 T retval;
145 bool success = parseDLPackTensor(tensor, retval, copy);
146 if (success)
147 {
148 PyCapsule_SetName(capsule, CV_DLPACK_USED_CAPSULE_NAME);
149 }
150 if (capsule != arr)
151 Py_DECREF(capsule);
152
153 return success ? pyopencv_from(retval) : nullptr;
154}

156static DLDataType GetDLPackType(size_t elemSize1, int depth) {
157 DLDataType dtype;
158 dtype.bits = static_cast<uint8_t>(8 * elemSize1);
159 dtype.lanes = 1;
160 switch (depth)
161 {
162 case CV_Bool: dtype.code = kDLBool; break;
163 case CV_16BF: dtype.code = kDLBfloat; break;
164 case CV_8S: case CV_16S: case CV_32S: case CV_64S: dtype.code = kDLInt; break;
165 case CV_8U: case CV_16U: case CV_32U: case CV_64U: dtype.code = kDLUInt; break;
166 case CV_16F: case CV_32F: case CV_64F: dtype.code = kDLFloat; break;
167 default:
168 CV_Error(Error::StsNotImplemented, "__dlpack__ data type");
169 }
170 return dtype;
171}

173static int DLPackTypeToCVType(const DLDataType& dtype, int channels) {
174 if (dtype.code == kDLInt)
175 {
176 switch (dtype.bits)
177 {
178 case 8: return CV_8SC(channels);
179 case 16: return CV_16SC(channels);
180 case 32: return CV_32SC(channels);
181 case 64: return CV_64SC(channels);
182 default:
183 {
184 PyErr_SetString(PyExc_BufferError,
185 format("Unsupported int dlpack depth: %d", dtype.bits).c_str());
186 return -1;
187 }
188 }
189 }
190 if (dtype.code == kDLUInt)
191 {
192 switch (dtype.bits)
193 {
194 case 8: return CV_8UC(channels);
195 case 16: return CV_16UC(channels);
196 case 32: return CV_32UC(channels);
197 case 64: return CV_64UC(channels);
198 default:
199 {
200 PyErr_SetString(PyExc_BufferError,
201 format("Unsupported uint dlpack depth: %d", dtype.bits).c_str());
202 return -1;
203 }
204 }
205 }
206 if (dtype.code == kDLFloat)
207 {
208 switch (dtype.bits)
209 {
210 case 16: return CV_16FC(channels);
211 case 32: return CV_32FC(channels);
212 case 64: return CV_64FC(channels);
213 default:
214 {
215 PyErr_SetString(PyExc_BufferError,
216 format("Unsupported float dlpack depth: %d", dtype.bits).c_str());
217 return -1;
218 }
219 }
220 }
221 if (dtype.code == kDLBool)
222 {
223 return CV_BoolC(channels);
224 }
225 if (dtype.code == kDLBfloat)
226 {
227 return CV_16BFC(channels);
228 }
229 PyErr_SetString(PyExc_BufferError, format("Unsupported dlpack data type: %d", dtype.code).c_str());
230 return -1;
231}
