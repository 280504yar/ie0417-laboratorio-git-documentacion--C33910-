93int solveLP(InputArray Func_, InputArray Constr_, OutputArray z_, double constr_eps)
94{
95 dprintf(("call to solveLP\n"));
96
97 //sanity check (size, type, no. of channels)
98 CV_Assert(Func_.type()==CV_64FC1 || Func_.type()==CV_32FC1);
99 CV_Assert(Constr_.type()==CV_64FC1 || Constr_.type()==CV_32FC1);
100 CV_Assert((Func_.rows()==1 && (Constr_.cols()-Func_.cols()==1))||
101 (Func_.cols()==1 && (Constr_.cols()-Func_.rows()==1)));
102 if (z_.fixedType())
103 CV_CheckType(z_.type(), z_.type() == CV_64FC1 || z_.type() == CV_32FC1 || z_.type() == CV_32SC1, "");
104
105 Mat Func = Func_.getMat();
106 Mat Constr = Constr_.getMat();
107
108 //copy arguments for we will shall modify them
109 Mat_<double> bigC=Mat_<double>(1,(Func.rows==1?Func.cols:Func.rows)+1),
110 bigB=Mat_<double>(Constr.rows,Constr.cols+1);
111 if(Func.rows==1){
112 Func.convertTo(bigC.colRange(1,bigC.cols),CV_64FC1);
113 }else{
114 Mat FuncT=Func.t();
115 FuncT.convertTo(bigC.colRange(1,bigC.cols),CV_64FC1);
116 }
117 Constr.convertTo(bigB.colRange(1,bigB.cols),CV_64FC1);
118 double v=0;
119 vector<int> N,B;
120 vector<unsigned int> indexToRow;
121
122 if(initialize_simplex(bigC,bigB,v,N,B,indexToRow)==SOLVELP_UNFEASIBLE){
123 return SOLVELP_UNFEASIBLE;
124 }
125 Mat_<double> c=bigC.colRange(1,bigC.cols),
126 b=bigB.colRange(1,bigB.cols);
127
128 int res=0;
129 if((res=inner_simplex(c,b,v,N,B,indexToRow))==SOLVELP_UNBOUNDED){
130 return SOLVELP_UNBOUNDED;
131 }
132
133 //return the optimal solution
134 Mat z(c.cols,1,CV_64FC1);
135 MatIterator_<double> it=z.begin<double>();
136 unsigned int nsize = (unsigned int)N.size();
137 for(int i=1;i<=c.cols;i++,it++){
138 if(indexToRow[i]<nsize){
139 *it=0;
140 }else{
141 *it=b.at<double>(indexToRow[i]-nsize,b.cols-1);
142 }
143 }
144
145 z.copyTo(z_);
146
147 //check constraints feasibility
148 Mat prod = Constr(Rect(0, 0, Constr.cols - 1, Constr.rows)) * z;
149 Mat constr_check = Constr.col(Constr.cols - 1) - prod;
150 double min_value = 0.0;
151 minMaxIdx(constr_check, &min_value);
152 if (min_value < -constr_eps)
153 {
154 return SOLVELP_LOST;
155 }
156
157 return res;
158}

167 N[0]=0;
168 for (std::vector<int>::iterator it = N.begin()+1 ; it != N.end(); ++it){

173 for (std::vector<int>::iterator it = B.begin()+1 ; it != B.end(); ++it){

178 for (std::vector<unsigned int>::iterator it = indexToRow.begin()+1 ; it != indexToRow.end(); ++it){
179 *it=it[-1]+1;
180 }
181 v=0;
182
183 int k=0;
184 {
185 double min=DBL_MAX;

188 min=b(i,b.cols-1);
189 k=i;
190 }
191 }

193

206 b(i,0)=-1;

265 for (std::vector<unsigned int>::iterator it = indexToRow.begin()+1 ; it != indexToRow.end(); ++it){
266 --(*it);
267 }
268 return 0;

271static int inner_simplex(Mat_<double>& c, Mat_<double>& b,double& v,vector<int>& N,vector<int>& B,vector<unsigned int>& indexToRow){
272
273 for(;;){
274 MatIterator_<double> pos_ptr;
275 int e=-1,pos_ctr=0,min_var=INT_MAX;
276 bool all_nonzero=true;
277 for(pos_ptr=c.begin();pos_ptr!=c.end();pos_ptr++,pos_ctr++){
278 if(*pos_ptr==0){
279 all_nonzero=false;
280 }
281 if(*pos_ptr>0){
282 if(N[pos_ctr]<min_var){
283 e=pos_ctr;
284 min_var=N[pos_ctr];
285 }
286 }
287 }
288 if(e==-1){
289 dprintf(("hello from e==-1\n"));
290 print_matrix(c);
291 if(all_nonzero==true){
292 return SOLVELP_SINGLE;
293 }else{
294 return SOLVELP_MULTI;
295 }
296 }
297
298 int l=-1;
299 min_var=INT_MAX;
300 double min=DBL_MAX;
301 int row_it=0;
302 MatIterator_<double> min_row_ptr=b.begin();
303 for(MatIterator_<double> it=b.begin();it!=b.end();it+=b.cols,row_it++){
304 double myite=0;
305 //check constraints, select the tightest one, reinforcing Bland's rule
306 if((myite=it[e])>0){
307 double val=it[b.cols-1]/myite;
308 if(val<min || (val==min && B[row_it]<min_var)){
309 min_var=B[row_it];
310 min_row_ptr=it;
311 min=val;
312 l=row_it;
313 }
314 }
315 }
316 if(l==-1){
317 return SOLVELP_UNBOUNDED;
318 }
319 dprintf(("the tightest constraint is in row %d with %g\n",l,min));
320
321 pivot(c,b,v,N,B,l,e,indexToRow);
322
323 dprintf(("objective, v=%g\n",v));
324 print_matrix(c);
325 dprintf(("constraints\n"));
326 print_matrix(b);
327 dprintf(("non-basic: "));
328 print_matrix(Mat(N));
329 dprintf(("basic: "));
330 print_matrix(Mat(B));
331 }
332}

334static inline void pivot(Mat_<double>& c,Mat_<double>& b,double& v,vector<int>& N,vector<int>& B,
335 int leaving_index,int entering_index,vector<unsigned int>& indexToRow){
336 double Coef=b(leaving_index,entering_index);
337 for(int i=0;i<b.cols;i++){
338 if(i==entering_index){
339 b(leaving_index,i)=1/Coef;
340 }else{
341 b(leaving_index,i)/=Coef;
342 }
343 }
344
345 for(int i=0;i<b.rows;i++){
346 if(i!=leaving_index){
347 double coef=b(i,entering_index);
348 for(int j=0;j<b.cols;j++){
349 if(j==entering_index){
350 b(i,j)=-coef*b(leaving_index,j);
351 }else{
352 b(i,j)-=(coef*b(leaving_index,j));
353 }
354 }
355 }
356 }
357
358 //objective function
359 Coef=c(0,entering_index);
360 for(int i=0;i<(b.cols-1);i++){
361 if(i==entering_index){
362 c(0,i)=-Coef*b(leaving_index,i);
363 }else{
364 c(0,i)-=Coef*b(leaving_index,i);
365 }
366 }
367 dprintf(("v was %g\n",v));
368 v+=Coef*b(leaving_index,b.cols-1);
369
370 SWAP(int,N[entering_index],B[leaving_index]);
371 SWAP(int,indexToRow[N[entering_index]],indexToRow[B[leaving_index]]);
372}
