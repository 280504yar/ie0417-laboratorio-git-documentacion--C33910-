34static inline
35void ptsort(struct pt *pts, int sz)
36{
37#define MAYBE_SWAP(arr,apos,bpos) \
38 if (arr[apos].theta > arr[bpos].theta) { \
39 tmp = arr[apos]; arr[apos] = arr[bpos]; arr[bpos] = tmp; \
40 };
41
42 if (sz <= 1)
43 return;
44
45 if (sz == 2) {
46 struct pt tmp;
47 MAYBE_SWAP(pts, 0, 1);
48 return;
49 }
50
51 // NB: Using less-branch-intensive sorting networks here on the
52 // hunch that it's better for performance.
53 if (sz == 3) { // 3 element bubble sort is optimal
54 struct pt tmp;
55 MAYBE_SWAP(pts, 0, 1);
56 MAYBE_SWAP(pts, 1, 2);
57 MAYBE_SWAP(pts, 0, 1);
58 return;
59 }
60
61 if (sz == 4) { // 4 element optimal sorting network.
62 struct pt tmp;
63 MAYBE_SWAP(pts, 0, 1); // sort each half, like a merge sort
64 MAYBE_SWAP(pts, 2, 3);
65 MAYBE_SWAP(pts, 0, 2); // minimum value is now at 0.
66 MAYBE_SWAP(pts, 1, 3); // maximum value is now at end.
67 MAYBE_SWAP(pts, 1, 2); // that only leaves the middle two.
68 return;
69 }
70
71 if (sz == 5) {
72 // this 9-step swap is optimal for a sorting network, but two
73 // steps slower than a generic sort.
74 struct pt tmp;
75 MAYBE_SWAP(pts, 0, 1); // sort each half (3+2), like a merge sort
76 MAYBE_SWAP(pts, 3, 4);
77 MAYBE_SWAP(pts, 1, 2);
78 MAYBE_SWAP(pts, 0, 1);
79 MAYBE_SWAP(pts, 0, 3); // minimum element now at 0
80 MAYBE_SWAP(pts, 2, 4); // maximum element now at end
81 MAYBE_SWAP(pts, 1, 2); // now resort the three elements 1-3.
82 MAYBE_SWAP(pts, 2, 3);
83 MAYBE_SWAP(pts, 1, 2);
84 return;
85 }
86
87#undef MAYBE_SWAP
88
89 ptsort_(pts, sz);
90}

92void ptsort_(struct pt *pts, int sz)
93{
94 // a merge sort with temp storage.
95
96 // Use stack storage if it's not too big.
97 AutoBuffer<struct pt, 1024> _tmp_stack(sz);
98 memcpy(_tmp_stack.data(), pts, sizeof(struct pt) * sz);
99
100 int asz = sz/2;
101 int bsz = sz - asz;
102
103 struct pt *as = &_tmp_stack[0];
104 struct pt *bs = &_tmp_stack[asz];
105
106 ptsort(as, asz);
107 ptsort(bs, bsz);
108
109#define MERGE(apos,bpos) \
110 if (as[apos].theta < bs[bpos].theta) \
111 pts[outpos++] = as[apos++]; \
112 else \
113 pts[outpos++] = bs[bpos++];
114
115 int apos = 0, bpos = 0, outpos = 0;
116 while (apos + 8 < asz && bpos + 8 < bsz) {
117 MERGE(apos,bpos); MERGE(apos,bpos); MERGE(apos,bpos); MERGE(apos,bpos);
118 MERGE(apos,bpos); MERGE(apos,bpos); MERGE(apos,bpos); MERGE(apos,bpos);
119 }
120
121 while (apos < asz && bpos < bsz) {
122 MERGE(apos,bpos);
123 }
124
125 if (apos < asz)
126 memcpy(&pts[outpos], &as[apos], (asz-apos)*sizeof(struct pt));
127 if (bpos < bsz)
128 memcpy(&pts[outpos], &bs[bpos], (bsz-bpos)*sizeof(struct pt));
129
130#undef MERGE
131}

139void fit_line(struct line_fit_pt *lfps, int sz, int i0, int i1, double *lineparm, double *err, double *mse){
140 CV_Assert(i0 != i1);
141 CV_Assert(i0 >= 0 && i1 >= 0 && i0 < sz && i1 < sz);
142
143 double Mx, My, Mxx, Myy, Mxy, W;
144 int N; // how many points are included in the set?
145
146 if (i0 < i1) {
147 N = i1 - i0 + 1;
148
149 Mx = lfps[i1].Mx;
150 My = lfps[i1].My;
151 Mxx = lfps[i1].Mxx;
152 Mxy = lfps[i1].Mxy;
153 Myy = lfps[i1].Myy;
154 W = lfps[i1].W;
155
156 if (i0 > 0) {
157 Mx -= lfps[i0-1].Mx;
158 My -= lfps[i0-1].My;
159 Mxx -= lfps[i0-1].Mxx;
160 Mxy -= lfps[i0-1].Mxy;
161 Myy -= lfps[i0-1].Myy;
162 W -= lfps[i0-1].W;
163 }
164
165 } else {
166 // i0 > i1, e.g. [15, 2]. Wrap around.
167 CV_Assert(i0 > 0);
168
169 Mx = lfps[sz-1].Mx - lfps[i0-1].Mx;
170 My = lfps[sz-1].My - lfps[i0-1].My;
171 Mxx = lfps[sz-1].Mxx - lfps[i0-1].Mxx;
172 Mxy = lfps[sz-1].Mxy - lfps[i0-1].Mxy;
173 Myy = lfps[sz-1].Myy - lfps[i0-1].Myy;
174 W = lfps[sz-1].W - lfps[i0-1].W;
175
176 Mx += lfps[i1].Mx;
177 My += lfps[i1].My;
178 Mxx += lfps[i1].Mxx;
179 Mxy += lfps[i1].Mxy;
180 Myy += lfps[i1].Myy;
181 W += lfps[i1].W;
182
183 N = sz - i0 + i1 + 1;
184 }
185
186 CV_Assert(N >= 2);
187
188 double Ex = Mx / W;
189 double Ey = My / W;
190 double Cxx = Mxx / W - Ex*Ex;
191 double Cxy = Mxy / W - Ex*Ey;
192 double Cyy = Myy / W - Ey*Ey;
193
194 double nx, ny;
195
196 if (1) {
197 // on iOS about 5% of total CPU spent in these trig functions.
198 // 85 ms per frame on 5S, example.pnm
199 //
200 // XXX this was using the double-precision atan2. Was there a case where
201 // we needed that precision? Seems doubtful.
202 float normal_theta = float(.5f * (CV_PI / 180)) * cv::fastAtan2((float)(-2*Cxy), (float)(Cyy - Cxx));
203 nx = cosf(normal_theta);
204 ny = sinf(normal_theta);
205 } else {
206 // 73.5 ms per frame on 5S, example.pnm
207 double ty = -2*Cxy;
208 double tx = (Cyy - Cxx);
209 double mag = ty*ty + tx*tx;
210
211 if (mag == 0) {
212 nx = 1;
213 ny = 0;
214 } else {
215 float norm = sqrtf((float)(ty*ty + tx*tx));
216 tx /= norm;
217
218 // ty is now sin(2theta)
219 // tx is now cos(2theta). We want sin(theta) and cos(theta)
220
221 // due to precision err, tx could still have slightly too large magnitude.
222 if (tx > 1) {
223 ny = 0;
224 nx = 1;
225 } else if (tx < -1) {
226 ny = 1;
227 nx = 0;
228 } else {
229 // half angle formula
230 ny = sqrtf((1.0f - (float)tx)*0.5f);
231 nx = sqrtf((1.0f + (float)tx)*0.5f);
232
233 // pick a consistent branch cut
234 if (ty < 0)
235 ny = - ny;
236 }
237 }
238 }
239
240 if (lineparm) {
241 lineparm[0] = Ex;
242 lineparm[1] = Ey;
243 lineparm[2] = nx;
244 lineparm[3] = ny;
245 }
246
247 // sum of squared errors =
248 //
249 // SUM_i ((p_x - ux)*nx + (p_y - uy)*ny)^2
250 // SUM_i nx*nx*(p_x - ux)^2 + 2nx*ny(p_x -ux)(p_y-uy) + ny*ny*(p_y-uy)*(p_y-uy)
251 // nx*nx*SUM_i((p_x -ux)^2) + 2nx*ny*SUM_i((p_x-ux)(p_y-uy)) + ny*ny*SUM_i((p_y-uy)^2)
252 //
253 // nx*nx*N*Cxx + 2nx*ny*N*Cxy + ny*ny*N*Cyy
254
255 // sum of squared errors
256 if (err)
257 *err = nx*nx*N*Cxx + 2*nx*ny*N*Cxy + ny*ny*N*Cyy;
258
259 // mean squared error
260 if (mse)
261 *mse = nx*nx*Cxx + 2*nx*ny*Cxy + ny*ny*Cyy;
262}

286int quad_segment_maxima(const DetectorParameters &td, int sz, struct line_fit_pt *lfps, int indices[4]){
287
288 // ksz: when fitting points, how many points on either side do we consider?
289 // (actual "kernel" width is 2ksz).
290 //
291 // This value should be about: 0.5 * (points along shortest edge).
292 //
293 // If all edges were equally-sized, that would give a value of
294 // sz/8. We make it somewhat smaller to account for tags at high
295 // aspects.
296
297 // XXX Tunable. Maybe make a multiple of JPEG block size to increase robustness
298 // to JPEG compression artifacts?
299 //int ksz = imin(20, sz / 12);
300 int ksz = 20 < (sz/12)? 20: (sz/12);
301
302 // can't fit a quad if there are too few points.
303 if (ksz < 2)
304 return 0;
305
306 // printf("sz %5d, ksz %3d\n", sz, ksz);
307
308 std::vector<double> errs(sz);
309
310 for (int i = 0; i < sz; i++) {
311 fit_line(lfps, sz, (i + sz - ksz) % sz, (i + ksz) % sz, NULL, &errs[i], NULL);
312 }
313
314 // apply a low-pass filter to errs
315 if (1) {
316 std::vector<double> y(sz);
317
318 // how much filter to apply?
319
320 // XXX Tunable
321 double sigma = 1; // was 3
322
323 // cutoff = exp(-j*j/(2*sigma*sigma));
324 // log(cutoff) = -j*j / (2*sigma*sigma)
325 // log(cutoff)*2*sigma*sigma = -j*j;
326
327 // how big a filter should we use? We make our kernel big
328 // enough such that we represent any values larger than
329 // 'cutoff'.
330
331 // XXX Tunable (though not super useful to change)
332 double cutoff = 0.05;
333 int fsz = cvFloor(sqrt(-log(cutoff)*2*sigma*sigma)) + 1;
334 fsz = 2*fsz + 1;
335
336 // For default values of cutoff = 0.05, sigma = 3,
337 // we have fsz = 17.
338 std::vector<float> f(fsz);
339
340 for (int i = 0; i < fsz; i++) {
341 int j = i - fsz / 2;
342 f[i] = (float)exp(-j*j/(2*sigma*sigma));
343 }
344
345 for (int iy = 0; iy < sz; iy++) {
346 double acc = 0;
347
348 for (int i = 0; i < fsz; i++) {
349 acc += errs[(iy + i - fsz / 2 + sz) % sz] * f[i];
350 }
351 y[iy] = acc;
352 }
353 std::copy(y.begin(), y.end(), errs.begin());
354 }
355
356 std::vector<int> maxima(sz);
357 std::vector<double> maxima_errs(sz);
358 int nmaxima = 0;
359
360 for (int i = 0; i < sz; i++) {
361 if (errs[i] > errs[(i+1)%sz] && errs[i] > errs[(i+sz-1)%sz]) {
362 maxima[nmaxima] = i;
363 maxima_errs[nmaxima] = errs[i];
364 nmaxima++;
365 }
366 }
367 // if we didn't get at least 4 maxima, we can't fit a quad.
368 if (nmaxima < 4)
369 return 0;
370
371 // select only the best maxima if we have too many
372 int max_nmaxima = td.aprilTagMaxNmaxima;
373
374 if (nmaxima > max_nmaxima) {
375 std::vector<double> maxima_errs_copy(maxima_errs.begin(), maxima_errs.begin()+nmaxima);
376
377 // throw out all but the best handful of maxima. Sorts descending.
378 qsort(maxima_errs_copy.data(), nmaxima, sizeof(double), err_compare_descending);
379
380 double maxima_thresh = maxima_errs_copy[max_nmaxima];
381 int out = 0;
382 for (int in = 0; in < nmaxima; in++) {
383 if (maxima_errs[in] <= maxima_thresh)
384 continue;
385 maxima[out++] = maxima[in];
386 }
387 nmaxima = out;
388 }
389
390 int best_indices[4];
391 double best_error = HUGE_VALF;
392
393 double err01, err12, err23, err30;
394 double mse01, mse12, mse23, mse30;
395 double params01[4], params12[4], params23[4], params30[4];
396
397 // disallow quads where the angle is less than a critical value.
398 double max_dot = cos(td.aprilTagCriticalRad); //25*M_PI/180);
399
400 for (int m0 = 0; m0 < nmaxima - 3; m0++) {
401 int i0 = maxima[m0];
402
403 for (int m1 = m0+1; m1 < nmaxima - 2; m1++) {
404 int i1 = maxima[m1];
405
406 fit_line(lfps, sz, i0, i1, params01, &err01, &mse01);
407
408 if (mse01 > td.aprilTagMaxLineFitMse)
409 continue;
410
411 for (int m2 = m1+1; m2 < nmaxima - 1; m2++) {
412 int i2 = maxima[m2];
413
414 fit_line(lfps, sz, i1, i2, params12, &err12, &mse12);
415 if (mse12 > td.aprilTagMaxLineFitMse)
416 continue;
417
418 double dot = params01[2]*params12[2] + params01[3]*params12[3];
419 if (fabs(dot) > max_dot)
420 continue;
421
422 for (int m3 = m2+1; m3 < nmaxima; m3++) {
423 int i3 = maxima[m3];
424
425 fit_line(lfps, sz, i2, i3, params23, &err23, &mse23);
426 if (mse23 > td.aprilTagMaxLineFitMse)
427 continue;
428
429 fit_line(lfps, sz, i3, i0, params30, &err30, &mse30);
430 if (mse30 > td.aprilTagMaxLineFitMse)
431 continue;
432
433 double err = err01 + err12 + err23 + err30;
434 if (err < best_error) {
435 best_error = err;
436 best_indices[0] = i0;
437 best_indices[1] = i1;
438 best_indices[2] = i2;
439 best_indices[3] = i3;
440 }
441 }
442 }
443 }
444 }
445
446 if (best_error == HUGE_VALF)
447 return 0;
448
449 for (int i = 0; i < 4; i++)
450 indices[i] = best_indices[i];
451
452 if (best_error / sz < td.aprilTagMaxLineFitMse)
453 return 1;
454 return 0;
455}

460int quad_segment_agg(int sz, struct line_fit_pt *lfps, int indices[4]){
461 //int sz = zarray_size(cluster);
462
463 zmaxheap_t *heap = zmaxheap_create(sizeof(struct remove_vertex*));
464
465 // We will initially allocate sz rvs. We then have two types of
466 // iterations: some iterations that are no-ops in terms of
467 // allocations, and those that remove a vertex and allocate two
468 // more children. This will happen at most (sz-4) times. Thus we
469 // need: sz + 2*(sz-4) entries.
470
471 int rvalloc_pos = 0;
472 int rvalloc_size = 3*sz;
473 cv::AutoBuffer<struct remove_vertex, 0> rvalloc_(std::max(1, rvalloc_size));
474 memset(rvalloc_.data(), 0, sizeof(rvalloc_[0]) * rvalloc_.size()); // TODO Add AutoBuffer zero fill
475 struct remove_vertex *rvalloc = rvalloc_.data();
476 cv::AutoBuffer<struct segment, 0> segs_(std::max(1, sz)); // TODO Add AutoBuffer zero fill
477 memset(segs_.data(), 0, sizeof(segs_[0]) * segs_.size());
478 struct segment *segs = segs_.data();
479
480 // populate with initial entries
481 for (int i = 0; i < sz; i++) {
482 struct remove_vertex *rv = &rvalloc[rvalloc_pos++];
483 rv->i = i;
484 if (i == 0) {
485 rv->left = sz-1;
486 rv->right = 1;
487 } else {
488 rv->left = i-1;
489 rv->right = (i+1) % sz;
490 }
491
492 fit_line(lfps, sz, rv->left, rv->right, NULL, NULL, &rv->err);
493
494 //TODO is finite CV_Assert():
495 CV_DbgAssert (!cvIsNaN(-rv->err) && "zmaxheap_add: Trying to add non-finite number to heap. NaN's prohibited, could allow INF with testing");
496 zmaxheap_add(heap, &rv, (float)-rv->err);
497
498 segs[i].left = rv->left;
499 segs[i].right = rv->right;
500 segs[i].is_vertex = 1;
501 }
502
503 int nvertices = sz;
504
505 while (nvertices > 4) {
506 CV_Assert(rvalloc_pos < rvalloc_size);
507
508 struct remove_vertex *rv;
509 float err;
510
511 int res = zmaxheap_remove_max(heap, &rv, &err);
512 if (!res)
513 return 0;
514 CV_Assert(res);
515
516 // is this remove_vertex valid? (Or has one of the left/right
517 // vertices changes since we last looked?)
518 if (!segs[rv->i].is_vertex ||
519 !segs[rv->left].is_vertex ||
520 !segs[rv->right].is_vertex) {
521 continue;
522 }
523
524 // we now merge.
525 CV_Assert(segs[rv->i].is_vertex);
526
527 segs[rv->i].is_vertex = 0;
528 segs[rv->left].right = rv->right;
529 segs[rv->right].left = rv->left;
530
531 // create the join to the left
532 if (1) {
533 struct remove_vertex *child = &rvalloc[rvalloc_pos++];
534 child->i = rv->left;
535 child->left = segs[rv->left].left;
536 child->right = rv->right;
537
538 fit_line(lfps, sz, child->left, child->right, NULL, NULL, &child->err);
539
540 //TODO is finite CV_Assert():
541 CV_DbgAssert (!cvIsNaN(-child->err) && "zmaxheap_add: Trying to add non-finite number to heap. NaN's prohibited, could allow INF with testing");
542 zmaxheap_add(heap, &child, (float)-child->err);
543 }
544
545 // create the join to the right
546 if (1) {
547 struct remove_vertex *child = &rvalloc[rvalloc_pos++];
548 child->i = rv->right;
549 child->left = rv->left;
550 child->right = segs[rv->right].right;
551
552 fit_line(lfps, sz, child->left, child->right, NULL, NULL, &child->err);
553
554 //TODO is finite CV_Assert():
555 CV_DbgAssert (!cvIsNaN(-child->err) && "zmaxheap_add: Trying to add non-finite number to heap. NaN's prohibited, could allow INF with testing");
556 zmaxheap_add(heap, &child, (float)-child->err);
557 }
558
559 // we now have one less vertex
560 nvertices--;
561 }
562
563 zmaxheap_destroy(heap);
564
565 int idx = 0;
566 for (int i = 0; i < sz; i++) {
567 if (segs[i].is_vertex) {
568 indices[idx++] = i;
569 }
570 }
571
572 return 1;
573}

579int fit_quad(const DetectorParameters &_params, const Mat im, zarray_t *cluster, struct sQuad *quad){
580 CV_Assert(cluster != NULL);
581
582 int res = 0;
583
584 int sz = _zarray_size(cluster);
585 if (sz < 4) // can't fit a quad to less than 4 points
586 return 0;
587
589 // Step 1. Sort points so they wrap around the center of the
590 // quad. We will constrain our quad fit to simply partition this
591 // ordered set into 4 groups.
592
593 // compute a bounding box so that we can order the points
594 // according to their angle WRT the center.
595 int32_t xmax = 0, xmin = INT32_MAX, ymax = 0, ymin = INT32_MAX;
596
597 for (int pidx = 0; pidx < sz; pidx++) {
598 struct pt *p;
599 _zarray_get_volatile(cluster, pidx, &p);
600
601 //(a > b) ? a : b;
602 //xmax = imax(xmax, p->x);
603 //xmin = imin(xmin, p->x);
604 //ymax = imax(ymax, p->y);
605 //ymin = imin(ymin, p->y);
606
607 xmax = xmax > p->x? xmax : p->x;
608 xmin = xmin < p->x? xmin : p->x;
609
610 ymax = ymax > p->y? ymax : p->y;
611 ymin = ymin < p->y? ymin : p->y;
612 }
613
614 // add some noise to (cx,cy) so that pixels get a more diverse set
615 // of theta estimates. This will help us remove more points.
616 // (Only helps a small amount. The actual noise values here don't
617 // matter much at all, but we want them [-1, 1]. (XXX with
618 // fixed-point, should range be bigger?)
619 double cx = (xmin + xmax) * 0.5 + 0.05118;
620 double cy = (ymin + ymax) * 0.5 + -0.028581;
621
622 double dot = 0;
623
624 for (int pidx = 0; pidx < sz; pidx++) {
625 struct pt *p;
626 _zarray_get_volatile(cluster, pidx, &p);
627
628 double dx = p->x - cx;
629 double dy = p->y - cy;
630
631 p->theta = cv::fastAtan2((float)dy, (float)dx) * (float)(CV_PI/180);
632
633 dot += dx*p->gx + dy*p->gy;
634 }
635
636 // Ensure that the black border is inside the white border.
637 if (dot < 0)
638 return 0;
639
640 // we now sort the points according to theta. This is a preparatory
641 // step for segmenting them into four lines.
642 if (1) {
643 // zarray_sort(cluster, pt_compare_theta);
644 ptsort((struct pt*) cluster->data, sz);
645
646 // remove duplicate points. (A byproduct of our segmentation system.)
647 if (1) {
648 int outpos = 1;
649
650 struct pt *last;
651 _zarray_get_volatile(cluster, 0, &last);
652
653 for (int i = 1; i < sz; i++) {
654
655 struct pt *p;
656 _zarray_get_volatile(cluster, i, &p);
657
658 if (p->x != last->x || p->y != last->y) {
659
660 if (i != outpos) {
661 struct pt *out;
662 _zarray_get_volatile(cluster, outpos, &out);
663 memcpy(out, p, sizeof(struct pt));
664 }
665
666 outpos++;
667 }
668
669 last = p;
670 }
671
672 cluster->size = outpos;
673 sz = outpos;
674 }
675
676 } else {
677 // This is a counting sort in which we retain at most one
678 // point for every bucket; the bucket index is computed from
679 // theta. Since a good quad completes a complete revolution,
680 // there's reason to think that we should get a good
681 // distribution of thetas. We might "lose" a few points due
682 // to collisions, but this shouldn't affect quality very much.
683
684 // XXX tunable. Increase to reduce the likelihood of "losing"
685 // points due to collisions.
686 int nbuckets = 4*sz;
687
688#define ASSOC 2
689 std::vector<std::vector<struct pt> > v(nbuckets, std::vector<struct pt>(ASSOC));
690
691 // put each point into a bucket.
692 for (int i = 0; i < sz; i++) {
693 struct pt *p;
694 _zarray_get_volatile(cluster, i, &p);
695
696 CV_Assert(p->theta >= -CV_PI && p->theta <= CV_PI);
697
698 int bucket = cvFloor((nbuckets - 1) * (p->theta + CV_PI) / (2*CV_PI));
699 CV_Assert(bucket >= 0 && bucket < nbuckets);
700
701 for (int j = 0; j < ASSOC; j++) {
702 if (v[bucket][j].theta == 0) {
703 v[bucket][j] = *p;
704 break;
705 }
706 }
707 }
708
709 // collect the points from the buckets and put them back into the array.
710 int outsz = 0;
711 for (int i = 0; i < nbuckets; i++) {
712 for (int j = 0; j < ASSOC; j++) {
713 if (v[i][j].theta != 0) {
714 _zarray_set(cluster, outsz, &v[i][j], NULL);
715 outsz++;
716 }
717 }
718 }
719
720 _zarray_truncate(cluster, outsz);
721 sz = outsz;
722 }
723
724 if (sz < 4)
725 return 0;
726
728 // Step 2. Precompute statistics that allow line fit queries to be
729 // efficiently computed for any contiguous range of indices.
730
731 cv::AutoBuffer<struct line_fit_pt, 64> lfps_(sz);
732 memset(lfps_.data(), 0, sizeof(lfps_[0]) * lfps_.size()); // TODO Add AutoBuffer zero fill
733 struct line_fit_pt *lfps = lfps_.data();
734
735 for (int i = 0; i < sz; i++) {
736 struct pt *p;
737 _zarray_get_volatile(cluster, i, &p);
738
739 if (i > 0) {
740 memcpy(&lfps[i], &lfps[i-1], sizeof(struct line_fit_pt));
741 }
742
743 if (0) {
744 // we now undo our fixed-point arithmetic.
745 double delta = 0.5;
746 double x = p->x * .5 + delta;
747 double y = p->y * .5 + delta;
748 double W;
749
750 for (int dy = -1; dy <= 1; dy++) {
751 int iy = cvFloor(y + dy);
752
753 if (iy < 0 || iy + 1 >= im.rows)
754 continue;
755
756 for (int dx = -1; dx <= 1; dx++) {
757 int ix = cvFloor(x + dx);
758
759 if (ix < 0 || ix + 1 >= im.cols)
760 continue;
761
762 int grad_x = im.data[iy * im.cols + ix + 1] -
763 im.data[iy * im.cols + ix - 1];
764
765 int grad_y = im.data[(iy+1) * im.cols + ix] -
766 im.data[(iy-1) * im.cols + ix];
767
768 W = sqrtf(float(grad_x*grad_x + grad_y*grad_y)) + 1;
769
770 // double fx = x + dx, fy = y + dy;
771 double fx = ix + .5, fy = iy + .5;
772 lfps[i].Mx += W * fx;
773 lfps[i].My += W * fy;
774 lfps[i].Mxx += W * fx * fx;
775 lfps[i].Mxy += W * fx * fy;
776 lfps[i].Myy += W * fy * fy;
777 lfps[i].W += W;
778 }
779 }
780 } else {
781 // we now undo our fixed-point arithmetic.
782 double delta = 0.5; // adjust for pixel center bias
783 double x = p->x * .5 + delta;
784 double y = p->y * .5 + delta;
785 int ix = cvFloor(x), iy = cvFloor(y);
786 double W = 1;
787
788 if (ix > 0 && ix+1 < im.cols && iy > 0 && iy+1 < im.rows) {
789 int grad_x = im.data[iy * im.cols + ix + 1] -
790 im.data[iy * im.cols + ix - 1];
791
792 int grad_y = im.data[(iy+1) * im.cols + ix] -
793 im.data[(iy-1) * im.cols + ix];
794
795 // XXX Tunable. How to shape the gradient magnitude?
796 W = sqrt(grad_x*grad_x + grad_y*grad_y) + 1;
797 }
798
799 double fx = x, fy = y;
800 lfps[i].Mx += W * fx;
801 lfps[i].My += W * fy;
802 lfps[i].Mxx += W * fx * fx;
803 lfps[i].Mxy += W * fx * fy;
804 lfps[i].Myy += W * fy * fy;
805 lfps[i].W += W;
806 }
807 }
808
809 int indices[4];
810 if (1) {
811 if (!quad_segment_maxima(_params, _zarray_size(cluster), lfps, indices))
812 goto finish;
813 } else {
814 if (!quad_segment_agg(sz, lfps, indices))
815 goto finish;
816 }
817
818 // printf("%d %d %d %d\n", indices[0], indices[1], indices[2], indices[3]);
819
820 if (0) {
821 // no refitting here; just use those points as the vertices.
822 // Note, this is useful for debugging, but pretty bad in
823 // practice since this code path also omits several
824 // plausibility checks that save us tons of time in quad
825 // decoding.
826 for (int i = 0; i < 4; i++) {
827 struct pt *p;
828 _zarray_get_volatile(cluster, indices[i], &p);
829
830 quad->p[i][0] = (float)(.5*p->x); // undo fixed-point arith.
831 quad->p[i][1] = (float)(.5*p->y);
832 }
833
834 res = 1;
835
836 } else {
837 double lines[4][4];
838
839 for (int i = 0; i < 4; i++) {
840 int i0 = indices[i];
841 int i1 = indices[(i+1)&3];
842
843 if (0) {
844 // if there are enough points, skip the points near the corners
845 // (because those tend not to be very good.)
846 if (i1-i0 > 8) {
847 int t = (i1-i0)/6;
848 if (t < 0)
849 t = -t;
850
851 i0 = (i0 + t) % sz;
852 i1 = (i1 + sz - t) % sz;
853 }
854 }
855
856 double err;
857 fit_line(lfps, sz, i0, i1, lines[i], NULL, &err);
858
859 if (err > _params.aprilTagMaxLineFitMse) {
860 res = 0;
861 goto finish;
862 }
863 }
864
865 for (int i = 0; i < 4; i++) {
866 // solve for the intersection of lines (i) and (i+1)&3.
867 // p0 + lambda0*u0 = p1 + lambda1*u1, where u0 and u1
868 // are the line directions.
869 //
870 // lambda0*u0 - lambda1*u1 = (p1 - p0)
871 //
872 // rearrange (solve for lambdas)
873 //
874 // [u0_x -u1_x ] [lambda0] = [ p1_x - p0_x ]
875 // [u0_y -u1_y ] [lambda1] [ p1_y - p0_y ]
876 //
877 // remember that lines[i][0,1] = p, lines[i][2,3] = NORMAL vector.
878 // We want the unit vector, so we need the perpendiculars. Thus, below
879 // we have swapped the x and y components and flipped the y components.
880
881 const int i1 = (i + 1) & 3;
882 double A00 = lines[i][3], A01 = -lines[i1][3];
883 double A10 = -lines[i][2], A11 = lines[i1][2];
884 double B0 = -lines[i][0] + lines[i1][0];
885 double B1 = -lines[i][1] + lines[i1][1];
886
887 double det = A00 * A11 - A10 * A01;
888 if (fabs(det) < 0.001) {
889 res = 0;
890 goto finish;
891 }
892
893 // inverse.
894 double det_inv = 1.0 / det;
895 double W00 = A11 * det_inv, W01 = -A01 * det_inv;
896
897 // solve
898 double L0 = W00*B0 + W01*B1;
899
900 // compute intersection
901 quad->p[i][0] = (float)(lines[i][0] + L0*A00);
902 quad->p[i][1] = (float)(lines[i][1] + L0*A10);
903
904#if !defined(NDEBUG)
905 {
906 // we should get the same intersection starting
907 // from point p1 and moving L1*u1.
908 double W10 = -A10 * det_inv, W11 = A00 * det_inv;
909 double L1 = W10*B0 + W11*B1;
910
911 double x = lines[i1][0] - L1*A01;
912 double y = lines[i1][1] - L1*A11;
913
914 CV_Assert(fabs(x - quad->p[i][0]) < 0.001);
915 CV_Assert(fabs(y - quad->p[i][1]) < 0.001);
916 }
917#endif // NDEBUG
918
919 res = 1;
920 }
921 }
922
923 // reject quads that are too small
924 if (1) {
925 double area = 0;
926
927 // get area of triangle formed by points 0, 1, 2, 0
928 double length[3], p;
929 for (int i = 0; i < 3; i++) {
930 int idxa = i; // 0, 1, 2,
931 int idxb = (i+1) % 3; // 1, 2, 0
932 //length[i] = sqrt(
933 // sq(quad->p[idxb][0] - quad->p[idxa][0])
934 // + sq(quad->p[idxb][1] - quad->p[idxa][1]));
935 double sq1 = quad->p[idxb][0] - quad->p[idxa][0];
936 sq1 = sq1 * sq1;
937 double sq2 = quad->p[idxb][1] - quad->p[idxa][1];
938 sq2 = sq2 * sq2;
939 length[i] = sqrt(sq1 + sq2);
940 }
941 p = (length[0] + length[1] + length[2]) / 2;
942
943 area += sqrt(p*(p-length[0])*(p-length[1])*(p-length[2]));
944
945 // get area of triangle formed by points 2, 3, 0, 2
946 for (int i = 0; i < 3; i++) {
947 int idxs[] = { 2, 3, 0, 2 };
948 int idxa = idxs[i];
949 int idxb = idxs[i+1];
950 //length[i] = sqrt(
951 // sq(quad->p[idxb][0] - quad->p[idxa][0])
952 // + sq(quad->p[idxb][1] - quad->p[idxa][1]));
953 double sq1 = quad->p[idxb][0] - quad->p[idxa][0];
954 sq1 = sq1 * sq1;
955 double sq2 = quad->p[idxb][1] - quad->p[idxa][1];
956 sq2 = sq2 * sq2;
957 length[i] = sqrt(sq1 + sq2);
958 }
959 p = (length[0] + length[1] + length[2]) / 2;
960
961 area += sqrt(p*(p-length[0])*(p-length[1])*(p-length[2]));
962
963 // we don't actually know the family yet (quad detection is generic.)
964 // This threshold is based on a 6x6 tag (which is actually 8x8)
965 // int d = fam->d + fam->black_border*2;
966 int d = 8;
967 if (area < d*d) {
968 res = 0;
969 goto finish;
970 }
971 }
972
973 // reject quads whose cumulative angle change isn't equal to 2PI
974 if(1){
975 double total = 0;
976
977 for (int i = 0; i < 4; i++) {
978 int i0 = i, i1 = (i+1)&3, i2 = (i+2)&3;
979
980 double theta0 = atan2f(quad->p[i0][1] - quad->p[i1][1],
981 quad->p[i0][0] - quad->p[i1][0]);
982 double theta1 = atan2f(quad->p[i2][1] - quad->p[i1][1],
983 quad->p[i2][0] - quad->p[i1][0]);
984
985 double dtheta = theta0 - theta1;
986 if (dtheta < 0)
987 dtheta += 2*CV_PI;
988
989 if (dtheta < _params.aprilTagCriticalRad || dtheta > (CV_PI - _params.aprilTagCriticalRad))
990 res = 0;
991
992 total += dtheta;
993 }
994
995 // looking for 2PI
996 if (total < 6.2 || total > 6.4) {
997 res = 0;
998 goto finish;
999 }
1000 }
1001
1002 finish:
1003
1004 return res;
1005}

1047void threshold(const Mat mIm, const DetectorParameters &parameters, Mat& mThresh){
1048 int w = mIm.cols, h = mIm.rows;
1049 int s = (unsigned) mIm.step;
1050 CV_Assert(w < 32768);
1051 CV_Assert(h < 32768);
1052
1053 CV_Assert(mThresh.step == (unsigned)s);
1054
1055 // The idea is to find the maximum and minimum values in a
1056 // window around each pixel. If it's a contrast-free region
1057 // (max-min is small), don't try to binarize. Otherwise,
1058 // threshold according to (max+min)/2.
1059 //
1060 // Mark low-contrast regions with value 127 so that we can skip
1061 // future work on these areas too.
1062
1063 // however, computing max/min around every pixel is needlessly
1064 // expensive. We compute max/min for tiles. To avoid artifacts
1065 // that arise when high-contrast features appear near a tile
1066 // edge (and thus moving from one tile to another results in a
1067 // large change in max/min value), the max/min values used for
1068 // any pixel are computed from all 3x3 surrounding tiles. Thus,
1069 // the max/min sampling area for nearby pixels overlap by at least
1070 // one tile.
1071 //
1072 // The important thing is that the windows be large enough to
1073 // capture edge transitions; the tag does not need to fit into
1074 // a tile.
1075
1076 // XXX Tunable. Generally, small tile sizes--- so long as they're
1077 // large enough to span a single tag edge--- seem to be a winner.
1078 const int tilesz = 4;
1079
1080 // the last (possibly partial) tiles along each row and column will
1081 // just use the min/max value from the last full tile.
1082 int tw = w / tilesz;
1083 int th = h / tilesz;
1084
1085 uint8_t *im_max = (uint8_t*)calloc(tw*th, sizeof(uint8_t));
1086 uint8_t *im_min = (uint8_t*)calloc(tw*th, sizeof(uint8_t));
1087
1088
1089 // first, collect min/max statistics for each tile
1090 for (int ty = 0; ty < th; ty++) {
1091 for (int tx = 0; tx < tw; tx++) {
1092 uint8_t max = 0, min = 255;
1093
1094 for (int dy = 0; dy < tilesz; dy++) {
1095
1096 for (int dx = 0; dx < tilesz; dx++) {
1097
1098 uint8_t v = mIm.data[(ty*tilesz+dy)*s + tx*tilesz + dx];
1099 if (v < min)
1100 min = v;
1101 if (v > max)
1102 max = v;
1103 }
1104 }
1105 im_max[ty*tw+tx] = max;
1106 im_min[ty*tw+tx] = min;
1107 }
1108 }
1109
1110 // second, apply 3x3 max/min convolution to "blur" these values
1111 // over larger areas. This reduces artifacts due to abrupt changes
1112 // in the threshold value.
1113 uint8_t *im_max_tmp = (uint8_t*)calloc(tw*th, sizeof(uint8_t));
1114 uint8_t *im_min_tmp = (uint8_t*)calloc(tw*th, sizeof(uint8_t));
1115
1116 for (int ty = 0; ty < th; ty++) {
1117 for (int tx = 0; tx < tw; tx++) {
1118 uint8_t max = 0, min = 255;
1119
1120 for (int dy = -1; dy <= 1; dy++) {
1121 if (ty+dy < 0 || ty+dy >= th)
1122 continue;
1123 for (int dx = -1; dx <= 1; dx++) {
1124 if (tx+dx < 0 || tx+dx >= tw)
1125 continue;
1126
1127 uint8_t m = im_max[(ty+dy)*tw+tx+dx];
1128 if (m > max)
1129 max = m;
1130 m = im_min[(ty+dy)*tw+tx+dx];
1131 if (m < min)
1132 min = m;
1133 }
1134 }
1135
1136 im_max_tmp[ty*tw + tx] = max;
1137 im_min_tmp[ty*tw + tx] = min;
1138 }
1139 }
1140 free(im_max);
1141 free(im_min);
1142 im_max = im_max_tmp;
1143 im_min = im_min_tmp;
1144
1145 for (int ty = 0; ty < th; ty++) {
1146 for (int tx = 0; tx < tw; tx++) {
1147
1148 int min_ = im_min[ty*tw + tx];
1149 int max_ = im_max[ty*tw + tx];
1150
1151 // low contrast region? (no edges)
1152 if (max_ - min_ < parameters.aprilTagMinWhiteBlackDiff) {
1153 for (int dy = 0; dy < tilesz; dy++) {
1154 int y = ty*tilesz + dy;
1155
1156 for (int dx = 0; dx < tilesz; dx++) {
1157 int x = tx*tilesz + dx;
1158
1159 //threshim->buf[y*s+x] = 127;
1160 mThresh.data[y*s+x] = 127;
1161 }
1162 }
1163 continue;
1164 }
1165
1166 // otherwise, actually threshold this tile.
1167
1168 // argument for biasing towards dark; specular highlights
1169 // can be substantially brighter than white tag parts
1170 uint8_t thresh = saturate_cast<uint8_t>((max_ + min_) / 2);
1171
1172 for (int dy = 0; dy < tilesz; dy++) {
1173 int y = ty*tilesz + dy;
1174
1175 for (int dx = 0; dx < tilesz; dx++) {
1176 int x = tx*tilesz + dx;
1177
1178 uint8_t v = mIm.data[y*s+x];
1179 mThresh.data[y*s+x] = (v > thresh) ? 255 : 0;
1180 }
1181 }
1182 }
1183 }
1184
1185 // we skipped over the non-full-sized tiles above. Fix those now.
1186 for (int y = 0; y < h; y++) {
1187
1188 // what is the first x coordinate we need to process in this row?
1189
1190 int x0;
1191
1192 if (y >= th*tilesz) {
1193 x0 = 0; // we're at the bottom; do the whole row.
1194 } else {
1195 x0 = tw*tilesz; // we only need to do the right most part.
1196 }
1197
1198 // compute tile coordinates and clamp.
1199 int ty = y / tilesz;
1200 if (ty >= th)
1201 ty = th - 1;
1202
1203 for (int x = x0; x < w; x++) {
1204 int tx = x / tilesz;
1205 if (tx >= tw)
1206 tx = tw - 1;
1207
1208 int max = im_max[ty*tw + tx];
1209 int min = im_min[ty*tw + tx];
1210 int thresh = min + (max - min) / 2;
1211
1212 uint8_t v = mIm.data[y*s+x];
1213 if (v > thresh){
1214 mThresh.data[y*s+x] = 255;
1215 }
1216 else{
1217 mThresh.data[y*s+x] = 0;
1218 }
1219 }
1220 }
1221 free(im_min);
1222 free(im_max);
1223
1224 // this is a dilate/erode deglitching scheme that does not improve
1225 // anything as far as I can tell.
1226 if (parameters.aprilTagDeglitch) {
1227 Mat tmp(h,w, mIm.type());
1228 for (int y = 1; y + 1 < h; y++) {
1229 for (int x = 1; x + 1 < w; x++) {
1230 uint8_t max = 0;
1231 for (int dy = -1; dy <= 1; dy++) {
1232 for (int dx = -1; dx <= 1; dx++) {
1233 uint8_t v = mThresh.data[(y+dy)*s + x + dx];
1234 if (v > max)
1235 max = v;
1236 }
1237 }
1238 tmp.data[y*s+x] = max;
1239 }
1240 }
1241
1242 for (int y = 1; y + 1 < h; y++) {
1243 for (int x = 1; x + 1 < w; x++) {
1244 uint8_t min = 255;
1245 for (int dy = -1; dy <= 1; dy++) {
1246 for (int dx = -1; dx <= 1; dx++) {
1247 uint8_t v = tmp.data[(y+dy)*s + x + dx];
1248 if (v < min)
1249 min = v;
1250 }
1251 }
1252 mThresh.data[y*s+x] = min;
1253 }
1254 }
1255 }
1256
1257}

1269zarray_t *apriltag_quad_thresh(const DetectorParameters &parameters, const Mat & mImg, std::vector<std::vector<Point> > &contours){
1270
1272 // step 1. threshold the image, creating the edge image.
1273
1274 int w = mImg.cols, h = mImg.rows;
1275
1276 Mat thold(h, w, mImg.type());
1277 threshold(mImg, parameters, thold);
1278
1279 int ts = thold.cols;
1280
1281#ifdef APRIL_DEBUG
1282 imwrite("2.2 debug_threshold.pnm", thold);
1283#endif
1284
1286 // step 2. find connected components.
1287
1288 UnionFind uf(w * h);
1289
1290 // TODO PARALLELIZE
1291 for (int y = 0; y < h - 1; y++) {
1292 uf.do_line(thold, w, ts, y);
1293 }
1294
1295 // XXX sizing??
1296 int nclustermap = 2*w*h - 1;
1297
1298 struct uint64_zarray_entry **clustermap = (struct uint64_zarray_entry**)calloc(nclustermap, sizeof(struct uint64_zarray_entry*));
1299
1300 for (int y = 1; y < h-1; y++) {
1301 for (int x = 1; x < w-1; x++) {
1302
1303 uint8_t v0 = thold.data[y*ts + x];
1304 if (v0 == 127)
1305 continue;
1306
1307 // XXX don't query this until we know we need it?
1308 uint64_t rep0 = uf.get_representative(y*w + x);
1309
1310 // whenever we find two adjacent pixels such that one is
1311 // white and the other black, we add the point half-way
1312 // between them to a cluster associated with the unique
1313 // ids of the white and black regions.
1314 //
1315 // We additionally compute the gradient direction (i.e., which
1316 // direction was the white pixel?) Note: if (v1-v0) == 255, then
1317 // (dx,dy) points towards the white pixel. if (v1-v0) == -255, then
1318 // (dx,dy) points towards the black pixel. p.gx and p.gy will thus
1319 // be -255, 0, or 255.
1320 //
1321 // Note that any given pixel might be added to multiple
1322 // different clusters. But in the common case, a given
1323 // pixel will be added multiple times to the same cluster,
1324 // which increases the size of the cluster and thus the
1325 // computational costs.
1326 //
1327 // A possible optimization would be to combine entries
1328 // within the same cluster.
1329
1330#define DO_CONN(dx, dy) \
1331 if (1) { \
1332 uint8_t v1 = thold.data[y*ts + dy*ts + x + dx]; \
