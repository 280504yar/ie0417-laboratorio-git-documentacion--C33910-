20template<typename T, typename VecT> static void
21vecsplit_( const T* src, T** dst, int len, int cn )
22{
23 const int VECSZ = VTraits<VecT>::vlanes();
24 int i, i0 = 0;
25 T* dst0 = dst[0];
26 T* dst1 = dst[1];
27
28 int r0 = (int)((size_t)(void*)dst0 % (VECSZ*sizeof(T)));
29 int r1 = (int)((size_t)(void*)dst1 % (VECSZ*sizeof(T)));
30 int r2 = cn > 2 ? (int)((size_t)(void*)dst[2] % (VECSZ*sizeof(T))) : r0;
31 int r3 = cn > 3 ? (int)((size_t)(void*)dst[3] % (VECSZ*sizeof(T))) : r0;
32
33 hal::StoreMode mode = hal::STORE_ALIGNED_NOCACHE;
34 if( (r0|r1|r2|r3) != 0 )
35 {
36 mode = hal::STORE_UNALIGNED;
37 if( r0 == r1 && r0 == r2 && r0 == r3 && r0 % sizeof(T) == 0 && len > VECSZ*2 )
38 i0 = VECSZ - (r0 / sizeof(T));
39 }
40
41 if( cn == 2 )
42 {
43 for( i = 0; i < len; i += VECSZ )
44 {
45 if( i > len - VECSZ )
46 {
47 i = len - VECSZ;
48 mode = hal::STORE_UNALIGNED;
49 }
50 VecT a, b;
51 v_load_deinterleave(src + i*cn, a, b);
52 v_store(dst0 + i, a, mode);
53 v_store(dst1 + i, b, mode);
54 if( i < i0 )
55 {
56 i = i0 - VECSZ;
57 mode = hal::STORE_ALIGNED_NOCACHE;
58 }
59 }
60 }
61 else if( cn == 3 )
62 {
63 T* dst2 = dst[2];
64 for( i = 0; i < len; i += VECSZ )
65 {
66 if( i > len - VECSZ )
67 {
68 i = len - VECSZ;
69 mode = hal::STORE_UNALIGNED;
70 }

72 v_load_deinterleave(src + i*cn, a, b, c);
73 v_store(dst0 + i, a, mode);
74 v_store(dst1 + i, b, mode);
75 v_store(dst2 + i, c, mode);

112template<typename T> static void
113split_( const T* src, T** dst, int len, int cn )
114{
115 int k = cn % 4 ? cn % 4 : 4;
116 int i, j;
117 if( k == 1 )
118 {
119 T* dst0 = dst[0];
120
121 if(cn == 1)
122 {
123 memcpy(dst0, src, len * sizeof(T));
124 }
125 else
126 {
127 for( i = 0, j = 0 ; i < len; i++, j += cn )
128 dst0[i] = src[j];
129 }
130 }
131 else if( k == 2 )
132 {
133 T *dst0 = dst[0], *dst1 = dst[1];
134 i = j = 0;
135
136 for( ; i < len; i++, j += cn )
137 {
138 dst0[i] = src[j];
139 dst1[i] = src[j+1];
140 }
141 }
142 else if( k == 3 )
143 {
144 T *dst0 = dst[0], *dst1 = dst[1], *dst2 = dst[2];
145 i = j = 0;
146
147 for( ; i < len; i++, j += cn )
148 {
149 dst0[i] = src[j];
150 dst1[i] = src[j+1];
151 dst2[i] = src[j+2];
152 }
153 }
154 else
155 {
156 T *dst0 = dst[0], *dst1 = dst[1], *dst2 = dst[2], *dst3 = dst[3];
157 i = j = 0;
158
159 for( ; i < len; i++, j += cn )
160 {
161 dst0[i] = src[j]; dst1[i] = src[j+1];
162 dst2[i] = src[j+2]; dst3[i] = src[j+3];
163 }
164 }
165
166 for( ; k < cn; k += 4 )
167 {
168 T *dst0 = dst[k], *dst1 = dst[k+1], *dst2 = dst[k+2], *dst3 = dst[k+3];
169 for( i = 0, j = k; i < len; i++, j += cn )
170 {
171 dst0[i] = src[j]; dst1[i] = src[j+1];
172 dst2[i] = src[j+2]; dst3[i] = src[j+3];
173 }
174 }
