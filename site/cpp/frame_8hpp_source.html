35 int cellKey(int row, int col) const { return row * gridCols + col; }

37 void buildGrid()
38 {
39 grid.clear();
40 if (imageSize.width <= 0 || imageSize.height <= 0) return;
41 gridCols = std::max(1, (imageSize.width + CELL_SIZE_PX - 1) / CELL_SIZE_PX);
42 gridRows = std::max(1, (imageSize.height + CELL_SIZE_PX - 1) / CELL_SIZE_PX);
43 for (size_t i = 0; i < undistKpts.size(); ++i)
44 {
45 int col = std::min(gridCols - 1, std::max(0, (int)(undistKpts[i].x / CELL_SIZE_PX)));
46 int row = std::min(gridRows - 1, std::max(0, (int)(undistKpts[i].y / CELL_SIZE_PX)));
47 grid[cellKey(row, col)].push_back(i);
48 }
49 }

51 std::vector<size_t> getKeypointsInRadius(float x, float y, float r) const
52 {
53 std::vector<size_t> result;
54 if (gridRows <= 0 || gridCols <= 0 || grid.empty()) return result;
55 const int colMin = std::max(0, (int)((x - r) / CELL_SIZE_PX));
56 const int colMax = std::min(gridCols-1, (int)((x + r) / CELL_SIZE_PX));
57 const int rowMin = std::max(0, (int)((y - r) / CELL_SIZE_PX));
58 const int rowMax = std::min(gridRows-1, (int)((y + r) / CELL_SIZE_PX));
59 const float r2 = r * r;
60 for (int row = rowMin; row <= rowMax; ++row)
61 for (int col = colMin; col <= colMax; ++col)
62 {
63 auto it = grid.find(cellKey(row, col));
64 if (it == grid.end()) continue;
65 for (size_t idx : it->second)
66 {
67 float dx = undistKpts[idx].x - x;
68 float dy = undistKpts[idx].y - y;
69 if (dx * dx + dy * dy <= r2)
70 result.push_back(idx);
71 }
72 }
73 return result;
74 }
75};
