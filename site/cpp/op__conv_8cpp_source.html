18OpConv::OpConv(const Mat& weightBlob, const std::vector<float>& biasvec, int _activType, const int _ngroups, const int _K,
19 const int _C, const int _Hk, const int _Wk, const int _stride_h, const int _stride_w,
20 const int _dilation_h, const int _dilation_w, const int _pad_left, const int _pad_top):
21 activ((FusedActivationType)_activType), ngroups(_ngroups), K(_K), C(_C), Hk(_Hk), Wk(_Wk), stride_h(_stride_h), stride_w(_stride_w),
22 dilation_h(_dilation_h), dilation_w(_dilation_w), pad_left(_pad_left), pad_top(_pad_top)
23{
24 type_ = kOpTypeConv;
25 CV_Assert(!weightBlob.empty());
26
27 Kg = K/ngroups, Cg = max(C/ngroups, 1);
28 ksize = Hk * Wk;
29 CgHkWk = Cg * ksize;
30 fast_1x1 = ksize == 1 && stride_w == 1 && stride_h == 1 && pad_top == 0 && pad_left == 0;
31
32 if (ngroups > 1 && ngroups == K && ngroups == C)
33 {
34 if (Hk == 3 && Wk == 3)
35 shader_name = "conv_depthwise_3x3_spv";
36 else
37 shader_name = "conv_depthwise_spv";
38
39 shaderType = kConvShaderTypeDepthWise;
40 STRIP_LEN = 16;
41 }
42 else if (fast_1x1) // 1x1
43 {
44 shader_name = "conv_1x1_fast_spv";
45 shaderType = kConvShaderTypeGeneric;
46 STRIP_LEN = 32;
47 }
48 else
49 {
50 shader_name = "conv_implicit_gemm_spv";
51 shaderType = kConvShaderTypeGeneric;
52 STRIP_LEN = 32;
53 }
54
55 CgHkWk_aligned = alignSize(CgHkWk, STRIP_LEN);
56 // repack the weight. The shape is from [K, C, Hk, Wk] to [ngroups, Ceil(K/group), Align(Cg*Hk*Wk, STRIP_LEN)]
57 if (shaderType == kConvShaderTypeGeneric)
58 {
59 std::vector<int> repackWeightShape = {ngroups, Kg, CgHkWk_aligned};
60
61 Mat repackWeight = Mat(repackWeightShape, CV_32FC1, Scalar_<float>(0.0f));
62 float* weightsBufPtr = repackWeight.ptr<float>();
63 const float* srcWeight = weightBlob.ptr<float>();
64 const size_t wstep = weightBlob.step1(); // Hk*Wk*Cg
65
66 // Pack the weight.
67 parallel_for_(Range(0, ngroups * Kg), [&](const Range& r0){
68 for (int gki = r0.start; gki < r0.end; gki++)
69 {
70 const float* wptr = srcWeight + gki * wstep;
71 float* packed_wptr = weightsBufPtr + gki * CgHkWk_aligned;
72
73 memcpy(packed_wptr, wptr, sizeof(wptr[0]) * CgHkWk);
74 }});
75
76 // Create weightTensor
77 Tensor weightTensor;
78 CV_Assert(repackWeight.isContinuous() && weightBlob.type() == CV_32F);
79 weightTensor.reshape((const char*)repackWeight.data, repackWeightShape);
80 weightTensorPtr = makePtr<Tensor>(weightTensor);
81 }
82 else
83 {
84 // Create weightTensor
85 Tensor weightTensor;
86 CV_Assert(weightBlob.isContinuous() && weightBlob.type() == CV_32F);
87 std::vector<int> matShape = shape(weightBlob).vec();
88 weightTensor.reshape((const char*)weightBlob.data, matShape); // This code will copy the src data from Mat to VkBuffer.
89
90 weightTensorPtr = makePtr<Tensor>(weightTensor);
91 }
92
93 if (!biasvec.empty())
94 {
95 int biasAlignedSize = alignSize(biasvec.size(), BLOCK_SIZE);
96 std::vector<int> biasShape = {biasAlignedSize};
97
98 biasCopy.resize(biasAlignedSize, 0.f);
99
100 for (int i = 0; i < biasvec.size(); i++)
101 {
102 biasCopy[i] = biasvec[i];
103 }
104
105 Tensor biasTensor;
106 biasTensor.reshape((const char*)biasCopy.data(), biasShape); // This code will copy the src data from Mat to VkBuffer.
107
108 biasTensorPtr = makePtr<Tensor>(biasTensor);
109 }
110 else
111 {
112 std::vector<int> shape = {K};
113 Tensor bias(0, shape);
114 biasTensorPtr = makePtr<Tensor>(bias);
115 }
116}

137bool OpConv::forward(std::vector<Tensor>& ins, std::vector<Tensor>& outs)
138{
139 CV_Assert(ins.size() == 1 && outs.size() == 1);
140 Shape inputShape = ins[0].getShape();
141 Shape outputShape = outs[0].getShape();
142 CV_Assert(inputShape.size() == outputShape.size());
143
144 batch = inputShape[kShapeIdxBatch];
145 Hi = inputShape[kShapeIdxHeight];
146 Wi = inputShape[kShapeIdxWidth];
147
148 H0 = outputShape[kShapeIdxHeight];
149 W0 = outputShape[kShapeIdxWidth];
150
151 firstForward();
152
153 std::vector<int> param = {Hi, Wi,
154 H0, W0,
155 stride_h, stride_w,
156 pad_top, pad_left,
157 Hk, Wk,
158 dilation_h, dilation_w,
159 Kg, Cg,
160 ngroups,
161 CgHkWk_aligned,
162 (int)activ,
163 0, 0};
164
165 std::vector<int> shape = {(int)param.size()};
166 destTypes = {
167 VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, // input
168 VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, // bias
169 VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, // weight
170 VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, // output
171 VK_DESCRIPTOR_TYPE_UNIFORM_BUFFER // param
172 };
173
174 Ptr<Pipeline> pipeline = pipelineFactoryPtr->getPipeline(shader_name, destTypes);
175 Ptr<Descriptor> desSet = pipeline->createSet();
176 Ptr<CommandBuffer> cmdBuffer = cmdPoolPtr->allocBuffer();
177
178 VkCommandBuffer cmdBufferReal = cmdBuffer->get();
179 desSet->writeTensor(ins[0], 0);
180 desSet->writeTensor(*biasTensorPtr, 1);
181 desSet->writeTensor(*weightTensorPtr, 2);
182 desSet->writeTensor(outs[0], 3);
183
184 if (shaderType == kConvShaderTypeGeneric)
185 {
186 for (int b = 0; b < batch; b++)
187 {
188 for (int g = 0; g < ngroups; g++)
189 {
190 param[17] = b;
191 param[18] = g;
192 Tensor paramTensor = Tensor(reinterpret_cast<const char *>(param.data()), shape, kFormatInt32, VK_BUFFER_USAGE_UNIFORM_BUFFER_BIT);
193 desSet->writeTensor(paramTensor, 4);
194
195 cmdBuffer->beginRecord();
196 pipeline->bind(cmdBufferReal, desSet->get());
197 vkCmdDispatch(cmdBufferReal, group_x_, group_y_, group_z_);
198 cmdBuffer->endRecord();
199
200 cmdPoolPtr->submitAndWait(cmdBufferReal);
201 }
202 }
203 }
204 else if (shaderType == kConvShaderTypeDepthWise)
205 {
206 for (int b = 0; b < batch; b++)
207 {
208 param[17] = b;
209 Tensor paramTensor = Tensor(reinterpret_cast<const char *>(param.data()), shape, kFormatInt32, VK_BUFFER_USAGE_UNIFORM_BUFFER_BIT);
210 desSet->writeTensor(paramTensor, 4);
211
212 cmdBuffer->beginRecord();
213 pipeline->bind(cmdBufferReal, desSet->get());
214 vkCmdDispatch(cmdBufferReal, group_x_, group_y_, group_z_);
215 cmdBuffer->endRecord();
216
217 cmdPoolPtr->submitAndWait(cmdBufferReal);
218 }
219 }
220
221 return true;
222}

224bool OpConv::computeGroupCount()
225{
226 int outplan = H0 * W0;
227 if (shaderType == kConvShaderTypeDepthWise)
228 {
229 group_x_ = alignSize(outplan, config.local_size_x) / config.local_size_x;
230 group_y_ = K;
231 group_z_ = 1;
232 return true;
233 }
234 else if (shaderType == kConvShaderTypeGeneric)
235 {
236 group_x_ = alignSize(Kg, config.local_size_x) / config.local_size_x;
237 group_y_ = alignSize(outplan, config.local_size_y) / config.local_size_y;
238 group_z_ = 1;
239 }
240 else if (shaderType == kConvShaderTest)
241 {
242 group_x_ = 1;
243 group_y_ = 1;
244 group_z_ = 1;
245 }
246 else
247 CV_Error(cv::Error::StsNotImplemented, "shader type is not supported at compute GroupCount.");
248
249 CV_Assert(group_x_ <= MAX_GROUP_COUNT_X);
250 CV_Assert(group_y_ <= MAX_GROUP_COUNT_Y);
251 CV_Assert(group_z_ <= MAX_GROUP_COUNT_Z);
252
253 return true;
254}
