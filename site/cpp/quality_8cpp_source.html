113 : error (error_), points_size (points_size_), threshold (threshold_), k_msac(k_msac_),

117 inline Score getScore (const Mat &model) const override {
118 error->setModelParameters(model);
119 float err, sum_errors = 0;
120 int inlier_number = 0;
121 const auto preemptive_thr = points_size + best_score;
122 for (int point = 0; point < points_size; point++) {
123 err = error->getError(point);
124 if (err < norm_thr) {
125 sum_errors -= (1 - err * one_over_thr);
126 if (err < threshold)
127 inlier_number++;
128 } else if (sum_errors + point > preemptive_thr)
129 break;
130 }
131 return {inlier_number, sum_errors};
132 }

139 if (err < norm_thr) {
140 sum_errors -= (1 - err * one_over_thr);

142 inlier_number++;
143 }

167class MagsacQualityImpl : public MagsacQuality {
168private:
169 const Ptr<Error> error;
170 const Ptr<GammaValues> gamma_generator;
171 const int points_size;
172 // for example, maximum standard deviation of noise.
173 const double maximum_threshold_sqr, tentative_inlier_threshold;
174 // Calculate the gamma value of k
175 const double gamma_value_of_k;
176 double previous_best_loss;
177 float maximum_sigma_2_per_2;
178 // Calculating 2^(DoF + 1) / \sigma_{max} which will be used for the estimation and,
179 // due to being constant, it is better to calculate it a priori.
180 double two_ad_dof_plus_one_per_maximum_sigma, rescale_err, norm_loss;
181 const std::vector<double> &stored_complete_gamma_values, &stored_lower_incomplete_gamma_values;
182 unsigned int stored_incomplete_gamma_number_min1;
183public:
184
185 MagsacQualityImpl (double maximum_thr, int points_size_, const Ptr<Error> &error_,
186 const Ptr<GammaValues> &gamma_generator_,
187 double tentative_inlier_threshold_, int DoF, double sigma_quantile,
188 double upper_incomplete_of_sigma_quantile)
189 : error (error_), gamma_generator(gamma_generator_), points_size(points_size_),
190 maximum_threshold_sqr(maximum_thr*maximum_thr),
191 tentative_inlier_threshold(tentative_inlier_threshold_),
192 gamma_value_of_k (upper_incomplete_of_sigma_quantile),
193 stored_complete_gamma_values (gamma_generator->getCompleteGammaValues()),
194 stored_lower_incomplete_gamma_values (gamma_generator->getIncompleteGammaValues()) {
195 previous_best_loss = std::numeric_limits<double>::max();
196 const auto maximum_sigma = (float)sqrt(maximum_threshold_sqr) / sigma_quantile;
197 const auto maximum_sigma_2 = (float) (maximum_sigma * maximum_sigma);
198 maximum_sigma_2_per_2 = maximum_sigma_2 / 2.f;
199 const auto maximum_sigma_2_times_2 = maximum_sigma_2 * 2.f;
200 two_ad_dof_plus_one_per_maximum_sigma = std::pow(2, (DoF + 1.0)*.5)/maximum_sigma;
201 rescale_err = gamma_generator->getScaleOfGammaCompleteValues() / maximum_sigma_2_times_2;
202 stored_incomplete_gamma_number_min1 = static_cast<unsigned int>(gamma_generator->getTableSize()-1);
203
204 double max_loss = 1e-10;
205 // MAGSAC maximum / minimum loss does not have to be in extremum residuals
206 // make 30 iterations to find maximum loss
207 const double step = maximum_threshold_sqr / 30;
208 double sqr_res = 0;
209 while (sqr_res < maximum_threshold_sqr) {
210 auto x= static_cast<unsigned int>(rescale_err * sqr_res);
211 if (x > stored_incomplete_gamma_number_min1)
212 x = stored_incomplete_gamma_number_min1;
213 const double loss = two_ad_dof_plus_one_per_maximum_sigma * (maximum_sigma_2_per_2 *
214 stored_lower_incomplete_gamma_values[x] + sqr_res * 0.25 *
215 (stored_complete_gamma_values[x] - gamma_value_of_k));
216 if (max_loss < loss)
217 max_loss = loss;
218 sqr_res += step;
219 }
220 norm_loss = two_ad_dof_plus_one_per_maximum_sigma / max_loss;
221 }
222
223 // https://github.com/danini/magsac
224 Score getScore (const Mat &model) const override {
225 error->setModelParameters(model);
226 double total_loss = 0.0;
227 int num_tentative_inliers = 0;
228 const auto preemptive_thr = points_size + previous_best_loss;
229 for (int point_idx = 0; point_idx < points_size; point_idx++) {
230 const float squared_residual = error->getError(point_idx);
231 if (squared_residual < tentative_inlier_threshold)
232 num_tentative_inliers++;
233 if (squared_residual < maximum_threshold_sqr) { // consider point as inlier
234 // Get the position of the gamma value in the lookup table
235 auto x = static_cast<unsigned int>(rescale_err * squared_residual);
236 // If the sought gamma value is not stored in the lookup, return the closest element
237 if (x > stored_incomplete_gamma_number_min1)
238 x = stored_incomplete_gamma_number_min1;
239 // Calculate the loss implied by the current point
240 total_loss -= (1 - (maximum_sigma_2_per_2 *
241 stored_lower_incomplete_gamma_values[x] + squared_residual * 0.25 *
242 (stored_complete_gamma_values[x] - gamma_value_of_k)) * norm_loss);
243 } else if (total_loss + point_idx > preemptive_thr)
244 break;
245 }
246 return {num_tentative_inliers, (float)total_loss};
247 }
248
249 Score getScore (const std::vector<float> &errors) const override {
250 double total_loss = 0.0;
251 int num_tentative_inliers = 0;
252 for (int point_idx = 0; point_idx < points_size; point_idx++) {
253 const float squared_residual = errors[point_idx];
254 if (squared_residual < tentative_inlier_threshold)
255 num_tentative_inliers++;
256 if (squared_residual < maximum_threshold_sqr) {
257 auto x = static_cast<unsigned int>(rescale_err * squared_residual);
258 if (x > stored_incomplete_gamma_number_min1)
259 x = stored_incomplete_gamma_number_min1;
260 total_loss -= (1 - (maximum_sigma_2_per_2 *
261 stored_lower_incomplete_gamma_values[x] + squared_residual * 0.25 *
262 (stored_complete_gamma_values[x] - gamma_value_of_k)) * norm_loss);
263 }
264 }
265 return {num_tentative_inliers, (float)total_loss};
266 }

268 void setBestScore (float best_loss) override {
269 if (previous_best_loss > best_loss) previous_best_loss = best_loss;
270 }

274 int getInliers (const Mat &model, std::vector<int> &inliers, double thr) const override
275 { return Quality::getInliers(error, model, inliers, thr); }

278 double getThreshold () const override { return tentative_inlier_threshold; }
279 int getPointsSize () const override { return points_size; }

352class AdaptiveSPRTImpl : public AdaptiveSPRT {
353private:
354 RNG rng;
355 const Ptr<Error> err;
356 const Ptr<Quality> quality;
357 const int points_size;
358 int highest_inlier_number, last_iteration;
359 // time t_M needed to instantiate a model hypothesis given a sample
360 // Let m_S be the number of models that are verified per sample
361 const double inlier_threshold, norm_thr, one_over_thr;
362
363 // alpha is false negative rate, alpha = 1 / A
364 double t_M, lowest_sum_errors, current_epsilon, current_delta, current_A,
365 delta_to_epsilon, complement_delta_to_complement_epsilon,
366 time_ver_corr_sprt = 0, time_ver_corr = 0,
367 one_over_complement_alpha, avg_num_checked_pts;
368
369 std::vector<SPRT_history> sprt_histories, empty;
370 std::vector<int> points_random_pool;
371 std::vector<float> errors;
372
373 bool do_sprt, adapt, IS_ADAPTIVE;
374 const ScoreMethod score_type;
375 double m_S;
376public:
377 AdaptiveSPRTImpl (int state, const Ptr<Quality> &quality_, int points_size_,
378 double inlier_threshold_, double prob_pt_of_good_model, double prob_pt_of_bad_model,
379 double time_sample, double avg_num_models, ScoreMethod score_type_,
380 double k_mlesac_, bool is_adaptive) : rng(state), err(quality_->getErrorFnc()),
381 quality(quality_), points_size(points_size_), inlier_threshold (quality->getThreshold()),
382 norm_thr(inlier_threshold_*k_mlesac_), one_over_thr (1/norm_thr), t_M (time_sample),
383 score_type (score_type_), m_S (avg_num_models) {
384
385 // Generate array of random points for randomized evaluation
386 points_random_pool = std::vector<int> (points_size_);
387 // fill values from 0 to points_size-1
388 for (int i = 0; i < points_size; i++)
389 points_random_pool[i] = i;
390 randShuffle(points_random_pool, 1, &rng);
391
392 // reserve (approximately) some space for sprt vector.
393 sprt_histories.reserve(20);
394
395 highest_inlier_number = last_iteration = 0;
396 lowest_sum_errors = std::numeric_limits<double>::max();
397 if (score_type_ != ScoreMethod::SCORE_METHOD_MSAC)
398 errors = std::vector<float>(points_size_);
399 IS_ADAPTIVE = is_adaptive;
400 delta_to_epsilon = one_over_complement_alpha = complement_delta_to_complement_epsilon = current_A = -1;
401 avg_num_checked_pts = points_size_;
402 adapt = IS_ADAPTIVE;
403 do_sprt = !IS_ADAPTIVE;
404 if (IS_ADAPTIVE) {
405 // all these variables will be initialized later
406 current_epsilon = prob_pt_of_good_model;
407 current_delta = prob_pt_of_bad_model;
408 } else {
409 current_epsilon = current_delta = 1e-5;
410 createTest(prob_pt_of_good_model, prob_pt_of_bad_model);
411 }
412 }
413
414 /*
415 * p(x(r)|Hb) p(x(j)|Hb)
416 * lambda(j) = Product (----------) = lambda(j-1) * ----------
417 * p(x(r)|Hg) p(x(j)|Hg)
418 * Set j = 1
419 * 1. Check whether j-th data point is consistent with the
420 * model
421 * 2. Compute the likelihood ratio λj eq. (1)
422 * 3. If λj > A, decide the model is ’bad’ (model "re-jected"),
423 * else increment j or continue testing
424 * 4. If j = N the number of correspondences decide model "accepted"
425 *
426 * Verifies model and returns model score.
427
428 * Returns true if model is good, false - otherwise.
429 * @model: model to verify
430 * @current_hypothesis: current RANSAC iteration
431 * Return: true if model is good, false - otherwise.
432 */
433 inline bool isModelGood (const Mat &model, Score &out_score) override {
434 // update error object with current model
435 bool last_model_is_good = true;
436 double sum_errors = 0;
437 int tested_inliers = 0;
438 if (! do_sprt || adapt) { // if adapt or not sprt then compute model score directly
439 out_score = quality->getScore(model);
440 tested_inliers = out_score.inlier_number;
441 sum_errors = out_score.score;
442 } else { // do sprt and not adapt
443 err->setModelParameters(model);
444 double lambda = 1;
445 int random_pool_idx = rng.uniform(0, points_size), tested_point;
446 if (score_type == ScoreMethod::SCORE_METHOD_MSAC) {
447 const auto preemptive_thr = points_size + lowest_sum_errors;
448 for (tested_point = 0; tested_point < points_size; tested_point++) {
449 if (random_pool_idx == points_size)
450 random_pool_idx = 0;
451 const float error = err->getError (points_random_pool[random_pool_idx++]);
452 if (error < inlier_threshold) {
453 tested_inliers++;
454 lambda *= delta_to_epsilon;
455 } else {
456 lambda *= complement_delta_to_complement_epsilon;
457 // since delta is always higher than epsilon, then lambda can increase only
458 // when point is not consistent with model
459 if (lambda > current_A)
460 break;
461 }
462 if (error < norm_thr)
463 sum_errors -= (1 - error * one_over_thr);
464 else if (sum_errors + tested_point > preemptive_thr)
465 break;
466 }
467 } else { // save errors into array here
468 for (tested_point = 0; tested_point < points_size; tested_point++) {
469 if (random_pool_idx == points_size)
470 random_pool_idx = 0;
471 const int pt = points_random_pool[random_pool_idx++];
472 const float error = err->getError (pt);
473 if (error < inlier_threshold) {
474 tested_inliers++;
475 lambda *= delta_to_epsilon;
476 } else {
477 lambda *= complement_delta_to_complement_epsilon;
478 if (lambda > current_A)
479 break;
480 }
481 errors[pt] = error;
482 }
483 }
484 last_model_is_good = tested_point == points_size;
485 }
486 if (last_model_is_good && do_sprt) {
487 out_score.inlier_number = tested_inliers;
488 if (score_type == ScoreMethod::SCORE_METHOD_MSAC)
489 out_score.score = static_cast<float>(sum_errors);
490 else if (score_type == ScoreMethod::SCORE_METHOD_RANSAC)
491 out_score.score = -static_cast<float>(tested_inliers);
492 else out_score = quality->getScore(errors);
493 }
494 return last_model_is_good;
495 }

497 // update SPRT parameters = called only once inside usac
498 void updateSPRT (double time_model_est, double time_corr_ver, double new_avg_models, double new_delta, double new_epsilon, const Score &best_score) override {
499 if (adapt) {
500 adapt = false;
501 m_S = new_avg_models;
502 t_M = time_model_est / time_corr_ver;
503 time_ver_corr = time_corr_ver;
504 time_ver_corr_sprt = time_corr_ver * 1.05;
505 createTest(new_epsilon, new_delta);
506 highest_inlier_number = best_score.inlier_number;
507 lowest_sum_errors = best_score.score;
508 }
509 }

511 const std::vector<SPRT_history> &getSPRTvector () const override { return adapt ? empty : sprt_histories; }
512 void update (const Score &score, int iteration) override {
513 if (adapt || highest_inlier_number > score.inlier_number)
514 return;
515
516 if (sprt_histories.size() == 1 && sprt_histories.back().tested_samples == 0)
517 sprt_histories.back().tested_samples = iteration;
518 else if (! sprt_histories.empty())
519 sprt_histories.back().tested_samples += iteration - last_iteration;
520
521 SPRT_history new_sprt_history;
522 new_sprt_history.epsilon = (double)score.inlier_number / points_size;
523 highest_inlier_number = score.inlier_number;
524 lowest_sum_errors = score.score;
525 createTest(static_cast<double>(highest_inlier_number) / points_size, current_delta);
526 new_sprt_history.delta = current_delta;
527 new_sprt_history.A = current_A;
528 sprt_histories.emplace_back(new_sprt_history);
529 last_iteration = iteration;
530 }
531 int avgNumCheckedPts () const override { return do_sprt ? (int)avg_num_checked_pts + 1 : points_size; }
532 void reset() override {
533 adapt = true;
534 do_sprt = false;
535 highest_inlier_number = last_iteration = 0;
536 lowest_sum_errors = DBL_MAX;
537 sprt_histories.clear();
538 }
539private:
540 // Update current epsilon, delta and threshold (A).
541 bool createTest (double epsilon, double delta) {
542 if (fabs(current_epsilon - epsilon) < FLT_EPSILON && fabs(current_delta - delta) < FLT_EPSILON)
543 return false;
544 // if epsilon is closed to 1 then set them to 0.99 to avoid numerical problems
545 if (epsilon > 0.999999) epsilon = 0.999;
546 // delta can't be higher than epsilon, because ratio delta / epsilon will be greater than 1
547 if (epsilon < delta) delta = epsilon-0.001;
548 // avoid delta going too high as it is very unlikely
549 // e.g., 30% of points are consistent with bad model is not very real
550 if (delta > 0.3) delta = 0.3;
551
552 const auto AC = estimateThresholdA (epsilon, delta);
553 current_A = AC.first;
554 const auto C = AC.second;
555 current_delta = delta;
556 current_epsilon = epsilon;
557 one_over_complement_alpha = 1 / (1 - 1 / current_A);
558
559 delta_to_epsilon = delta / epsilon;
560 complement_delta_to_complement_epsilon = (1 - delta) / (1 - epsilon);
561
562 if (IS_ADAPTIVE) {
563 avg_num_checked_pts = std::min((log(current_A) / C) * one_over_complement_alpha, (double)points_size);
564 do_sprt = time_ver_corr_sprt * avg_num_checked_pts < time_ver_corr * points_size;
565 }
566 return true;
567 }
568 std::pair<double,double> estimateThresholdA (double epsilon, double delta) {
569 const double C = (1 - delta) * log ((1 - delta) / (1 - epsilon)) + delta * log (delta / epsilon);
570 // K = K1/K2 + 1 = (t_M / P_g) / (m_S / (C * P_g)) + 1 = (t_M * C)/m_S + 1
571 const double K = t_M * C / m_S + 1;
572 double An, An_1 = K;
573 // compute A using a recursive relation
574 // A* = lim(n->inf)(An), the series typically converges within 4 iterations
575 for (int i = 0; i < 10; i++) {
576 An = K + log(An_1);
577 if (fabs(An - An_1) < FLT_EPSILON)
578 break;
579 An_1 = An;
580 }
581 return std::make_pair(An, C);
582 }
583};
