103float KeyPoint::overlap( const KeyPoint& kp1, const KeyPoint& kp2 )
104{
105 float a = kp1.size * 0.5f;
106 float b = kp2.size * 0.5f;
107 float a_2 = a * a;
108 float b_2 = b * b;
109
110 Point2f p1 = kp1.pt;
111 Point2f p2 = kp2.pt;
112 float c = (float)norm( p1 - p2 );
113
114 float ovrl = 0.f;
115
116 // one circle is completely encovered by the other => no intersection points!
117 if( std::min( a, b ) + c <= std::max( a, b ) )
118 return std::min( a_2, b_2 ) / std::max( a_2, b_2 );
119
120 if( c < a + b ) // circles intersect
121 {
122 float c_2 = c * c;
123 float cosAlpha = ( b_2 + c_2 - a_2 ) / ( kp2.size * c );
124 float cosBeta = ( a_2 + c_2 - b_2 ) / ( kp1.size * c );
125 float alpha = acos( cosAlpha );
126 float beta = acos( cosBeta );
127 float sinAlpha = sin(alpha);
128 float sinBeta = sin(beta);
129
130 float segmentAreaA = a_2 * beta;
131 float segmentAreaB = b_2 * alpha;
132
133 float triangleAreaA = a_2 * sinBeta * cosBeta;
134 float triangleAreaB = b_2 * sinAlpha * cosAlpha;
135
136 float intersectionArea = segmentAreaA + segmentAreaB - triangleAreaA - triangleAreaB;
137 float unionArea = (a_2 + b_2) * (float)CV_PI - intersectionArea;
138
139 ovrl = intersectionArea / unionArea;
140 }
141
142 return ovrl;
143}

147RotatedRect::RotatedRect(const Point2f& _point1, const Point2f& _point2, const Point2f& _point3)
148{
149 Point2f _center = 0.5f * (_point1 + _point3);
150 Vec2f vecs[2];
151 vecs[0] = Vec2f(_point1 - _point2);
152 vecs[1] = Vec2f(_point2 - _point3);
153 double x = std::max(norm(_point1), std::max(norm(_point2), norm(_point3)));
154 double a = std::min(norm(vecs[0]), norm(vecs[1]));
155 // check that given sides are perpendicular
156 CV_Assert( std::fabs(vecs[0].ddot(vecs[1])) * a <= FLT_EPSILON * 9 * x * (norm(vecs[0]) * norm(vecs[1])) );
157
158 // wd_i stores which vector (0,1) or (1,2) will make the width
159 // One of them will definitely have slope within -1 to 1
160 int wd_i = 0;
161 if( std::fabs(vecs[1][1]) < std::fabs(vecs[1][0]) ) wd_i = 1;
162 int ht_i = (wd_i + 1) % 2;
163
164 float _angle = std::atan(vecs[wd_i][1] / vecs[wd_i][0]) * 180.0f / (float) CV_PI;
165 float _width = (float) norm(vecs[wd_i]);
166 float _height = (float) norm(vecs[ht_i]);
167
168 center = _center;
169 size = Size2f(_width, _height);
170 angle = _angle;
171}

173void RotatedRect::points(Point2f pt[]) const
174{
175 double _angle = angle*CV_PI/180.;
176 float b = (float)cos(_angle)*0.5f;
177 float a = (float)sin(_angle)*0.5f;
178
179 const float ah = a*size.height;
180 const float aw = a*size.width;
181 const float bh = b*size.height;
182 const float bw = b*size.width;
183 pt[0].x = center.x - ah - bw;
184 pt[0].y = center.y + bh - aw;
185 pt[1].x = center.x + ah - bw;
186 pt[1].y = center.y - bh - aw;
187 pt[2].x = center.x + ah + bw;
188 pt[2].y = center.y - bh + aw;
189 pt[3].x = center.x - ah + bw;
190 pt[3].y = center.y + bh + aw;
191}

198Rect RotatedRect::boundingRect() const
199{
200 Point2f pt[4];
201 points(pt);
202 Rect r(cvFloor(std::min(std::min(std::min(pt[0].x, pt[1].x), pt[2].x), pt[3].x)),
203 cvFloor(std::min(std::min(std::min(pt[0].y, pt[1].y), pt[2].y), pt[3].y)),
204 cvCeil(std::max(std::max(std::max(pt[0].x, pt[1].x), pt[2].x), pt[3].x)),
205 cvCeil(std::max(std::max(std::max(pt[0].y, pt[1].y), pt[2].y), pt[3].y)));
206 r.width -= r.x - 1;
207 r.height -= r.y - 1;
208 return r;
209}
