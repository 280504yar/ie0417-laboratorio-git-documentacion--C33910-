60Ptr<BackendNode> Layer::initCUDA(
61 void* context,
62 const std::vector<Ptr<BackendWrapper>>& inputs,
63 const std::vector<Ptr<BackendWrapper>>& outputs)
64{
65#ifdef HAVE_CUDA
66 // Adapt the classic wrapper-based entry point to the array-based one, so ops ported to the
67 // new graph engine only need to override initCUDA(context, inputs, outputs) with GpuMatND.
68 std::vector<cuda::GpuMatND> inGpu(inputs.size()), outGpu(outputs.size());
69 for (size_t i = 0; i < inputs.size(); i++)
70 inGpu[i] = inputs[i].dynamicCast<CUDABackendWrapper>()->getDeviceMatND();
71 for (size_t i = 0; i < outputs.size(); i++)
72 outGpu[i] = outputs[i].dynamicCast<CUDABackendWrapper>()->getDeviceMatND();
73 return initCUDA(context, inGpu, outGpu);
74#else
75 CV_UNUSED(context); CV_UNUSED(inputs); CV_UNUSED(outputs);
76 CV_Error(Error::StsNotImplemented, "CUDA pipeline of " + type + " layers is not defined.");
77 return Ptr<BackendNode>();
78#endif
79}

210void Layer::forward_fallback(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr)
211{
212 CV_TRACE_FUNCTION();
213 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
214
215 if (preferableTarget == DNN_TARGET_OPENCL_FP16 && inputs_arr.depth() == CV_16F)
216 {
217 std::vector<UMat> inputs;
218 std::vector<UMat> outputs;
219 std::vector<UMat> internals;
220
221 std::vector<UMat> orig_inputs;
222 std::vector<UMat> orig_outputs;
223 std::vector<UMat> orig_internals;
224
225 inputs_arr.getUMatVector(orig_inputs);
226 outputs_arr.getUMatVector(orig_outputs);
227 internals_arr.getUMatVector(orig_internals);
228
229 inputs.resize(orig_inputs.size());
230 for (size_t i = 0; i < orig_inputs.size(); i++)
231 if (orig_inputs[i].depth() == CV_16F)
232 orig_inputs[i].convertTo(inputs[i], CV_32F);
233 else
234 inputs[i] = orig_inputs[i];
235
236 outputs.resize(orig_outputs.size());
237 for (size_t i = 0; i < orig_outputs.size(); i++)
238 if (orig_outputs[i].depth() == CV_16F)
239 outputs[i].create(shape(orig_outputs[i]), CV_32F);
240 else
241 outputs[i] = orig_outputs[i];
242
243 internals.resize(orig_internals.size());
244 for (size_t i = 0; i < orig_internals.size(); i++)
245 if (orig_internals[i].depth() == CV_16F)
246 internals[i].create(shape(orig_internals[i]), CV_32F);
247 else
248 internals[i] = orig_internals[i];
249
250 forward(inputs, outputs, internals);
251
252 for (size_t i = 0; i < outputs.size(); i++)
253 if (orig_outputs[i].depth() == CV_16F)
254 outputs[i].convertTo(orig_outputs[i], CV_16F);
255 else
256 outputs[i] = orig_outputs[i];
257
258 // sync results back
259 outputs_arr.assign(orig_outputs);
260 internals_arr.assign(orig_internals);
261 return;
262 }
263 std::vector<Mat> inpvec;
264 std::vector<Mat> outputs;
265 std::vector<Mat> internals;
266
267 inputs_arr.getMatVector(inpvec);
268 outputs_arr.getMatVector(outputs);
269 internals_arr.getMatVector(internals);
270
271 std::vector<Mat*> inputs(inpvec.size());
272 for (int i = 0; i < inpvec.size(); i++)
273 inputs[i] = &inpvec[i];
274
275 this->forward(inputs, outputs, internals);
276
277 // sync results back
278 outputs_arr.assign(outputs);
279 internals_arr.assign(internals);
280}

362std::ostream& LayerInfo::dump(std::ostream& strm, int indent, bool comma) const
363{
364 CV_Assert(netimpl);
365 size_t ninputs = inputs.size();
366 size_t noutputs = outputs.size();
367 size_t nblobs = blobs.size();
368 const std::vector<Ptr<Graph> >* subgraphs_ = subgraphs();
369 size_t nsubgraphs = subgraphs_ ? subgraphs_->size() : 0;
370 Net::Impl* netimpl = getNetImpl(this);
371 int delta_indent = netimpl->dump_indent;
372 int subindent = indent + delta_indent;
373 int argindent = subindent + delta_indent;
374 prindent(strm, indent);
375 std::string opname = type;
376 strm << opname << " {\n";
377 prindent(strm, subindent);
378 strm << "name: \"" << name << "\",\n";
379
380 if (!blobs.empty()) {
381 prindent(strm, subindent);
382 strm << "blobs: [\n";
383 for (size_t i = 0; i < nblobs; i++) {
384 if (i > 0)
385 strm << ",\n";
386 const Mat& blob = blobs[i];
387 prindent(strm, argindent);
388 netimpl->dumpTypeShape(strm, blob.type(), blob.shape());
389 }
390 strm << "\n";
391 prindent(strm, subindent);
392 strm << "],\n";
393 }
394 dumpAttrs(strm, subindent);
395 prindent(strm, subindent);
396 strm << "inputs: [\n";
397 for (size_t i = 0; i < ninputs; i++) {
398 netimpl->dumpArg(strm, inputs[i], argindent, i+1 < ninputs, true);
399 }
400 prindent(strm, subindent);
401 strm << "],\n";
402 prindent(strm, subindent);
403 strm << "outputs: [\n";
404 for (size_t i = 0; i < noutputs; i++) {
405 netimpl->dumpArg(strm, outputs[i], argindent, i+1 < noutputs, true);
406 }
407 prindent(strm, subindent);
408 strm << "],\n";
409
410 if (nsubgraphs > 0) {
411 std::vector<std::string> names;
412 if (opname == "If")
413 names = {"then", "else"};
414 else if (opname == "Loop" || opname == "Scan")
415 names = {"body"};
416 else {
417 CV_Error(Error::StsError,
418 format("unsupported operation '%s' with subgraphs",
419 std::string(opname).c_str()));
420 }
421 CV_Assert(names.size() == nsubgraphs);
422 for (size_t i = 0; i < nsubgraphs; i++) {
423 prindent(strm, subindent);
424 strm << names[i] << ": ";
425 subgraphs_->at(i)->dump(strm, argindent, i+1 < nsubgraphs);
426 }
427 }
428 prindent(strm, indent);
429 strm << '}';
430 if (comma)
431 strm << ',';
432 strm << '\n';
433 return strm;
434}
