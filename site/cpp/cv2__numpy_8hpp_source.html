106template<>
107bool canBeSafelyCasted<size_t>(PyObject* obj, PyArray_Descr* to)
108{
109 PyArray_Descr* from = PyArray_DescrFromScalar(obj);
110 if (PyArray_CanCastTo(from, to))
111 {
112 return true;
113 }
114 else
115 {
116 // False negative scenarios:
117 // - Signed input is positive so it can be safely cast to unsigned output
118 // - Input has wider limits but value is representable within output limits
119 // - All the above
120 if (PyDataType_ISSIGNED(from))
121 {
122 int64_t input = 0;
123 PyArray_CastScalarToCtype(obj, &input, getNumpyTypeDescriptor<int64_t>());
124 return (input >= 0) && isRepresentable<size_t>(static_cast<uint64_t>(input));
125 }
126 else
127 {
128 uint64_t input = 0;
129 PyArray_CastScalarToCtype(obj, &input, getNumpyTypeDescriptor<uint64_t>());
130 return isRepresentable<size_t>(input);
131 }
132 return false;
133 }
134}

137template<class T>
138bool parseNumpyScalar(PyObject* obj, T& value)
139{
140 if (PyArray_CheckScalar(obj))
141 {
142 // According to the numpy documentation:
143 // There are 21 statically-defined PyArray_Descr objects for the built-in data-types
144 // So descriptor pointer is not owning.
145 PyArray_Descr* to = getNumpyTypeDescriptor<T>();
146 if (canBeSafelyCasted<T>(obj, to))
147 {
148 PyArray_CastScalarToCtype(obj, &value, to);
149 return true;
150 }
151 }
152 return false;
153}

181template <class T, std::size_t N>
182bool parseSequence(PyObject* obj, RefWrapper<T> (&value)[N], const ArgInfo& info)
183{
184 if (!obj || obj == Py_None)
185 {
186 return true;
187 }
188 if (!PySequence_Check(obj))
189 {
190 failmsg("Can't parse '%s'. Input argument doesn't provide sequence "
191 "protocol", info.name);
192 return false;
193 }
194 const std::size_t sequenceSize = PySequence_Size(obj);
195 if (sequenceSize != N)
196 {
197 failmsg("Can't parse '%s'. Expected sequence length %lu, got %lu",
198 info.name, N, sequenceSize);
199 return false;
200 }
201 for (std::size_t i = 0; i < N; ++i)
202 {
203 SafeSeqItem seqItem(obj, i);
204 if (!pyopencv_to(seqItem.item, value[i].get(), info))
205 {
206 failmsg("Can't parse '%s'. Sequence item with index %lu has a "
207 "wrong type", info.name, i);
208 return false;
209 }
210 }
211 return true;
212}
