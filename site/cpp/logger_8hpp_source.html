109#define CV_LOGTAG_PTR_CAST(expr) static_cast<const cv::utils::logging::LogTag*>(expr)
110
111// CV_LOGTAG_EXPAND_NAME is intended to be re-defined (undef and then define again)
112// to allows logging users to use a shorter name argument when calling
113// CV_LOG_WITH_TAG or its related macros such as CV_LOG_INFO.
114//
115// This macro is intended to modify the tag argument as a string (token), via
116// preprocessor token pasting or metaprogramming techniques. A typical usage
117// is to apply a prefix, such as
118// ...... #define CV_LOGTAG_EXPAND_NAME(tag) cv_logtag_##tag
119//
120// It is permitted to re-define to a hard-coded expression, ignoring the tag.
121// This would work identically like the CV_LOGTAG_FALLBACK macro.
122//
123// Important: When the logging macro is called with tag being NULL, a user-defined
124// CV_LOGTAG_EXPAND_NAME may expand it into cv_logtag_0, cv_logtag_NULL, or
125// cv_logtag_nullptr. Use with care. Also be mindful of C++ symbol redefinitions.
126//
127// If there is significant amount of logging code with tag being NULL, it is
128// recommended to use (re-define) CV_LOGTAG_FALLBACK to inject locally a default
129// tag at the beginning of a compilation unit, to minimize lines of code changes.
130//
131#define CV_LOGTAG_EXPAND_NAME(tag) tag
132
133// CV_LOGTAG_FALLBACK is intended to be re-defined (undef and then define again)
134// by any other compilation units to provide a log tag when the logging statement
135// does not specify one. The macro needs to expand into a C++ expression that can
136// be static_cast into (cv::utils::logging::LogTag*). Null (nullptr) is permitted.
137#define CV_LOGTAG_FALLBACK nullptr
138
139// CV_LOGTAG_GLOBAL is the tag used when a log tag is not specified in the logging
140// statement nor the compilation unit. The macro needs to expand into a C++
141// expression that can be static_cast into (cv::utils::logging::LogTag*). Must be
142// non-null. Do not re-define.
143#define CV_LOGTAG_GLOBAL cv::utils::logging::internal::getGlobalLogTag()

145#define CV_LOG_WITH_TAG(tag, msgLevel, extra_check0, extra_check1, ...) \
146 for(;;) { \
147 extra_check0; \
148 const auto cv_temp_msglevel = (cv::utils::logging::LogLevel)(msgLevel); \
149 if (cv_temp_msglevel >= (CV_LOG_STRIP_LEVEL)) break; \
150 auto cv_temp_logtagptr = CV_LOGTAG_PTR_CAST(CV_LOGTAG_EXPAND_NAME(tag)); \
151 if (!cv_temp_logtagptr) cv_temp_logtagptr = CV_LOGTAG_PTR_CAST(CV_LOGTAG_FALLBACK); \
152 if (!cv_temp_logtagptr) cv_temp_logtagptr = CV_LOGTAG_PTR_CAST(CV_LOGTAG_GLOBAL); \
153 if (cv_temp_logtagptr && (cv_temp_msglevel > cv_temp_logtagptr->level)) break; \
154 extra_check1; \
155 std::stringstream cv_temp_logstream; \
156 cv_temp_logstream << __VA_ARGS__; \
157 cv::utils::logging::internal::writeLogMessageEx( \
158 cv_temp_msglevel, \
159 (cv_temp_logtagptr ? cv_temp_logtagptr->name : nullptr), \
160 __FILE__, \
161 __LINE__, \
162 CV_Func, \
163 cv_temp_logstream.str().c_str()); \
164 break; \
165 }
