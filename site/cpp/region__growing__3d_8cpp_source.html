19int
20RegionGrowing3DImpl::segment(OutputArrayOfArrays regions_idx_, OutputArray labels_,
21 InputArray input_pts_, InputArray normals_, InputArrayOfArrays nn_idx_)
22{
23 Mat ori_pts, normals;
24 getPointsMatFromInputArray(input_pts_, ori_pts, 0);
25 getPointsMatFromInputArray(normals_, normals, 0);
26
27 std::vector<Mat> nn_idx;
28 nn_idx_.getMatVector(nn_idx);
29
30 int pts_size = ori_pts.rows;
31 bool has_curvatures = !curvatures.empty();
32 bool has_seeds = !seeds.empty();
33
34 CV_CheckGE(max_size, min_size,
35 "The maximum size should be grater than or equal to the minimum size.");
36 CV_CheckEQ(pts_size, (int) nn_idx.size(),
37 "The point number of NN search result should be equal to the size of the point cloud.");
38 CV_CheckEQ(pts_size, normals.rows,
39 "The number of points in the normals should be equal to that in the point cloud.");
40 if (has_curvatures)
41 {
42 CV_CheckEQ(pts_size, curvatures.cols,
43 "The number of points in the curvatures should be equal to that in the point cloud.");
44 }
45 if (has_seeds)
46 {
47 CV_CheckGE(pts_size, seeds.cols,
48 "The number of seed should be less than or equal to the number of points in the point cloud.");
49 }
50
51 if (!has_seeds && !has_curvatures)
52 {
53 // If the user does not set the seeds and curvatures, use the natural order of the points
54 std::vector<int> *natural_order_seeds = new std::vector<int>(pts_size);
55 std::iota(natural_order_seeds->begin(), natural_order_seeds->end(), 0);
56 seeds = Mat(*natural_order_seeds);
57 }
58 else if (!has_seeds && has_curvatures)
59 {
60 // If the user sets the curvatures without setting the seeds, seeds will be sorted in
61 // ascending order of curvatures
62 sortIdx(curvatures, seeds, SORT_EVERY_ROW + SORT_ASCENDING);
63 }
64
65 const int *seeds_ptr = (int *) seeds.data;
66 const float *curvatures_ptr = (float *) curvatures.data;
67
68 int seeds_num = seeds.cols, flag = 1;
69 double cos_smoothness_thr = std::cos(smoothness_thr);
70 double cos_smoothness_thr_square = cos_smoothness_thr * cos_smoothness_thr;
71 // Used to determine if the point can be grown
72 std::vector<bool> has_grown(pts_size, false);
73 // Store the indexes of the points in each region
74 std::vector<std::vector<int>> regions_idx;
75 for (int i = 0; i < seeds_num; ++i)
76 {
77 int cur_seed = seeds_ptr[i];
78
79 // 1. If the number of regions is satisfied then stop running.
80 // 2. Filter out seeds with curvature greater than the threshold.
81 // If no seed points have been set, it will use seeds sorted in ascending order of curvatures.
82 // When the curvature doesn't satisfy the threshold, the seed points that follow will not satisfy the threshold either.
83 if (flag > region_num ||
84 (!has_seeds && has_curvatures && curvatures_ptr[cur_seed] > curvature_thr))
85 {
86 break;
87 }
88 // 1. If current seed has been grown then grow the next one.
89 // 2. Filter out seeds with curvature greater than the threshold.
90 if (has_grown[cur_seed] ||
91 (has_seeds && has_curvatures && curvatures_ptr[cur_seed] > curvature_thr))
92 {
93 continue;
94 }
95
96 Mat base_normal;
97 if (!smooth_mode) base_normal = normals.row(cur_seed);
98
99 has_grown[cur_seed] = true;
100 std::vector<int> region = {cur_seed};
101 std::queue<int> grow_list;
102 grow_list.push(cur_seed);
103 while (!grow_list.empty())
104 {
105 int cur_idx = grow_list.front();
106 grow_list.pop();
107 if (smooth_mode) base_normal = normals.row(cur_idx);
108
109 // The maximum size that may be used for this row
110 int bound = max_neighbor_num > nn_idx[cur_idx].cols ? nn_idx[cur_idx].cols
111 : max_neighbor_num;
112 const int *nn_idx_ptr_base = (int *) nn_idx[cur_idx].data;
113 // Start from index 1 because the first one of knn_idx is itself
114 for (int j = 1; j < bound; ++j)
115 {
116 int cur_nei_idx = nn_idx_ptr_base[j];
117 // If the index is less than 0,
118 // the nn_idx of this row will no longer have any information.
119 if (cur_nei_idx < 0) break;
120 // 1. If current point has been grown then grow the next one
121 // 2. Filter out points with curvature greater than the threshold
122 if (has_grown[cur_nei_idx] ||
123 (has_curvatures && curvatures_ptr[cur_nei_idx] > curvature_thr))
124 {
125 continue;
126 }
127
128 Mat cur_normal = normals.row(cur_nei_idx);
129 double n12 = base_normal.dot(cur_normal);
130 double n1m = base_normal.dot(base_normal);
131 double n2m = cur_normal.dot(cur_normal);
132 // If the smoothness threshold is satisfied, this neighbor will be pushed to the growth list
133 if (n12 * n12 / (n1m * n2m) >= cos_smoothness_thr_square)
134 {
135 has_grown[cur_nei_idx] = true;
136 region.emplace_back(cur_nei_idx);
137 grow_list.push(cur_nei_idx);
138 }
139 }
140 }
141
142 int region_size = (int) region.size();
143 if (min_size <= region_size && region_size <= max_size)
144 {
145 regions_idx.emplace_back(Mat(region));
146 flag++;
147 }
148 else if (region_size > max_size)
149 {
150 break;
151 }
152 }
153
154 if (need_sort)
155 {
156 // Compare the size of two regions in descending order
157 auto compareRegionSize = [](const std::vector<int> &r1,
158 const std::vector<int> &r2) -> bool
159 {
160 return r1.size() > r2.size();
161 };
162 sort(regions_idx.begin(), regions_idx.end(), compareRegionSize);
163 }
164
165 std::vector<std::vector<int>> &_regions_idx = *(std::vector<std::vector<int>> *) regions_idx_.getObj();
166 _regions_idx.resize(regions_idx.size());
167 Mat labels = Mat::zeros(pts_size, 1, CV_32S);
168 int *labels_ptr = (int *) labels.data;
169 for (int i = 0; i < (int) regions_idx.size(); ++i)
170 {
171 Mat(1, (int) regions_idx[i].size(), CV_32S, regions_idx[i].data()).copyTo(_regions_idx[i]);
172 for (int j: regions_idx[i])
173 {
174 labels_ptr[j] = i + 1;
175 }
176 }
177 labels.copyTo(labels_);
178
179 return flag - 1;
180}
