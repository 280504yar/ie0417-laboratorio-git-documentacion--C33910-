31 inline uint32_t get_representative(uint32_t id) {
32 uint32_t root = id;
33
34 // chase down the root
35 while (data[root].parent != root) {
36 root = data[root].parent;
37 }
38
39 // go back and collapse the tree.
40 //
41 // XXX: on some of our workloads that have very shallow trees
42 // (e.g. image segmentation), we are actually faster not doing
43 // this...
44 while (data[id].parent != root) {
45 uint32_t tmp = data[id].parent;
46 data[id].parent = root;
47 id = tmp;
48 }
49
50 return root;
51 }

53 inline uint32_t get_set_size(uint32_t id) {
54 uint32_t repid = get_representative(id);
55 return data[repid].size;
56 }

58 inline uint32_t connect(uint32_t aid, uint32_t bid) {
59 uint32_t aroot = get_representative(aid);
60 uint32_t broot = get_representative(bid);
61
62 if (aroot == broot)
63 return aroot;
64
65 // we don't perform "union by rank", but we perform a similar
66 // operation (but probably without the same asymptotic guarantee):
67 // We join trees based on the number of *elements* (as opposed to
68 // rank) contained within each tree. I.e., we use size as a proxy
69 // for rank. In my testing, it's often *faster* to use size than
70 // rank, perhaps because the rank of the tree isn't that critical
71 // if there are very few nodes in it.
72 uint32_t asize = data[aroot].size;
73 uint32_t bsize = data[broot].size;
74
75 // optimization idea: We could shortcut some or all of the tree
76 // that is grafted onto the other tree. Pro: those nodes were just
77 // read and so are probably in cache. Con: it might end up being
78 // wasted effort -- the tree might be grafted onto another tree in
79 // a moment!
80 if (asize > bsize) {
81 data[broot].parent = aroot;
82 data[aroot].size += bsize;
83 return aroot;
84 } else {
85 data[aroot].parent = broot;
86 data[broot].size += asize;
87 return broot;
88 }
89 }

91 #define DO_UNIONFIND(dx, dy) if (im.data[y*s + dy*s + x + dx] == v) connect(y*w + x, y*w + dy*w + x + dx);
92 void do_line(Mat &im, int w, int s, int y) {
93 CV_Assert(y+1 < im.rows);
94 CV_Assert(!im.empty());
95
96 for (int x = 1; x < w - 1; x++) {
97 uint8_t v = im.data[y*s + x];
98
99 if (v == 127)
100 continue;
101
102 // (dx,dy) pairs for 8 connectivity:
103 // (REFERENCE) (1, 0)
104 // (-1, 1) (0, 1) (1, 1)
105 //
106 DO_UNIONFIND(1, 0);
107 DO_UNIONFIND(0, 1);
108 if (v == 255) {
109 DO_UNIONFIND(-1, 1);
110 DO_UNIONFIND(1, 1);
111 }
112 }
113 }
