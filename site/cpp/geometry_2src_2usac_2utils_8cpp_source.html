11/*
12SolvePoly is used to find only real roots of N-degree polynomial using Sturm sequence.
13It recursively finds interval where a root lies, and the actual root is found using Regula-Falsi method.
14*/
15class SolvePoly : public SolverPoly {
16private:
17 static int sgn(double val) {
18 return (double(0) < val) - (val < double(0));
19 }
20 class Poly {
21 public:
22 Poly () = default;
23 Poly (const std::vector<double> &coef_) {
24 coef = coef_;
25 checkDegree();
26 }
27 Poly (const Poly &p) { coef = p.coef; }
28 // a_n x^n + a_n-1 x^(n-1) + ... + a_1 x + a_0
29 // coef[i] = a_i
30 std::vector<double> coef = {0};
31 inline int degree() const { return (int)coef.size()-1; }
32 void multiplyScalar (double s) {
33 // multiplies polynom by scalar
34 if (fabs(s) < DBL_EPSILON) { // check if scalar is 0
35 coef = {0};
36 return;
37 }
38 for (double &c : coef) c *= s;
39 }
40 void checkDegree() {
41 int deg = degree(); // approximate degree
42 // check if coefficients of the highest power is non-zero
43 while (fabs(coef[deg]) < DBL_EPSILON) {
44 coef.pop_back(); // remove last zero element
45 if (--deg == 0)
46 break;
47 }
48 }
49 double eval (double x) const {
50 // Horner method a0 + x (a1 + x (a2 + x (a3 + ... + x (an-1 + x an))))
51 const int d = degree();
52 double y = coef[d];
53 for (int i = d; i >= 1; i--)
54 y = coef[i-1] + x * y;
55 return y;
56 }
57 // at +inf and -inf
58 std::pair<int,int> signsAtInf () const {
59 // lim x->+-inf p(x) = lim x->+-inf a_n x^n
60 const int d = degree();
61 const int s = sgn(coef[d]); // sign of the highest coefficient
62 // compare even and odd degree
63 return std::make_pair(s, d % 2 == 0 ? s : -s);
64 }
65 Poly derivative () const {
66 Poly deriv;
67 if (degree() == 0)
68 return deriv;
69 // derive.degree = poly.degree-1;
70 deriv.coef = std::vector<double>(coef.size()-1);
71 for (int i = degree(); i > 0; i--)
72 // (a_n * x^n)' = n * a_n * x^(n-1)
73 deriv.coef[i-1] = i * coef[i];
74 return deriv;
75 }
76 void copyFrom (const Poly &p) { coef = p.coef; }
77 };
78 // return remainder
79 static void dividePoly (const Poly &p1, const Poly &p2, /*Poly &quotient,*/ Poly &remainder) {
80 remainder.copyFrom(p1);
81 int p2_degree = p2.degree(), remainder_degree = remainder.degree();
82 if (p1.degree() < p2_degree)
83 return;
84 if (p2_degree == 0) { // special case for dividing polynomial by constant
85 remainder.multiplyScalar(1/p2.coef[0]);
86 // quotient.coef[0] = p2.coef[0];
87 return;
88 }
89 // quotient.coef = std::vector<double>(p1.degree() - p2_degree + 1, 0);
90 const double p2_term = 1/p2.coef[p2_degree];
91 while (remainder_degree >= p2_degree) {
92 const double temp = remainder.coef[remainder_degree] * p2_term;
93 // quotient.coef[remainder_degree-p2_degree] = temp;
94 // polynoms now have the same degree, but p2 is shorter than remainder
95 for (int i = p2_degree, j = remainder_degree; i >= 0; i--, j--)
96 remainder.coef[j] -= temp * p2.coef[i];
97 remainder.checkDegree();
98 remainder_degree = remainder.degree();
99 }
100 }

102 constexpr static int REGULA_FALSI_MAX_ITERS = 500, MAX_POWER = 10, MAX_LEVEL = 200;
103 constexpr static double TOLERANCE = 1e-10, DIFF_TOLERANCE = 1e-7;
104
105 static bool findRootRegulaFalsi (const Poly &poly, double min, double max, double &root) {
106 double f_min = poly.eval(min), f_max = poly.eval(max);
107 if (f_min * f_max > 0 || min > max) {// conditions are not fulfilled
108 return false;
109 }
110 int sign = 0, iter = 0;
111 for (; iter < REGULA_FALSI_MAX_ITERS; iter++) {
112 root = (f_min * max - f_max * min) / (f_min - f_max);
113 const double f_root = poly.eval(root);
114 if (fabs(f_root) < TOLERANCE || fabs(min - max) < DIFF_TOLERANCE) {
115 return true; // root is found
116 }
117
118 if (f_root * f_max > 0) {
119 max = root; f_max = f_root;
120 if (sign == -1)
121 f_min *= 0.5;
122 sign = -1;
123 } else if (f_min * f_root > 0) {
124 min = root; f_min = f_root;
125 if (sign == 1)
126 f_max *= 0.5;
127 sign = 1;
128 }
129 }
130 return false;
131 }

133 static int numberOfSignChanges (const std::vector<Poly> &sturm, double x) {
134 int prev_sign = 0, sign_changes = 0;
135 for (const auto &poly : sturm) {
136 const int s = sgn(poly.eval(x));
137 if (s != 0 && prev_sign != 0 && s != prev_sign)
138 sign_changes++;
139 prev_sign = s;
140 }
141 return sign_changes;
142 }

144 static void findRootsRecursive (const Poly &poly, const std::vector<Poly> &sturm, double min, double max,
145 int sign_changes_at_min, int sign_changes_at_max, std::vector<double> &roots, int level) {
146 const int num_roots = sign_changes_at_min - sign_changes_at_max;
147 if (level == MAX_LEVEL) {
148 // roots are too close
149 const double mid = (min + max) * 0.5;
150 if (fabs(poly.eval(mid)) < DBL_EPSILON) {
151 roots.emplace_back(mid);
152 }
153 } else if (num_roots == 1) {
154 double root;
155 if (findRootRegulaFalsi(poly, min, max, root)) {
156 roots.emplace_back(root);
157 }
158 } else if (num_roots > 1) { // at least 2 roots
159 const double mid = (min + max) * 0.5;
160 const int sign_changes_at_mid = numberOfSignChanges(sturm, mid);
161 // try to split interval equally for the roots
162 if (sign_changes_at_min - sign_changes_at_mid > 0)
163 findRootsRecursive(poly, sturm, min, mid, sign_changes_at_min, sign_changes_at_mid, roots, level+1);
164 if (sign_changes_at_mid - sign_changes_at_max > 0)
165 findRootsRecursive(poly, sturm, mid, max, sign_changes_at_mid, sign_changes_at_max, roots, level+1);
166 }
167 }
168public:
169 int getRealRoots (const std::vector<double> &coeffs, std::vector<double> &real_roots) override {
170 if (coeffs.empty())
171 return 0;
172 for (auto c : coeffs)
173 if (cvIsNaN(c) || cvIsInf(c))
174 return 0;
175 Poly input(coeffs);
176 if (input.degree() < 1)
177 return 0;
178 // derivative of input polynomial
179 const Poly input_der = input.derivative();
181 Poly p (input), q (input_der), remainder;
182 std::vector<std::pair<int,int>> signs_at_inf; signs_at_inf.reserve(p.degree()); // +inf, -inf pair
183 signs_at_inf.emplace_back(p.signsAtInf());
184 signs_at_inf.emplace_back(q.signsAtInf());
185 std::vector<Poly> sturm_sequence; sturm_sequence.reserve(input.degree());
186 sturm_sequence.emplace_back(input);
187 sturm_sequence.emplace_back(input_der);
188 while (q.degree() > 0) {
189 dividePoly(p, q, remainder);
190 remainder.multiplyScalar(-1);
191 p.copyFrom(q);
192 q.copyFrom(remainder);
193 sturm_sequence.emplace_back(remainder);
194 signs_at_inf.emplace_back(remainder.signsAtInf());
195 }
197 int num_sign_changes_at_pos_inf = 0, num_sign_changes_at_neg_inf = 0;
198 int prev_sign_pos_inf = signs_at_inf[0].first, prev_sign_neg_inf = signs_at_inf[0].second;
199 for (int i = 1; i < (int)signs_at_inf.size(); i++) {
200 const auto s_pos_inf = signs_at_inf[i].first, s_neg_inf = signs_at_inf[i].second;
201 // zeros must be ignored
202 if (s_pos_inf != 0) {
203 if (prev_sign_pos_inf != 0 && prev_sign_pos_inf != s_pos_inf)
204 num_sign_changes_at_pos_inf++;
205 prev_sign_pos_inf = s_pos_inf;
206 }
207 if (s_neg_inf != 0) {
208 if (prev_sign_neg_inf != 0 && prev_sign_neg_inf != s_neg_inf)
209 num_sign_changes_at_neg_inf++;
210 prev_sign_neg_inf = s_neg_inf;
211 }
212 }
214 double root_neg_bound = -0.01, root_pos_bound = 0.01;
215 int num_sign_changes_min_x = -1, num_sign_changes_pos_x = -1; // -1 = unknown, trigger next if condition
216 for (int i = 0; i < MAX_POWER; i++) {
217 if (num_sign_changes_min_x != num_sign_changes_at_neg_inf) {
218 root_neg_bound *= 10;
219 num_sign_changes_min_x = numberOfSignChanges(sturm_sequence, root_neg_bound);
220 }
221 if (num_sign_changes_pos_x != num_sign_changes_at_pos_inf) {
222 root_pos_bound *= 10;
223 num_sign_changes_pos_x = numberOfSignChanges(sturm_sequence, root_pos_bound);
224 }
225 }
227 real_roots.clear();
228 findRootsRecursive(input, sturm_sequence, root_neg_bound, root_pos_bound, num_sign_changes_min_x, num_sign_changes_pos_x, real_roots, 0 /*level*/);
230 if ((int)real_roots.size() > input.degree())
231 real_roots.resize(input.degree()); // must not happen, unless some roots repeat
232 return (int) real_roots.size();
233 }
234};

236 return makePtr<SolvePoly>();
237}

245 * K1, K2 are 3x3 intrinsics matrices
246 * points is matrix of size |N| x 4
247 * Assume K = [k11 k12 k13
248 * 0 k22 k23

251void Utils::calibratePoints (const Mat &K1, const Mat &K2, const Mat &points, Mat &calib_points) {
252 const auto * const points_ = (float *) points.data;
253 const auto * const k1 = (double *) K1.data;
254 const auto inv1_k11 = float(1 / k1[0]); // 1 / k11
255 const auto inv1_k12 = float(-k1[1] / (k1[0]*k1[4])); // -k12 / (k11*k22)
256 // (-k13*k22 + k12*k23) / (k11*k22)
257 const auto inv1_k13 = float((-k1[2]*k1[4] + k1[1]*k1[5]) / (k1[0]*k1[4]));
258 const auto inv1_k22 = float(1 / k1[4]); // 1 / k22
259 const auto inv1_k23 = float(-k1[5] / k1[4]); // -k23 / k22
260
261 const auto * const k2 = (double *) K2.data;
262 const auto inv2_k11 = float(1 / k2[0]);
263 const auto inv2_k12 = float(-k2[1] / (k2[0]*k2[4]));
264 const auto inv2_k13 = float((-k2[2]*k2[4] + k2[1]*k2[5]) / (k2[0]*k2[4]));
265 const auto inv2_k22 = float(1 / k2[4]);
266 const auto inv2_k23 = float(-k2[5] / k2[4]);
267
268 calib_points = Mat ( points.rows, 4, points.type());
269 auto * calib_points_ = (float *) calib_points.data;
270
271 for (int i = 0; i < points.rows; i++) {
272 const int idx = 4*i;
273 (*calib_points_++) = inv1_k11 * points_[idx ] + inv1_k12 * points_[idx+1] + inv1_k13;
274 (*calib_points_++) = inv1_k22 * points_[idx+1] + inv1_k23;
275 (*calib_points_++) = inv2_k11 * points_[idx+2] + inv2_k12 * points_[idx+3] + inv2_k13;
276 (*calib_points_++) = inv2_k22 * points_[idx+3] + inv2_k23;
277 }
278}

280/*
281 * K is 3x3 intrinsic matrix
282 * points is matrix of size |N| x 5, first two columns are image points [u_i, v_i]
283 * calib_norm_pts are K^-1 [u v 1]^T / ||K^-1 [u v 1]^T||
284 */
285void Utils::calibrateAndNormalizePointsPnP (const Mat &K, const Mat &pts, Mat &calib_norm_pts) {
286 const auto * const points = (float *) pts.data;
287 const auto * const k = (double *) K.data;
288 const auto inv_k11 = float(1 / k[0]);
289 const auto inv_k12 = float(-k[1] / (k[0]*k[4]));
290 const auto inv_k13 = float((-k[2]*k[4] + k[1]*k[5]) / (k[0]*k[4]));
291 const auto inv_k22 = float(1 / k[4]);
292 const auto inv_k23 = float(-k[5] / k[4]);
293
294 calib_norm_pts = Mat (pts.rows, 3, pts.type());
295 auto * calib_norm_pts_ = (float *) calib_norm_pts.data;
296
297 for (int i = 0; i < pts.rows; i++) {
298 const int idx = 5 * i;
299 const float k_inv_u = inv_k11 * points[idx] + inv_k12 * points[idx+1] + inv_k13;
300 const float k_inv_v = inv_k22 * points[idx+1] + inv_k23;
301 const float norm = 1.f / sqrtf(k_inv_u*k_inv_u + k_inv_v*k_inv_v + 1);
302 (*calib_norm_pts_++) = k_inv_u * norm;
303 (*calib_norm_pts_++) = k_inv_v * norm;
304 (*calib_norm_pts_++) = norm;
305 }
306}

308void Utils::normalizeAndDecalibPointsPnP (const Mat &K_, Mat &pts, Mat &calib_norm_pts) {
309 const auto * const K = (double *) K_.data;
310 const auto k11 = (float)K[0], k12 = (float)K[1], k13 = (float)K[2],
311 k22 = (float)K[4], k23 = (float)K[5];
312 calib_norm_pts = Mat (pts.rows, 3, pts.type());
313 auto * points = (float *) pts.data;
314 auto * calib_norm_pts_ = (float *) calib_norm_pts.data;
315
316 for (int i = 0; i < pts.rows; i++) {
317 const int idx = 5 * i;
318 const float k_inv_u = points[idx ];
319 const float k_inv_v = points[idx+1];
320 const float norm = 1.f / sqrtf(k_inv_u*k_inv_u + k_inv_v*k_inv_v + 1);
321 (*calib_norm_pts_++) = k_inv_u * norm;
322 (*calib_norm_pts_++) = k_inv_v * norm;
323 (*calib_norm_pts_++) = norm;
324 points[idx ] = k11 * k_inv_u + k12 * k_inv_v + k13;
325 points[idx+1] = k22 * k_inv_v + k23;
326 }

328/*
329 * decompose Projection Matrix to calibration, rotation and translation
330 * Assume K = [fx 0 tx
331 * 0 fy ty
332 * 0 0 1]
333 */
334void Utils::decomposeProjection (const Mat &P, Matx33d &K, Matx33d &R, Vec3d &t, bool same_focal) {
335 const Matx33d M = P.colRange(0,3);
336 double scale = norm(M.row(2)); scale *= scale;
337 K = Matx33d::eye();
338 K(1,2) = M.row(1).dot(M.row(2)) / scale;
339 K(0,2) = M.row(0).dot(M.row(2)) / scale;
340 K(1,1) = sqrt(M.row(1).dot(M.row(1)) / scale - K(1,2)*K(1,2));
341 K(0,0) = sqrt(M.row(0).dot(M.row(0)) / scale - K(0,2)*K(0,2));
342 if (same_focal)
343 K(0,0) = K(1,1) = (K(0,0) + K(1,1)) / 2;
344 R = K.inv() * M / sqrt(scale);
345 if (determinant(M) < 0) R *= -1;
346 t = R * M.inv() * Vec3d(P.col(3));
347}

349double Utils::getPoissonCDF (double lambda, int inliers) {
350 double exp_lambda = exp(-lambda), cdf = exp_lambda, lambda_i_div_fact_i = 1;
351 for (int i = 1; i <= inliers; i++) {
352 lambda_i_div_fact_i *= (lambda / i);
353 cdf += exp_lambda * lambda_i_div_fact_i;
354 if (fabs(cdf - 1) < DBL_EPSILON) // cdf is almost 1
355 break;
356 }
357 return cdf;
358}

360// since F(E) has rank 2 we use cross product to compute epipole,
361// since the third column / row is linearly dependent on two first
362// this is faster than SVD
363// It is recommended to normalize F, such that ||F|| = 1
364Vec3d Utils::getLeftEpipole (const Mat &F/*E*/) {
365 Vec3d _e = F.col(0).cross(F.col(2)); // F^T e' = 0; e'^T F = 0
366 const auto * const e = _e.val;
367 if (e[0] <= DBL_EPSILON && e[0] > -DBL_EPSILON &&
368 e[1] <= DBL_EPSILON && e[1] > -DBL_EPSILON &&
369 e[2] <= DBL_EPSILON && e[2] > -DBL_EPSILON)
370 _e = Vec3d(Mat(F.col(1))).cross(F.col(2)); // if e' is zero
371 return _e; // e'
372}
373Vec3d Utils::getRightEpipole (const Mat &F/*E*/) {
374 Vec3d _e = F.row(0).cross(F.row(2)); // Fe = 0
375 const auto * const e = _e.val;
376 if (e[0] <= DBL_EPSILON && e[0] > -DBL_EPSILON &&
377 e[1] <= DBL_EPSILON && e[1] > -DBL_EPSILON &&
378 e[2] <= DBL_EPSILON && e[2] > -DBL_EPSILON)
379 _e = F.row(1).cross(F.row(2)); // if e is zero
380 return _e;
381}

383void Utils::densitySort (const Mat &points, int knn, Mat &sorted_points, std::vector<int> &sorted_mask) {
384 // mask of sorted points (array of indexes)
385 const int points_size = points.rows, dim = points.cols;
386 sorted_mask = std::vector<int >(points_size);
387 for (int i = 0; i < points_size; i++)
388 sorted_mask[i] = i;
389
390 // get neighbors
391 cv::Ptr<FlannNeighborhoodGraph> graph = FlannNeighborhoodGraph::create(points, points_size, knn,
392 true /*get distances */, 6, 1);
393
394 std::vector<double> sum_knn_distances (points_size, 0);
395 for (int p = 0; p < points_size; p++) {
396 const std::vector<double> &dists = graph->getNeighborsDistances(p);
397 for (int k = 0; k < knn; k++)
398 sum_knn_distances[p] += dists[k];
399 }
400
401 // compare by sum of distances to k nearest neighbors.
402 std::sort(sorted_mask.begin(), sorted_mask.end(), [&](int a, int b) {
403 return sum_knn_distances[a] < sum_knn_distances[b];
404 });
405
406 // copy array of points to array with sorted points
407 // using @sorted_idx mask of sorted points indexes
408
409 sorted_points = Mat(points_size, dim, points.type());
410 const auto * const points_ptr = (float *) points.data;
411 auto * spoints_ptr = (float *) sorted_points.data;
412 for (int i = 0; i < points_size; i++) {
413 const int pt2 = sorted_mask[i] * dim;
414 for (int j = 0; j < dim; j++)
415 (*spoints_ptr++) = points_ptr[pt2+j];
416 }
417}

425/*
426 * Eliminate matrix of m rows and n columns to be upper triangular.
427 */
428bool Math::eliminateUpperTriangular (std::vector<double> &a, int m, int n) {
429 for (int r = 0; r < m; r++){
430 double pivot = a[r*n+r];
431 int row_with_pivot = r;
432
433 // find the maximum pivot value among r-th column
434 for (int k = r+1; k < m; k++)
435 if (fabs(pivot) < fabs(a[k*n+r])) {
436 pivot = a[k*n+r];
437 row_with_pivot = k;
438 }
439
440 // if pivot value is 0 continue
441 if (fabs(pivot) < DBL_EPSILON)
442 continue;
443
444 // swap row with maximum pivot value with current row
445 for (int c = r; c < n; c++)
446 std::swap(a[row_with_pivot*n+c], a[r*n+c]);
447
448 // eliminate other rows
449 for (int j = r+1; j < m; j++){
450 const int row_idx1 = j*n, row_idx2 = r*n;
451 const auto fac = a[row_idx1+r] / pivot;
452 a[row_idx1+r] = 0; // zero eliminated element
453 for (int c = r+1; c < n; c++)
454 a[row_idx1+c] -= fac * a[row_idx2+c];
455 }
456 }
457 return true;
458}

460double Utils::intersectionOverUnion (const std::vector<bool> &a, const std::vector<bool> &b) {
461 int intersects = 0, unions = 0;
462 for (int i = 0; i < (int)a.size(); i++)
463 if (a[i] || b[i]) {
464 unions++; // one value is true
465 if (a[i] && b[i])
466 intersects++; // a[i] == b[i] and if they both true
467 }
468 if (unions == 0) return 0.0;
469 return (double) intersects / unions;
470}

473class UniformRandomGeneratorImpl : public UniformRandomGenerator {
474private:
475 int subset_size = 0, max_range = 0;
476 std::vector<int> subset;
477 RNG rng;
478public:
479 explicit UniformRandomGeneratorImpl (int state) : rng(state) {}
480
481 // interval is <0; max_range);
482 UniformRandomGeneratorImpl (int state, int max_range_, int subset_size_) : rng(state) {
483 subset_size = subset_size_;
484 max_range = max_range_;
485 subset = std::vector<int>(subset_size_);
486 }
487 int getRandomNumber () override {
488 return rng.uniform(0, max_range);
489 }
490 int getRandomNumber (int max_rng) override {
491 return rng.uniform(0, max_rng);
492 }
493 // closed range
494 void resetGenerator (int max_range_) override {
495 CV_CheckGE(0, max_range_, "max range must be greater than 0");
496 max_range = max_range_;
497 }
498
499 void generateUniqueRandomSet (std::vector<int>& sample) override {
500 CV_CheckLE(subset_size, max_range, "RandomGenerator. Subset size must be LE than range!");
501 int j, num;
502 sample[0] = rng.uniform(0, max_range);
503 for (int i = 1; i < subset_size;) {
504 num = rng.uniform(0, max_range);
505 // check if value is in array
506 for (j = i - 1; j >= 0; j--)
507 if (num == sample[j])
508 // if so, generate again
509 break;
510 // success, value is not in array, so it is unique, add to sample.
511 if (j == -1) sample[i++] = num;
512 }
513 }
514
515 // interval is <0; max_range)
516 void generateUniqueRandomSet (std::vector<int>& sample, int max_range_) override {
517 /*
518 * if subset size is bigger than range then array cannot be unique,
519 * so function has infinite loop.
520 */
521 CV_CheckLE(subset_size, max_range_, "RandomGenerator. Subset size must be LE than range!");
522 int num, j;
523 sample[0] = rng.uniform(0, max_range_);

572// @k_minth - desired k-th minimal element. For median is half of array
573// closed working interval of array <@left; @right>
574float quicksort_median (std::vector<float> &array, int k_minth, int left, int right);
575float quicksort_median (std::vector<float> &array, int k_minth, int left, int right) {
576 // length is 0, return single value
577 if (right - left == 0) return array[left];
578
579 // get pivot, the rightest value in array
580 const auto pivot = array[right];
581 int right_ = right - 1; // -1, not including pivot
582 // counter of values smaller equal than pivot
583 int j = left, values_less_eq_pivot = 1; // 1, inludes pivot already
584 for (; j <= right_;) {
585 if (array[j] <= pivot) {
586 j++;
587 values_less_eq_pivot++;
588 } else
589 // value is bigger than pivot, swap with right_ value
590 // swap values in array and decrease interval
591 std::swap(array[j], array[right_--]);
592 }
593 if (values_less_eq_pivot == k_minth) return pivot;
594 if (k_minth > values_less_eq_pivot)
595 return quicksort_median(array, k_minth - values_less_eq_pivot, j, right-1);
596 else
597 return quicksort_median(array, k_minth, left, j-1);
598}
599
600// find median using quicksort with complexity O(log n)
601// Note, function changes order of values in array
602float Utils::findMedian (std::vector<float> &array) {
603 const int length = static_cast<int>(array.size());
604 if (length % 2) {
605 // odd number of values
606 return quicksort_median (array, length/2+1, 0, length-1);
607 } else {
608 // even: return average
609 return (quicksort_median(array, length/2 , 0, length-1) +
610 quicksort_median(array, length/2+1, 0, length-1))*.5f;
611 }

615struct FlannMatcher
616{
617 Ptr<flann::IndexParams> indexParams;
618 Ptr<flann::SearchParams> searchParams;
619 Ptr<flann::Index> flannIndex;
620
621 FlannMatcher( const Ptr<flann::IndexParams>& _indexParams, const Ptr<flann::SearchParams>& _searchParams )
622 : indexParams(_indexParams), searchParams(_searchParams)
623 {
624 CV_Assert( _indexParams );
625 CV_Assert( _searchParams );
626 }
627
628 void radiusMatch( const Mat& queryDescriptors, const Mat& trainDescriptors,
629 std::vector<std::vector<DMatch> >& matches, float maxDistance)
630 {
631 CV_Assert( maxDistance > std::numeric_limits<float>::epsilon() );
632
633 if( trainDescriptors.empty() || queryDescriptors.empty() )
634 return;
635
636 matches.clear();
637
638 flannIndex = makePtr<flann::Index>( trainDescriptors, *indexParams );
639
640 Mat indices( queryDescriptors.rows, 1, CV_32SC1, Scalar::all(-1) );
641 Mat dists( queryDescriptors.rows, 1, CV_32FC1, Scalar::all(-1) );
642 for( int qIdx = 0; qIdx < queryDescriptors.rows; qIdx++ )
643 {
644 Mat queryDescriptorsRow = queryDescriptors.row(qIdx);
645 Mat indicesRow = indices.row(qIdx);
646 Mat distsRow = dists.row(qIdx);
647 flannIndex->radiusSearch( queryDescriptorsRow, indicesRow, distsRow, maxDistance*maxDistance, 1, *searchParams );
648 }
649
650 convertToDMatches( indices, dists, matches );
651 }
652
653 void convertToDMatches( const Mat& indices, const Mat& dists, std::vector<std::vector<DMatch> >& matches )
654 {
655 matches.resize( indices.rows );
656 for( int i = 0; i < indices.rows; i++ )
657 {
658 for( int j = 0; j < indices.cols; j++ )
659 {
660 int idx = indices.at<int>(i, j);
661 if( idx >= 0 )
662 {
663 int imgIdx = 0;
664 int trainIdx = idx;
665 float dist = 0;
666 if (dists.type() == CV_32S)
667 dist = static_cast<float>( dists.at<int>(i,j) );
668 else
669 dist = std::sqrt(dists.at<float>(i,j));
670 matches[i].push_back( DMatch( i, trainIdx, imgIdx, dist ) );
671 }
672 }
673 }
674 }

773class GridNeighborhoodGraphImpl : public GridNeighborhoodGraph {
774private:
775 // This struct is used for the nearest neighbors search by griding two images.
776 struct CellCoord {
777 int c1x, c1y, c2x, c2y;
778 CellCoord (int c1x_, int c1y_, int c2x_, int c2y_) {
779 c1x = c1x_; c1y = c1y_; c2x = c2x_; c2y = c2y_;
780 }
781 bool operator==(const CellCoord &o) const {
782 return c1x == o.c1x && c1y == o.c1y && c2x == o.c2x && c2y == o.c2y;
783 }
784 bool operator<(const CellCoord &o) const {
785 if (c1x < o.c1x) return true;
786 if (c1x == o.c1x && c1y < o.c1y) return true;
787 if (c1x == o.c1x && c1y == o.c1y && c2x < o.c2x) return true;
788 return c1x == o.c1x && c1y == o.c1y && c2x == o.c2x && c2y < o.c2y;
789 }
790 };
791
792 std::vector<std::vector<int>> graph;
793public:
794 GridNeighborhoodGraphImpl (const Mat &container_, int points_size,
795 int cell_size_x_img1, int cell_size_y_img1, int cell_size_x_img2, int cell_size_y_img2,
796 int max_neighbors) {
797
798 std::map<CellCoord, std::vector<int >> neighbors_map;
799 const auto * const container = (float *) container_.data;
800 // <int, int, int, int> -> {neighbors set}
801 // Key is cell position. The value is indexes of neighbors.
802
803 const float cell_sz_x1 = 1.f / (float) cell_size_x_img1,
804 cell_sz_y1 = 1.f / (float) cell_size_y_img1,
805 cell_sz_x2 = 1.f / (float) cell_size_x_img2,
806 cell_sz_y2 = 1.f / (float) cell_size_y_img2;
807 const int dimension = container_.cols;
808 for (int i = 0; i < points_size; i++) {
809 const int idx = dimension * i;
810 neighbors_map[CellCoord((int)(container[idx ] * cell_sz_x1),
811 (int)(container[idx+1] * cell_sz_y1),
812 (int)(container[idx+2] * cell_sz_x2),
813 (int)(container[idx+3] * cell_sz_y2))].emplace_back(i);
814 }
815
816 //--------- create a graph ----------
817 graph = std::vector<std::vector<int>>(points_size);
818
819 // store neighbors cells into graph (2D vector)
820 for (const auto &cell : neighbors_map) {
821 const int neighbors_in_cell = static_cast<int>(cell.second.size());
822 // only one point in cell -> no neighbors
823 if (neighbors_in_cell < 2) continue;
824
825 const std::vector<int> &neighbors = cell.second;
826 // ---------- fill graph -----
827 for (int v_in_cell : neighbors) {
828 // there is always at least one neighbor
829 auto &graph_row = graph[v_in_cell];
830 graph_row = std::vector<int>(std::min(max_neighbors, neighbors_in_cell-1));
831 int j = 0;
832 for (int n : neighbors)
833 if (n != v_in_cell){
834 graph_row[j++] = n;
835 if (j >= max_neighbors)
836 break;
837 }
838 }
839 }
840 }
841 const std::vector<std::vector<int>> &getGraph () const override { return graph; }
842 inline const std::vector<int> &getNeighbors(int point_idx) const override {
843 // Note, neighbors vector also includes point_idx!
844 // return neighbors_map[vertices_to_cells[point_idx]];
845 return graph[point_idx];
846 }
847};
