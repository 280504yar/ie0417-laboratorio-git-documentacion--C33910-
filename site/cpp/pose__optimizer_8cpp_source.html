15int poseInlierCheck(Frame& frame, const Mat& K, double reprojThresh)
16{
17 const double fx = K.at<double>(0, 0);
18 const double fy = K.at<double>(1, 1);
19 const double cx = K.at<double>(0, 2);
20 const double cy = K.at<double>(1, 2);
21 const Matx44d& T = frame.poseCw;
22
23 int nInliers = 0;
24 for (size_t i = 0; i < frame.mapPoints.size(); ++i)
25 {
26 frame.outliers[i] = true;
27 MapPoint* mp = frame.mapPoints[i];
28 if (!mp || mp->bad) continue;
29
30 const double Xc = T(0,0)*mp->pos.x + T(0,1)*mp->pos.y + T(0,2)*mp->pos.z + T(0,3);
31 const double Yc = T(1,0)*mp->pos.x + T(1,1)*mp->pos.y + T(1,2)*mp->pos.z + T(1,3);
32 const double Zc = T(2,0)*mp->pos.x + T(2,1)*mp->pos.y + T(2,2)*mp->pos.z + T(2,3);
33 if (Zc <= 0.0) continue;
34
35 const double u = fx * Xc / Zc + cx;
36 const double v = fy * Yc / Zc + cy;
37 const double dx = u - static_cast<double>(frame.undistKpts[i].x);
38 const double dy = v - static_cast<double>(frame.undistKpts[i].y);
39
40 if (std::sqrt(dx * dx + dy * dy) <= reprojThresh)
41 {
42 frame.outliers[i] = false;
43 ++nInliers;
44 }
45 }
46 return nInliers;
47}
