43 MatShape getOutShape(const MatShape& inpshape, const Mat& shapeTensor) const
44 {
45 MatShape shape0 = tensorToShape(shapeTensor);
46 MatShape shape = inpshape.expand(shape0);
47 // according to ONNX specification, the specified shape can be smaller than the input!
48 // so we comment off the check
49 // CV_Assert(shape == shape0); // check that input can be expanded to the specified shape
50 return shape;
51 }

53 bool getMemoryShapes(const std::vector<MatShape>& inputs,
54 const int,
55 std::vector<MatShape> &outputs,
56 std::vector<MatShape> &internals) const CV_OVERRIDE
57 {
58 CV_Assert(!dynamicOutputShapes());
59
60 size_t ninputs = inputs.size();
61 CV_Assert(ninputs == (size_t)2);
62 Net::Impl* netimpl_ = getNetImpl(this);
63
64 Mat shapeTensor = netimpl_->argTensor(this->inputs[1]);

66 outputs.assign(1, getOutShape(inputs[0], shapeTensor));
67 internals.clear();

88 void forward(InputArrayOfArrays inputs_arr,
89 OutputArrayOfArrays outputs_arr,
90 OutputArrayOfArrays) CV_OVERRIDE
91 {
92 CV_TRACE_FUNCTION();
93 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
94
95 Size size = inputs_arr.size();
96 int ninputs = size.area();
97 CV_Assert(ninputs == 2);
98
99 Mat inp = inputs_arr.getMat(0);
100 int inptype = inp.type();
101 Mat shapeTensor = inputs_arr.getMat(1);
102
103 MatShape outshape = getOutShape(inp.shape(), shapeTensor);
104
105 auto kind = outputs_arr.kind();
106 if (kind == _InputArray::STD_VECTOR_MAT) {
107 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
108 outs.resize(1);
109 outs[0].fit(outshape, inptype);
110 broadcast(inp, outshape, outs[0]);
111 } else if (kind == _InputArray::STD_VECTOR_UMAT) {
112 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
113 outs.resize(1);
114 outs[0].fit(outshape, inptype);
115 Mat temp(outshape, inptype);
116 broadcast(inp, outshape, temp);
117 temp.copyTo(outs[0]);
118 } else {
119 CV_Error(Error::StsNotImplemented, "");
120 }
121 }
122};
