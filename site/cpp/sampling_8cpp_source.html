27int voxelGridSampling(OutputArray sampled_point_flags, InputArray input_pts,
28 const float length, const float width, const float height)
29{
30 CV_CheckGT(length, 0.0f, "Invalid length of grid");
31 CV_CheckGT(width, 0.0f, "Invalid width of grid");
32 CV_CheckGT(height, 0.0f, "Invalid height of grid");
33
34 // Get input point cloud data
35 Mat ori_pts;
36 getPointsMatFromInputArray(input_pts, ori_pts, 0);
37
38 const int ori_pts_size = ori_pts.rows;
39
40 float *const ori_pts_ptr = (float *) ori_pts.data;
41
42 // Compute the minimum and maximum bounding box values
43 float x_min, x_max, y_min, y_max, z_min, z_max;
44 x_max = x_min = ori_pts_ptr[0];
45 y_max = y_min = ori_pts_ptr[1];
46 z_max = z_min = ori_pts_ptr[2];
47
48 for (int i = 1; i < ori_pts_size; ++i)
49 {
50 float *const ptr_base = ori_pts_ptr + 3 * i;
51 float x = ptr_base[0], y = ptr_base[1], z = ptr_base[2];
52
53 x_min = std::min(x_min, x);
54 x_max = std::max(x_max, x);
55
56 y_min = std::min(y_min, y);
57 y_max = std::max(y_max, y);
58
59 z_min = std::min(z_min, z);
60 z_max = std::max(z_max, z);
61 }
62
63 // Up to 2^64 grids for key type int64_t
64 // For larger point clouds or smaller grid sizes, use string or hierarchical map
65 typedef int64_t keyType;
66 std::unordered_map<keyType, std::vector<int>> grids;
67
68// int init_size = ori_pts_size * 0.02;
69// grids.reserve(init_size);
70
71 // Divide points into different grids
72
73 keyType offset_y = static_cast<keyType>(cvCeil((x_max - x_min) / length));
74 keyType offset_z = offset_y * static_cast<keyType>(cvCeil((y_max - y_min) / width));
75
76 for (int i = 0; i < ori_pts_size; ++i)
77 {
78 int ii = 3 * i;
79 keyType hx = static_cast<keyType>((ori_pts_ptr[ii] - x_min) / length);
80 keyType hy = static_cast<keyType>((ori_pts_ptr[ii + 1] - y_min) / width);
81 keyType hz = static_cast<keyType>((ori_pts_ptr[ii + 2] - z_min) / height);
82 // Convert three-dimensional coordinates to one-dimensional coordinates key
83 // Place the stacked three-dimensional grids(boxes) into one dimension (placed in a straight line)
84 keyType key = hx + hy * offset_y + hz * offset_z;
85
86 std::unordered_map<keyType, std::vector<int>>::iterator iter = grids.find(key);
87 if (iter == grids.end())
88 grids[key] = {i};
89 else
90 iter->second.push_back(i);
91 }
92
93 const int pts_new_size = static_cast<int>(grids.size());
94 std::vector<char> _sampled_point_flags(ori_pts_size, 0);
95
96 // Take out the points in the grid and calculate the point closest to the centroid
97 std::unordered_map<keyType, std::vector<int>>::iterator grid_iter = grids.begin();
98
99 for (int label_id = 0; label_id < pts_new_size; ++label_id, ++grid_iter)
100 {
101 std::vector<int> grid_pts = grid_iter->second;
102 int grid_pts_cnt = static_cast<int>(grid_iter->second.size());
103 int sampled_point_idx = grid_pts[0];
104 // 1. one point in the grid, select it directly, no need to calculate.
105 // 2. two points in the grid, the distance from these two points to their centroid is the same,
106 // can directly select one of them, also do not need to calculate
107 if (grid_pts_cnt > 2)
108 {
109 // Calculate the centroid position
110 float sum_x = 0, sum_y = 0, sum_z = 0;
111 for (const int &item: grid_pts)
112 {
113 float *const ptr_base = ori_pts_ptr + 3 * item;
114 sum_x += ptr_base[0];
115 sum_y += ptr_base[1];
116 sum_z += ptr_base[2];
117 }
118
119 float centroid_x = sum_x / grid_pts_cnt, centroid_y = sum_y / grid_pts_cnt, centroid_z =
120 sum_z / grid_pts_cnt;
121
122 // Find the point closest to the centroid
123 float min_dist_square = FLT_MAX;
124 for (const int &item: grid_pts)
125 {
126 float *const ptr_base = ori_pts_ptr + item * 3;
127 float x = ptr_base[0], y = ptr_base[1], z = ptr_base[2];
128
129 float dist_square = (x - centroid_x) * (x - centroid_x) +
130 (y - centroid_y) * (y - centroid_y) +
131 (z - centroid_z) * (z - centroid_z);
132 if (dist_square < min_dist_square)
133 {
134 min_dist_square = dist_square;
135 sampled_point_idx = item;
136 }
137 }
138 }
139
140 _sampled_point_flags[sampled_point_idx] = 1;
141 }
142 Mat(_sampled_point_flags).copyTo(sampled_point_flags);
143
144 return pts_new_size;
145} // voxelGrid()

147void
148randomSampling(OutputArray sampled_pts, InputArray input_pts, const int sampled_pts_size, RNG *rng)
149{
150 CV_CheckGT(sampled_pts_size, 0, "The point cloud size after sampling must be greater than 0.");
151 CV_CheckDepth(sampled_pts.depth(), sampled_pts.isMat() || sampled_pts.depth() == CV_32F,
152 "The output data type only supports Mat, vector<Point3f>.");
153
154 // Get input point cloud data
155 Mat ori_pts;
156 getPointsMatFromInputArray(input_pts, ori_pts, 0);
157
158 const int ori_pts_size = ori_pts.rows;
159 CV_CheckLT(sampled_pts_size, ori_pts_size,
160 "The sampled point cloud size must be smaller than the original point cloud size.");
161
162 std::vector<int> pts_idxs(ori_pts_size);
163 for (int i = 0; i < ori_pts_size; ++i) pts_idxs[i] = i;
164 randShuffle(pts_idxs, 1, rng);
165
166 int channels = sampled_pts.channels();
167 if (channels == 3 && sampled_pts.isVector())
168 {
169 // std::vector<cv::Point3f>
170 sampled_pts.create(1, sampled_pts_size, CV_32FC3);
171 }
172 else
173 {
174 // std::vector<float> or cv::Mat
175 sampled_pts.create(sampled_pts_size, 3, CV_32F);
176 }
177
178 Mat out = sampled_pts.getMat();
179
180 float *const ori_pts_ptr = (float *) ori_pts.data;
181 float *const sampled_pts_ptr = (float *) out.data;
182 for (int i = 0; i < sampled_pts_size; ++i)
183 {
184 float *const ori_pts_ptr_base = ori_pts_ptr + pts_idxs[i] * 3;
185 float *const sampled_pts_ptr_base = sampled_pts_ptr + i * 3;
186 sampled_pts_ptr_base[0] = ori_pts_ptr_base[0];
187 sampled_pts_ptr_base[1] = ori_pts_ptr_base[1];
188 sampled_pts_ptr_base[2] = ori_pts_ptr_base[2];
189 }
190
191} // randomSampling()

194randomSampling(OutputArray sampled_pts, InputArray input_pts, const float sampled_scale, RNG *rng)
195{
196 CV_CheckGT(sampled_scale, 0.0f, "The point cloud sampled scale must greater than 0.");
197 CV_CheckLT(sampled_scale, 1.0f, "The point cloud sampled scale must less than 1.");
198 Mat ori_pts;
199 getPointsMatFromInputArray(input_pts, ori_pts, 0);
200 randomSampling(sampled_pts, input_pts, cvCeil(sampled_scale * ori_pts.rows), rng);
201} // randomSampling()

215int farthestPointSampling(OutputArray sampled_point_flags, InputArray input_pts,
216 const int sampled_pts_size, const float dist_lower_limit, RNG *rng)
217{
218 CV_CheckGT(sampled_pts_size, 0, "The point cloud size after sampling must be greater than 0.");
219 CV_CheckGE(dist_lower_limit, 0.0f,
220 "The distance lower bound must be greater than or equal to 0.");
221
222 // Get input point cloud data
223 Mat ori_pts;
224 // In order to keep the points continuous in memory (which allows better support for SIMD),
225 // the position of the points may be changed, data copying is mandatory
226 getPointsMatFromInputArray(input_pts, ori_pts, 1, true);
227
228 const int ori_pts_size = ori_pts.rows * ori_pts.cols / 3;
229 CV_CheckLT(sampled_pts_size, ori_pts_size,
230 "The sampled point cloud size must be smaller than the original point cloud size.");
231
232
233 // idx arr [ . . . . . . . . . ] --- sampling ---> [ . . . . . . . . . ]
234 // C S | C
235 // _idxs records the original location/id of the point
236 AutoBuffer<int> _idxs(ori_pts_size);
237 // _dist_square records the distance from point(in C) to S
238 AutoBuffer<float> _dist_square(ori_pts_size);
239 int *idxs = _idxs.data();
240 float *dist_square = _dist_square.data();
241 for (int i = 0; i < ori_pts_size; ++i)
242 {
243 idxs[i] = i;
244 dist_square[i] = FLT_MAX;
245 }
246
247 // Randomly take a seed point from C and take it from C to S
248 int seed = (int) ((rng ? rng->next() : theRNG().next()) % ori_pts_size);
249 idxs[0] = seed;
250 idxs[seed] = 0;
251
252 // Pointer (base address) of access point data x,y,z
253 float *const ori_pts_ptr_x = (float *) ori_pts.data;
254 float *const ori_pts_ptr_y = ori_pts_ptr_x + ori_pts_size;
255 float *const ori_pts_ptr_z = ori_pts_ptr_y + ori_pts_size;
256
257 // Ensure that the point(in C) data x,y,z is continuous in the memory respectively
258 _swap(ori_pts_ptr_x[seed], ori_pts_ptr_x[0]);
259 _swap(ori_pts_ptr_y[seed], ori_pts_ptr_y[0]);
260 _swap(ori_pts_ptr_z[seed], ori_pts_ptr_z[0]);
261
262 int sampled_cnt = 1;
263 const float dist_lower_limit_square = dist_lower_limit * dist_lower_limit;
264 while (sampled_cnt < sampled_pts_size)
265 {
266 int last_pt = sampled_cnt - 1;
267 float last_pt_x = ori_pts_ptr_x[last_pt];
268 float last_pt_y = ori_pts_ptr_y[last_pt];
269 float last_pt_z = ori_pts_ptr_z[last_pt];
270
271 // Calculate the distance from point in C to set S
272 float max_dist_square = 0;
273 int next_pt = sampled_cnt;
274 int i = sampled_cnt;
275#if CV_SIMD
276 v_float32 v_last_p_x = vx_setall_f32(last_pt_x);
277 v_float32 v_last_p_y = vx_setall_f32(last_pt_y);
278 v_float32 v_last_p_z = vx_setall_f32(last_pt_z);
279
280 for (; i <= ori_pts_size - VTraits<v_float32>::vlanes(); i += VTraits<v_float32>::vlanes())
281 {
282 v_float32 vx_diff = v_sub(v_last_p_x, vx_load(ori_pts_ptr_x + i));
283 v_float32 vy_diff = v_sub(v_last_p_y, vx_load(ori_pts_ptr_y + i));
284 v_float32 vz_diff = v_sub(v_last_p_z, vx_load(ori_pts_ptr_z + i));
285
286 v_float32 v_next_dist_square =
287 v_add(v_add(v_mul(vx_diff, vx_diff), v_mul(vy_diff, vy_diff)), v_mul(vz_diff, vz_diff));
288
289 // Update the distance from the points(in C) to S
290 float *dist_square_ptr = dist_square + i;
291 v_float32 v_dist_square = vx_load(dist_square_ptr);
292 v_dist_square = v_min(v_dist_square, v_next_dist_square);
293 vx_store(dist_square_ptr, v_dist_square);
294
295 // Find a point in C that is the farthest away from S and take it from C to S
296 if (v_check_any(v_gt(v_dist_square, vx_setall_f32(max_dist_square))))
297 {
298 for (int m = 0; m < VTraits<v_float32>::vlanes(); ++m)
299 {
300 if (dist_square_ptr[m] > max_dist_square)
301 {
302 next_pt = i + m;
303 max_dist_square = dist_square_ptr[m];
304 }
305 }
306 }
307
308 }
309
310#endif
311 for (; i < ori_pts_size; ++i)
312 {
313 float x_diff = (last_pt_x - ori_pts_ptr_x[i]);
314 float y_diff = (last_pt_y - ori_pts_ptr_y[i]);
315 float z_diff = (last_pt_z - ori_pts_ptr_z[i]);
316 float next_dist_square = x_diff * x_diff + y_diff * y_diff + z_diff * z_diff;
317 // Update the distance from the points(in C) to S
318 dist_square[i] = std::min(dist_square[i], next_dist_square);
319 // Find a point in C that is the farthest away from S and take it from C to S
320 if (dist_square[i] > max_dist_square)
321 {
322 next_pt = i;
323 max_dist_square = dist_square[i];
324 }
325 }
326
327
328 if (max_dist_square < dist_lower_limit_square)
329 break;
330
331 // Take point next_pt from C to S
332 _swap(idxs[next_pt], idxs[sampled_cnt]);
333 _swap(dist_square[next_pt], dist_square[sampled_cnt]);
334
335 // Ensure that the point(in C) data x,y,z is continuous in the memory respectively
336 _swap(ori_pts_ptr_x[next_pt], ori_pts_ptr_x[sampled_cnt]);
337 _swap(ori_pts_ptr_y[next_pt], ori_pts_ptr_y[sampled_cnt]);
338 _swap(ori_pts_ptr_z[next_pt], ori_pts_ptr_z[sampled_cnt]);
339
340 ++sampled_cnt;
341 }
342
343 std::vector<char> _sampled_point_flags(ori_pts_size, 0);
344 for (int j = 0; j < sampled_cnt; ++j)
345 {
346 _sampled_point_flags[idxs[j]] = 1;
347 }
348
349 Mat(_sampled_point_flags).copyTo(sampled_point_flags);
350
351 return sampled_cnt;
352} // farthestPointSampling()

354int farthestPointSampling(OutputArray sampled_point_flags, InputArray input_pts,
355 const float sampled_scale, const float dist_lower_limit, RNG *rng)
356{
357 CV_CheckGT(sampled_scale, 0.0f, "The point cloud sampled scale must greater than 0.");
