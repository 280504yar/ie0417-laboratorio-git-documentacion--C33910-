5// OpenCV-backed implementation of MLAS's threading primitives.
6//
7// Replaces lib/threading.cpp from upstream MLAS (which, when built with
8// BUILD_MLAS_NO_ONNXRUNTIME and a nullptr ThreadPool, runs everything in a
9// serial for-loop). MLAS internally calls MlasGetMaximumThreadCount() to
10// pick a partition count; for the standalone build that returns 1, so even
11// with a parallel `MlasTrySimpleParallel` MLAS would still emit a single
12// iteration. We patch both halves:
13//
14// 1. MlasGetMaximumThreadCount() in mlasi.h returns cv::getNumThreads()
15// when MLAS_OPENCV_THREADING is defined (see the small patch in
16// mlasi.h flagged with that define).
17// 2. The three threaded entry points below dispatch to cv::parallel_for_.
18//
19// Compiled into the opencv_dnn_mlas object library only — the
20// MLAS_OPENCV_THREADING guard makes the file a no-op when picked up by
21// opencv_dnn's recursive src glob, since MLAS internal headers and the
22// MLAS_GEMM_ONLY / BUILD_MLAS_NO_ONNXRUNTIME defines are not in scope there.

44void
45MlasExecuteThreaded(
46 MLAS_THREADED_ROUTINE* ThreadedRoutine,
47 void* Context,
48 ptrdiff_t Iterations,
49 MLAS_THREADPOOL* /*ThreadPool*/)
50{
51 if (Iterations <= 0) return;
52 if (Iterations == 1) { ThreadedRoutine(Context, 0); return; }
53
54 cv::parallel_for_(cv::Range(0, static_cast<int>(Iterations)),
55 [&](const cv::Range& r) {
56 for (int tid = r.start; tid < r.end; tid++) {
57 ThreadedRoutine(Context, static_cast<ptrdiff_t>(tid));
58 }
59 });
60}

79void
80MlasTryBatchParallel(
81 MLAS_THREADPOOL* /*ThreadPool*/,
82 const std::ptrdiff_t Iterations,
83 const std::function<void(std::ptrdiff_t tid)>& Work)
84{
85 // MLAS only calls this for "non-performance-critical" small batches, but
86 // there is no reason to serialize it on a 24-core host either.
87 if (Iterations <= 0) return;
88 if (Iterations == 1) { Work(0); return; }
89
90 cv::parallel_for_(cv::Range(0, static_cast<int>(Iterations)),
91 [&](const cv::Range& r) {
92 for (int tid = r.start; tid < r.end; tid++) {
93 Work(static_cast<std::ptrdiff_t>(tid));
94 }
95 });
