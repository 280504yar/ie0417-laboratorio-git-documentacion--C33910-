17class BufferArea::Block
18{
19private:
20 inline size_t reserve_count() const
21 {
22 return alignment / type_size - 1;
23 }
24public:
25 Block(void **ptr_, ushort type_size_, size_t count_, ushort alignment_)
26 : ptr(ptr_), raw_mem(0), count(count_), type_size(type_size_), alignment(alignment_)
27 {
28 CV_Assert(ptr && *ptr == NULL);
29 }
30 void cleanup() const
31 {
32 CV_DbgAssert(ptr);
33 if (raw_mem)
34 fastFree(raw_mem);
35 }
36 size_t getByteCount() const
37 {
38 return type_size * (count + reserve_count());
39 }
40 void real_allocate()
41 {
42 CV_Assert(ptr && *ptr == NULL);
43 const size_t allocated_count = count + reserve_count();
44 raw_mem = fastMalloc(type_size * allocated_count);
45 if (alignment != type_size)
46 {
47 *ptr = alignPtr(raw_mem, alignment);
48 CV_Assert(reinterpret_cast<size_t>(*ptr) % alignment == 0);
49 CV_Assert(static_cast<uchar*>(*ptr) + type_size * count <= static_cast<uchar*>(raw_mem) + type_size * allocated_count);
50 }
51 else
52 {
53 *ptr = raw_mem;
54 }
55 }
56#ifndef OPENCV_ENABLE_MEMORY_SANITIZER
57 void * fast_allocate(void * buf) const
58 {
59 CV_Assert(ptr && *ptr == NULL);
60 buf = alignPtr(buf, alignment);
61 CV_Assert(reinterpret_cast<size_t>(buf) % alignment == 0);
62 *ptr = buf;
63 return static_cast<void*>(static_cast<uchar*>(*ptr) + type_size * count);
64 }
65#endif
66 bool operator==(void **other) const
67 {
68 CV_Assert(ptr && other);
69 return *ptr == *other;
70 }
71 void zeroFill() const
72 {
73 CV_Assert(ptr && *ptr);
74 memset(static_cast<uchar*>(*ptr), 0, count * type_size);
75 }
76private:
77 void **ptr;
78 void * raw_mem;
79 size_t count;
80 ushort type_size;
81 ushort alignment;
82};

142{
143#ifndef OPENCV_ENABLE_MEMORY_SANITIZER
144 if (!safe)
145 {
146 CV_Assert(totalSize > 0);
147 CV_Assert(oneBuf == NULL);
148 CV_Assert(!blocks.empty());
149 oneBuf = fastMalloc(totalSize);
150 void * ptr = oneBuf;
151 for(std::vector<Block>::const_iterator i = blocks.begin(); i != blocks.end(); ++i)

153 ptr = i->fast_allocate(ptr);
