102// Can `depth` hold `v` exactly? Floats: yes (close enough for our promotion). Integers: only if v
103// is integral and in range. Used so a const operand (e.g. 2.5 in a*2.5) is NOT quantized into a
104// narrow-integer direct kernel - such an op must fall back to the float working type instead.
105static bool depthRepresents(double v, int depth)
106{
107 if (isFloatDepth(depth)) return true;
108 if (v != std::floor(v)) return false;
109 switch (depth)
110 {
111 case CV_8U: return v >= 0 && v <= 255;
112 case CV_8S: return v >= -128 && v <= 127;
113 case CV_16U: return v >= 0 && v <= 65535;
114 case CV_16S: return v >= -32768 && v <= 32767;
115 case CV_32U: return v >= 0 && v <= 4294967295.0;
116 case CV_32S: return v >= -2147483648.0 && v <= 2147483647.0;
117 case CV_64U: return v >= 0 && v <= 18446744073709551615.0;
118 case CV_64S: return v >= -9223372036854775808.0 && v <= 9223372036854775807.0;
119 default: return false;
120 }
121}

125static void intRange(int depth, double& lo, double& hi)
126{
127 switch (depth)
128 {
129 case CV_8U: lo = 0; hi = 255; break;
130 case CV_8S: lo = -128; hi = 127; break;
131 case CV_16U: lo = 0; hi = 65535; break;
132 case CV_16S: lo = -32768; hi = 32767; break;
133 case CV_32U: lo = 0; hi = 4294967295.0; break;
134 case CV_32S: lo = -2147483648.0; hi = 2147483647.0; break;

141// numpy-style arithmetic promotion - INTEGER-PRESERVING and COMMUTATIVE. Same signedness -> the wider
142// integer (keeps the sign). Mixed sign -> a SIGNED result wide enough to hold the unsigned operand's
143// range (8u+8s -> 16s, 16u+16s -> 32s, 32u+32s -> 64s; 64-bit mixed -> 64F, no 128-bit int exists).
144// Any float -> the smallest float covering both operands (8-bit int / 16F / 16BF -> a 16-bit float,
145// 16-bit int / 32F -> 32F, 32/64-bit int / 64F -> 64F; two DISTINCT 16-floats widen to 32F). A flexible
146// operand (EW_DEPTH_NONE == -1) lands in the reserved slot 0 of every LUT (size 0, unsigned, non-float),
147// so the integer path just returns the other operand - no explicit NONE guard needed. Declared in
148// ew_op.hpp.
149//
150// Packed lookup tables, one field per depth at slot (depth+1):
151// leszlut - 3-bit size class: 8-bit -> 1, 16-bit -> 2, 32-bit -> 3, 64-bit -> 4 (NONE/Bool -> 0)
152// signlut - 1 bit, set for signed integer depths
153// fltlut - 1 bit, set for float depths
154int promoteArith(int a, int b)
155{
156 constexpr uint64_t leszlut = 034412243322110ULL;
157 constexpr unsigned signlut = 0b1001111110100u;
158 constexpr unsigned fltlut = 0b1111000000u;
159
160 if (a == b) return a;
161
162 int wa = int((leszlut >> (a+1)*3) & 7u), wb = int((leszlut >> (b+1)*3) & 7u);
163 int fa = int((fltlut >> (a+1)) & 1u), fb = int((fltlut >> (b+1)) & 1u);
164
165 if (fa + fb == 0) // both integer
166 {
167 int sa = int((signlut >> (a+1)) & 1u), sb = int((signlut >> (b+1)) & 1u);
168 if (sa == sb) // same signedness -> the wider one (keeps the sign)
169 return wa >= wb ? a : b;
170 // mixed sign -> a SIGNED result holding the unsigned operand: the signed width if it is already
171 // wider, else one step past the unsigned width; past 64 bits there is no int -> f64.
172 int rw = std::max(wa + (1 - sa), wb + (1 - sb));
173 constexpr int ilut = (CV_64F << 5*5) | (CV_64S << 4*5) | (CV_32S << 3*5) |
174 (CV_16S << 2*5) | (CV_8S << 1*5) | (CV_8S << 0*5);
175 return (ilut >> rw*5) & CV_MAT_DEPTH_MASK;
176 }
177
178 // at least one float -> the smallest float covering both operands. Lift each integer operand to the
179 // float size class that holds it (8-bit -> 16-float, 16-bit -> 32F, 32/64-bit -> 64F); a float
180 // operand keeps its own. Size 2 with exactly one float -> that float (a 8-bit int + 16-float pair);
181 // otherwise 32F, or 64F once the size class reaches 4.
182 wa += 1 - fa;
183 wb += 1 - fb;
184 int maxw = std::max(wa, wb);
185 if (maxw == 2 && fa + fb == 1)
186 return fa*a + (1 - fa)*b;
187 return CV_32F + (maxw >= 4);
188}

204// numpy-ish promotion of two KNOWN depths for the COMPUTE type (float dominates; the wider integer
205// otherwise, and - unlike promoteArith - two WIDE integers float-promote: promote2(16U,64S)=64F).
206// This is cv::add's "wtype": the type both operands are brought to before the op runs.
207static int promote2(int a, int b)
208{
209 if (a == b) return a;
210
211 constexpr unsigned lbits = 3, lmask = (1u << lbits) - 1u;
212 const uint64_t typelut = (uint64_t)((0ULL << CV_8U*lbits) | (0ULL << CV_8S*lbits) |
213 (1ULL << CV_16U*lbits) | (1ULL << CV_16S*lbits) |
214 (2ULL << CV_32U*lbits) | (2ULL << CV_32S*lbits) |
215 (3ULL << CV_16F*lbits) | (3ULL << CV_16BF*lbits) |
216 (3ULL << CV_32F*lbits) | (4ULL << CV_64F*lbits) |
217 (4ULL << CV_64S*lbits) | (4ULL << CV_64U*lbits));
218 unsigned pr_a = unsigned((typelut >> (a*lbits)) & lmask);
219 unsigned pr_b = unsigned((typelut >> (b*lbits)) & lmask);
220 unsigned max_pr = std::max(pr_a, pr_b);
221 constexpr unsigned dbits = CV_CN_SHIFT, dmask = (1u << dbits) - 1u;
222 const unsigned ctypelut = ((CV_16S << 0*dbits) | (CV_32S << 1*dbits) | (CV_64S << 2*dbits) |
223 (CV_32F << 3*dbits) | (CV_64F << 4*dbits));
224 return int((ctypelut >> (max_pr*dbits)) & dmask);
225}

227// Common type for a bit-pattern op (AND/OR/XOR). Unlike promoteArith there is NO numeric promotion:
228// a bitwise op keeps the operand's own type, and a scalar operand simply takes the array's type and
229// is reinterpreted by its bits. A flexible CONST (EW_DEPTH_NONE) yields to the concrete operand; two
230// concrete depths must share the element WIDTH (cv::bitwise requires equal types) - if they differ we
231// keep the wider one (the kernel dispatches by element size). NONE x NONE stays NONE (caller defaults).
232static int promoteBitwise(int a, int b)
233{
234 if (a == EW_DEPTH_NONE) return b;
235 if (b == EW_DEPTH_NONE) return a;
236 if (a == b) return a;
237 return CV_ELEM_SIZE1(a) >= CV_ELEM_SIZE1(b) ? a : b;
238}

240// A wide type in which add(depth,depth->wide) exists and the sum is held without a premature clamp
241// (the signed-widening fallback for ADD/SUB, whose difference may go negative).
242static int safeWide(int depth)
243{
244 constexpr unsigned lbits = 3, lmask = (1u << lbits) - 1u;
245 const uint64_t typelut = (uint64_t)((0ULL << CV_8U*lbits) | (0ULL << CV_8S*lbits) |
246 (1ULL << CV_16U*lbits) | (1ULL << CV_16S*lbits) |
247 (2ULL << CV_32U*lbits) | (2ULL << CV_32S*lbits) |
248 (3ULL << CV_16F*lbits) | (3ULL << CV_16BF*lbits) |
249 (3ULL << CV_32F*lbits) | (4ULL << CV_64F*lbits) |
250 (4ULL << CV_64S*lbits) | (4ULL << CV_64U*lbits));
251 unsigned pr_depth = unsigned((typelut >> (depth*lbits)) & lmask);
252 constexpr unsigned dbits = CV_CN_SHIFT, dmask = (1u << dbits) - 1u;
253 const unsigned ctypelut = ((CV_16S << 0*dbits) | (CV_32S << 1*dbits) | (CV_64S << 2*dbits) |
254 (CV_32F << 3*dbits) | (CV_64F << 4*dbits));
255 return int((ctypelut >> (pr_depth*dbits)) & dmask);
256}

258// Inline capacity for the small per-const double scratch used during type inference; larger
259// channel counts (e.g. Vec<_,16>) just spill the AutoBuffer to the heap.

262// Append `channels` values of depth `srcdepth` (raw bytes at `data`, or zero-filled if null) to
263// constbuf, padded to a whole number of uint64_t slots. Returns the offset (in uint64_t units).
264static size_t appendConstBuf(AutoBuffer<uint64_t, 16>& constbuf, int srcdepth, const void* data, int channels)
265{
266 const int cn = std::max(1, channels);
267 const size_t nbytes = (size_t)cn * CV_ELEM_SIZE1(srcdepth);
268 const size_t nu64 = (nbytes + sizeof(uint64_t) - 1) / sizeof(uint64_t);
269 const size_t ofs = constbuf.size();
270 constbuf.resize(ofs + nu64);
271 uchar* dst = (uchar*)(constbuf.data() + ofs);
272 if (data) memcpy(dst, data, nbytes); else memset(dst, 0, nbytes);
273 return ofs;
274}

276// Read a CONST slot's source values (constbuf, `srcdepth`) as doubles into `buf`. Returns channels.
277static int constDoubles(const TExpr& e, int s, AutoBuffer<double, MAX_LOCAL_CN>& buf)
278{
279 const TExpr::Arg& a = e.arginfo[s];
280 const int cn = std::max(1, a.channels);
281 buf.resize(cn);
282 const uchar* src = (const uchar*)(e.constbuf.data() + a.constofs);

290// Materialize a flexible CONST `s` at depth `d`, or return `s` unchanged for a typed operand
291// (the emit* layer then casts it to the compute depth). Used by the emit* policy layers.
292static inline bool isFlexConst(const TExpr& e, int s)
293{
294 return e.arginfo[s].kind == TExpr::CONST && e.arginfo[s].depth == EW_DEPTH_NONE;

297// Can a flexible CONST `s` be represented exactly at depth `d`? (typed operands trivially "fit").
298static bool constFits(const TExpr& e, int s, int d)
299{
300 if (!isFlexConst(e, s)) return true;
301 AutoBuffer<double, MAX_LOCAL_CN> v;
302 int cn = constDoubles(e, s, v);
303 for (int ch = 0; ch < cn; ch++)
304 if (!depthRepresents(v[ch], d)) return false;
305 return true;
306}

308// ---------------------------------------------------------------------------
309// TExpr::emitBinary(): type policy for a 2-input op. Derive the result depth, the compute depth and a
310// wide fallback per op family, then cast both operands and emit (direct, else wide+narrow). Operands
311// may be typed (INPUT/TEMP) or a flexible CONST (materialized at the compute depth here).
312// ---------------------------------------------------------------------------
313int TExpr::emitBinary(TOp op, int a, int b, int rdepth, const Scalar& params)
314{
315 // addWeighted a*alpha + b*beta + gamma (params = {alpha, beta, gamma}): ONE fused kernel (two v_fma).
316 // Inputs are the same type T (cast to a common type if not). The kernel outputs T/f32 (small ints,
317 // f16/bf16, f32) or f64 directly; for any other requested rdepth it computes in the work type W and a
318 // final cast narrows it.
319 if (op == OP_ADDW)
320 {
321 int Tt = arginfo[a].depth;
322 if (arginfo[a].depth != arginfo[b].depth)
323 {
324 Tt = promoteArith(arginfo[a].depth, arginfo[b].depth);
325 a = maybeAddCast(a, Tt); b = maybeAddCast(b, Tt);
326 }
327 if (rdepth == EW_DEPTH_NONE) rdepth = Tt; // default dtype = input depth
328 TKernel k = getElemwiseFunc(OP_ADDW, Tt, Tt, EW_DEPTH_NONE, rdepth);
329 int outD = rdepth;
330 if (!k.fptr) // no direct T->rdepth kernel: compute in W, cast
331 {
332 outD = (Tt==CV_32U || Tt==CV_32S || Tt==CV_64U || Tt==CV_64S || Tt==CV_64F || rdepth==CV_64F)
333 ? CV_64F : CV_32F;
334 k = getElemwiseFunc(OP_ADDW, Tt, Tt, EW_DEPTH_NONE, outD);
335 }
336 const int out = addTemp(outD);
337 addInsn(OP_ADDW, a, b, 0, out, k, Scalar(params[0], params[1], params[2]));
338 if (outD == rdepth) return out;
339 const int out2 = addTemp(rdepth);
340 addInsn(OP_CAST, out, 0, 0, out2);
341 return out2;
342 }
343
344 const int nd0 = isFlexConst(*this, a) ? EW_DEPTH_NONE : arginfo[a].depth;
345 const int nd1 = isFlexConst(*this, b) ? EW_DEPTH_NONE : arginfo[b].depth;
346 const ElemwiseCategory cat = opCategory(op);
347
348 // result depth (auto unless forced): compare -> mask (u8); everything else -> promoteArith.
349 int result = rdepth;
350 if (result == EW_DEPTH_NONE)
351 {
352 result = (cat == CAT_COMPARE) ? CV_8U
353 : (cat == CAT_BITWISE) ? promoteBitwise(nd0, nd1) // no numeric promotion for bit ops
354 : promoteArith(nd0, nd1);
355 if (result == EW_DEPTH_NONE) result = CV_32F; // const (op) const
356 }
357
358 // common type over the CONCRETE operands (a flexible const does not force it).
359 int base;
360 if (nd0 == EW_DEPTH_NONE && nd1 == EW_DEPTH_NONE) base = result;
361 else if (nd0 == EW_DEPTH_NONE) base = nd1;
362 else if (nd1 == EW_DEPTH_NONE) base = nd0;
363 else base = promote2(nd0, nd1);
364
365 // COMPARE of an INTEGER array against a threshold that doesn't fit that type as-is (fractional,
366 // out-of-range, or EQ/NE of a non-representable value): emit a SINGLE NATIVE integer compare instead
367 // of widening both sides to f64. Per channel the relation is either a REAL boundary (a>=B / a<=B) or
368 // a CONSTANT (always-false / always-true). Boundaries: a>t==a>=floor(t)+1; a>=t==a>=ceil(t);
369 // a<t==a<=ceil(t)-1; a<=t==a<=floor(t); a==(non-rep) never true; a!=(non-rep) always true. One family
370 // op (GE for GT/GE, LE for LT/LE, else EQ/NE) runs with a placeholder threshold on const channels;
371 // the per-channel fix-up result = (rawmask & M) | V is FOLDED into the compare kernel via its flags
372 // (M=255 real / 0 const, V=255 always-true / 0 else) - no extra pass, no separate patch kernel.
373 if (cat == CAT_COMPARE && nd0 != EW_DEPTH_NONE && !isFloatDepth(nd0) &&
374 isFlexConst(*this, b) && !constFits(*this, b, base))
375 {
376 double lo, hi; intRange(nd0, lo, hi);
377 if (hi > lo) // hi<=lo => a depth intRange doesn't model (CV_Bool ...): leave to the f64 path
378 {
379 enum { REAL, CFALSE, CTRUE };
380 AutoBuffer<double, MAX_LOCAL_CN> tv; int cn = constDoubles(*this, b, tv);
381 CV_Assert(cn <= 4); // a CONST is <= 4 channels (addConst) - fixed arrays, no heap
382 int kind[4] = {}; double bound[4] = {}; // {} for -Wmaybe-uninitialized only
383 // (filled for all cn used below)
384 const TOp fam = (op == OP_CMP_GT || op == OP_CMP_GE) ? OP_CMP_GE
385 : (op == OP_CMP_LT || op == OP_CMP_LE) ? OP_CMP_LE : op; // EQ/NE unchanged
386 for (int c = 0; c < cn; c++)
387 {
388 const double t = tv[c]; int k; double B = t;
389 switch (op)
390 {
391 case OP_CMP_GT: B = std::floor(t)+1; k = B > hi ? CFALSE : B <= lo ? CTRUE : REAL; break;
392 case OP_CMP_GE: B = std::ceil(t); k = B > hi ? CFALSE : B <= lo ? CTRUE : REAL; break;
393 case OP_CMP_LT: B = std::ceil(t)-1; k = B < lo ? CFALSE : B >= hi ? CTRUE : REAL; break;
394 case OP_CMP_LE: B = std::floor(t); k = B < lo ? CFALSE : B >= hi ? CTRUE : REAL; break;
395 case OP_CMP_EQ: k = depthRepresents(t, nd0) ? REAL : CFALSE; break; // a==non-rep -> false
396 default: k = depthRepresents(t, nd0) ? REAL : CTRUE; break; // NE: a!=non-rep -> true
397 }
398 kind[c] = k; bound[c] = B;
399 }
400 // If every channel lands on ONE op (single-channel, all-real, all-false, all-true, or
401 // real+true for a GE/LE family) the compare ALONE yields the result - a plain native compare,
402 // no fix-up. REAL -> fam(bound); forced-TRUE -> the family's always-true form (GE lo / LE hi);
403 // forced-FALSE -> "a < lo".
404 auto ucOp = [&](int c){ return kind[c]==REAL ? fam : kind[c]==CFALSE ? OP_CMP_LT
405 : (fam == OP_CMP_LE ? OP_CMP_LE : OP_CMP_GE); };
406 auto ucThr = [&](int c){ return kind[c]==REAL ? bound[c]
407 : (kind[c]==CTRUE && fam==OP_CMP_LE) ? hi : lo; };
408 const TOp u0 = ucOp(0);
409 bool uniform = true;
410 for (int c = 1; c < cn && uniform; c++) uniform = (ucOp(c) == u0);
411 if (uniform)
412 {
413 Scalar s; for (int c = 0; c < cn; c++) s[c] = ucThr(c);
414 b = addConst(EW_DEPTH_NONE, s, cn); op = u0; // rewrite threshold + op -> fall through
415 }
416 else
417 {
418 // genuine per-channel op split (only a multi-channel scalar can cause it -> the executor
419 // lays it out as a short-row tile). One family compare with a placeholder threshold on
420 // const channels; the per-channel fix-up (rawmask & M) | V is FOLDED into the kernel via
421 // its flags (M=255 real / 0 const, V=255 always-true / 0 else) - no extra pass.
422 Scalar ts; int patchFlags = 0;
423 for (int c = 0; c < cn; c++)
424 {
425 ts[c] = kind[c]==REAL ? bound[c] : lo; // in-range placeholder for a const channel
426 const int mbits = kind[c]==REAL ? 3 : 0; // M: real -> 255 (keep), const -> 0
427 const int vbits = kind[c]==CTRUE ? 3 : 0; // V: always-true -> 255, else 0
428 patchFlags |= (mbits | (vbits << 2)) << (EW_CMP_PATCH_SHIFT + c*4);
429 }
430 TKernel kern = getElemwiseFunc(fam, base, base, EW_DEPTH_NONE, result);
431 kern.flags |= EW_CMP_PATCH | patchFlags;
432 const int out = addTemp(result);
433 addInsn(fam, a, addConst(base, ts, cn), 0, out, kern);
434 return out;
435 }
436 }
437 }

439 // compute depth + wide fallback per family:
440 // MIN/MAX, AND/OR/XOR : T x T -> T, never widen (depth = result)
441 // ABSDIFF : T x T -> unsigned same width for signed ints (8s->8u, ...)
442 // COMPARE : common type -> mask (depth = base)
443 // MUL/DIV : float work (f64 if wide) (matches cv::multiply/divide)
444 // POW : float work
445 // ADD/SUB : common type, signed wide fallback
446 int depth, wide;
447 bool divGuard = false;
448 switch (op)
449 {
450 case OP_MIN: case OP_MAX:
451 case OP_AND: case OP_OR: case OP_XOR:
452 depth = result; wide = result; break;
453 case OP_ABSDIFF:
454 {
455 // operands meet at the integer-preserving common type; |a-b| of a signed integer needs the
456 // UNSIGNED type of the same width (8s->8u, ...), which is both the result and the safe type.
457 int common = (nd0 == EW_DEPTH_NONE) ? nd1
458 : (nd1 == EW_DEPTH_NONE) ? nd0 : promoteArith(nd0, nd1);
459 if (common == EW_DEPTH_NONE) common = result; // both operands flexible consts
460 depth = common;
461 wide = absdiffResultDepth(common);
462 if (rdepth == EW_DEPTH_NONE) result = wide; // auto result: unsigned same width
463 break;
464 }
465 case OP_CMP_EQ: case OP_CMP_NE: case OP_CMP_LT:
466 case OP_CMP_LE: case OP_CMP_GT: case OP_CMP_GE:
467 // compare in the common operand type; if a flexible-const threshold does not FIT that type
468 // (out of range, e.g. u8 > -10, or fractional, e.g. u8 > 2.5) fall back to f64 so it is NOT
469 // saturated into the operand type (which would move the boundary). f64 is exact for every
470 // integer array up to 32-bit; only a 64-bit-int array vs an out-of-range threshold stays
471 // approximate (a pathological case). array-vs-array never has a const, so it stays integer.
472 depth = base; wide = CV_64F; break;
473 case OP_MUL: case OP_DIV:
474 {
475 depth = base;
476 const bool w = (depth==CV_32U || depth==CV_32S || depth==CV_64U ||
477 depth==CV_64S || depth==CV_64F);
478 wide = w ? CV_64F : CV_32F;
479 if (op == OP_DIV)
480 {
481 // guard b==0 -> 0 when BOTH operands are integer (matches cv::divide); a flexible const
482 // counts as integer iff its value is integral. promote2 can float two wide ints, so this
483 // must read the ORIGINAL operands, not `depth`.
484 auto intOperand = [&](int s, int nd) {
485 if (nd != EW_DEPTH_NONE) return !isFloatDepth(nd);
486 AutoBuffer<double, MAX_LOCAL_CN> v; constDoubles(*this, s, v);
487 return v[0] == std::floor(v[0]); };
488 divGuard = intOperand(a, nd0) && intOperand(b, nd1);
489 }
490 break;
491 }
492 case OP_POW: case OP_HYPOT: case OP_ATAN2:
493 depth = (base == CV_64F) ? CV_64F : CV_32F; wide = depth; break;
494 default: // OP_ADD, OP_SUB
495 depth = base; wide = safeWide(depth); break;
496 }
497
498 // a flexible const that can't be represented exactly at `depth` (e.g. 2.5 in a u8 op) forces the
499 // wide working path; a NON-INTEGRAL const additionally forces a FLOAT compute (an integer wide
500 // would quantize the fraction - e.g. add(u8, 1.7) must not round 1.7 to 2).
501 auto fracConst = [&](int s) {
502 if (!isFlexConst(*this, s)) return false;
503 AutoBuffer<double, MAX_LOCAL_CN> v; int n = constDoubles(*this, s, v);
504 for (int ch = 0; ch < n; ch++)
505 if (v[ch] != std::floor(v[ch])) return true;
506 return false;
507 };
508 // BITWISE never widens or floats: a bit-pattern op keeps the array's own integer type, and an
509 // out-of-range/fractional scalar is just saturate/round-cast into it (matching scalarToRawData /
510 // cv::bitwise's classic convertAndUnrollScalar). Skipping the widening below leaves depth == result.
511 if (cat != CAT_BITWISE && (!constFits(*this, a, depth) || !constFits(*this, b, depth)))
512 {
513 depth = wide;
514 const bool frac = fracConst(a) || fracConst(b);
515 // MIN/MAX just SELECT an operand, so a fractional scalar threshold is round-cast into the
516 // array type (min(u8, 3.7) == min(u8, 4)) - matching the classic cv::min/max - instead of
517 // promoting the whole op to float (add/sub/absdiff DO need the float path to keep the fraction).
518 if (frac && !isFloatDepth(depth) && op != OP_MIN && op != OP_MAX)
519 depth = wide = (result == CV_64F || base == CV_64F) ? CV_64F : CV_32F;
520 // mul/div carry a scale-like float const at full f64 precision (cv::multiply/divide compute in
521 // f64), so don't settle for f32 - e.g. 110 * 147.2863... must round to 16201, not 16202.
522 if (frac && (op == OP_MUL || op == OP_DIV) && depth == CV_32F)
523 depth = wide = CV_64F;
524 }
525
526 int s0 = isFlexConst(*this, a) ? typedConstFrom(a, depth) : a;
527 int s1 = isFlexConst(*this, b) ? typedConstFrom(b, depth) : b;
528 int c0 = maybeAddCast(s0, depth), c1 = maybeAddCast(s1, depth);
529
530 // emit `depth -> result` directly when a kernel exists, else compute in `wide` and cast down.
531 // OP_DIV resolves via getDivFunc (carrying the /0 guard); every other op via getElemwiseFunc.
532 auto resolve = [&](int rd) {
533 return op == OP_DIV ? getDivFunc(depth, rd, divGuard)
534 : getElemwiseFunc(op, depth, depth, EW_DEPTH_NONE, rd);
535 };
536 TKernel k = resolve(result);
537 if (k.fptr)
538 {
539 int res = addTemp(result);
540 addInsn(op, c0, c1, 0, res, k, params);
541 return res;
542 }
543 k = resolve(wide);
544 CV_Assert(k.fptr && "ew: no kernel for this op/type combination");
545 int w = addTemp(wide);
546 addInsn(op, c0, c1, 0, w, k, params);
547 return maybeAddCast(w, result);
548}

550// ---------------------------------------------------------------------------
551// TExpr::emitUnary(): type policy for a 1-input op. MATH ops (sqrt/exp/...) compute in the float
552// domain; NEG/ABS keep the operand type; NOT stays integer. (OP_CAST is not routed here - an explicit
553// cast is just maybeAddCast / a typed addConst at the call site.)
554// ---------------------------------------------------------------------------
555int TExpr::emitUnary(TOp op, int a, int rdepth, const Scalar& params)
556{
557 const int nd = isFlexConst(*this, a) ? EW_DEPTH_NONE : arginfo[a].depth;
558 const ElemwiseCategory cat = opCategory(op);
559
560 // NEG and ABS have no kernels of their own - they are compositions over the binary family
561 // with a zero constant (flexible, so emitBinary types it as the operand's own type).
562 if (op == OP_NEG)
563 return emitBinary(OP_SUB, addConst(EW_DEPTH_NONE, Scalar(0.), 1), a, rdepth, params);
564 if (op == OP_ABS)
565 {
566 // peephole: abs(x - y) -> absdiff(x, y), ALWAYS. Strictly speaking the two differ on
567 // integers - the literal subtract saturates first (u8: max(x-y, 0); signed: clipped
568 // difference), absdiff computes the true |x - y| - but whoever writes abs(a - b) MEANS
569 // absdiff; the saturation artifacts are never the desired result. So we deliberately
570 // "don't notice" the difference and hand out the useful semantics. The sub is necessarily
571 // the last instruction and its result the last temp (abs is emitted right after its
572 // argument) - retire both, the moveToOutput manoeuvre.
573 if (!prog.empty() && arginfo[a].kind == TEMP &&
574 prog.back().op == OP_SUB && prog.back().result == a &&
575 arginfo[a].index == ntemps - 1)
576 {
577 const int x = prog.back().arg0, y = prog.back().arg1;
578 prog.pop_back();
579 arginfo[a].kind = NONE;
580 ntemps--;
581 return emitBinary(OP_ABSDIFF, x, y, rdepth, params);
582 }
583 // abs IS absdiff(a, 0), including the auto result type: a signed |a| lands in the
584 // UNSIGNED type of the same width (|-128| = 128 fits u8 exactly; pinning the result to
585 // the signed operand type would saturate it to 127). Fully uniform with the peephole.
586 return emitBinary(OP_ABSDIFF, a, addConst(EW_DEPTH_NONE, Scalar(0.), 1), rdepth, params);
587 }
588
589 int result = rdepth;
590 if (result == EW_DEPTH_NONE)
591 {
592 switch (cat)
593 {
594 // math is T -> T over the float depths (f16/bf16/f32/f64 kernels exist natively);
595 // an integer input computes - and lands - in the float domain
596 case CAT_MATH: result = isFloatDepth(nd) ? nd : CV_32F; break;
597 case CAT_BITWISE: result = (nd == EW_DEPTH_NONE) ? CV_32S : nd; break; // NOT
598 default: result = (nd == EW_DEPTH_NONE) ? CV_32F : nd; break; // NEG, ABS
599 }
600 }
601
602 int depth, wide;
603 switch (cat)
604 {
605 case CAT_MATH:
606 // T == result over a float depth => the native kernel (incl. the f16/bf16 in-kernel f32
607 // hub - no materialized f32 temps). Everything else computes in f32/f64 and casts.
608 depth = (isFloatDepth(nd) && nd == result) ? nd
609 : (nd == CV_64F || result == CV_64F) ? CV_64F : CV_32F;
610 wide = depth;
611 break;
612 case CAT_BITWISE: depth = result; wide = result; break;
613 default: depth = (nd == EW_DEPTH_NONE) ? result : nd; wide = result; break; // NEG/ABS
614 }
615
616 if (!constFits(*this, a, depth)) depth = (cat == CAT_MATH) ? depth : CV_32F;
617 int s0 = isFlexConst(*this, a) ? typedConstFrom(a, depth) : a;
618 int c0 = maybeAddCast(s0, depth);
619
620 TKernel k = getElemwiseFunc(op, depth, EW_DEPTH_NONE, EW_DEPTH_NONE, result);
621 if (k.fptr)
622 {
623 int res = addTemp(result);
624 addInsn(op, c0, 0, 0, res, k, params);
625 return res;
626 }
627 k = getElemwiseFunc(op, depth, EW_DEPTH_NONE, EW_DEPTH_NONE, wide);
628 CV_Assert(k.fptr && "ew: no kernel for this op/type combination");
629 int w = addTemp(wide);
630 addInsn(op, c0, 0, 0, w, k, params);
631 return maybeAddCast(w, result);
632}

634// ---------------------------------------------------------------------------
635// TExpr::emitTernary(): clamp(x,lo,hi) and select(mask,x,y). clamp brings all three data operands to
636// a common type and emits there; select keeps arg0 (the mask) untouched and only unifies the two
637// branches. (Both are minimal: not yet test-covered, no kernels wired up.)
638// ---------------------------------------------------------------------------
639int TExpr::emitTernary(TOp op, int a, int b, int c, int rdepth)
640{
641 auto dep = [&](int s) { return isFlexConst(*this, s) ? EW_DEPTH_NONE : arginfo[s].depth; };
642
643 if (op == OP_SELECT) // select(mask=a, x=b, y=c)
644 {
645 // the kernel consumes a 1-byte mask as-is (u8/s8/bool, mask != 0 semantics). Any other
646 // mask type - a wider array or a literal - is normalized by an explicit `mask != 0`
647 // compare (u8 result), NOT by a value cast (which would saturate/round the values).
648 if (isFlexConst(*this, a) || CV_ELEM_SIZE1(arginfo[a].depth) != 1)
649 a = emitBinary(OP_CMP_NE, a, addConst(EW_DEPTH_NONE, Scalar(0.), 1), CV_8U, Scalar());
650 int result = rdepth != EW_DEPTH_NONE ? rdepth : promoteArith(dep(b), dep(c));
651 if (result == EW_DEPTH_NONE) result = CV_32F;
652 // a literal branch (select(m, x, 0)) is a flexible const: type it at `result` directly
653 // (a value conversion in constbuf), never through an OP_CAST of a depth-less slot
654 int sb = isFlexConst(*this, b) ? typedConstFrom(b, result) : maybeAddCast(b, result);
655 int sc = isFlexConst(*this, c) ? typedConstFrom(c, result) : maybeAddCast(c, result);
656 int res = addTemp(result);
657 addInsn(OP_SELECT, a, sb, sc, res);
658 return res;
659 }
660
661 // clamp(x,lo,hi): unify all three operands at `result` (compute == result, no wide fallback).
662 // The x operand dominates the auto type: clamp(u8_img, 10, 200) must stay u8, and literal
663 // bounds are flexible consts typed at `result` directly (never OP_CAST of a depth-less slot).
664 int result = rdepth != EW_DEPTH_NONE ? rdepth
665 : promoteArith(promoteArith(dep(a), dep(b)), dep(c));
666 if (result == EW_DEPTH_NONE) result = CV_32F;
667 auto typed = [&](int s) {
668 return isFlexConst(*this, s) ? typedConstFrom(s, result) : maybeAddCast(s, result);
669 };
670 int c0 = typed(a), c1 = typed(b), c2 = typed(c);
671 TKernel k = getElemwiseFunc(op, result, result, result, result);
672 CV_Assert(k.fptr && "ew: no kernel for this op/type combination");
673 int res = addTemp(result);
674 addInsn(op, c0, c1, c2, res, k);
675 return res;
676}

678// ---------------------------------------------------------------------------
679// TExpr::moveToOutput(): land `temp` in the existing slot `out`. Prefer redirecting `temp`'s single
680// producer to write `out` directly (no copy), dropping `temp` when it was the last-added slot so
681// ntemps stays minimal (keeps compile()'s no-temp fast exit for single-op programs). Otherwise copy.
682// ---------------------------------------------------------------------------
683int TExpr::moveToOutput(int temp, int out)
684{
685 const int ninsn = (int)prog.size();
686 int producer = -1;
687 bool usedAsArg = false;
688 for (int i = 0; i < ninsn; i++)
689 {
690 const TExpr::Insn& ins = prog[i];
691 if (ins.result == temp) producer = i;
692 if (ins.arg0 == temp || ins.arg1 == temp || ins.arg2 == temp) usedAsArg = true;
693 }
694 if (arginfo[temp].kind == TEMP && producer >= 0 && !usedAsArg &&
695 arginfo[temp].depth == arginfo[out].depth)
696 {
697 // MOVE semantics: redirect `temp`'s single producer to write `out` directly, then leave the
698 // source slot EMPTY - reclassify it to NONE. A NONE slot gets no physical buffer (compile's
699 // liveness only walks TEMP slots) and is skipped everywhere in exec, so no stale dead-TEMP slot
700 // remains (which, buffer or not, would still cost per-tile setup in the general path) - and no
701 // slot removal / index renumbering is needed. This keeps single-op programs at zero temps =>
702 // compile()'s no-temp early-out (zero heap traffic) still fires. Every caller moves the result
703 // it JUST emitted, so `temp` is always the last-created temp - decrement ntemps to retire its
704 // index and keep the temp indices dense (0..ntemps-1) for compile()'s liveness arrays.
705 CV_Assert(arginfo[temp].index == ntemps - 1);
706 prog[producer].result = out;
707 arginfo[temp].kind = NONE;
708 ntemps--;
709 return out;
710 }
711 addInsn(OP_CAST, temp, 0, 0, out); // same/different depth copy into `out`
712 return out;
713}

723// ---------------------------------------------------------------------------
724// TExpr::compile(): finalize the already-typed program. Bind every instruction's kernel (skipping
725// any whose pointer was pre-set, e.g. div's caller-known /0 policy) and pack the temps into a
726// minimal set of reusable physical buffers (liveness). All per-arg bookkeeping here is transient.
727// ---------------------------------------------------------------------------
728void TExpr::compile()
729{
730 const int ninsn = (int)prog.size();
731
732 // Kernels are bound eagerly by addInsn at build time, so there is no binding pass here - compile()
733 // only counts consts and packs temp buffers.
734
735 // Materialize consts: convert each live CONST's source values (still in `srcdepth` at constofs)
736 // to its resolved `depth`, appending the result to constbuf; constofs then points at the converted
737 // values and srcdepth becomes the resolved depth. Flexible literals (depth==NONE) are dead - the
738 // emit* layers replaced them with typed copies - and are skipped. nconsts sizes exec's per-const
739 // header buffer and gates the fast path. (getConvertFunc may reallocate constbuf, so snapshot the
740 // source bytes first.)
741 const int nslots = (int)arginfo.size();
742 nconsts = 0;
743 for (int s = 1; s < nslots; s++)
744 {
745 Arg& a = arginfo[s];
746 if (a.kind != CONST || a.depth == EW_DEPTH_NONE) continue;
747 const int cn = std::max(1, a.channels), sd = a.srcdepth, dd = a.depth;
748 const size_t sesz = CV_ELEM_SIZE1(sd), desz = CV_ELEM_SIZE1(dd);
749 AutoBuffer<uchar, 64> srcbytes((size_t)cn * sesz);
750 memcpy(srcbytes.data(), (const uchar*)(constbuf.data() + a.constofs), (size_t)cn * sesz);
751 const size_t ofs = appendConstBuf(constbuf, dd, nullptr, cn); // reserve converted region
752 uchar* dst = (uchar*)(constbuf.data() + ofs);
753 if (sd == dd) memcpy(dst, srcbytes.data(), (size_t)cn * desz);
754 else getConvertFunc(sd, dd)(srcbytes.data(), 0, nullptr, 0, dst, 0, Size(cn, 1), nullptr);
755 a.constofs = ofs; a.srcdepth = dd;
756 nconsts++;
757 }
758
759 // ---- liveness: pack temps into a minimal set of reusable physical buffers. A program with no
760 // temps (single-op add/sub/mul/...) needs none of this - early out with ZERO heap traffic,
761 // so building such a program (the dominant cv::add-style call) allocates nothing. The temp
762 // case uses stack-backed AutoBuffers (inline for typical small expressions). ----
763 nbuffers = 0;
764 bufEszPrefix.resize(1); bufEszPrefix[0] = 0;
765 capElems = INT_MAX; // no temps => exec runs each tile in one block, no scratch
766 if (ntemps == 0) return;
767
768 AutoBuffer<int, 32> tempOfSlot(nslots);
769 for (int s = 0; s < nslots; s++) tempOfSlot[s] = -1;
770 for (int s = 1; s < nslots; s++)
771 if (arginfo[s].kind == TEMP) tempOfSlot[s] = arginfo[s].index;
772
773 AutoBuffer<int, 16> lastUse(ntemps);
774 for (int t = 0; t < ntemps; t++) lastUse[t] = -1;
775 for (int i = 0; i < ninsn; i++)
776 {
777 const TExpr::Insn& ins = prog[i];
778 int as[3] = { ins.arg0, ins.arg1, ins.arg2 };
779 for (int k = 0; k < 3; k++)
780 if (tempOfSlot[as[k]] >= 0) lastUse[tempOfSlot[as[k]]] = i;
781 }
782
783 bufferOfTemp.resize(ntemps);
784 for (int t = 0; t < ntemps; t++) bufferOfTemp[t] = -1;
785 AutoBuffer<int, 16> freeBufs(ntemps);
786 int nfree = 0, nbuf = 0;
787 for (int i = 0; i < ninsn; i++)
788 {
789 const TExpr::Insn& ins = prog[i];
790 int rt = tempOfSlot[ins.result];
791 if (rt >= 0 && bufferOfTemp[rt] < 0)
792 bufferOfTemp[rt] = nfree > 0 ? freeBufs[--nfree] : nbuf++;
793 int as[3] = { ins.arg0, ins.arg1, ins.arg2 };
794 for (int k = 0; k < 3; k++)
795 {
796 int t = tempOfSlot[as[k]];
797 if (t >= 0 && lastUse[t] == i)
798 {
799 freeBufs[nfree++] = bufferOfTemp[t];
800 lastUse[t] = -1; // the same temp may be several args of this insn ("d*d") -
801 // release its buffer once, or freeBufs overflows/double-frees
802 }
803 }
804 }
805
806 nbuffers = nbuf;
807
808 // Byte layout of the physical temp buffers, per output element: prefix sums of each buffer's
809 // max element size. bufEszPrefix[b]*region = buffer b's byte offset in the scratch; [nbuffers] =
810 // total temp bytes/element. Precomputed here (depths are build-time) so exec()/its fast path
811 // never recompute it per call, and the fast path can size its scratch in O(1).
812 bufEszPrefix.resize(nbuffers + 1);
813 for (int b = 0; b <= nbuffers; b++) bufEszPrefix[b] = 0;
814 for (int s = 1; s < nslots; s++)
815 if (arginfo[s].kind == TEMP)
816 {
817 int b = bufferOfTemp[arginfo[s].index], e = (int)CV_ELEM_SIZE1(arginfo[s].depth);
818 if (e > bufEszPrefix[b + 1]) bufEszPrefix[b + 1] = e; // max elem size in buffer b
819 }
820 for (int b = 0; b < nbuffers; b++) bufEszPrefix[b + 1] += bufEszPrefix[b]; // -> prefix sums
821
822 // L1 fragment cap: # elements one ~16KB scratch fragment holds (exec fragments the strip so the
823 // intermediates stay hot). Pure function of the temp byte layout, hence computed here once.
824 const int totalEsz = bufEszPrefix[nbuffers];
825 capElems = totalEsz > 0 ? std::max(64, (16 * 1024) / totalEsz) : INT_MAX;
826}

831// Logical shape of a Mat with channels as the innermost dimension; steps in elemsize1 units.
832// Used here only to infer the broadcast RESULT shape (spatial + channels) for output allocation;
833// broadcastOp does its own channel-aware layout for the traversal itself.
834static void matLogical(const Mat& m, MatShape& shp, EwSteps& step, int& esz1)
835{
836 esz1 = (int)m.elemSize1();
837 int nd = m.dims, cn = m.channels();
838 shp.resize(nd + 1);
839 for (int i = 0; i < nd; i++)
840 {
841 shp[i] = m.size[i];
842 step[i] = m.step[i] / esz1;
843 }
844 shp[nd] = cn;
845 step[nd] = 1;
846}

848// numpy-style broadcast of several right-aligned shapes.
849static bool broadcastShape(const MatShape* shps, int K, MatShape& out)
850{
851 int nd = 0;
852 for (int k = 0; k < K; k++) nd = std::max(nd, shps[k].dims);
853 out.assign(nd, 1);
854 for (int k = 0; k < K; k++)
855 {
856 const MatShape& s = shps[k];
857 int off = nd - s.dims;
858 for (int i = 0; i < s.dims; i++)
859 {
860 int d = s[i], &o = out[off + i];
861 if (o == 1) o = d;
862 else if (d != 1 && d != o) return false;
863 }
864 }
865 return true;
866}

868// Rough per-element cost of an op for the parallel_for_ stripe-count hint, in units of ~1/4 cycle
869// per element of the VECTORIZED kernel (measured on the SIMD paths; e.g. the atan2/exp polynomials
870// run at ~1.5 cycles/element, not the ~30 a scalar-cost model would suggest - the old values
871// over-split the work into stripes too small to amortize the per-job dispatch overhead).
872// Unknown ops default to ~division. cv::expression will own this once it drives broadcastOp.
873static int opCost(TOp op)
874{
875 switch (op)
876 {
877 case OP_ADD: case OP_SUB: case OP_MUL: case OP_MIN: case OP_MAX:
878 case OP_ABSDIFF: case OP_AND: case OP_OR: case OP_XOR: case OP_NOT:
879 case OP_NEG: case OP_ABS: case OP_CAST: case OP_RELU: case OP_SELECT:
880 case OP_CMP_EQ: case OP_CMP_NE: case OP_CMP_LT:
881 case OP_CMP_LE: case OP_CMP_GT: case OP_CMP_GE: return 1;
882 case OP_DIV: case OP_SQRT: case OP_HYPOT: case OP_CONVERT_SCALE: return 3;
883 case OP_SIN: case OP_COS: case OP_TANH: case OP_ERF: case OP_ATAN2:
884 case OP_EXP: case OP_LOG: case OP_POW: return 6;
885 default: return 3;
886 }
887}

906// Human-readable dump of the program (slot table + instruction list). Const values are shown in f64.
907void TExpr::dump(std::ostream& os) const
908{
909 auto dn = [](int d) -> const char* {
910 switch (d) {
911 case EW_DEPTH_NONE: return "flex";
912 case CV_8U: return "u8"; case CV_8S: return "s8";
913 case CV_16U: return "u16"; case CV_16S: return "s16";
914 case CV_32U: return "u32"; case CV_32S: return "s32";
915 case CV_64U: return "u64"; case CV_64S: return "s64";
916 case CV_16F: return "f16"; case CV_16BF:return "bf16";
917 case CV_32F: return "f32"; case CV_64F: return "f64";
918 case CV_Bool:return "bool";
919 default: return "?"; }
920 };
921 auto kn = [](ArgKind k) -> const char* {
922 switch (k) { case NONE: return "none"; case INPUT: return "in"; case CONST: return "const";
923 case TEMP: return "temp"; case OUTPUT: return "out"; } return "?";
924 };
925 os << "TExpr: inputs=" << ninputs << " outputs=" << noutputs << " temps=" << ntemps
926 << " buffers=" << nbuffers << " consts=" << nconsts << " insns=" << (int)prog.size() << "\n";
927 os << " slots:\n";
928 for (int s = 0; s < (int)arginfo.size(); s++) {

942 }

944 }

955 if (!ins.kernel.fptr) os << " [UNBOUND]";
956 os << "\n";
957 }

1058struct EwBody {
1059 const TExpr::Insn* prog;
1060 // slot -> a NON-NEGATIVE index whose meaning is given by slotKind[s]:
1061 // TExpr::INPUT / TExpr::OUTPUT : index into tile.slices (the broadcast operand list; consts,
1062 // now 0-dim broadcast operands, are relabeled INPUT and live here)
1063 // TExpr::TEMP : physical temp-buffer id
1064 const int* slotMap;
1065 const signed char* slotKind; // [nslots] arginfo[s].kind, for resolving slotMap[s]
1066 const int* bufEszPrefix; // [nbuffers+1] prefix sums of temp elem sizes; [nbuffers] = total
1067 int ninsn, nslots, nbuffers, capElems; // capElems: L1 fragment element cap (0 if no temps)
1068};

1070void TExpr::outputShape(const Mat* const* inputs, MatShape& spatial, int& channels) const
1071{
1072 // Fast path: every real input shares one shape + channel count => the result IS inputs[0]'s
1073 // (consts add no dims/channels). Otherwise re-broadcast the logical (spatial + channel) shapes.
1074 const int rcn0 = ninputs >= 1 ? inputs[0]->channels() : 1;
1075 bool sameShape = ninputs >= 1;
1076 for (int i = 1; sameShape && i < ninputs; i++)
1077 {
1078 const Mat& a = *inputs[i];
1079 if (a.dims != inputs[0]->dims || a.channels() != rcn0) sameShape = false;
1080 else for (int d = 0; d < a.dims; d++) if (a.size[d] != inputs[0]->size[d]) { sameShape = false; break; }
1081 }
1082 if (sameShape) { spatial = inputs[0]->size; channels = rcn0; return; }
1083
1084 AutoBuffer<MatShape, 16> bshapes(std::max(ninputs, 1));
1085 MatShape shp; EwSteps step; int esz1;
1086 for (int i = 0; i < ninputs; i++) { matLogical(*inputs[i], shp, step, esz1); bshapes[i] = shp; }
1087 MatShape full;
1088 CV_Assert(broadcastShape(bshapes.data(), ninputs, full) && "ew: inputs not broadcast-compatible");
1089 const int ndFull = (int)full.size();
1090 channels = full[ndFull - 1];
1091 spatial = full; spatial.resize(ndFull - 1);
1092}

1094// Convenience overload: inputs as a contiguous Mat array -> build the pointer array + forward.
1095void TExpr::outputShape(const Mat* inputs, MatShape& spatial, int& channels) const
1096{
1097 AutoBuffer<const Mat*, 16> ptrs(std::max(ninputs, 1));
1098 for (int i = 0; i < ninputs; i++) ptrs[i] = &inputs[i];
1099 outputShape(ptrs.data(), spatial, channels);
1100}

1102// At namespace scope, NOT inside exec(): MSVC 2019 loses the constexpr-ness of function-local
1103// constants used as template arguments inside a lambda (C2975).
1104static constexpr int LOCAL_HDRS = 3;
1105static constexpr int LOCAL_CONSTS = 4;
1106static constexpr int LOCAL_OPS = 16;
1107
1108void TExpr::exec(const Mat* const* inputs, Mat* outputs)
1109{
1110 using BrTile = BroadcastOp::Tile;
1111 using BrSlice = BroadcastOp::Slice;
1112 const int nslots = (int)arginfo.size();
1113 CV_Assert(nslots >= 1 && arginfo[0].kind == TExpr::NONE);
1114
1115 // ---- cheap fast path: a const-free program over small, same-shape, continuous arrays. Channels
1116 // fold into one flat contiguous run, so the whole job is a flat (total x 1) strip - run the
1117 // program directly here and skip the per-call prep loop, broadcastOp's geometry, 2D tiling
1118 // and the parallel framework (all pure overhead at this size). Temps are allowed: their
1119 // byte layout (bufEszPrefix) is known from compile(), so we size an L1 scratch in O(1) and
1120 // walk the strip in L1-sized fragments (the intermediates stay hot). nconsts/nbuffers/
1121 // bufEszPrefix come from compile(); the rest is a quick check of the args. ----
1122 if (nconsts == 0 && ninputs >= 1)
1123 {
1124 constexpr size_t EW_FASTPATH_MAX = 1u << 17; // above this the tiled/parallel path wins
1125 const Mat& r = *inputs[0];
1126 const int rcn = r.channels();
1127 bool ok = r.isContinuous();
1128 for (int i = 1; ok && i < ninputs; i++)
1129 {
1130 const Mat& a = *inputs[i];
1131 ok = ok && a.isContinuous() && a.channels() == rcn && a.size == r.size;
1132 }
1133 // the flat strip writes the output(s) contiguously too, so it can't serve a non-contiguous
1134 // (cropped-ROI) destination - those fall to the general strided path below (an empty output,
1135 // which exec will allocate contiguous, counts as continuous here).
1136 for (int s = 1; ok && s < nslots; s++)
1137 if (arginfo[s].kind == OUTPUT) ok = ok && outputs[arginfo[s].index].isContinuous();
1138 const size_t total = ok ? r.total() * rcn : 0;
1139 if (ok && total <= EW_FASTPATH_MAX)
1140 {
1141 const MatShape rshape = r.size; // spatial dims (channels separate), as a MatShape
1142 for (int s = 1; s < nslots; s++)
1143 if (arginfo[s].kind == OUTPUT)
1144 outputs[arginfo[s].index].create(rshape, CV_MAKETYPE(arginfo[s].depth, rcn));
1145 const int ninsn = (int)prog.size();
1146
1147 if (nbuffers == 0)
1148 {
1149 // No temps (single-op add/sub/mul/...): one flat (total x 1) pass, operands point
1150 // straight at the Mat data - no scratch, no fragment loop. Referenced slots are INPUT/
1151 // OUTPUT (a moved-from NONE slot may exist but is never an instruction argument).
1152 auto ptrOf = [&](int s) -> void* {
1153 if (s <= 0) return nullptr;
1154 const Arg& ai = arginfo[s];
1155 if (ai.kind == INPUT) return (void*)inputs[ai.index]->data;
1156 if (ai.kind == OUTPUT) return (void*)outputs[ai.index].data;
1157 return nullptr; // NONE
1158 };
1159 const int w = (int)total;
1160 for (int n = 0; n < ninsn; n++)
1161 {
1162 const Insn& ins = prog[n];
1163 runInsn(ins, ptrOf(ins.arg0), 0, 1, ptrOf(ins.arg1), 0, 1,
1164 ptrOf(ins.arg2), 0, 1, ptrOf(ins.result), 0, w, 1);
1165 }
1166 return;
1167 }
1168
1169 // Temps present: walk the strip in L1-sized fragments so the intermediates stay hot.
1170 // Byte layout (bufEszPrefix) and the fragment cap (capElems) both come from compile();
1171 // scratch is totalEsz*wf0 (<= ~16KB).
1172 const int totalEsz = bufEszPrefix[nbuffers];
1173 const int wf0 = std::min((int)total, capElems);
1174 // Scratch for the temp buffers. AutoBuffer no longer value-inits its tail, so a fresh per-call
1175 // buffer is free (we only WRITE to it); the inline 16KB covers the L1-capped size, heap backs
1176 // the rare larger case.
1177 AutoBuffer<uchar, 16*1024 + 256> scratchBuf((size_t)totalEsz * (size_t)wf0);
1178 uchar* scratch = scratchBuf.data();
1179 for (int x0 = 0; x0 < (int)total; x0 += wf0)
1180 {
1181 const int wf = std::min(wf0, (int)total - x0);
1182 auto ptrOf = [&](int s) -> void* {
1183 if (s <= 0) return nullptr;
1184 const Arg& ai = arginfo[s];
1185 size_t esz = CV_ELEM_SIZE1(ai.depth);
1186 if (ai.kind == INPUT) return (uchar*)inputs[ai.index]->data + (size_t)x0 * esz;
1187 if (ai.kind == OUTPUT) return (uchar*)outputs[ai.index].data + (size_t)x0 * esz;
1188 if (ai.kind != TEMP) return nullptr; // NONE: moved-from
1189 const int b = bufferOfTemp.empty() ? ai.index : bufferOfTemp[ai.index]; // TEMP
1190 return scratch + (size_t)bufEszPrefix[b] * (size_t)wf0; // fragment-local
1191 };
1192 for (int n = 0; n < ninsn; n++)
1193 {
1194 const Insn& ins = prog[n];
1195 runInsn(ins, ptrOf(ins.arg0), 0, 1, ptrOf(ins.arg1), 0, 1,
1196 ptrOf(ins.arg2), 0, 1, ptrOf(ins.result), 0, wf, 1);
1197 }
1198 }
1199 return;
1200 }
1201 }
1202
1203 // Scalars (TExpr::CONST) never influence the result shape or channel count - the output geometry
1204 // comes from the real array inputs alone (a scalar broadcasts into whatever they produce).
1205 // A flexible CONST (depth == EW_DEPTH_NONE) is a leftover literal: the emit* layers materialize a
1206 // typed copy at each use, so the original is dead - skipped entirely (no header built for it).
1207
1208 // ---- 1. result shape (spatial dims + channel count), channels innermost ----
1209 // Fast path: every input shares one shape+channels => the result IS inputs[0]'s shape. Skips
1210 // the matLogical + broadcastShape rebuild/re-broadcast done only to size the output. Consts are
1211 // irrelevant here (they don't add dims/channels), so they never break the fast path.
1212 MatShape spatial;
1213 int rchannels;
1214 outputShape(inputs, spatial, rchannels);
1215
1216 // ---- 2. ONE pass over slots: allocate outputs, build the operand pointer list + slot map,
1217 // per-slot element size, and per-temp-buffer element size. No Mat copies (arr[] holds
1218 // pointers); only real inputs + outputs become broadcast operands. TExpr::CONST scalars are
1219 // NOT broadcast operands and get NO Mat header: each is materialized once into a typed
1220 // scratch (constStore) and exposed to the body as a fixed slice {ptr=&value, 0, 0} - a
1221 // scalar the kernels read by broadcast. slotMap[s] stays a plain non-negative index;
1222 // its meaning (arr / temp / const) is recovered from arginfo[s].kind (see EwBody).
1223 // In-place: an input shares its data buffer with an output. An output create() may realloc that
1224 // buffer while an input still needs the old contents, so we incref (header-copy) every input and
1225 // read through the copies. Cheap data-pointer aliasing test (nullptr data => never matches).
1226 bool inplace = false;
1227 for (int i = 0; i < ninputs && !inplace; i++)
1228 for (int j = 0; j < noutputs; j++)
1229 if (inputs[i]->data && inputs[i]->data == outputs[j].data) { inplace = true; break; }
1230 int nhdrs = inplace ? ninputs : 0;
1231
1232 AutoBuffer<const Mat*, LOCAL_OPS> arr(ninputs + noutputs + nconsts);
1233 AutoBuffer<Mat, LOCAL_HDRS> hdrs(std::max(nhdrs, 1)); // non-owning input headers (in-place incref)
1234 AutoBuffer<const Mat*, LOCAL_HDRS> inptr(std::max(nhdrs, 1)); // repointed input list (in-place only)
1235 AutoBuffer<Mat, LOCAL_CONSTS> constHdrBuf(std::max(nconsts, 1)); // 0-dim headers over constbuf
1236 AutoBuffer<int, LOCAL_OPS> slotMap(nslots);
1237 AutoBuffer<signed char, LOCAL_OPS> slotKind(nslots);
1238 int narr = 0, nc = 0;
1239 if (inplace) {
1240 // save (incref) inputs in the case of in-place operation
1241 // to protect them from premature deallocation
1242 for (int j = 0; j < ninputs; j++) { hdrs[j] = *inputs[j]; inptr[j] = &hdrs[j]; }
1243 inputs = inptr.data();
1244 }
1245 slotKind[0] = (signed char)TExpr::NONE;
1246 for (int s = 1; s < nslots; s++)
1247 {
1248 const TExpr::Arg& ai = arginfo[s];
1249 slotKind[s] = (signed char)ai.kind;
1250 if (ai.kind == TExpr::INPUT) {
1251 slotMap[s] = narr;
1252 arr[narr++] = inputs[ai.index];
1253 }
1254 else if (ai.kind == TExpr::OUTPUT) {
1255 outputs[ai.index].create(spatial, CV_MAKETYPE(ai.depth, rchannels));
1256 slotMap[s] = narr; arr[narr++] = &outputs[ai.index];
1257 }
1258 else if (ai.kind == TExpr::CONST && ai.depth == EW_DEPTH_NONE) {
1259 slotKind[s] = (signed char)TExpr::NONE; // dead flexible literal: never referenced
1260 slotMap[s] = 0;
1261 }
1262 else if (ai.kind == TExpr::CONST) {
1263 // a materialized const rides the broadcast machinery as a 0-dim, per-channel operand: a
1264 // multichannel const forces CH_DIM (per-channel scalars); a single value broadcasts
1265 // everywhere. compile() already converted its values (in constbuf) to ai.depth.
1266 const int c = std::max(1, ai.channels);
1267 constHdrBuf[nc] = Mat(MatShape::scalar(), CV_MAKETYPE(ai.depth, c),
1268 (void*)(constbuf.data() + ai.constofs));
1269 slotKind[s] = (signed char)TExpr::INPUT; // to the body it is just an array operand
1270 slotMap[s] = narr; arr[narr++] = &constHdrBuf[nc];
1271 nc++;
1272 }
1273 else if (ai.kind == TExpr::TEMP) {
1274 slotMap[s] = bufferOfTemp.empty() ? ai.index : bufferOfTemp[ai.index]; // physical buffer id
1275 }
1276 else { // TExpr::NONE: a moved-from temp (or dead literal) - inert, no operand, no buffer. Never
1277 // referenced by any instruction; do NOT touch bufferOfTemp (its retired index may be >= ntemps).
1278 slotMap[s] = 0;
1279 }
1280 }
1281
1282 // Temp-buffer byte layout (prefix sums, bufEszPrefix) and the L1 fragment cap (capElems) were
1283 // both precomputed by compile(); the body uses them directly (prefix[buf]*region = a buffer's
1284 // byte offset). capElems == INT_MAX when there are no temps => the body runs each tile in a
1285 // single block (no scratch).
1286
1287 // ---- 3. ONE pass over instructions: summed per-element cost (kernels were bound at compile()). ----
1288 const int ninsn = (int)prog.size();
1289 long long costPerElem = 0;
1290 for (int n = 0; n < ninsn; n++)
1291 costPerElem += opCost(prog[n].op);
1292
1293 // ---- 4. parallel work hint: total output scalars x summed per-element op cost / budget. ----
1294 long long otot = (long long)rchannels;
1295 for (int d = 0; d < (int)spatial.size(); d++) otot *= spatial[d];
1296 // ~4 stripes per thread is plenty of granularity for element-wise work; more only multiplies
1297 // the per-job dispatch overhead (notably on the macOS/GCD backend). The absolute ceiling is
1298 // 32 pieces, EXCEPT on machines with many (heterogeneous) cores: there anything coarser than
1299 // ~3 pieces per worker makes the slow (efficiency) cores equal-share bottlenecks, so the
1300 // ceiling grows as 3*nthreads instead.
1301 // getNumThreads() is clamped from below: some backends may report 0 (WINRT, plugins).
1302 const double T = (double)std::max(getNumThreads(), 1);
1303 const double nstripes = std::min(
1304 (double)otot * (double)std::max<long long>(costPerElem, 1) * (1./ (double)(1 << 18)),
1305 std::min(4.*T, std::max(32., 3.*T)));
1306
1307 EwBody body;
1308 body.prog = prog.data();
1309 body.ninsn = ninsn;
1310 body.slotMap = slotMap.data();
1311
1312 // ---- 5. drive: broadcastOp does geometry + 2D tiling + parallelism; the body runs the
1313 // frozen program on each tile (temps tile-local, re-pointed from the tile slices).
1314 // expandChannels=true => the body always sees single-channel data.
1315 //
1316 // All state the body needs is packed into one POD (EwBody) so the lambda captures a
1317 // SINGLE reference: the closure is then one pointer, fits std::function's small-buffer
1318 // and never heap-allocates. Inside, the hot fields are copied into locals so the
1319 // per-tile/per-insn loops read them from registers, not through the captured pointer. -
1320
1321 body.bufEszPrefix = bufEszPrefix.data();
1322 body.slotKind = slotKind.data();
1323 body.nslots = nslots;
1324 body.nbuffers = nbuffers;
1325 body.capElems = capElems;
1326
1327 broadcastOp(arr.data(), narr, [&](const BrTile& tile)
1328 {
1329 EwBody& bc = body;
1330 const TExpr::Insn* insns = bc.prog; // hot fields -> locals (registers/stack)
1331 const int* const smap = bc.slotMap;
1332 const signed char* const skind = bc.slotKind;
1333 const int nin = bc.ninsn, nsl = bc.nslots;
1334 const int w = tile.width, h = tile.height;
1335
1336 // Run the program over L1-sized 2D blocks so the intermediates stay in cache. ONE rule covers
1337 // every tile shape: keep the strip as WIDE as fits L1 (long inner loop => good SIMD + hits the
1338 // kernels' width-specific branches), then add as many rows as still fit (bw*bh <= capElems).
1339 // For the dominant 1D tile (w huge, h==1) this is a column strip; for a tall-thin tile
1340 // (w==channels, h huge - e.g. a masked op) it instead splits the LONG axis (height), so the
1341 // temp block stays contiguous and the kernels keep full width. With NO temps capElems==INT_MAX
1342 // => bw=w, bh=h: one block over the whole tile, region temp store is empty (zero bytes), and
1343 // each operand slot is re-pointed once straight at its tile slice. Each temp buffer occupies
1344 // [bufEszPrefix[buf]*region, ...) bytes in tstore; region = bw*bh.
1345 const int* const eszPrefix = bc.bufEszPrefix;
1346 const int bw = std::min(w, bc.capElems);
1347 const int bh = std::min(h, std::max(1, bc.capElems / std::max(1, bw)));
1348 const size_t region = alignSize((size_t)bw * bh, 8);
1349 AutoBuffer<uchar, 16*1024 + 256> tstoreBuf((size_t)eszPrefix[bc.nbuffers] * region); // inline (<= ~16KB)
1350 uchar* tstore = tstoreBuf.data();
1351
1352 AutoBuffer<BrSlice, LOCAL_OPS> args(nsl);
1353 for (int y0 = 0; y0 < h; y0 += bh)
1354 {
1355 const int hf = std::min(bh, h - y0);
1356 for (int x0 = 0; x0 < w; x0 += bw)
1357 {
1358 const int wf = std::min(bw, w - x0);
1359 for (int s = 1; s < nsl; s++)
1360 {

1400// ---------------------------------------------------------------------------
1401// Manual program builders (stand-ins for the future engine-backed cv::add etc.): they skip the
1402// source graph and emit the program directly through TExpr's addInput/addTemp/addOutput/addInsn.
1403// ---------------------------------------------------------------------------
1404
1405// Compose a binary op (ADD/SUB/MUL/DIV/MIN/MAX/ABSDIFF/CMP_*) for any (depth0, depth1, rdepth): just
1406// the operand/output plumbing around emitBinary, which owns the whole type policy (compute type, wide
1407// fallback, div's /0 guard, mul/div `scale` in params[0]). emitBinary returns the result slot;
1408// moveToOutput lands it in the output with no dead temp (so a single-op program keeps zero temps).
1409//
1410// maskDepth != EW_DEPTH_NONE adds a write-mask (input #2): the arithmetic result lands in a temp and a
1411// final select(mask, r, dst) -> dst overwrites only the masked subset of the (pre-existing) output,
1412// leaving the rest UNCHANGED (matching cv::add/... with a mask); the output slot rides as both the
1413// select's arg2 and its result (the kernel is alias-safe). select is always the LAST instruction.
1414// The mask is a single-channel 1-byte array (bool/u8/s8) the size of the output spatial shape; it
1415// rides the normal broadcast machinery, so nothing special is needed in the executor.
1416void makeBinaryArithProgram(TExpr& p, TOp op, int depth0, int depth1, int rdepth,
1417 int maskDepth, double scale)
1418{
1419 p.clear();
1420 if (rdepth < 0) // rdepth == -1 => auto (like cv::'s dtype=-1)
1421 {
1422 if (opCategory(op) == CAT_COMPARE)
1423 rdepth = CV_8U; // compare -> u8 mask
1424 else
1425 {
1426 rdepth = promoteArith(depth0, depth1);
1427 if (op == OP_ABSDIFF) rdepth = absdiffResultDepth(rdepth); // signed |a-b| -> unsigned same width
1428 }
1429 }
1430 const bool masked = maskDepth != EW_DEPTH_NONE;
1431 int sIn0 = p.addInput(depth0);
1432 int sIn1 = p.addInput(depth1);
1433 int sMask = masked ? p.addInput(maskDepth) : 0;
1434 int sOut = p.addOutput(rdepth);
1435
1436 int r = p.emitBinary(op, sIn0, sIn1, rdepth, Scalar(scale));
1437 if (masked)
1438 p.addInsn(OP_SELECT, sMask, r, sOut, sOut); // dst = mask ? r : dst
1439 else
1440 p.moveToOutput(r, sOut); // straight into the output, no dead temp
1441
1442 p.compile(); // pack temp buffers + count consts (kernels already bound by addInsn)
1443}
1444
1445// addWeighted(a, alpha, b, beta, gamma) = a*alpha + b*beta + gamma. One fused kernel (two v_fma) via
1446// emitBinary(OP_ADDW); a final cast is appended only when the requested rdepth isn't a type the kernel
1447// emits directly. alpha/beta/gamma travel in the instruction's params block, not as operands.
1448void makeAddWeightedProgram(TExpr& p, int depth0, int depth1, int rdepth,
1449 double alpha, double beta, double gamma)
1450{
1451 p.clear();
1452 int sA = p.addInput(depth0);
1453 int sB = p.addInput(depth1);
1454 int sOut = p.addOutput(rdepth);
1455 int r = p.emitBinary(OP_ADDW, sA, sB, rdepth, Scalar(alpha, beta, gamma));
1456 p.moveToOutput(r, sOut);
1457 p.compile(); // pack temp buffers + count consts (kernels already bound by addInsn)
1458}

1466enum TokType { T_NUM, T_INPUT, T_IDENT, T_OP, T_LPAREN, T_RPAREN, T_COMMA, T_SEMI, T_ASSIGN, T_END };
1467
1468struct Token
1469{
1470 TokType type = T_END;
1471 double num = 0;
1472 int input = 0;

1474};

1476// --- lexer -------------------------------------------------------------------------------
1477struct Lexer
1478{
1479 std::string_view s;
1480 size_t pos = 0;
1481
1482 explicit Lexer(std::string_view src) : s(src) {}
1483
1484 static bool isIdentStart(char c) { return std::isalpha((unsigned char)c) || c == '_'; }
1485 static bool isIdentChar(char c) { return std::isalnum((unsigned char)c) || c == '_'; }
1486
1487 Token next()
1488 {
1489 while (pos < s.size() && std::isspace((unsigned char)s[pos])) pos++;
1490 Token t;
1491 if (pos >= s.size()) { t.type = T_END; return t; }
1492
1493 char c = s[pos];
1494
1495 if (c == '{') // input placeholder {N}
1496 {
1497 pos++;
1498 size_t start = pos;
1499 while (pos < s.size() && s[pos] != '}') pos++;
1500 CV_Assert(pos < s.size() && "ew::expression: unterminated '{'");
1501 t.type = T_INPUT;
1502 t.input = std::atoi(std::string(s.substr(start, pos - start)).c_str());
1503 pos++; // consume '}'
1504 return t;
1505 }
1506 if (std::isdigit((unsigned char)c) || (c == '.' && pos + 1 < s.size() &&
1507 std::isdigit((unsigned char)s[pos + 1])))
1508 {
1509 char* end = nullptr;
1510 std::string num(s.substr(pos));
1511 t.type = T_NUM;
1512 t.num = std::strtod(num.c_str(), &end);
1513 pos += (size_t)(end - num.c_str());
1514 return t;
1515 }
1516 if (isIdentStart(c))
1517 {
1518 size_t start = pos;
1519 while (pos < s.size() && isIdentChar(s[pos])) pos++;
1520 t.type = T_IDENT;
1521 t.text.assign(s.substr(start, pos - start));
1522 return t;
1523 }
1524 switch (c)
1525 {
1526 case '(': pos++; t.type = T_LPAREN; return t;
1527 case ')': pos++; t.type = T_RPAREN; return t;
1528 case ',': pos++; t.type = T_COMMA; return t;
1529 case ';': pos++; t.type = T_SEMI; return t;
1530 }
1531 // multi/!single-char operators
1532 auto two = [&](const char* op) {
1533 return pos + 1 < s.size() && s[pos] == op[0] && s[pos + 1] == op[1];
1534 };
1535 t.type = T_OP;
1536 if (two("<=") || two(">=") || two("==") || two("!=") || two("**"))
1537 { t.text.assign(s.substr(pos, 2)); pos += 2; return t; }
1538 if (c == '=') { pos++; t.type = T_ASSIGN; return t; }
1539 CV_Assert(std::strchr("+-*/<>&|^!?:", c) && "ew::expression: unexpected character");
1540 t.text.assign(1, c); pos++;
1541 return t;
1542 }
1543};

1546static int binPrec(const std::string& op)
1547{

1549 if (op == "*" || op == "/") return 7;
1550 if (op == "+" || op == "-") return 6;
1551 if (op == "<" || op == "<=" || op == ">" || op == ">=") return 5;
1552 if (op == "==" || op == "!=") return 4;
1553 if (op == "&") return 3;
1554 if (op == "^") return 2;
1555 if (op == "|") return 1;

1598{
1599 struct E { const char* n; TOp op; };
1600 static const E unary[] = { {"abs",OP_ABS},{"sqrt",OP_SQRT},{"exp",OP_EXP},{"log",OP_LOG},
1601 {"sin",OP_SIN},{"cos",OP_COS},{"tanh",OP_TANH},{"erf",OP_ERF},

1603 static const E binary[] = { {"max",OP_MAX},{"min",OP_MIN},{"pow",OP_POW},{"absdiff",OP_ABSDIFF},
1604 {"hypot",OP_HYPOT},{"mag",OP_HYPOT}, // mag = the cv::magnitude-flavored alias
1605 {"atan2",OP_ATAN2} };
1606 static const E tern[] = { {"clamp",OP_CLAMP},{"select",OP_SELECT} };
1607 for (const E& e : unary) if (name == e.n) { arity = 1; return e.op; }
1608 for (const E& e : binary) if (name == e.n) { arity = 2; return e.op; }
1609 for (const E& e : tern) if (name == e.n) { arity = 3; return e.op; }
1610 arity = 0; return OP_NOP;

1613// --- parser ------------------------------------------------------------------------------
1614// Builds the program DIRECTLY into the TExpr (no intermediate graph), exactly like the hand
1615// builders: each parse step calls e.emitUnary()/emitBinary()/addConst()/output() and returns the arg
1616// SLOT holding its value. Input depths are known up front (from the input Mats), so every operand is
1617// typed as it is parsed - the emit* layers infer result depths and insert casts on the spot.
1618struct Parser
1619{
1620 Lexer lex;
1621 Token cur;
1622 TExpr& e;
1623 const int* inputSlot; // slot id of each external input (precreated)
1624 int ninputs;
1625 std::map<std::string, int> env; // named temporaries -> slot
1626
1627 Parser(std::string_view src, TExpr& expr, const int* islot, int nin)
1628 : lex(src), e(expr), inputSlot(islot), ninputs(nin) { cur = lex.next(); }
1629
1630 void advance() { cur = lex.next(); }
1631 bool isOp(const char* op) const { return cur.type == T_OP && cur.text == op; }
1632
1633 void expect(TokType t, const char* what)
1634 {
1635 CV_Assert(cur.type == t && what);
1636 advance();
1637 }
1638
1639 // A flexible CONST slot holds a literal whose depth the emit* layers pick per use.
1640 bool isFlexConst(int slot) const
1641 {
1642 return e.arginfo[slot].kind == TExpr::CONST && e.arginfo[slot].depth == EW_DEPTH_NONE;
1643 }
1644
1645 int parsePrimary()
1646 {
1647 if (cur.type == T_NUM) { int s = e.addConst(EW_DEPTH_NONE, Scalar(cur.num), 1); advance(); return s; }
1648 if (cur.type == T_INPUT) { int idx = cur.input; advance();
1649 CV_Assert(idx >= 0 && idx < ninputs && "ew::expression: input index out of range");
1650 return inputSlot[idx]; }
1651 if (cur.type == T_LPAREN){ advance(); int x = parseTernary(); expect(T_RPAREN, "expected ')'"); return x; }
1652 if (cur.type == T_IDENT)
1653 {
1654 std::string name = cur.text; advance();
1655 if (cur.type != T_LPAREN) // variable reference
1656 {
1657 auto it = env.find(name);
1658 CV_Assert(it != env.end() && "ew::expression: undefined name");
1659 return it->second;
1660 }
1661 advance(); // consume '('
1662 // every function takes <= 3 args (clamp/select); a stack buffer avoids a std::vector
1663 // heap alloc. The {} is for gcc's -Wmaybe-uninitialized only.
1664 std::array<int, 8> args = {}; int nargs = 0;
1665 if (cur.type != T_RPAREN)
1666 {
1667 args[nargs++] = parseTernary();
1668 while (cur.type == T_COMMA) { advance();
1669 CV_Assert(nargs < (int)args.size() && "ew::expression: too many arguments");
1670 args[nargs++] = parseTernary(); }
1671 }
1672 expect(T_RPAREN, "expected ')'");
1673
1674 int td = typeDepth(name);
1675 if (td >= 0)
1676 {
1677 CV_Assert(nargs == 1); // type cast: just convert the operand
1678 int s = args[0];
1679 return isFlexConst(s) ? e.typedConstFrom(s, td)
1680 : e.maybeAddCast(s, td);
1681 }
1682
1683 int arity = 0; TOp op = fnOp(name, arity);
1684 CV_Assert(arity != 0 && "ew::expression: unknown function");
1685 CV_Assert(nargs == arity && "ew::expression: wrong number of arguments");
1686 if (arity == 1) return e.emitUnary(op, args[0]);
1687 if (arity == 2) return e.emitBinary(op, args[0], args[1]);
1688 return e.emitTernary(op, args[0], args[1], args[2]);
1689 }
1690 CV_Error(Error::StsParseError, "ew::expression: expected a primary expression");
1691 }
1692
1693 int parseUnary()
1694 {
1695 if (isOp("-") || isOp("!"))
1696 {
1697 std::string op = cur.text; advance();
1698 int operand = parseUnary();
1699 // constant-fold a leading sign so that "-1.5" does not need an OP_NEG kernel
1700 if (op == "-" && isFlexConst(operand))
1701 {
1702 AutoBuffer<double, MAX_LOCAL_CN> v; int cn = constDoubles(e, operand, v);
1703 Scalar neg; for (int i = 0; i < cn && i < 4; i++) neg[i] = -v[i];
1704 return e.addConst(EW_DEPTH_NONE, neg, std::min(cn, 4));
1705 }
1706 return e.emitUnary(op == "-" ? OP_NEG : OP_NOT, operand);
1707 }
1708 return parsePrimary();
1709 }
1710
1711 int parseExpr(int minPrec)
1712 {
1713 int left = parseUnary();
1714 while (cur.type == T_OP)
1715 {
1716 int p = binPrec(cur.text);
1717 if (p < minPrec) break;
1718 std::string op = cur.text; advance();
1719 // '**' is right-associative (a ** b ** c == a ** (b ** c)): recurse at the SAME
1720 // precedence so the right side swallows further '**'s; everything else at p+1.
1721 int right = parseExpr(op == "**" ? p : p + 1);
1722 left = e.emitBinary(binOp(op), left, right);
1723 }
1724 return left;
1725 }
1726
1727 // The ternary conditional cond ? a : b == select(cond, a, b). Lowest precedence (below all
1728 // arithmetic/compare/bitwise), right-associative: f1 ? a : f2 ? b : c groups as
1729 // f1 ? a : (f2 ? b : c). This is the general expression entry point.
1730 int parseTernary()
1731 {
1732 int cond = parseExpr(0);
1733 if (!isOp("?")) return cond;
1734 advance();
1735 int thenv = parseTernary();
1736 CV_Assert(isOp(":") && "ew::expression: expected ':' in a ?: conditional");
1737 advance();
1738 int elsev = parseTernary(); // right-associative chain
1739 return e.emitTernary(OP_SELECT, cond, thenv, elsev);
1740 }
1741
1742 // final result: a single expression, or a top-level tuple (e0, e1, ...).
1743 void parseResult()
1744 {
1745 if (cur.type == T_LPAREN)
1746 {
1747 // backtrack point: save the lexer AND the program length, since the trial parse below
1748 // emits instructions/slots that must be rolled back if this turns out NOT to be a tuple.
1749 Lexer save = lex; Token savedCur = cur;
1750 size_t nInsn = e.prog.size(), nArg = e.arginfo.size(); int nTemp = e.ntemps;
1751 advance();
1752 int e0 = parseTernary();
1753 if (cur.type == T_COMMA)
1754 {
1755 e.output(e0);
1756 while (cur.type == T_COMMA) { advance(); e.output(parseTernary()); }
1757 expect(T_RPAREN, "expected ')'");
1758 return;
1759 }
1760 lex = save; cur = savedCur; // not a tuple -> undo the trial parse
1761 e.prog.resize(nInsn); e.arginfo.resize(nArg); e.ntemps = nTemp;
1762 }
1763 e.output(parseTernary());
1764 }
1765
1766 void parse()
1767 {
1768 while (true)
1769 {
1770 if (cur.type == T_IDENT)
1771 {
1772 Lexer save = lex; Token savedCur = cur; // lexer-only backtrack (nothing emitted yet)
1773 std::string name = cur.text; advance();
1774 if (cur.type == T_ASSIGN)
1775 {
1776 advance();
1777 env[name] = parseTernary();
1778 expect(T_SEMI, "expected ';' after assignment");
1779 continue;
1780 }
1781 lex = save; cur = savedCur; // not an assignment
1782 }
1783 parseResult();
1784 if (cur.type == T_SEMI) advance();
1785 break;
1786 }
1787 CV_Assert(cur.type == T_END && "ew::expression: trailing tokens");
1788 CV_Assert(e.noutputs > 0 && "ew::expression: no result");
1789 }
