46template <typename T>
47size_t convertDepthToFloat(const cv::Mat& depth, const cv::Mat& mask, float scale, cv::Mat_<float> &u_mat, cv::Mat_<float> &v_mat, cv::Mat_<float> &z_mat)
48{
49 CV_Assert(depth.size == mask.size);
50
51 cv::Size depth_size = depth.size();
52
53 cv::Mat_<uchar> uchar_mask = mask;
54
55 if ((mask.depth() != CV_8S) && (mask.depth() != CV_8U) && (mask.depth() != CV_Bool))
56 mask.convertTo(uchar_mask, CV_8U);
57
58 u_mat = cv::Mat_<float>(depth_size.area(), 1);
59 v_mat = cv::Mat_<float>(depth_size.area(), 1);
60 z_mat = cv::Mat_<float>(depth_size.area(), 1);
61
62 // Raw data from the Kinect has int
63 size_t n_points = 0;
64
65 for (int v = 0; v < depth_size.height; v++)
66 {
67 uchar* r = uchar_mask.ptr<uchar>(v, 0);
68
69 for (int u = 0; u < depth_size.width; u++, ++r)
70 if (*r)
71 {
72 u_mat((int)n_points, 0) = (float)u;
73 v_mat((int)n_points, 0) = (float)v;
74 T depth_i = depth.at<T>(v, u);
75
76 if (cvIsNaN((float)depth_i) || (depth_i == std::numeric_limits<T>::min()) || (depth_i == std::numeric_limits<T>::max()))
77 z_mat((int)n_points, 0) = std::numeric_limits<float>::quiet_NaN();
78 else
79 z_mat((int)n_points, 0) = depth_i * scale;
80
81 ++n_points;
82 }
83 }
84
85 return n_points;
86}

92template <typename T>
93void convertDepthToFloat(const cv::Mat& depth, float scale, const cv::Mat &uv_mat, cv::Mat_<float> &z_mat)
94{
95 z_mat = cv::Mat_<float>(uv_mat.size());
96
97 // Raw data from the Kinect has int
98 float* z_mat_iter = reinterpret_cast<float*>(z_mat.data);
99
100 for (cv::Mat_<cv::Vec2f>::const_iterator uv_iter = uv_mat.begin<cv::Vec2f>(), uv_end = uv_mat.end<cv::Vec2f>();
101 uv_iter != uv_end; ++uv_iter, ++z_mat_iter)
102 {
103 T depth_i = depth.at < T >((int)(*uv_iter)[1], (int)(*uv_iter)[0]);
104
105 if (cvIsNaN((float)depth_i) || (depth_i == std::numeric_limits < T > ::min())
106 || (depth_i == std::numeric_limits < T > ::max()))
107 *z_mat_iter = std::numeric_limits<float>::quiet_NaN();
108 else
109 *z_mat_iter = depth_i * scale;
110 }
111}
