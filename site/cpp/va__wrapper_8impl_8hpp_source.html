48static std::shared_ptr<cv::plugin::impl::DynamicLib> loadLibVA()
49{
50 std::shared_ptr<cv::plugin::impl::DynamicLib> lib;
51 const std::string envPath = utils::getConfigurationParameterString("OPENCV_LIBVA_RUNTIME");
52 if (!envPath.empty())
53 {
54 lib = std::make_shared<cv::plugin::impl::DynamicLib>(envPath);
55 return lib;
56 }
57 static const char* const candidates[] = {
58 "libva.so",
59 "libva.so.2",
60 "libva.so.1",
61 };
62 for (int i = 0; i < 3; ++i)
63 {
64 lib = std::make_shared<cv::plugin::impl::DynamicLib>(candidates[i]);
65 if (lib->isLoaded())
66 break;
67 }
68 return lib;
69}
70static void init_libva()
71{
72 static bool initialized = false;
73 static auto library = loadLibVA();
74 if (!initialized)
75 {
76 if (!library || !library->isLoaded())
77 {
78 library.reset();
79 CV_Error(cv::Error::StsBadFunc, "OpenCV can't load VA library (libva)");
80 }
81 auto& lib = *library.get();
82#define VA_LOAD_SYMBOL(name) fn_ ## name = reinterpret_cast<FN_ ## name>(lib.getSymbol(#name)); \
83 if (!fn_ ## name) \
84 { \
85 library.reset(); \
86 initialized = true; \
87 CV_Error_(cv::Error::StsBadFunc, ("OpenCV can't load VA library (libva), missing symbol: %s", #name)); \
88 }
89
90 VA_LOAD_SYMBOL(vaDeriveImage);
91 VA_LOAD_SYMBOL(vaDestroyImage);
92 VA_LOAD_SYMBOL(vaMapBuffer);
93 VA_LOAD_SYMBOL(vaSyncSurface);
94 VA_LOAD_SYMBOL(vaUnmapBuffer);
95 VA_LOAD_SYMBOL(vaMaxNumImageFormats);
96 VA_LOAD_SYMBOL(vaQueryImageFormats);
97 VA_LOAD_SYMBOL(vaCreateImage);
98 VA_LOAD_SYMBOL(vaPutImage);
99 VA_LOAD_SYMBOL(vaGetImage);
100 initialized = true;
101 }
102 if (!library)
103 CV_Error(cv::Error::StsBadFunc, "OpenCV can't load/initialize VA library (libva)");
104}
