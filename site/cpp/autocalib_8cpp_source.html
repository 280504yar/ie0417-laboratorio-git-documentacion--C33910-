63void focalsFromHomography(const Mat& H, double &f0, double &f1, bool &f0_ok, bool &f1_ok)
64{
65 CV_Assert(H.type() == CV_64F && H.size() == Size(3, 3));
66
67 const double* h = H.ptr<double>();
68
69 double d1, d2; // Denominators
70 double v1, v2; // Focal squares value candidates
71
72 f1_ok = true;
73 d1 = h[6] * h[7];
74 d2 = (h[7] - h[6]) * (h[7] + h[6]);
75 v1 = -(h[0] * h[1] + h[3] * h[4]) / d1;
76 v2 = (h[0] * h[0] + h[3] * h[3] - h[1] * h[1] - h[4] * h[4]) / d2;
77 if (v1 < v2)
78 {
79 std::swap(v1, v2);
80 std::swap(d1, d2);
81 }
82 if (v1 > 0 && v2 > 0) f1 = std::sqrt(std::abs(d1) > std::abs(d2) ? v1 : v2);
83 else if (v1 > 0) f1 = std::sqrt(v1);
84 else f1_ok = false;
85
86 f0_ok = true;
87 d1 = h[0] * h[3] + h[1] * h[4];
88 d2 = h[0] * h[0] + h[1] * h[1] - h[3] * h[3] - h[4] * h[4];
89 v1 = -h[2] * h[5] / d1;
90 v2 = (h[5] * h[5] - h[2] * h[2]) / d2;
91 if (v1 < v2)
92 {
93 std::swap(v1, v2);
94 std::swap(d1, d2);
95 }
96 if (v1 > 0 && v2 > 0) f0 = std::sqrt(std::abs(d1) > std::abs(d2) ? v1 : v2);
97 else if (v1 > 0) f0 = std::sqrt(v1);
98 else f0_ok = false;
99}

102void estimateFocal(const std::vector<ImageFeatures> &features, const std::vector<MatchesInfo> &pairwise_matches,
103 std::vector<double> &focals)
104{
105 const int num_images = static_cast<int>(features.size());
106 focals.resize(num_images);
107
108 std::vector<double> all_focals;
109
110 for (int i = 0; i < num_images; ++i)
111 {
112 for (int j = 0; j < num_images; ++j)
113 {
114 const MatchesInfo &m = pairwise_matches[i*num_images + j];
115 if (m.H.empty())
116 continue;
117 double f0, f1;
118 bool f0ok, f1ok;
119 focalsFromHomography(m.H, f0, f1, f0ok, f1ok);
120 if (f0ok && f1ok)
121 all_focals.push_back(std::sqrt(f0 * f1));
122 }
123 }
124
125 if (static_cast<int>(all_focals.size()) >= num_images - 1)
126 {
127 double median;
128
129 std::sort(all_focals.begin(), all_focals.end());
130 if (all_focals.size() % 2 == 1)
131 median = all_focals[all_focals.size() / 2];
132 else
133 median = (all_focals[all_focals.size() / 2 - 1] + all_focals[all_focals.size() / 2]) * 0.5;
134
135 for (int i = 0; i < num_images; ++i)
136 focals[i] = median;
137 }
138 else
139 {
140 LOGLN("Can't estimate focal length, will use naive approach");
141 double focals_sum = 0;
142 for (int i = 0; i < num_images; ++i)
143 focals_sum += features[i].img_size.width + features[i].img_size.height;
144 for (int i = 0; i < num_images; ++i)
145 focals[i] = focals_sum / num_images;
146 }
147}

150bool calibrateRotatingCamera(const std::vector<Mat> &Hs, Mat &K)
151{
152 int m = static_cast<int>(Hs.size());
153 CV_Assert(m >= 1);
154
155 std::vector<Mat> Hs_(m);
156 for (int i = 0; i < m; ++i)
157 {
158 CV_Assert(Hs[i].size() == Size(3, 3) && Hs[i].type() == CV_64F);
159 Hs_[i] = Hs[i] / std::cbrt(determinant(Hs[i]));
160 }
161
162 const int idx_map[3][3] = {{0, 1, 2}, {1, 3, 4}, {2, 4, 5}};
163 Mat_<double> A(6*m, 6);
164 A.setTo(0);
165
166 int eq_idx = 0;
167 for (int k = 0; k < m; ++k)
168 {
169 Mat_<double> H(Hs_[k]);
170 for (int i = 0; i < 3; ++i)
171 {
172 for (int j = i; j < 3; ++j, ++eq_idx)
173 {
174 for (int l = 0; l < 3; ++l)
175 {
176 for (int s = 0; s < 3; ++s)
177 {
178 int idx = idx_map[l][s];
179 A(eq_idx, idx) += H(i,l) * H(j,s);
180 }
181 }
182 A(eq_idx, idx_map[i][j]) -= 1;
183 }
184 }
185 }
186
187 Mat_<double> wcoef;
188 SVD::solveZ(A, wcoef);
189
190 Mat_<double> W(3,3);
191 for (int i = 0; i < 3; ++i)
192 for (int j = i; j < 3; ++j)
193 W(i,j) = W(j,i) = wcoef(idx_map[i][j], 0) / wcoef(5,0);
194 if (!decomposeCholesky(W.ptr<double>(), W.step, 3))
195 return false;
196 W(0,1) = W(0,2) = W(1,2) = 0;
197 K = W.t();
198 return true;
199}
