33void reduceAllFloatParallel_(const Mat& src, Mat& dst, int reduce_type_int) {
34 const Reduce2Layer::ReduceType rt = (Reduce2Layer::ReduceType)reduce_type_int;
35 const float* p = src.ptr<const float>();
36 const size_t total = src.total();
37 const int nThreads = std::max(1, cv::getNumThreads());
38 const int stripes = std::max(1, std::min<int>(
39 (int)((total + 16383) / 16384), nThreads));
40
41 float init_f = 0.0f;
42 if (rt == Reduce2Layer::ReduceType::MAX) init_f = -FLT_MAX;
43 else if (rt == Reduce2Layer::ReduceType::MIN) init_f = FLT_MAX;
44 else if (rt == Reduce2Layer::ReduceType::PROD) init_f = 1.0f;
45
46 std::vector<float> partial_f(stripes, init_f);
47
48 parallel_for_(Range(0, stripes), [&](const Range& r) {
49 for (int s = r.start; s < r.end; s++) {
50 size_t start = (size_t)s * total / stripes;
51 size_t end = (size_t)(s + 1) * total / stripes;
52 const float* q = p + start;
53 size_t n = end - start, i = 0;
54#if (CV_SIMD || CV_SIMD_SCALABLE)
55 const int L = VTraits<v_float32>::vlanes();
56#endif
57 switch (rt) {
58 case Reduce2Layer::ReduceType::MAX: {
59 float acc = -FLT_MAX;
60#if (CV_SIMD || CV_SIMD_SCALABLE)
61 v_float32 va = vx_setall_f32(-FLT_MAX);
62 for (; i + L <= n; i += L) va = v_max(va, vx_load(q + i));
63 acc = v_reduce_max(va);
64#endif
65 for (; i < n; i++) acc = acc > q[i] ? acc : q[i];
66 partial_f[s] = acc;
67 break;
68 }
69 case Reduce2Layer::ReduceType::MIN: {
70 float acc = FLT_MAX;
71#if (CV_SIMD || CV_SIMD_SCALABLE)
72 v_float32 va = vx_setall_f32(FLT_MAX);
73 for (; i + L <= n; i += L) va = v_min(va, vx_load(q + i));
74 acc = v_reduce_min(va);
75#endif
76 for (; i < n; i++) acc = acc < q[i] ? acc : q[i];
77 partial_f[s] = acc;
78 break;
79 }
80 case Reduce2Layer::ReduceType::SUM:
81 case Reduce2Layer::ReduceType::MEAN:
82 case Reduce2Layer::ReduceType::LOG_SUM: {
83 float acc = 0.0f;
84#if (CV_SIMD || CV_SIMD_SCALABLE)
85 v_float32 va = vx_setzero_f32();
86 for (; i + L <= n; i += L) va = v_add(va, vx_load(q + i));
87 acc += v_reduce_sum(va);
88#endif
89 for (; i < n; i++) acc += q[i];
90 partial_f[s] = acc;
91 break;
92 }
93 case Reduce2Layer::ReduceType::L1: {
94 float acc = 0.0f;
95#if (CV_SIMD || CV_SIMD_SCALABLE)
96 v_float32 va = vx_setzero_f32();
97 for (; i + L <= n; i += L) va = v_add(va, v_abs(vx_load(q + i)));
98 acc += v_reduce_sum(va);
99#endif
100 for (; i < n; i++) acc += std::fabs(q[i]);
101 partial_f[s] = acc;
102 break;
103 }
104 case Reduce2Layer::ReduceType::L2:
105 case Reduce2Layer::ReduceType::SUM_SQUARE: {
106 float acc = 0.0f;
107#if (CV_SIMD || CV_SIMD_SCALABLE)
108 v_float32 va = vx_setzero_f32();
109 for (; i + L <= n; i += L) {
110 v_float32 x = vx_load(q + i);
111 va = v_add(va, v_mul(x, x));
112 }
113 acc += v_reduce_sum(va);
114#endif
115 for (; i < n; i++) { float x = q[i]; acc += x * x; }
116 partial_f[s] = acc;
117 break;
118 }
119 case Reduce2Layer::ReduceType::PROD: {
120 float acc = 1.0f;
121#if (CV_SIMD || CV_SIMD_SCALABLE)
122 v_float32 va = vx_setall_f32(1.0f);
123 for (; i + L <= n; i += L) va = v_mul(va, vx_load(q + i));
124 float buf[VTraits<v_float32>::max_nlanes];
125 v_store(buf, va);
126 for (int k = 0; k < L; k++) acc *= buf[k];
127#endif
128 for (; i < n; i++) acc *= q[i];
129 partial_f[s] = acc;
130 break;
131 }
132 default: CV_Error(Error::StsInternal, "reduceAllFloatParallel_: unhandled type");
133 }
134 }
135 });
136
137 float* out = dst.ptr<float>();
138 switch (rt) {
139 case Reduce2Layer::ReduceType::MAX: {
140 float m = -FLT_MAX;
141 for (int s = 0; s < stripes; s++) m = m > partial_f[s] ? m : partial_f[s];
142 *out = m;
143 break;
144 }
145 case Reduce2Layer::ReduceType::MIN: {
146 float m = FLT_MAX;
147 for (int s = 0; s < stripes; s++) m = m < partial_f[s] ? m : partial_f[s];
148 *out = m;
149 break;
150 }
151 case Reduce2Layer::ReduceType::SUM: {
152 float acc = 0.0f; for (int s = 0; s < stripes; s++) acc += partial_f[s];
153 *out = acc;
154 break;
155 }
156 case Reduce2Layer::ReduceType::MEAN: {
157 float acc = 0.0f; for (int s = 0; s < stripes; s++) acc += partial_f[s];
158 *out = total > 0 ? (float)((double)acc / (double)total) : 0.0f;
159 break;
160 }
161 case Reduce2Layer::ReduceType::L1:
162 case Reduce2Layer::ReduceType::SUM_SQUARE: {
163 float acc = 0.0f; for (int s = 0; s < stripes; s++) acc += partial_f[s];
164 *out = acc;
165 break;
166 }
167 case Reduce2Layer::ReduceType::L2: {
168 float acc = 0.0f; for (int s = 0; s < stripes; s++) acc += partial_f[s];
169 *out = std::sqrt(acc);
170 break;
171 }
172 case Reduce2Layer::ReduceType::LOG_SUM: {
173 float acc = 0.0f; for (int s = 0; s < stripes; s++) acc += partial_f[s];
174 *out = total > 0 ? std::log(acc) : -std::numeric_limits<float>::infinity();
175 break;
176 }
177 case Reduce2Layer::ReduceType::PROD: {
178 float acc = 1.0f; for (int s = 0; s < stripes; s++) acc *= partial_f[s];
179 *out = acc;
180 break;
181 }
182 default: CV_Error(Error::StsInternal, "reduceAllFloatParallel_: unhandled type");
183 }

186void reduceLastAxesFloatParallel_(const Mat& src, Mat& dst, size_t innerLen, int reduce_type_int) {
187 const Reduce2Layer::ReduceType rt = (Reduce2Layer::ReduceType)reduce_type_int;
188 const float* p = src.ptr<const float>();
189 float* q = dst.ptr<float>();
190 const size_t nOut = src.total() / innerLen;
191
192 const float inv_inner = innerLen > 0 ? 1.0f / (float)innerLen : 0.0f;
193
194 parallel_for_(Range(0, (int)nOut), [&](const Range& r) {
195 for (int row = r.start; row < r.end; row++) {
196 const float* s0 = p + (size_t)row * innerLen;
197 size_t i = 0, n = innerLen;
198#if (CV_SIMD || CV_SIMD_SCALABLE)
199 const int L = VTraits<v_float32>::vlanes();
200#endif
201 switch (rt) {
202 case Reduce2Layer::ReduceType::MAX: {
203 float acc = -FLT_MAX;
204#if (CV_SIMD || CV_SIMD_SCALABLE)
205 if (n >= (size_t)L) {
206 v_float32 va = vx_setall_f32(-FLT_MAX);
207 for (; i + L <= n; i += L) va = v_max(va, vx_load(s0 + i));
208 acc = v_reduce_max(va);
209 }
210#endif
211 for (; i < n; i++) acc = acc > s0[i] ? acc : s0[i];
212 q[row] = acc;
213 break;
214 }
215 case Reduce2Layer::ReduceType::MIN: {
216 float acc = FLT_MAX;
217#if (CV_SIMD || CV_SIMD_SCALABLE)
218 if (n >= (size_t)L) {
219 v_float32 va = vx_setall_f32(FLT_MAX);
220 for (; i + L <= n; i += L) va = v_min(va, vx_load(s0 + i));
221 acc = v_reduce_min(va);
222 }
223#endif
224 for (; i < n; i++) acc = acc < s0[i] ? acc : s0[i];
225 q[row] = acc;
226 break;
227 }
228 case Reduce2Layer::ReduceType::SUM:
229 case Reduce2Layer::ReduceType::MEAN:
230 case Reduce2Layer::ReduceType::LOG_SUM: {
231 float acc = 0.0f;
232#if (CV_SIMD || CV_SIMD_SCALABLE)
233 if (n >= (size_t)L) {
234 v_float32 va = vx_setzero_f32();
235 for (; i + L <= n; i += L) va = v_add(va, vx_load(s0 + i));
236 acc += v_reduce_sum(va);
237 }
238#endif
239 for (; i < n; i++) acc += s0[i];
240 if (rt == Reduce2Layer::ReduceType::MEAN) q[row] = acc * inv_inner;
241 else if (rt == Reduce2Layer::ReduceType::LOG_SUM) q[row] = std::log(acc);
242 else q[row] = acc;
243 break;
244 }
245 case Reduce2Layer::ReduceType::L1: {
246 float acc = 0.0f;
247#if (CV_SIMD || CV_SIMD_SCALABLE)
248 if (n >= (size_t)L) {
249 v_float32 va = vx_setzero_f32();
250 for (; i + L <= n; i += L) va = v_add(va, v_abs(vx_load(s0 + i)));
251 acc += v_reduce_sum(va);
252 }
253#endif
254 for (; i < n; i++) acc += std::fabs(s0[i]);
255 q[row] = acc;
256 break;
257 }
258 case Reduce2Layer::ReduceType::L2:
259 case Reduce2Layer::ReduceType::SUM_SQUARE: {
260 float acc = 0.0f;
261#if (CV_SIMD || CV_SIMD_SCALABLE)
262 if (n >= (size_t)L) {
263 v_float32 va = vx_setzero_f32();
264 for (; i + L <= n; i += L) {
265 v_float32 x = vx_load(s0 + i);
266 va = v_add(va, v_mul(x, x));
267 }
268 acc += v_reduce_sum(va);
269 }
270#endif
271 for (; i < n; i++) { float x = s0[i]; acc += x * x; }
272 q[row] = (rt == Reduce2Layer::ReduceType::L2) ? std::sqrt(acc) : acc;
273 break;
274 }
275 case Reduce2Layer::ReduceType::PROD: {
276 float acc = 1.0f;
277#if (CV_SIMD || CV_SIMD_SCALABLE)
278 if (n >= (size_t)L) {
279 v_float32 va = vx_setall_f32(1.0f);
280 for (; i + L <= n; i += L) va = v_mul(va, vx_load(s0 + i));
281 float buf[VTraits<v_float32>::max_nlanes];
282 v_store(buf, va);
283 for (int k = 0; k < L; k++) acc *= buf[k];
284 }
285#endif
286 for (; i < n; i++) acc *= s0[i];
287 q[row] = acc;
288 break;
289 }
290 default: CV_Error(Error::StsInternal, "reduceLastAxesFloatParallel_: unhandled type");
291 }
292 }
293 }, (double)nOut * (double)innerLen / 16384.0);
294}
