53template <typename F>
54Mat elementWise(const Mat& src, F&& lambda, Mat dst=Mat())
55{
56 if (dst.empty() || !dst.isContinuous() || dst.total() != src.total() || dst.type() != src.type())
57 dst = Mat(src.rows, src.cols, src.type());
58 const int channel = src.channels();
59 if (src.isContinuous()) {
60 const int num_elements = (int)src.total()*channel;
61 const double *psrc = (double*)src.data;
62 double *pdst = (double*)dst.data;
63 const int batch = getNumThreads() > 1 ? 128 : num_elements;
64 const int N = (num_elements / batch) + ((num_elements % batch) > 0);
65 parallel_for_(Range(0, N),[&](const Range& range) {
66 const int start = range.start * batch;
67 const int end = std::min(range.end*batch, num_elements);
68 for (int i = start; i < end; i++) {
69 pdst[i] = lambda(psrc[i]);
70 }
71 });
72 return dst;
73 }
74 switch (channel)
75 {
76 case 1:
77 {
78
79 MatIterator_<double> it, end;
80 for (it = dst.begin<double>(), end = dst.end<double>(); it != end; ++it)
81 {
82 (*it) = lambda((*it));
83 }
84 break;
85 }
86 case 3:
87 {
88 MatIterator_<Vec3d> it, end;
89 for (it = dst.begin<Vec3d>(), end = dst.end<Vec3d>(); it != end; ++it)
90 {
91 for (int j = 0; j < 3; j++)
92 {
93 (*it)[j] = lambda((*it)[j]);
94 }
95 }
96 break;
97 }
98 default:
99 CV_Error(Error::StsBadArg, "Wrong channel!" );
100 break;
101 }
102 return dst;
103}

126template <typename F>
127Mat distanceWise(Mat& src, Mat& ref, F&& lambda)
128{
129 Mat dst = Mat(src.size(), CV_64FC1);
130 MatIterator_<Vec3d> it_src = src.begin<Vec3d>(), end_src = src.end<Vec3d>(),
131 it_ref = ref.begin<Vec3d>();
132 MatIterator_<double> it_dst = dst.begin<double>();
133 for (; it_src != end_src; ++it_src, ++it_ref, ++it_dst)
134 {
135 *it_dst = lambda(*it_src, *it_ref);
136 }
137 return dst;
138}
