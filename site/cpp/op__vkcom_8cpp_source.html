21void Net::Impl::initVkComBackend()
22{
23 CV_TRACE_FUNCTION();
24 CV_Assert(preferableBackend == DNN_BACKEND_VKCOM);
25
26 context = vkcom::Context::create();
27
28 for (MapIdToLayerData::iterator it = layers.begin(); it != layers.end(); it++)
29 {
30 LayerData &ld = it->second;
31 Ptr<Layer> layer = ld.layerInstance;
32 if (!layer->supportBackend(preferableBackend))
33 {
34 continue;
35 }
36
37 try
38 {
39 ld.backendNodes[DNN_BACKEND_VKCOM] = layer->initVkCom(ld.inputBlobsWrappers, ld.outputBlobsWrappers);
40 }
41 catch (const cv::Exception& e)
42 {
43 CV_LOG_ERROR(NULL, "initVkCom failed, fallback to CPU implementation. " << e.what());
44 ld.backendNodes[DNN_BACKEND_VKCOM] = Ptr<BackendNode>();
45 }
46 }
47}

53int transFusedActivType(Ptr<ActivationLayer> &actLayer)
54{
55 if (actLayer)
56 {
57 Ptr<ReLULayer> activ_relu = actLayer.dynamicCast<ReLULayer>();
58 Ptr<ReLU6Layer> activ_relu6 = actLayer.dynamicCast<ReLU6Layer>();
59
60 if (!activ_relu.empty())
61 {
62 if (activ_relu->negativeSlope == 0.0f)
63 {
64 return 1; // kFusedActivRelu
65 }
66 else // Leaky ReLU
67 {
68 return -1; // kFusedActivNone
69 }
70 }
71 else if (!activ_relu6.empty())
72 {
73 return 2; // kFusedActivRelu6
74 }
75 else
76 return -1; // kFusedActivUnsupport
77 }
78 else
79 return 0; // kFusedActivNone
80}

82void copyToTensor(vkcom::Tensor &dst, const Mat &src)
83{
84 CV_Assert(src.isContinuous() && src.type() == CV_32F);
85
86 std::vector<int> mat_shape = shape(src).vec();
87
88 // The following code will copy the src data from CPU Mat to GPU VkBuffer.
89 dst.reshape((const char*)src.data, mat_shape);
90}

92void copyToMat(Mat &dst, vkcom::Tensor &src)
93{
94 CV_Assert(dst.type() == CV_32F);
95
96 std::vector<int> shape = src.getShape();
97 void *data = src.map();
98 Mat tmp(shape, CV_32F, data);
99 tmp.copyTo(dst);
100 src.unMap();
101}

142bool VkComBackendNode::forward()
143{
144 for (int i = 0, n = inputsWrapper_.size(); i < n; ++i)
