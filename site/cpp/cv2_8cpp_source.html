160static PyObject* createSubmodule(PyObject* parent_module, const std::string& name)
161{
162 if (!parent_module)
163 {
164 return PyErr_Format(PyExc_ImportError,
165 "Bindings generation error. "
166 "Parent module is NULL during the submodule '%s' creation",
167 name.c_str()
168 );
169 }
170 if (strEndsWith(name, '.'))
171 {
172 return PyErr_Format(PyExc_ImportError,
173 "Bindings generation error. "
174 "Submodule can't end with a dot. Got: %s", name.c_str()
175 );
176 }
177
178 const std::string parent_name = PyModule_GetName(parent_module);
179
182 if (name == parent_name) {
183 return parent_module;
184 }
185
186 if (!strStartsWith(name, parent_name))
187 {
188 return PyErr_Format(PyExc_ImportError,
189 "Bindings generation error. "
190 "Submodule name should always start with a parent module name. "
191 "Parent name: %s. Submodule name: %s", parent_name.c_str(),
192 name.c_str()
193 );
194 }
195
196 size_t submodule_name_end = name.find('.', parent_name.size() + 1);
198 if (submodule_name_end == std::string::npos)
199 {
200 submodule_name_end = name.size();
201 }
202
203 PyObject* submodule = parent_module;
204
205 for (size_t submodule_name_start = parent_name.size() + 1;
206 submodule_name_start < name.size(); )
207 {
208 const std::string submodule_name = name.substr(submodule_name_start,
209 submodule_name_end - submodule_name_start);
210
211 const std::string full_submodule_name = name.substr(0, submodule_name_end);
212
213
214 PyObject* parent_module_dict = PyModule_GetDict(submodule);
217 submodule = PyDict_GetItemString(parent_module_dict,
218 submodule_name.c_str());
219 if (!submodule)
220 {
222 submodule = PyImport_AddModule(full_submodule_name.c_str());
223 if (!submodule)
224 {
226 return NULL;
227 }
231 if (PyDict_SetItemString(parent_module_dict, submodule_name.c_str(), submodule) < 0) {
232 return PyErr_Format(PyExc_ImportError,
233 "Can't register a submodule '%s' (full name: '%s')",
234 submodule_name.c_str(), full_submodule_name.c_str()
235 );
236 }
237 }
238
239 submodule_name_start = submodule_name_end + 1;
240
241 submodule_name_end = name.find('.', submodule_name_start);
242 if (submodule_name_end == std::string::npos) {
243 submodule_name_end = name.size();
244 }
245 }
246 return submodule;
247}

249static bool init_submodule(PyObject * root, const char * name, PyMethodDef * methods, ConstDef * consts)
250{
251 // traverse and create nested submodules
252 PyObject* submodule = createSubmodule(root, name);
253 if (!submodule)
254 {
255 return false;
256 }
257 // populate module's dict
258 PyObject * d = PyModule_GetDict(submodule);
259 for (PyMethodDef * m = methods; m->ml_name != NULL; ++m)
260 {
261 PyObject * method_obj = PyCFunction_NewEx(m, NULL, NULL);
262 if (PyDict_SetItemString(d, m->ml_name, method_obj) < 0)
263 {
264 PyErr_Format(PyExc_ImportError,
265 "Can't register function %s in module: %s", m->ml_name, name
266 );
267 Py_CLEAR(method_obj);
268 return false;
269 }
270 Py_DECREF(method_obj);
271 }
272 for (ConstDef * c = consts; c->name != NULL; ++c)
273 {
274 PyObject* const_obj = PyLong_FromLongLong(c->val);
275 if (PyDict_SetItemString(d, c->name, const_obj) < 0)
276 {
277 PyErr_Format(PyExc_ImportError,
278 "Can't register constant %s in module %s", c->name, name
279 );
280 Py_CLEAR(const_obj);
281 return false;
282 }
283 Py_DECREF(const_obj);
284 }
285 return true;
286}

304static inline
305bool registerTypeInClassScope(PyObject* cls, const char* type_name, PyObject* type_obj)
306{
307 if (!PyType_CheckExact(cls)) {
308 PyErr_Format(PyExc_ImportError,
309 "Failed to register type '%s' in class scope. "
310 "Scope class object has a wrong type", type_name
311 );
312 return false;
313 }
314 if (PyObject_SetAttrString(cls, type_name, type_obj) < 0)
315 {
316 #ifndef Py_LIMITED_API
317 PyObject* cls_dict = reinterpret_cast<PyTypeObject*>(cls)->tp_dict;
318 if (PyDict_SetItemString(cls_dict, type_name, type_obj) >= 0) {
321 PyErr_Clear();
322 return true;
323 }
324 #endif
325 const std::string cls_name = getPyObjectNameAttr(cls);
326 PyErr_Format(PyExc_ImportError,
327 "Failed to register type '%s' in '%s' class scope. Can't update scope dictionary",
328 type_name, cls_name.c_str()
329 );
330 return false;
331 }
332 return true;
333}

335static inline
336PyObject* getScopeFromTypeObject(PyObject* obj, const std::string& scope_name)
337{
338 if (!PyType_CheckExact(obj)) {
339 const std::string type_name = getPyObjectNameAttr(obj);
340 return PyErr_Format(PyExc_ImportError,
341 "Failed to get scope from type '%s' "
342 "Scope class object has a wrong type", type_name.c_str()
343 );
344 }
346#if defined(Py_LIMITED_API)
347 return PyObject_GetAttrString(obj, scope_name.c_str());
348#else
350 PyObject* type_dict = reinterpret_cast<PyTypeObject*>(obj)->tp_dict;
351 if (!type_dict) {
352 const std::string type_name = getPyObjectNameAttr(obj);
353 return PyErr_Format(PyExc_ImportError,
354 "Failed to get scope from type '%s' "
355 "Type dictionary is not available", type_name.c_str()
356 );
357 }
358 return PyDict_GetItemString(type_dict, scope_name.c_str());
359#endif // Py_LIMITED_API
360}

362static inline
363PyObject* findTypeScope(PyObject* root_module, const std::string& scope_name)
364{
365 PyObject* scope = root_module;
366 if (scope_name.empty())
367 {
368 return scope;
369 }
371 size_t name_end = scope_name.find('.', 1);
372 if (name_end == std::string::npos)
373 {
374 name_end = scope_name.size();
375 }
376 for (size_t name_start = 1; name_start < scope_name.size() && scope; )
377 {
378 const std::string current_scope_name = scope_name.substr(name_start,
379 name_end - name_start);
380
381 if (PyModule_CheckExact(scope))
382 {
383 PyObject* scope_dict = PyModule_GetDict(scope);
384 if (!scope_dict)
385 {
386 return PyErr_Format(PyExc_ImportError,
387 "Scope '%s' dictionary is not available during the search for "
388 " the '%s' scope object", current_scope_name.c_str(),
389 scope_name.c_str()
390 );
391 }
392
393 scope = PyDict_GetItemString(scope_dict, current_scope_name.c_str());
394 }
395 else if (PyType_CheckExact(scope))
396 {
397 scope = getScopeFromTypeObject(scope, current_scope_name);
398 }
399 else
400 {
401 return PyErr_Format(PyExc_ImportError,
402 "Can't find scope '%s'. '%s' doesn't reference a module or a class",
403 scope_name.c_str(), current_scope_name.c_str()
404 );
405 }
406
407
408 name_start = name_end + 1;
409 name_end = scope_name.find('.', name_start);
410 if (name_end == std::string::npos)
411 {
412 name_end = scope_name.size();
413 }
414 }
415 if (!scope)
416 {
417 return PyErr_Format(PyExc_ImportError,
418 "Module or class with name '%s' can't be found in '%s' module",
419 scope_name.c_str(), PyModule_GetName(root_module)
420 );
421 }
422 return scope;
423}

425static bool registerNewType(PyObject* root_module, const char* type_name,
426 PyObject* type_obj, const std::string& scope_name)
427{
428 PyObject* scope = findTypeScope(root_module, scope_name);
429
432 if (!scope) {
433 return false;
434 }
435
436 if (PyModule_CheckExact(scope))
437 {
438 if (!registerTypeInModuleScope(scope, type_name, type_obj))
439 {
440 return false;
441 }
442 }
443 else
444 {
447 if (!registerTypeInClassScope(scope, type_name, type_obj))
448 {
449 return false;
450 }
451 }
452
457 if (scope != root_module)
458 {
459 std::string type_name_str(type_name);
460
461 std::string alias_name;
462 alias_name.reserve(scope_name.size() + type_name_str.size());
463 std::replace_copy(scope_name.begin() + 1, scope_name.end(), std::back_inserter(alias_name), '.', '_');
464 alias_name += '_';
465 alias_name += type_name_str;
466
467 return registerTypeInModuleScope(root_module, alias_name.c_str(), type_obj);
468 }
469 return true;
470}
