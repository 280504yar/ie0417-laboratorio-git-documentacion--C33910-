31static bool ocl_convertScaleAbs( InputArray _src, OutputArray _dst, double alpha, double beta )
32{
33 const ocl::Device & d = ocl::Device::getDefault();
34
35 int type = _src.type(), depth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type);
36 bool doubleSupport = d.doubleFPConfig() > 0;
37 if (!doubleSupport && depth == CV_64F)
38 return false;
39
40 _dst.createSameSize(_src, CV_8UC(cn));
41 int kercn = 1;
42 if (d.isIntel())
43 {
44 static const int vectorWidths[] = {4, 4, 4, 4, 4, 4, 4, -1};
45 kercn = ocl::checkOptimalVectorWidth( vectorWidths, _src, _dst,
46 noArray(), noArray(), noArray(),
47 noArray(), noArray(), noArray(),
48 noArray(), ocl::OCL_VECTOR_MAX);
49 }
50 else
51 kercn = ocl::predictOptimalVectorWidthMax(_src, _dst);
52
53 int rowsPerWI = d.isIntel() ? 4 : 1;
54 char cvt[2][50];
55 int wdepth = std::max(depth, CV_32F);
56 String build_opt = format("-D OP_CONVERT_SCALE_ABS -D UNARY_OP -D dstT=%s -D DEPTH_dst=%d -D srcT1=%s"
57 " -D workT=%s -D wdepth=%d -D convertToWT1=%s -D convertToDT=%s"
58 " -D workT1=%s -D rowsPerWI=%d%s",
59 ocl::typeToStr(CV_8UC(kercn)), CV_8U,
60 ocl::typeToStr(CV_MAKE_TYPE(depth, kercn)),
61 ocl::typeToStr(CV_MAKE_TYPE(wdepth, kercn)), wdepth,
62 ocl::convertTypeStr(depth, wdepth, kercn, cvt[0], sizeof(cvt[0])),
63 ocl::convertTypeStr(wdepth, CV_8U, kercn, cvt[1], sizeof(cvt[1])),
64 ocl::typeToStr(wdepth), rowsPerWI,
65 doubleSupport ? " -D DOUBLE_SUPPORT" : "");
66 ocl::Kernel k("KF", ocl::core::arithm_oclsrc, build_opt);
67 if (k.empty())
68 return false;
69
70 UMat src = _src.getUMat();
71 UMat dst = _dst.getUMat();
72
73 ocl::KernelArg srcarg = ocl::KernelArg::ReadOnlyNoSize(src),
74 dstarg = ocl::KernelArg::WriteOnly(dst, cn, kercn);
75
76 if (wdepth == CV_32F)
77 k.args(srcarg, dstarg, (float)alpha, (float)beta);
78 else if (wdepth == CV_64F)
79 k.args(srcarg, dstarg, alpha, beta);
80
81 size_t globalsize[2] = { (size_t)src.cols * cn / kercn, ((size_t)src.rows + rowsPerWI - 1) / rowsPerWI };
82 return k.run(2, globalsize, NULL, false);
83}

87void convertScaleAbs(InputArray _src, OutputArray _dst, double alpha, double beta)
88{
89 CV_INSTRUMENT_REGION();
90
91 CV_OCL_RUN(_src.dims() <= 2 && _dst.isUMat(),
92 ocl_convertScaleAbs(_src, _dst, alpha, beta))
93
94 Mat src = _src.getMat();
95 int cn = src.channels();
96 double scale[] = {alpha, beta};
97 _dst.create( src.size, CV_8UC(cn) );
98 Mat dst = _dst.getMat();
99 BinaryFunc func = getCvtScaleAbsFunc(src.depth());
100 CV_Assert( func != 0 );
101
102 if( src.dims <= 2 )
103 {
104 Size sz = getContinuousSize2D(src, dst, cn);
105 func( src.ptr(), src.step, 0, 0, dst.ptr(), dst.step, sz, scale );
106 }
107 else
108 {
109 const Mat* arrays[] = {&src, &dst, 0};
110 uchar* ptrs[2] = {};
111 NAryMatIterator it(arrays, ptrs);
112 Size sz((int)it.size*cn, 1);
113
114 for( size_t i = 0; i < it.nplanes; i++, ++it )
115 func( ptrs[0], 0, 0, 0, ptrs[1], 0, sz, scale );
116 }
117}
