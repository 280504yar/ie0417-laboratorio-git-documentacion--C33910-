33static void batchnorm(const Mat& inp, Mat& out, const Mat& scale,
34 const Mat& bias, DataLayout defaultLayout)
35{
36 CV_Assert_N(inp.isContinuous(), out.isContinuous());
37 CV_Assert_N(scale.dims == 1, bias.dims == 1);
38 CV_Assert_N(scale.type() == CV_32F, bias.type() == CV_32F);
39
40 MatShape shape = inp.shape();
41 DataLayout layout = shape.layout;
42 if (layout == DATA_LAYOUT_UNKNOWN)
43 layout = defaultLayout;
44 CV_Assert(layout == DATA_LAYOUT_BLOCK ||
45 layout == DATA_LAYOUT_NCHW ||
46 layout == DATA_LAYOUT_NHWC);
47 int N = shape[0];
48 int C_ = layout == DATA_LAYOUT_BLOCK ? shape.C :
49 layout == DATA_LAYOUT_NCHW ? shape[1] : shape[shape.dims-1];
50 CV_Assert_N(scale.cols == C_, bias.cols == C_);
51 int C1_ = layout != DATA_LAYOUT_NHWC ? shape[1] : 1;
52 int C0_ = layout != DATA_LAYOUT_NCHW ? shape[shape.dims-1] : 1;
53 int type_ = inp.type();
54 CV_SIMD_ONLY(int vlanes_ = VTraits<v_float32>::vlanes());
55
56 size_t esz = inp.elemSize();
57
58 CV_Assert(type_ == CV_32F || type_ == CV_16F || type_ == CV_16BF);
59
60 CV_Assert(inp.shape() == out.shape());
61 CV_Assert(inp.isContinuous());
62 CV_Assert(out.isContinuous());
63
64 int planesize_ = 1;
65 for (int i = 1; i < shape.dims; i++) {
66 planesize_ *= shape[i];
67 }
68 planesize_ /= (C0_*C1_);
69
70 parallel_for_(Range(0, N*C1_), [&](const Range& r) {
71 int C0 = C0_, C1 = C1_, C = C_;
72 int planesize_C0 = planesize_*C0;
73 int type = type_;
74 CV_SIMD_ONLY(int vlanes = vlanes_;
75 constexpr int max_lanes = VTraits<v_float32>::max_nlanes;
76 constexpr int MAX_UNROLL = 4;
77 float scalebuf[max_lanes*MAX_UNROLL];
78 float biasbuf[max_lanes*MAX_UNROLL]);
79
80 for (int k = r.start; k < r.end; k++) {
81 int i = 0, n = k/C1, c1 = k % C1;
82 int c_start = c1*C0, c_end = std::min(C, c_start + C0), c_delta = c_end - c_start;
83 size_t ofs_k = (n*C1 + c1)*planesize_C0*esz;
84 const uchar* inptr_ = inp.data + ofs_k;
85 uchar* outptr_ = out.data + ofs_k;
86 const float* scaleptr = scale.ptr<float>() + c_start;
87 const float* biasptr = bias.ptr<float>() + c_start;
88
89 if (C0 == 1) {
90 // NCHW case
91 float scale_c = 0.f, bias_c = 0.f;
92 if (c_start < c_end) {
93 scale_c = scaleptr[0];
94 bias_c = biasptr[0];
95 }
96 CV_SIMD_ONLY(v_float32 vscale_c = vx_setall_f32(scale_c);
97 v_float32 vbias_c = vx_setall_f32(bias_c));
98 if (type == CV_32F) {
99 const float* inptr = (const float*)inptr_;
100 float* outptr = (float*)outptr_;
101 CV_SIMD_ONLY(for (; i <= planesize_C0 - vlanes*2; i += vlanes*2) {
102 v_float32 x0 = vx_load(inptr + i);
103 v_float32 x1 = vx_load(inptr + i + vlanes);
104 x0 = v_fma(x0, vscale_c, vbias_c);
105 x1 = v_fma(x1, vscale_c, vbias_c);
106 v_store(outptr + i, x0);
107 v_store(outptr + i + vlanes, x1);
108 });
109 for (; i < planesize_C0; i++) {
110 outptr[i] = inptr[i]*scale_c + bias_c;
111 }
112 } else if (type == CV_16F) {
113 const hfloat* inptr = (const hfloat*)inptr_;
114 hfloat* outptr = (hfloat*)outptr_;
115 CV_SIMD_ONLY(for (; i <= planesize_C0 - vlanes*2; i += vlanes*2) {
116 v_float32 x0 = vx_load_expand(inptr + i);
117 v_float32 x1 = vx_load_expand(inptr + i + vlanes);
118 x0 = v_fma(x0, vscale_c, vbias_c);
119 x1 = v_fma(x1, vscale_c, vbias_c);
120 v_pack_store(outptr + i, x0);
121 v_pack_store(outptr + i + vlanes, x1);
122 });
123 for (; i < planesize_C0; i++) {
124 outptr[i] = hfloat(float(inptr[i])*scale_c + bias_c);
125 }
126 } else if (type == CV_16BF) {
127 const bfloat* inptr = (const bfloat*)inptr_;
128 bfloat* outptr = (bfloat*)outptr_;
129 CV_SIMD_ONLY(for (; i <= planesize_C0 - vlanes*2; i += vlanes*2) {
130 v_float32 x0 = vx_load_expand(inptr + i);
131 v_float32 x1 = vx_load_expand(inptr + i + vlanes);
132 x0 = v_fma(x0, vscale_c, vbias_c);
133 x1 = v_fma(x1, vscale_c, vbias_c);
134 v_pack_store(outptr + i, x0);
135 v_pack_store(outptr + i + vlanes, x1);
136 });
137 for (; i < planesize_C0; i++) {
138 outptr[i] = bfloat(float(inptr[i])*scale_c + bias_c);
139 }
140 }
141 }
142 #if CV_SIMD || CV_SIMD_SCALABLE
143 /*
144 [TODO] support C0 == vlanes/2, maybe C0 == vlanes/4.
145 in this case, load everything into vsc0 and vb0, process
146 most part of the plane using vector code as if C0 == vlanes and
147 then process the tail using scalar code
148 */
149 else if (C0 == vlanes*4 || C0 == vlanes*2 || C0 == vlanes) {
150 // accelerated block layout case
151 int c = 0;
152 for (; c < c_delta; c++) {
153 scalebuf[c] = scaleptr[c];
154 biasbuf[c] = biasptr[c];
155 }
156 for (; c < MAX_UNROLL*vlanes; c++) {
157 scalebuf[c] = biasbuf[c] = 0.f;
158 }
159 v_float32 vsc0, vsc1, vsc2, vsc3;
160 v_float32 vb0, vb1, vb2, vb3;
161 vsc0 = vx_load(scalebuf);
162 vsc1 = vx_load(scalebuf + vlanes);
163 vsc2 = vx_load(scalebuf + vlanes*2);
164 vsc3 = vx_load(scalebuf + vlanes*3);
165 vb0 = vx_load(biasbuf);
166 vb1 = vx_load(biasbuf + vlanes);
167 vb2 = vx_load(biasbuf + vlanes*2);
168 vb3 = vx_load(biasbuf + vlanes*3);
169 if (type == CV_32F) {
170 const float* inptr = (const float*)inptr_;
171 float* outptr = (float*)outptr_;
172 if (C0 == vlanes*4) {
173 for (; i < planesize_C0; i += C0) {
174 v_float32 x0 = vx_load(inptr + i);
175 v_float32 x1 = vx_load(inptr + i + vlanes);
176 v_float32 x2 = vx_load(inptr + i + vlanes*2);
177 v_float32 x3 = vx_load(inptr + i + vlanes*3);
178 x0 = v_fma(x0, vsc0, vb0);
179 x1 = v_fma(x1, vsc1, vb1);
180 x2 = v_fma(x2, vsc2, vb2);
181 x3 = v_fma(x3, vsc3, vb3);
182 v_store(outptr + i, x0);
183 v_store(outptr + i + vlanes, x1);
184 v_store(outptr + i + vlanes*2, x2);
185 v_store(outptr + i + vlanes*3, x3);
186 }
187 } else if (C0 == vlanes*2) {
188 for (; i < planesize_C0; i += C0) {
189 v_float32 x0 = vx_load(inptr + i);
190 v_float32 x1 = vx_load(inptr + i + vlanes);
191 x0 = v_fma(x0, vsc0, vb0);
192 x1 = v_fma(x1, vsc1, vb1);
193 v_store(outptr + i, x0);
194 v_store(outptr + i + vlanes, x1);
195 }
196 } else {
197 for (; i < planesize_C0; i += C0) {
198 v_float32 x0 = vx_load(inptr + i);
199 x0 = v_fma(x0, vsc0, vb0);
200 v_store(outptr + i, x0);
201 }
202 }
203 } else if (type == CV_16F) {
204 const hfloat* inptr = (const hfloat*)inptr_;
205 hfloat* outptr = (hfloat*)outptr_;
206 if (type == CV_32F) {
207 if (C0 == vlanes*4) {
208 for (; i < planesize_C0; i += C0) {
209 v_float32 x0 = vx_load_expand(inptr + i);
210 v_float32 x1 = vx_load_expand(inptr + i + vlanes);
211 v_float32 x2 = vx_load_expand(inptr + i + vlanes*2);
212 v_float32 x3 = vx_load_expand(inptr + i + vlanes*3);
213 x0 = v_fma(x0, vsc0, vb0);
214 x1 = v_fma(x1, vsc1, vb1);
215 x2 = v_fma(x2, vsc2, vb2);
216 x3 = v_fma(x3, vsc3, vb3);
217 v_pack_store(outptr + i, x0);
218 v_pack_store(outptr + i + vlanes, x1);
219 v_pack_store(outptr + i + vlanes*2, x2);
220 v_pack_store(outptr + i + vlanes*3, x3);
221 }
222 } else if (C0 == vlanes*2) {
223 for (; i < planesize_C0; i += C0) {
224 v_float32 x0 = vx_load_expand(inptr + i);
225 v_float32 x1 = vx_load_expand(inptr + i + vlanes);
226 x0 = v_fma(x0, vsc0, vb0);
227 x1 = v_fma(x1, vsc1, vb1);
228 v_pack_store(outptr + i, x0);
229 v_pack_store(outptr + i + vlanes, x1);
230 }
231 } else {
232 for (; i < planesize_C0; i += C0) {
233 v_float32 x0 = vx_load_expand(inptr + i);
234 x0 = v_fma(x0, vsc0, vb0);
235 v_pack_store(outptr + i, x0);
236 }
237 }
238 }
239 } else if (type == CV_16BF) {
240 const bfloat* inptr = (const bfloat*)inptr_;
241 bfloat* outptr = (bfloat*)outptr_;
242 if (C0 == vlanes*4) {
243 for (; i < planesize_C0; i += C0) {
244 v_float32 x0 = vx_load_expand(inptr + i);
245 v_float32 x1 = vx_load_expand(inptr + i + vlanes);
246 v_float32 x2 = vx_load_expand(inptr + i + vlanes*2);
247 v_float32 x3 = vx_load_expand(inptr + i + vlanes*3);
248 x0 = v_fma(x0, vsc0, vb0);
249 x1 = v_fma(x1, vsc1, vb1);
250 x2 = v_fma(x2, vsc2, vb2);
251 x3 = v_fma(x3, vsc3, vb3);
252 v_pack_store(outptr + i, x0);
253 v_pack_store(outptr + i + vlanes, x1);
254 v_pack_store(outptr + i + vlanes*2, x2);
255 v_pack_store(outptr + i + vlanes*3, x3);
256 }
257 } else if (C0 == vlanes*2) {
258 for (; i < planesize_C0; i += C0) {
259 v_float32 x0 = vx_load_expand(inptr + i);
260 v_float32 x1 = vx_load_expand(inptr + i + vlanes);
261 x0 = v_fma(x0, vsc0, vb0);
262 x1 = v_fma(x1, vsc1, vb1);
263 v_pack_store(outptr + i, x0);
264 v_pack_store(outptr + i + vlanes, x1);
265 }
266 } else {
267 for (; i < planesize_C0; i += C0) {
268 v_float32 x0 = vx_load_expand(inptr + i);
269 x0 = v_fma(x0, vsc0, vb0);
270 v_pack_store(outptr + i, x0);
271 }
272 }
273 }
274 }
275 #endif
276 else {
277 // general block layout or NHWC case
278 if (type == CV_32F) {
279 const float* inptr = (const float*)inptr_;
280 float* outptr = (float*)outptr_;
281 for (; i < planesize_C0; i += C0) {
282 int c = 0;
283 CV_SIMD_ONLY(for (; c <= c_delta - vlanes*2; c += vlanes*2) {
284 v_float32 x0 = vx_load(inptr + i + c);
285 v_float32 x1 = vx_load(inptr + i + c + vlanes);
286 v_float32 sc0 = vx_load(scaleptr + c);
287 v_float32 sc1 = vx_load(scaleptr + c + vlanes);
288 v_float32 b0 = vx_load(biasptr + c);
289 v_float32 b1 = vx_load(biasptr + c + vlanes);
290 x0 = v_fma(x0, sc0, b0);
291 x1 = v_fma(x1, sc1, b1);
292 v_store(outptr + i + c, x0);
293 v_store(outptr + i + c + vlanes, x1);
294 });
295 for (; c < c_delta; c++)
296 outptr[i + c] = inptr[i + c]*scaleptr[c] + biasptr[c];
297 for (; c < C0; c++)
298 outptr[i + c] = 0.f;
299 }
300 } else if (type == CV_16F) {
301 const hfloat* inptr = (const hfloat*)inptr_;
302 hfloat* outptr = (hfloat*)outptr_;
303 hfloat z = hfloat(0.f);
304 for (; i < planesize_C0; i += C0) {
305 int c = 0;
306 CV_SIMD_ONLY(for (; c <= c_delta - vlanes*2; c += vlanes*2) {
307 v_float32 x0 = vx_load_expand(inptr + i + c);
308 v_float32 x1 = vx_load_expand(inptr + i + c + vlanes);
309 v_float32 sc0 = vx_load(scaleptr + c);
310 v_float32 sc1 = vx_load(scaleptr + c + vlanes);
311 v_float32 b0 = vx_load(biasptr + c);
312 v_float32 b1 = vx_load(biasptr + c + vlanes);
313 x0 = v_fma(x0, sc0, b0);
314 x1 = v_fma(x1, sc1, b1);
315 v_pack_store(outptr + i + c, x0);
316 v_pack_store(outptr + i + c + vlanes, x1);
317 });
318 for (; c < c_delta; c++)
319 outptr[i + c] = hfloat(float(inptr[i + c])*scaleptr[c] + biasptr[c]);
320 for (; c < C0; c++)
321 outptr[i + c] = z;
322 }
323 } else if (type == CV_16BF) {
324 const bfloat* inptr = (const bfloat*)inptr_;
325 bfloat* outptr = (bfloat*)outptr_;
326 bfloat z = bfloat(0.f);
327 for (; i < planesize_C0; i += C0) {
328 int c = 0;
329 CV_SIMD_ONLY(for (; c <= c_delta - vlanes*2; c += vlanes*2) {
330 v_float32 x0 = vx_load_expand(inptr + i + c);
331 v_float32 x1 = vx_load_expand(inptr + i + c + vlanes);
332 v_float32 sc0 = vx_load(scaleptr + c);
333 v_float32 sc1 = vx_load(scaleptr + c + vlanes);
334 v_float32 b0 = vx_load(biasptr + c);
335 v_float32 b1 = vx_load(biasptr + c + vlanes);
336 x0 = v_fma(x0, sc0, b0);
337 x1 = v_fma(x1, sc1, b1);
338 v_pack_store(outptr + i + c, x0);
339 v_pack_store(outptr + i + c + vlanes, x1);
340 });
341 for (; c < c_delta; c++)
342 outptr[i + c] = bfloat(float(inptr[i + c])*scaleptr[c] + biasptr[c]);
343 for (; c < C0; c++)
344 outptr[i + c] = z;
345 }
346 }
347 }
348 }
349 }, (planesize_*C0_ > 1000000 ? N*C1_ : 1));

509void BatchNorm2Layer::getScaleBias(InputArray scale_, InputArray bias_,
510 InputArray mean_, InputArray variance_, float eps,
511 OutputArray outscale_, OutputArray outbias_)
512{
513 Mat scale = scale_.getMat(), bias = bias_.getMat();
514 Mat mean = mean_.getMat(), var = variance_.getMat();
515
516 int sctype = scale.type(), btype = bias.type();
517 int mtype = mean.type(), vtype = var.type();
518
519 CV_Assert(sctype == CV_32F || sctype == CV_16F || sctype == CV_16BF);
520 CV_Assert(btype == CV_32F || btype == CV_16F || btype == CV_16BF);
521 CV_Assert(mtype == CV_32F || mtype == CV_16F || mtype == CV_16BF);
522 CV_Assert(vtype == CV_32F || vtype == CV_16F || vtype == CV_16BF);
523
524 CV_Assert_N(scale.dims == 1, bias.dims == 1, mean.dims == 1, var.dims == 1);
525 int C = scale.cols;
526 CV_Assert_N(bias.cols == C, mean.cols == C, var.cols == C);
527
528 Mat outscale(1, &C, CV_32F);
529 Mat outbias(1, &C, CV_32F);
530
531 const uchar* scdata = scale.data;
532 const uchar* bdata = bias.data;
533 const uchar* mdata = mean.data;
534 const uchar* vdata = var.data;
535 float* outsc = outscale.ptr<float>();
536 float* outb = outbias.ptr<float>();
537
538 #undef LOAD_AS_FLOAT
539 #define LOAD_AS_FLOAT(typ, ptr, i) \
540 (typ == CV_32F ? ((const float*)ptr)[i] : \

545 for (int i = 0; i < C; i++) {

552 float outbval = b - m*outscval;
553 outsc[i] = outscval;
554 outb[i] = outbval;
555 }
