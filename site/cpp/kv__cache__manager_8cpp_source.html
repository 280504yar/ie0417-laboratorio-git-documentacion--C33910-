68void KVCacheManager::buildRoutes()
69{
70 presentToPastRoutes.clear();
71 hasRoutes = false;
72
73 if (!netimpl || !netimpl->mainGraph)
74 return;
75
76 const std::vector<Arg>& gr_outputs = netimpl->mainGraph->outputs();
77 for (const Arg& out_arg : gr_outputs)
78 {
79 const ArgData& out_adata = netimpl->args.at(out_arg.idx);
80 const std::string& out_name = out_adata.name;
81
82 if (out_name.compare(0, 8, "present.") != 0)
83 continue;
84
85 std::string past_name = "past_key_values." + out_name.substr(8);
86 auto it = netimpl->argnames.find(past_name);
87 if (it == netimpl->argnames.end())
88 continue;
89
90 Arg past_arg((int)it->second);
91 if (netimpl->args.at(past_arg.idx).kind != DNN_ARG_INPUT)
92 continue;
93
94 presentToPastRoutes.emplace_back(out_arg.idx, past_arg.idx);
95 }
96
97 hasRoutes = !presentToPastRoutes.empty();
98 if (hasRoutes)
99 initPastTensors();
100}

102void KVCacheManager::initPastTensors()
103{
104 for (const auto& route : presentToPastRoutes)
105 {
106 const ArgData& past_adata = netimpl->args.at(route.second);
107 const MatShape& decl_shape = past_adata.shape;
108 if (decl_shape.dims <= 0)
109 continue;
110
111 // Replace symbolic dims (stored as <=0): batch (dim 0) -> 1, all others -> 0 for empty-sequence state.
112 std::vector<int> shape_vec(decl_shape.dims);
113 for (int d = 0; d < decl_shape.dims; d++)
114 shape_vec[d] = (decl_shape[d] > 0) ? decl_shape[d] : (d == 0 ? 1 : 0);
115
116 int dtype = past_adata.type;
117 if (dtype < 0)
118 dtype = CV_32F;
119
120 Mat& past_t = netimpl->__tensors__.at(route.second);
121 past_t = Mat(shape_vec, dtype, Scalar(0));
122 // The consumer's signature changed; the op loop re-finalizes it.
123 }
124}

126void KVCacheManager::applyRoutes()
127{
128 for (const auto& route : presentToPastRoutes)
129 {
130 const Mat& present_t = netimpl->argTensor(Arg(route.first));
131 Mat& past_t = netimpl->__tensors__.at(route.second);
132 if (present_t.empty())
133 continue;
134 // Grown buffer changes the consumer's signature; the op loop re-finalizes it.
135 present_t.copyTo(past_t);

145void KVCache::grow(const Mat& newData) {
146 CV_Assert(newData.dims == 4 || newData.dims == 3);
147
148 if (nHeads == -1) {
149 CV_Assert(newData.dims == 4); // to derive shape from data, we need 4D
150 headDim = newData.size[3];
151 nHeads = newData.size[1];
152 batchSize = newData.size[0];
153 } else {
154 if (headDim == -1) {
155 if (newData.dims == 4)
156 headDim = newData.size[3];
157 else{
158 CV_Assert(newData.dims == 3);
159 CV_Assert(newData.size[2] % nHeads == 0);
160 headDim = newData.size[2] / nHeads;
161 }
162 } else {
163 if (newData.dims == 4)
164 CV_Assert(newData.size[3] == headDim);
165 else
166 CV_Assert(newData.size[2] == headDim * nHeads);
167 }
168 if (batchSize == -1)
169 batchSize = newData.size[0];
170 }
171
172 int T = newData.dims == 4 ? newData.size[2] : newData.size[1];
173
174 if (pages.empty()) {
175 // Prefill: allocate the page pool, honouring any reserved capacity.
176 int neededPages = (T + pageSize - 1) / pageSize;
177 int reservedPages = (int)(((int64_t)reservedTokens + pageSize - 1) / pageSize);
178 int totalPages = std::max(neededPages, reservedPages);
179 for (int i = 0; i < totalPages; i++) {
180 int page_size = isKCache ?
181 (int)fastGemmPackBSize(pageSize, headDim, opt):
182 (int)fastGemmPackBSize(headDim, pageSize, opt);
183
184 pages.push_back(
185 Mat({batchSize, nHeads, page_size}, CV_32F, Scalar(0))
186 );
187 }
188 growPrefill(newData, T);
189 } else {
190 // Non-empty cache: append T (>=1) tokens (T>1 = chunked prefill / speculative decode).
191 appendTokens(newData, T);
192 }

196void KVCache::appendTokens(const Mat& newData, int T) {
197 if (T == 1) {
198 growGenerate(newData);
199 return;
200 }
201 const int seqAxis = newData.dims == 4 ? 2 : 1;
202 std::vector<Range> ranges(newData.dims, Range::all());
203 for (int t = 0; t < T; t++) {
204 ranges[seqAxis] = Range(t, t + 1);
205 Mat token = newData(ranges).clone(); // contiguous single-token slice
206 growGenerate(token);
207 }

210void KVCache::growPrefill(const Mat& newData, int T){
211 int totalPages = (T + pageSize - 1) / pageSize;
212 int ps = isKCache ? (int)fastGemmPackBSize(pageSize, headDim, opt)
213 : (int)fastGemmPackBSize(headDim, pageSize, opt);
214
215 bool is3Dlayout = newData.dims == 3;
216
217 int total = totalPages * batchSize * nHeads;
218
219 cv::TLSData<std::vector<float>> tls_temp_buf;
220
221 auto fn = [&](const Range& range) {
222 for (int i = range.start; i < range.end; i++) {
223 int page = i / (batchSize * nHeads);
224 int b = (i - page * batchSize * nHeads) / nHeads;
225 int h = i % nHeads;
226
227 // source
228 size_t step_source = b * nHeads * T * headDim;
229 if(is3Dlayout)
230 step_source += page * pageSize * nHeads * headDim +
231 h * headDim;
232 else
233 step_source += h * headDim * T + page * pageSize * headDim;
234 const auto* source = newData.ptr<float>() + step_source;
235
236 int chunk_T = std::min(pageSize, T - page * pageSize);
237 const float* actual_source = source;
238
239 int lds = is3Dlayout ? headDim * nHeads : headDim;
240
241 if (chunk_T < pageSize) {
242 std::vector<float>& temp_buf = *tls_temp_buf.get();
243 temp_buf.assign(pageSize * headDim, 0.0f);
244 for (int i = 0; i < chunk_T; i++) {
245 std::memcpy(temp_buf.data() + i * headDim, source + i * lds, headDim * sizeof(float));
246 }
247 actual_source = temp_buf.data();
248 }
249
250 // dst
251 size_t step_dst = b * nHeads * ps +
252 h * ps;
253 auto* dst = pages[page].ptr<float>() + step_dst;
254
255 const int N = headDim;
256 const int K = pageSize;
257
258 fastGemmPackB(
259 isKCache,
260 N, K,
261 actual_source, chunk_T < pageSize ? headDim : lds,
262 dst,
263 opt
264 );
265 }
266 };
267 parallel_for_(Range(0, total), fn);
268 nTokens += T;
269}

271void KCache::growGenerate(const Mat& newData){
272 int cur_page = nTokens / pageSize;
273 const int Ps = fastGemmPackBSize(pageSize, headDim, opt);
274 int t0 = nTokens % pageSize;
275 const int batch_size = newData.size[0];
276
277 if (cur_page >= (int)pages.size()) {
278 pages.push_back(Mat({batchSize, nHeads, Ps}, CV_32F, Scalar(0)));
279 }
280
281 auto* page = pages[cur_page].ptr<float>();
282 const auto* data = newData.ptr<float>();
283
284 const int nstripes = batch_size * nHeads;
285 parallel_for_(Range(0, nstripes), [&](const Range& r) {
286 for (int i = r.start; i < r.end; i++) {
287 int b = i / nHeads, h = i % nHeads;
288 const float* src = data + (b * nHeads + h) * headDim;
289 float* dst = page + b * nHeads * Ps + h * Ps + t0;
290 for (int j = 0; j < headDim; j++)
291 dst[j * pageSize] = src[j];
292 }
293 });
294
295 nTokens += 1;
296}

298void VCache::growGenerate(const Mat& newData){
299 const int batch_size = newData.size[0];
300 const int Nr = fastGemmNR(opt);
301 const int Ps = fastGemmPackBSize(headDim, pageSize, opt);
302 const int t0 = nTokens % pageSize;
303 const int step_packed = pageSize * Nr;
304 int cur_page = nTokens / pageSize;
305
306 if (cur_page >= (int)pages.size()) {
307 pages.push_back(Mat({batchSize, nHeads, Ps}, CV_32F, Scalar(0)));
308 }
309
310 auto* page = pages[cur_page].ptr<float>();
311 const auto* data = newData.ptr<float>();
312
313 const int nstripes = batch_size * nHeads;
314 parallel_for_(Range(0, nstripes), [&](const Range& r) {
315 for (int i = r.start; i < r.end; i++) {
316 int b = i / nHeads, h = i % nHeads;
317 for (int j = 0; j <= (headDim - 1) / Nr; j++) {
318 int step = b * nHeads * headDim + h * headDim + j * Nr;
319 int copy_size = std::min(Nr, headDim - j * Nr);
320
321 auto* cur_page_ptr = page + b * nHeads * Ps + h * Ps + t0 * Nr + j * step_packed;
322 const float* src_ptr = data + step;
323 std::memcpy(cur_page_ptr, src_ptr, copy_size * sizeof(float));
324 if (copy_size < Nr) {
325 float replication_val = src_ptr[0];
326 for (int k = copy_size; k < Nr; k++) {
327 cur_page_ptr[k] = replication_val;
328 }
329 }
330 }
331 }
332 });
333
334 nTokens += 1;
335}
