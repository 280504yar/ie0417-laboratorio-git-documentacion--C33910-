33 public:

37 PaddingOp(csl::Stream stream_, PaddingType type_, T value_, const std::vector<cv::Range>& ranges)
38 : stream(std::move(stream_)), type{ type_ }, value{ value_ }, dstRanges(std::move(ranges))

42 void forward(
43 const std::vector<cv::Ptr<BackendWrapper>>& inputs,
44 const std::vector<cv::Ptr<BackendWrapper>>& outputs,
45 csl::Workspace& workspace) override
46 {
47 CV_Assert(inputs.size() == 1 && outputs.size() == 1);
48
49 auto input_wrapper = inputs[0].dynamicCast<wrapper_type>();
50 auto input = input_wrapper->getView();
51
52 auto output_wrapper = outputs[0].dynamicCast<wrapper_type>();
53 auto output = output_wrapper->getSpan();
54
55 /* suppose we require padding for the first spatial axis (H in NCHW or D in NCDHW)
56 *
57 * there could be a case where the batch axis, channel axis, and the first spatial axis are all one
58 * this would result in effective rank being less than the number of axes requiring padding
59 */
60 /* the effective rank of the input may be smaller than the effective rank of the output but the converse is never true
61 * input: [1, 1, 1, 3]; effective rank = 1
62 * output: [1, 1, 3, 3]; effective rank = 2
63 *
64 * hence, we use the effective rank of the output tensor for the padding operation
65 */
66 auto effective_rank = get_effective_rank(output);
67 CV_Assert(get_effective_rank(input) <= effective_rank);
68 effective_rank = std::max(effective_rank, dstRanges.size());
69
70 for (int i = effective_rank - dstRanges.size(); i < effective_rank; i++)
71 {
72 if (dstRanges[i] == Range::all())
73 CV_Assert(input.get_axis_size(i) == output.get_axis_size(i));
74 else
75 CV_Assert(input.get_axis_size(i) == dstRanges[i].size());
76 }
77
78 if (type == PaddingType::CONSTANT)
79 {
80 kernels::fill<T>(stream, output, value);
81
82 std::vector<std::size_t> offsets(effective_rank, 0);
83 for (int i = 0; i < dstRanges.size(); i++)
84 {
85 const auto delta = effective_rank - dstRanges.size();
86 if (dstRanges[i] != Range::all())
87 offsets[delta + i] = dstRanges[i].start;
88 }
89
90 kernels::concat_with_offsets<T>(stream, output, input, offsets);
91 }
92 else if (type == PaddingType::REFLECTION101)
93 {
94 std::vector<std::pair<std::size_t, std::size_t>> ranges(effective_rank);
95 for (int i = 0; i < effective_rank; i++)
96 {
97 const auto delta = effective_rank - dstRanges.size();
98 if (i < delta || dstRanges[i - delta] == Range::all())
99 ranges[i] = { 0, input.get_axis_size(i) };
100 else
101 ranges[i] = { dstRanges[i].start, dstRanges[i].end };
102 }
103
104 kernels::copy_with_reflection101<T>(stream, output, input, ranges);
105 }
106 }
107
108 private:
109 csl::Stream stream;
110 PaddingType type;
111 T value;
112
113 std::vector<cv::Range> dstRanges;
114 };
