33 void writeTag( const char* key, int tag_type, const std::vector<std::string>& attrlist=std::vector<std::string>() )
34 {
35 char* ptr = fs->bufferPtr();
36 int i, len = 0;
37 FStructData& current_struct = fs->getCurrentStruct();
38 int struct_flags = current_struct.flags;
39
40 if( key && key[0] == '\0' )
41 key = 0;
42
43 if( tag_type == CV_XML_OPENING_TAG || tag_type == CV_XML_EMPTY_TAG )
44 {
45 if( FileNode::isCollection(struct_flags) )
46 {
47 if( FileNode::isMap(struct_flags) ^ (key != 0) )
48 CV_Error( cv::Error::StsBadArg, "An attempt to add element without a key to a map, "
49 "or add element with key to sequence" );
50 }
51 else
52 {
53 struct_flags = FileNode::EMPTY + (key ? FileNode::MAP : FileNode::SEQ);
54 //fs->is_first = 0;
55 }
56
57 if( !FileNode::isEmptyCollection(struct_flags) )
58 ptr = fs->flush();
59 }
60
61 if( !key )
62 key = "_";
63 else if( key[0] == '_' && key[1] == '\0' )
64 CV_Error( cv::Error::StsBadArg, "A single _ is a reserved tag name" );
65
66 len = (int)strlen( key );
67 *ptr++ = '<';
68 if( tag_type == CV_XML_CLOSING_TAG )
69 {
70 if( !attrlist.empty() )
71 CV_Error( cv::Error::StsBadArg, "Closing tag should not include any attributes" );
72 *ptr++ = '/';
73 }
74
75 if( !cv_isalpha(key[0]) && key[0] != '_' )
76 CV_Error( cv::Error::StsBadArg, "Key should start with a letter or _" );
77
78 ptr = fs->resizeWriteBuffer( ptr, len );
79 for( i = 0; i < len; i++ )
80 {
81 char c = key[i];
82 if( !cv_isalnum(c) && c != '_' && c != '-' )
83 CV_Error( cv::Error::StsBadArg, "Key name may only contain alphanumeric characters [a-zA-Z0-9], '-' and '_'" );
84 ptr[i] = c;
85 }
86 ptr += len;
87
88 int nattr = (int)attrlist.size();
89 CV_Assert( nattr % 2 == 0 );
90
91 for( i = 0; i < nattr; i += 2 )
92 {
93 size_t len0 = attrlist[i].size();
94 size_t len1 = attrlist[i+1].size();
95 CV_Assert( len0 > 0 );
96
97 ptr = fs->resizeWriteBuffer( ptr, (int)(len0 + len1 + 4) );
98 *ptr++ = ' ';
99
100 memcpy( ptr, attrlist[i].c_str(), len0 );
101 ptr += len0;
102 *ptr++ = '=';
103 *ptr++ = '\"';
104 if( len1 > 0 )
105 memcpy( ptr, attrlist[i+1].c_str(), len1 );
106 ptr += len1;
107 *ptr++ = '\"';
108 }
109
110 if( tag_type == CV_XML_EMPTY_TAG )
111 *ptr++ = '/';
112 *ptr++ = '>';
113 fs->setBufferPtr(ptr);
114 current_struct.flags = struct_flags & ~FileNode::EMPTY;
115 }

160 void write(const char* key, const char* str, bool quote)
161 {
162 char buf[CV_FS_MAX_LEN*6+16];
163 char* data = (char*)str;
164 int i, len;
165
166 if( !str )
167 CV_Error( cv::Error::StsNullPtr, "Null string pointer" );
168
169 len = (int)strlen(str);
170 if( len > CV_FS_MAX_LEN )
171 CV_Error( cv::Error::StsBadArg, "The written string is too long" );
172
173 if( quote || len == 0 || str[0] != '\"' || str[0] != str[len-1] )
174 {
175 bool need_quote = quote || len == 0;
176 data = buf;
177 *data++ = '\"';
178 for( i = 0; i < len; i++ )
179 {
180 char c = str[i];
181
182 if( (uchar)c >= 128 || c == ' ' )
183 {
184 *data++ = c;
185 need_quote = true;
186 }
187 else if( !cv_isprint(c) || c == '<' || c == '>' || c == '&' || c == '\'' || c == '\"' )
188 {
189 *data++ = '&';
190 if( c == '<' )
191 {
192 memcpy(data, "lt", 2);
193 data += 2;
194 }
195 else if( c == '>' )
196 {
197 memcpy(data, "gt", 2);
198 data += 2;
199 }
200 else if( c == '&' )
201 {
202 memcpy(data, "amp", 3);
203 data += 3;
204 }
205 else if( c == '\'' )
206 {
207 memcpy(data, "apos", 4);
208 data += 4;
209 }
210 else if( c == '\"' )
211 {
212 memcpy( data, "quot", 4);
213 data += 4;
214 }
215 else
216 {
217 snprintf( data, sizeof(buf) - (data - buf), "#x%02x", (uchar)c );
218 data += 4;
219 }
220 *data++ = ';';
221 need_quote = 1;
222 }
223 else
224 *data++ = c;
225 }
226 if( !need_quote && (cv_isdigit(str[0]) ||
227 str[0] == '+' || str[0] == '-' || str[0] == '.' ))
228 need_quote = true;
229
230 if( need_quote )
231 *data++ = '\"';
232 len = (int)(data - buf) - !need_quote;
233 *data++ = '\0';
234 data = buf + (int)!need_quote;
235 }
236
237 writeScalar( key, data );
238 }

240 void writeScalar(const char* key, const char* data)
241 {
242 fs->check_if_write_struct_is_delayed(false);
243 if ( fs->get_state_of_writing_base64() == FileStorage_API::Uncertain )
244 {
245 fs->switch_to_Base64_state( FileStorage_API::NotUse );
246 }
247 else if ( fs->get_state_of_writing_base64() == FileStorage_API::InUse )
248 {
249 CV_Error( cv::Error::StsError, "At present, output Base64 data only." );
250 }
251
252 int len = (int)strlen(data);
253 if( key && *key == '\0' )
254 key = 0;
255
256 FStructData& current_struct = fs->getCurrentStruct();
257 int struct_flags = current_struct.flags;
258
259 if( FileNode::isMap(struct_flags) ||
260 (!FileNode::isCollection(struct_flags) && key) )
261 {
262 writeTag( key, CV_XML_OPENING_TAG );
263 char* ptr = fs->resizeWriteBuffer( fs->bufferPtr(), len );
264 memcpy( ptr, data, len );
265 fs->setBufferPtr( ptr + len );
266 writeTag( key, CV_XML_CLOSING_TAG );
267 }
268 else
269 {
270 char* ptr = fs->bufferPtr();
271 int new_offset = (int)(ptr - fs->bufferStart()) + len;
272
273 if( key )
274 CV_Error( cv::Error::StsBadArg, "elements with keys can not be written to sequence" );
275
276 current_struct.flags = FileNode::SEQ;
277
278 if( (new_offset > fs->wrapMargin() && new_offset - current_struct.indent > 10) ||
279 (ptr > fs->bufferStart() && ptr[-1] == '>') )
280 {
281 ptr = fs->flush();
282 }
283 else if( ptr > fs->bufferStart() + current_struct.indent && ptr[-1] != '>' )
284 *ptr++ = ' ';
285
286 memcpy( ptr, data, len );
287 fs->setBufferPtr(ptr + len);
288 }

291 void writeComment(const char* comment, bool eol_comment)
292 {
293 FStructData& current_struct = fs->getCurrentStruct();
294 int len;
295 int multiline;
296 const char* eol;
297 char* ptr;
298
299 if( !comment )
300 CV_Error( cv::Error::StsNullPtr, "Null comment" );
301
302 if( strstr(comment, "--") != 0 )
303 CV_Error( cv::Error::StsBadArg, "Double hyphen \'--\' is not allowed in the comments" );
304
305 len = (int)strlen(comment);
306 eol = strchr(comment, '\n');
307 multiline = eol != 0;
308 ptr = fs->bufferPtr();
309
310 if( multiline || !eol_comment || fs->bufferEnd() - ptr < len + 5 )
311 ptr = fs->flush();
312 else if( ptr > fs->bufferStart() + current_struct.indent )
313 *ptr++ = ' ';
314
315 if( !multiline )
316 {
317 ptr = fs->resizeWriteBuffer( ptr, len + 5+4+1 );
318 snprintf( ptr, len + 5+4+1, "<!-- %s -->", comment );
319 len = (int)strlen(ptr);
320 }
321 else
322 {
323 strcpy( ptr, "<!--" );
324 len = 4;
325 }
326
327 fs->setBufferPtr(ptr + len);
328 ptr = fs->flush();
329
330 if( multiline )
331 {
332 while( comment )
333 {
334 if( eol )
335 {
336 ptr = fs->resizeWriteBuffer( ptr, (int)(eol - comment) + 1 );
337 memcpy( ptr, comment, eol - comment + 1 );
338 ptr += eol - comment;
339 comment = eol + 1;
340 eol = strchr( comment, '\n' );
341 }
342 else
343 {
344 len = (int)strlen(comment);
345 ptr = fs->resizeWriteBuffer( ptr, len );
346 memcpy( ptr, comment, len );
347 ptr += len;
348 comment = 0;
349 }
350 fs->setBufferPtr(ptr);
351 ptr = fs->flush();
352 }
353 strcpy( ptr, "-->" );
354 fs->setBufferPtr(ptr + 3);
355 fs->flush();
356 }
357 }

377 char* skipSpaces( char* ptr, int mode )
378 {
379 if (!ptr)
380 CV_PARSE_ERROR_CPP("Invalid input");
381
382 int level = 0;
383
384 for(;;)
385 {
386 char c;
387 ptr--;
388
389 if( mode == CV_XML_INSIDE_COMMENT )
390 {
391 do c = *++ptr;
392 while( cv_isprint_or_tab(c) && (c != '-' || ptr[1] != '-' || ptr[2] != '>') );
393
394 if( c == '-' )
395 {
396 CV_Assert( ptr[1] == '-' && ptr[2] == '>' );
397 mode = 0;
398 ptr += 3;
399 }
400 }
401 else if( mode == CV_XML_INSIDE_DIRECTIVE )
402 {
403 // !!!NOTE!!! This is not quite correct, but should work in most cases
404 do
405 {
406 c = *++ptr;
407 level += c == '<';
408 level -= c == '>';
409 if( level < 0 )
410 return ptr;
411 } while( cv_isprint_or_tab(c) );
412 }
413 else
414 {
415 do c = *++ptr;
416 while( c == ' ' || c == '\t' );
417
418 if( c == '<' && ptr[1] == '!' && ptr[2] == '-' && ptr[3] == '-' )
419 {
420 if( mode != 0 )
421 CV_PARSE_ERROR_CPP( "Comments are not allowed here" );
422 mode = CV_XML_INSIDE_COMMENT;
423 ptr += 4;
424 }
425 else if( cv_isprint(c) )
426 break;
427 }
428
429 if( !cv_isprint(*ptr) )
430 {
431 if( *ptr != '\0' && *ptr != '\n' && *ptr != '\r' )
432 CV_PARSE_ERROR_CPP( "Invalid character in the stream" );
433 ptr = fs->gets();
434 if( !ptr || *ptr == '\0' )
435 break;
436 }
437 }
438 return ptr;
439 }

461 char* parseValue( char* ptr, FileNode& node )
462 {
463 if (!ptr)
464 CV_PARSE_ERROR_CPP("Invalid input");
465
466 FileNode new_elem;
467 bool have_space = true;
468 int value_type = node.type();
469 std::string key, key2, type_name;
470
471 for(;;)
472 {
473 char c = *ptr, d;
474 char* endptr;
475
476 // FIXIT ptr[1], ptr[2] - out of bounds read without check or data fetch (#11061)
477 if( cv_isspace(c) || c == '\0' ||
478 (c == '<' && ptr[1] == '!' && ptr[2] == '-') )
479 {
480 ptr = skipSpaces( ptr, 0 );
481 if (!ptr)
482 CV_PARSE_ERROR_CPP("Invalid input");
483 have_space = true;
484 c = *ptr;
485 }
486
487 d = ptr[1]; // FIXIT ptr[1] - out of bounds read without check or data fetch (#11061)
488
489 if( c =='<' || c == '\0' )
490 {
491 int tag_type = 0;
492 int elem_type = FileNode::NONE;
493
494 if( d == '/' || c == '\0' )
495 break;
496
497 ptr = parseTag( ptr, key, type_name, tag_type );
498
499 if( tag_type == CV_XML_DIRECTIVE_TAG )
500 CV_PARSE_ERROR_CPP( "Directive tags are not allowed here" );
501 if( tag_type == CV_XML_EMPTY_TAG )
502 CV_PARSE_ERROR_CPP( "Empty tags are not supported" );
503
504 CV_Assert(tag_type == CV_XML_OPENING_TAG);
505
506 /* for base64 string */
507 bool binary_string = false;
508
509 if( !type_name.empty() )
510 {
511 const char* tn = type_name.c_str();
512 if( strcmp(tn, "str") == 0 )
513 elem_type = FileNode::STRING;
514 else if( strcmp( tn, "map" ) == 0 )
515 elem_type = FileNode::MAP;
516 else if( strcmp( tn, "seq" ) == 0 )
517 elem_type = FileNode::SEQ;
518 else if( strcmp( tn, "binary") == 0)
519 binary_string = true;
520 }
521
522 new_elem = fs->addNode(node, key, elem_type, 0);
523 if (!binary_string)
524 ptr = parseValue(ptr, new_elem);
525 else
526 {
527 ptr = fs->parseBase64( ptr, 0, new_elem);
528 ptr = skipSpaces( ptr, 0 );
529 if (!ptr)
530 CV_PARSE_ERROR_CPP("Invalid input");
531 }
532
533 ptr = parseTag( ptr, key2, type_name, tag_type );
534 if( tag_type != CV_XML_CLOSING_TAG || key2 != key )
535 CV_PARSE_ERROR_CPP( "Mismatched closing tag" );
536 have_space = true;
537 }
538 else
539 {
540 if( !have_space )
541 CV_PARSE_ERROR_CPP( "There should be space between literals" );
542
543 FileNode* elem = &node;
544 if( node.type() != FileNode::NONE )
545 {
546 fs->convertToCollection( FileNode::SEQ, node );
547 new_elem = fs->addNode(node, std::string(), FileNode::NONE, 0);
548 elem = &new_elem;
549 }
550
551 if( value_type != FileNode::STRING &&
552 (cv_isdigit(c) || ((c == '-' || c == '+') &&
553 (cv_isdigit(d) || d == '.')) || (c == '.' && cv_isalnum(d))) ) // a number
554 {
555 endptr = ptr + (c == '-' || c == '+');
556 while( cv_isdigit(*endptr) )
557 endptr++;
558 if( *endptr == '.' || *endptr == 'e' )
559 {
560 double fval = fs->strtod( ptr, &endptr );
561 elem->setValue(FileNode::REAL, &fval);
562 }
563 else
564 {
565 int64_t ival = strtoll( ptr, &endptr, 0 );
566 elem->setValue(FileNode::INT, &ival);
567 }
568
569 if( endptr == ptr )
570 CV_PARSE_ERROR_CPP( "Invalid numeric value (inconsistent explicit type specification?)" );
571
572 ptr = endptr;
573 CV_PERSISTENCE_CHECK_END_OF_BUFFER_BUG_CPP();
574 }
575 else
576 {
577 // string
578 int i = 0, len, is_quoted = 0;
579 if( c == '\"' )
580 is_quoted = 1;
581 else
582 --ptr;
583 strbuf[0] = '\0';
584
585 for( ;; )
586 {
587 c = *++ptr;
588 CV_PERSISTENCE_CHECK_END_OF_BUFFER_BUG_CPP();
589
590 if( !cv_isalnum(c) )
591 {
592 if( c == '\"' )
593 {
594 if( !is_quoted )
595 CV_PARSE_ERROR_CPP( "Literal \" is not allowed within a string. Use &quot;" );
596 ++ptr;
597 break;
598 }
599 else if( !cv_isprint(c) || c == '<' || (!is_quoted && cv_isspace(c)))
600 {
601 if( is_quoted )
602 CV_PARSE_ERROR_CPP( "Closing \" is expected" );
603 break;
604 }
605 else if( c == '\'' || c == '>' )
606 {
607 CV_PARSE_ERROR_CPP( "Literal \' or > are not allowed. Use &apos; or &gt;" );
608 }
609 else if( c == '&' )
610 {
611 if( *++ptr == '#' )
612 {
613 int val, base = 10;
614 ptr++;
615 if( *ptr == 'x' )
616 {
617 base = 16;
618 ptr++;
619 }
620 val = (int)strtol( ptr, &endptr, base );
621 if( (unsigned)val > (unsigned)255 ||
622 !endptr || *endptr != ';' )
623 CV_PARSE_ERROR_CPP( "Invalid numeric value in the string" );
624 c = (char)val;
625 }
626 else
627 {
628 endptr = ptr;
629 do c = *++endptr;
630 while( cv_isalnum(c) );
631 if( c != ';' )
632 CV_PARSE_ERROR_CPP( "Invalid character in the symbol entity name" );
633 len = (int)(endptr - ptr);
634 if( len == 2 && memcmp( ptr, "lt", len ) == 0 )
635 c = '<';
636 else if( len == 2 && memcmp( ptr, "gt", len ) == 0 )
637 c = '>';
638 else if( len == 3 && memcmp( ptr, "amp", len ) == 0 )
639 c = '&';
640 else if( len == 4 && memcmp( ptr, "apos", len ) == 0 )
641 c = '\'';
642 else if( len == 4 && memcmp( ptr, "quot", len ) == 0 )
643 c = '\"';
644 else
645 {
646 if (len + 2 + i >= CV_FS_MAX_LEN)
647 CV_PARSE_ERROR_CPP("string is too long");
648 memcpy( strbuf + i, ptr-1, len + 2 );
649 i += len + 2;
650 }
651 }
652 ptr = endptr;
653 CV_PERSISTENCE_CHECK_END_OF_BUFFER_BUG_CPP();
654 }
655 }
656 if (i + 1 >= CV_FS_MAX_LEN)
657 CV_PARSE_ERROR_CPP("Too long string literal");
658 strbuf[i++] = c;
659 }
660 elem->setValue(FileNode::STRING, strbuf, i);
661 }
662
663 if( value_type != FileNode::NONE && value_type != FileNode::SEQ && value_type != FileNode::MAP )
664 break;
665 have_space = false;
666 }
667 }
668 fs->finalizeCollection(node);
669
670 return ptr;
671 }

673 char* parseTag( char* ptr, std::string& tag_name,
674 std::string& type_name, int& tag_type )
675 {
676 if (!ptr)
677 CV_PARSE_ERROR_CPP("Invalid tag input");
678
679 if( *ptr == '\0' )
680 CV_PARSE_ERROR_CPP( "Unexpected end of the stream" );
681
682 if( *ptr != '<' )
683 CV_PARSE_ERROR_CPP( "Tag should start with \'<\'" );
684
685 ptr++;
686 CV_PERSISTENCE_CHECK_END_OF_BUFFER_BUG_CPP();
687
688 if( cv_isalnum(*ptr) || *ptr == '_' )
689 tag_type = CV_XML_OPENING_TAG;
690 else if( *ptr == '/' )
691 {
692 tag_type = CV_XML_CLOSING_TAG;
693 ptr++;
694 }
695 else if( *ptr == '?' )
696 {
697 tag_type = CV_XML_HEADER_TAG;
698 ptr++;
699 }
700 else if( *ptr == '!' )
701 {
702 tag_type = CV_XML_DIRECTIVE_TAG;
703 CV_Assert( ptr[1] != '-' || ptr[2] != '-' );
704 ptr++;
705 }
706 else
707 CV_PARSE_ERROR_CPP( "Unknown tag type" );
708
709 tag_name.clear();
710 type_name.clear();
711
712 for(;;)
713 {
714 char c, *endptr;
715 if( !cv_isalpha(*ptr) && *ptr != '_' )
716 CV_PARSE_ERROR_CPP( "Name should start with a letter or underscore" );
717
718 endptr = ptr - 1;
719 do c = *++endptr;
720 while( cv_isalnum(c) || c == '_' || c == '-' );
721
722 std::string attrname(ptr, (size_t)(endptr - ptr));
723 ptr = endptr;
724 CV_PERSISTENCE_CHECK_END_OF_BUFFER_BUG_CPP();
725
726 if( tag_name.empty() )
727 tag_name = attrname;
728 else
729 {
730 if( tag_type == CV_XML_CLOSING_TAG )
731 CV_PARSE_ERROR_CPP( "Closing tag should not contain any attributes" );
732
733 if( *ptr != '=' )
734 {
735 ptr = skipSpaces( ptr, CV_XML_INSIDE_TAG );
736 if (!ptr)
737 CV_PARSE_ERROR_CPP("Invalid attribute");
738 if( *ptr != '=' )
739 CV_PARSE_ERROR_CPP( "Attribute name should be followed by \'=\'" );
740 }
741
742 c = *++ptr;
743 if( c != '\"' && c != '\'' )
744 {
745 ptr = skipSpaces( ptr, CV_XML_INSIDE_TAG );
746 if(!ptr)
747 CV_PARSE_ERROR_CPP("Invalid attribute value");
748 if( *ptr != '\"' && *ptr != '\'' )
749 CV_PARSE_ERROR_CPP( "Attribute value should be put into single or double quotes" );
750 }
751
752 char quote = *ptr++;
753 endptr = ptr;
754 for(;;)
755 {
756 c = *endptr++;
757 if( c == quote )
758 break;
759 if( c == '\0' )
760 CV_PARSE_ERROR_CPP( "Unexpected end of line" );
761 }
762
763 if( attrname == "type_id" )
764 {
765 CV_Assert( type_name.empty() );
766 type_name = std::string(ptr, (size_t)(endptr - 1 - ptr));
767 }
768
769 ptr = endptr;
770 }
771
772 c = *ptr;
