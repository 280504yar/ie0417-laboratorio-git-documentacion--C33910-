248- (void)setDesiredCameraPosition:(AVCaptureDevicePosition)desiredPosition;
249{
250 for (AVCaptureDevice *device in [AVCaptureDevice devicesWithMediaType:AVMediaTypeVideo]) {
251 if ([device position] == desiredPosition) {
252 [self.captureSession beginConfiguration];
253
254 NSError* error = nil;
255 AVCaptureDeviceInput *input = [AVCaptureDeviceInput deviceInputWithDevice:device error:&error];
256 if (!input) {
257 NSLog(@"error creating input %@", [error description]);
258 }
259
260 // support for autofocus
261 if ([device isFocusModeSupported:AVCaptureFocusModeContinuousAutoFocus]) {
262 error = nil;
263 if ([device lockForConfiguration:&error]) {
264 device.focusMode = AVCaptureFocusModeContinuousAutoFocus;
265 [device unlockForConfiguration];
266 } else {
267 NSLog(@"unable to lock device for autofocus configuration %@", [error description]);
268 }
269 }
270 [self.captureSession addInput:input];
271
272 for (AVCaptureInput *oldInput in self.captureSession.inputs) {
273 [self.captureSession removeInput:oldInput];
274 }
275 [self.captureSession addInput:input];
276 [self.captureSession commitConfiguration];
277
278 break;
279 }
280 }

298 [self createVideoPreviewLayer];
299 } else {
300 [self createCustomVideoPreview];
301 }

304 }

306 [self.captureSession startRunning];
