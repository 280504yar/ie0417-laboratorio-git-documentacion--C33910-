35 CUDA4DNN_CHECK_CUDNN(cudnnCreateActivationDescriptor(&descriptor));
36 try {
37 const auto mode = [type] {
38 switch(type) {
39 case ActivationType::IDENTITY: return CUDNN_ACTIVATION_IDENTITY;
40 case ActivationType::RELU: return CUDNN_ACTIVATION_RELU;
41 case ActivationType::CLIPPED_RELU: return CUDNN_ACTIVATION_CLIPPED_RELU;
42 case ActivationType::SIGMOID: return CUDNN_ACTIVATION_SIGMOID;
43 case ActivationType::TANH: return CUDNN_ACTIVATION_TANH;
44 case ActivationType::ELU: return CUDNN_ACTIVATION_ELU;
45 }
46 CV_Assert(0);
47 return CUDNN_ACTIVATION_IDENTITY;
48 } ();
49
50 CUDA4DNN_CHECK_CUDNN(cudnnSetActivationDescriptor(descriptor, mode, CUDNN_NOT_PROPAGATE_NAN, relu_ceiling_or_elu_alpha));
51 } catch(...) {
52 /* cudnnDestroyActivationDescriptor will not fail for a valid descriptor object */
53 CUDA4DNN_CHECK_CUDNN(cudnnDestroyActivationDescriptor(descriptor));
54 throw;
