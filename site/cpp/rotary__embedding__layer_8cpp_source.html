14static void gather(
15 const uchar* data, const int64_t* pos_ids,
16 const int batch_size, const int seq_len, const int seq_len_max,
17 const int dhalf, const int elem_size,
18 uchar* out
19)
20{
21
22 Range r(0, seq_len * batch_size);
23 bool outOfRangeIdx = false;
24 for (int i = r.start; i < r.end; ++i)
25 {
26 const int t = i % seq_len;
27 const int b = i / seq_len;
28
29 int64_t pos_id = pos_ids[b * seq_len + t];
30 if (pos_id >= seq_len_max)
31 {
32 outOfRangeIdx = true;
33 break;
34 }
35 const uchar* data_in_ptr = data + pos_id * dhalf * elem_size;
36 uchar* out_ptr = out + ( b * seq_len + t ) * dhalf * elem_size;
37
38 memcpy(out_ptr, data_in_ptr, dhalf * elem_size);
39 }
40
41 if (outOfRangeIdx)
42 {
43 CV_Error(Error::StsOutOfRange, "some of indices are outside of range");
44 }
45}

47static void rotate(
48 const float*data_in, float* data_out,
49 const float*cos_cache, const float*sin_cache,
50 const int batch_size, const int seq_len, const int n_heads, const int dim_head,
51 const int rotary_dim, const bool is_data_4d
52){
53 auto fn = [&](const Range &r) {
54 const int dhalf = rotary_dim / 2;
55 for (int i = r.start; i < r.end; ++i)
56 {
57 const int t = i % seq_len;
58 const int b = i / (n_heads * seq_len);
59 const int n = (i - b * n_heads * seq_len - t) / seq_len;
60
61 const int offset = (is_data_4d ?
62 seq_len * (b * n_heads + n ) + t :
63 n_heads * (b * seq_len + t ) + n) * dim_head;
64
65 const float* real = data_in + offset;
66 const float* imag = data_in + offset + dhalf;
67
68 const float*sin_cache_ptr = sin_cache + (b * seq_len + t) * dhalf;
69 const float*cos_cache_ptr = cos_cache + (b * seq_len + t) * dhalf;
70
71 float* out_real = data_out + offset;
72 float* out_imag = data_out + offset + dhalf;
73
74 int d = 0;
75 #if (CV_SIMD || CV_SIMD_SCALABLE)
76 const size_t w = VTraits<v_float32>::vlanes();
77 for (; d + w <= dhalf; d += w)
78 {
79 // Alignment requirement: if CV_STRONG_ALIGNMENT=1 then passed pointer must be aligned (sizeof(lane type) should be enough
80 // should be fullfilled by design
81 v_float32 vreal = vx_load(real + d);
82 v_float32 vimag = vx_load(imag + d);
83 v_float32 vsin = vx_load(sin_cache_ptr + d);
84 v_float32 vcos = vx_load(cos_cache_ptr + d);
85
86 v_float32 vout_real = v_sub(v_mul(vcos, vreal), v_mul(vsin, vimag));
87 v_float32 vout_imag = v_add(v_mul(vsin, vreal), v_mul(vcos, vimag));
88
89 v_store(out_real + d, vout_real);
90 v_store(out_imag + d, vout_imag);
91 }
92 #endif
93 // scalar tail
94 for (; d < dhalf; ++d)
95 {
96 float r = real[d];
97 float im = imag[d];
98 float s = sin_cache_ptr[d];
99 float c = cos_cache_ptr[d];
100 out_real[d] = r * c - im * s;
101 out_imag[d] = im * c + r * s;
102 }
103
104 // copy not rotated part
105 if (dim_head > rotary_dim)
106 {
107 const float* data_in_ptr = data_in + offset + rotary_dim;
108 float* out_ptr = data_out + offset + rotary_dim;
109 memcpy(out_ptr, data_in_ptr, (dim_head - rotary_dim) * sizeof(float));
110 }
111 }
112 };
113
114 const size_t loops = n_heads * seq_len * batch_size;
115 double nstripes = loops * dim_head * (1 / 1024.0);
116 parallel_for_(Range(0, loops), fn, nstripes);
117}

119static void rotate_interleaved(
120 const float*data_in, float* data_out,
121 const float*cos_cache, const float*sin_cache,
122 const int batch_size, const int seq_len, const int n_heads, const int dim_head,
123 const int rotary_dim, const bool is_data_4d
124){
125 auto fn = [&](const Range &r) {
126 const int dhalf = rotary_dim / 2;
127
128 for (int i = r.start; i < r.end; ++i)
129 {
130 const int t = i % seq_len;
131 const int b = i / (n_heads * seq_len);
132 const int n = (i - b * n_heads * seq_len - t) / seq_len;
133
134 const int offset = (is_data_4d ?
135 seq_len * (b * n_heads + n ) + t :
136 n_heads * (b * seq_len + t ) + n) * dim_head;
137
138 const float* data_in_ptr = data_in + offset;
139 float* out_ptr = data_out + offset;
140
141 const float*sin_cache_ptr = sin_cache + (b * seq_len + t) * dhalf;
142 const float*cos_cache_ptr = cos_cache + (b * seq_len + t) * dhalf;
143
144 int d = 0;
145 #if (CV_SIMD || CV_SIMD_SCALABLE)
146 const size_t w = VTraits<v_float32>::vlanes();
147 for (; d + w <= dhalf; d += w)
148 {
149 v_float32 vimag, vreal;
150 v_float32 vsin = vx_load(sin_cache_ptr + d);
151 v_float32 vcos = vx_load(cos_cache_ptr + d);
152 v_load_deinterleave(data_in_ptr + 2*d, vreal, vimag);
153
154 v_float32 vout_real = v_sub(v_mul(vcos, vreal), v_mul(vsin, vimag));
155 v_float32 vout_imag = v_add(v_mul(vsin, vreal), v_mul(vcos, vimag));
156
157 v_store_interleave(out_ptr + 2*d, vout_real, vout_imag);
158 }
159 #endif
160 // scalar tail
161 for (; d < dhalf; ++d)
162 {
163 float r = data_in_ptr[2*d];
164 float im = data_in_ptr[2*d + 1];
165 float s = sin_cache_ptr[d];
166 float c = cos_cache_ptr[d];
167 out_ptr[2*d] = r * c - im * s;
168 out_ptr[2*d + 1] = im * c + r * s;
169 }
170 // copy not rotated part
171 if (dim_head > rotary_dim)
172 {
173 const float* data_in_ptr_tail = data_in_ptr + rotary_dim;
174 float* out_ptr_tail = out_ptr + rotary_dim;
175 memcpy(out_ptr_tail, data_in_ptr_tail, (dim_head - rotary_dim) * sizeof(float));
176 }
177 }
178 };
179
180 const size_t loops = n_heads * seq_len * batch_size;
181 double nstripes = loops * dim_head * (1 / 1024.0);
182 parallel_for_(Range(0, loops), fn, nstripes);
183}

214 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
215 const int requiredOutputs,
216 std::vector<MatShape> &outputs,
217 std::vector<MatShape> &internals) const CV_OVERRIDE {
218
219 CV_CheckTrue(inputs.size() >= 3, "RotaryEmbeddingLayer: at least three inputs are required");
220
221 const MatShape& x_input_shape = inputs[0];
222 CV_CheckTrue(
223 x_input_shape.dims == 4 || num_heads > -1,
224 "RotaryEmbeddingLayer: input must have 4 dimensions or num_heads must be specified"
225 );
226 const MatShape& cos_cache_shape = inputs[1];
227 const MatShape& sin_cache_shape = inputs[2];
228 CV_CheckTrue(
229 cos_cache_shape.dims == sin_cache_shape.dims,
230 "RotaryEmbeddingLayer: cos_cache and sin_cache must have the same number of dimensions"
231 );
232 CV_CheckTrue(
233 cos_cache_shape.dims == 3 || inputs.size() > 3,
234 "RotaryEmbeddingLayer: provide position_ids or specify sin_cache and cos_cache in format BxTxD"
235 );
236 CV_CheckTrue(
237 cos_cache_shape.dims == sin_cache_shape.dims,
238 "RotaryEmbeddingLayer: cos_cache and sin_cache must have the same number of dimensions"
239 );
240 for (int i = 0; i < cos_cache_shape.dims; ++i)
241 {
242 CV_CheckTrue(
243 cos_cache_shape[i] == sin_cache_shape[i],
244 "RotaryEmbeddingLayer: cos_cache and sin_cache must have the same shape"
245 );
246 }
247
248 outputs.assign(1, inputs[0]);
249
250 const bool do_gather = inputs.size() > 3;
251 if (do_gather)
252 {
253 CV_CheckTrue(
254 inputs[3].dims == 2,
255 "RotaryEmbeddingLayer: position_ids must have 2 dimensions (BxT)"
256 );
257 CV_CheckTrue(
258 inputs[1].dims == 2 && inputs[2].dims == 2,
259 "RotaryEmbeddingLayer: when using position_ids, sin_cache and cos_cache must have 2 dimensions (TxD)"
260 );
261 const int batch_size = static_cast<int>(inputs[3][0]);
262 const int seq_len = static_cast<int>(inputs[3][1]);
263 const int d_half = static_cast<int>(cos_cache_shape[cos_cache_shape.dims - 1]);
264 // cos
265 internals.push_back(MatShape{batch_size, seq_len, d_half});
266 // sin
267 internals.push_back(MatShape{batch_size, seq_len, d_half});
268 } else {
269 CV_CheckTrue(
270 cos_cache_shape.dims == 3,
271 "RotaryEmbeddingLayer: sin_cache and cos_cache must have 3 dimensions (BxTxD) when position_ids are not provided"
272 );
273 }
274
275 if (x_input_shape.dims == 4)
276 {
277 CV_CheckTrue(x_input_shape[3] % 2 == 0,
278 "RotaryEmbeddingLayer: head size must be even");
279 } else if( x_input_shape.dims == 3 ) {
280 CV_CheckTrue(num_heads > 0, "RotaryEmbeddingLayer: num_heads must be provided for 3d input");
281 CV_CheckTrue(x_input_shape[2] % num_heads == 0,
282 "RotaryEmbeddingLayer: input's last dimension must be divisible by num_heads");
283 }
284 return true;
285 }

287 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE {
288 std::vector<Mat> inputs, outputs, internals;
289 inputs_arr.getMatVector(inputs);
290 outputs_arr.getMatVector(outputs);
291 internals_arr.getMatVector(internals);
292 const bool do_gather = inputs.size() > 3;
293
294 if (num_heads == -1)
295 // if input is 3d and num_heads is not given (-1),
296 // getMemoryShapes throws error
297 // so here input must be 4d
298 num_heads = inputs[0].size[1];
299
300 const Mat& input = inputs[0];
301 Mat& output = outputs[0];
302
303 const Mat& cos_cache = do_gather ? internals[0] : inputs[1];
304 const Mat& sin_cache = do_gather ? internals[1] : inputs[2];
305
306 const bool is_data_4d = input.dims == 4;
307 const int dim_head = is_data_4d ? input.size[3] : (input.size[2] / num_heads);
308 const int rotary_dim = rotary_embedding_dim > 0 ? rotary_embedding_dim : dim_head;
309 CV_CheckTrue(rotary_dim % 2 == 0, "RotaryEmbeddingLayer: rotary_dim must be even");
310 const int seq_len = is_data_4d ? input.size[2] : input.size[1];
311
312 if (do_gather)
313 {
314 const Mat& position_ids = inputs[3];
315 const int batch_size = position_ids.size[0];
316 const int seq_len = position_ids.size[1];
317 const int seq_len_max = inputs[1].size[0];
318 const int dhalf = rotary_dim / 2;
319 const Mat* caches[2] = { &inputs[1], &inputs[2] };
320 for (int i = 0; i < 2; ++i)
321 {
322 gather(
323 caches[i]->ptr<uchar>(), position_ids.ptr<int64_t>(),
324 batch_size, seq_len, seq_len_max,
325 dhalf, sizeof(float),
326 internals[i].ptr<uchar>()
327 );
328 }
329 }
330
331 if (interleaved)
332 {
333 rotate_interleaved(
334 input.ptr<const float>(), output.ptr<float>(),
335 cos_cache.ptr<const float>(), sin_cache.ptr<const float>(),
336 input.size[0], seq_len, num_heads, dim_head,
337 rotary_dim, is_data_4d
338 );
339 }
340 else
341 {
342 rotate(
343 input.ptr<const float>(), output.ptr<float>(),
344 cos_cache.ptr<const float>(), sin_cache.ptr<const float>(),
345 input.size[0], seq_len, num_heads, dim_head,
346 rotary_dim, is_data_4d
347 );
348 }
349 }
350
351 private:
352 int num_heads = -1;
353 int interleaved;
354 int rotary_embedding_dim = -1;
355};
