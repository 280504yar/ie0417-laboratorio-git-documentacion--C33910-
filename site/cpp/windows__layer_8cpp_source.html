31template<typename T>
32static void BlackmanWindowFill(Mat& out, int N, double N1, bool useDouble)
33{
34 CV_Assert(out.dims == 1);
35 CV_Assert((int)out.total() == N);
36
37 if (useDouble)
38 {
39 const double alpha = 0.42;
40 const double beta = 0.08;
41 const double pi = CV_PI;
42 const double coeff1 = (2.0 * pi) / N1;
43 const double coeff2 = (4.0 * pi) / N1;
44
45 cv::AutoBuffer<double> _w(N);
46 double* w = _w.data();
47
48 int i = 0;
49#if CV_SIMD_64F || CV_SIMD_SCALABLE_64F
50 const int nlanes64 = VTraits<v_float64>::vlanes();
51 const int max_nlanes = VTraits<v_float64>::max_nlanes;
52 std::array<double, max_nlanes> index;
53 std::iota(index.data(), index.data() + max_nlanes, 0.0);
54 v_float64 vindex = vx_load(index.data());
55 v_float64 delta = vx_setall_f64(nlanes64);
56 v_float64 vcoeff1 = vx_setall_f64(coeff1);
57 v_float64 vcoeff2 = vx_setall_f64(coeff2);
58 v_float64 valpha = vx_setall_f64(alpha);
59 v_float64 vbeta = vx_setall_f64(beta);
60 v_float64 vnegHalf = vx_setall_f64(-0.5);
61
62 for (; i <= N - nlanes64; i += nlanes64)
63 {
64 v_float64 varg1 = v_mul(vcoeff1, vindex);
65 v_float64 varg2 = v_mul(vcoeff2, vindex);
66 v_float64 vc1 = v_cos(varg1);
67 v_float64 vc2 = v_cos(varg2);
68 v_float64 v = v_add(valpha, v_add(v_mul(vc1, vnegHalf), v_mul(vc2, vbeta)));
69 vx_store(w + i, v);
70 vindex = v_add(vindex, delta);
71 }
72#endif
73
74 for (; i < N; ++i)
75 {
76 double arg1 = coeff1 * i;
77 double arg2 = coeff2 * i;
78 double v = std::cos(arg1) * (-0.5);
79 v += std::cos(arg2) * beta;
80 v += alpha;
81 w[i] = v;
82 }
83
84 T* dst = out.ptr<T>();
85 for (int n = 0; n < N; ++n)
86 dst[n] = saturate_cast<T>(w[n]);
87 }
88 else
89 {
90 const float alpha = 0.42f;
91 const float beta = 0.08f;
92 const float pi = (float)CV_PI;
93 const float coeff1 = (2.0f * pi) / (float)N1;
94 const float coeff2 = (4.0f * pi) / (float)N1;
95
96 cv::AutoBuffer<float> _w(N);
97 float* w = _w.data();
98
99 int i = 0;
100#if (defined(CV_SIMD_32F) && CV_SIMD_32F) || (defined(CV_SIMD_SCALABLE_32F) && CV_SIMD_SCALABLE_32F)
101 const int nlanes32 = VTraits<v_float32>::vlanes();
102 const int max_nlanes32 = VTraits<v_float32>::max_nlanes;
103 std::array<float, max_nlanes32> index;
104 std::iota(index.data(), index.data() + max_nlanes32, 0.0f);
105 v_float32 vindex = vx_load(index.data());
106 v_float32 delta = vx_setall_f32(nlanes32);
107 v_float32 vcoeff1 = vx_setall_f32(coeff1);
108 v_float32 vcoeff2 = vx_setall_f32(coeff2);
109 v_float32 valpha = vx_setall_f32(alpha);
110 v_float32 vbeta = vx_setall_f32(beta);
111 v_float32 vnegHalf = vx_setall_f32(-0.5f);
112
113 for (; i <= N - nlanes32; i += nlanes32)
114 {
115 v_float32 varg1 = v_mul(vcoeff1, vindex);
116 v_float32 varg2 = v_mul(vcoeff2, vindex);
117 v_float32 vc1 = v_cos(varg1);
118 v_float32 vc2 = v_cos(varg2);
119 v_float32 v = v_add(valpha, v_add(v_mul(vc1, vnegHalf), v_mul(vc2, vbeta)));
120 vx_store(w + i, v);
121 vindex = v_add(vindex, delta);
122 }
123#endif
124
125 for (; i < N; ++i)
126 {
127 float arg1 = coeff1 * i;
128 float arg2 = coeff2 * i;
129 float v = std::cos(arg1) * (-0.5f);
130 v += std::cos(arg2) * beta;
131 v += alpha;
132 w[i] = v;
133 }
134
135 T* dst = out.ptr<T>();
136 for (int n = 0; n < N; ++n)
137 dst[n] = saturate_cast<T>(w[n]);
138 }
139}

141template<typename T>
142static void HannWindowFill(Mat& out, int N, double N1, bool useDouble)
143{
144 CV_Assert(out.dims == 1);
145 CV_Assert((int)out.total() == N);
146
147 if (useDouble)
148 {
149 cv::AutoBuffer<double> _w(N);
150 double* w = _w.data();
151
152 const double pi = CV_PI;
153 const double coeff = (2.0 * pi) / N1;
154
155 int i = 0;
156#if CV_SIMD_64F || CV_SIMD_SCALABLE_64F
157 const int nlanes64 = VTraits<v_float64>::vlanes();
158 const int max_nlanes = VTraits<v_float64>::max_nlanes;
159 std::array<double, max_nlanes> index;
160 std::iota(index.data(), index.data() + max_nlanes, 0.0);
161 v_float64 vindex = vx_load(index.data());
162 v_float64 delta = vx_setall_f64(nlanes64);
163 v_float64 vcoeff = vx_setall_f64(coeff);
164 v_float64 one = vx_setall_f64(1.0);
165 v_float64 half = vx_setall_f64(0.5);
166
167 for (; i <= N - nlanes64; i += nlanes64)
168 {
169 v_float64 v = v_mul(half, v_sub(one, v_cos(v_mul(vcoeff, vindex))));
170 vx_store(w + i, v);
171 vindex = v_add(vindex, delta);
172 }
173#endif
174
175 for (; i < N; ++i)
176 w[i] = 0.5 * (1.0 - std::cos(coeff * i));
177
178 T* dst = out.ptr<T>();
179 for (int n = 0; n < N; ++n)
180 dst[n] = saturate_cast<T>(w[n]);
181 }
182 else
183 {
184 const float pi = (float)CV_PI;
185 const float coeff = (2.0f * pi) / (float)N1;
186
187 cv::AutoBuffer<float> _w(N);
188 float* w = _w.data();
189
190 int i = 0;
191#if (defined(CV_SIMD_32F) && CV_SIMD_32F) || (defined(CV_SIMD_SCALABLE_32F) && CV_SIMD_SCALABLE_32F)
192 const int nlanes32 = VTraits<v_float32>::vlanes();
193 const int max_nlanes32 = VTraits<v_float32>::max_nlanes;
194 std::array<float, max_nlanes32> index;
195 std::iota(index.data(), index.data() + max_nlanes32, 0.0f);
196 v_float32 vindex = vx_load(index.data());
197 v_float32 delta = vx_setall_f32(nlanes32);
198 v_float32 vcoeff = vx_setall_f32(coeff);
199 v_float32 one = vx_setall_f32(1.0f);
200 v_float32 half = vx_setall_f32(0.5f);
201
202 for (; i <= N - nlanes32; i += nlanes32)
203 {
204 v_float32 v = v_mul(half, v_sub(one, v_cos(v_mul(vcoeff, vindex))));
205 vx_store(w + i, v);
206 vindex = v_add(vindex, delta);
207 }
208#endif
209
210 for (; i < N; ++i)
211 w[i] = 0.5f * (1.0f - std::cos(coeff * i));
212
213 T* dst = out.ptr<T>();
214 for (int n = 0; n < N; ++n)
215 dst[n] = saturate_cast<T>(w[n]);
216 }
217}

219template<typename T>
220static void HammingWindowFill(Mat& out, int N, double N1, bool useDouble)
221{
222 CV_Assert(out.dims == 1);
223 CV_Assert((int)out.total() == N);
224
225 if (useDouble)
226 {
227 const double alpha = 25.0 / 46.0;
228 const double beta = 1.0 - alpha;
229 const double pi = CV_PI;
230 const double coeff = (2.0 * pi) / N1;
231
232 cv::AutoBuffer<double> _w(N);
233 double* w = _w.data();
234
235 int i = 0;
236#if CV_SIMD_64F || CV_SIMD_SCALABLE_64F
237 const int nlanes64 = VTraits<v_float64>::vlanes();
238 const int max_nlanes = VTraits<v_float64>::max_nlanes;
239 std::array<double, max_nlanes> index;
240 std::iota(index.data(), index.data() + max_nlanes, 0.0);
241 v_float64 vindex = vx_load(index.data());
242 v_float64 delta = vx_setall_f64(nlanes64);
243 v_float64 vcoeff = vx_setall_f64(coeff);
244 v_float64 valpha = vx_setall_f64(alpha);
245 v_float64 vbeta = vx_setall_f64(beta);
246
247 for (; i <= N - nlanes64; i += nlanes64)
248 {
249 v_float64 varg = v_mul(vcoeff, vindex);
250 v_float64 vc = v_cos(varg);
251 v_float64 v = v_sub(valpha, v_mul(vc, vbeta));
252 vx_store(w + i, v);
253 vindex = v_add(vindex, delta);
254 }
255#endif
256
257 for (; i < N; ++i)
258 {
259 double arg = coeff * i;
260 w[i] = alpha - std::cos(arg) * beta;
261 }
262
263 T* dst = out.ptr<T>();
264 for (int n = 0; n < N; ++n)
265 dst[n] = saturate_cast<T>(w[n]);
266 }
267 else
268 {
269 const float alpha = (float)(25.0 / 46.0);
270 const float beta = 1.0f - alpha;
271 const float pi = (float)CV_PI;
272 const float coeff = (2.0f * pi) / (float)N1;
273
274 cv::AutoBuffer<float> _w(N);
275 float* w = _w.data();
276
277 int i = 0;
278#if (defined(CV_SIMD_32F) && CV_SIMD_32F) || (defined(CV_SIMD_SCALABLE_32F) && CV_SIMD_SCALABLE_32F)
279 const int nlanes32 = VTraits<v_float32>::vlanes();
280 const int max_nlanes32 = VTraits<v_float32>::max_nlanes;
281 std::array<float, max_nlanes32> index;
282 std::iota(index.data(), index.data() + max_nlanes32, 0.0f);
283 v_float32 vindex = vx_load(index.data());
284 v_float32 delta = vx_setall_f32(nlanes32);
285 v_float32 vcoeff = vx_setall_f32(coeff);
286 v_float32 valpha = vx_setall_f32(alpha);
287 v_float32 vbeta = vx_setall_f32(beta);
288
289 for (; i <= N - nlanes32; i += nlanes32)
290 {
291 v_float32 varg = v_mul(vcoeff, vindex);
292 v_float32 vc = v_cos(varg);
293 v_float32 v = v_sub(valpha, v_mul(vc, vbeta));
294 vx_store(w + i, v);
295 vindex = v_add(vindex, delta);
296 }
297#endif
298
299 for (; i < N; ++i)
300 {
301 float arg = coeff * i;
302 w[i] = alpha - std::cos(arg) * beta;
303 }
304
305 T* dst = out.ptr<T>();
306 for (int n = 0; n < N; ++n)
307 dst[n] = saturate_cast<T>(w[n]);
308 }

311template<typename BaseLayer>
312class WindowLayerImpl CV_FINAL : public BaseLayer
313{
314public:
315 WindowLayerImpl(const LayerParams& params, WindowKind kind_)
316 : outputType(onnxDataTypeToCV(static_cast<OnnxDataType>(
317 params.get<int>("output_type", static_cast<int>(ONNX_FLOAT))))),
318 periodic(params.get<int>("periodic", 1)),
319 kind(kind_)
320 {
321 this->setParamsFrom(params);
322 }
323
324 bool supportBackend(int backendId) CV_OVERRIDE
325 {
326 return backendId == DNN_BACKEND_OPENCV;
327 }
328
329 bool dynamicOutputShapes() const CV_OVERRIDE
330 {
331 return true;
332 }
333
334 bool getMemoryShapes(const std::vector<MatShape>& inputs,
335 const int /*requiredOutputs*/,
336 std::vector<MatShape>& outputs,
337 std::vector<MatShape>& internals) const CV_OVERRIDE
338 {
339 CV_Assert(inputs.size() == 1);
340 outputs.assign(1, MatShape());
341 internals.clear();
342 return true;
343 }
344
345 void getTypes(const std::vector<MatType>& /*inputs*/,
346 const int requiredOutputs,
347 const int requiredInternals,
348 std::vector<MatType>& outputs,
349 std::vector<MatType>& internals) const CV_OVERRIDE
350 {
351 int t = outputType;
352 if (t < 0)
353 t = CV_32F;
354
355 outputs.assign(requiredOutputs, MatType(t));
356 internals.assign(requiredInternals, MatType(-1));
357 }
358
359 void forward(InputArrayOfArrays inputs_arr,
360 OutputArrayOfArrays outputs_arr,
361 OutputArrayOfArrays /*internals_arr*/) CV_OVERRIDE
362 {
363 forwardImpl(this->name, outputType, inputs_arr, outputs_arr);
364 }
365
366private:
367 void forwardImpl(const String& name,
368 int outputType_,
369 InputArrayOfArrays inputs_arr,
370 OutputArrayOfArrays outputs_arr) const
371 {
372 CV_TRACE_FUNCTION();
373 CV_TRACE_ARG_VALUE(layer_name, "name", name.c_str());
374
375 Size sz = inputs_arr.size();
376 CV_Assert(sz.area() == 1);
377
378 Mat sizeTensor = inputs_arr.getMat(0);
379
380 CV_Assert(!sizeTensor.empty());
381 CV_Assert(sizeTensor.total() == (size_t)1);
382
383 int64_t size64 = 0;
384 tensorToScalar(sizeTensor, CV_64S, &size64);
385
386 const char* layerPrefix = nullptr;
387 switch (kind)
388 {
389 case WindowKind::Blackman: layerPrefix = "BlackmanWindow"; break;
390 case WindowKind::Hann: layerPrefix = "HannWindow"; break;
391 case WindowKind::Hamming: layerPrefix = "HammingWindow"; break;
392 default: layerPrefix = "Window"; break;
393 }
394
395 if (size64 <= 0)
396 CV_Error(Error::StsOutOfRange, String(layerPrefix) + ": size must be > 0");
397 int N = saturate_cast<int>(size64);
398
399 int outType = outputType_ >= 0 ? outputType_ : CV_32F;
400 int depth = CV_MAT_DEPTH(outType);
401
402 MatShape outShape(1);
403 outShape[0] = N;
404
405 int outKind = outputs_arr.kind();
406 Mat Y;
407 std::vector<Mat>* out_mats = nullptr;
408 std::vector<UMat>* out_umats = nullptr;
409
410 if (outKind == _InputArray::STD_VECTOR_MAT)
411 {
412 out_mats = &outputs_arr.getMatVecRef();
413 out_mats->resize(1);
414 out_mats->at(0).fit(outShape, outType);
415 Y = out_mats->at(0);
416 }
417 else if (outKind == _InputArray::STD_VECTOR_UMAT)
418 {
419 out_umats = &outputs_arr.getUMatVecRef();
420 out_umats->resize(1);
421 out_umats->at(0).fit(outShape, outType);
422 Y = Mat(outShape, outType);
423 }
424 else
425 {
426 CV_Error(Error::StsNotImplemented, String(layerPrefix) + ": unsupported output kind");
427 }
428
429 const double N1 = (periodic == 0) ? std::max(N - 1, 1) : std::max(N, 1);
430
431 switch (kind)
432 {
433 case WindowKind::Blackman:
434 switch (depth)
435 {
436 case CV_64F: BlackmanWindowFill<double>(Y, N, N1, true); break;
437 case CV_32F: BlackmanWindowFill<float>(Y, N, N1, false); break;
438 case CV_16F: BlackmanWindowFill<hfloat>(Y, N, N1, false); break;
439 case CV_16BF: BlackmanWindowFill<bfloat>(Y, N, N1, false); break;
440 default:
441 CV_Error(Error::BadDepth, "BlackmanWindow: unsupported output depth");
442 }
443 break;
444 case WindowKind::Hann:
445 switch (depth)
446 {
447 case CV_64F: HannWindowFill<double>(Y, N, N1, true); break;
448 case CV_32F: HannWindowFill<float>(Y, N, N1, false); break;
449 case CV_16F: HannWindowFill<hfloat>(Y, N, N1, false); break;
450 case CV_16BF: HannWindowFill<bfloat>(Y, N, N1, false); break;
451 default:
452 CV_Error(Error::BadDepth, "HannWindow: unsupported output depth");
453 }
454 break;
455 case WindowKind::Hamming:
456 switch (depth)
457 {
458 case CV_64F: HammingWindowFill<double>(Y, N, N1, true); break;
459 case CV_32F: HammingWindowFill<float>(Y, N, N1, false); break;
460 case CV_16F: HammingWindowFill<hfloat>(Y, N, N1, false); break;
461 case CV_16BF: HammingWindowFill<bfloat>(Y, N, N1, false); break;
462 default:
463 CV_Error(Error::BadDepth, "HammingWindow: unsupported output depth");
464 }
465 break;
466 default:
467 CV_Error(Error::StsInternal, "Unknown WindowKind");
468 }
469
470 if (outKind == _InputArray::STD_VECTOR_UMAT)
471 {
472 Y.copyTo(out_umats->at(0));
473 }
474 }
475
476 int outputType;
477 int periodic;
478 WindowKind kind;
479};
