10template<typename T1, typename T2> void
11convertData_(const void* _from, void* _to, int cn)
12{
13 const T1* from = (const T1*)_from;
14 T2* to = (T2*)_to;
15 if( cn == 1 )
16 *to = saturate_cast<T2>(*from);
17 else
18 for( int i = 0; i < cn; i++ )
19 to[i] = saturate_cast<T2>(from[i]);
20}

22template<typename T1, typename T2> void
23convertScaleData_(const void* _from, void* _to, int cn, double alpha, double beta)
24{
25 const T1* from = (const T1*)_from;
26 T2* to = (T2*)_to;
27 if( cn == 1 )
28 *to = saturate_cast<T2>(*from*alpha + beta);
29 else
30 for( int i = 0; i < cn; i++ )
31 to[i] = saturate_cast<T2>(from[i]*alpha + beta);
32}

129static inline void copyElem(const uchar* from, uchar* to, size_t elemSize)
130{
131 size_t i;
132 for( i = 0; i + sizeof(int) <= elemSize; i += sizeof(int) )
133 *(int*)(to + i) = *(const int*)(from + i);
134 for( ; i < elemSize; i++ )
135 to[i] = from[i];
136}

138static inline bool isZeroElem(const uchar* data, size_t elemSize)
139{
140 size_t i;
141 for( i = 0; i + sizeof(int) <= elemSize; i += sizeof(int) )
142 if( *(int*)(data + i) != 0 )
143 return false;
144 for( ; i < elemSize; i++ )
145 if( data[i] != 0 )
146 return false;
147 return true;
148}

273SparseMat::SparseMat(const Mat& m)
274: flags(MAGIC_VAL), hdr(0)
275{
276 create( m.dims, m.size.p, m.type() );
277
278 int i, idx[CV_MAX_DIM] = {0}, d = m.dims, lastSize = m.size[d - 1];
279 size_t esz = m.elemSize();
280 const uchar* dptr = m.ptr();
281
282 for(;;)
283 {
284 for( i = 0; i < lastSize; i++, dptr += esz )
285 {
286 if( isZeroElem(dptr, esz) )
287 continue;
288 idx[d-1] = i;
289 uchar* to = newNode(idx, hash(idx));
290 copyElem( dptr, to, esz );
291 }
292
293 for( i = d - 2; i >= 0; i-- )
294 {
295 dptr += m.step[i] - m.size[i+1]*m.step[i+1];
296 if( ++idx[i] < m.size[i] )
297 break;
298 idx[i] = 0;
299 }
300 if( i < 0 )
301 break;
302 }
303}

305void SparseMat::create(int d, const int* _sizes, int _type)
306{
307 CV_Assert( _sizes && 0 < d && d <= CV_MAX_DIM );
308 for( int i = 0; i < d; i++ )
309 CV_Assert( _sizes[i] > 0 );
310 _type = CV_MAT_TYPE(_type);
311 if( hdr && _type == type() && hdr->dims == d && hdr->refcount == 1 )
312 {
313 int i;
314 for( i = 0; i < d; i++ )
315 if( _sizes[i] != hdr->size[i] )
316 break;
317 if( i == d )
318 {
319 clear();
320 return;
321 }
322 }
323 int _sizes_backup[CV_MAX_DIM]; // #5991
324 if (hdr && _sizes == hdr->size)
325 {
326 for(int i = 0; i < d; i++ )
327 _sizes_backup[i] = _sizes[i];
328 _sizes = _sizes_backup;
329 }
330 release();
331 flags = MAGIC_VAL | _type;
332 hdr = new Hdr(d, _sizes, _type);
333}

374void SparseMat::convertTo( SparseMat& m, int rtype, double alpha ) const
375{
376 int cn = channels();
377 if( rtype < 0 )
378 rtype = type();
379 rtype = CV_MAKETYPE(rtype, cn);
380 if( hdr == m.hdr && rtype != type() )
381 {
382 SparseMat temp;
383 convertTo(temp, rtype, alpha);
384 m = temp;
385 return;
386 }
387
388 CV_Assert(hdr != 0);
389 if( hdr != m.hdr )
390 m.create( hdr->dims, hdr->size, rtype );
391
392 SparseMatConstIterator from = begin();
393 size_t N = nzcount();
394
395 if( alpha == 1 )
396 {
397 ConvertData cvtfunc = getConvertElem(type(), rtype);
398 CV_Assert(cvtfunc);
399 for( size_t i = 0; i < N; i++, ++from )
400 {
401 const Node* n = from.node();
402 uchar* to = hdr == m.hdr ? from.ptr : m.newNode(n->idx, n->hashval);
403 cvtfunc( from.ptr, to, cn );
404 }
405 }
406 else
407 {
408 ConvertScaleData cvtfunc = getConvertScaleElem(type(), rtype);
409 CV_Assert(cvtfunc);
410 for( size_t i = 0; i < N; i++, ++from )
411 {
412 const Node* n = from.node();
413 uchar* to = hdr == m.hdr ? from.ptr : m.newNode(n->idx, n->hashval);
414 cvtfunc( from.ptr, to, cn, alpha, 0 );
415 }
416 }
417}

421{

473 nidx = elem->next;
474 }

476 if( createMissing )

478 int idx[] = { i0 };
479 return newNode( idx, h );
480 }
481 return NULL;

618void SparseMat::resizeHashTab(size_t newsize)
619{
620 newsize = std::max(newsize, (size_t)8);
621 if((newsize & (newsize-1)) != 0)
622 newsize = (size_t)1 << cvCeil(std::log((double)newsize)/CV_LOG2);
623
624 size_t hsize = hdr->hashtab.size();
625 std::vector<size_t> _newh(newsize);
626 size_t* newh = &_newh[0];
627 for( size_t i = 0; i < newsize; i++ )
628 newh[i] = 0;
629 uchar* pool = &hdr->pool[0];
630 for( size_t i = 0; i < hsize; i++ )
631 {
632 size_t nidx = hdr->hashtab[i];
633 while( nidx )
634 {
635 Node* elem = (Node*)(pool + nidx);
636 size_t next = elem->next;
637 size_t newhidx = elem->hashval & (newsize - 1);
638 elem->next = newh[newhidx];
639 newh[newhidx] = nidx;
640 nidx = next;
641 }
642 }
643 hdr->hashtab = _newh;
644}

646uchar* SparseMat::newNode(const int* idx, size_t hashval)
647{
648 const int HASH_MAX_FILL_FACTOR=3;
649 CV_Assert(hdr);
650 size_t hsize = hdr->hashtab.size();
651 if( ++hdr->nodeCount > hsize*HASH_MAX_FILL_FACTOR )
652 {
653 resizeHashTab(std::max(hsize*2, (size_t)8));
654 hsize = hdr->hashtab.size();
655 }
656
657 if( !hdr->freeList )
658 {
659 size_t i, nsz = hdr->nodeSize, psize = hdr->pool.size(),
660 newpsize = std::max(psize*3/2, 8*nsz);
661 newpsize = (newpsize/nsz)*nsz;
662 hdr->pool.resize(newpsize);
663 uchar* pool = &hdr->pool[0];
664 hdr->freeList = std::max(psize, nsz);
665 for( i = hdr->freeList; i < newpsize - nsz; i += nsz )
666 ((Node*)(pool + i))->next = i + nsz;
667 ((Node*)(pool + i))->next = 0;
668 }
669 size_t nidx = hdr->freeList;
670 Node* elem = (Node*)&hdr->pool[nidx];
671 hdr->freeList = elem->next;
672 elem->hashval = hashval;
673 size_t hidx = hashval & (hsize - 1);
674 elem->next = hdr->hashtab[hidx];
675 hdr->hashtab[hidx] = nidx;
676
677 int i, d = hdr->dims;
678 for( i = 0; i < d; i++ )
679 elem->idx[i] = idx[i];

711double norm( const SparseMat& src, int normType )
712{
713 CV_INSTRUMENT_REGION();
714
715 SparseMatConstIterator it = src.begin();
716
717 size_t i, N = src.nzcount();
718 normType &= NORM_TYPE_MASK;
719 int type = src.type();
720 double result = 0;
721
722 CV_Assert( normType == NORM_INF || normType == NORM_L1 || normType == NORM_L2 );
723
724 if( type == CV_32F )
725 {
726 if( normType == NORM_INF )
727 for( i = 0; i < N; i++, ++it )
728 {
729 CV_Assert(it.ptr);
730 result = std::max(result, std::abs((double)it.value<float>()));
731 }
732 else if( normType == NORM_L1 )
733 for( i = 0; i < N; i++, ++it )
734 {
735 CV_Assert(it.ptr);
736 result += std::abs(it.value<float>());
737 }
738 else
739 for( i = 0; i < N; i++, ++it )
740 {
741 CV_Assert(it.ptr);
742 double v = it.value<float>();
743 result += v*v;
744 }
745 }
746 else if( type == CV_64F )
747 {
748 if( normType == NORM_INF )
749 for( i = 0; i < N; i++, ++it )
750 {
751 CV_Assert(it.ptr);
752 result = std::max(result, std::abs(it.value<double>()));
753 }
754 else if( normType == NORM_L1 )
755 for( i = 0; i < N; i++, ++it )
756 {
757 CV_Assert(it.ptr);
758 result += std::abs(it.value<double>());
759 }
760 else
761 for( i = 0; i < N; i++, ++it )
762 {
763 CV_Assert(it.ptr);
764 double v = it.value<double>();
765 result += v*v;
766 }
767 }
768 else
769 CV_Error( cv::Error::StsUnsupportedFormat, "Only 32f and 64f are supported" );
770
771 if( normType == NORM_L2 )
772 result = std::sqrt(result);
773 return result;
774}

777{
778 CV_INSTRUMENT_REGION();
779
780 SparseMatConstIterator it = src.begin();

782 int type = src.type();

784

790 CV_Assert(it.ptr);

811 for( i = 0; i < N; i++, ++it )
812 {
813 CV_Assert(it.ptr);
814 double v = it.value<double>();
815 if( v < minval )
816 {
817 minval = v;

819 }

822 maxval = v;

828 if( _maxval )
829 *_maxval = maxval;
830 }
831 else
