315int CvCaptureCAM::startCaptureDevice(int cameraNum) {
316 NSAutoreleasePool *localpool = [[NSAutoreleasePool alloc] init];
317
318#if defined(__MAC_OS_X_VERSION_MAX_ALLOWED) && __MAC_OS_X_VERSION_MAX_ALLOWED >= 101400
319 if (@available(macOS 10.14, *))
320 {
321 AVAuthorizationStatus status = [AVCaptureDevice authorizationStatusForMediaType:AVMediaTypeVideo];
322 if (status == AVAuthorizationStatusDenied)
323 {
324 fprintf(stderr, "OpenCV: camera access has been denied. Either run 'tccutil reset Camera' "
325 "command in same terminal to reset application authorization status, "
326 "either modify 'System Preferences -> Security & Privacy -> Camera' "
327 "settings for your application.\n");
328 [localpool drain];
329 return 0;
330 }
331 else if (status != AVAuthorizationStatusAuthorized)
332 {
333 if (!cv::utils::getConfigurationParameterBool("OPENCV_AVFOUNDATION_SKIP_AUTH", false))
334 {
335 fprintf(stderr, "OpenCV: not authorized to capture video (status %ld), requesting...\n", status);
336 [AVCaptureDevice requestAccessForMediaType:AVMediaTypeVideo completionHandler:^(BOOL) { /* we don't care */}];
337 if ([NSThread isMainThread])
338 {
339 // we run the main loop for 0.1 sec to show the message
340 [[NSRunLoop mainRunLoop] runUntilDate:[NSDate dateWithTimeIntervalSinceNow:0.1]];
341 }
342 else
343 {
344 fprintf(stderr, "OpenCV: can not spin main run loop from other thread, set "
345 "OPENCV_AVFOUNDATION_SKIP_AUTH=1 to disable authorization request "
346 "and perform it in your application.\n");
347 }
348 }
349 else
350 {
351 fprintf(stderr, "OpenCV: not authorized to capture video (status %ld), set "
352 "OPENCV_AVFOUNDATION_SKIP_AUTH=0 to enable authorization request or "
353 "perform it in your application.\n", status);
354 }
355 [localpool drain];
356 return 0;
357 }
358 }
359#endif
360
361 // get capture device
362 NSArray *devices = [[AVCaptureDevice devicesWithMediaType: AVMediaTypeVideo]
363 arrayByAddingObjectsFromArray:[AVCaptureDevice devicesWithMediaType:AVMediaTypeMuxed]];
364
365 if ( devices.count == 0 ) {
366 fprintf(stderr, "OpenCV: AVFoundation didn't find any attached Video Input Devices!\n");
367 [localpool drain];
368 return 0;
369 }
370
371 if ( cameraNum < 0 || devices.count <= NSUInteger(cameraNum) ) {
372 fprintf(stderr, "OpenCV: out device of bound (0-%ld): %d\n", devices.count-1, cameraNum);
373 [localpool drain];
374 return 0;
375 }
376
377 // Preserve devices ordering on the system
378 // see AVCaptureDevice::uniqueID property documentation for more info
379 devices = [devices
380 sortedArrayUsingComparator:^NSComparisonResult(AVCaptureDevice *d1,
381 AVCaptureDevice *d2) {
382 return [d1.uniqueID compare:d2.uniqueID];
383 }
384 ];
385
386 mCaptureDevice = devices[cameraNum];
387
388 if ( ! mCaptureDevice ) {
389 fprintf(stderr, "OpenCV: device %d not able to use.\n", cameraNum);
390 [localpool drain];
391 return 0;
392 }
393
394 // get input device
395 NSError *error = nil;
396 mCaptureDeviceInput = [[AVCaptureDeviceInput alloc] initWithDevice: mCaptureDevice
397 error: &error];
398 if ( error ) {
399 fprintf(stderr, "OpenCV: error in [AVCaptureDeviceInput initWithDevice:error:]\n");
400 NSLog(@"OpenCV: %@", error.localizedDescription);
401 [localpool drain];
402 return 0;
403 }
404
405 // create output
406 mCapture = [[CaptureDelegate alloc] init];
407 mCaptureVideoDataOutput = [[AVCaptureVideoDataOutput alloc] init];
408 dispatch_queue_t queue = dispatch_queue_create("cameraQueue", DISPATCH_QUEUE_SERIAL);
409 [mCaptureVideoDataOutput setSampleBufferDelegate: mCapture queue: queue];
410 dispatch_release(queue);
411
412 OSType pixelFormat = kCVPixelFormatType_32BGRA;
413 //OSType pixelFormat = kCVPixelFormatType_422YpCbCr8;
414 NSDictionary *pixelBufferOptions;
415 if (width > 0 && height > 0) {
416 pixelBufferOptions =
417 @{
418 (id)kCVPixelBufferWidthKey: @(1.0*width),
419 (id)kCVPixelBufferHeightKey: @(1.0*height),
420 (id)kCVPixelBufferPixelFormatTypeKey: @(pixelFormat)
421 };
422 } else {
423 pixelBufferOptions =
424 @{
425 (id)kCVPixelBufferPixelFormatTypeKey: @(pixelFormat)
426 };
427 }
428 mCaptureVideoDataOutput.videoSettings = pixelBufferOptions;
429 mCaptureVideoDataOutput.alwaysDiscardsLateVideoFrames = YES;
430
431 // create session
432 mCaptureSession = [[AVCaptureSession alloc] init];
433 mCaptureSession.sessionPreset = AVCaptureSessionPresetMedium;
434 [mCaptureSession addInput: mCaptureDeviceInput];
435 [mCaptureSession addOutput: mCaptureVideoDataOutput];
436
437 [mCaptureSession startRunning];
438
439 // flush old position image
440 grabFrame(1);
441
442 [localpool drain];
443 return 1;
444}

446void CvCaptureCAM::setWidthHeight() {
447 NSMutableDictionary *pixelBufferOptions = [mCaptureVideoDataOutput.videoSettings mutableCopy];
448
449 while ( true ) {
450 // auto matching
451 pixelBufferOptions[(id)kCVPixelBufferWidthKey] = @(1.0*width);
452 pixelBufferOptions[(id)kCVPixelBufferHeightKey] = @(1.0*height);
453 mCaptureVideoDataOutput.videoSettings = pixelBufferOptions;
454
455 // compare matched size and my options
456 CMFormatDescriptionRef format = mCaptureDevice.activeFormat.formatDescription;
457 CMVideoDimensions deviceSize = CMVideoFormatDescriptionGetDimensions(format);
458 if ( deviceSize.width == width && deviceSize.height == height ) {
459 break;
460 }
461
462 // fit my options to matched size
463 width = deviceSize.width;
464 height = deviceSize.height;
465 }
466
467 // flush old size image
468 grabFrame(1);
469
470 [pixelBufferOptions release];
471}

505bool CvCaptureCAM::setProperty(int property_id, double value) {
506 NSAutoreleasePool *localpool = [[NSAutoreleasePool alloc] init];
507
508 bool isSucceeded = false;
509
510 switch (property_id) {
511 case cv::CAP_PROP_FRAME_WIDTH:
512 width = value;
513 settingWidth = 1;
514 if (settingWidth && settingHeight) {
515 setWidthHeight();
516 settingWidth = 0;
517 settingHeight = 0;
518 }
519 isSucceeded = true;
520 break;
521 case cv::CAP_PROP_FRAME_HEIGHT:
522 height = value;
523 settingHeight = 1;
524 if (settingWidth && settingHeight) {
525 setWidthHeight();
526 settingWidth = 0;
527 settingHeight = 0;
528 }
529 isSucceeded = true;
530 break;
531 case cv::CAP_PROP_FPS:
532 if ( [mCaptureDevice lockForConfiguration: NULL] ) {
533 NSArray * ranges = mCaptureDevice.activeFormat.videoSupportedFrameRateRanges;
534 AVFrameRateRange *matchedRange = ranges[0];
535 double minDiff = fabs(matchedRange.maxFrameRate - value);
536 for ( AVFrameRateRange *range in ranges ) {
537 double diff = fabs(range.maxFrameRate - value);
538 if ( diff < minDiff ) {
539 minDiff = diff;
540 matchedRange = range;
541 }
542 }
543 mCaptureDevice.activeVideoMinFrameDuration = matchedRange.minFrameDuration;
544 mCaptureDevice.activeVideoMaxFrameDuration = matchedRange.minFrameDuration;
545 isSucceeded = true;
546 [mCaptureDevice unlockForConfiguration];
547 }
548 break;
549 default:
550 break;
551 }
552
553 [localpool drain];
554 return isSucceeded;
555}

609-(BOOL) grabImageUntilDate: (NSDate *)limit {
610 BOOL isGrabbed = NO;
611 [mHasNewFrame lock];
612
613 if ( mGrabbedPixels ) {
614 CVBufferRelease(mGrabbedPixels);
615 }
616 if ( [mHasNewFrame waitUntilDate: limit] ) {
617 isGrabbed = YES;
618 mGrabbedPixels = CVBufferRetain(mCurrentImageBuffer);
619 }
620
621 [mHasNewFrame unlock];
622 return isGrabbed;

625-(bool) updateImage {
626 if ( ! mGrabbedPixels ) {
627 return false;
628 }
629
630 CVPixelBufferLockBaseAddress(mGrabbedPixels, 0);
631 uchar *baseaddress = reinterpret_cast<uchar*>(CVPixelBufferGetBaseAddress(mGrabbedPixels));
632
633 cv::Size sz { (int)CVPixelBufferGetWidth(mGrabbedPixels), (int)CVPixelBufferGetHeight(mGrabbedPixels) };
634 size_t rowBytes = CVPixelBufferGetBytesPerRow(mGrabbedPixels);
635 OSType pixelFormat = CVPixelBufferGetPixelFormatType(mGrabbedPixels);
636
637 bool res = false;
638 if (rowBytes != 0 && (pixelFormat == kCVPixelFormatType_32BGRA || pixelFormat == kCVPixelFormatType_422YpCbCr8)) {
639 mOutImage.create(sz, CV_8UC3);
640 if ( pixelFormat == kCVPixelFormatType_32BGRA ) {
641 cv::Mat devImage(sz, CV_8UC4, baseaddress, rowBytes);
642 cv::cvtColor(devImage, mOutImage, cv::COLOR_BGRA2BGR);
643 res = true;
644 } else if ( pixelFormat == kCVPixelFormatType_422YpCbCr8 ) {
645 cv::Mat devImage(sz, CV_8UC2, baseaddress, rowBytes);
646 cv::cvtColor(devImage, mOutImage, cv::COLOR_YUV2BGR_UYVY);
647 res = true;
648 }
649 } else {
650 fprintf(stderr, "OpenCV: rowBytes == 0 or unknown pixel format 0x%08X\n", pixelFormat);
651 mOutImage.create(cv::Size(0, 0), mOutImage.type());
652 }
653
654 CVPixelBufferUnlockBaseAddress(mGrabbedPixels, 0);
655 CVBufferRelease(mGrabbedPixels);
656 mGrabbedPixels = NULL;
657
658 return res;
659}

739bool CvCaptureFile::setupReadingAt(CMTime position) {
740 if (mAssetReader) {
741 if (mAssetReader.status == AVAssetReaderStatusReading) {
742 [mAssetReader cancelReading];
743 }
744 [mAssetReader release];
745 mAssetReader = nil;
746 }
747 if (mTrackOutput) {
748 [mTrackOutput release];
749 mTrackOutput = nil;
750 }
751
752 // Capture in a pixel format that can be converted efficiently to the output mode.
753 OSType pixelFormat;
754 if (mMode == CV_CAP_MODE_BGR || mMode == CV_CAP_MODE_RGB) {
755 // For CV_CAP_MODE_BGR, read frames as BGRA (AV Foundation's YUV->RGB conversion is slightly faster than OpenCV's cv::COLOR_YUV2BGR_YV12)
756 // kCVPixelFormatType_32ABGR is reportedly faster on OS X, but OpenCV doesn't have a CV_ABGR2BGR conversion.
757 // kCVPixelFormatType_24RGB is significantly slower than kCVPixelFormatType_32BGRA.
758 pixelFormat = kCVPixelFormatType_32BGRA;
759 mFormat = CV_8UC3;
760 } else if (mMode == CV_CAP_MODE_GRAY) {
761 // For CV_CAP_MODE_GRAY, read frames as 420v (faster than 420f or 422 -- at least for H.264 files)

822 return cv::Mat();

825 NSAutoreleasePool *localpool = [[NSAutoreleasePool alloc] init];
826
827 CVPixelBufferLockBaseAddress(mGrabbedPixels, 0);
828 uchar *baseaddress;
829 size_t rowBytes;

833

837 sz.height = CVPixelBufferGetHeightOfPlane(mGrabbedPixels, 0);

857 outChannels = 3;
858 } else if (mMode == CV_CAP_MODE_GRAY) {
859 outChannels = 1;
860 } else if (mMode == CV_CAP_MODE_YUYV) {
861 outChannels = 2;

864 CVPixelBufferUnlockBaseAddress(mGrabbedPixels, 0);
865 CVBufferRelease(mGrabbedPixels);

871 // (Not all of these conversions are used in production, but they were all tested to find the fastest options.)
872 int deviceChannels;
873 int cvtCode;
874

876 deviceChannels = 4;
877
878 if (mMode == CV_CAP_MODE_BGR) {
879 cvtCode = cv::COLOR_BGRA2BGR;
880 } else if (mMode == CV_CAP_MODE_RGB) {
881 cvtCode = cv::COLOR_BGRA2RGB;
882 } else if (mMode == CV_CAP_MODE_GRAY) {
883 cvtCode = cv::COLOR_BGRA2GRAY;
884 } else {
885 CVPixelBufferUnlockBaseAddress(mGrabbedPixels, 0);
886 CVBufferRelease(mGrabbedPixels);
887 mGrabbedPixels = NULL;
888 fprintf(stderr, "OpenCV: unsupported pixel conversion mode\n");
889 return cv::Mat();
890 }
891 } else if ( pixelFormat == kCVPixelFormatType_24RGB ) {
892 deviceChannels = 3;
893
894 if (mMode == CV_CAP_MODE_BGR) {
895 cvtCode = cv::COLOR_RGB2BGR;
896 } else if (mMode == CV_CAP_MODE_RGB) {
897 cvtCode = -1;
898 } else if (mMode == CV_CAP_MODE_GRAY) {
899 cvtCode = cv::COLOR_RGB2GRAY;
900 } else {
901 CVPixelBufferUnlockBaseAddress(mGrabbedPixels, 0);
902 CVBufferRelease(mGrabbedPixels);
903 mGrabbedPixels = NULL;
904 fprintf(stderr, "OpenCV: unsupported pixel conversion mode\n");
905 return cv::Mat();
906 }
907 } else if ( pixelFormat == kCVPixelFormatType_422YpCbCr8 ) { // 422 (2vuy, UYVY)
908 deviceChannels = 2;
909
910 if (mMode == CV_CAP_MODE_BGR) {
911 cvtCode = cv::COLOR_YUV2BGR_UYVY;
912 } else if (mMode == CV_CAP_MODE_RGB) {
913 cvtCode = cv::COLOR_YUV2RGB_UYVY;
914 } else if (mMode == CV_CAP_MODE_GRAY) {
915 cvtCode = cv::COLOR_YUV2GRAY_UYVY;

920 CVBufferRelease(mGrabbedPixels);

922 fprintf(stderr, "OpenCV: unsupported pixel conversion mode\n");

927 // cvtColor(cv::COLOR_YUV2GRAY_420) is expecting a single buffer with both the Y plane and the CrCb planes.
928 // So, lie about the height of the buffer. cvtColor(cv::COLOR_YUV2GRAY_420) will only read the first 2/3 of it.
929 sz.height = sz.height * 3 / 2;
930 deviceChannels = 1;
931
932 if (mMode == CV_CAP_MODE_BGR) {
933 cvtCode = cv::COLOR_YUV2BGR_YV12;
934 } else if (mMode == CV_CAP_MODE_RGB) {

936 } else if (mMode == CV_CAP_MODE_GRAY) {
937 cvtCode = cv::COLOR_YUV2GRAY_420;
938 } else {
939 CVPixelBufferUnlockBaseAddress(mGrabbedPixels, 0);
940 CVBufferRelease(mGrabbedPixels);
941 mGrabbedPixels = NULL;
942 fprintf(stderr, "OpenCV: unsupported pixel conversion mode\n");
943 return cv::Mat();
944 }
945 } else {
946 fprintf(stderr, "OpenCV: unsupported pixel format 0x%08X\n", pixelFormat);
947 CVPixelBufferUnlockBaseAddress(mGrabbedPixels, 0);
948 CVBufferRelease(mGrabbedPixels);
949 mGrabbedPixels = NULL;
950 return cv::Mat();
951 }
952
953 mOutImage.create(sz, CV_MAKE_TYPE(CV_8U, outChannels));
954 cv::Mat devImage(sz, CV_MAKE_TYPE(CV_8U, deviceChannels), baseaddress, rowBytes);
955 // Convert the device image into the output image.
956 if (cvtCode == -1) {
957 // Copy.
958 devImage.copyTo(mOutImage);
959 } else {
960 cv::cvtColor(devImage, mOutImage, cvtCode);
961 }
962
963
964 CVPixelBufferUnlockBaseAddress(mGrabbedPixels, 0);
965
966 [localpool drain];
967
968 return mOutImage;
969}
