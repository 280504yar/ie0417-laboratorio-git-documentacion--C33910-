9class EpipolarGeometryDegeneracyImpl : public EpipolarGeometryDegeneracy {
10private:
11 Mat points_mat;
12 const int min_sample_size;
13public:
14 explicit EpipolarGeometryDegeneracyImpl (const Mat &points_, int sample_size_) :
15 points_mat(points_), min_sample_size (sample_size_)
16 {
17 CV_DbgAssert(!points_mat.empty() && points_mat.isContinuous());
18 }
19 /*
20 * Do oriented constraint to verify if epipolar geometry is in front or behind the camera.
21 * Return: true if all points are in front of the camers w.r.t. tested epipolar geometry - satisfies constraint.
22 * false - otherwise.
23 * x'^T F x = 0
24 * e' × x' ~+ Fx <=> λe' × x' = Fx, λ > 0
25 * e × x ~+ x'^T F
26 */
27 inline bool isModelValid(const Mat &F_, const std::vector<int> &sample) const override {
28 const Vec3d ep = Utils::getRightEpipole(F_);
29 const auto * const e = ep.val; // of size 3x1
30 const auto * const F = (double *) F_.data;
31 const float * points = points_mat.ptr<float>();
32
33 // without loss of generality, let the first point in sample be in front of the camera.
34 int pt = 4*sample[0];
35 // check only two first elements of vectors (e × x) and (x'^T F)
36 // s1 = (x'^T F)[0] = x2 * F11 + y2 * F21 + 1 * F31
37 // s2 = (e × x)[0] = e'_2 * 1 - e'_3 * y1
38 // sign1 = s1 * s2
39 const double sign1 = (F[0]*points[pt+2]+F[3]*points[pt+3]+F[6])*(e[1]-e[2]*points[pt+1]);
40
41 for (int i = 1; i < min_sample_size; i++) {
42 pt = 4 * sample[i];
43 // if signum of the first point and tested point differs
44 // then two points are on different sides of the camera.
45 if (sign1*(F[0]*points[pt+2]+F[3]*points[pt+3]+F[6])*(e[1]-e[2]*points[pt+1])<0)
46 return false;
47 }
48 return true;
49 }
50};
51void EpipolarGeometryDegeneracy::recoverRank (Mat &model, bool is_fundamental_mat) {
52 /*
53 * Do singular value decomposition.
54 * Make last eigen value zero of diagonal matrix of singular values.
55 */
56 Matx33d U, Vt;
57 Vec3d w;
58 SVD::compute(model, w, U, Vt, SVD::MODIFY_A);
59 if (is_fundamental_mat)
60 model = Mat(U * Matx33d(w(0), 0, 0, 0, w(1), 0, 0, 0, 0) * Vt);
61 else {
62 const double mean_singular_val = (w[0] + w[1]) * 0.5;
63 model = Mat(U * Matx33d(mean_singular_val, 0, 0, 0, mean_singular_val, 0, 0, 0, 0) * Vt);
64 }

82 inline bool isSampleGood (const std::vector<int> &sample) const override {
83 const int smpl1 = 4*sample[0], smpl2 = 4*sample[1], smpl3 = 4*sample[2], smpl4 = 4*sample[3];
84 // planar correspondences must lie on the same side of any line from two points in sample
85 const float * points = points_mat.ptr<float>();
86 const float x1 = points[smpl1], y1 = points[smpl1+1], X1 = points[smpl1+2], Y1 = points[smpl1+3];
87 const float x2 = points[smpl2], y2 = points[smpl2+1], X2 = points[smpl2+2], Y2 = points[smpl2+3];
88 const float x3 = points[smpl3], y3 = points[smpl3+1], X3 = points[smpl3+2], Y3 = points[smpl3+3];
89 const float x4 = points[smpl4], y4 = points[smpl4+1], X4 = points[smpl4+2], Y4 = points[smpl4+3];
90 // line from points 1 and 2
91 const float ab_cross_x = y1 - y2, ab_cross_y = x2 - x1, ab_cross_z = x1 * y2 - y1 * x2;
92 const float AB_cross_x = Y1 - Y2, AB_cross_y = X2 - X1, AB_cross_z = X1 * Y2 - Y1 * X2;
93
94 // check if points 3 and 4 are on the same side of line ab on both images
95 if ((ab_cross_x * x3 + ab_cross_y * y3 + ab_cross_z) *
96 (AB_cross_x * X3 + AB_cross_y * Y3 + AB_cross_z) < 0)
97 return false;
98 if ((ab_cross_x * x4 + ab_cross_y * y4 + ab_cross_z) *
99 (AB_cross_x * X4 + AB_cross_y * Y4 + AB_cross_z) < 0)
100 return false;
101
102 // line from points 3 and 4
103 const float cd_cross_x = y3 - y4, cd_cross_y = x4 - x3, cd_cross_z = x3 * y4 - y3 * x4;
104 const float CD_cross_x = Y3 - Y4, CD_cross_y = X4 - X3, CD_cross_z = X3 * Y4 - Y3 * X4;
105
106 // check if points 1 and 2 are on the same side of line cd on both images
107 if ((cd_cross_x * x1 + cd_cross_y * y1 + cd_cross_z) *
108 (CD_cross_x * X1 + CD_cross_y * Y1 + CD_cross_z) < 0)
109 return false;
110 if ((cd_cross_x * x2 + cd_cross_y * y2 + cd_cross_z) *
111 (CD_cross_x * X2 + CD_cross_y * Y2 + CD_cross_z) < 0)
112 return false;
113
114 // Checks if points are not collinear
115 // If area of triangle constructed with 3 points is less than threshold then points are collinear:
116 // |x1 y1 1| |x1 y1 1|
117 // (1/2) det |x2 y2 1| = (1/2) det |x2-x1 y2-y1 0| = det |x2-x1 y2-y1| < 2 * threshold
118 // |x3 y3 1| |x3-x1 y3-y1 0| |x3-x1 y3-y1|
119 // for points on the first image
120 if (fabsf((x2-x1) * (y3-y1) - (y2-y1) * (x3-x1)) < TOLERANCE) return false; //1,2,3
121 if (fabsf((x2-x1) * (y4-y1) - (y2-y1) * (x4-x1)) < TOLERANCE) return false; //1,2,4
122 if (fabsf((x3-x1) * (y4-y1) - (y3-y1) * (x4-x1)) < TOLERANCE) return false; //1,3,4
123 if (fabsf((x3-x2) * (y4-y2) - (y3-y2) * (x4-x2)) < TOLERANCE) return false; //2,3,4
124 // for points on the second image
125 if (fabsf((X2-X1) * (Y3-Y1) - (Y2-Y1) * (X3-X1)) < TOLERANCE) return false; //1,2,3
126 if (fabsf((X2-X1) * (Y4-Y1) - (Y2-Y1) * (X4-X1)) < TOLERANCE) return false; //1,2,4
127 if (fabsf((X3-X1) * (Y4-Y1) - (Y3-Y1) * (X4-X1)) < TOLERANCE) return false; //1,3,4
128 if (fabsf((X3-X2) * (Y4-Y2) - (Y3-Y2) * (X4-X2)) < TOLERANCE) return false; //2,3,4
129
130 return true;
131 }

257 bool estimateHfrom3Points (const Mat &F_best, const std::vector<int> &sample, Mat &H_best) {
258 Score H_best_score;
259 // find e', null vector of F^T
260 const Vec3d e_prime = Utils::getLeftEpipole(F_best);
261 const Matx33d A = Math::getSkewSymmetric(e_prime) * Matx33d(F_best);
262 bool is_degenerate = false;
263 int idx = -1;
264 for (const auto &h_i : h_sample) { // only 5 samples
265 idx++;
266 Matx33d H;
267 if (!getH(A, e_prime, 4 * sample[h_i[0]], 4 * sample[h_i[1]], 4 * sample[h_i[2]], H))
268 continue;
269 h_reproj_error->setModelParameters(Mat(H));
270 const auto &ver_pts = h_sample_ver[idx];
271 int inliers_in_plane = 3; // 3 points are always inliers
272 for (int s : ver_pts)
273 if (h_reproj_error->getError(sample[s]) < homography_threshold) {
274 if (++inliers_in_plane >= H_INLS_DEGEN_SAMPLE)
275 break;
276 }
277 if (inliers_in_plane >= H_INLS_DEGEN_SAMPLE) {
278 is_degenerate = true;
279 const auto h_score = h_repr_quality->getScore(Mat(H));
280 if (h_score.isBetter(H_best_score)) {
281 H_best_score = h_score;
282 H_best = Mat(H);
283 }
284 }
285 }
286 if (!is_degenerate)
287 return false;
288 int h_inls_cnt = optimizeH(H_best, H_best_score);
289 for (int iter = 0; iter < 2; iter++) {
290 if (h_non_min_solver->estimate(h_inliers, h_inls_cnt, h_models, weights) == 0)
291 break;
292 const auto h_score = h_repr_quality->getScore(h_models[0]);
293 if (h_score.isBetter(H_best_score)) {
294 H_best_score = h_score;
295 h_models[0].copyTo(H_best);
296 h_inls_cnt = h_repr_quality->getInliers(H_best, h_inliers);
297 } else break;
298 }
299 getOutliersH(H_best);
300 return true;
301 }

326 random_gen_H->setSubsetSize(h_inls_cnt <= MAX_H_SUBSET ? (int)(0.8*h_inls_cnt) : MAX_H_SUBSET);
327 }
328 H_best_score = h_score;
329 }
330 }
331 }

335 bool recoverIfDegenerate (const std::vector<int> &sample, const Mat &F_best, const Score &F_best_score,
336 Mat &non_degenerate_model, Score &non_degenerate_model_score) override {
337 const auto swapF = [&] (const Mat &_F, const Score &_score) {
338 const auto non_min_solver = EpipolarNonMinimalSolver::create(points_mat, true);
339 if (! optimizeF(_F, _score, non_degenerate_model, non_degenerate_model_score)) {
340 _F.copyTo(non_degenerate_model);
341 non_degenerate_model_score = _score;
342 }
343 };
344 Mat F_from_H, F_from_E, H_best; Score F_from_H_score, F_from_E_score;
345 if (! estimateHfrom3Points(F_best, sample, H_best)) {
346 return false; // non degenerate
347 }
348 if (true_K_given) {
349 if (getFfromTrueK(H_best, F_from_H, F_from_H_score)) {
350 if (F_from_H_score.isBetter(F_from_E_score))
351 swapF(F_from_H, F_from_H_score);
352 else swapF(F_from_E, F_from_E_score);
353 return true;
354 } else {
355 non_degenerate_model_score = Score();
356 return true; // no translation
357 }
358 }
359 const int non_planar_support_degen_F = getNonPlanarSupport(F_best);
360 Score F_pl_par_score, F_calib_score; Mat F_pl_par, F_calib;
361 if (calibDegensac(H_best, F_calib, F_calib_score, non_planar_support_degen_F, F_best_score)) {
362 if (planeAndParallaxRANSAC(H_best, h_outliers, num_h_outliers, max_iters_plane_and_parallax, true,
363 F_best_score, non_planar_support_degen_F, F_pl_par, F_pl_par_score) && F_pl_par_score.isBetter(F_calib_score)
364 && getNonPlanarSupport(F_pl_par) > getNonPlanarSupport(F_calib)) {
365 swapF(F_pl_par, F_pl_par_score);
366 return true;
367 }
368 swapF(F_calib, F_calib_score);
369 return true;
370 } else {
371 if (planeAndParallaxRANSAC(H_best, h_outliers, num_h_outliers, max_iters_plane_and_parallax, true,
372 F_best_score, non_planar_support_degen_F, F_pl_par, F_pl_par_score)) {
373 swapF(F_pl_par, F_pl_par_score);
374 return true;
375 }
376 }
377 if (! isFDegenerate(non_planar_support_degen_F)) {
378 return false;
379 }
380 non_degenerate_model_score = Score();
381 return true;

384 // RANSAC with plane-and-parallax to find new Fundamental matrix
385 bool getFfromTrueK (const Matx33d &H, Mat &F_from_K, Score &F_from_K_score) {
386 std::vector<Matx33d> R; std::vector<Vec3d> t;
387 if (decomposeHomographyMat(true_K2_inv * H * true_K1, Matx33d::eye(), R, t, noArray()) == 1) {
388 // std::cout << "Warning: translation is zero!\n";
389 return false; // is degenerate
390 }
391 // sign of translation does not make difference
392 const Mat F1 = Mat(true_K2_inv_t * Math::getSkewSymmetric(t[0]) * R[0] * true_K1_inv);
393 const Mat F2 = Mat(true_K2_inv_t * Math::getSkewSymmetric(t[1]) * R[1] * true_K1_inv);
394 const auto score_f1 = quality->getScore(f_error->getErrors(F1)), score_f2 = quality->getScore(f_error->getErrors(F2));
395 if (score_f1.isBetter(score_f2)) {
396 F_from_K = F1; F_from_K_score = score_f1;
397 } else {
398 F_from_K = F2; F_from_K_score = score_f2;
399 }
400 return true;
401 }
402 bool planeAndParallaxRANSAC (const Matx33d &H, std::vector<int> &non_planar_pts, int num_non_planar_pts,
403 int max_iters_pl_par, bool use_preemptive, const Score &score_degen_F, int non_planar_support_degen_F,
404 Mat &F_new, Score &F_new_score) {
405 if (num_non_planar_pts < 2)
406 return false;
407 num_models_used_so_far = 0; // reset estimation of lambda for plane-and-parallax
408 int max_iters = max_iters_pl_par, non_planar_support = 0, iters = 0;
409 const float * points = points_mat.ptr<float>();
410 std::vector<Matx33d> F_good;

542 bool verifyFundamental (const Mat &F_best, const Score &F_score, const std::vector<bool> &inliers_mask, Mat &F_new, Score &new_score) override {
543 const int f_sample_size = 3, max_H_iters = 5; // 3.52 = log(0.01) / log(1 - std::pow(0.9, 3));
544 int num_f_inliers = 0;
545 std::vector<int> inliers(points_size), f_sample(f_sample_size);
546 for (int i = 0; i < points_size; i++) if (inliers_mask[i]) inliers[num_f_inliers++] = i;
547 const auto sampler = UniformSampler::create(0, f_sample_size, num_f_inliers);
548 // find e', null space of F^T
549 const Vec3d e_prime = Utils::getLeftEpipole(F_best);
550 const Matx33d A = Math::getSkewSymmetric(e_prime) * Matx33d(F_best);
551 Score H_best_score; Mat H_best;
552 for (int iter = 0; iter < max_H_iters; iter++) {
553 sampler->generateSample(f_sample);
554 Matx33d H;
555 if (!getH(A, e_prime, 4*inliers[f_sample[0]], 4*inliers[f_sample[1]], 4*inliers[f_sample[2]], H))
556 continue;
557 const auto h_score = h_repr_quality->getScore(Mat(H));
558 if (h_score.isBetter(H_best_score)) {
559 H_best_score = h_score; H_best = Mat(H);
560 }
561 }
562 if (H_best.empty()) return false; // non-degenerate
563 optimizeH(H_best, H_best_score);
564 getOutliersH(H_best);
565 const int non_planar_support_best_F = getNonPlanarSupport(F_best);
566 const bool is_F_degen = isFDegenerate(non_planar_support_best_F);
567 Mat F_from_K; Score F_from_K_score;
568 bool success = false;
569 // generate non-degenerate F even though so-far-the-best one may not be degenerate
570 if (true_K_given) {
571 // use GT calibration
572 if (getFfromTrueK(H_best, F_from_K, F_from_K_score)) {
573 new_score = F_from_K_score;
574 F_from_K.copyTo(F_new);
575 success = true;
576 }
577 } else {
578 // use calibrated DEGENSAC
579 if (calibDegensac(H_best, F_from_K, F_from_K_score, non_planar_support_best_F, F_score)) {
580 new_score = F_from_K_score;
581 F_from_K.copyTo(F_new);
582 success = true;
583 }
584 }
585 if (!is_F_degen) {
586 return false;
587 } else if (success) // F is degenerate
588 return true; // but successfully recovered
589
590 // recover degenerate F using plane-and-parallax
591 Score plane_parallax_score; Mat F_plane_parallax;
592 if (planeAndParallaxRANSAC(H_best, h_outliers, num_h_outliers, 20, true,
593 F_score, non_planar_support_best_F, F_plane_parallax, plane_parallax_score)) {
594 new_score = plane_parallax_score;
595 F_plane_parallax.copyTo(F_new);
596 return true;
597 }
598 // plane-and-parallax failed. A previous non-degenerate so-far-the-best model will be used instead
599 new_score = Score();
600 return true;

615 bool getH (const Matx33d &A, const Vec3d &e_prime, int smpl1, int smpl2, int smpl3, Matx33d &H) {
616 const float * points = points_mat.ptr<float>();
617 Vec3d p1(points[smpl1 ], points[smpl1+1], 1), p2(points[smpl2 ], points[smpl2+1], 1), p3(points[smpl3 ], points[smpl3+1], 1);
618 Vec3d P1(points[smpl1+2], points[smpl1+3], 1), P2(points[smpl2+2], points[smpl2+3], 1), P3(points[smpl3+2], points[smpl3+3], 1);
619 const Matx33d M (p1[0], p1[1], 1, p2[0], p2[1], 1, p3[0], p3[1], 1);
620 if (p1.cross(p2).dot(p3) * P1.cross(P2).dot(P3) < 0) return false;
621 // (x′i × e')
622 const Vec3d P1e = P1.cross(e_prime), P2e = P2.cross(e_prime), P3e = P3.cross(e_prime);
623 // x′i × (A xi))^T (x′i × e′) / ‖x′i×e′‖^2,
624 const Vec3d b (P1.cross(A * p1).dot(P1e) / (P1e[0]*P1e[0]+P1e[1]*P1e[1]+P1e[2]*P1e[2]),
625 P2.cross(A * p2).dot(P2e) / (P2e[0]*P2e[0]+P2e[1]*P2e[1]+P2e[2]*P2e[2]),
626 P3.cross(A * p3).dot(P3e) / (P3e[0]*P3e[0]+P3e[1]*P3e[1]+P3e[2]*P3e[2]));
627
628 H = A - e_prime * (M.inv() * b).t();
629 return true;
