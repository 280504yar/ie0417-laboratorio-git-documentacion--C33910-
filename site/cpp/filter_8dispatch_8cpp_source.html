233int getKernelType(InputArray filter_kernel, Point anchor)
234{
235 Mat _kernel = filter_kernel.getMat();
236 CV_Assert( _kernel.channels() == 1 );
237 int i, sz = _kernel.rows*_kernel.cols;
238
239 Mat kernel;
240 _kernel.convertTo(kernel, CV_64F);
241
242 const double* coeffs = kernel.ptr<double>();
243 double sum = 0;
244 int type = KERNEL_SMOOTH + KERNEL_INTEGER;
245 if( (_kernel.rows == 1 || _kernel.cols == 1) &&
246 anchor.x*2 + 1 == _kernel.cols &&
247 anchor.y*2 + 1 == _kernel.rows )
248 type |= (KERNEL_SYMMETRICAL + KERNEL_ASYMMETRICAL);
249
250 for( i = 0; i < sz; i++ )
251 {
252 double a = coeffs[i], b = coeffs[sz - i - 1];
253 if( a != b )
254 type &= ~KERNEL_SYMMETRICAL;
255 if( a != -b )
256 type &= ~KERNEL_ASYMMETRICAL;
257 if( a < 0 )
258 type &= ~KERNEL_SMOOTH;
259 if( a != saturate_cast<int>(a) )
260 type &= ~KERNEL_INTEGER;
261 sum += a;
262 }
263
264 if( fabs(sum - 1) > FLT_EPSILON*(fabs(sum) + 1) )
265 type &= ~KERNEL_SMOOTH;
266 return type;
267}

296static bool createBitExactKernel_32S(const Mat& kernel, Mat& kernel_dst, int bits)
297{
298 kernel.convertTo(kernel_dst, CV_32S, (1 << bits));
299 Mat_<double> kernel_64f;
300 kernel.convertTo(kernel_64f, CV_64F, (1 << bits));
301 int ksize = (int)kernel.total();
302 const double eps = 10 * FLT_EPSILON * (1 << bits);
303 for (int i = 0; i < ksize; i++)
304 {
305 int bitExactValue = kernel_dst.at<int>(i);
306 double approxValue = kernel_64f.at<double>(i);
307 if (fabs(approxValue - bitExactValue) > eps)
308 return false;
309 }
310 return true;
311}

313Ptr<FilterEngine> createSeparableLinearFilter(
314 int _srcType, int _dstType,
315 InputArray __rowKernel, InputArray __columnKernel,
316 Point _anchor, double _delta,
317 int _rowBorderType, int _columnBorderType,
318 const Scalar& _borderValue)
319{
320 Mat _rowKernel = __rowKernel.getMat(), _columnKernel = __columnKernel.getMat();
321 _srcType = CV_MAT_TYPE(_srcType);
322 _dstType = CV_MAT_TYPE(_dstType);
323 int sdepth = CV_MAT_DEPTH(_srcType), ddepth = CV_MAT_DEPTH(_dstType);
324 int cn = CV_MAT_CN(_srcType);
325 CV_Assert( cn == CV_MAT_CN(_dstType) );
326 int rsize = _rowKernel.rows + _rowKernel.cols - 1;
327 int csize = _columnKernel.rows + _columnKernel.cols - 1;
328 if( _anchor.x < 0 )
329 _anchor.x = rsize/2;
330 if( _anchor.y < 0 )
331 _anchor.y = csize/2;
332 int rtype = getKernelType(_rowKernel,
333 _rowKernel.rows == 1 ? Point(_anchor.x, 0) : Point(0, _anchor.x));
334 int ctype = getKernelType(_columnKernel,
335 _columnKernel.rows == 1 ? Point(_anchor.y, 0) : Point(0, _anchor.y));
336 Mat rowKernel, columnKernel;
337
338 bool isBitExactMode = false;
339 int bdepth = std::max(CV_32F,std::max(sdepth, ddepth));
340 int bits = 0;
341
342 if( sdepth == CV_8U &&
343 ((rtype == KERNEL_SMOOTH+KERNEL_SYMMETRICAL &&
344 ctype == KERNEL_SMOOTH+KERNEL_SYMMETRICAL &&
345 ddepth == CV_8U) ||
346 ((rtype & (KERNEL_SYMMETRICAL+KERNEL_ASYMMETRICAL)) &&
347 (ctype & (KERNEL_SYMMETRICAL+KERNEL_ASYMMETRICAL)) &&
348 (rtype & ctype & KERNEL_INTEGER) &&
349 ddepth == CV_16S)) )
350 {
351 int bits_ = ddepth == CV_8U ? 8 : 0;
352 bool isValidBitExactRowKernel = createBitExactKernel_32S(_rowKernel, rowKernel, bits_);
353 bool isValidBitExactColumnKernel = createBitExactKernel_32S(_columnKernel, columnKernel, bits_);
354 if (!isValidBitExactRowKernel)
355 {
356 CV_LOG_DEBUG(NULL, "createSeparableLinearFilter: bit-exact row-kernel can't be applied: ksize=" << _rowKernel.total());
357 }
358 else if (!isValidBitExactColumnKernel)
359 {
360 CV_LOG_DEBUG(NULL, "createSeparableLinearFilter: bit-exact column-kernel can't be applied: ksize=" << _columnKernel.total());
361 }
362 else
363 {
364 bdepth = CV_32S;
365 bits = bits_;
366 bits *= 2;
367 _delta *= (1 << bits);
368 isBitExactMode = true;
369 }
370 }
371 if (!isBitExactMode)
372 {
373 if( _rowKernel.type() != bdepth )
374 _rowKernel.convertTo( rowKernel, bdepth );
375 else
376 rowKernel = _rowKernel;
377 if( _columnKernel.type() != bdepth )
378 _columnKernel.convertTo( columnKernel, bdepth );
379 else
380 columnKernel = _columnKernel;
381 }
382
383 int _bufType = CV_MAKETYPE(bdepth, cn);
384 Ptr<BaseRowFilter> _rowFilter = getLinearRowFilter(
385 _srcType, _bufType, rowKernel, _anchor.x, rtype);
386 Ptr<BaseColumnFilter> _columnFilter = getLinearColumnFilter(
387 _bufType, _dstType, columnKernel, _anchor.y, ctype, _delta, bits );
388
389 return Ptr<FilterEngine>( new FilterEngine(Ptr<BaseFilter>(), _rowFilter, _columnFilter,
390 _srcType, _dstType, _bufType, _rowBorderType, _columnBorderType, _borderValue ));
391}

398void preprocess2DKernel( const Mat& kernel, std::vector<Point>& coords, std::vector<uchar>& coeffs )
399{
400 int i, j, k, nz = countNonZero(kernel), ktype = kernel.type();
401 if(nz == 0)
402 nz = 1;
403 CV_Assert( ktype == CV_8U || ktype == CV_32S || ktype == CV_32F || ktype == CV_64F );
404 coords.resize(nz);
405 coeffs.resize(nz*getElemSize(ktype));
406 uchar* _coeffs = &coeffs[0];
407
408 for( i = k = 0; i < kernel.rows; i++ )
409 {
410 const uchar* krow = kernel.ptr(i);
411 for( j = 0; j < kernel.cols; j++ )
412 {
413 if( ktype == CV_8U )
414 {
415 uchar val = krow[j];
416 if( val == 0 )
417 continue;
418 coords[k] = Point(j,i);
419 _coeffs[k++] = val;
420 }
421 else if( ktype == CV_32S )
422 {
423 int val = ((const int*)krow)[j];
424 if( val == 0 )
425 continue;
426 coords[k] = Point(j,i);
427 ((int*)_coeffs)[k++] = val;
428 }
429 else if( ktype == CV_32F )
430 {
431 float val = ((const float*)krow)[j];
432 if( val == 0 )
433 continue;
434 coords[k] = Point(j,i);
435 ((float*)_coeffs)[k++] = val;
436 }
437 else
438 {
439 double val = ((const double*)krow)[j];
440 if( val == 0 )
441 continue;
442 coords[k] = Point(j,i);
443 ((double*)_coeffs)[k++] = val;
444 }
445 }
446 }
447}

455 Filter2D( const Mat& _kernel, Point _anchor,

470 void operator()(const uchar** src, uchar* dst, int dststep, int count, int width, int cn) CV_OVERRIDE
471 {
472 KT _delta = delta;
473 const Point* pt = &coords[0];
474 const KT* kf = (const KT*)&coeffs[0];
475 int i, k, nz = (int)coords.size();
476 AutoBuffer<const ST*> _kp(nz);
477 const ST** kp = _kp.data();
478 CastOp castOp = castOp0;
479
480 width *= cn;
481 for( ; count > 0; count--, dst += dststep, src++ )
482 {
483 DT* D = (DT*)dst;
484
485 for( k = 0; k < nz; k++ )
486 kp[k] = (const ST*)src[pt[k].y] + pt[k].x*cn;
487
488 i = vecOp((const uchar**)kp, dst, width);
489 #if CV_ENABLE_UNROLLED
490 for( ; i <= width - 4; i += 4 )
491 {
492 KT s0 = _delta, s1 = _delta, s2 = _delta, s3 = _delta;
493
494 for( k = 0; k < nz; k++ )
495 {
496 const ST* sptr = kp[k] + i;
497 KT f = kf[k];
498 s0 += f*sptr[0];
499 s1 += f*sptr[1];
500 s2 += f*sptr[2];
501 s3 += f*sptr[3];
502 }
503
504 D[i] = castOp(s0); D[i+1] = castOp(s1);
505 D[i+2] = castOp(s2); D[i+3] = castOp(s3);
506 }
507 #endif
508 for( ; i < width; i++ )
509 {
510 KT s0 = _delta;
511 for( k = 0; k < nz; k++ )
512 s0 += kf[k]*kp[k][i];
513 D[i] = castOp(s0);
514 }
515 }
516 }

519 std::vector<uchar> coeffs;
520 KT delta;
521 CastOp castOp0;
522 VecOp vecOp;

530// prepare kernel: transpose and make double rows (+align). Returns size of aligned row
531// Samples:
532// a b c
533// Input: d e f
534// g h i
535// Output, last two zeros is the alignment:
536// a d g a d g 0 0
537// b e h b e h 0 0
538// c f i c f i 0 0
539template <typename T>
540static int _prepareKernelFilter2D(std::vector<T> & data, const Mat & kernel)
541{
542 Mat _kernel; kernel.convertTo(_kernel, DataDepth<T>::value);
543 int size_y_aligned = ROUNDUP(kernel.rows * 2, 4);
544 data.clear(); data.resize(size_y_aligned * kernel.cols, 0);
545 for (int x = 0; x < kernel.cols; x++)
546 {
547 for (int y = 0; y < kernel.rows; y++)
548 {
549 data[x * size_y_aligned + y] = _kernel.at<T>(y, x);
550 data[x * size_y_aligned + y + kernel.rows] = _kernel.at<T>(y, x);
551 }
552 }
553 return size_y_aligned;
554}

556static bool ocl_filter2D( InputArray _src, OutputArray _dst, int ddepth,
557 InputArray _kernel, Point anchor,
558 double delta, int borderType )
559{
560 int type = _src.type(), sdepth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type);
561 ddepth = ddepth < 0 ? sdepth : ddepth;
562 int dtype = CV_MAKE_TYPE(ddepth, cn), wdepth = std::max(std::max(sdepth, ddepth), CV_32F),
563 wtype = CV_MAKE_TYPE(wdepth, cn);
564 if (cn > 4)
565 return false;
566
567 Size ksize = _kernel.size();
568 if (anchor.x < 0)
569 anchor.x = ksize.width / 2;
570 if (anchor.y < 0)
571 anchor.y = ksize.height / 2;
572
573 bool isolated = (borderType & BORDER_ISOLATED) != 0;
574 borderType &= ~BORDER_ISOLATED;
575 const cv::ocl::Device &device = cv::ocl::Device::getDefault();
576 bool doubleSupport = device.doubleFPConfig() > 0;
577 if (wdepth == CV_64F && !doubleSupport)
578 return false;
579
580 const char * const borderMap[] = { "BORDER_CONSTANT", "BORDER_REPLICATE", "BORDER_REFLECT",
581 "BORDER_WRAP", "BORDER_REFLECT_101" };
582
583 cv::Mat kernelMat = _kernel.getMat();
584 cv::Size sz = _src.size(), wholeSize;
585 size_t globalsize[2] = { (size_t)sz.width, (size_t)sz.height };
586 size_t localsize_general[2] = {0, 1};
587 size_t* localsize = NULL;
588
589 ocl::Kernel k;
590 UMat src = _src.getUMat();
591 if (!isolated)
592 {
593 Point ofs;
594 src.locateROI(wholeSize, ofs);
595 }
596
597 size_t tryWorkItems = device.maxWorkGroupSize();
598 if (device.isIntel() && 128 < tryWorkItems)
599 tryWorkItems = 128;
600 char cvt[2][50];
601
602 // For smaller filter kernels, there is a special kernel that is more
603 // efficient than the general one.
604 if (device.isIntel() && (device.type() & ocl::Device::TYPE_GPU) &&
605 ((ksize.width < 5 && ksize.height < 5) ||
606 (ksize.width == 5 && ksize.height == 5 && cn == 1)))
607 {
608 kernelMat = kernelMat.reshape(0, 1);
609 String kerStr = ocl::kernelToStr(kernelMat, CV_32F);
610 int h = isolated ? sz.height : wholeSize.height;
611 int w = isolated ? sz.width : wholeSize.width;
612
613 if (w < ksize.width || h < ksize.height)
614 return false;
615
616 // Figure out what vector size to use for loading the pixels.
617 int pxLoadNumPixels = cn != 1 || sz.width % 4 ? 1 : 4;
618 int pxLoadVecSize = cn * pxLoadNumPixels;
619
620 // Figure out how many pixels per work item to compute in X and Y
621 // directions. Too many and we run out of registers.
622 int pxPerWorkItemX = 1;
623 int pxPerWorkItemY = 1;
624 if (cn <= 2 && ksize.width <= 4 && ksize.height <= 4)
625 {
626 pxPerWorkItemX = sz.width % 8 ? sz.width % 4 ? sz.width % 2 ? 1 : 2 : 4 : 8;
627 pxPerWorkItemY = sz.height % 2 ? 1 : 2;
628 }
629 else if (cn < 4 || (ksize.width <= 4 && ksize.height <= 4))
630 {
631 pxPerWorkItemX = sz.width % 2 ? 1 : 2;
632 pxPerWorkItemY = sz.height % 2 ? 1 : 2;
633 }
634 globalsize[0] = sz.width / pxPerWorkItemX;
635 globalsize[1] = sz.height / pxPerWorkItemY;
636
637 // Need some padding in the private array for pixels
638 int privDataWidth = ROUNDUP(pxPerWorkItemX + ksize.width - 1, pxLoadNumPixels);
639
640 // Make the global size a nice round number so the runtime can pick
641 // from reasonable choices for the workgroup size
642 const int wgRound = 256;
643 globalsize[0] = ROUNDUP(globalsize[0], wgRound);
644
645 char build_options[1024];

1237 int kernel_width, int kernel_height,
1238 int anchor_x, int anchor_y,
1239 double delta, int borderType)
1240{
1241 {
1242 int sdepth = CV_MAT_DEPTH(stype);
1243 int ddepth = CV_MAT_DEPTH(dtype);
1244 int dft_filter_size = checkHardwareSupport(CV_CPU_SSE3) && ((sdepth == CV_8U && (ddepth == CV_8U || ddepth == CV_16S)) || (sdepth == CV_32F && ddepth == CV_32F)) ? 130 : 50;
1245 if (kernel_width * kernel_height < dft_filter_size)
1246 return false;
1247
1248 // detect roi case
1249 if( (offset_x != 0) || (offset_y != 0) )
1250 {
1251 return false;
1252 }
1253 if( (width != full_width) || (height != full_height) )
1254 {
1255 return false;
1256 }
1257 }
1258
1259 Point anchor = Point(anchor_x, anchor_y);
1260 Mat kernel = Mat(Size(kernel_width, kernel_height), kernel_type, kernel_data, kernel_step);
1261
1262 Mat src(Size(width, height), stype, src_data, src_step);
1263 Mat dst(Size(width, height), dtype, dst_data, dst_step);
1264 Mat temp;
1265 int src_channels = CV_MAT_CN(stype);
1266 int dst_channels = CV_MAT_CN(dtype);
1267 int ddepth = CV_MAT_DEPTH(dtype);
1268 // crossCorr doesn't accept non-zero delta with multiple channels
1269 if (src_channels != 1 && delta != 0) {
1270 // The semantics of filter2D require that the delta be applied
1271 // as floating-point math. So wee need an intermediate Mat
1272 // with a float datatype. If the dest is already floats,
1273 // we just use that.
1274 int corrDepth = ddepth;
1275 if ((ddepth == CV_32F || ddepth == CV_64F) && src_data != dst_data) {
1276 temp = Mat(Size(width, height), dtype, dst_data, dst_step);
1277 } else {
1278 corrDepth = ddepth == CV_64F ? CV_64F : CV_32F;
1279 temp.create(Size(width, height), CV_MAKETYPE(corrDepth, dst_channels));
1280 }
1281 crossCorr(src, kernel, temp, anchor, 0, borderType);
1282 add(temp, delta, temp);
1283 if (temp.data != dst_data) {
1284 temp.convertTo(dst, dst.type());
1285 }
1286 } else {
1287 if (src_data != dst_data)
1288 temp = Mat(Size(width, height), dtype, dst_data, dst_step);
1289 else
1290 temp.create(Size(width, height), dtype);
1291 crossCorr(src, kernel, temp, anchor, delta, borderType);
1292 if (temp.data != dst_data)
1293 temp.copyTo(dst);
1294 }
1295 return true;
