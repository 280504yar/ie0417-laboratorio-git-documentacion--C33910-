77class BaseConvolutionLayerImpl : public ConvolutionLayer
78{
79public:
80 bool fusedWeights, fusedBias;
81 std::vector<double> weightsMultipliers;
82 int groups;
83 BaseConvolutionLayerImpl(const LayerParams &params)
84 {
85 setParamsFrom(params);
86 getConvolutionKernelParams(params, kernel_size, pads_begin, pads_end, strides, dilations,
87 padMode, adjust_pads, useWinograd);
88
89 numOutput = -1;
90 groups = params.get<int>("group", 1);
91
92 if (kernel_size.size() == 2) {
93 kernel = Size(kernel_size[1], kernel_size[0]);
94 stride = Size(strides[1], strides[0]);
95 pad = Size(pads_begin[1], pads_begin[0]);
96 dilation = Size(dilations[1], dilations[0]);
97
98 adjustPad.height = adjust_pads[0];
99 adjustPad.width = adjust_pads[1];
100 }
101
102 for (int i = 0; i < adjust_pads.size(); i++) {
103 CV_Assert(adjust_pads[i] < strides[i]);
104 }
105
106 fusedWeights = false;
107 fusedBias = false;
108 }
109
110 bool hasBias() const
111 {
112 return blobs.size() >= 2;
113 }
114
115 virtual MatShape computeColRowShape(const MatShape &inpShape, const MatShape &outShape) const = 0;
116 bool is1x1() const
117 {
118 return (kernel.height == 1 && kernel.width == 1) &&
119 (stride.height == 1 && stride.width == 1) &&
120 (dilation.height == 1 && dilation.width == 1);
121 }
122
123 virtual bool tryFuse(Ptr<Layer>& top) CV_OVERRIDE
124 {
125 if (fusedAdd) // If the Conv layer has fused Add layer, it cannot fuse other layers.
126 return false;
127
128 Ptr<BlankLayer> blank_layer = top.dynamicCast<BlankLayer>();
129 if (blank_layer)
130 return true;
131
132 Mat w, b;
133 top->getScaleShift(w, b);
134 if (!w.empty() || !b.empty())
135 {
136 fuseWeights(w, b);
137 fusedWeights = fusedWeights || !w.empty();
138 fusedBias = fusedBias || (hasBias() && !w.empty()) || !b.empty();
139 return true;
140 }
141 return false;
142 }
143
144 virtual void fuseWeights(const Mat& w_, const Mat& b_) = 0;
145};

156 MatShape computeColRowShape(const MatShape &inpShape, const MatShape &outShape) const CV_OVERRIDE
157 {
158 int dims = inpShape.size();
159 int inpD = dims == 5 ? inpShape[2] : 1;
160 int inpH = inpShape[dims - 2];
161 int inpW = inpShape.back();
162 int outCn = outShape[1];
163 int outGroupCn = outCn / groups;
164 int ksize = outGroupCn * std::accumulate(kernel_size.begin(), kernel_size.end(),
165 1, std::multiplies<size_t>());
166 return shape(ksize, inpD * inpH * inpW);
167 }

171 if (backendId == DNN_BACKEND_CUDA)

173 /* only deconvolution 2d and 3d supported */

195 bool getMemoryShapes(const std::vector<MatShape> &inputs,
196 const int requiredOutputs,
197 std::vector<MatShape> &outputs,
198 std::vector<MatShape> &internals) const CV_OVERRIDE
199 {
200 CV_Assert(inputs.size() != 0);
201 CV_Assert(inputs.size() > 1 || !blobs.empty());
202 MatShape weightShape = blobs.empty() ? inputs[1] : blobs[0].shape();
203
204 int outCn = numOutput;
205 if (outCn < 0) {
206 outCn = weightShape[1]*groups;
207 }
208 std::vector<int> outShape;
209 outShape.push_back(inputs[0][0]); // batch
210 outShape.push_back(outCn);
211
212 int spatialDims = kernel_size.size();
213 CV_Assert(spatialDims >= 1 && spatialDims <= 3);
214
215 int expectedDims = inputs[0].size();
216
217 // If the network is effectively 1-D (input shape N x C x W) but the
218 // parser has duplicated the kernel/stride/etc. parameters so that
219 // `spatialDims == 2`, switch to the dedicated 1-D branch.
220 if (expectedDims == 3)
221 spatialDims = 1;
222
223 if (spatialDims == 1) {
224 CV_Assert(expectedDims == 3 || expectedDims == 4);
225 } else {
226 CV_Assert(expectedDims == 2 + spatialDims);
227 }
228
229 if (padMode.empty())
230 {
231 for (int i = 0; i < spatialDims; i++) {
232 outShape.push_back(strides[i] * (inputs[0][2 + i] - 1) + ((kernel_size[i] - 1) * dilations[i] + 1) - pads_begin[i] - pads_end[i] + adjust_pads[i]);
233 }
234 }
235 else if (padMode == "VALID")
236 {
237 for (int i = 0; i < spatialDims; i++) {
238 outShape.push_back(strides[i] * (inputs[0][2 + i] - 1) + ((kernel_size[i] - 1) * dilations[i] + 1) + adjust_pads[i]);
239 }
240 }
241 else if (padMode == "SAME")
242 {
243 for (int i = 0; i < spatialDims; i++)
244 outShape.push_back(strides[i] * (inputs[0][2 + i] - 1) + 1 + adjust_pads[i]);
245 }
246 else
247 CV_Error(Error::StsError, "Unsupported padding mode " + padMode);
248 CV_Assert(outCn % weightShape[1] == 0);
249
250 int inpCn = inputs[0][1];
251 CV_Assert(inpCn % groups == 0 && outCn % groups == 0);
252 CV_Assert(weightShape[0] == inpCn);
253
254 outputs.resize(1, MatShape(outShape));
255
256 if (!is1x1())
257 internals.push_back(computeColRowShape(inputs[0], outputs[0]));
258
259 return false;
260 }

273 void fuseWeights(const Mat& w_, const Mat& b_) CV_OVERRIDE
274 {
275 if (numOutput < 0) {
276 CV_Assert(!blobs.empty());
277 numOutput = blobs[0].size[1] * groups;
278 }
279 weightsMultipliers.assign(numOutput, 1.0);
280
281 if (weightsMat.empty() && !blobs.empty())
282 transpose(blobs[0].reshape(1, blobs[0].size[0]), weightsMat);
283
284 if (biasesMat.empty()) {
285 Mat inpBias = blobs.size() >= 2 ? blobs[1] : Mat();
286 Mat biasesMat_ = !inpBias.empty() ? inpBias.reshape(1, numOutput) : Mat::zeros(numOutput, 1, CV_32F);
287 biasesMat_.copyTo(biasesMat);
288 }
289
290 Mat w = w_.total() == 1 ? Mat(1, numOutput, CV_32F, Scalar(w_.at<float>(0))) : w_;
291 Mat b = b_.total() == 1 ? Mat(1, numOutput, CV_32F, Scalar(b_.at<float>(0))) : b_;
292
293 CV_Assert_N(!weightsMat.empty(),
294 w.empty() || numOutput == w.total(),
295 b.empty() || numOutput == b.total());
296
297 if (!w.empty())
298 {
299 transpose(blobs[0].reshape(1, blobs[0].size[0]), weightsMat);
300 weightsMat = weightsMat.reshape(1, numOutput);
301 for (int i = 0; i < numOutput; ++i)
302 {
303 double wi = w.at<float>(i);
304 weightsMultipliers[i] *= wi;
305 cv::multiply(weightsMat.row(i), weightsMultipliers[i], weightsMat.row(i));
306 biasesMat.at<float>(i) *= wi;
307 }
308 weightsMat = weightsMat.reshape(1, weightsMat.total() / blobs[0].size[0]);
309 }
310
311 if (!b.empty())
312 {
313 cv::add(biasesMat, b.reshape(1, numOutput), biasesMat);
314 }

317 class MatMulInvoker : public ParallelLoopBody
318 {
319 public:
320 MatMulInvoker(const Mat& a, const Mat& b, Mat& c, int nstripes)
321 {
322 a_ = &a;
323 b_ = &b;
324 c_ = &c;
325 nstripes_ = nstripes;
326 useAVX = checkHardwareSupport(CPU_AVX);
327 useAVX2 = checkHardwareSupport(CPU_AVX2);
328 useAVX512 = CV_CPU_HAS_SUPPORT_AVX512_SKX;
329 useRVV = checkHardwareSupport(CPU_RVV);
330 useLASX = checkHardwareSupport(CPU_LASX);
331 }
332
333 void operator()(const Range& range_) const CV_OVERRIDE
334 {
335 int stripeSize = (int)alignSize((b_->cols + nstripes_ - 1)/nstripes_, 16);
336 Range range(range_.start*stripeSize, std::min(range_.end*stripeSize, b_->cols));
337 int mmax = a_->rows;
338 int nmax = range.end - range.start;
339 int kmax = a_->cols;
340 int m, n, k;
341 const float* aptr = a_->ptr<float>();
342 const float* bptr = b_->ptr<float>() + range.start;
343 float* cptr = c_->ptr<float>() + range.start;
344 size_t astep = a_->step1();
345 size_t bstep = b_->step1();
346 size_t cstep = c_->step1();
347
348 #if CV_TRY_AVX512_SKX
349 if( useAVX512 )
350 opt_AVX512_SKX::fastGEMM( aptr, astep, bptr, bstep, cptr, cstep, mmax, kmax, nmax );
351 else
352 #endif
353 #if CV_TRY_AVX2
354 if( useAVX2 )
355 opt_AVX2::fastGEMM( aptr, astep, bptr, bstep, cptr, cstep, mmax, kmax, nmax );
356 else
357 #endif
358 #if CV_TRY_AVX
359 if( useAVX )
360 opt_AVX::fastGEMM( aptr, astep, bptr, bstep, cptr, cstep, mmax, kmax, nmax );
361 else
362 #endif
363 #if CV_TRY_RVV && CV_RVV
364 if( useRVV ) {
365 opt_RVV::fastGEMM( aptr, astep, bptr, bstep, cptr, cstep, mmax, kmax, nmax );
366 }
367 else
368 #endif
369 #if CV_TRY_LASX
370 if( useLASX )
371 opt_LASX::fastGEMM( aptr, astep, bptr, bstep, cptr, cstep, mmax, kmax, nmax );
372 else
373 #endif
374 for( m = 0; m < mmax; m += 2 )
375 {
376 float* dst0 = cptr + cstep*m;
377 float* dst1 = cptr + cstep*std::min(m+1, mmax-1);
378 const float* aptr0 = aptr + astep*m;
379 const float* aptr1 = aptr + astep*std::min(m+1, mmax-1);
380
381 for( n = 0; n < nmax; n++ )
382 {
383 dst0[n] = 0.f;
384 dst1[n] = 0.f;
385 }
386
387 for( k = 0; k < kmax; k += 4 )
388 {
389 float alpha00 = aptr0[k];
390 float alpha01 = aptr1[k];
391 float alpha10 = 0.f, alpha11 = 0.f;
392 float alpha20 = 0.f, alpha21 = 0.f;
393 float alpha30 = 0.f, alpha31 = 0.f;
394 const float* bptr0 = bptr + k*bstep;
395 const float* bptr1 = bptr0;
396 const float* bptr2 = bptr0;
397 const float* bptr3 = bptr0;
398
399 if( k+1 < kmax )
400 {
401 alpha10 = aptr0[k+1];
402 alpha11 = aptr1[k+1];
403 bptr1 = bptr0 + bstep;
404 if( k+2 < kmax )
405 {
406 alpha20 = aptr0[k+2];
407 alpha21 = aptr1[k+2];
408 bptr2 = bptr1 + bstep;
409 if( k+3 < kmax )
410 {
411 alpha30 = aptr0[k+3];
412 alpha31 = aptr1[k+3];
413 bptr3 = bptr2 + bstep;
414 }
415 }
416 }
417 n = 0;
418
419 #if CV_SIMD128
420 v_float32x4 a00 = v_setall_f32(alpha00);
421 v_float32x4 a01 = v_setall_f32(alpha01);
422 v_float32x4 a10 = v_setall_f32(alpha10);
423 v_float32x4 a11 = v_setall_f32(alpha11);
424 v_float32x4 a20 = v_setall_f32(alpha20);
425 v_float32x4 a21 = v_setall_f32(alpha21);
426 v_float32x4 a30 = v_setall_f32(alpha30);
427 v_float32x4 a31 = v_setall_f32(alpha31);
428
429 for( ; n <= nmax - 4; n += 4 )
430 {
431 v_float32x4 d0 = v_load(dst0 + n);
432 v_float32x4 d1 = v_load(dst1 + n);
433 v_float32x4 b0 = v_load(bptr0 + n);
434 v_float32x4 b1 = v_load(bptr1 + n);
435 v_float32x4 b2 = v_load(bptr2 + n);
436 v_float32x4 b3 = v_load(bptr3 + n);
437 // TODO try to improve pipeline width
438 d0 = v_fma(b0, a00, d0);
439 d1 = v_fma(b0, a01, d1);
440 d0 = v_fma(b1, a10, d0);
441 d1 = v_fma(b1, a11, d1);
442 d0 = v_fma(b2, a20, d0);
443 d1 = v_fma(b2, a21, d1);
444 d0 = v_fma(b3, a30, d0);
445 d1 = v_fma(b3, a31, d1);
446 v_store(dst0 + n, d0);
447 v_store(dst1 + n, d1);
448 }
449 #endif
450
451 for( ; n < nmax; n++ )
452 {
453 float b0 = bptr0[n];
454 float b1 = bptr1[n];
455 float b2 = bptr2[n];
456 float b3 = bptr3[n];
457 float d0 = dst0[n] + alpha00*b0 + alpha10*b1 + alpha20*b2 + alpha30*b3;
458 float d1 = dst1[n] + alpha01*b0 + alpha11*b1 + alpha21*b2 + alpha31*b3;
459 dst0[n] = d0;
460 dst1[n] = d1;
461 }
462 }
463 }
464 }

497 static void run(const float* data_col,
498 int channels,
499 const std::vector<int>& output_shape,
500 const std::vector<int>& kernel_shape,
501 const std::vector<int>& pads,
502 const std::vector<int>& strides,
503 const std::vector<int>& dilations,
504 const std::vector<int>& input_shape,
505 float* data_im,
506 const float* biasvec,
507 bool is1x1)

516 t.kernel_shape = kernel_shape;
517 t.pads = pads;
518 t.strides = strides;

527
528 virtual void operator ()(const Range &r) const CV_OVERRIDE
529 {
530 const float* data_col_ = data_col;
531 float* data_im_ = data_im;
532 bool is1x1_ = is1x1;
533 const float* biasvec_ = biasvec;
534
535 int ndims = output_shape.size();
536
537 // Calculate total output size
538 int total_output_size = channels;
539 int input_spatial_size = 1;
540 for (int i = 0; i < ndims; i++) {
541 total_output_size *= output_shape[i];
542 input_spatial_size *= input_shape[i];
543 }
544
545 size_t stripeSize = (total_output_size + nstripes - 1) / nstripes;
546 size_t startIndex = r.start * stripeSize;
547 size_t endIndex = std::min(r.end * stripeSize, (size_t)total_output_size);
548
549 for (size_t index = startIndex; index < endIndex; index++)
550 {
551 // Convert linear index to multi-dimensional coordinates
552 std::vector<int> coords(ndims + 1); // +1 for channel dimension
553 size_t idx = index;
554
555 // Extract spatial coordinates and channel
556 for (int i = ndims - 1; i >= 0; i--) {
557 coords[i + 1] = idx % output_shape[i];
558 idx /= output_shape[i];
559 }
560 coords[0] = idx; // channel
561
562 float val = 0.0f;
563
564 if( is1x1_ )
565 val = data_im_[index];
566 else {
567 std::vector<int> kernel_coords(ndims);
568 std::function<void(int)> iterate_kernel = [&](int dim) {
569 if (dim == ndims) {
570 std::vector<int> input_coords(ndims);
571 bool valid = true;
572
573 for (int i = 0; i < ndims; i++) {
574 // Apply dilation to kernel coordinates
575 int dilated_kernel_pos = kernel_coords[i] * dilations[i];
576 input_coords[i] = coords[i + 1] + pads[i] - dilated_kernel_pos;
577 if (input_coords[i] < 0 || input_coords[i] % strides[i] != 0) {
578 valid = false;
579 break;
580 }
581 input_coords[i] /= strides[i];
582 if (input_coords[i] >= input_shape[i]) {
583 valid = false;
584 break;
585 }
586 }
587
588 if (valid) {
589 // Calculate offset in column matrix
590 int col_offset = coords[0]; // channel
591 for (int i = 0; i < ndims; i++) {
592 col_offset = col_offset * kernel_shape[i] + kernel_coords[i];
593 }
594 col_offset *= input_spatial_size;
595
596 // Calculate input position in flattened input
597 int input_pos = 0;
598 for (int i = 0; i < ndims; i++) {
599 input_pos = input_pos * input_shape[i] + input_coords[i];
600 }
601
602 val += data_col_[col_offset + input_pos];
603 }
604 } else {
605 for (int k = 0; k < kernel_shape[dim]; k++) {
606 kernel_coords[dim] = k;
607 iterate_kernel(dim + 1);
608 }
609 }
610 };
611
612 iterate_kernel(0);
613 }
614 data_im_[index] = val + biasvec_[coords[0]];
615 }
616 }

619#ifdef HAVE_OPENCL
620 bool forward_ocl(InputArrayOfArrays inputs_, OutputArrayOfArrays outputs_, OutputArrayOfArrays internals_)
621 {
622 std::vector<UMat> inputs;
623 std::vector<UMat> outputs;
624 std::vector<UMat> internals;
625
626 if (inputs_.depth() == CV_16F)
627 return false;
628
629 inputs_.getUMatVector(inputs);
630 outputs_.getUMatVector(outputs);
631 internals_.getUMatVector(internals);
632
633 int outCn = numOutput;
634 int inpCn = inputs[0].size[1];
635
636 if (is1x1())
637 return false;
638
639 if (umat_weights.empty() || inputs.size() >= 2) {
640 Mat temp;
641 if (fusedWeights)
642 weightsMat.copyTo(umat_weights);
643 else if (!blobs.empty()) {
644 transpose(blobs[0].reshape(1, inpCn), temp);
645 temp.copyTo(umat_weights);
646 }
647 else {
648 transpose(inputs[1].reshape(1, inpCn), temp);
649 temp.copyTo(umat_weights);
650 }
651 }
652
653 if (umat_biases.empty() || inputs.size() >= 3) {
654 if (fusedBias)
655 biasesMat.copyTo(umat_biases);
656 else if (blobs.size() > 1)
657 blobs[1].reshape(1, outCn).copyTo(umat_biases);
658 else if (inputs.size() >= 3)
659 inputs[2].reshape(1, outCn).copyTo(umat_biases);
660 else
661 umat_biases = UMat::zeros(outCn, 1, CV_32F);
662 }
