13Net readNet(const String& _model, const String& _config, const String& _framework, int engine)
14{
15 String framework = toLowerCase(_framework);
16 String model = _model;
17 String config = _config;
18 const std::string modelExt = model.substr(model.rfind('.') + 1);
19 const std::string configExt = config.substr(config.rfind('.') + 1);
20 if (framework == "caffe" || modelExt == "caffemodel" || configExt == "caffemodel" ||
21 modelExt == "prototxt" || configExt == "prototxt")
22 {
23 CV_Error(Error::StsError, "Caffe importer has been removed. Please use ONNX-converted models or use an older OpenCV version.");
24 }
25 if (framework == "tensorflow" || modelExt == "pb" || configExt == "pb" || modelExt == "pbtxt" || configExt == "pbtxt")
26 {
27 if (modelExt == "pbtxt" || configExt == "pb")
28 std::swap(model, config);
29 return readNetFromTensorflow(model, config, engine);
30 }
31 if (framework == "tflite" || modelExt == "tflite")
32 {
33 return readNetFromTFLite(model, engine);
34 }
35 if (framework == "darknet" || modelExt == "weights" || configExt == "weights" || modelExt == "cfg" || configExt == "cfg")
36 {
37 if (modelExt == "cfg" || configExt == "weights")
38 std::swap(model, config);
39 CV_Error(Error::StsError, "Darknet importer has been removed. Please use readNet(\"darknet\", ...) with ONNX-converted models or use an older OpenCV version.");
40 }
41 if (framework == "dldt" || framework == "openvino" ||
42 modelExt == "bin" || configExt == "bin" ||
43 modelExt == "xml" || configExt == "xml")
44 {
45 if (modelExt == "xml" || configExt == "bin" || modelExt == "onnx")
46 std::swap(model, config);
47 return readNetFromModelOptimizer(config, model);
48 }
49 if (framework == "onnx" || modelExt == "onnx")
50 {
51 return readNetFromONNX(model, engine);
52 }
53 CV_Error(Error::StsError, "Cannot determine an origin framework of files: " + model + (config.empty() ? "" : ", " + config));
54}

56Net readNet(const String& _framework, const std::vector<uchar>& bufferModel,
57 const std::vector<uchar>& bufferConfig, int engine)
58{
59 String framework = toLowerCase(_framework);
60 if (framework == "caffe")
61 CV_Error(Error::StsError, "Caffe importer has been removed. Please use ONNX-converted models or use an older OpenCV version.");
62 if (framework == "onnx")
63 return readNetFromONNX(bufferModel, engine);
64 else if (framework == "tensorflow")
65 return readNetFromTensorflow(bufferModel, bufferConfig, engine);
66 else if (framework == "darknet")
67 CV_Error(Error::StsError, "Darknet importer has been removed. Please use ONNX-converted models or use an older OpenCV version.");
68 else if (framework == "dldt" || framework == "openvino")
69 return readNetFromModelOptimizer(bufferConfig, bufferModel);
70 else if (framework == "tflite")
71 return readNetFromTFLite(bufferModel, engine);
72 CV_Error(Error::StsError, "Cannot determine an origin framework with a name " + framework);
73}
