34
35void Net::Impl::fuseLayers(const std::vector<LayerPin>& blobsToKeep_)
36{
37 CV_TRACE_FUNCTION();
38
39 if(!fusion || (preferableBackend != DNN_BACKEND_OPENCV &&
40 preferableBackend != DNN_BACKEND_CUDA &&
41 preferableBackend != DNN_BACKEND_INFERENCE_ENGINE_NGRAPH &&
42 preferableBackend != DNN_BACKEND_TIMVX &&
43 preferableBackend != DNN_BACKEND_VKCOM))
44 return;
45
46#if 0 // FIXIT mode without fusion is broken due to unsupported layers and handling of "custom" nodes
47 if (preferableBackend == DNN_BACKEND_INFERENCE_ENGINE_NGRAPH)
48 return;
49#endif
50
51 // scan through all the layers. If there is convolution layer followed by the activation layer,
52 // we try to embed this activation into the convolution and disable separate execution of the activation
53
54 // FIXIT replace by layersToKeep to avoid hacks like "LayerPin(lid, 0)"
55 std::set<LayerPin> pinsToKeep(blobsToKeep_.begin(),
56 blobsToKeep_.end());
57 for (MapIdToLayerData::const_iterator it = layers.begin(); it != layers.end(); it++)
58 {
59 int lid = it->first;
60 LayerData& ld = layers[lid];
61 if (ld.skip)
62 {
63 printf_(("skipped %s: %s\n", ld.layerInstance->name.c_str(), ld.layerInstance->type.c_str()));
64 continue;
65 }
66 printf_(("analyzing %s: %s\n", ld.layerInstance->name.c_str(), ld.layerInstance->type.c_str()));
67
68 // the optimization #1. try to fuse batch norm, scaling and/or activation layers
69 // with the current layer if they follow it. Normally, the are fused with the convolution layer,
70 // but some of them (like activation) may be fused with fully-connected, elemwise (+) and
71 // some other layers.
72 Ptr<Layer>& currLayer = ld.layerInstance;
73 if (ld.consumers.size() == 1 && pinsToKeep.count(LayerPin(lid, 0)) == 0)
74 {
75 LayerData* nextData = &layers[ld.consumers[0].lid];
76 LayerPin lpNext(ld.consumers[0].lid, 0);
77 while (nextData)
78 {
79#ifdef HAVE_INF_ENGINE
80 if (preferableBackend == DNN_BACKEND_INFERENCE_ENGINE_NGRAPH && pinsToKeep.count(lpNext) != 0)
81 {
82 CV_LOG_DEBUG(NULL, "DNN/IE: skip fusing with 'output' node: " << nextData->name << "@" << nextData->type);
83 break;
84 }
85#endif
86 /* we use `tryFuse` member of convolution layer to fuse eltwise/naryEltwise later
87 * it's not intended to be fused here; hence, we stop when we encounter eltwise
88 */
89 if (preferableBackend == DNN_BACKEND_CUDA && ld.type == "Convolution" &&
90 (nextData->type == "Eltwise" || nextData->type == "NaryEltwise"))
91 break;
92 Ptr<Layer> nextLayer = nextData->layerInstance;
93 if (currLayer->tryFuse(nextLayer))
94 {
95 printf_(("\tfused with %s\n", nextLayer->name.c_str()));
96 nextData->skip = true;
97 ld.outputBlobs = layers[lpNext.lid].outputBlobs;
98 ld.outputBlobsWrappers = layers[lpNext.lid].outputBlobsWrappers;
99 if (nextData->consumers.size() == 1)
100 {
101 int nextLayerId = nextData->consumers[0].lid;
102 nextData = &layers[nextLayerId];
103 lpNext = LayerPin(nextLayerId, 0);
104 }
105 else
106 {
107 nextData = 0;
108 break;
109 }
110 }
111 else
112 break;
113 }
114
115 if (preferableBackend != DNN_BACKEND_OPENCV && preferableBackend != DNN_BACKEND_CUDA
116 && preferableBackend != DNN_BACKEND_VKCOM)
117 continue; // Go to the next layer.
118
119 // TODO: OpenCL target support more fusion styles.
120 if ( preferableBackend == DNN_BACKEND_OPENCV && IS_DNN_OPENCL_TARGET(preferableTarget) &&
121 (!cv::ocl::useOpenCL() || (ld.layerInstance->type != "Convolution" &&
122 ld.layerInstance->type != "MVN" && ld.layerInstance->type != "Pooling" &&
123 ld.layerInstance->type != "Concat")) )
124 continue;
125
126 if (preferableBackend == DNN_BACKEND_CUDA && IS_DNN_CUDA_TARGET(preferableTarget)
127 && ld.layerInstance->type != "Convolution"
128 && ld.layerInstance->type != "Concat")
129 continue;
130
131 while (nextData)
132 {
133 // For now, OpenCL target support fusion with activation of ReLU/ChannelsPReLU/Power/Tanh
134 if (IS_DNN_OPENCL_TARGET(preferableTarget) &&
135 nextData->type != "ReLU" &&
136 nextData->type != "ChannelsPReLU" &&
137 nextData->type != "ReLU6" &&
138 nextData->type != "TanH" &&
139 nextData->type != "Power")
140 break;
141
142 Ptr<ActivationLayer> nextActivLayer = nextData->layerInstance.dynamicCast<ActivationLayer>();
143 if (nextActivLayer.empty())
144 break;
145
146 // For now, Vulkan target support fusion with activation of ReLU/ReLU6
147 if (IS_DNN_VULKAN_TARGET(preferableTarget))
148 {
149 if (nextData->type == "ReLU")
150 {
151 Ptr<ReLULayer> nextReLULayer = nextData->layerInstance.dynamicCast<ReLULayer>();
152 CV_Assert(nextReLULayer);
153 if (nextReLULayer->negativeSlope != 0.0f)
154 break; // Skip LeakyReLU
155 }
156 else if (nextData->type == "ReLU6")
157 {
158 Ptr<ReLU6Layer> nextReLU6Layer = nextData->layerInstance.dynamicCast<ReLU6Layer>();
159 CV_Assert(nextReLU6Layer);
160
161 if( fabs(nextReLU6Layer->minValue) > FLT_EPSILON || fabs(nextReLU6Layer->maxValue - 6.0f) > FLT_EPSILON)
162 break; // Skip ReLU6 if the minValue != 0 or maxValue != 6.
163 }
164 else
165 break;
166 }
167
168 if (currLayer->setActivation(nextActivLayer))
169 {
170 printf_(("\tfused with %s\n", nextActivLayer->name.c_str()));
171 nextData->skip = true;
172 ld.outputBlobs = layers[lpNext.lid].outputBlobs;
173 ld.outputBlobsWrappers = layers[lpNext.lid].outputBlobsWrappers;
174 if (nextData->consumers.size() == 1)
175 {
176 int nextLayerId = nextData->consumers[0].lid;
177 nextData = &layers[nextLayerId];
178 lpNext = LayerPin(nextLayerId, 0);
179 }
180 else
181 {
182 nextData = 0;
183 break;
184 }
185 }
186 else
187 break;
188 }
189
190 // CPU: fuse Convolution 2D layer followed by Add + activation.
191 while (nextData && (IS_DNN_CPU_TARGET(preferableTarget)) && ld.layerInstance->type == "Convolution")
192 {
193 // Note that we can only deal with conv + Add + activ here.
194 // To avoid the order like: conv + activ + add, if we found the conv has been fused with activ, we break.
195 Ptr<ConvolutionLayer> convLayer = ld.layerInstance.dynamicCast<ConvolutionLayer>();
196
197 // Only Convolution layer without fusion Activation supports this fusion, other-wise, we skip.
198 if (convLayer->fusedActivation)
199 break;
200
201 // For now, there are currently two layers in OpenCV that run the Add operator.
202 Ptr<NaryEltwiseLayer> nextNaryEltwiseLayer = nextData->layerInstance.dynamicCast<NaryEltwiseLayer>();
203 Ptr<EltwiseLayer> nextEltwiseLayer = nextData->layerInstance.dynamicCast<EltwiseLayer>();
204 if (nextNaryEltwiseLayer.empty() && nextEltwiseLayer.empty())
205 break;
206
207 if (nextData->inputBlobsId.size() != 2)
208 break;
209
210 if (!nextData->params.has("operation") || toLowerCase(nextData->params.get<String>("operation")) != "add")
211 {
212 CV_LOG_DEBUG(NULL, "DNN/CPU: fusion with NaryEltwise or Eltwise Layer operation is not supported: "
213 << toLowerCase(nextData->params.get<String>("operation", "sum")));
214 break;
215 }
216
217 // This optimization is for cases like
218 // some_layer conv
219 // | |
220 // +-- eltwise or (naryEltwise) --+
221 // |
222 // activ
223 // This way all the element-wise computations
224 // (i.e. some_layer+conv) would be done at [conv] layer.
225 // So we need to replace [conv]'s output blob to [eltwise]'s one
226 // considering that [activ] is an in-place layer.
227 // Also we need to move all the consumers' references.
228 // To prevent memory collisions (i.e. when input of
229 // [conv] and output of [eltwise or naryEltwise] is the same blob)
230 // we allocate a new blob.
231 {
232 LayerData *naryOrEltwiseData = nextData;
233
234 // Eltwise or NaryEltwise layer has two inputs. We need to determine which
235 // is a base convolution layer and which could be used as it's bias.
236 LayerData* biasLayerData = 0;
237 for (int i = 0; i < 2; ++i)
238 {
239 LayerData *downLayerData = &layers[naryOrEltwiseData->inputBlobsId[i].lid];
240 CV_Assert(downLayerData);
241 // If the current downLayerData is skip, it means it is fused into the parent node.
242 while (downLayerData->skip)
243 {
244 if (downLayerData->inputBlobsId.size() == 1)
245 downLayerData = &layers[downLayerData->inputBlobsId[0].lid];
246 else
247 {
248 downLayerData = 0;
249 break;
250 }
251 }
252
253 if (downLayerData && ld.id == downLayerData->id)
254 {
255 biasLayerData = &layers[naryOrEltwiseData->inputBlobsId[1 - i].lid];
256 break;
257 }
258 }
259
260 // We check if biasLayerData is expected layer.
261 if (!biasLayerData)
262 break;
263
264 // We check if the bias output shape and the ld output shape are the same.
265 MatShape biasOutShape = shape(biasLayerData->outputBlobs[0]);
266 MatShape ldOutShape = shape(ld.outputBlobs[0]);
267 if (biasOutShape != ldOutShape)
268 break;
269
270 CV_Assert(biasLayerData);
271 {
272 // fuse naryEltwise layer
273 // bias must already be computed to fuse => bias layer must appear before convolution
274 if (biasLayerData->id < ld.id && biasLayerData->consumers.size() == 1)
275 {
276 // conv + naryEltwise.
277 CV_Assert_N(biasLayerData->outputBlobs.size() == 1, ld.inputBlobs.size() == 1);
278 CV_Assert_N(biasLayerData->outputBlobsWrappers.size() == 1, ld.inputBlobsWrappers.size() == 1);
279
280 printf_(("\tfused with %s\n", nextNaryEltwiseLayer->name.c_str()));
281 naryOrEltwiseData->skip = true;
282
283
284 CV_Assert_N(ld.outputBlobs.size() == 1, ld.outputBlobsWrappers.size() == 1);
285 // Note: Here's a trick. We set the output of conv as the output of biasLayer.
286 ld.outputBlobs[0] = ld.outputBlobs[0].clone();
287 ld.outputBlobsWrappers[0] = wrap(ld.outputBlobs[0]);
288
289 // Recursively modifies the output data of biasLayerData and its parent.
290 std::vector<LayerData*> skipDataList;
291 skipDataList.push_back(biasLayerData);
292
293 while (!skipDataList.empty())
294 {
295 LayerData* skipData = skipDataList.back();
296 skipDataList.pop_back();
297
298 CV_Assert(skipData->outputBlobs.size() == 1);
299 skipData->outputBlobs[0] = ld.outputBlobs[0];
300 skipData->outputBlobsWrappers[0] = ld.outputBlobsWrappers[0];
301 if (skipData->skip)
302 {
303 for (auto& inputLayerId : skipData->inputLayersId)
304 {
305 LayerData* inputld = &layers[inputLayerId];
306
307 if (inputld && inputld->outputBlobs.size() == 1)
308 skipDataList.push_back(inputld);
309 }
310 }
311 }
312
313 naryOrEltwiseData->outputBlobs = ld.outputBlobs;
314 naryOrEltwiseData->outputBlobsWrappers = ld.outputBlobsWrappers;
315
316 // set the fusedAdd flag in [Conv];
317 convLayer->fusedAdd = true;
318 LayerData* finalData = naryOrEltwiseData;
319 /* After fused Conv + naryEltwise or eltwise, we can fuse activation if:
320 * => activation layer that follows is the only consumer of eltwise output
321 * => activation layer does not process multiple inputs
322 * => we do not require to keep the output of eltwise
323 */
324 if (naryOrEltwiseData->consumers.size() == 1)
325 {
326 Ptr<ActivationLayer> nextFusabeleActivLayer;
327 LayerData* nextAct = &layers[naryOrEltwiseData->consumers[0].lid];
328
329 if (nextData->outputBlobs.size() == 1)
330 nextFusabeleActivLayer = nextAct->layerInstance.dynamicCast<ActivationLayer>();
331
332 if (!nextFusabeleActivLayer.empty())
333 {
334 convLayer->setActivation(nextFusabeleActivLayer);
335 nextAct->skip = true;
336
337 nextAct->outputBlobs = ld.outputBlobs;
338 nextAct->outputBlobsWrappers = ld.outputBlobsWrappers;
339 }
340 }
341
342 // Move references of finalData (eltwise or activation) layer consumers to the newly allocated blob.
343 for (int i = 0; i < finalData->consumers.size(); ++i)
344 {
345 LayerData& consumer = layers[finalData->consumers[i].lid];
346 for (int j = 0; j < consumer.inputBlobsId.size(); ++j)
347 {
348 if (consumer.inputBlobsId[j].lid == finalData->id)
349 {
350 consumer.inputBlobs[j] = &ld.outputBlobs[0];
351 consumer.inputBlobsWrappers[j] = ld.outputBlobsWrappers[0];
352 break;
353 }
354 }
355 }
356 }
357 }
358 }
359 break;
360 }
361
362 // OpenCL: fuse convolution layer followed by eltwise + relu
363 // CUDA: fuse convolution layer followed by eltwise/naryEltwise (and optional activation)
364 while (nextData &&
365 (IS_DNN_OPENCL_TARGET(preferableTarget) || IS_DNN_CUDA_TARGET(preferableTarget)) &&
366 ld.layerInstance->type == "Convolution"
367 ) // semantic of 'if'
368 {
369 Ptr<EltwiseLayer> nextEltwiseLayer = nextData->layerInstance.dynamicCast<EltwiseLayer>();
370 Ptr<NaryEltwiseLayer> nextNaryEltwiseLayer = nextData->layerInstance.dynamicCast<NaryEltwiseLayer>();
371 if (nextEltwiseLayer.empty() && nextNaryEltwiseLayer.empty())
372 break;
373 LayerData *naryOrEltwiseData = nextData;
374
375 // TODO: fused the Conv+NaryEltwise on OpenCL backend. At present, we can only support it at CUDA backend.
376 if (IS_DNN_OPENCL_TARGET(preferableTarget) && nextNaryEltwiseLayer)
377 break;
378
379#ifdef HAVE_CUDA
380 // CUDA backend supports fusion with eltwise sum (without variable channels)
381 if (IS_DNN_CUDA_TARGET(preferableTarget) && (!nextEltwiseLayer.empty() || !nextNaryEltwiseLayer.empty()))
382 {
383 // we create a temporary backend node for eltwise layer to obtain the eltwise configuration
384 cuda4dnn::csl::CSLContext context; // assume that initCUDA and EltwiseOp do not use the context during init
385
386 if (!nextData->layerInstance->supportBackend(DNN_BACKEND_CUDA))
387 break;
388
389 const auto node = nextData->layerInstance->initCUDA(&context, nextData->inputBlobsWrappers, nextData->outputBlobsWrappers);
390 auto eltwiseNode = node.dynamicCast<cuda4dnn::EltwiseOpBase>();
391
392 // broadcasting not supported in fused ops
393 auto required_shape = shape(nextData->outputBlobs[0]);
394 for (int i = 0; i < nextData->inputBlobs.size(); i++)
395 {
396 if (shape(*nextData->inputBlobs[i]) != required_shape)
397 {
398 eltwiseNode.reset();
399 break;
400 }
401 }
402
403 // CUDA backend uses EltwiseOp when all operands have the same number of channels; otherwise, ShortcutOp is used.
404 // Hence, a successful cast to EltwiseOp implies that the number of channels is same in all operand tensors.
405 if (eltwiseNode.empty() || eltwiseNode->op != cuda4dnn::EltwiseOpType::SUM || !eltwiseNode->coeffs.empty())
406 break;
407 }
408#endif
409
410 if (IS_DNN_OPENCL_TARGET(preferableTarget) && pinsToKeep.count(lpNext) != 0)
411 break;
412 if (nextData->inputBlobsId.size() != 2)
413 break;
414
415 if (IS_DNN_OPENCL_TARGET(preferableTarget))
416 {
417 if (!nextData->params.has("operation") || toLowerCase(nextData->params.get<String>("operation")) == "sum")
418 {
419 if (nextData->params.has("coeff"))
420 {
421 DictValue paramCoeff = nextData->params.get("coeff");
422 int n = paramCoeff.size();
423 bool isCoeffOneOne = (n == 2);
424 for (int i = 0; isCoeffOneOne && i < n; i++)
425 {
426 float c = paramCoeff.get<float>(i);
427 isCoeffOneOne &= (c == 1.0f);
428 }
429 if (!isCoeffOneOne)
430 {
431 CV_LOG_DEBUG(NULL, "DNN/OpenCL: fusion of 'Sum' without coeffs (or {1.0, 1.0}) is supported only");
432 break;
433 }
434 }
435 }
436 else
437 {
438 CV_LOG_DEBUG(NULL, "DNN/OpenCL: fusion with eltwise operation is not supported: " << nextData->params.get<String>("operation"));
439 break;
440 }
441 }
442
443 {
444 LayerData *eltwiseData = nextData;
445
446 // Eltwise/NaryEltwise layer has two inputs. We need to determine which
447 // is a base convolution layer and which could be used as it's bias.
448 LayerData* biasLayerData = 0;
449 for (int i = 0; i < 2; ++i)
450 {
451 LayerData *downLayerData = &layers[eltwiseData->inputBlobsId[i].lid];
452 CV_Assert(downLayerData);
453 while (downLayerData->skip)
454 {
455 if (downLayerData->inputBlobsId.size() == 1)
456 downLayerData = &layers[downLayerData->inputBlobsId[0].lid];
457 else
458 {
459 downLayerData = 0;
460 break;
461 }
462 }
463 if (downLayerData && ld.id == downLayerData->id)
464 {
465 biasLayerData = &layers[eltwiseData->inputBlobsId[1 - i].lid];
466 break;
467 }
468 }
469 CV_Assert(biasLayerData);
470 {
471 // fuse eltwise + activation layer
472 // bias must already be computed to fuse => bias layer must appear before convolution
473 if (biasLayerData->id < ld.id)
474 {
475 /* we can fuse activation if:
476 * => activation layer that follows is the only consumer of eltwise output
477 * => activation layer does not process multiple inputs
478 * => we do not require to keep the output of eltwise
479 */
480 Ptr<ActivationLayer> nextFusabeleActivLayer;
481 if (eltwiseData->consumers.size() == 1 && pinsToKeep.count(lpNext) == 0)
482 {
483 nextData = &layers[eltwiseData->consumers[0].lid];
484 lpNext = LayerPin(eltwiseData->consumers[0].lid, 0);
485 CV_Assert(nextData);
486 if (nextData->outputBlobs.size() == 1)
487 nextFusabeleActivLayer = nextData->layerInstance.dynamicCast<ActivationLayer>();
488 }
489 else
490 {
491 // OCL backend cannot fuse in this case but the CUDA backend can continue with just eltwise
492 nextData = 0;
493 }
494
495 // the requirements of OCV OpenCL backend and CUDA backend are different
496 // we need to check them separately; hence, the fuse variables
497 bool fuse_eltwise = false, fuse_activation = false;
498
499 Ptr<PowerLayer> activ_power;
500 if (IS_DNN_OPENCL_TARGET(preferableTarget) && !nextFusabeleActivLayer.empty() &&
501 nextData &&
502 (!nextData->type.compare("ReLU") ||
503 !nextData->type.compare("ChannelsPReLU") ||
504 (!nextData->type.compare("Power") && (activ_power = nextFusabeleActivLayer.dynamicCast<PowerLayer>()) && activ_power->scale == 1.0f)
505 ) &&
506 currLayer->setActivation(nextFusabeleActivLayer))
507 {
508 fuse_eltwise = true;
509 fuse_activation = true;
510 }
511
512 if (IS_DNN_CUDA_TARGET(preferableTarget))
513 {
514 /* supported fusion options:
515 * => convolution + eltwise
516 * => activation(convolution) + eltwise
517 * > convolution + activation would have been fused already; we have to fuse eltwise
518 * => activation(convolution + eltwise)
519 * > fuse eltwise and then activation
520 */
521 Ptr<Layer> layer = nullptr;
522 if (nextNaryEltwiseLayer)
523 layer = nextNaryEltwiseLayer.staticCast<Layer>();
524 else if (nextEltwiseLayer)
525 layer = nextEltwiseLayer.staticCast<Layer>();
526 else
527 CV_Error(Error::StsError, "Both nextNaryEltwiseLayer and nextEltwiseLayer are empty!");
528
529 if (currLayer->tryFuse(layer))
530 {
531 fuse_eltwise = true; /* eltwise was successfully fused */
532 if (!nextFusabeleActivLayer.empty() && nextData)
533 {
534 if ((!nextData->type.compare("ReLU") ||
535 !nextData->type.compare("ReLU6") ||
536 !nextData->type.compare("Power") ||
537 !nextData->type.compare("TanH") ||
538 !nextData->type.compare("Sigmoid") ||
539 !nextData->type.compare("Swish") ||
540 !nextData->type.compare("Mish")) &&
541 currLayer->setActivation(nextFusabeleActivLayer))
542 {
543 // activation was fused
544 fuse_activation = true;
545 }
546 }
547 }
548 }
549
550 CV_Assert(!fuse_activation || fuse_eltwise); /* cannot fuse activation without eltwise */
551 if(fuse_eltwise && fuse_activation)
552 {
553 CV_Assert(nextData);
554 CV_Assert_N(biasLayerData->outputBlobsWrappers.size() == 1, ld.inputBlobsWrappers.size() == 1);
555 ld.inputBlobsWrappers.push_back(biasLayerData->outputBlobsWrappers[0]);
556
557 if (nextEltwiseLayer)
558 printf_(("\tfused with %s\n", nextEltwiseLayer->name.c_str()));
559 else if (nextNaryEltwiseLayer)
560 printf_(("\tfused with %s\n", nextEltwiseLayer->name.c_str()));
561 else
562 CV_Error(Error::StsError, "Both nextNaryEltwiseLayer and nextEltwiseLayer are empty!");
563
564 printf_(("\tfused with %s\n", nextFusabeleActivLayer->name.c_str()));
565 eltwiseData->skip = true;
566 nextData->skip = true;
567 // This optimization for cases like
568 // some_layer conv
569 // | |
570 // +-- eltwise --+
571 // |
572 // activ
573 // This way all the element-wise computations
574 // (i.e. some_layer+conv or some_layer*conv)
575 // would be done at [conv] layer. So we need to
576 // replace [conv]'s output blob to [eltwise]'s one
577 // considering that [activ] is an in-place layer.
578 // Also we need to move all the consumers' references.
579 // To prevent memory collisions (i.e. when input of
580 // [conv] and output of [eltwise] is the same blob)
581 // we allocate a new blob.
582 CV_Assert_N(ld.outputBlobs.size() == 1, ld.outputBlobsWrappers.size() == 1);
583 ld.outputBlobs[0] = ld.outputBlobs[0].clone();
584 ld.outputBlobsWrappers[0] = wrap(ld.outputBlobs[0]);
585
586 eltwiseData->outputBlobs = ld.outputBlobs;
587 nextData->outputBlobs = ld.outputBlobs;
588 eltwiseData->outputBlobsWrappers = ld.outputBlobsWrappers;
589 nextData->outputBlobsWrappers = ld.outputBlobsWrappers;
590
591 // Move references of [activ] layer consumers to the newly allocated blob.
592 for (int i = 0; i < nextData->consumers.size(); ++i)
593 {
594 LayerData& consumer = layers[nextData->consumers[i].lid];
595 for (int j = 0; j < consumer.inputBlobsId.size(); ++j)
596 {
597 if (consumer.inputBlobsId[j].lid == lpNext.lid)
598 {
599 consumer.inputBlobs[j] = &ld.outputBlobs[0];
600 consumer.inputBlobsWrappers[j] = ld.outputBlobsWrappers[0];
601 break;
602 }
603 }
604 }
605 }
606 else if (fuse_eltwise) // conv + eltwise/naryEltwise (note: conv could have fused activations before eltwise)
607 {
608 CV_Assert(IS_DNN_CUDA_TARGET(preferableTarget));
609 CV_Assert_N(biasLayerData->outputBlobsWrappers.size() >= 1, ld.inputBlobsWrappers.size() == 1);
610 // Iterate over eltwise inputs to find exact output id
611 for (const auto& pin : naryOrEltwiseData->inputBlobsId)
612 {
613 if (pin.lid == biasLayerData->id)
614 {
615 ld.inputBlobsWrappers.push_back(biasLayerData->outputBlobsWrappers[pin.oid]);
616 break;
617 }
618 }
619 CV_Assert(ld.inputBlobsWrappers.size() == 2); // Check input was found
620
621 if (nextEltwiseLayer)
622 printf_(("\tfused with %s\n", nextEltwiseLayer->name.c_str()));
623 else if (nextNaryEltwiseLayer)
624 printf_(("\tfused with %s\n", nextEltwiseLayer->name.c_str()));
625 else
626 CV_Error(Error::StsError, "Both nextNaryEltwiseLayer and nextEltwiseLayer are empty!");
627
628 eltwiseData->skip = true;
629 // This optimization is for cases like
630 // some_layer conv (maybe fused with activ)
631 // | |
632 // +-- eltwise --+
633 //
634 // This way all the element-wise computations
635 // (i.e. some_layer+conv or some_layer*conv)
636 // would be done at [conv] layer. So we need to
637 // replace [conv]'s output blob to [eltwise]'s one.
638 // Also, we need to move all the consumers' references.
639 // To prevent memory collisions (i.e. when input of
640 // [conv] and output of [eltwise] is the same blob)
641 // we allocate a new blob.
642 CV_Assert_N(ld.outputBlobs.size() == 1, ld.outputBlobsWrappers.size() == 1);
643 ld.outputBlobs[0] = ld.outputBlobs[0].clone();
644 ld.outputBlobsWrappers[0] = wrap(ld.outputBlobs[0]);
645
646 eltwiseData->outputBlobs = ld.outputBlobs;
647 eltwiseData->outputBlobsWrappers = ld.outputBlobsWrappers;
648
649 // Move references of [eltwise] layer consumers to the newly allocated blob.
650 for (int i = 0; i < eltwiseData->consumers.size(); ++i)
651 {
652 LayerData& consumer = layers[eltwiseData->consumers[i].lid];
653 for (int j = 0; j < consumer.inputBlobsId.size(); ++j)
654 {
655 if (consumer.inputBlobsId[j].lid == eltwiseData->id)
656 {
657 consumer.inputBlobs[j] = &ld.outputBlobs[0];
658 consumer.inputBlobsWrappers[j] = ld.outputBlobsWrappers[0];
659 break;
660 }
661 }
662 }
663 }
664 }
665 }
666 }
667
668 break;
669 }
670 }
671
672 if (preferableBackend != DNN_BACKEND_OPENCV && preferableBackend != DNN_BACKEND_CUDA)
673 continue; // Go to the next layer.
674
675 Ptr<ConcatLayer> concatLayer = ld.layerInstance.dynamicCast<ConcatLayer>();
676 if( !concatLayer.empty() && !concatLayer->padding && ld.outputBlobs.size() == 1 )
677 {
678 Mat& output = ld.outputBlobs[0];
679 UMat umat_output;
680#ifdef HAVE_OPENCL
681 if (!ld.outputBlobsWrappers.empty() &&
682 (preferableBackend == DNN_BACKEND_OPENCV && IS_DNN_OPENCL_TARGET(preferableTarget)))
683 {
684 size_t i, ninputs = ld.inputBlobsId.size();
685 bool conv_layer = true;
686 for( i = 0; i < ninputs; i++ )
687 {
688 LayerPin pin = ld.inputBlobsId[i];
689 LayerData* inp_i_data = &layers[pin.lid];
690 while(inp_i_data->skip &&
691 inp_i_data->inputBlobsId.size() == 1 &&
692 inp_i_data->consumers.size() == 1)
693 {
694 pin = inp_i_data->inputBlobsId[0];
695 inp_i_data = &layers[pin.lid];
696 }
697 conv_layer = conv_layer && (getLayerInstance(*inp_i_data)->type == "Convolution");
698 }
699 if (!conv_layer)
700 continue;
701 std::vector<UMat> umat_outputBlobs;
702 umat_outputBlobs = OpenCLBackendWrapper::getUMatVector(ld.outputBlobsWrappers);
703 umat_output = umat_outputBlobs[0];
704 }
705#endif
706
707 // TODO: in general, this optimization can always be done, but
708 // many layers currently check that the input/output blobs are
709 // continuous arrays. Unfortunately, this is not true when
710 // the concatenation optimization is applied with batch_size > 1.
711 // so, for now, we only apply this optimization in the most popular
712 // case batch_size == 1.
713 int axis = normalize_axis(concatLayer->axis, output.dims);
714 if( output.total(0, axis) == 1 )
715 {
716 size_t i, ninputs = ld.inputBlobsId.size();
717 std::vector<LayerPin> realinputs(ninputs);
718 for( i = 0; i < ninputs; i++ )
719 {
720 LayerPin pin = ld.inputBlobsId[i];
721 LayerData* inp_i_data = &layers[pin.lid];
722 while(inp_i_data->skip &&
723 inp_i_data->inputBlobsId.size() == 1 &&
724 inp_i_data->consumers.size() == 1)
725 {
726 pin = inp_i_data->inputBlobsId[0];
727 inp_i_data = &layers[pin.lid];
728 }
729 printf_(("\treal input for %s is %s\n",
730 layers[ld.inputBlobsId[i].lid].getLayerInstance()->name.c_str(),
731 inp_i_data->getLayerInstance()->name.c_str()));
732
733 if(inp_i_data->skip || inp_i_data->consumers.size() != 1)
734 break;
735#ifdef HAVE_CUDA
736 /* Risk: Not every operation in "NaryEltwise" is supported in the CUDA backend. There is a chance
737 that Concat's output is filled with data in both host and device, leading to data missing.
738 See https://github.com/opencv/opencv/issues/24721 for more details.
739 */
740 if (preferableBackend == DNN_BACKEND_CUDA &&
741 (inp_i_data->layerInstance->supportBackend(DNN_BACKEND_CUDA) == false ||
742 (inp_i_data->layerInstance->type != "Convolution" &&
743 inp_i_data->layerInstance->type != "Pooling" &&
744 inp_i_data->layerInstance->type != "Resize" &&
745 inp_i_data->layerInstance->type != "Flatten" &&
746 inp_i_data->layerInstance->type != "Permute" &&
747 inp_i_data->layerInstance->type != "Reorg" &&
748 inp_i_data->layerInstance->type != "Eltwise" &&
749 inp_i_data->layerInstance->type != "NaryEltwise" &&
750 inp_i_data->layerInstance.dynamicCast<ActivationLayer>().empty())))
751 {
752 break;
753 }
754#endif
755 realinputs[i] = pin;
756 }
757
758 if( i >= ninputs )
759 {
760 // Allocate new memory to prevent collisions during memory
761 // reusing (see https://github.com/opencv/opencv/pull/10456).
762 output = output.clone();
763#ifdef HAVE_OPENCL
764 if (preferableBackend == DNN_BACKEND_OPENCV &&
765 IS_DNN_OPENCL_TARGET(preferableTarget))
766 {
767 std::vector<UMat> umats(1);
768 umat_output = umat_output.clone();
769 umats[0] = umat_output;
770 OpenCLBackendWrapper::update(ld.outputBlobsWrappers, umats);
771 }
772#endif
773
774#ifdef HAVE_CUDA
775 if (preferableBackend == DNN_BACKEND_CUDA)
776 ld.outputBlobsWrappers[0] = wrap(output);
777#endif
778 std::vector<Range> chrange(output.dims, Range::all());
779 int ofs = 0;
780 for( i = 0; i < ninputs; i++ )
781 {
782 LayerPin pin = realinputs[i];
783 LayerData* inp_i_data = &layers[pin.lid];
784 int channels_i = ld.inputBlobs[i]->size[axis];
785 chrange[axis] = Range(ofs, ofs + channels_i);
786 printf_(("\toutput %s(%d) to channels (%d, %d)\n", inp_i_data->layerInstance->name.c_str(),
787 pin.oid, ofs, ofs + channels_i));
788 ofs += channels_i;
789 Mat output_slice = output(chrange);
790 Mat& curr_output = inp_i_data->outputBlobs[pin.oid];
791 CV_Assert(output_slice.isContinuous() && output_slice.size == curr_output.size);
792 Mat* oldPtr = &curr_output;
793 curr_output = output_slice;
794#ifdef HAVE_OPENCL
795 if (preferableBackend == DNN_BACKEND_OPENCV && IS_DNN_OPENCL_TARGET(preferableTarget))
796 {
797 std::vector<UMat> umats(inp_i_data->outputBlobsWrappers.size());
798 umats[pin.oid] = umat_output(chrange);
799 OpenCLBackendWrapper::update(inp_i_data->outputBlobsWrappers, umats);
800 }
801#endif
802#ifdef HAVE_CUDA
803 if (preferableBackend == DNN_BACKEND_CUDA)
804 {
805 auto cuda_wrapper = wrap(output).dynamicCast<CUDABackendWrapper>();
806 auto offset = chrange[axis].start * output_slice.total(axis + 1, output.dims);
807 auto new_shape = shape(output_slice);
808 cuda_wrapper->update(new_shape, offset);
809 inp_i_data->outputBlobsWrappers[pin.oid] = cuda_wrapper.staticCast<BackendWrapper>();
810 }
811#endif
812 // Layers that refer old input Mat will refer to the
813 // new data but the same Mat object.
814 CV_Assert_N(curr_output.data == output_slice.data, oldPtr == &curr_output);
815 }
816
817#ifdef HAVE_CUDA
818 if (preferableBackend == DNN_BACKEND_CUDA)
819 {
820 for (int i = 0; i < ld.consumers.size(); i++)
821 {
822 LayerData& consumer = layers[ld.consumers[i].lid];
823 for (int j = 0; j < consumer.inputBlobsId.size(); j++)
824 {
825 if (consumer.inputBlobsId[j].lid == ld.id)
826 {
827 CV_Assert(consumer.inputBlobs[j]->data == ld.outputBlobs[0].data);
828 consumer.inputBlobsWrappers[j] = ld.outputBlobsWrappers[0];
829 break;
830 }
831 }
832 }
833 }
834#endif
835 ld.skip = true;
836 printf_(("\toptimized out Concat layer %s\n", concatLayer->name.c_str()));
837 }
838 }
839 }
840 }
