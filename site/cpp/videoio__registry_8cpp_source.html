212class VideoBackendRegistry
213{
214protected:
215 std::vector<VideoBackendInfo> enabledBackends;
216 VideoBackendRegistry()
217 {
218 const int N = sizeof(builtin_backends)/sizeof(builtin_backends[0]);
219 enabledBackends.assign(builtin_backends, builtin_backends + N);
220 for (int i = 0; i < N; i++)
221 {
222 VideoBackendInfo& info = enabledBackends[i];
223 info.priority = 1000 - i * 10;
224 }
225 CV_LOG_DEBUG(NULL, "VIDEOIO: Builtin backends(" << N << "): " << dumpBackends());
226 if (readPrioritySettings())
227 {
228 CV_LOG_INFO(NULL, "VIDEOIO: Updated backends priorities: " << dumpBackends());
229 }
230 int enabled = 0;
231 for (int i = 0; i < N; i++)
232 {
233 VideoBackendInfo& info = enabledBackends[enabled];
234 if (enabled != i)
235 info = enabledBackends[i];
236 size_t param_priority = utils::getConfigurationParameterSizeT(cv::format("OPENCV_VIDEOIO_PRIORITY_%s", info.name).c_str(), (size_t)info.priority);
237 CV_Assert(param_priority == (size_t)(int)param_priority); // overflow check
238 if (param_priority > 0)
239 {
240 info.priority = (int)param_priority;
241 enabled++;
242 }
243 else
244 {
245 CV_LOG_INFO(NULL, "VIDEOIO: Disable backend: " << info.name);
246 }
247 }
248 enabledBackends.resize(enabled);
249 CV_LOG_DEBUG(NULL, "VIDEOIO: Available backends(" << enabled << "): " << dumpBackends());
250 std::sort(enabledBackends.begin(), enabledBackends.end(), sortByPriority);
251 CV_LOG_INFO(NULL, "VIDEOIO: Enabled backends(" << enabled << ", sorted by priority): " << dumpBackends());
252 }

254 static std::vector<std::string> tokenize_string(const std::string& input, char token)
255 {
256 std::vector<std::string> result;
257 std::string::size_type prev_pos = 0, pos = 0;
258 while((pos = input.find(token, pos)) != std::string::npos)
259 {
260 result.push_back(input.substr(prev_pos, pos-prev_pos));
261 prev_pos = ++pos;
262 }
263 result.push_back(input.substr(prev_pos));
264 return result;
265 }
266 bool readPrioritySettings()
267 {
268 bool hasChanges = false;
269 cv::String prioritized_backends = utils::getConfigurationParameterString("OPENCV_VIDEOIO_PRIORITY_LIST");
270 if (prioritized_backends.empty())
271 return hasChanges;
272 CV_LOG_INFO(NULL, "VIDEOIO: Configured priority list (OPENCV_VIDEOIO_PRIORITY_LIST): " << prioritized_backends);
273 const std::vector<std::string> names = tokenize_string(prioritized_backends, ',');
274 for (size_t i = 0; i < names.size(); i++)
275 {
276 const std::string& name = names[i];
277 bool found = false;
278 for (size_t k = 0; k < enabledBackends.size(); k++)
279 {
280 VideoBackendInfo& info = enabledBackends[k];
281 if (name == info.name)
282 {
283 info.priority = (int)(100000 + (names.size() - i) * 1000);
284 CV_LOG_DEBUG(NULL, "VIDEOIO: New backend priority: '" << name << "' => " << info.priority);
285 found = true;
286 hasChanges = true;
287 break;
288 }
289 }
290 if (!found)
291 {
292 CV_LOG_WARNING(NULL, "VIDEOIO: Can't prioritize unknown/unavailable backend: '" << name << "'");
293 }
294 }
295 return hasChanges;
296 }
297public:
298 std::string dumpBackends() const
299 {
300 std::ostringstream os;
301 for (size_t i = 0; i < enabledBackends.size(); i++)
302 {
303 if (i > 0) os << "; ";
304 const VideoBackendInfo& info = enabledBackends[i];
305 os << info.name << '(' << info.priority << ')';
306 }
307 return os.str();
308 }

403{
404 if (api == CAP_ANY)
405 return "CAP_ANY"; // special case, not a part of backends list
406 const int N = sizeof(builtin_backends)/sizeof(builtin_backends[0]);
