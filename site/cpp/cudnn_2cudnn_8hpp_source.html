194 template <class ...Sizes>
195 TensorDescriptor(Sizes ...sizes) {
196 static_assert(sizeof...(Sizes) <= CUDNN_DIM_MAX, "required rank exceeds maximum supported rank");
197 std::array<int, sizeof...(Sizes)> dims = { static_cast<int>(sizes)... };
198 constructor(std::begin(dims), std::end(dims));
199 }
200
201 ~TensorDescriptor() noexcept {
202 if (descriptor != nullptr) {
203 /* cudnnDestroyTensorDescriptor will not fail */
204 CUDA4DNN_CHECK_CUDNN(cudnnDestroyTensorDescriptor(descriptor));
205 }
206 }
207
208 TensorDescriptor& operator=(const TensorDescriptor&) = delete;
209 TensorDescriptor& operator=(TensorDescriptor&& other) noexcept {
210 descriptor = other.descriptor;
211 other.descriptor = nullptr;
212 return *this;
213 };
214
215 cudnnTensorDescriptor_t get() const noexcept { return descriptor; }
216
217 private:
218 template <class ForwardItr>
219 void constructor(ForwardItr start, ForwardItr end) {
220 CV_Assert(start != end);
221 CV_Assert(std::distance(start, end) <= CUDNN_DIM_MAX);
222
223 CUDA4DNN_CHECK_CUDNN(cudnnCreateTensorDescriptor(&descriptor));
224 try {
225 /* cuDNN documentation recommends using the 4d tensor API whenever possible
226 * hence, we create a 4d tensor descriptors for 3d tensor
227 */
228 const auto rank = std::distance(start, end);
229 if (rank <= 4) {
230 std::array<int, 4> dims;
231 std::fill(std::begin(dims), std::end(dims), 1);
232
233 /* suppose we have a 3d tensor, the first axis is the batch axis and
234 * the second axis is the channel axis (generally)
235 *
236 * cuDNN frequently assumes that the first axis is the batch axis and the
237 * second axis is the channel axis; hence, we copy the shape of a lower rank
238 * tensor to the beginning of `dims`
239 */
240 std::copy(start, end, std::begin(dims));
241
242 CUDA4DNN_CHECK_CUDNN(
243 cudnnSetTensor4dDescriptor(descriptor,
244 CUDNN_TENSOR_NCHW, detail::get_data_type<T>(),
245 dims[0], dims[1], dims[2], dims[3]
246 )
247 );
248 } else {
249 std::vector<int> stride(rank);
250 stride.back() = 1;
251 /* WHAT WE HAVE NOW:
252 * stride[-1] = 1
253 * stride[-2] = garbage
254 * stride[-3] = garbage
255 * stride[-4] = garbage
256 * ...
257 */
258
259 std::copy(start + 1, end, stride.begin());
260 /* WHAT WE HAVE NOW:
261 * stride[-1] = 1
262 * stride[-2] = dim[-1]
263 * stride[-3] = dim[-2]
264 * stride[-4] = dim[-3]
265 * ...
266 */
267
268 std::partial_sum(stride.rbegin(), stride.rend(), stride.rbegin(), std::multiplies<int>());
269 /* WHAT WE HAVE NOW:
270 * stride[-1] = 1
271 * stride[-2] = stride[-1] * dim[-1]
272 * stride[-3] = stride[-2] * dim[-2]
273 * stride[-4] = stride[-3] * dim[-3]
274 * ...
275 */
276
277 std::vector<int> dims(start, end);
278 CUDA4DNN_CHECK_CUDNN(
279 cudnnSetTensorNdDescriptor(descriptor,
280 detail::get_data_type<T>(), rank,
281 dims.data(), stride.data()
282 )
283 );
284 }
285 } catch (...) {
286 /* cudnnDestroyTensorDescriptor will not fail */
287 CUDA4DNN_CHECK_CUDNN(cudnnDestroyTensorDescriptor(descriptor));
288 throw;
289 }
290 }
291
292 cudnnTensorDescriptor_t descriptor;
293 };
