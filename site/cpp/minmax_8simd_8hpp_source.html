21minMaxIdx_( const T* src, const uchar* mask, WT* _minVal, WT* _maxVal,
22 size_t* _minIdx, size_t* _maxIdx, int len, size_t startIdx )
23{
24 WT minVal = *_minVal, maxVal = *_maxVal;
25 size_t minIdx = *_minIdx, maxIdx = *_maxIdx;
26 int i = 0;
27
28 if (minIdx == 0 || maxIdx == 0) {
29 if (mask) {
30 for (; i < len; i++) {
31 if (mask[i]) {
32 minVal = maxVal = (WT)src[i];
33 minIdx = maxIdx = startIdx + i;
34 i++;
35 break;
36 }
37 }
38 }
39 else if (len > 0) {
40 minVal = maxVal = (WT)src[0];
41 minIdx = maxIdx = startIdx;
42 i++;
43 }
44 }
45
46 if( !mask )
47 {
48 for( ; i < len; i++ )
49 {
50 WT val = (WT)src[i];
51 if( val < minVal )
52 {
53 minVal = val;
54 minIdx = startIdx + i;
55 }
56 if( val > maxVal )
57 {
58 maxVal = val;
59 maxIdx = startIdx + i;
60 }
61 }
62 }
63 else
64 {
65 for( ; i < len; i++ )
66 {
67 WT val = (WT)src[i];
68 uchar m = mask[i];
69 if( m && val < minVal )
70 {
71 minVal = val;
72 minIdx = startIdx + i;
73 }
74 if( m && val > maxVal )
75 {
76 maxVal = val;
77 maxIdx = startIdx + i;
78 }
79 }
80 }
81
82 *_minIdx = minIdx;
83 *_maxIdx = maxIdx;
84 *_minVal = minVal;
85 *_maxVal = maxVal;
86}

95static int minMaxInit(const uchar* mask, int len)
96{
97 int i = 0;
98 SIMD_ONLY(
99 int vlanes = VTraits<v_uint8>::vlanes();
100 v_uint8 v_zero = vx_setzero_u8();
101 for (; i < len; i += vlanes) {
102 if (i + vlanes > len) {
103 if (i == 0)
104 break;
105 i = len - vlanes;
106 }
107 v_uint8 mask_i = v_ne(vx_load(mask + i), v_zero);
108 if (v_check_any(mask_i))
109 return i + v_scan_forward(mask_i);
110 })
111 for (; i < len; i++) {
112 if (mask[i] != 0)
113 return i;
114 }
115 return -1;
116}

120#undef DEFINE_MINMAXIDX_SMALLINT_FUNC
121#define DEFINE_MINMAXIDX_SMALLINT_FUNC(funcname, suffix, usuffix, T, UT, VT, UVT, WT, BLOCK_SIZE, load_mask) \
122static void funcname(const T* src, const uchar* mask, WT* _minVal, WT* _maxVal, \
123 size_t* _minIdx, size_t* _maxIdx, int len, size_t startIdx) \
124{ \
125 T minVal = T(*_minVal), maxVal = T(*_maxVal); \
126 size_t minIdx = *_minIdx, maxIdx = *_maxIdx; \
127 int i = 0; \
128 /* initialize minVal/maxVal/minIdx/maxIdx to the proper values in the beginning */ \
129 if (minIdx == 0) { \
130 if (mask) { \
131 i = minMaxInit(mask, len); \
132 if (i < 0) \
133 return; \
134 } \
135 minVal = maxVal = src[i]; \
136 minIdx = maxIdx = startIdx + i; \
137 i++; \
138 } \
139 SIMD_ONLY( \
140 const int vlanes = VTraits<VT>::vlanes(); \
141 const int block_size0 = BLOCK_SIZE - vlanes; \
142 if (len-i >= vlanes && block_size0 > 0 && block_size0 % vlanes == 0) { \
143 UT idxbuf[VTraits<UVT>::max_nlanes]; \
144 for (int j = 0; j < vlanes; j++) \
145 idxbuf[j] = (UT)j; \
146 UVT v_idx0 = vx_load(idxbuf); \
147 UVT v_idx_delta = vx_setall_##usuffix((UT)vlanes); \
148 UVT v_invalid_idx = vx_setall_##usuffix((UT)-1); \
149 VT v_minval = vx_setall_##suffix(minVal); \
150 VT v_maxval = vx_setall_##suffix(maxVal); \
151 int block_size = block_size0; \
152 /* process data by blocks: */ \
153 /* - for u8/s8 data each block contains up to 256-vlanes elements */ \
154 /* - for u16/s16 data each block contains up to 65536-vlanes elements */ \
155 /* inside each block we can store the relative (local) index (v_locidx) */ \
156 /* in a compact way: 8 bits per lane for u8/s8 data, */ \
157 /* 16 bits per lane for u16/s16 data */ \
158 /* 0b111...111 is "invalid index", meaning that this */ \
159 /* particular lane has not been updated. */ \
160 /* after each block we update minVal, maxVal, minIdx and maxIdx */ \
161 for (; i <= len - vlanes; i += block_size) { \
162 block_size = std::min(block_size, (len - i) & -vlanes); \
163 UVT v_locidx = v_idx0; \
164 UVT v_minidx = v_invalid_idx; \
165 UVT v_maxidx = v_invalid_idx; \
166 if (!mask) { \
167 for (int j = 0; j < block_size; j += vlanes) { \
168 VT data = vx_load(src + i + j); \
169 UVT lt_min = v_reinterpret_as_##usuffix(v_lt(data, v_minval)); \
170 UVT gt_max = v_reinterpret_as_##usuffix(v_gt(data, v_maxval)); \
171 v_minidx = v_select(lt_min, v_locidx, v_minidx); \
172 v_maxidx = v_select(gt_max, v_locidx, v_maxidx); \
173 v_minval = v_min(v_minval, data); \
174 v_maxval = v_max(v_maxval, data); \
175 v_locidx = v_add(v_locidx, v_idx_delta); \
176 } \
177 } else { \
178 UVT v_zero = vx_setzero_##usuffix(); \
179 for (int j = 0; j < block_size; j += vlanes) { \
180 VT data = vx_load(src + i + j); \
181 UVT msk = v_ne(load_mask(mask + i + j), v_zero); \
182 UVT lt_min = v_reinterpret_as_##usuffix(v_lt(data, v_minval)); \
183 UVT gt_max = v_reinterpret_as_##usuffix(v_gt(data, v_maxval)); \
184 lt_min = v_and(lt_min, msk); \
185 gt_max = v_and(gt_max, msk); \
186 v_minidx = v_select(lt_min, v_locidx, v_minidx); \
187 v_maxidx = v_select(gt_max, v_locidx, v_maxidx); \
188 VT lt_min_data = v_reinterpret_as_##suffix(lt_min); \
189 VT gt_max_data = v_reinterpret_as_##suffix(gt_max); \
190 v_minval = v_select(lt_min_data, data, v_minval); \
191 v_maxval = v_select(gt_max_data, data, v_maxval); \
192 v_locidx = v_add(v_locidx, v_idx_delta); \
193 } \
194 } \
195 /* for both minimum and maximum we check whether global extremum */ \
196 /* and its index need to be updated. If yes, we compute */ \
197 /* the smallest index within the block where the new global */ \
198 /* extremum value occurs */ \
199 UVT idxmask = v_ne(v_minidx, v_invalid_idx); \
200 if (v_check_any(idxmask)) { \
201 minVal = (T)v_reduce_min(v_minval); \
202 VT invmask = v_ne(v_minval, vx_setall_##suffix(minVal)); \
203 v_minidx = v_or(v_minidx, v_reinterpret_as_##usuffix(invmask)); \
204 minIdx = startIdx + i + v_reduce_min(v_minidx); \
205 v_minval = vx_setall_##suffix(minVal); \
206 } \
207 idxmask = v_ne(v_maxidx, v_invalid_idx); \
208 if (v_check_any(idxmask)) { \
209 maxVal = (T)v_reduce_max(v_maxval); \
210 VT invmask = v_ne(v_maxval, vx_setall_##suffix(maxVal)); \
211 v_maxidx = v_or(v_maxidx, v_reinterpret_as_##usuffix(invmask)); \
212 maxIdx = startIdx + i + v_reduce_min(v_maxidx); \
213 v_maxval = vx_setall_##suffix(maxVal); \
214 } \
215 } \
216 }) \
217 *_minVal = (WT)minVal; \
218 *_maxVal = (WT)maxVal; \
219 *_minIdx = minIdx; \
220 *_maxIdx = maxIdx; \
221 /* [TODO]: unlike sum, countNonZero and other reduce operations, */ \
222 /* in the case of minMaxIdx we can process the tail using */ \
223 /* vector overlapping technique (as in arithmetic operations) */ \
224 if (i < len) { \
225 src += i; \
226 if (mask) mask += i; \
227 startIdx += i; \
228 len -= i; \
229 minMaxIdx_(src, mask, _minVal, _maxVal, _minIdx, _maxIdx, len, startIdx); \
230 } \
231}
