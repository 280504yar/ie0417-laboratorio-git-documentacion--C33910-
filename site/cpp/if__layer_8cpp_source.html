52 int branch(InputArray arr) const CV_OVERRIDE
53 {
54 Mat buf, *inp;
55 if (arr.kind() == _InputArray::MAT) {
56 inp = (Mat*)arr.getObj();
57 } else {
58 buf = arr.getMat();
59 inp = &buf;
60 }
61 CV_Assert(inp->total() == 1u);
62 bool flag;
63 switch (inp->depth())
64 {
65 case CV_8U: case CV_8S: case CV_Bool:
66 flag = *inp->ptr<char>() != 0; break;
67 case CV_16U: case CV_16S:
68 flag = *inp->ptr<short>() != 0; break;
69 case CV_16F:
70 flag = *inp->ptr<hfloat>() != 0; break;
71 case CV_16BF:
72 flag = *inp->ptr<hfloat>() != 0; break;
73 case CV_32U: case CV_32S:
74 flag = *inp->ptr<int>() != 0; break;
75 case CV_32F:
76 flag = *inp->ptr<float>() != 0; break;
77 case CV_64U: case CV_64S:
78 flag = *inp->ptr<long long>() != 0; break;
79 case CV_64F:
80 flag = *inp->ptr<double>() != 0; break;
81 default:
82 CV_Error_(Error::StsBadArg,
83 ("If-layer condition: unsupported tensor type %s",
84 typeToString(inp->type()).c_str()));
85 }
86 return (int)!flag;
87 }
