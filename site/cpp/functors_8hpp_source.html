116template <>
117struct MishFunctor<float> {
118 struct Params {
119 CUDA4DNN_HOST_DEVICE Params() { }
120 };
121
122 CUDA4DNN_DEVICE MishFunctor() { }
123 CUDA4DNN_DEVICE MishFunctor(const Params& params) { }
124
125 CUDA4DNN_DEVICE float operator()(float value) {
126 // f(x) = x * tanh(log1pexp(x));
127 using csl::device::fast_divide;
128 using csl::device::fast_exp;
129
130 auto e = fast_exp(value);
131 auto n = e * e + 2 * e;
132 if (value <= -0.6f)
133 return value * fast_divide(n, n + 2);
134 return value - 2 * fast_divide(value, n + 2);
135 }
