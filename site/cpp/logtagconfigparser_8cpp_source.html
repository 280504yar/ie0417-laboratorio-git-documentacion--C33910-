67void LogTagConfigParser::segmentTokens()
68{
69 const size_t len = m_input.length();
70 std::vector<std::pair<size_t, size_t>> startStops;
71 bool wasSeparator = true;
72 for (size_t pos = 0u; pos < len; ++pos)
73 {
74 char c = m_input[pos];
75 bool isSeparator = (c == ' ' || c == '\t' || c == ';');
76 if (!isSeparator)
77 {
78 if (wasSeparator)
79 {
80 startStops.emplace_back(pos, pos + 1u);
81 }
82 else
83 {
84 startStops.back().second = pos + 1u;
85 }
86 }
87 wasSeparator = isSeparator;
88 }
89 for (const auto& startStop : startStops)
90 {
91 const auto s = m_input.substr(startStop.first, startStop.second - startStop.first);
92 parseNameAndLevel(s);
93 }
94}

96void LogTagConfigParser::parseNameAndLevel(const std::string& s)
97{
98 const size_t npos = std::string::npos;
99 const size_t len = s.length();
100 size_t colonIdx = s.find_first_of(":=");
101 if (colonIdx == npos)
102 {
103 // See if the whole string is a log level
104 auto parsedLevel = parseLogLevel(s);
105 if (parsedLevel.second)
106 {
107 // If it is, assume the log level is for global
108 parseWildcard("", parsedLevel.first);
109 return;
110 }
111 else
112 {
113 // not sure what to do.
114 m_malformed.push_back(s);
115 return;
116 }
117 }
118 if (colonIdx == 0u || colonIdx + 1u == len)
119 {
120 // malformed (colon or equal sign at beginning or end), cannot do anything
121 m_malformed.push_back(s);
122 return;
123 }
124 size_t colonIdx2 = s.find_first_of(":=", colonIdx + 1u);
125 if (colonIdx2 != npos)
126 {
127 // malformed (more than one colon or equal sign), cannot do anything
128 m_malformed.push_back(s);
129 return;
130 }
131 auto parsedLevel = parseLogLevel(s.substr(colonIdx + 1u));
132 if (parsedLevel.second)
133 {
134 parseWildcard(s.substr(0u, colonIdx), parsedLevel.first);
135 return;
136 }
137 else
138 {
139 // Cannot recognize the right side of the colon or equal sign.
140 // Not sure what to do.
141 m_malformed.push_back(s);
142 return;
143 }
144}

146void LogTagConfigParser::parseWildcard(const std::string& name, LogLevel level)
147{
148 const size_t npos = std::string::npos;
149 const size_t len = name.length();
150 if (len == 0u)
151 {
152 m_parsedGlobal.level = level;
153 return;
154 }
155 const bool hasPrefixWildcard = (name[0u] == '*');
156 if (hasPrefixWildcard && len == 1u)
157 {
158 m_parsedGlobal.level = level;
159 return;
160 }
161 const size_t firstNonWildcard = name.find_first_not_of("*.");
162 if (hasPrefixWildcard && firstNonWildcard == npos)
163 {
164 m_parsedGlobal.level = level;
165 return;
166 }
167 const bool hasSuffixWildcard = (name[len - 1u] == '*');
168 const size_t lastNonWildcard = name.find_last_not_of("*.");
169 std::string trimmedNamePart = name.substr(firstNonWildcard, lastNonWildcard - firstNonWildcard + 1u);
170 // The case of a single asterisk has been handled above;
171 // here we only handle the explicit use of "global" in the log config string.
172 const bool isGlobal = (trimmedNamePart == "global");
173 if (isGlobal)
174 {
175 m_parsedGlobal.level = level;
176 return;
177 }
178 LogTagConfig result(trimmedNamePart, level, false, hasPrefixWildcard, hasSuffixWildcard);
179 if (hasPrefixWildcard)
180 {
181 m_parsedAnyPart.emplace_back(std::move(result));
182 }
183 else if (hasSuffixWildcard)
184 {
185 m_parsedFirstPart.emplace_back(std::move(result));
186 }
187 else
188 {
189 m_parsedFullName.emplace_back(std::move(result));
190 }
191}

193std::pair<LogLevel, bool> LogTagConfigParser::parseLogLevel(const std::string& s)
194{
195 const auto falseDontCare = std::make_pair(LOG_LEVEL_VERBOSE, false);
196 const auto make_parsed_result = [](LogLevel lev) -> std::pair<LogLevel, bool>
197 {
198 return std::make_pair(lev, true);
199 };
200 const size_t len = s.length();
201 if (len >= 1u)
202 {
203 const char c = (char)std::toupper(s[0]);
204 switch (c)
205 {
206 case '0':
207 if (len == 1u)
208 {
209 return make_parsed_result(LOG_LEVEL_SILENT);
210 }
211 break;
212 case 'D':
213 if (len == 1u ||
214 (len == 5u && cv::toUpperCase(s) == "DEBUG"))
215 {
216 return make_parsed_result(LOG_LEVEL_DEBUG);
217 }
218 if ((len == 7u && cv::toUpperCase(s) == "DISABLE") ||
219 (len == 8u && cv::toUpperCase(s) == "DISABLED"))
220 {
221 return make_parsed_result(LOG_LEVEL_SILENT);
222 }
223 break;
224 case 'E':
225 if (len == 1u ||
226 (len == 5u && cv::toUpperCase(s) == "ERROR"))
227 {
228 return make_parsed_result(LOG_LEVEL_ERROR);
229 }
230 break;
231 case 'F':
232 if (len == 1u ||
233 (len == 5u && cv::toUpperCase(s) == "FATAL"))
234 {
235 return make_parsed_result(LOG_LEVEL_FATAL);
236 }
237 break;
238 case 'I':
239 if (len == 1u ||
240 (len == 4u && cv::toUpperCase(s) == "INFO"))
241 {
242 return make_parsed_result(LOG_LEVEL_INFO);
243 }
244 break;
245 case 'O':
246 if (len == 3u && cv::toUpperCase(s) == "OFF")
247 {
248 return make_parsed_result(LOG_LEVEL_SILENT);
249 }
250 break;
251 case 'S':
252 if (len == 1u ||
253 (len == 6u && cv::toUpperCase(s) == "SILENT"))
254 {
255 return make_parsed_result(LOG_LEVEL_SILENT);
256 }
257 break;
258 case 'V':
259 if (len == 1u ||
260 (len == 7u && cv::toUpperCase(s) == "VERBOSE"))
261 {
262 return make_parsed_result(LOG_LEVEL_VERBOSE);
263 }
264 break;
265 case 'W':
266 if (len == 1u ||
267 (len == 4u && cv::toUpperCase(s) == "WARN") ||
268 (len == 7u && cv::toUpperCase(s) == "WARNING") ||
269 (len == 8u && cv::toUpperCase(s) == "WARNINGS"))
270 {
271 return make_parsed_result(LOG_LEVEL_WARNING);
272 }
273 break;
274 default:
275 break;
276 }
277 // fall through
278 }
279 return falseDontCare;
280}
