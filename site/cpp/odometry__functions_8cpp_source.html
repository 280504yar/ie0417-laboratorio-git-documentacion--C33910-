17static void randomSubsetOfMask(InputOutputArray _mask, float part)
18{
19 const int minPointsCount = 1000; // minimum point count (we can process them fast)
20 const int nonzeros = countNonZero(_mask);
21 const int needCount = std::max(minPointsCount, int(_mask.total() * part));
22 if (needCount < nonzeros)
23 {
24 RNG rng;
25 Mat mask = _mask.getMat();
26 Mat subset(mask.size(), CV_8UC1, Scalar(0));
27
28 int subsetSize = 0;
29 while (subsetSize < needCount)
30 {
31 int y = rng(mask.rows);
32 int x = rng(mask.cols);
33 if (mask.at<uchar>(y, x))
34 {
35 subset.at<uchar>(y, x) = 255;
36 mask.at<uchar>(y, x) = 0;
37 subsetSize++;
38 }
39 }
40 _mask.assign(subset);
41 }
42}

45static UMat prepareScaledDepth(OdometryFrame& frame)
46{
47 UMat depth;
48 frame.getDepth(depth);
49 CV_Assert(!depth.empty());
50 bool isFloat = (depth.type() == CV_32FC1 || depth.type() == CV_64FC1);
51
52 // Odometry works well with depth values in range [0, 10)
53 // If it's bigger, let's scale it down by 5000, a typical depth factor
54 UMat depthFlt;
55 if (isFloat)
56 {
57 depth.convertTo(depthFlt, CV_32FC1);
58 patchNaNs(depthFlt, 0);
59 }
60
61 double maxv;
62 cv::minMaxLoc(isFloat ? depthFlt : depth, nullptr, &maxv);
63
64 if (!isFloat || maxv > 10)
65 {
66 depth.convertTo(depthFlt, CV_32FC1, maxv > 10 ? (1.f / 5000.f) : 1.f);
67 patchNaNs(depthFlt, 0);
68 }
69
70 frame.impl->scaledDepth = depthFlt;
71
72 return depthFlt;
73}

85static void preparePyramidMask(UMat mask, const std::vector<UMat> pyramidDepth, int nLevels, float minDepth, float maxDepth,
86 std::vector<UMat>& pyramidMask)
87{
88 minDepth = std::max(0.f, minDepth);
89
90 buildPyramid(mask, pyramidMask, nLevels - 1);
91
92 for (int i = 0; i < nLevels; i++)
93 {
94 UMat maski = pyramidMask[i];
95 threshold(maski, maski, 254, 255, THRESH_TOZERO);
96 const UMat depthi = pyramidDepth[i];
97
98 UMat gtmin, ltmax, tmpMask;
99 cv::compare(depthi, Scalar(minDepth), gtmin, CMP_GT);
100 cv::compare(depthi, Scalar(maxDepth), ltmax, CMP_LT);
101 cv::bitwise_and(gtmin, ltmax, tmpMask);
102 cv::bitwise_and(maski, tmpMask, maski);
103 }

153static void preparePyramidTexturedMask(const std::vector<UMat>& pyramid_dI_dx, const std::vector<UMat>& pyramid_dI_dy,
154 std::vector<float> minGradMagnitudes, const std::vector<UMat>& pyramidMask, double maxPointsPart,
155 std::vector<UMat>& pyramidTexturedMask, double sobelScale)
156{
157 int nLevels = (int)pyramid_dI_dx.size();
158
159 const float sobelScale2_inv = (float)(1. / (sobelScale * sobelScale));
160 pyramidTexturedMask.resize(nLevels, UMat());
161 for (int i = 0; i < nLevels; i++)
162 {
163 const float minScaledGradMagnitude2 = minGradMagnitudes[i] * minGradMagnitudes[i] * sobelScale2_inv;
164 const Mat dIdx = pyramid_dI_dx[i].getMat(ACCESS_READ);
165 const Mat dIdy = pyramid_dI_dy[i].getMat(ACCESS_READ);
166
167 Mat texturedMask(dIdx.size(), CV_8UC1, Scalar(0));
168
169 for (int y = 0; y < dIdx.rows; y++)
170 {
171 const short *dIdx_row = dIdx.ptr<short>(y);
172 const short *dIdy_row = dIdy.ptr<short>(y);
173 uchar *texturedMask_row = texturedMask.ptr<uchar>(y);
174 for (int x = 0; x < dIdx.cols; x++)
175 {
176 float magnitude2 = static_cast<float>(dIdx_row[x] * dIdx_row[x] + dIdy_row[x] * dIdy_row[x]);
177 if (magnitude2 >= minScaledGradMagnitude2)
178 texturedMask_row[x] = 255;
179 }
180 }
181 Mat texMask = texturedMask & pyramidMask[i].getMat(ACCESS_READ);
182 randomSubsetOfMask(texMask, (float)maxPointsPart);
183 texMask.copyTo(pyramidTexturedMask[i]);
184 }
185}

188static void preparePyramidNormals(const UMat &normals, const std::vector<UMat> &pyramidDepth, std::vector<UMat> &pyramidNormals)
189{
190 int nLevels = (int)pyramidDepth.size();
191
192 buildPyramid(normals, pyramidNormals, nLevels - 1);
193 // renormalize normals
194 for (int i = 1; i < nLevels; i++)
195 {
196 Mat currNormals = pyramidNormals[i].getMat(ACCESS_RW);
197 CV_Assert(currNormals.type() == CV_32FC4);
198 for (int y = 0; y < currNormals.rows; y++)
199 {
200 Vec4f *normals_row = currNormals.ptr<Vec4f>(y);
201 for (int x = 0; x < currNormals.cols; x++)
202 {
203 Vec4f n4 = normals_row[x];
204 Point3f n(n4[0], n4[1], n4[2]);
205 double nrm = norm(n);
206 n *= 1.f / nrm;
207 normals_row[x] = Vec4f(n.x, n.y, n.z, 0);
208 }
209 }
210 }

214static void preparePyramidNormalsMask(const std::vector<UMat> &pyramidNormals, const std::vector<UMat> &pyramidMask, double maxPointsPart,
215 std::vector<UMat> &pyramidNormalsMask)
216{
217 int nLevels = (int)pyramidNormals.size();
218 pyramidNormalsMask.resize(nLevels, UMat());
219 for (int i = 0; i < nLevels; i++)
220 {
221 Mat pyrMask = pyramidMask[i].getMat(ACCESS_READ);
222
223 const Mat normals = pyramidNormals[i].getMat(ACCESS_READ);
224 Mat_<uchar> normalsMask(pyrMask.size(), (uchar)255);
225 for (int y = 0; y < normalsMask.rows; y++)
226 {
227 const Vec4f *normals_row = normals.ptr<Vec4f>(y);
228 uchar *normalsMask_row = normalsMask.ptr<uchar>(y);
229 for (int x = 0; x < normalsMask.cols; x++)
230 {
231 Vec4f n = normals_row[x];
232 if (!(std::isfinite(n[0]) && std::isfinite(n[1]) && std::isfinite(n[2])))
233 {
234 normalsMask_row[x] = 0;
235 }
236 }
237 }
238 cv::bitwise_and(pyrMask, normalsMask, normalsMask);
239
240 randomSubsetOfMask(normalsMask, (float)maxPointsPart);
241 normalsMask.copyTo(pyramidNormalsMask[i]);
242 }

246static void prepareRGBFrameBase(OdometryFrame& frame, OdometrySettings settings)
247{
248 UMat grayImage;
249 frame.getGrayImage(grayImage);
250 if (grayImage.empty())
251 {
252 UMat image;
253 frame.getImage(image);
254 CV_Assert(!image.empty() && image.depth() == CV_8U);
255
256 int ch = image.channels();
257 if (ch == 3 || ch == 4)
258 cvtColor(image, grayImage, ch == 3 ? COLOR_BGR2GRAY : COLOR_BGRA2GRAY, 1);
259 else if (ch == 1)
260 grayImage = image;
261 else
262 CV_Error(Error::StsBadArg, "Image should have 3 or 4 channels (RGB) or 1 channel (grayscale)");
263 grayImage.convertTo(grayImage, CV_8U);
264 frame.impl->imageGray = grayImage;
265 }
266
267 //TODO: don't use scaled when scale bug is fixed
268 UMat scaledDepth;
269 frame.getProcessedDepth(scaledDepth);
270 if (scaledDepth.empty())
271 {
272 scaledDepth = prepareScaledDepth(frame);
273 CV_Assert(scaledDepth.size() == grayImage.size());
274 }
275
276 UMat depthMask;
277 // ignore small, negative, Inf, NaN values
278 cv::compare(scaledDepth, Scalar(FLT_EPSILON), depthMask, CMP_GT);
279
280 UMat mask;
281 frame.getMask(mask);
282 if (mask.empty())
283 {
284 frame.impl->mask = depthMask;
285 }
286 else
287 {
288 CV_Assert(mask.type() == CV_8UC1 || mask.type() == CV_8SC1 || mask.type() == CV_BoolC1);
289 CV_Assert(mask.size() == grayImage.size());
290 cv::bitwise_and(mask, depthMask, frame.impl->mask);
291 }
292 frame.getMask(mask);
293
294 std::vector<int> iterCounts;
295 settings.getIterCounts(iterCounts);
296
297 int maxLevel = (int)iterCounts.size() - 1;
298 std::vector<UMat>& ipyramids = frame.impl->pyramids[OdometryFramePyramidType::PYR_IMAGE];
299 if (ipyramids.empty())
300 buildPyramid(grayImage, ipyramids, maxLevel);
301
302 std::vector<UMat>& dpyramids = frame.impl->pyramids[OdometryFramePyramidType::PYR_DEPTH];
303 if (dpyramids.empty())
304 preparePyramidDepth(scaledDepth, dpyramids, maxLevel);
305
306 std::vector<UMat>& mpyramids = frame.impl->pyramids[OdometryFramePyramidType::PYR_MASK];
307 if (mpyramids.empty())
308 preparePyramidMask(mask, dpyramids, maxLevel + 1, settings.getMinDepth(), settings.getMaxDepth(), mpyramids);
309}

412{
413 Matx33f cameraMatrix;
414 settings.getCameraMatrix(cameraMatrix);
415
416 UMat scaledDepth, mask, normals;
417 frame.getProcessedDepth(scaledDepth);

427 normalsComputer->getRows() != scaledDepth.rows ||
428 normalsComputer->getCols() != scaledDepth.cols ||

437 cameraMatrix,
438 normalWinSize,
439 diffThreshold,
440 normalMethod);
441 }

443 normalsComputer->apply(c0, normals);
444 frame.impl->normals = normals;
445 }
446 CV_Assert(normals.type() == CV_32FC4);
447

495bool RGBDICPOdometryImpl(OutputArray _Rt, const Mat& initRt,
496 const OdometryFrame& srcFrame,
497 const OdometryFrame& dstFrame,
498 const Matx33f& cameraMatrix,
499 float maxDepthDiff, float angleThreshold, const std::vector<int>& iterCounts,
500 double maxTranslation, double maxRotation, double sobelScale,
501 OdometryType method, OdometryTransformType transformType, OdometryAlgoType algtype)
502{
503 if (srcFrame.getPyramidLevels() != (int)(iterCounts.size()))
504 CV_Error(Error::StsBadArg, "srcFrame has incorrect number of pyramid levels. Did you forget to call prepareFrame()?");
505 if (dstFrame.getPyramidLevels() != (int)(iterCounts.size()))
506 CV_Error(Error::StsBadArg, "dstFrame has incorrect number of pyramid levels. Did you forget to call prepareFrame()?");
507
508 int transformDim = getTransformDim(transformType);
509
510 const int minOverdetermScale = 20;
511 const int minCorrespsCount = minOverdetermScale * transformDim;
512
513 std::vector<Matx33f> pyramidCameraMatrix;
514 buildPyramidCameraMatrix(cameraMatrix, (int)iterCounts.size(), pyramidCameraMatrix);
515
516 Mat resultRt = initRt.empty() ? Mat::eye(4,4,CV_64FC1) : initRt.clone();
517 Mat currRt, ksi;
518 Affine3f transform = Affine3f::Identity();
519
520 bool isOk = false;
521 for(int level = (int)iterCounts.size() - 1; level >= 0; level--)
522 {
523 const Matx33f& levelCameraMatrix = pyramidCameraMatrix[level];
524 const Mat srcLevelDepth, dstLevelDepth;
525 const Mat srcLevelImage, dstLevelImage;
526 srcFrame.getPyramidAt(srcLevelDepth, OdometryFramePyramidType::PYR_DEPTH, level);
527 dstFrame.getPyramidAt(dstLevelDepth, OdometryFramePyramidType::PYR_DEPTH, level);
528
529 if (method != OdometryType::DEPTH) // RGB(D)
530 {
531 srcFrame.getPyramidAt(srcLevelImage, OdometryFramePyramidType::PYR_IMAGE, level);
532 dstFrame.getPyramidAt(dstLevelImage, OdometryFramePyramidType::PYR_IMAGE, level);
533 }
534
535 const double fx = levelCameraMatrix(0, 0);
536 const double fy = levelCameraMatrix(1, 1);
537 const double determinantThreshold = 1e-6;
538
539 Mat AtA_rgbd, AtB_rgbd, AtA_icp, AtB_icp;
540
541 // Run transformation search on current level iteratively.
542 for(int iter = 0; iter < iterCounts[level]; iter ++)
543 {
544 Mat resultRt_inv = resultRt.inv(DECOMP_SVD);
545 Mat corresps_rgbd, corresps_icp, diffs_rgbd;
546 Mat dummy;
547 double sigma_rgbd = 0, dummyFloat = 0;
548
549 const Mat pyramidMask;
550 srcFrame.getPyramidAt(pyramidMask, OdometryFramePyramidType::PYR_MASK, level);
551
552 if(method != OdometryType::DEPTH) // RGB(D)
553 {
554 const Mat pyramidTexturedMask;
555 dstFrame.getPyramidAt(pyramidTexturedMask, OdometryFramePyramidType::PYR_TEXMASK, level);
556 computeCorresps(levelCameraMatrix, resultRt,
557 srcLevelImage, srcLevelDepth, pyramidMask,
558 dstLevelImage, dstLevelDepth, pyramidTexturedMask, maxDepthDiff,
559 corresps_rgbd, diffs_rgbd, sigma_rgbd, OdometryType::RGB);
560 }
561
562 if(method != OdometryType::RGB) // ICP, RGBD
563 {
564 if (algtype == OdometryAlgoType::COMMON)
565 {
566 const Mat pyramidNormalsMask;
567 dstFrame.getPyramidAt(pyramidNormalsMask, OdometryFramePyramidType::PYR_NORMMASK, level);
568 computeCorresps(levelCameraMatrix, resultRt,
569 Mat(), srcLevelDepth, pyramidMask,
570 Mat(), dstLevelDepth, pyramidNormalsMask, maxDepthDiff,
571 corresps_icp, dummy, dummyFloat, OdometryType::DEPTH);
572 }
573 }
574
575 if(corresps_rgbd.rows < minCorrespsCount && corresps_icp.rows < minCorrespsCount && algtype != OdometryAlgoType::FAST)
576 break;
577
578 const UMat srcPyrCloud;
579 srcFrame.getPyramidAt(srcPyrCloud, OdometryFramePyramidType::PYR_CLOUD, level);
580
581
582 Mat AtA(transformDim, transformDim, CV_64FC1, Scalar(0)), AtB(transformDim, 1, CV_64FC1, Scalar(0));
583 if(corresps_rgbd.rows >= minCorrespsCount)
584 {
585 const Mat srcPyrImage, dstPyrImage, dstPyrIdx, dstPyrIdy;
586 srcFrame.getPyramidAt(srcPyrImage, OdometryFramePyramidType::PYR_IMAGE, level);
587 dstFrame.getPyramidAt(dstPyrImage, OdometryFramePyramidType::PYR_IMAGE, level);
588 dstFrame.getPyramidAt(dstPyrIdx, OdometryFramePyramidType::PYR_DIX, level);
589 dstFrame.getPyramidAt(dstPyrIdy, OdometryFramePyramidType::PYR_DIY, level);
590 calcRgbdLsmMatrices(srcPyrCloud.getMat(ACCESS_READ), resultRt, dstPyrIdx, dstPyrIdy,
591 corresps_rgbd, diffs_rgbd, sigma_rgbd, fx, fy, sobelScale,
592 AtA_rgbd, AtB_rgbd, transformType);
593 AtA += AtA_rgbd;
594 AtB += AtB_rgbd;
595 }
596 if(corresps_icp.rows >= minCorrespsCount || algtype == OdometryAlgoType::FAST)
597 {
598 if (algtype == OdometryAlgoType::COMMON)
599 {
600 const Mat dstPyrCloud, dstPyrNormals;
601 dstFrame.getPyramidAt(dstPyrCloud, OdometryFramePyramidType::PYR_CLOUD, level);
602 dstFrame.getPyramidAt(dstPyrNormals, OdometryFramePyramidType::PYR_NORM, level);
603 calcICPLsmMatrices(srcPyrCloud.getMat(ACCESS_READ), resultRt, dstPyrCloud, dstPyrNormals,
604 corresps_icp, AtA_icp, AtB_icp, transformType);
605 }
606 else
607 {
608 const UMat dstPyrCloud, dstPyrNormals, srcPyrNormals;
609 dstFrame.getPyramidAt(dstPyrCloud, OdometryFramePyramidType::PYR_CLOUD, level);
610 dstFrame.getPyramidAt(dstPyrNormals, OdometryFramePyramidType::PYR_NORM, level);
611 srcFrame.getPyramidAt(srcPyrNormals, OdometryFramePyramidType::PYR_NORM, level);
612 cv::Matx66f A;
613 cv::Vec6f b;
614 calcICPLsmMatricesFast(cameraMatrix, dstPyrCloud, dstPyrNormals, srcPyrCloud, srcPyrNormals, transform, level, maxDepthDiff, angleThreshold, A, b);
615 AtA_icp = Mat(A);
616 AtB_icp = Mat(b);
617 }
618
619 AtA += AtA_icp;
620 AtB += AtB_icp;
621 }
622
623 bool solutionExist = solveSystem(AtA, AtB, determinantThreshold, ksi);
624
625 if (!solutionExist)
626 {
627 break;
628 }
629
630 Mat tmp61(6, 1, CV_64FC1, Scalar(0));
631 if(transformType == OdometryTransformType::ROTATION)
632 {
633 ksi.copyTo(tmp61.rowRange(0,3));
634 ksi = tmp61;
635 }
636 else if(transformType == OdometryTransformType::TRANSLATION)
637 {
638 ksi.copyTo(tmp61.rowRange(3,6));
639 ksi = tmp61;
640 }
641
642 computeProjectiveMatrix(ksi, currRt);
643 resultRt = currRt * resultRt;
644
645 //TODO: fixit, transform is used for Fast ICP only
646 Vec6f x(ksi);
647 Affine3f tinc(Vec3f(x.val), Vec3f(x.val + 3));
648 transform = tinc * transform;
649
650 isOk = true;
651 }
652
653 }
654
655 _Rt.create(resultRt.size(), resultRt.type());
656 Mat Rt = _Rt.getMat();
657 resultRt.copyTo(Rt);
658 if(isOk)
659 {
660 Mat deltaRt;
661 if(initRt.empty())
662 deltaRt = resultRt;
663 else
664 deltaRt = resultRt * initRt.inv(DECOMP_SVD);
665
666 isOk = testDeltaTransformation(deltaRt, maxTranslation, maxRotation);
667 }
668
669 return isOk;
670}

672// Rotate dst by RtInv to get corresponding src pixels
673// In RGB case compute sigma and diffs too
674void computeCorresps(const Matx33f& _K, const Mat& rt,
675 const Mat& imageSrc, const Mat& depthSrc, const Mat& validMaskSrc,
676 const Mat& imageDst, const Mat& depthDst, const Mat& selectMaskDst, float maxDepthDiff,
677 Mat& _corresps, Mat& _diffs, double& _sigma, OdometryType method)
678{
679 Mat mrtInv = rt.inv(DECOMP_SVD);
680 Matx44d rtInv = mrtInv;
681
682 Mat corresps(depthDst.size(), CV_16SC2, Scalar::all(-1));
683 Mat diffs;
684 if (method == OdometryType::RGB)
685 diffs = Mat(depthDst.size(), CV_32F, Scalar::all(-1));
686
687 // src_2d = K * src_3d, src_3d = K_inv * [src_2d | z]
688 //
689
690 Matx33d K(_K);
691 Matx33d K_inv = K.inv(DECOMP_SVD);
692 Rect r(0, 0, depthDst.cols, depthDst.rows);
693 Matx31d tinv = rtInv.get_minor<3, 1>(0, 3);
694 Matx31d ktinvm = K * tinv;
695 //const double* Kt_ptr = Kt.ptr<const double>();
696 Point3d ktinv(ktinvm(0, 0), ktinvm(1, 0), ktinvm(2, 0));
697
698 AutoBuffer<float> buf(3 * (depthDst.cols + depthDst.rows));
699 float* KRK_inv0_u1 = buf.data();
700 float* KRK_inv1_v1_plus_KRK_inv2 = buf.data() + depthDst.cols;
701 float* KRK_inv3_u1 = buf.data() + depthDst.cols + depthDst.rows;
702 float* KRK_inv4_v1_plus_KRK_inv5 = buf.data() + depthDst.cols * 2 + depthDst.rows;
703 float* KRK_inv6_u1 = buf.data() + depthDst.cols * 2 + depthDst.rows * 2;
704 float* KRK_inv7_v1_plus_KRK_inv8 = buf.data() + depthDst.cols * 3 + depthDst.rows * 2;
705 {
706 Matx33d rinv = rtInv.get_minor<3, 3>(0, 0);
707
708 Matx33d kriki = K * rinv * K_inv;
709 for (int udst = 0; udst < depthDst.cols; udst++)
710 {
711 KRK_inv0_u1[udst] = (float)(kriki(0, 0) * udst);
712 KRK_inv3_u1[udst] = (float)(kriki(1, 0) * udst);
713 KRK_inv6_u1[udst] = (float)(kriki(2, 0) * udst);
714 }
715
716 for (int vdst = 0; vdst < depthDst.rows; vdst++)
717 {
718 KRK_inv1_v1_plus_KRK_inv2[vdst] = (float)(kriki(0, 1) * vdst + kriki(0, 2));
719 KRK_inv4_v1_plus_KRK_inv5[vdst] = (float)(kriki(1, 1) * vdst + kriki(1, 2));
720 KRK_inv7_v1_plus_KRK_inv8[vdst] = (float)(kriki(2, 1) * vdst + kriki(2, 2));
721 }
722 }
723
724 double sigma = 0;
725 int correspCount = 0;
726 for (int vdst = 0; vdst < depthDst.rows; vdst++)
727 {
728 const float* depthDst_row = depthDst.ptr<float>(vdst);
729 const uchar* maskDst_row = selectMaskDst.ptr<uchar>(vdst);
730 for (int udst = 0; udst < depthDst.cols; udst++)
731 {
732 float ddst = depthDst_row[udst];
733
734 if (maskDst_row[udst])
735 {
736 float transformed_ddst = static_cast<float>(ddst * (KRK_inv6_u1[udst] + KRK_inv7_v1_plus_KRK_inv8[vdst]) + ktinv.z);
737
738 if (transformed_ddst > 0)
739 {
740 float transformed_ddst_inv = 1.f / transformed_ddst;
741 int usrc = cvRound(transformed_ddst_inv * (ddst * (KRK_inv0_u1[udst] + KRK_inv1_v1_plus_KRK_inv2[vdst]) + ktinv.x));
742 int vsrc = cvRound(transformed_ddst_inv * (ddst * (KRK_inv3_u1[udst] + KRK_inv4_v1_plus_KRK_inv5[vdst]) + ktinv.y));
743 if (r.contains(Point(usrc, vsrc)))
744 {
745 float dsrc = depthSrc.at<float>(vsrc, usrc);
746 if (validMaskSrc.at<uchar>(vsrc, usrc) && std::abs(transformed_ddst - dsrc) <= maxDepthDiff)
747 {
748 CV_DbgAssert(!cvIsNaN(dsrc));
749 Vec2s& c = corresps.at<Vec2s>(vsrc, usrc);
750 float diff = 0;
751 if (c[0] != -1)
752 {
753 diff = 0;
754 int exist_u1 = c[0], exist_v1 = c[1];
755
756 float exist_d1 = (float)(depthDst.at<float>(exist_v1, exist_u1) *
757 (KRK_inv6_u1[exist_u1] + KRK_inv7_v1_plus_KRK_inv8[exist_v1]) + ktinv.z);
758
759 if (transformed_ddst > exist_d1)
760 continue;
761 if (method == OdometryType::RGB)
762 diff = static_cast<float>(static_cast<int>(imageSrc.at<uchar>(vsrc, usrc)) -
763 static_cast<int>(imageDst.at<uchar>(vdst, udst)));
764 }
765 else
766 {
767 if (method == OdometryType::RGB)
768 diff = static_cast<float>(static_cast<int>(imageSrc.at<uchar>(vsrc, usrc)) -
769 static_cast<int>(imageDst.at<uchar>(vdst, udst)));
770 correspCount++;
771 }
772 c = Vec2s((short)udst, (short)vdst);
773 if (method == OdometryType::RGB)
774 {
775 diffs.at<float>(vsrc, usrc) = diff;
776 sigma += diff * diff;
777 }
778 }
779 }
780 }
781 }
782 }
783 }
784
785 _sigma = std::sqrt(sigma / double(correspCount));
786
787 _corresps.create(correspCount, 1, CV_32SC4);
788 Vec4i* corresps_ptr = _corresps.ptr<Vec4i>();
789 float* diffs_ptr = nullptr;
790 if (method == OdometryType::RGB)
791 {
792 _diffs.create(correspCount, 1, CV_32F);
793 diffs_ptr = _diffs.ptr<float>();
794 }
795 for (int vsrc = 0, i = 0; vsrc < corresps.rows; vsrc++)
796 {
797 const Vec2s* corresps_row = corresps.ptr<Vec2s>(vsrc);
798 const float* diffs_row = nullptr;
799 if (method == OdometryType::RGB)
800 diffs_row = diffs.ptr<float>(vsrc);
801 for (int usrc = 0; usrc < corresps.cols; usrc++)
802 {
803 const Vec2s& c = corresps_row[usrc];
804 const float& d = diffs_row[usrc];
805 if (c[0] != -1)
806 {
807 corresps_ptr[i] = Vec4i(usrc, vsrc, c[0], c[1]);
808 if (method == OdometryType::RGB)
809 diffs_ptr[i] = d;
810 i++;
811 }
812 }
813 }
814}

816void calcRgbdLsmMatrices(const Mat& cloud0, const Mat& Rt,
817 const Mat& dI_dx1, const Mat& dI_dy1,
818 const Mat& corresps, const Mat& _diffs, const double _sigma,
819 double fx, double fy, double sobelScaleIn,
820 Mat& AtA, Mat& AtB, OdometryTransformType transformType)
821{
822 int transformDim = getTransformDim(transformType);
823 AtA = Mat(transformDim, transformDim, CV_64FC1, Scalar(0));
824 AtB = Mat(transformDim, 1, CV_64FC1, Scalar(0));
825 double* AtB_ptr = AtB.ptr<double>();
826
827 CV_Assert(Rt.type() == CV_64FC1);
828 Affine3d rtmat(Rt);
829
830 const float* diffs_ptr = _diffs.ptr<float>();
831 const Vec4i* corresps_ptr = corresps.ptr<Vec4i>();

833
834 std::vector<double> A_buf(transformDim);
835 double* A_ptr = &A_buf[0];
836
837 for (int correspIndex = 0; correspIndex < corresps.rows; correspIndex++)

839 const Vec4i& c = corresps_ptr[correspIndex];
840 int u0 = c[0], v0 = c[1];
841 int u1 = c[2], v1 = c[3];
842

844 w = w > DBL_EPSILON ? 1. / w : 1.;
845
846 double w_sobelScale = w * sobelScaleIn;

852 A_ptr,
853 w_sobelScale * dI_dx1.at<short int>(v1, u1),
854 w_sobelScale * dI_dy1.at<short int>(v1, u1),
855 tp0, fx, fy);

858 {
859 double* AtA_ptr = AtA.ptr<double>(y);
860 for (int x = y; x < transformDim; x++)
861 {
862 AtA_ptr[x] += A_ptr[y] * A_ptr[x];
863 }
864 AtB_ptr[y] += A_ptr[y] * w * diffs_ptr[correspIndex];
865 }
866 }
867
868 for (int y = 0; y < transformDim; y++)
869 for (int x = y + 1; x < transformDim; x++)
870 AtA.at<double>(x, y) = AtA.at<double>(y, x);
871}

874void calcICPLsmMatrices(const Mat& cloud0, const Mat& Rt,
875 const Mat& cloud1, const Mat& normals1,
876 const Mat& corresps,
877 Mat& AtA, Mat& AtB, OdometryTransformType transformType)
878{
879 int transformDim = getTransformDim(transformType);
880 AtA = Mat(transformDim, transformDim, CV_64FC1, Scalar(0));
881 AtB = Mat(transformDim, 1, CV_64FC1, Scalar(0));
882 double* AtB_ptr = AtB.ptr<double>();
883
884 const int correspsCount = corresps.rows;
885
886 CV_Assert(Rt.type() == CV_64FC1);
887 const double* Rt_ptr = Rt.ptr<const double>();
888
889 AutoBuffer<float> diffs(correspsCount);
890 float* diffs_ptr = diffs.data();
891
892 AutoBuffer<Point3f> transformedPoints0(correspsCount);
893 Point3f* tps0_ptr = transformedPoints0.data();
894
895 const Vec4i* corresps_ptr = corresps.ptr<Vec4i>();
896
897 double sigma = 0;
898 for (int correspIndex = 0; correspIndex < corresps.rows; correspIndex++)
899 {
900 const Vec4i& c = corresps_ptr[correspIndex];
901 int u0 = c[0], v0 = c[1];
902 int u1 = c[2], v1 = c[3];
903
904 const Vec4f& p0 = cloud0.at<Vec4f>(v0, u0);
905 Point3f tp0;
906 tp0.x = (float)(p0[0] * Rt_ptr[0] + p0[1] * Rt_ptr[1] + p0[2] * Rt_ptr[2] + Rt_ptr[3]);
907 tp0.y = (float)(p0[0] * Rt_ptr[4] + p0[1] * Rt_ptr[5] + p0[2] * Rt_ptr[6] + Rt_ptr[7]);
908 tp0.z = (float)(p0[0] * Rt_ptr[8] + p0[1] * Rt_ptr[9] + p0[2] * Rt_ptr[10] + Rt_ptr[11]);
909
910 Vec4f n1 = normals1.at<Vec4f>(v1, u1);
911 Vec4f _v = cloud1.at<Vec4f>(v1, u1);
912 Point3f v = Point3f(_v[0], _v[1], _v[2]) - tp0;
913
914 tps0_ptr[correspIndex] = tp0;
915 diffs_ptr[correspIndex] = n1[0] * v.x + n1[1] * v.y + n1[2] * v.z;
916 sigma += diffs_ptr[correspIndex] * diffs_ptr[correspIndex];
917 }
918
919 sigma = std::sqrt(sigma / correspsCount);
920
921 std::vector<double> A_buf(transformDim);
922 double* A_ptr = &A_buf[0];
923 for (int correspIndex = 0; correspIndex < corresps.rows; correspIndex++)
924 {
925 const Vec4i& c = corresps_ptr[correspIndex];
926 int u1 = c[2], v1 = c[3];
927
928 double w = sigma +std::abs(diffs_ptr[correspIndex]);
929 w = w > DBL_EPSILON ? 1. / w : 1.;
930
931 Vec4f n4 = normals1.at<Vec4f>(v1, u1);
932 Vec4f p1 = cloud1.at<Vec4f>(v1, u1);
933
934 icpCoeffsFunc(transformType,
935 A_ptr, tps0_ptr[correspIndex], Point3d(p1[0], p1[1], p1[2]), Vec3d(n4[0], n4[1], n4[2]) * w);
936 for (int y = 0; y < transformDim; y++)
937 {
938 double* AtA_ptr = AtA.ptr<double>(y);
939 for (int x = y; x < transformDim; x++)
940 {
941 AtA_ptr[x] += A_ptr[y] * A_ptr[x];
942 }
943 AtB_ptr[y] += A_ptr[y] * w * diffs_ptr[correspIndex];
944 }
945 }
946
947 for (int y = 0; y < transformDim; y++)
948 for (int x = y + 1; x < transformDim; x++)
949 AtA.at<double>(x, y) = AtA.at<double>(y, x);
950}

953{
954 CV_Assert(ksi.size() == Size(1, 6) && ksi.type() == CV_64FC1);
955

961}

1155 VxN.z = v_get0(v_reinterpret_as_f32(v_extract<2>(v_reinterpret_as_u32(VxNv), v_setzero_u32())));

1160 // which is [A^T*A | A^T*b]
1161 // and gather sum
1162

1165 v_float32x4 nyzx;

1168 v_uint32x4 yz00 = v_extract<1>(aa, v_setzero_u32());
1169 v_uint32x4 x0y0, tmp;
1170 v_zip(aa, v_setzero_u32(), x0y0, tmp);
1171 nyzx = v_reinterpret_as_f32(v_combine_low(yz00, x0y0));
1172 }

1175 for (size_t i = 0; i < utBufferSize; i++)
1176 vutg[i] = v_load_aligned(upperTriangle + i * 4);
1177

1179 v_float32x4 v;

1181 v = v_setall_f32(VxN.x);

1183 v_store_aligned(upperTriangle + p * 4, v_muladd(v, n, vutg[p])); p++;

1185 v = v_setall_f32(VxN.y);
1186 v_store_aligned(upperTriangle + p * 4, v_muladd(v, vd, vutg[p])); p++;
1187 v_store_aligned(upperTriangle + p * 4, v_muladd(v, n, vutg[p])); p++;

1317 {
