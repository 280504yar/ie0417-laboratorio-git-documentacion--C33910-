1# Iterate all GLSL shaders (with suffix '.comp') in current directory.
2#
3# Use glslangValidator to compile them to SPIR-V shaders and write them
4# into .cpp files as unsigned int array.
5#
6# Also generate a header file 'spv_shader.hpp' to extern declare these shaders.
7
8import re

12dir = "./"
13license_decl = \

16'// of this distribution and at http://opencv.org/license.html.\n\n'
17
18precomp = '#include \"../../precomp.hpp\"\n'
19ns_head = '\nnamespace cv { namespace dnn { namespace vkcom {\n\n'
20ns_tail = '\n}}} // namespace cv::dnn::vkcom\n'
21
22headfile = open('spv_shader.hpp', 'w')
23headfile.write(license_decl)

25headfile.write('#define OPENCV_DNN_SPV_SHADER_HPP\n\n')
26headfile.write(ns_head)

28cppfile = open('spv_shader.cpp', 'w')

32cppfile.write(ns_head)

34cmd_remove = ''
35null_out = ''
36if sys.platform.find('win32') != -1:
37 cmd_remove = 'del'
38 null_out = ' >>nul 2>nul'
39elif sys.platform.find('linux') != -1:
40 cmd_remove = 'rm'

42else:

45insertList = []
46externList = []
47
48list = os.listdir(dir)
49for i in range(0, len(list)):
50 if (os.path.splitext(list[i])[-1] != '.comp'):
51 continue

54
55
56 bin_file = prefix + '.tmp'

58 print('Run cmd = ', cmd)

63
64 spv_txt_file = prefix + '.spv'
65 cmd = 'glslangValidator -V ' + path + ' -S comp -o ' + spv_txt_file + ' -x' #+ null_out
66 os.system(cmd)

68 infile_name = spv_txt_file

74 outfile.write(license_decl)
75 outfile.write(precomp)

77 # xxx.spv ==> xxx_spv.cpp
78 fmt = 'extern const unsigned int %s[%d] = {\n' % (array_name, size/4)
79 outfile.write(fmt)
80 for eachLine in infile:

82 continue
83 newline = ' ' + eachLine.replace('\t','')
84 outfile.write(newline)
85 infile.close()

91 externList.append(fmt)

93 insertList.append(fmt)
94
95 os.system(cmd_remove + ' ' + bin_file)

98for fmt in externList:

112cppfile.write('void initSPVMaps()\n{\n')
