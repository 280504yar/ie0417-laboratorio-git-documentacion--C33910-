16double deltaCIE94(const Vec3d& lab1, const Vec3d& lab2, double kH,
17 double kC, double kL, double k1, double k2)
18{
19 double dl = lab1[0] - lab2[0];
20 double c1 = sqrt(pow(lab1[1], 2) + pow(lab1[2], 2));
21 double c2 = sqrt(pow(lab2[1], 2) + pow(lab2[2], 2));
22 double dc = c1 - c2;
23 double da = lab1[1] - lab2[1];
24 double db = lab1[2] - lab2[2];
25 double dh = pow(da, 2) + pow(db, 2) - pow(dc, 2);
26 double sc = 1.0 + k1 * c1;
27 double sh = 1.0 + k2 * c1;
28 double sl = 1.0;
29 double res = pow(dl / (kL * sl), 2) + pow(dc / (kC * sc), 2) + dh / pow(kH * sh, 2);
30
31 return res > 0 ? sqrt(res) : 0;
32}

46double deltaCIEDE2000_(const Vec3d& lab1, const Vec3d& lab2, double kL,
47 double kC, double kH)
48{
49 double deltaLApo = lab2[0] - lab1[0];
50 double lBarApo = (lab1[0] + lab2[0]) / 2.0;
51 double C1 = sqrt(pow(lab1[1], 2) + pow(lab1[2], 2));
52 double C2 = sqrt(pow(lab2[1], 2) + pow(lab2[2], 2));
53 double cBar = (C1 + C2) / 2.0;
54 double G = sqrt(pow(cBar, 7) / (pow(cBar, 7) + pow(25, 7)));
55 double a1Apo = lab1[1] + lab1[1] / 2.0 * (1.0 - G);
56 double a2Apo = lab2[1] + lab2[1] / 2.0 * (1.0 - G);
57 double c1Apo = sqrt(pow(a1Apo, 2) + pow(lab1[2], 2));
58 double c2Apo = sqrt(pow(a2Apo, 2) + pow(lab2[2], 2));
59 double cBarApo = (c1Apo + c2Apo) / 2.0;
60 double deltaCApo = c2Apo - c1Apo;
61
62 double h1Apo;
63 if (c1Apo == 0)
64 {
65 h1Apo = 0.0;
66 }
67 else
68 {
69 h1Apo = atan2(lab1[2], a1Apo);
70 if (h1Apo < 0.0)
71 h1Apo += 2. * CV_PI;
72 }
73
74 double h2Apo;
75 if (c2Apo == 0)
76 {
77 h2Apo = 0.0;
78 }
79 else
80 {
81 h2Apo = atan2(lab2[2], a2Apo);
82 if (h2Apo < 0.0)
83 h2Apo += 2. * CV_PI;
84 }
85
86 double deltaHApo;
87 if (abs(h2Apo - h1Apo) <= CV_PI)
88 {
89 deltaHApo = h2Apo - h1Apo;
90 }
91 else if (h2Apo <= h1Apo)
92 {
93 deltaHApo = h2Apo - h1Apo + 2. * CV_PI;
94 }
95 else
96 {
97 deltaHApo = h2Apo - h1Apo - 2. * CV_PI;
98 }
99
100 double hBarApo;
101 if (c1Apo == 0 || c2Apo == 0)
102 {
103 hBarApo = h1Apo + h2Apo;
104 }
105 else if (abs(h1Apo - h2Apo) <= CV_PI)
106 {
107 hBarApo = (h1Apo + h2Apo) / 2.0;
108 }
109 else if (h1Apo + h2Apo < 2. * CV_PI)
110 {
111 hBarApo = (h1Apo + h2Apo + 2. * CV_PI) / 2.0;
112 }
113 else
114 {
115 hBarApo = (h1Apo + h2Apo - 2. * CV_PI) / 2.0;
116 }
117
118 double deltaH_Apo = 2.0 * sqrt(c1Apo * c2Apo) * sin(deltaHApo / 2.0);
119 double T = 1.0 - 0.17 * cos(hBarApo - toRad(30.)) + 0.24 * cos(2.0 * hBarApo) + 0.32 * cos(3.0 * hBarApo + toRad(6.0)) - 0.2 * cos(4.0 * hBarApo - toRad(63.0));
120 double sC = 1.0 + 0.045 * cBarApo;
121 double sH = 1.0 + 0.015 * cBarApo * T;
122 double sL = 1.0 + ((0.015 * pow(lBarApo - 50.0, 2.0)) / sqrt(20.0 + pow(lBarApo - 50.0, 2.0)));
123 double rC = 2.0 * sqrt(pow(cBarApo, 7.0) / (pow(cBarApo, 7.0) + pow(25, 7)));
124 double rT = -sin(toRad(60.0) * exp(-pow((hBarApo - toRad(275.0)) / toRad(25.0), 2.0))) * rC;
125 double res = (pow(deltaLApo / (kL * sL), 2.0) + pow(deltaCApo / (kC * sC), 2.0) + pow(deltaH_Apo / (kH * sH), 2.0) + rT * (deltaCApo / (kC * sC)) * (deltaH_Apo / (kH * sH)));
126 return res > 0 ? sqrt(res) : 0;
127}

134double deltaCMC(const Vec3d& lab1, const Vec3d& lab2, double kL, double kC)
135{
136 double dL = lab2[0] - lab1[0];
137 double da = lab2[1] - lab1[1];
138 double db = lab2[2] - lab1[2];
139 double C1 = sqrt(pow(lab1[1], 2.0) + pow(lab1[2], 2.0));
140 double C2 = sqrt(pow(lab2[1], 2.0) + pow(lab2[2], 2.0));
141 double dC = C2 - C1;
142 double dH = sqrt(pow(da, 2) + pow(db, 2) - pow(dC, 2));
143
144 double H1;
145 if (C1 == 0.)
146 {
147 H1 = 0.0;
148 }
149 else
150 {
151 H1 = atan2(lab1[2], lab1[1]);
152 if (H1 < 0.0)
153 H1 += 2. * CV_PI;
154 }
155
156 double F = pow(C1, 2) / sqrt(pow(C1, 4) + 1900);
157 double T = (H1 > toRad(164) && H1 <= toRad(345))
158 ? 0.56 + abs(0.2 * cos(H1 + toRad(168)))
159 : 0.36 + abs(0.4 * cos(H1 + toRad(35)));
160 double sL = lab1[0] < 16. ? 0.511 : (0.040975 * lab1[0]) / (1.0 + 0.01765 * lab1[0]);
161 double sC = (0.0638 * C1) / (1.0 + 0.0131 * C1) + 0.638;
162 double sH = sC * (F * T + 1.0 - F);
163
164 return sqrt(pow(dL / (kL * sL), 2.0) + pow(dC / (kC * sC), 2.0) + pow(dH / sH, 2.0));
165}
