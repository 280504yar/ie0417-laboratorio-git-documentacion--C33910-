117bool VideoCapture::open(const String& filename, int apiPreference, const std::vector<int>& params)
118{
119 CV_INSTRUMENT_REGION();
120
121 if (isOpened())
122 {
123 release();
124 }
125
126 const VideoCaptureParameters parameters(params);
127 const std::vector<VideoBackendInfo> backends = cv::videoio_registry::getAvailableBackends_CaptureByFilename();
128 for (size_t i = 0; i < backends.size(); i++)
129 {
130 const VideoBackendInfo& info = backends[i];
131 if (apiPreference == CAP_ANY || apiPreference == info.id)
132 {
133 if (!info.backendFactory)
134 {
135 CV_LOG_DEBUG(NULL, "VIDEOIO(" << info.name << "): factory is not available (plugins require filesystem support)");
136 continue;
137 }
138 CV_CAPTURE_LOG_DEBUG(NULL,
139 cv::format("VIDEOIO(%s): trying capture filename='%s' ...",
140 info.name, filename.c_str()));
141 CV_Assert(!info.backendFactory.empty());
142 const Ptr<IBackend> backend = info.backendFactory->getBackend();
143 if (!backend.empty())
144 {
145 try
146 {
147 icap = backend->createCapture(filename, parameters);
148 if (!icap.empty())
149 {
150 CV_CAPTURE_LOG_DEBUG(NULL,
151 cv::format("VIDEOIO(%s): created, isOpened=%d",
152 info.name, icap->isOpened()));
153 if (icap->isOpened())
154 {
155 return true;
156 }
157 icap.release();
158 }
159 else
160 {
161 CV_CAPTURE_LOG_DEBUG(NULL,
162 cv::format("VIDEOIO(%s): can't create capture",
163 info.name));
164 }
165 }
166 catch (const cv::Exception& e)
167 {
168 if (throwOnFail && apiPreference != CAP_ANY)
169 {
170 throw;
171 }
172 CV_LOG_WARNING(NULL,
173 cv::format("VIDEOIO(%s): raised OpenCV exception:\n\n%s\n",
174 info.name, e.what()));
175 }
176 catch (const std::exception& e)
177 {
178 if (throwOnFail && apiPreference != CAP_ANY)
179 {
180 throw;
181 }
182 CV_LOG_WARNING(NULL, cv::format("VIDEOIO(%s): raised C++ exception:\n\n%s\n",
183 info.name, e.what()));
184 }
185 catch (...)
186 {
187 if (throwOnFail && apiPreference != CAP_ANY)
188 {
189 throw;
190 }
191 CV_LOG_WARNING(NULL,
192 cv::format("VIDEOIO(%s): raised unknown C++ exception!\n\n",
193 info.name));
194 }
195 }
196 else
197 {
198 CV_CAPTURE_LOG_DEBUG(NULL,
199 cv::format("VIDEOIO(%s): backend is not available "
200 "(plugin is missing, or can't be loaded due "
201 "dependencies or it is not compatible)",
202 info.name));
203 }
204 }
205 }
206
207 if(apiPreference != CAP_ANY)
208 {
209 bool found = cv::videoio_registry::isBackendBuiltIn(static_cast<VideoCaptureAPIs>(apiPreference));
210 if (found)
211 {
212 CV_LOG_WARNING(NULL, cv::format("VIDEOIO(%s): backend is generally available "
213 "but can't be used to capture by name",
214 cv::videoio_registry::getBackendName(static_cast<VideoCaptureAPIs>(apiPreference)).c_str()));
215 }
216 }
217
218 if (throwOnFail)
219 {
220 CV_Error_(Error::StsError, ("could not open '%s'", filename.c_str()));
221 }
222
223 if (cv::videoio_registry::checkDeprecatedBackend(apiPreference))
224 {
225 CV_LOG_DEBUG(NULL,
226 cv::format("VIDEOIO(%s): backend is removed from OpenCV",
227 cv::videoio_registry::getBackendName((VideoCaptureAPIs) apiPreference).c_str()));
228 }
229 else
230 {
231 CV_LOG_DEBUG(NULL, "VIDEOIO: chosen backend does not work or wrong. "
232 "Please make sure that your computer support chosen backend and OpenCV built "
233 "with right flags.");
234 }
235
236 return false;
237}

611double VideoCapture::get(int propId) const
612{

614 {
615 int api = 0;
616 if (icap && icap->isOpened())
617 {
618 api = icap->getCaptureDomain();

625 }

630bool VideoCapture::waitAny(const std::vector<VideoCapture>& streams,
631 CV_OUT std::vector<int>& readyIndex, int64 timeoutNs)
632{
633 CV_Assert(!streams.empty());
634
635 VideoCaptureAPIs backend = (VideoCaptureAPIs)streams[0].icap->getCaptureDomain();
636
637 for (size_t i = 1; i < streams.size(); ++i)
638 {
639 VideoCaptureAPIs backend_i = (VideoCaptureAPIs)streams[i].icap->getCaptureDomain();
640 CV_CheckEQ((int)backend, (int)backend_i, "All captures must have the same backend");
641 }
642
643#if (defined HAVE_CAMV4L2 || defined HAVE_VIDEOIO) // see cap_v4l.cpp guard
644 if (backend == CAP_V4L2)
645 {
646 return VideoCapture_V4L_waitAny(streams, readyIndex, timeoutNs);
647 }
648#else
649 CV_UNUSED(readyIndex);
650 CV_UNUSED(timeoutNs);
651#endif
652 CV_Error(Error::StsNotImplemented, "VideoCapture::waitAny() is supported by V4L backend only");
653}

719bool VideoWriter::open(const String& filename, int apiPreference, int fourcc, double fps,
720 const Size& frameSize, const std::vector<int>& params)
721{
722 CV_INSTRUMENT_REGION();
723
724 if (isOpened())
725 {
726 release();
727 }
728
729 const VideoWriterParameters parameters(params);
730 for (const auto& info : videoio_registry::getAvailableBackends_Writer())
731 {
732 if (apiPreference == CAP_ANY || apiPreference == info.id)
733 {
734 CV_WRITER_LOG_DEBUG(NULL,
735 cv::format("VIDEOIO(%s): trying writer with filename='%s' "
736 "fourcc=0x%08x fps=%g sz=%dx%d isColor=%d...",
737 info.name, filename.c_str(), (unsigned)fourcc, fps,
738 frameSize.width, frameSize.height,
739 parameters.get(VIDEOWRITER_PROP_IS_COLOR, true)));
740 CV_Assert(!info.backendFactory.empty());
741 const Ptr<IBackend> backend = info.backendFactory->getBackend();
742 if (!backend.empty())
743 {
744 try
745 {
746 iwriter = backend->createWriter(filename, fourcc, fps, frameSize, parameters);
747 if (!iwriter.empty())
748 {
749
750 CV_WRITER_LOG_DEBUG(NULL,
751 cv::format("VIDEOIO(%s): created, isOpened=%d",
752 info.name, iwriter->isOpened()));
753 if (param_VIDEOIO_DEBUG || param_VIDEOWRITER_DEBUG)
754 {
755 for (int key: parameters.getUnused())
756 {
757 CV_LOG_WARNING(NULL,
758 cv::format("VIDEOIO(%s): parameter with key '%d' was unused",
759 info.name, key));
760 }
761 }
762 if (iwriter->isOpened())
763 {
764 return true;
765 }
766 iwriter.release();
767 }
768 else
769 {
770 CV_WRITER_LOG_DEBUG(NULL, cv::format("VIDEOIO(%s): can't create writer",
771 info.name));
772 }
773 }
774 catch (const cv::Exception& e)
775 {
776 CV_LOG_WARNING(NULL,
777 cv::format("VIDEOIO(%s): raised OpenCV exception:\n\n%s\n",
778 info.name, e.what()));
779 }
780 catch (const std::exception& e)
781 {
782 CV_LOG_WARNING(NULL, cv::format("VIDEOIO(%s): raised C++ exception:\n\n%s\n",
783 info.name, e.what()));
784 }
785 catch (...)
786 {
787 CV_LOG_WARNING(NULL,
788 cv::format("VIDEOIO(%s): raised unknown C++ exception!\n\n",
789 info.name));
790 }
791 }
