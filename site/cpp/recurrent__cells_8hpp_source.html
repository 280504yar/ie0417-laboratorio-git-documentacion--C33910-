26template<class T>
27class LSTMOp final : public CUDABackendNode
28{
29public:
30 using wrapper_type = GetCUDABackendWrapperType<T>;
31
32 LSTMOp(csl::Stream stream_, csl::cudnn::Handle handle, const Mat& filters, const Mat& h0,
33 const Mat& c0, const RNNConfiguration& config)
34 : stream(std::move(stream_))
35 {
36 typename csl::LSTM<T>::params_type params{
37 {filters.total(), 1, 1}, // reshape
38 config.seqLength,
39 config.numLayers,
40 config.hiddenSize,
41 config.inputSize,
42 config.miniBatch,
43 config.bidirectional,
44 0.0, /* dropout */
45 csl::cudnn::RNNDescriptor<T>::RNNMode::LSTM
46 };
47
48 lstm = csl::LSTM<T>(handle, params);
49 auto correct_shape_filters = filters.reshape(1, {static_cast<int>(filters.total()), 1, 1});
50 filtersTensor = csl::makeTensorHeader<T>(correct_shape_filters);
51 csl::copyMatToTensor<T>(correct_shape_filters, filtersTensor, stream);
52
53 h0Tensor = csl::makeTensorHeader<T>(h0);
54 csl::copyMatToTensor<T>(h0, h0Tensor, stream);
55
56 c0Tensor = csl::makeTensorHeader<T>(c0);
57 csl::copyMatToTensor<T>(c0, c0Tensor, stream);
58 }

60 void forward(const std::vector<cv::Ptr<BackendWrapper>>& inputs,
61 const std::vector<cv::Ptr<BackendWrapper>>& outputs,
62 csl::Workspace& workspace) override
63 {
64 CV_Assert(inputs.size() == 1 && !outputs.empty());
65
66 auto input_wrapper = inputs[0].dynamicCast<wrapper_type>();
67 auto input = input_wrapper->getView();
68
69 auto y_output_wrapper = outputs[0].dynamicCast<wrapper_type>();
70 auto y_output = y_output_wrapper->getSpan();
71
72 Ptr<wrapper_type> yc_output_wrapper = outputs.size() == 2 ? outputs[1].dynamicCast<wrapper_type>() : Ptr<wrapper_type>();
73 csl::TensorSpan<T> yc_output = yc_output_wrapper.empty() ? csl::TensorSpan<T>() : yc_output_wrapper->getSpan();
74
75 lstm.inference(input, y_output, yc_output, filtersTensor, h0Tensor, c0Tensor, workspace);
76 }
77
78 std::size_t get_workspace_memory_in_bytes() const noexcept override
79 {
80 return lstm.get_workspace_memory_in_bytes();
81 }
82
83private:
84 csl::LSTM<T> lstm;
85 csl::Stream stream;
86 csl::Tensor<T> filtersTensor;
87 csl::Tensor<T> h0Tensor;
88 csl::Tensor<T> c0Tensor;
89};
