85// select YCrCb (crcb) or YUV (yuv) coefficients into coeffs; size N deduced
86template<typename CT, int N>
87static inline void selectYuvCoeffs(CT (&coeffs)[N], bool isCrCb, const CT (&crcb)[N], const CT (&yuv)[N])
88{
89 for(int i = 0; i < N; i++)
90 coeffs[i] = isCrCb ? crcb[i] : yuv[i];

95template<typename _Tp> struct RGB2YCrCb_f
96{
97 typedef _Tp channel_type;
98
99 RGB2YCrCb_f(int _srccn, int _blueIdx, bool _isCrCb) :
100 srccn(_srccn), blueIdx(_blueIdx), isCrCb(_isCrCb)
101 {
102 static const float coeffs_crb[] = { R2YF, G2YF, B2YF, YCRF, YCBF };
103 static const float coeffs_yuv[] = { R2YF, G2YF, B2YF, R2VF, B2UF };
104 selectYuvCoeffs(coeffs, isCrCb, coeffs_crb, coeffs_yuv);
105 if(blueIdx == 0)
106 std::swap(coeffs[0], coeffs[2]);
107 }
108
109 void operator()(const _Tp* src, _Tp* dst, int n) const
110 {
111 int scn = srccn, bidx = blueIdx;
112 int yuvOrder = !isCrCb; //1 if YUV, 0 if YCrCb
113 const _Tp delta = ColorChannel<_Tp>::half();
114 float C0 = coeffs[0], C1 = coeffs[1], C2 = coeffs[2], C3 = coeffs[3], C4 = coeffs[4];
115 n *= 3;
116 for(int i = 0; i < n; i += 3, src += scn)
117 {
118 _Tp Y = saturate_cast<_Tp>(src[0]*C0 + src[1]*C1 + src[2]*C2);
119 _Tp Cr = saturate_cast<_Tp>((src[bidx^2] - Y)*C3 + delta);
120 _Tp Cb = saturate_cast<_Tp>((src[bidx] - Y)*C4 + delta);
121 dst[i] = Y; dst[i+1+yuvOrder] = Cr; dst[i+2-yuvOrder] = Cb;
122 }
123 }
124 int srccn, blueIdx;
125 bool isCrCb;
126 float coeffs[5];
127};

278#if (CV_SIMD || CV_SIMD_SCALABLE)

283 v_int16 g2y = vx_setall_s16((short)C1);

285 v_int16 one = vx_setall_s16(1);
286 v_int16 z = vx_setzero_s16();
287
288 v_int16 bg2y, r12y;
289 v_int16 dummy;
290 v_zip(b2y, g2y, bg2y, dummy);
291 v_zip(r2y, one, r12y, dummy);
292
293 v_int16 vdescale = vx_setall_s16(1 << (shift-1));
294 v_int32 vc3 = vx_setall_s32(C3);
295 v_int32 vc4 = vx_setall_s32(C4);
296 v_int32 vdd = vx_setall_s32(sdelta + descale);
297
298 for(; i < n;
299 i += vsize, src += vsize*scn, dst += vsize*3)
300 {
301 if ( i > n - vsize ) {
302 if (i == 0 || src == dst) {
303 break;
304 }

308 dst -= backup * 3;
309 }

312 {
313 v_load_deinterleave(src, b, g, r);
314 }
315 else
316 {
317 v_load_deinterleave(src, b, g, r, a);
318 }
319
320 v_uint16 y, cr, cb;
321
322 v_int16 sb = v_reinterpret_as_s16(b);
323 v_int16 sr = v_reinterpret_as_s16(r);
324 v_int16 sg = v_reinterpret_as_s16(g);
325
326 v_int16 bg0, bg1;
327 v_int16 rd0, rd1;
328 v_zip(sb, sg, bg0, bg1);
329 v_zip(sr, vdescale, rd0, rd1);
330
331 // fixing 16bit signed multiplication
332 v_int16 mr, mg, mb;
333 mr = v_and(v_lt(sr, z), r2y);
334 mg = v_and(v_lt(sg, z), g2y);
335 mb = v_and(v_lt(sb, z), b2y);
336 v_int16 fixmul = v_shl(v_add_wrap(mr, v_add_wrap(mg, mb)), fix_shift);
337

342
343 if(bidx)
344 v_swap(r, b);
345

347 v_uint32 r0, r1, b0, b1;
348 v_expand(r, r0, r1);
349 v_expand(b, b0, b1);
350
351 v_uint32 uy0, uy1;
352 v_expand(y, uy0, uy1);
353
354 v_int32 sr0 = v_reinterpret_as_s32(r0);
355 v_int32 sr1 = v_reinterpret_as_s32(r1);
356 v_int32 sb0 = v_reinterpret_as_s32(b0);
357 v_int32 sb1 = v_reinterpret_as_s32(b1);
358 v_int32 sy0 = v_reinterpret_as_s32(uy0);
359 v_int32 sy1 = v_reinterpret_as_s32(uy1);
360
361 sr0 = v_sub(sr0, sy0); sr1 = v_sub(sr1, sy1);
362 sb0 = v_sub(sb0, sy0); sb1 = v_sub(sb1, sy1);
363
364 v_int32 v_scr0, v_scr1, v_scb0, v_scb1;
365
366 v_scr0 = v_shr(v_add(v_mul(sr0, vc3), vdd), shift);
367 v_scr1 = v_shr(v_add(v_mul(sr1, vc3), vdd), shift);
368 v_scb0 = v_shr(v_add(v_mul(sb0, vc4), vdd), shift);
369 v_scb1 = v_shr(v_add(v_mul(sb1, vc4), vdd), shift);
370
371 // saturate and pack
372 cr = v_pack_u(v_scr0, v_scr1);
373 cb = v_pack_u(v_scb0, v_scb1);
374
375 if(yuvOrder)
376 {
377 v_store_interleave(dst, y, cb, cr);
378 }
379 else
380 {
381 v_store_interleave(dst, y, cr, cb);
382 }
383 }
384 vx_cleanup();
385#endif
386 for( ; i < n; i++, src += scn, dst += 3)
387 {
388 int Y = CV_DESCALE(src[0]*C0 + src[1]*C1 + src[2]*C2, shift);
389 int Cr = CV_DESCALE((src[bidx^2] - Y)*C3 + sdelta, shift);
390 int Cb = CV_DESCALE((src[bidx] - Y)*C4 + sdelta, shift);

392 dst[1+yuvOrder] = saturate_cast<ushort>(Cr);

1841static const int RGB2YUV422_SHIFT = 14;
1842
1843// Coefficients based on ITU.BT-601, ISBN 1-878707-09-4 (https://fourcc.org/fccyvrgb.php)
1844// The conversion coefficients for RGB to YUV422 are based on the ones for RGB to YUV.
1845// For both Y components, the coefficients are applied as given in the link to each input RGB pixel
1846// separately. For U and V, they are reduced by half to account for two RGB pixels contributing
1847// to the same U and V values. In other words, the U and V contributions from the two RGB pixels
1848// are averaged. The integer versions are obtained by multiplying the float versions by 16384
1849// and rounding to the nearest integer so that resulting values are in these bounds:
1850// Y : [16, 235]; Cb, Cr: [16, 240] centered at 128
1851
1852static const int R2Y422 = 4211; // 0.299077 * (236 - 16) / 256 * 16384
1853static const int G2Y422 = 8258; // 0.586506 * (236 - 16) / 256 * 16384
1854static const int B2Y422 = 1606; // 0.114062 * (236 - 16) / 256 * 16384
1855
1856static const int R2U422 = -1212; // -0.148 * 8192
1857static const int G2U422 = -2384; // -0.291 * 8192
1858static const int B2U422 = 3596; // 0.439 * 8192
1859static const int G2V422 = -3015; // -0.368 * 8192
1860static const int B2V422 = -582; // -0.071 * 8192
1861
1862static inline void RGB2Y(const uchar r, const uchar g, const uchar b, uchar& y)
1863{
1864 int y_ = r * R2Y422 + g * G2Y422 + b * B2Y422 + (1 << RGB2YUV422_SHIFT) * 16;
1865 y = saturate_cast<uchar>(((1 << (RGB2YUV422_SHIFT-1)) + y_) >> RGB2YUV422_SHIFT);
1866}

1868static inline void RGB2UV(const uchar r1, const uchar g1, const uchar b1,
1869 const uchar r2, const uchar g2, const uchar b2,
1870 uchar& u, uchar& v)
1871{
1872 int sr = r1 + r2, sg = g1 + g2, sb = b1 + b2;
1873
1874 int u_ = sr * R2U422 + sg * G2U422 + sb * B2U422 + (1 << (RGB2YUV422_SHIFT-1)) * 256;
1875 u = saturate_cast<uchar>(((1 << (RGB2YUV422_SHIFT-1)) + u_) >> RGB2YUV422_SHIFT);
1876
1877 int v_ = sr * B2U422 + sg * G2V422 + sb * B2V422 + (1 << (RGB2YUV422_SHIFT-1)) * 256;
1878 v = saturate_cast<uchar>(((1 << (RGB2YUV422_SHIFT-1)) + v_) >> RGB2YUV422_SHIFT);
1879}

1881#if (CV_SIMD || CV_SIMD_SCALABLE)
1882// SIMD RGB2Y: v_dotprod (pmaddwd) keeps the multiply in 16-bit, avoiding emulated 32-bit mul on pre-SSE4.1
1883static inline v_uint8 v_RGB2Y(const v_uint8& r, const v_uint8& g, const v_uint8& b)
1884{
1885 v_int16 crg, cb, dummy;
1886 v_zip(vx_setall_s16((short)R2Y422), vx_setall_s16((short)G2Y422), crg, dummy);
1887 v_zip(vx_setall_s16((short)B2Y422), vx_setall_s16(0), cb, dummy);
1888 v_int32 vbias = vx_setall_s32((1 << RGB2YUV422_SHIFT) * 16 + (1 << (RGB2YUV422_SHIFT - 1)));
1889 v_int16 vzero = vx_setzero_s16();
1890
1891 v_uint16 r0, r1, g0, g1, b0, b1;
1892 v_expand(r, r0, r1); v_expand(g, g0, g1); v_expand(b, b0, b1);
1893 v_int16 sr0 = v_reinterpret_as_s16(r0), sr1 = v_reinterpret_as_s16(r1);
1894 v_int16 sg0 = v_reinterpret_as_s16(g0), sg1 = v_reinterpret_as_s16(g1);
1895 v_int16 sb0 = v_reinterpret_as_s16(b0), sb1 = v_reinterpret_as_s16(b1);
1896
1897 v_int16 rg00, rg01, rg10, rg11, bz00, bz01, bz10, bz11;
1898 v_zip(sr0, sg0, rg00, rg01); v_zip(sr1, sg1, rg10, rg11);
1899 v_zip(sb0, vzero, bz00, bz01); v_zip(sb1, vzero, bz10, bz11);
1900
1901 v_uint32 y00, y01, y10, y11;
1902 y00 = v_shr<RGB2YUV422_SHIFT>(v_reinterpret_as_u32(v_add(v_add(v_dotprod(rg00, crg), v_dotprod(bz00, cb)), vbias)));
1903 y01 = v_shr<RGB2YUV422_SHIFT>(v_reinterpret_as_u32(v_add(v_add(v_dotprod(rg01, crg), v_dotprod(bz01, cb)), vbias)));
1904 y10 = v_shr<RGB2YUV422_SHIFT>(v_reinterpret_as_u32(v_add(v_add(v_dotprod(rg10, crg), v_dotprod(bz10, cb)), vbias)));
1905 y11 = v_shr<RGB2YUV422_SHIFT>(v_reinterpret_as_u32(v_add(v_add(v_dotprod(rg11, crg), v_dotprod(bz11, cb)), vbias)));
1906
1907 return v_pack(v_pack(y00, y01), v_pack(y10, y11));
1908}
