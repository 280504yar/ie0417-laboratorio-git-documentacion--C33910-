37void Board::Impl::matchImagePoints(InputArrayOfArrays detectedCorners, InputArray detectedIds, OutputArray _objPoints,
38 OutputArray imgPoints) const {
39 CV_Assert(detectedIds.total() == detectedCorners.total());
40 CV_Assert(detectedIds.total() > 0ull);
41 CV_Assert(detectedCorners.depth() == CV_32F);
42
43 size_t nDetectedMarkers = detectedIds.total();
44
45 vector<Point3f> objPnts;
46 objPnts.reserve(nDetectedMarkers);
47
48 vector<Point2f> imgPnts;
49 imgPnts.reserve(nDetectedMarkers);
50
51 // look for detected markers that belong to the board and get their information
52 Mat detectedIdsMat = detectedIds.getMat();
53 vector<Mat> detectedCornersVecMat;
54
55 detectedCorners.getMatVector(detectedCornersVecMat);
56 CV_Assert((int)detectedCornersVecMat.front().total()*detectedCornersVecMat.front().channels() == 8);
57
58 for(unsigned int i = 0; i < nDetectedMarkers; i++) {
59 int currentId = detectedIdsMat.at<int>(i);
60 for(unsigned int j = 0; j < ids.size(); j++) {
61 if(currentId == ids[j]) {
62 for(int p = 0; p < 4; p++) {
63 objPnts.push_back(objPoints[j][p]);
64 imgPnts.push_back(detectedCornersVecMat[i].ptr<Point2f>(0)[p]);
65 }
66 }
67 }
68 }
69
70 // create output
71 Mat(objPnts).copyTo(_objPoints);
72 Mat(imgPnts).copyTo(imgPoints);
73}

75void Board::Impl::generateImage(Size outSize, OutputArray img, int marginSize, int borderBits) const {
76 CV_Assert(!outSize.empty());
77 CV_Assert(marginSize >= 0);
78
79 img.create(outSize, CV_8UC1);
80 Mat out = img.getMat();
81 out.setTo(Scalar::all(255));
82 out.adjustROI(-marginSize, -marginSize, -marginSize, -marginSize);
83
84 // calculate max and min values in XY plane
85 CV_Assert(objPoints.size() > 0);
86 float minX, maxX, minY, maxY;
87 minX = maxX = objPoints[0][0].x;
88 minY = maxY = objPoints[0][0].y;
89
90 for(unsigned int i = 0; i < objPoints.size(); i++) {
91 for(int j = 0; j < 4; j++) {
92 minX = min(minX, objPoints[i][j].x);
93 maxX = max(maxX, objPoints[i][j].x);
94 minY = min(minY, objPoints[i][j].y);
95 maxY = max(maxY, objPoints[i][j].y);
96 }
97 }
98
99 float sizeX = maxX - minX;
100 float sizeY = maxY - minY;
101
102 // now paint each marker
103 Mat marker;
104 Point2f outCorners[3];
105 Point2f inCorners[3];
106 for(unsigned int m = 0; m < objPoints.size(); m++) {
107 // transform corners to markerZone coordinates
108 for(int j = 0; j < 3; j++) {
109 Point2f pf = Point2f(objPoints[m][j].x, objPoints[m][j].y);
110 // move top left to 0, 0
111 pf -= Point2f(minX, minY);
112 pf.x = pf.x / sizeX * float(out.cols);
113 pf.y = pf.y / sizeY * float(out.rows);
114 outCorners[j] = pf;
115 }
116
117 // get marker
118 Point2f vecWidth = outCorners[1] - outCorners[0];
119 float width = (float)cv::norm(vecWidth);
120
121 Point2f vecHeight = outCorners[2] - outCorners[0];
122 float height = (float)cv::norm(vecHeight);
123
124 Size dst_sz(cvRound(width), cvRound(height));
125 dst_sz.width = dst_sz.height = std::min(dst_sz.width, dst_sz.height); //marker should be square
126 dictionary.generateImageMarker(ids[m], dst_sz.width, marker, borderBits);
127
128 if((outCorners[0].y == outCorners[1].y) && (outCorners[1].x == outCorners[2].x)) {
129 // marker is aligned to image axes
130 marker.copyTo(out(Rect(outCorners[0], dst_sz)));
131 continue;
132 }
133
134 // interpolate tiny marker to marker position in markerZone
135 inCorners[0] = Point2f(-0.5f, -0.5f);
136 inCorners[1] = Point2f(marker.cols - 0.5f, -0.5f);
137 inCorners[2] = Point2f(marker.cols - 0.5f, marker.rows - 0.5f);
138
139 // remove perspective
140 Mat transformation = getAffineTransform(inCorners, outCorners);
141 warpAffine(marker, out, transformation, out.size(), INTER_LINEAR,
142 BORDER_TRANSPARENT);
143 }
144}

156Board::Board(InputArrayOfArrays objPoints, const Dictionary &dictionary, InputArray ids):
157 Board(new Board::Impl(dictionary)) {
158 CV_Assert(objPoints.total() == ids.total());
159 CV_Assert(objPoints.type() == CV_32FC3 || objPoints.type() == CV_32FC1);
160
161 vector<vector<Point3f> > obj_points_vector;
162 Point3f rightBottomBorder = Point3f(0.f, 0.f, 0.f);
163 for (unsigned int i = 0; i < objPoints.total(); i++) {
164 vector<Point3f> corners;
165 Mat corners_mat = objPoints.getMat(i);
166
167 if (corners_mat.type() == CV_32FC1)
168 corners_mat = corners_mat.reshape(3);
169 CV_Assert(corners_mat.total() == 4);
170
171 for (int j = 0; j < 4; j++) {
172 const Point3f &corner = corners_mat.at<Point3f>(j);
173 corners.push_back(corner);
174 rightBottomBorder.x = std::max(rightBottomBorder.x, corner.x);
175 rightBottomBorder.y = std::max(rightBottomBorder.y, corner.y);
176 rightBottomBorder.z = std::max(rightBottomBorder.z, corner.z);
177 }
178 obj_points_vector.push_back(corners);
179 }
180
181 ids.copyTo(impl->ids);
182 impl->objPoints = obj_points_vector;
183 impl->rightBottomBorder = rightBottomBorder;
184}

242GridBoard::GridBoard(const Size& size, float markerLength, float markerSeparation,
243 const Dictionary &dictionary, InputArray ids):
244 Board(new GridBoardImpl(dictionary, size, markerLength, markerSeparation)) {
245 float onePin = markerLength / ((float)(dictionary.markerSize+2));
246 if (markerSeparation < onePin*.7f) {
247 CV_LOG_WARNING(NULL, "Marker border " << markerSeparation << " is less than 70% of ArUco pin size "
248 << onePin << ". Please increase markerSeparation or decrease markerLength for stable board detection");

254

256 ids.copyTo(impl->ids);
257 } else {
258 impl->ids = std::vector<int>(totalMarkers);
259 std::iota(impl->ids.begin(), impl->ids.end(), 0);
260 }
261
262 // calculate Board objPoints
263 for (int y = 0; y < size.height; y++) {
264 for (int x = 0; x < size.width; x++) {
265 vector <Point3f> corners(4);
266 corners[0] = Point3f(x * (markerLength + markerSeparation),
267 y * (markerLength + markerSeparation), 0);
268 corners[1] = corners[0] + Point3f(markerLength, 0, 0);
269 corners[2] = corners[0] + Point3f(markerLength, markerLength, 0);
270 corners[3] = corners[0] + Point3f(0, markerLength, 0);
271 objPoints.push_back(corners);
272 }
273 }
274 impl->objPoints = objPoints;
275 impl->rightBottomBorder = Point3f(size.width * markerLength + markerSeparation * (size.width - 1),
276 size.height * markerLength + markerSeparation * (size.height - 1), 0.f);
277}

332void CharucoBoardImpl::createCharucoBoard() {
333 float diffSquareMarkerLength = (squareLength - markerLength) / 2;
334 int totalMarkers = (int)(ids.size());
335 // calculate Board objPoints
336 int nextId = 0;
337 objPoints.clear();
338 for(int y = 0; y < size.height; y++) {
339 for(int x = 0; x < size.width; x++) {
340
341 if(legacyPattern && (size.height % 2 == 0)) { // legacy behavior only for even row count patterns
342 if((y + 1) % 2 == x % 2) continue; // black corner, no marker here
343 } else {
344 if(y % 2 == x % 2) continue; // black corner, no marker here
345 }
346
347 vector<Point3f> corners(4);
348 corners[0] = Point3f(x * squareLength + diffSquareMarkerLength,
349 y * squareLength + diffSquareMarkerLength, 0);
350 corners[1] = corners[0] + Point3f(markerLength, 0, 0);
351 corners[2] = corners[0] + Point3f(markerLength, markerLength, 0);
352 corners[3] = corners[0] + Point3f(0, markerLength, 0);
353 objPoints.push_back(corners);
354 // first ids in dictionary
355 if (totalMarkers == 0)
356 ids.push_back(nextId);
357 nextId++;
358 }
359 }
360 if (totalMarkers > 0 && nextId != totalMarkers)
361 CV_Error(cv::Error::StsBadSize, "Size of ids must be equal to the number of markers: "+std::to_string(nextId));
362
363 // now fill chessboardCorners
364 chessboardCorners.clear();
365 for(int y = 0; y < size.height - 1; y++) {
366 for(int x = 0; x < size.width - 1; x++) {
367 Point3f corner;
368 corner.x = (x + 1) * squareLength;
369 corner.y = (y + 1) * squareLength;
370 corner.z = 0;
371 chessboardCorners.push_back(corner);
372 }
373 }
374 rightBottomBorder = Point3f(size.width * squareLength, size.height * squareLength, 0.f);
375 calcNearestMarkerCorners();
376}

379void CharucoBoardImpl::calcNearestMarkerCorners() {
380 nearestMarkerIdx.clear();
381 nearestMarkerCorners.clear();
382 nearestMarkerIdx.resize(chessboardCorners.size());
383 nearestMarkerCorners.resize(chessboardCorners.size());
384 unsigned int nMarkers = (unsigned int)objPoints.size();
385 unsigned int nCharucoCorners = (unsigned int)chessboardCorners.size();
386 for(unsigned int i = 0; i < nCharucoCorners; i++) {
387 double minDist = -1; // distance of closest markers
388 Point3f charucoCorner = chessboardCorners[i];
389 for(unsigned int j = 0; j < nMarkers; j++) {
390 // calculate distance from marker center to charuco corner
391 Point3f center = Point3f(0, 0, 0);
392 for(unsigned int k = 0; k < 4; k++)
393 center += objPoints[j][k];
394 center /= 4.;
395 double sqDistance;
396 Point3f distVector = charucoCorner - center;
397 sqDistance = distVector.x * distVector.x + distVector.y * distVector.y;
398 if(j == 0 || fabs(sqDistance - minDist) < cv::pow(0.01 * squareLength, 2)) {
399 // if same minimum distance (or first iteration), add to nearestMarkerIdx vector
400 nearestMarkerIdx[i].push_back(j);
401 minDist = sqDistance;
402 } else if(sqDistance < minDist) {
403 // if finding a closest marker to the charuco corner
404 nearestMarkerIdx[i].clear(); // remove any previous added marker
405 nearestMarkerIdx[i].push_back(j); // add the new closest marker index
406 minDist = sqDistance;
407 }
408 }
409 // for each of the closest markers, search the marker corner index closer
410 // to the charuco corner
411 for(unsigned int j = 0; j < nearestMarkerIdx[i].size(); j++) {
412 nearestMarkerCorners[i].resize(nearestMarkerIdx[i].size());
413 double minDistCorner = -1;
414 for(unsigned int k = 0; k < 4; k++) {
415 double sqDistance;
416 Point3f distVector = charucoCorner - objPoints[nearestMarkerIdx[i][j]][k];
417 sqDistance = distVector.x * distVector.x + distVector.y * distVector.y;
418 if(k == 0 || sqDistance < minDistCorner) {
419 // if this corner is closer to the charuco corner, assing its index
420 // to nearestMarkerCorners
421 minDistCorner = sqDistance;
422 nearestMarkerCorners[i][j] = k;
423 }
424 }
425 }
426 }

429void CharucoBoardImpl::matchImagePoints(InputArrayOfArrays detectedCharuco, InputArray detectedIds,
430 OutputArray outObjPoints, OutputArray outImgPoints) const {
431 CV_CheckEQ(detectedIds.total(), detectedCharuco.total(), "Number of corners and ids must be equal");
432 CV_Assert(detectedIds.total() > 0ull);
433 CV_Assert(detectedCharuco.depth() == CV_32F);
434 // detectedCharuco includes charuco corners as vector<Point2f> or Mat.
435 // Python bindings could add extra dimension to detectedCharuco and therefore vector<Mat> case is additionally processed.
436 CV_Assert((detectedCharuco.isMat() || detectedCharuco.isVector() || detectedCharuco.isMatVector() || detectedCharuco.isUMatVector())
437 && detectedCharuco.depth() == CV_32F);
438 size_t nDetected = detectedCharuco.total();
439 vector<Point3f> objPnts(nDetected);
440 vector<Point2f> imgPnts(nDetected);
441 Mat detectedCharucoMat, detectedIdsMat = detectedIds.getMat();
442 if (!detectedCharuco.isMatVector()) {
443 detectedCharucoMat = detectedCharuco.getMat();
444 CV_Assert(detectedCharucoMat.checkVector(2));
445 }
446
447 std::vector<Mat> detectedCharucoVecMat;
448 if (detectedCharuco.isMatVector()) {
449 detectedCharuco.getMatVector(detectedCharucoVecMat);
450 }
451
452 for(size_t i = 0ull; i < nDetected; i++) {
453 int pointId = detectedIdsMat.at<int>((int)i);
454 CV_Assert(pointId >= 0 && pointId < (int)chessboardCorners.size());
455 objPnts[i] = chessboardCorners[pointId];
456 if (detectedCharuco.isMatVector()) {
457 CV_Assert((int)detectedCharucoVecMat[i].total() * detectedCharucoVecMat[i].channels() == 2);
458 imgPnts[i] = detectedCharucoVecMat[i].ptr<Point2f>(0)[0];
459 }
460 else
461 imgPnts[i] = detectedCharucoMat.ptr<Point2f>(0)[i];
462 }
463 Mat(objPnts).copyTo(outObjPoints);
464 Mat(imgPnts).copyTo(outImgPoints);
465}

467void CharucoBoardImpl::generateImage(Size outSize, OutputArray img, int marginSize, int borderBits) const {
468 CV_Assert(!outSize.empty());
469 CV_Assert(marginSize >= 0);
470
471 img.create(outSize, CV_8UC1);
472 img.setTo(255);
473 Mat out = img.getMat();
474 Mat noMarginsImg =
475 out.colRange(marginSize, out.cols - marginSize).rowRange(marginSize, out.rows - marginSize);
476
477 // the size of the chessboard square depends on the location of the chessboard
478 float pixInSquare = 0.f;
479 // the size of the chessboard in pixels
480 Size pixInChessboard(noMarginsImg.cols, noMarginsImg.rows);
481
482 // determine the zone where the chessboard is placed
483 float pixInSquareX = (float)noMarginsImg.cols / (float)size.width;
484 float pixInSquareY = (float)noMarginsImg.rows / (float)size.height;
485 Point startChessboard(0, 0);
486 if (pixInSquareX <= pixInSquareY) {
487 // the width of "noMarginsImg" image determines the dimensions of the chessboard
488 pixInSquare = pixInSquareX;
489 pixInChessboard.height = cvRound(pixInSquare*size.height);
490 int rowsMargin = (noMarginsImg.rows - pixInChessboard.height) / 2;
491 startChessboard.y = rowsMargin;
492 }
493 else {
494 // the height of "noMarginsImg" image determines the dimensions of the chessboard
495 pixInSquare = pixInSquareY;
496 pixInChessboard.width = cvRound(pixInSquare*size.width);
497 int colsMargin = (noMarginsImg.cols - pixInChessboard.width) / 2;
498 startChessboard.x = colsMargin;
499 }
500 // determine the zone where the chessboard is located
501 Mat chessboardZoneImg = noMarginsImg(Rect(startChessboard, pixInChessboard));
502
503 // marker size in pixels
504 const float pixInMarker = markerLength/squareLength*pixInSquare;
505 // the size of the marker margin in pixels
506 const float pixInMarginMarker = 0.5f*(pixInSquare - pixInMarker);
507
508 // determine the zone where the aruco markers are located
509 int endArucoX = cvRound(pixInSquare*(size.width-1)+pixInMarginMarker+pixInMarker);
510 int endArucoY = cvRound(pixInSquare*(size.height-1)+pixInMarginMarker+pixInMarker);
511 Mat arucoZone = chessboardZoneImg(Range(cvRound(pixInMarginMarker), endArucoY), Range(cvRound(pixInMarginMarker), endArucoX));
512
513 // draw markers
514 Board::Impl::generateImage(arucoZone.size(), arucoZone, 0, borderBits);
515
516 // now draw black squares
517 for(int y = 0; y < size.height; y++) {
518 for(int x = 0; x < size.width; x++) {
519
520 if(legacyPattern && (size.height % 2 == 0)) { // legacy behavior only for even row count patterns
521 if((y + 1) % 2 != x % 2) continue; // white corner, don't do anything
522 } else {
523 if(y % 2 != x % 2) continue; // white corner, don't do anything
524 }
525
526 float startX = pixInSquare * float(x);
527 float startY = pixInSquare * float(y);
528
529 Mat squareZone = chessboardZoneImg(Range(cvRound(startY), cvRound(startY + pixInSquare)),
530 Range(cvRound(startX), cvRound(startX + pixInSquare)));
531
532 squareZone.setTo(0);
533 }
534 }
535}

584bool CharucoBoard::checkCharucoCornersCollinear(InputArray charucoIds) const {
585 CV_Assert(impl);
586 Mat charucoIdsMat = charucoIds.getMat();
587
588 unsigned int nCharucoCorners = (unsigned int)charucoIdsMat.total();
589 if (nCharucoCorners <= 2)
590 return true;
591
592 // only test if there are 3 or more corners
593 auto board = static_pointer_cast<CharucoBoardImpl>(impl);
594 CV_Assert(board->chessboardCorners.size() >= charucoIdsMat.total());
595
596 Vec<double, 3> point0(board->chessboardCorners[charucoIdsMat.at<int>(0)].x,
597 board->chessboardCorners[charucoIdsMat.at<int>(0)].y, 1);
598
599 Vec<double, 3> point1(board->chessboardCorners[charucoIdsMat.at<int>(1)].x,
600 board->chessboardCorners[charucoIdsMat.at<int>(1)].y, 1);
601
602 // create a line from the first two points.
603 Vec<double, 3> testLine = point0.cross(point1);
604 Vec<double, 3> testPoint(0, 0, 1);
605
