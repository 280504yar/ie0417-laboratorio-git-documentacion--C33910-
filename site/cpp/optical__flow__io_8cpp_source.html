48const float FLOW_TAG_FLOAT = 202021.25f;
49const char *FLOW_TAG_STRING = "PIEH";
50CV_EXPORTS_W Mat readOpticalFlow( const String& path )
51{
52 Mat_<Point2f> flow;
53 std::ifstream file(path.c_str(), std::ios_base::binary);
54 if ( !file.good() )
55 return Mat(); // no file - return empty matrix
56
57 float tag;
58 file.read((char*) &tag, sizeof(float));
59 if ( tag != FLOW_TAG_FLOAT )
60 return Mat();
61
62 int width, height;
63
64 file.read((char*) &width, 4);
65 file.read((char*) &height, 4);
66
67 flow.create(height, width);
68
69 for ( int i = 0; i < flow.rows; ++i )
70 {
71 for ( int j = 0; j < flow.cols; ++j )
72 {
73 Point2f u;
74 file.read((char*) &u.x, sizeof(float));
75 file.read((char*) &u.y, sizeof(float));
76 if ( !file.good() )
77 {
78 flow.release();
79 return Mat();
80 }
81
82 flow(i, j) = u;
83 }
84 }
85 file.close();
86 return Mat(flow);
87}

89CV_EXPORTS_W bool writeOpticalFlow( const String& path, InputArray flow )
90{
91 const int nChannels = 2;
92
93 Mat input = flow.getMat();
94 if ( input.channels() != nChannels || input.depth() != CV_32F || path.length() == 0 )
95 return false;
96
97 std::ofstream file(path.c_str(), std::ofstream::binary);
98 if ( !file.good() )
99 return false;
100
101 int nRows, nCols;
102
103 nRows = (int) input.size().height;
104 nCols = (int) input.size().width;
105
106 const int headerSize = 12;
107 char header[headerSize];
108 memcpy(header, FLOW_TAG_STRING, 4);
109 // size of ints is known - has been asserted in the current function
110 memcpy(header + 4, reinterpret_cast<const char*>(&nCols), sizeof(nCols));
111 memcpy(header + 8, reinterpret_cast<const char*>(&nRows), sizeof(nRows));
112 file.write(header, headerSize);
113 if ( !file.good() )
114 return false;
115
116 int row;
117 char* p;
118 for ( row = 0; row < nRows; row++ )
119 {
120 p = input.ptr<char>(row);
121 file.write(p, nCols * nChannels * sizeof(float));
122 if ( !file.good() )
123 return false;
124 }
125 file.close();
126 return true;
127}
