109struct BlobManager
110{
111public:
112 // Increase references counter to layer output.
113 void addReference(const LayerPin& lp)
114 {
115 std::map<LayerPin, int>::iterator it = refCounter.find(lp);
116 if (it == refCounter.end())
117 refCounter[lp] = 1;
118 else
119 it->second += 1;
120 }
121
122 void addReferences(const std::vector<LayerPin>& pins)
123 {
124 for (int i = 0; i < pins.size(); i++)
125 {
126 addReference(pins[i]);
127 }
128 }
129
130 // Returns number of references to allocated memory that used in specific
131 // layer blob.
132 int numReferences(const LayerPin& lp)
133 {
134 std::map<LayerPin, LayerPin>::const_iterator mapIt = reuseMap.find(lp);
135 CV_Assert(mapIt != reuseMap.end());
136 LayerPin memHost = mapIt->second;
137
138 std::map<LayerPin, int>::const_iterator refIt = refCounter.find(memHost);
139 CV_Assert(refIt != refCounter.end());
140 return refIt->second;
141 }
142
143 // Reuse data allocated in <host> inside the <user> blob.
144 void reuse(const LayerPin& host, const LayerPin& user)
145 {
146 CV_Assert(reuseMap.find(user) == reuseMap.end());
147 CV_Assert(reuseMap.find(host) != reuseMap.end());
148 LayerPin memHost = reuseMap[host];
149 reuseMap[user] = memHost;
150 if (refCounter.find(memHost) != refCounter.end())
151 {
152 std::map<LayerPin, int>::iterator userRefIt = refCounter.find(user);
153 if (userRefIt != refCounter.end())
154 {
155 refCounter[memHost] += userRefIt->second;
156 refCounter.erase(userRefIt);
157 }
158 else
159 refCounter[memHost] += 1;
160 }
161 }
162
163 // Decrease references counter to allocated memory inside specific blob.
164 void releaseReference(const LayerPin& lp)
165 {
166 std::map<LayerPin, LayerPin>::const_iterator mapIt = reuseMap.find(lp);
167 CV_Assert(mapIt != reuseMap.end());
168
169 std::map<LayerPin, int>::iterator refIt = refCounter.find(mapIt->second);
170 CV_Assert(refIt != refCounter.end());
171 CV_Assert(refIt->second > 0);
172 refIt->second -= 1;
173 }
174
175 void releaseReferences(const std::vector<LayerPin>& pins)
176 {
177 for (int i = 0; i < pins.size(); i++)
178 {
179 releaseReference(pins[i]);
180 }
181 }
182
183 void reuseOrCreate(const MatShape& shape, const LayerPin& lp, Mat& dst, const int& dtype)
184 {
185 if (!getParam_DNN_DISABLE_MEMORY_OPTIMIZATIONS())
186 {
187 Mat bestBlob;
188 LayerPin bestBlobPin;
189
190 std::map<LayerPin, Mat>::const_iterator hostIt;
191 std::map<LayerPin, int>::const_iterator refIt;
192
193 const int targetTotal = (int)total(shape);
194 size_t bestBlobTotal = INT_MAX;
195
196 for (hostIt = memHosts.begin(); hostIt != memHosts.end(); ++hostIt)
197 {
198 refIt = refCounter.find(hostIt->first);
199 // Use only blobs that had references before because if not,
200 // it might be used as output.
201 if (refIt != refCounter.end() && refIt->second == 0)
202 {
203 const Mat& unusedBlob = hostIt->second;
204 if (unusedBlob.total() >= targetTotal && unusedBlob.total() < bestBlobTotal && unusedBlob.type() == dtype)
205 {
206 bestBlobPin = hostIt->first;
207 bestBlob = unusedBlob;
208 bestBlobTotal = unusedBlob.total();
209 }
210 }
211 }
212 if (!bestBlob.empty())
213 {
214 reuse(bestBlobPin, lp);
215 dst = bestBlob.reshape(1, 1).colRange(0, targetTotal).reshape(1, shape);
216 dst.size.dims = dst.dims = shape.size();
217 return;
218 }
219 }
220
221 {
222 // if dst already has been allocated with total(shape) elements,
223 // it won't be recreated and pointer of dst.data remains the same.
224 dst.create(shape, dtype);
225 addHost(lp, dst);
226 }
227 }

229 void allocateBlobsForLayer(LayerData& ld, const LayerShapes& layerShapes,
230 std::vector<LayerPin>& pinsForInternalBlobs)
231 {
232 CV_TRACE_FUNCTION();
233
234 pinsForInternalBlobs.clear();
235
236 std::vector<Mat>&outputBlobs = ld.outputBlobs,
237 &internalBlobs = ld.internals;
238
239 const ShapesVec &outShapes = layerShapes.out,
240 internalShapes = layerShapes.internal;
241 const TypesVec &outTypes = layerShapes.outTypes,
242 &internalTypes = layerShapes.internalTypes;
243 CV_CheckEQ(outShapes.size(), outTypes.size(), "Numbers shapes and types shoud be equal");
244 CV_CheckEQ(internalShapes.size(), internalTypes.size(), "Numbers shapes and types shoud be equal");
245
246 outputBlobs.resize(std::max((size_t)1, outShapes.size())); // layer produce at least one output blob
247 internalBlobs.resize(internalShapes.size());
248
249 CV_Assert(ld.requiredOutputs.size() <= outShapes.size());
250
251 // Check that layer could work in-place.
252 bool inPlace = false;
253 if (layerShapes.supportInPlace)
254 {
255 if (ld.inputBlobs.size() == 1)
256 {
257 // Get number of references to the input memory.
258 int numRef = numReferences(ld.inputBlobsId[0]);
259 // If current layer is one and only customer of this blob.
260 inPlace = numRef == 1;
261 }
262 }
263
264 ShapesVec shapes(outShapes);
265 TypesVec types(outTypes);
266 shapes.insert(shapes.end(), internalShapes.begin(), internalShapes.end());
267 types.insert(types.end(), internalTypes.begin(), internalTypes.end());
268 std::vector<Mat*> blobs;
269 for (int i = 0; i < outputBlobs.size(); i++)
270 {
271 blobs.push_back(&outputBlobs[i]);
272 }
273
274 for (int i = 0; i < internalBlobs.size(); i++)
275 {
276 blobs.push_back(&internalBlobs[i]);
277 if (total(internalShapes[i]))
278 {
279 pinsForInternalBlobs.push_back(LayerPin(ld.id, ld.outputBlobs.size() + i));
280 }
281 }
282
283 addReferences(pinsForInternalBlobs);
284
285 std::map<int, std::vector<int>> idxSizes;
286 for (int i = 0; i < shapes.size(); i++)
287 {
288 idxSizes[total(shapes[i])].push_back(i);
289 }
290
291 std::map<int, std::vector<int>>::reverse_iterator it;
292 for (it = idxSizes.rbegin(); it != idxSizes.rend(); it++)
293 {
294 for (int j = 0; j < it->second.size(); j++)
295 {
296 int index = it->second[j];
297 if (total(shapes[index]))
298 {
299 LayerPin blobPin(ld.id, index);
300 if (index < outShapes.size() && inPlace)
301 {
302 CV_CheckEQ(ld.inputBlobs[0]->total(), total(shapes[index]), "");
303 CV_CheckTypeEQ(ld.inputBlobs[0]->type(), types[index], "blob can't be reused if it has different type");
304 ld.outputBlobs[index] = ld.inputBlobs[0]->reshape(1, shapes[index]);
305 reuse(ld.inputBlobsId[0], blobPin);
306 }
307 else
308 reuseOrCreate(shapes[index], blobPin, *blobs[index], types[index]);
309 }
310 }
311 }
312 }

324private:
325 // Register allocated memory.
326 void addHost(const LayerPin& lp, const Mat& mat)
327 {
328 CV_Assert(memHosts.find(lp) == memHosts.end());

335 // For origin blobs key == value.
336 std::map<LayerPin, LayerPin> reuseMap;
337 std::map<LayerPin, Mat> memHosts;
338}; // BlobManager
