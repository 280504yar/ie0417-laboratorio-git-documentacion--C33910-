41static inline float rotatedRectIOU(const RotatedRect& a, const RotatedRect& b)
42{
43 std::vector<Point2f> inter;
44 int res = rotatedRectangleIntersection(a, b, inter);
45 if (inter.empty() || res == INTERSECT_NONE)
46 return 0.0f;
47 if (res == INTERSECT_FULL)
48 return 1.0f;
49 float interArea = contourArea(inter);
50 return interArea / (a.size.area() + b.size.area() - interArea);
51}

62template<class Rect_t>
63static inline void NMSBoxesBatchedImpl(const std::vector<Rect_t>& bboxes,
64 const std::vector<float>& scores, const std::vector<int>& class_ids,
65 const float score_threshold, const float nms_threshold,
66 std::vector<int>& indices, const float eta, const int top_k)
67{
68 double x1, y1, x2, y2, max_coord = 0;
69 for (int i = 0; i < bboxes.size(); i++)
70 {
71 x1 = bboxes[i].x;
72 y1 = bboxes[i].y;
73 x2 = x1 + bboxes[i].width;
74 y2 = y1 + bboxes[i].height;
75
76 max_coord = std::max(x1, max_coord);
77 max_coord = std::max(y1, max_coord);
78 max_coord = std::max(x2, max_coord);
79 max_coord = std::max(y2, max_coord);
80 }
81
82 // calculate offset and add offset to each bbox
83 std::vector<Rect_t> bboxes_offset;
84 double offset;
85 for (int i = 0; i < bboxes.size(); i++)
86 {
87 offset = class_ids[i] * (max_coord + 1);
88 bboxes_offset.push_back(
89 Rect_t(bboxes[i].x + offset, bboxes[i].y + offset,
90 bboxes[i].width, bboxes[i].height)
91 );
92 }
93
94 NMSFast_(bboxes_offset, scores, score_threshold, nms_threshold, eta, top_k, indices, rectOverlap);
95}

117void softNMSBoxes(const std::vector<Rect>& bboxes,
118 const std::vector<float>& scores,
119 std::vector<float>& updated_scores,
120 const float score_threshold,
121 const float nms_threshold,
122 std::vector<int>& indices,
123 size_t top_k,
124 const float sigma,
125 SoftNMSMethod method)
126{
127 CV_Assert_N(bboxes.size() == scores.size(), score_threshold >= 0,
128 nms_threshold >= 0, sigma >= 0);
129
130 indices.clear();
131 updated_scores.clear();
132
133 std::vector<std::pair<float, size_t> > score_index_vec(scores.size());
134 for (size_t i = 0; i < scores.size(); i++)
135 {
136 score_index_vec[i].first = scores[i];
137 score_index_vec[i].second = i;
138 }
139
140 const auto score_cmp = [](const std::pair<float, size_t>& a, const std::pair<float, size_t>& b)
141 {
142 return a.first == b.first ? a.second > b.second : a.first < b.first;
143 };
144
145 top_k = top_k == 0 ? scores.size() : std::min(top_k, scores.size());
146 ptrdiff_t start = 0;
147 while (indices.size() < top_k)
148 {
149 auto it = std::max_element(score_index_vec.begin() + start, score_index_vec.end(), score_cmp);
150
151 float bscore = it->first;
152 size_t bidx = it->second;
153
154 if (bscore < score_threshold)
155 {
156 break;
157 }
158
159 indices.push_back(static_cast<int>(bidx));
160 updated_scores.push_back(bscore);
161 std::swap(score_index_vec[start], *it); // first start elements are chosen
162
163 for (size_t i = start + 1; i < scores.size(); ++i)
164 {
165 float& bscore_i = score_index_vec[i].first;
166 const size_t bidx_i = score_index_vec[i].second;
167
168 if (bscore_i < score_threshold)
169 {
170 continue;
171 }
172
173 float overlap = rectOverlap(bboxes[bidx], bboxes[bidx_i]);
174
175 switch (method)
176 {
177 case SoftNMSMethod::SOFTNMS_LINEAR:
178 if (overlap > nms_threshold)
179 {
180 bscore_i *= 1.f - overlap;
181 }
182 break;
183 case SoftNMSMethod::SOFTNMS_GAUSSIAN:
184 bscore_i *= exp(-(overlap * overlap) / sigma);
185 break;
186 default:
187 CV_Error(Error::StsBadArg, "Not supported SoftNMS method.");
188 }
189 }
190 ++start;
191 }
192}
