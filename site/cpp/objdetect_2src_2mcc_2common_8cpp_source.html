36Rect poly2mask(const std::vector<Point2f> &poly, Size size, InputOutputArray mask)
37{
38 // Create Polygon from vertices
39 std::vector<Point> roi_poly;
40 approxPolyDP(poly, roi_poly, 1.0, true);
41
42 Rect roi = boundingRect(roi_poly);
43
44 // Fill polygon white
45 fillConvexPoly(mask, &roi_poly[0], (int)roi_poly.size(), 1, 8, 0);
46
47 roi &= Rect(0, 0, size.width, size.height);
48 if (roi.empty())
49 roi = Rect(0, 0, 1, 1);
50 return roi;
51}

53float perimeter(const std::vector<Point2f> &ps)
54{
55 float sum = 0, dx, dy;
56
57 for (size_t i = 0; i < ps.size(); i++)
58 {
59 int i2 = (i + 1) % (int)ps.size();
60
61 dx = ps[i].x - ps[i2].x;
62 dy = ps[i].y - ps[i2].y;
63
64 sum += sqrt(dx * dx + dy * dy);
65 }
66
67 return sum;
68}

85void polyanticlockwise(std::vector<Point2f> &points)
86{
87 // Sort the points in anti-clockwise order
88 // Trace a line between the first and second point.
89 // If the third point is at the right side, then the points are anti-clockwise
90 Point2f v1 = points[1] - points[0];
91 Point2f v2 = points[2] - points[0];
92
93 //if the third point is in the left side, then sort in anti-clockwise order
94 if ((v1.x * v2.y) - (v1.y * v2.x) < 0.0)
95 std::swap(points[1], points[3]);
96}
