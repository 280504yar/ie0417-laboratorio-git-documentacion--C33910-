20 MatMulInt8LayerImpl(const MatMulInt8Params& p)
21 {
22 name = p.name;
23 type = "MatMulInt8";
24 input_sc = p.input_sc;
25 input_zp = p.input_zp;
26 output_sc = p.output_sc;
27 output_zp = p.output_zp;
28 output_type = p.output_type;
29 per_channel = p.per_channel;
30 inp_dims = p.inp_dims;
31 num_output = p.num_output;
32
33 if (!p.weights.empty()) {
34 int numOutput = p.num_output;
35 int innerSize = (int)p.weights.total() / numOutput;
36
37 CV_Assert(p.weights.dims >= 2 && (size_t)(innerSize * numOutput) == p.weights.total());
38 CV_Assert((size_t)numOutput == p.bias.total());
39
40 blobs = { p.weights.clone(), p.bias.clone(), p.outputMultiplier.clone() };
41 weightsMat = blobs[0] = blobs[0].reshape(1, numOutput);
42 int vecsize = weightsMat.cols;
43 if (vecsize % VEC_ALIGN != 0)
44 {
45 int vecsize_aligned = (int)alignSize(vecsize, VEC_ALIGN);
46 Mat weightsBuf(weightsMat.rows, vecsize_aligned, weightsMat.type());
47 Mat wpadding = weightsBuf.colRange(vecsize, vecsize_aligned);
48 wpadding.setTo(Scalar::all(0));
49 weightsMat = weightsBuf.colRange(0, vecsize);
50 blobs[0].copyTo(weightsMat);
51 }
52 biasMat = blobs[1] = blobs[1].reshape(1, 1);
53 outputMultiplier = blobs[2];
54 }
55 }

93 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr,
94 OutputArrayOfArrays internals_arr) CV_OVERRIDE
95 {
96 CV_TRACE_FUNCTION();
97 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
98
99 std::vector<Mat> input, output;
100 inputs_arr.getMatVector(input);
101 outputs_arr.getMatVector(output);
102
103 // Flatten all dimensions except the last (K) into a single batch dim.
104 // Input: [B1, B2, ..., M, K] → [B1*B2*...*M, K]
105 int ndims = input[0].dims;
106 int K = input[0].size[ndims - 1];
107 int outerSize = (int)(input[0].total() / K);
108
109 Mat srcMat0 = input[0].reshape(1, outerSize);
110 Mat srcMat;
111 if (srcMat0.type() == CV_8U) {
112 srcMat0.convertTo(srcMat, CV_8S, 1, -128);
113 } else {
114 srcMat = srcMat0;
115 }
116
117 Mat dstMat = output[0].reshape(1, outerSize);
118 Mat dstMatInt32 = Mat(shape(dstMat), CV_32S);
119
120 const int nstripes = outerSize <= 4 ? 1 : getNumThreads();
121 runInt8Gemm(srcMat, weightsMat, biasMat, outputMultiplier, dstMatInt32, nstripes, output_zp);
122 if (output_type == CV_8U) {
123 dstMatInt32.convertTo(dstMat, output_type, 1, 128);
124 } else {
125 dstMatInt32.convertTo(dstMat, output_type);
126 }
127 }

129 virtual int64 getFLOPS(const std::vector<MatShape> &inputs,
130 const std::vector<MatShape> &outputs) const CV_OVERRIDE
131 {
132 CV_UNUSED(inputs);
133 long flops = 0;

136 flops += total(outputs[i]) * innerSize;

141private:
142 // Int8 GEMM: C_int32 = A_int8 @ W_int8^T + bias_int32
143 // Then: output = clamp(round(C_int32 * multiplier) + output_zp)
144 static void runInt8Gemm(const Mat& srcMat, const Mat& weights, const Mat& biasMat,
145 const Mat& outputMultiplier, Mat& dstMat, int nstripes, int outZp)
146 {
147 CV_Assert(srcMat.dims == 2 && srcMat.cols == weights.cols &&
148 dstMat.rows == srcMat.rows && dstMat.cols == weights.rows &&
149 srcMat.type() == weights.type() && srcMat.type() == CV_8S &&
150 dstMat.type() == CV_32S && biasMat.type() == CV_32S &&
151 biasMat.isContinuous() && (int)biasMat.total() == dstMat.cols);
152
153 Int8GemmBody p;
154 p.srcMat = &srcMat;
155 p.weights = &weights;
156 p.biasMat = &biasMat;
157 p.outputMultiplier = &outputMultiplier;
158 p.dstMat = &dstMat;
159 p.nstripes = nstripes;
160 p.outZp = outZp;
161 p.useAVX2 = checkHardwareSupport(CPU_AVX2);
162 p.useAVX512 = CV_CPU_HAS_SUPPORT_AVX512_SKX;
163 p.useLASX = checkHardwareSupport(CPU_LASX);
164 p.useRVV = checkHardwareSupport(CPU_RVV);
165
166 parallel_for_(Range(0, nstripes), p, nstripes);
167 }

169 class Int8GemmBody : public ParallelLoopBody
170 {
171 public:
172 Int8GemmBody() : srcMat(0), weights(0), biasMat(0), outputMultiplier(0),
173 dstMat(0), nstripes(0), outZp(0),
174 useAVX2(false), useAVX512(false), useLASX(false), useRVV(false) {}

176 void operator()(const Range& r) const CV_OVERRIDE
177 {
178 int valign = MatMulInt8LayerImpl::VEC_ALIGN;
179 int nsamples = srcMat->rows;
180 int nw0 = weights->rows;
181 int k, vecsize = srcMat->cols;
182 int vecsize_aligned = (int)alignSize(vecsize, VEC_ALIGN);
183 size_t total = (size_t)nsamples * nw0;
184 size_t stripeSize = (total + nstripes - 1) / nstripes;
185 size_t stripeStart = r.start * stripeSize;
186 size_t stripeEnd = r.end == nstripes ? total : std::min(r.end * stripeSize, total);
187 size_t wstep = weights->step1();
188 AutoBuffer<int8_t> srcbuf(vecsize_aligned + valign);
189 int8_t* sptr = alignPtr(srcbuf.data(), (int)(valign * sizeof(int8_t)));
190
191 for (k = vecsize; k < vecsize_aligned; k++)
192 sptr[k] = 0;
193
194 for (size_t ofs = stripeStart; ofs < stripeEnd; )
195 {
196 int sampleIdx = (int)(ofs / nw0);
197 int delta = (int)(ofs - (size_t)sampleIdx * nw0);
198 const int8_t* sptr_ = srcMat->ptr<int8_t>(sampleIdx);
199 const int8_t* wptr = weights->ptr<int8_t>(delta);
200 int* dptr = dstMat->ptr<int>(sampleIdx) + delta;
201 const int* biasptr = biasMat->ptr<int>() + delta;
202 const float* multptr = outputMultiplier->ptr<float>() + delta;
203 int nw = std::min(nw0 - delta, (int)(stripeEnd - ofs));
204
205 memcpy(sptr, sptr_, vecsize * sizeof(sptr[0]));
206 #if CV_TRY_AVX512_SKX
207 if (useAVX512)
208 opt_AVX512_SKX::fastGEMM1T(sptr, wptr, wstep, biasptr, multptr, dptr, nw, vecsize, outZp);
209 else
210 #endif
211 #if CV_TRY_AVX2
212 if (useAVX2)
213 opt_AVX2::fastGEMM1T(sptr, wptr, wstep, biasptr, multptr, dptr, nw, vecsize, outZp);
214 else
215 #endif
216 #if CV_TRY_LASX
217 if (useLASX)
218 opt_LASX::fastGEMM1T(sptr, wptr, wstep, biasptr, multptr, dptr, nw, vecsize, outZp);
219 else
220 #endif
221 #if CV_TRY_RVV && CV_RVV
222 if (useRVV)
223 opt_RVV::fastGEMM1T(sptr, wptr, wstep, biasptr, multptr, dptr, nw, vecsize, outZp);
224 else
225 #endif
226 #if CV_RVP052
227 if (1)
228 opt_RVP052::fastGEMM1T(sptr, wptr, wstep, biasptr, multptr, dptr, nw, vecsize, outZp);
229 else
230 #endif
231 {
232 int i = 0;
233 #if CV_SIMD128
234 for (; i <= nw - 4; i += 4, wptr += 4 * wstep)
235 {
236 v_int32x4 vs0 = v_setzero_s32(), vs1 = v_setzero_s32(),
237 vs2 = v_setzero_s32(), vs3 = v_setzero_s32();
238 v_int32x4 outzp = v_setall_s32(outZp), outmin = v_setall_s32(-128), outmax = v_setall_s32(127);
239 v_int32x4 s = v_load(biasptr + i);
240 v_float32x4 mult = v_load(multptr + i);
241
242 for (k = 0; k < vecsize; k += 16)
243 {
244 v_int8x16 v = v_load_aligned(sptr + k);
245 vs0 = v_dotprod_expand_fast(v, v_load_aligned(wptr + k), vs0);
246 vs1 = v_dotprod_expand_fast(v, v_load_aligned(wptr + wstep + k), vs1);
247 vs2 = v_dotprod_expand_fast(v, v_load_aligned(wptr + wstep * 2 + k), vs2);
248 vs3 = v_dotprod_expand_fast(v, v_load_aligned(wptr + wstep * 3 + k), vs3);
249 }
250
251 s = v_add(s, v_int32x4(v_reduce_sum(vs0), v_reduce_sum(vs1), v_reduce_sum(vs2), v_reduce_sum(vs3)));
252 v_int32x4 out = v_add(outzp, v_round(v_mul(v_cvt_f32(s), mult)));
253 v_store(dptr + i, v_min(v_max(out, outmin), outmax));
254 }
255 #endif
256
257 for (; i < nw; i++, wptr += wstep)
258 {
259 int s0 = biasptr[i];
260 float mult0 = multptr[i];
261
262 for (k = 0; k < vecsize; k++)
263 {
264 int8_t v = sptr[k];
265 s0 += (int)v * wptr[k];
266 }
267 int out0 = outZp + (int)std::round(s0 * mult0);
268 dptr[i] = std::min(std::max(out0, -128), 127);
269 }
270 }
271
272 ofs += nw;
273 }
274 }

281 bool useLASX;
282 bool useRVV;
