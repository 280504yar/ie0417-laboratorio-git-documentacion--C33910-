51void RBaseStream::readBlock()
52{
53 setPos( getPos() ); // normalize position
54
55 if( m_file == 0 )
56 {
57 if( m_block_pos == 0 && m_current < m_end )
58 return;
59 throw RBS_THROW_EOS;
60 }
61
62#ifdef _WIN32
63 // See https://learn.microsoft.com/en-us/cpp/c-runtime-library/reference/fseek-fseeki64?view=msvc-170
64 // See https://en.wikipedia.org/wiki/64-bit_computing#64-bit_data_models
65 // - Windows uses LLP64 data model, sizeof long int = 32.
66 // - Linux uses LP64 data model, sizeof long int = 64.
67 // So for Windows, we have to use _fseeki64() instead of fseek().
68 _fseeki64( m_file, m_block_pos, SEEK_SET );
69#else
70 fseek( m_file, m_block_pos, SEEK_SET );
71#endif
72
73 size_t readed = fread( m_start, 1, m_block_size, m_file );
74 m_end = m_start + readed;
75
76 if( readed == 0 || m_current >= m_end )
77 throw RBS_THROW_EOS;
78}

133void RBaseStream::setPos( int64_t pos )
134{
135 CV_Assert(isOpened() && pos >= 0);
136
137 if( !m_file )
138 {
139 m_current = m_start + pos;
140 m_block_pos = 0;
141 return;
142 }
143
144 int offset = pos % m_block_size;
145 int64_t old_block_pos = m_block_pos;
146 m_block_pos = pos - offset;
147 m_current = m_start + offset;
148 if (old_block_pos != m_block_pos)
149 readBlock();
150}

195int RLByteStream::getBytes( void* buffer, int count )
196{
197 uchar* data = (uchar*)buffer;
198 int readed = 0;
199 CV_Assert(count >= 0);
200
201 while( count > 0 )
202 {
203 int l;
204
205 for(;;)
206 {
207 l = (int)(m_end - m_current);
208 if( l > count ) l = count;
209 if( l > 0 ) break;
210 readBlock();
211 }
212 memcpy( data, m_current, l );
213 m_current += l;
214 data += l;
215 count -= l;
216 readed += l;
217 }
218 return readed;
219}

347bool WBaseStream::writeBlock()
348{
349 int size = (int)(m_current - m_start);
350
351 CV_Assert(isOpened());
352 if( size == 0 )
353 return true;
354
355 if( m_buf )
356 {
357 size_t sz = m_buf->size();
358 m_buf->resize( sz + size );
359 memcpy( &(*m_buf)[sz], m_start, size );
360 m_current = m_start;
361 m_block_pos += size;
362 return true;
363 }
364 else
365 {
366 size_t written = fwrite( m_start, 1, size, m_file );
367 m_current = m_start;
368 m_block_pos += size;
369 return written == (size_t)size;
370 }
371}
