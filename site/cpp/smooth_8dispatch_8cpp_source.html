78static
79softdouble getGaussianKernelBitExact(std::vector<softdouble>& result, int n, double sigma)
80{
81 CV_Assert(n > 0);
82 //TODO: incorrect SURF implementation requests kernel with n = 20 (PATCH_SZ): https://github.com/opencv/opencv/issues/15856
83 //CV_Assert((n & 1) == 1); // odd
84
85 if (sigma <= 0)
86 {
87 if (n == 1)
88 {
89 result = std::vector<softdouble>(1, softdouble::one());
90 return softdouble::one();
91 }
92 else if (n == 3)
93 {
94 softdouble v3[] = {
95 softdouble::fromRaw(0x3fd0000000000000), // 0.25
96 softdouble::fromRaw(0x3fe0000000000000), // 0.5
97 softdouble::fromRaw(0x3fd0000000000000) // 0.25
98 };
99 result.assign(v3, v3 + 3);
100 return softdouble::one();
101 }
102 else if (n == 5)
103 {
104 softdouble v5[] = {
105 softdouble::fromRaw(0x3fb0000000000000), // 0.0625
106 softdouble::fromRaw(0x3fd0000000000000), // 0.25
107 softdouble::fromRaw(0x3fd8000000000000), // 0.375
108 softdouble::fromRaw(0x3fd0000000000000), // 0.25
109 softdouble::fromRaw(0x3fb0000000000000) // 0.0625
110 };
111 result.assign(v5, v5 + 5);
112 return softdouble::one();
113 }
114 else if (n == 7)
115 {
116 softdouble v7[] = {
117 softdouble::fromRaw(0x3fa0000000000000), // 0.03125
118 softdouble::fromRaw(0x3fbc000000000000), // 0.109375
119 softdouble::fromRaw(0x3fcc000000000000), // 0.21875
120 softdouble::fromRaw(0x3fd2000000000000), // 0.28125
121 softdouble::fromRaw(0x3fcc000000000000), // 0.21875
122 softdouble::fromRaw(0x3fbc000000000000), // 0.109375
123 softdouble::fromRaw(0x3fa0000000000000) // 0.03125
124 };
125 result.assign(v7, v7 + 7);
126 return softdouble::one();
127 }
128 else if (n == 9)
129 {
130 softdouble v9[] = {
131 softdouble::fromRaw(0x3f90000000000000), // 4 / 256
132 softdouble::fromRaw(0x3faa000000000000), // 13 / 256
133 softdouble::fromRaw(0x3fbe000000000000), // 30 / 256
134 softdouble::fromRaw(0x3fc9800000000000), // 51 / 256
135 softdouble::fromRaw(0x3fce000000000000), // 60 / 256
136 softdouble::fromRaw(0x3fc9800000000000), // 51 / 256
137 softdouble::fromRaw(0x3fbe000000000000), // 30 / 256
138 softdouble::fromRaw(0x3faa000000000000), // 13 / 256
139 softdouble::fromRaw(0x3f90000000000000) // 4 / 256
140 };
141 result.assign(v9, v9 + 9);
142 return softdouble::one();
143 }
144 }
145
146 softdouble sd_0_15 = softdouble::fromRaw(0x3fc3333333333333); // 0.15
147 softdouble sd_0_35 = softdouble::fromRaw(0x3fd6666666666666); // 0.35
148 softdouble sd_minus_0_125 = softdouble::fromRaw(0xbfc0000000000000); // -0.5*0.25
149
150 softdouble sigmaX = sigma > 0 ? softdouble(sigma) : mulAdd(softdouble(n), sd_0_15, sd_0_35);// softdouble(((n-1)*0.5 - 1)*0.3 + 0.8)
151 softdouble scale2X = sd_minus_0_125/(sigmaX*sigmaX);
152
153 int n2_ = (n - 1) / 2;
154 cv::AutoBuffer<softdouble> values(n2_ + 1);
155 softdouble sum = softdouble::zero();
156 for (int i = 0, x = 1 - n; i < n2_; i++, x+=2)
157 {
158 // x = i - (n - 1)*0.5
159 // t = std::exp(scale2X*x*x)
160 softdouble t = exp(softdouble(x*x)*scale2X);
161 values[i] = t;
162 sum += t;
163 }
164 sum *= softdouble(2);
165 //values[n2_] = softdouble::one(); // x=0 in exp(softdouble(x*x)*scale2X);
166 sum += softdouble::one();
167 if ((n & 1) == 0)
168 {
169 //values[n2_ + 1] = softdouble::one();
170 sum += softdouble::one();
171 }
172
173 // normalize: sum(k[i]) = 1
174 softdouble mul1 = softdouble::one()/sum;
175
176 result.resize(n);
177
178 softdouble sum2 = softdouble::zero();
179 for (int i = 0; i < n2_; i++ )
180 {
181 softdouble t = values[i] * mul1;
182 result[i] = t;
183 result[n - 1 - i] = t;
184 sum2 += t;
185 }
186 sum2 *= softdouble(2);
187 result[n2_] = /*values[n2_]*/ softdouble::one() * mul1;
188 sum2 += result[n2_];
189 if ((n & 1) == 0)
190 {
191 result[n2_ + 1] = result[n2_];
192 sum2 += result[n2_];
193 }
194
195 return sum2;
196}

198Mat getGaussianKernel(int n, double sigma, int ktype)
199{
200 CV_CheckDepth(ktype, ktype == CV_32F || ktype == CV_64F, "");
201 Mat kernel(n, 1, ktype);
202
203 std::vector<softdouble> kernel_bitexact;
204 getGaussianKernelBitExact(kernel_bitexact, n, sigma);
205
206 if (ktype == CV_32F)
207 {
208 for (int i = 0; i < n; i++)
209 kernel.at<float>(i) = (float)kernel_bitexact[i];
210 }
211 else
212 {
213 CV_DbgAssert(ktype == CV_64F);
214 for (int i = 0; i < n; i++)
215 kernel.at<double>(i) = kernel_bitexact[i];
216 }
217
218 return kernel;
219}

221static
222softdouble getGaussianKernelFixedPoint_ED(CV_OUT std::vector<int64_t>& result, const std::vector<softdouble> kernel_bitexact, int fractionBits)
223{
224 const int n = (int)kernel_bitexact.size();
225 CV_Assert((n & 1) == 1); // odd
226
227 CV_CheckGT(fractionBits, 0, "");
228 CV_CheckLE(fractionBits, 32, "");
229
230 int64_t fractionMultiplier = CV_BIG_INT(1) << fractionBits;
231 softdouble fractionMultiplier_sd(fractionMultiplier);
232
233 result.resize(n);
234
235 int n2_ = n / 2; // n is odd
236 softdouble err = softdouble::zero();
237 int64_t sum = 0;
238 for (int i = 0; i < n2_; i++)
239 {
240 //softdouble err0 = err;
241 softdouble adj_v = kernel_bitexact[i] * fractionMultiplier_sd + err;
242 int64_t v0 = cvRound(adj_v); // cvFloor() provides bad results
243 err = adj_v - softdouble(v0);
244 //printf("%3d: adj_v=%8.3f(%8.3f+%8.3f) v0=%d ed_err=%8.3f\n", i, (double)adj_v, (double)(kernel_bitexact[i] * fractionMultiplier_sd), (double)err0, (int)v0, (double)err);
245
246 result[i] = v0;
247 result[n - 1 - i] = v0;
248 sum += v0;
249 }
250 sum *= 2;
251 softdouble adj_v_center = kernel_bitexact[n2_] * fractionMultiplier_sd + err;
252 int64_t v_center = fractionMultiplier - sum;
253 result[n2_] = v_center;
254 //printf("center = %g ===> %g ===> %g\n", (double)(kernel_bitexact[n2_] * fractionMultiplier), (double)adj_v_center, (double)v_center);
255 return (adj_v_center - softdouble(v_center));
256}

258static void getGaussianKernel(int n, double sigma, int ktype, Mat& res) { res = getGaussianKernel(n, sigma, ktype); }
259template <typename FT> static void getGaussianKernel(int n, double sigma, int, std::vector<FT>& res)
260{
261 std::vector<softdouble> res_sd;
262 softdouble s0 = getGaussianKernelBitExact(res_sd, n, sigma);
263 CV_UNUSED(s0);
264
265 std::vector<int64_t> fixed_256;
266 softdouble approx_err = getGaussianKernelFixedPoint_ED(fixed_256, res_sd, FT::fixedShift);
267 CV_UNUSED(approx_err);
268
269 res.resize(n);
270 for (int i = 0; i < n; i++)
271 {
272 res[i] = FT::fromRaw((typename FT::raw_t)fixed_256[i]);
273 //printf("%03d: %d\n", i, res[i].raw());
274 }
275}

277template <typename T>
278static void createGaussianKernels( T & kx, T & ky, int type, Size &ksize,
279 double sigma1, double sigma2 )
280{
281 int depth = CV_MAT_DEPTH(type);
282 if( sigma2 <= 0 )
283 sigma2 = sigma1;
284
285 // automatic detection of kernel size from sigma
286 if( ksize.width <= 0 && sigma1 > 0 )
287 ksize.width = cvRound(sigma1*(depth == CV_8U ? 3 : 4)*2 + 1)|1;
288 if( ksize.height <= 0 && sigma2 > 0 )
289 ksize.height = cvRound(sigma2*(depth == CV_8U ? 3 : 4)*2 + 1)|1;
290
291 CV_Assert( ksize.width > 0 && ksize.width % 2 == 1 &&
292 ksize.height > 0 && ksize.height % 2 == 1 );
293
294 sigma1 = std::max( sigma1, 0. );
295 sigma2 = std::max( sigma2, 0. );
296
297 getGaussianKernel( ksize.width, sigma1, std::max(depth, CV_32F), kx );
298 if( ksize.height == ksize.width && std::abs(sigma1 - sigma2) < DBL_EPSILON )
299 ky = kx;
300 else
301 getGaussianKernel( ksize.height, sigma2, std::max(depth, CV_32F), ky );
302}

304Ptr<FilterEngine> createGaussianFilter( int type, Size ksize,
305 double sigma1, double sigma2,
306 int borderType )
307{
308 Mat kx, ky;
309 createGaussianKernels(kx, ky, type, ksize, sigma1, sigma2);
310
311 return createSeparableLinearFilter( type, type, kx, ky, Point(-1,-1), 0, borderType );
312}

316static bool ocl_GaussianBlur_8UC1(InputArray _src, OutputArray _dst, Size ksize, int ddepth,
317 InputArray _kernelX, InputArray _kernelY, int borderType)
318{
319 const ocl::Device & dev = ocl::Device::getDefault();
320 int type = _src.type(), sdepth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type);
321
322 if ( !(dev.isIntel() && (type == CV_8UC1) &&
323 (_src.offset() == 0) && (_src.step() % 4 == 0) &&
324 ((ksize.width == 5 && (_src.cols() % 4 == 0)) ||
325 (ksize.width == 3 && (_src.cols() % 16 == 0) && (_src.rows() % 2 == 0)))) )
326 return false;
327
328 Mat kernelX = _kernelX.getMat().reshape(1, 1);
329 if (kernelX.cols % 2 != 1)
330 return false;
331 Mat kernelY = _kernelY.getMat().reshape(1, 1);
332 if (kernelY.cols % 2 != 1)
333 return false;
334
335 if (ddepth < 0)
336 ddepth = sdepth;
337
338 Size size = _src.size();
339 size_t globalsize[2] = { 0, 0 };
340 size_t localsize[2] = { 0, 0 };
341
342 if (ksize.width == 3)
343 {
344 globalsize[0] = size.width / 16;
345 globalsize[1] = size.height / 2;
346 }
347 else if (ksize.width == 5)
348 {
349 globalsize[0] = size.width / 4;
350 globalsize[1] = size.height / 1;
351 }
352
353 const char * const borderMap[] = { "BORDER_CONSTANT", "BORDER_REPLICATE", "BORDER_REFLECT", 0, "BORDER_REFLECT_101" };
354 char build_opts[1024];
355 snprintf(build_opts, sizeof(build_opts), "-D %s %s%s", borderMap[borderType & ~BORDER_ISOLATED],
356 ocl::kernelToStr(kernelX, CV_32F, "KERNEL_MATRIX_X").c_str(),
357 ocl::kernelToStr(kernelY, CV_32F, "KERNEL_MATRIX_Y").c_str());
358
359 ocl::Kernel kernel;
360
361 if (ksize.width == 3)
362 kernel.create("gaussianBlur3x3_8UC1_cols16_rows2", cv::ocl::imgproc::gaussianBlur3x3_oclsrc, build_opts);
363 else if (ksize.width == 5)
364 kernel.create("gaussianBlur5x5_8UC1_cols4", cv::ocl::imgproc::gaussianBlur5x5_oclsrc, build_opts);
365
366 if (kernel.empty())
367 return false;
368
369 UMat src = _src.getUMat();
370 _dst.create(size, CV_MAKETYPE(ddepth, cn));
371 if (!(_dst.offset() == 0 && _dst.step() % 4 == 0))
372 return false;
373 UMat dst = _dst.getUMat();
374
375 int idxArg = kernel.set(0, ocl::KernelArg::PtrReadOnly(src));
376 idxArg = kernel.set(idxArg, (int)src.step);
377 idxArg = kernel.set(idxArg, ocl::KernelArg::PtrWriteOnly(dst));
378 idxArg = kernel.set(idxArg, (int)dst.step);
379 idxArg = kernel.set(idxArg, (int)dst.rows);
380 idxArg = kernel.set(idxArg, (int)dst.cols);
381
382 return kernel.run(2, globalsize, (localsize[0] == 0) ? NULL : localsize, false);
383}

445static bool ipp_GaussianBlur(cv::Mat& src, cv::Mat& dst, Size ksize,
446 double sigma1, double sigma2, int borderType )
447{
448#ifdef HAVE_IPP_IW
449 CV_INSTRUMENT_REGION_IPP();
450
451#if IPP_VERSION_X100 < 201800 && ((defined _MSC_VER && defined _M_IX86) || (defined __GNUC__ && defined __i386__))
452 CV_UNUSED(src); CV_UNUSED(dst); CV_UNUSED(ksize); CV_UNUSED(sigma1); CV_UNUSED(sigma2); CV_UNUSED(borderType);
453 return false; // bug on ia32
454#else
455 if(sigma1 != sigma2)
456 return false;
457
458 if(sigma1 < FLT_EPSILON)
459 return false;

461 if(ksize.width != ksize.height)
462 return false;

464 // Acquire data and begin processing
465 try
466 {
467 ::ipp::IwiImage iwSrc = ippiGetImage(src);

469 ::ipp::IwiBorderSize borderSize = ::ipp::iwiSizeToBorderSize(ippiGetSize(ksize));

498 catch (const ::ipp::IwException &)

501 }
502
503 return true;
504#endif
505#else
506 CV_UNUSED(src); CV_UNUSED(dst); CV_UNUSED(ksize); CV_UNUSED(sigma1); CV_UNUSED(sigma2); CV_UNUSED(borderType);
507 return false;
508#endif

512template<typename T>
513static bool validateGaussianBlurKernel(std::vector<T>& kernel)
514{
515 softdouble validation_sum = softdouble::zero();
516 for (size_t i = 0; i < kernel.size(); i++)
517 {
518 validation_sum += softdouble((double)kernel[i]);
519 }
520
521 bool isValid = validation_sum == softdouble::one();
522 return isValid;
523}

525void GaussianBlur(InputArray _src, OutputArray _dst, Size ksize,
526 double sigma1, double sigma2,
527 int borderType, AlgorithmHint hint)
528{
529 CV_INSTRUMENT_REGION();
530
531 if (hint == cv::ALGO_HINT_DEFAULT)
532 hint = cv::getDefaultAlgorithmHint();
533
534 CV_Assert(!_src.empty());
535
536 int type = _src.type();
537 Size size = _src.size();
538 _dst.create( size, type );
539
540 if( (borderType & ~BORDER_ISOLATED) != BORDER_CONSTANT &&
541 ((borderType & BORDER_ISOLATED) != 0 || !_src.getMat().isSubmatrix()) )
542 {
543 if( size.height == 1 )
544 ksize.height = 1;
545 if( size.width == 1 )
546 ksize.width = 1;
547 }
548
549 if( ksize.width == 1 && ksize.height == 1 )
550 {
551 _src.copyTo(_dst);
552 return;
553 }
554
555 if (sigma2 <= 0)
556 sigma2 = sigma1;
557
558 bool useOpenCL = ocl::isOpenCLActivated() && _dst.isUMat() && _src.dims() <= 2 &&
559 _src.rows() >= ksize.height && _src.cols() >= ksize.width &&
560 ksize.width > 1 && ksize.height > 1;
561 CV_UNUSED(useOpenCL);
562
563 int sdepth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type);
564
565 Mat kx, ky;
566 createGaussianKernels(kx, ky, type, ksize, sigma1, sigma2);
567
568 CV_OCL_RUN(useOpenCL && sdepth == CV_8U &&
569 ((ksize.width == 3 && ksize.height == 3) ||
570 (ksize.width == 5 && ksize.height == 5)),
571 ocl_GaussianBlur_8UC1(_src, _dst, ksize, CV_MAT_DEPTH(type), kx, ky, borderType)
572 );
573
574 if(sdepth == CV_8U && ((borderType & BORDER_ISOLATED) || !_src.isSubmatrix()))
575 {
576 std::vector<ufixedpoint16> fkx, fky;
577 createGaussianKernels(fkx, fky, type, ksize, sigma1, sigma2);
578
579 static bool param_check_gaussian_blur_bitexact_kernels = utils::getConfigurationParameterBool("OPENCV_GAUSSIANBLUR_CHECK_BITEXACT_KERNELS", false);
580 if (param_check_gaussian_blur_bitexact_kernels && !validateGaussianBlurKernel(fkx))
581 {
582 CV_LOG_INFO(NULL, "GaussianBlur: bit-exact fx kernel can't be applied: ksize=" << ksize << " sigma=" << Size2d(sigma1, sigma2));
583 }
584 else if (param_check_gaussian_blur_bitexact_kernels && !validateGaussianBlurKernel(fky))
585 {
586 CV_LOG_INFO(NULL, "GaussianBlur: bit-exact fy kernel can't be applied: ksize=" << ksize << " sigma=" << Size2d(sigma1, sigma2));
587 }
588 else
589 {
590 CV_OCL_RUN(useOpenCL,
591 ocl_sepFilter2D_BitExact(_src, _dst, sdepth,
592 ksize,
593 (const uint16_t*)&fkx[0], (const uint16_t*)&fky[0],
594 Point(-1, -1), 0, borderType,
595 8/*shift_bits*/)
596 );
597
598 Mat src = _src.getMat();
599 Mat dst = _dst.getMat();
600
601 if (src.data == dst.data)
602 src = src.clone();
603
604 if ((sigma1 == 0.0) && (sigma2 == 0.0) && (ksize.height == ksize.width))
605 {
606 Point ofs;
607 Size wsz(src.cols, src.rows);
608 Mat src2 = src;
609 if(!(borderType & BORDER_ISOLATED))
