11void write( FileStorage& fs, const String& name, const Mat& m )
12{
13 char dt[22];
14
15 if( m.dims == 2 || m.empty() )
16 {
17 fs.startWriteStruct(name, FileNode::MAP, String("opencv-matrix"));
18 fs << "rows" << m.rows;
19 fs << "cols" << m.cols;
20 fs << "dt" << fs::encodeFormat( m.type(), dt, sizeof(dt) );
21 fs << "data" << "[:";
22 for( int i = 0; i < m.rows; i++ )
23 fs.writeRaw(dt, m.ptr(i), m.cols*m.elemSize());
24 fs << "]";
25 fs.endWriteStruct();
26 }
27 else
28 {
29 fs.startWriteStruct(name, FileNode::MAP, String("opencv-nd-matrix"));
30 fs << "sizes" << "[:";
31 fs.writeRaw( "i", m.size.p, m.dims*sizeof(int) );
32 fs << "]";
33 fs << "dt" << fs::encodeFormat( m.type(), dt, sizeof(dt) );
34 fs << "data" << "[:";
35 const Mat* arrays[] = {&m, 0};
36 uchar* ptrs[1] = {};
37 NAryMatIterator it(arrays, ptrs);
38 size_t total = it.size*m.elemSize();
39
40 for( size_t i = 0; i < it.nplanes; i++, ++it )
41 fs.writeRaw( dt, ptrs[0], total );
42 fs << "]";
43 fs.endWriteStruct();

64void write( FileStorage& fs, const String& name, const SparseMat& m )
65{
66 char dt[22];
67
68 fs.startWriteStruct(name, FileNode::MAP, String("opencv-sparse-matrix"));
69 fs << "sizes" << "[:";
70 int dims = m.dims();
71 if( dims > 0 )
72 fs.writeRaw("i", m.hdr->size, dims*sizeof(int) );
73 fs << "]";
74 fs << "dt" << fs::encodeFormat( m.type(), dt, sizeof(dt) );
75 fs << "data" << "[:";
76
77 size_t i = 0, n = m.nzcount();
78 AutoBuffer<const SparseMat::Node*> elems(n);
79 SparseMatConstIterator it = m.begin(), it_end = m.end();
80
81 for( ; it != it_end; ++it )
82 {
83 CV_Assert(it.node() != 0);
84 elems[i++] = it.node();
85 }
86
87 std::sort(elems.begin(), elems.end(), SparseNodeCmp(dims));
88 const SparseMat::Node* prev_node = 0;
89 size_t esz = m.elemSize();
90
91 for( i = 0; i < n; i++ )
92 {
93 const SparseMat::Node* node = elems[i];
94 int k = 0;
95
96 if( prev_node )
97 {
98 for( ; k < dims; k++ )
99 if( node->idx[k] != prev_node->idx[k] )
100 break;
101 CV_Assert( k < dims );
102 if( k < dims - 1 )
103 writeScalar( fs, k - dims + 1 );
104 }
105 for( ; k < dims; k++ )
106 writeScalar( fs, node->idx[k] );
107 prev_node = node;
108
109 const uchar* value = &m.value<uchar>(node);
110 fs.writeRaw(dt, value, esz);
111 }
112
113 fs << "]" << "}";
114}

116void read(const FileNode& node, Mat& m, const Mat& default_mat)
117{
118 if( node.empty() )
119 {
120 default_mat.copyTo(m);
121 return;
122 }
123
124 std::string dt;
125 int rows, cols, elem_type;
126
127 read(node["dt"], dt, std::string());
128 CV_Assert( !dt.empty() );
129
130 elem_type = fs::decodeSimpleFormat( dt.c_str() );
131
132 read(node["rows"], rows, -1);
133 if( rows >= 0 )
134 {
135 read(node["cols"], cols, -1);
136 m.create(rows, cols, elem_type);
137 }
138 else
139 {
140 int sizes[CV_MAX_DIM] = {0}, dims;
141 FileNode sizes_node = node["sizes"];
142 CV_Assert( !sizes_node.empty() );
143
144 dims = (int)sizes_node.size();
145 CV_Assert( dims >= 0 && dims <= CV_MAX_DIM );
146 sizes_node.readRaw("i", sizes, dims*sizeof(sizes[0]));
147
148 m.create(dims, sizes, elem_type);
149 }
150
151 FileNode data_node = node["data"];
152 CV_Assert(!data_node.empty());
153
154 size_t nelems = data_node.size();
155 if (nelems == 0)
156 {
157 m = Mat();
158 return;
159 }
160 CV_Assert(nelems == m.total()*m.channels());
161
162 data_node.readRaw(dt, (uchar*)m.ptr(), m.total()*m.elemSize());
163}

165void read( const FileNode& node, SparseMat& m, const SparseMat& default_mat )
166{
167 if( node.empty() )
168 {
169 default_mat.copyTo(m);
170 return;
171 }
172
173 std::string dt;
174 read(node["dt"], dt, std::string());
175 CV_Assert( !dt.empty() );
176
177 int elem_type = fs::decodeSimpleFormat( dt.c_str() );
178
179 int sizes[CV_MAX_DIM] = {0};
180 FileNode sizes_node = node["sizes"];
181 CV_Assert( !sizes_node.empty() );
182
183 int dims = (int)sizes_node.size();
184 CV_Assert( dims > 0 && dims <= CV_MAX_DIM );
185 sizes_node.readRaw("i", sizes, dims*sizeof(sizes[0]));
186
187 m.create(dims, sizes, elem_type);
188
189 FileNode data = node["data"];
190 CV_Assert( data.isSeq() );
191
192 int cn = CV_MAT_CN(elem_type);
193 int idx[CV_MAX_DIM] = {0};
194 size_t i, sz = data.size();
195 size_t esz = m.elemSize();
196 FileNodeIterator it = data.begin();
197
198 for( i = 0; i < sz; )
199 {
200 FileNode n = *it;
201 int k = (int)n;
202 if( i > 0 && k >= 0 )
203 idx[dims-1] = k;
204 else
205 {
206 if( i > 0 )
207 k = dims + k - 1;
208 else
209 idx[0] = k, k = 1;
210 for( ; k < dims; k++ )
211 {
212 ++it;
213 i++;
214 n = *it;
215 CV_Assert( n.isInt() );
216 int idx_k = (int)n;
217 CV_Assert( idx_k >= 0 );
218 idx[k] = idx_k;
219 }
220 }
221 ++it;
222 i++;
223 uchar* valptr = m.ptr(idx, true);
224 it.readRaw(dt, valptr, esz);
225 i += cn;
226 }
227}
