370static inline Point normalizeAnchor( Point anchor, Size ksize )
371{
372 if( anchor.x == -1 )
373 anchor.x = ksize.width/2;
374 if( anchor.y == -1 )
375 anchor.y = ksize.height/2;
376 CV_Assert( anchor.inside(Rect(0, 0, ksize.width, ksize.height)) );
377 return anchor;
378}
