78void FramebufferWindow::imshow(InputArray image)
79{
80 CV_LOG_DEBUG(NULL, "UI: FramebufferWindow::imshow(InputArray image)");
81 currentImg = image.getMat().clone();
82
83 CV_LOG_INFO(NULL, "UI: InputArray image: "
84 << cv::typeToString(image.type()) << " size " << image.size());
85
86 if (currentImg.empty())
87 {
88 CV_LOG_WARNING(NULL, "UI: image is empty");
89 return;
90 }
91 CV_CheckEQ(currentImg.dims, 2, "UI: dims != 2");
92
93 Mat img = image.getMat();
94 switch (img.channels())
95 {
96 case 1:
97 {
98 Mat tmp;
99 switch(img.type())
100 {
101 case CV_8U:
102 tmp = img;
103 break;
104 case CV_8S:
105 cv::convertScaleAbs(img, tmp, 1, 127);
106 break;
107 case CV_16S:
108 cv::convertScaleAbs(img, tmp, 1/255., 127);
109 break;
110 case CV_16U:
111 cv::convertScaleAbs(img, tmp, 1/255.);
112 break;
113 case CV_32F:
114 case CV_64F: // assuming image has values in range [0, 1)
115 img.convertTo(tmp, CV_8U, 255., 0.);
116 break;
117 }
118 Mat rgb(img.rows, img.cols, CV_8UC3);
119 cvtColor(tmp, rgb, COLOR_GRAY2RGB);
120 img = rgb;
121 }
122 break;
123 case 3:
124 case 4:
125 {
126 Mat tmp(img.rows, img.cols, CV_8UC3);
127 convertToShow(img, tmp, true);
128 img = tmp;
129 }
130 break;
131 default:
132 CV_Error(cv::Error::StsBadArg, "Bad image: wrong number of channels");
133 }
134 {
135 Mat bgra(img.rows, img.cols, CV_8UC4);
136 cvtColor(img, bgra, COLOR_RGB2BGRA, bgra.channels());
137 img = bgra;
138 }
139
140 int newWidth = windowRect.width;
141 int newHeight = windowRect.height;
142 int cntChannel = img.channels();
143 cv::Size imgSize = currentImg.size();
144
145 if (flags & WINDOW_AUTOSIZE)
146 {
147 windowRect.width = imgSize.width;
148 windowRect.height = imgSize.height;
149 newWidth = windowRect.width;
150 newHeight = windowRect.height;
151 }
152
153 if (flags & WINDOW_FREERATIO)
154 {
155 newWidth = windowRect.width;
156 newHeight = windowRect.height;
157 }
158 else //WINDOW_KEEPRATIO
159 {
160 double aspect_ratio = ((double)img.cols) / img.rows;
161 newWidth = windowRect.width;
162 newHeight = (int)(windowRect.width / aspect_ratio);
163
164 if (newHeight > windowRect.height)
165 {
166 newWidth = (int)(windowRect.height * aspect_ratio);
167 newHeight = windowRect.height;
168 }
169 }
170
171 if ((newWidth != img.cols) && (newHeight != img.rows))
172 {
173 Mat imResize;
174 cv::resize(img, imResize, cv::Size(newWidth, newHeight), INTER_LINEAR);
175 img = imResize;
176 }
177
178 CV_LOG_INFO(NULL, "UI: Formated image: "
179 << cv::typeToString(img.type()) << " size " << img.size());
180
181 if (backend.getMode() == FB_MODE_EMU)
182 {
183 CV_LOG_WARNING(NULL, "UI: FramebufferWindow::imshow is used in EMU mode");
184 return;
185 }
186
187 if (backend.getFBPointer() == MAP_FAILED)
188 {
189 CV_LOG_ERROR(NULL, "UI: Framebuffer is not mapped");
190 return;
191 }
192
193 int xOffset = backend.getFBXOffset();
194 int yOffset = backend.getFBYOffset();
195 int fbHeight = backend.getFBHeight();
196 int fbWidth = backend.getFBWidth();
197 int lineLength = backend.getFBLineLength();
198
199 int img_start_x;
200 int img_start_y;
201 int img_end_x;
202 int img_end_y;
203 int fb_start_x;
204 int fb_start_y;
205
206 if (windowRect.y - yOffset < 0)
207 {
208 img_start_y = - (windowRect.y - yOffset);
209 }
210 else
211 {
212 img_start_y = 0;
213 }
214 if (windowRect.x - xOffset < 0)
215 {
216 img_start_x = - (windowRect.x - xOffset);
217 }
218 else
219 {
220 img_start_x = 0;
221 }
222
223 if (windowRect.y + yOffset + img.rows > fbHeight)
224 {
225 img_end_y = fbHeight - windowRect.y - yOffset;
226 }
227 else
228 {
229 img_end_y = img.rows;
230 }
231 if (windowRect.x + xOffset + img.cols > fbWidth)
232 {
233 img_end_x = fbWidth - windowRect.x - xOffset;
234 }
235 else
236 {
237 img_end_x = img.cols;
238 }
239
240 if (windowRect.y + yOffset >= 0)
241 {
242 fb_start_y = windowRect.y + yOffset;
243 }
244 else
245 {
246 fb_start_y = 0;
247 }
248 if (windowRect.x + xOffset >= 0)
249 {
250 fb_start_x = windowRect.x + xOffset;
251 }
252 else
253 {
254 fb_start_x = 0;
255 }
256
257 for (int y = img_start_y; y < img_end_y; y++)
258 {
259 std::memcpy(backend.getFBPointer() +
260 (fb_start_y + y - img_start_y) * lineLength + fb_start_x * cntChannel,
261 img.ptr<unsigned char>(y) + img_start_x * cntChannel,
262 (img_end_x - img_start_x) * cntChannel);
263 }

360int FramebufferBackend::fbOpenAndGetInfo()
361{
362 std::string fbFileName = getFBFileName();
363 CV_LOG_INFO(NULL, "UI: FramebufferWindow::The following is used as a framebuffer file: \n"
364 << fbFileName);
365
366 int fb_fd = open(fbFileName.c_str(), O_RDWR);
367 if (fb_fd == -1)
368 {
369 CV_LOG_ERROR(NULL, "UI: can't open framebuffer");
370 return -1;
371 }
372
373 if (ioctl(fb_fd, FBIOGET_FSCREENINFO, &fixInfo))
374 {
375 CV_LOG_ERROR(NULL, "UI: can't read fix info for framebuffer");
376 return -1;
377 }
378
379 if (ioctl(fb_fd, FBIOGET_VSCREENINFO, &varInfo))
380 {
381 CV_LOG_ERROR(NULL, "UI: can't read var info for framebuffer");
382 return -1;
383 }
384
385 CV_LOG_INFO(NULL, "UI: framebuffer info: \n"
386 << " red offset " << varInfo.red.offset << " length " << varInfo.red.length << "\n"
387 << " green offset " << varInfo.green.offset << " length " << varInfo.green.length << "\n"
388 << " blue offset " << varInfo.blue.offset << " length " << varInfo.blue.length << "\n"
389 << "transp offset " << varInfo.transp.offset << " length " <<varInfo.transp.length << "\n"
390 << "bits_per_pixel " << varInfo.bits_per_pixel);
391
392 if ((varInfo.red.offset != 16) && (varInfo.red.length != 8) &&
393 (varInfo.green.offset != 8) && (varInfo.green.length != 8) &&
394 (varInfo.blue.offset != 0) && (varInfo.blue.length != 8) &&
395 (varInfo.bits_per_pixel != 32) )
396 {
397 close(fb_fd);
398 CV_LOG_ERROR(NULL, "UI: Framebuffer format is not supported "
399 << "(use BGRA format with bits_per_pixel = 32)");
400 return -1;
401 }
402
403 fbWidth = varInfo.xres;
404 fbHeight = varInfo.yres;
405 fbXOffset = varInfo.xoffset;
406 fbYOffset = varInfo.yoffset;
407 fbBitsPerPixel = varInfo.bits_per_pixel;
408 fbLineLength = fixInfo.line_length;
409
410 fbScreenSize = max(varInfo.xres, varInfo.xres_virtual) *
411 max(varInfo.yres, varInfo.yres_virtual) *
412 fbBitsPerPixel / 8;
413
414 fbPointer = (unsigned char*)
415 mmap(0, fbScreenSize, PROT_READ | PROT_WRITE, MAP_SHARED, fb_fd, 0);
416
417 if (fbPointer == MAP_FAILED)
418 {
419 CV_LOG_ERROR(NULL, "UI: can't mmap framebuffer");
420 return -1;
421 }
422
423 return fb_fd;
424}

426int FramebufferBackend::XvfbOpenAndGetInfo()
427{
428 int fb_fd = -1;
429
430#ifdef HAVE_FRAMEBUFFER_XVFB
431 std::string fbFileName = getFBFileName();
432 CV_LOG_INFO(NULL, "UI: FramebufferWindow::The following is used as a framebuffer file: \n"
433 << fbFileName);
434
435 fb_fd = open(fbFileName.c_str(), O_RDWR);
436 if (fb_fd == -1)
437 {
438 CV_LOG_ERROR(NULL, "UI: can't open framebuffer");
439 return -1;
440 }
441
442 XWDFileHeader *xwd_header;
443
444 xwd_header = (XWDFileHeader*)
445 mmap(NULL, sizeof(XWDFileHeader), PROT_READ, MAP_SHARED, fb_fd, 0);
446
447 if (xwd_header == MAP_FAILED)
448 {
449 CV_LOG_ERROR(NULL, "UI: can't mmap xwd header");
450 return -1;
451 }
452
453 if (C32INT(&(xwd_header->pixmap_format)) != ZPixmap)
454 {
455 CV_LOG_ERROR(NULL, "Unsupported pixmap format: " << xwd_header->pixmap_format);
456 return -1;
457 }
458
459 if (xwd_header->xoffset != 0)
460 {
461 CV_LOG_ERROR(NULL, "UI: Unsupported xoffset value: " << xwd_header->xoffset );
462 return -1;
463 }
464
465 unsigned int r = C32INT(&(xwd_header->red_mask));
466 unsigned int g = C32INT(&(xwd_header->green_mask));
467 unsigned int b = C32INT(&(xwd_header->blue_mask));
468
469 fbWidth = C32INT(&(xwd_header->pixmap_width));
470 fbHeight = C32INT(&(xwd_header->pixmap_height));
471 fbXOffset = 0;
472 fbYOffset = 0;
473 fbLineLength = C32INT(&(xwd_header->bytes_per_line));
474 fbBitsPerPixel = C32INT(&(xwd_header->bits_per_pixel));
475
476 CV_LOG_INFO(NULL, "UI: XVFB info: \n"
477 << " red_mask " << r << "\n"
478 << " green_mask " << g << "\n"
479 << " blue_mask " << b << "\n"
480 << "bits_per_pixel " << fbBitsPerPixel);
481
482 if ((r != 16711680 ) && (g != 65280 ) && (b != 255 ) &&
483 (fbBitsPerPixel != 32))
484 {
485 CV_LOG_ERROR(NULL, "UI: Framebuffer format is not supported "
486 << "(use BGRA format with bits_per_pixel = 32)");
487 return -1;
488 }
489
490 xvfb_len_header = C32INT(&(xwd_header->header_size));
491 xvfb_len_colors = sizeof(XWDColor) * C32INT(&(xwd_header->ncolors));
492 xvfb_len_pixmap = C32INT(&(xwd_header->bytes_per_line)) *
493 C32INT(&(xwd_header->pixmap_height));
494
495 munmap(xwd_header, sizeof(XWDFileHeader));
496
497 fbScreenSize = xvfb_len_header + xvfb_len_colors + xvfb_len_pixmap;
498 xwd_header = (XWDFileHeader*)
499 mmap(NULL, fbScreenSize, PROT_READ | PROT_WRITE, MAP_SHARED, fb_fd, 0);
500
501 fbPointer = (unsigned char*)xwd_header;
502 fbPointer_dist = xvfb_len_header + xvfb_len_colors;
503
504#else
505 CV_LOG_WARNING(NULL, "UI: To use virtual framebuffer, "
506 << "compile OpenCV with the WITH_FRAMEBUFFER_XVFB=ON");
507#endif
508
509 return fb_fd;
510}

572FramebufferBackend::FramebufferBackend():mode(FB_MODE_FB), fbPointer_dist(0)
573{
574 CV_LOG_DEBUG(NULL, "UI: FramebufferWindow::FramebufferBackend()");
575
576 std::string fbModeStr = getFBMode();
577
578 if (fbModeStr == "EMU")
579 {
580 mode = FB_MODE_EMU;
581 CV_LOG_WARNING(NULL, "UI: FramebufferWindow is trying to use EMU mode");
582 }
583 if (fbModeStr == "FB")
584 {
585 mode = FB_MODE_FB;
586 CV_LOG_WARNING(NULL, "UI: FramebufferWindow is trying to use FB mode");
587 }
588 if (fbModeStr == "XVFB")
589 {
590 mode = FB_MODE_XVFB;
591 CV_LOG_WARNING(NULL, "UI: FramebufferWindow is trying to use XVFB mode");
592 }
593
594 fbID = -1;
595 if (mode == FB_MODE_FB)
596 {
597 fbID = fbOpenAndGetInfo();
598 }
599 if (mode == FB_MODE_XVFB)
600 {
601 fbID = XvfbOpenAndGetInfo();
602 }
603
604 CV_LOG_INFO(NULL, "UI: FramebufferWindow::fbID " << fbID);
605
606 if (fbID == -1)
607 {
608 mode = FB_MODE_EMU;
609 fbWidth = 640;
610 fbHeight = 480;
611 fbXOffset = 0;
612 fbYOffset = 0;
613 fbBitsPerPixel = 0;
614 fbLineLength = 0;
615
616 CV_LOG_WARNING(NULL, "UI: FramebufferWindow is used in EMU mode");
617 return;
618 }
619
620 CV_LOG_INFO(NULL, "UI: Framebuffer's width, height, bits per pix: "
621 << fbWidth << " " << fbHeight << " " << fbBitsPerPixel);
622
623 CV_LOG_INFO(NULL, "UI: Framebuffer's offsets (x, y), line length: "
624 << fbXOffset << " " << fbYOffset << " " << fbLineLength);
625
626 backgroundBuff = Mat(fbHeight, fbWidth, CV_8UC4);
627 int cntChannel = 4;
628 for (int y = fbYOffset; y < backgroundBuff.rows + fbYOffset; y++)
629 {
630 std::memcpy(backgroundBuff.ptr<unsigned char>(y - fbYOffset),
631 getFBPointer() + y * fbLineLength + fbXOffset * cntChannel,
632 backgroundBuff.cols * cntChannel);
633 }
634}

637{
638 CV_LOG_DEBUG(NULL, "UI: FramebufferBackend::~FramebufferBackend()");
639 if(fbID == -1) return;
640

670void FramebufferBackend::initTermios(int echo, int wait)
671{
672 tcgetattr(0, &old);
673 current = old;
674 current.c_lflag &= ~ICANON;
675 current.c_lflag &= ~ISIG;
676 current.c_cc[VMIN] = wait;
677 if (echo)
678 {
679 current.c_lflag |= ECHO;
680 }
681 else
682 {
683 current.c_lflag &= ~ECHO;
684 }
685 tcsetattr(0, TCSANOW, &current);

719int FramebufferBackend::waitKeyEx(int delay)
720{
721 CV_LOG_DEBUG(NULL, "UI: FramebufferBackend::waitKeyEx(int delay = " << delay << ")");
722
723 int code = -1;
724
725 if (delay <= 0)
726 {
727 int ch = getch_(0, 1);
728 CV_LOG_INFO(NULL, "UI: FramebufferBackend::getch_() take value = " << (int)ch);
729 code = ch;
730
731 while ((ch = getch_(0, 0)) >= 0)
732 {
733 CV_LOG_INFO(NULL, "UI: FramebufferBackend::getch_() take value = "
734 << (int)ch << " (additional code on <stdin>)");
735 code = ch;
736 }
737 }
738 else
739 {
740 bool f_kbhit = false;
741 while (!(f_kbhit = kbhit()) && (delay > 0))
742 {
743 delay -= 1;
744 usleep(1000);
745 }
746 if (f_kbhit)
747 {
748 CV_LOG_INFO(NULL, "UI: FramebufferBackend kbhit is True ");
749
750 int ch = getch_(0, 1);
751 CV_LOG_INFO(NULL, "UI: FramebufferBackend::getch_() take value = " << (int)ch);
752 code = ch;
753
754 while ((ch = getch_(0, 0)) >= 0)
755 {
756 CV_LOG_INFO(NULL, "UI: FramebufferBackend::getch_() take value = "
757 << (int)ch << " (additional code on <stdin>)");
758 code = ch;
759 }
760 }
761 }
762
763 CV_LOG_INFO(NULL, "UI: FramebufferBackend::waitKeyEx() result code = " << code);
764 return code;
765}

767int FramebufferBackend::pollKey()
768{
769 CV_LOG_DEBUG(NULL, "UI: FramebufferBackend::pollKey()");
770 int code = -1;
771 bool f_kbhit = false;
772 f_kbhit = kbhit();
773
