65void Timelapser::process(InputArray _img, InputArray /*_mask*/, Point tl)
66{
67 CV_INSTRUMENT_REGION();
68
69 dst_.setTo(Scalar::all(0));
70
71 Mat img = _img.getMat();
72 Mat dst = dst_.getMat(ACCESS_RW);
73
74 CV_Assert(img.type() == CV_16SC3);
75 int dx = tl.x - dst_roi_.x;
76 int dy = tl.y - dst_roi_.y;
77
78 for (int y = 0; y < img.rows; ++y)
79 {
80 const Point3_<short> *src_row = img.ptr<Point3_<short> >(y);
81
82 for (int x = 0; x < img.cols; ++x)
83 {
84 if (test_point(Point(tl.x + x, tl.y + y)))
85 {
86 Point3_<short> *dst_row = dst.ptr<Point3_<short> >(dy + y);
87 dst_row[dx + x] = src_row[x];
88 }
89 }
90 }
91}
