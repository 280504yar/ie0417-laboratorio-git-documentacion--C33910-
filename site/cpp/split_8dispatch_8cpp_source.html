72static bool ipp_split(const Mat& src, Mat* mv, int channels)
73{
74#ifdef HAVE_IPP_IW_LL
75 CV_INSTRUMENT_REGION_IPP();
76
77 if(channels != 3 && channels != 4)
78 return false;
79
80 if(src.dims <= 2)
81 {
82 IppiSize size = ippiSize(src.size());
83 void *dstPtrs[4] = {NULL};
84 size_t dstStep = mv[0].step;
85 for(int i = 0; i < channels; i++)
86 {
87 dstPtrs[i] = mv[i].ptr();
88 if(dstStep != mv[i].step)
89 return false;
90 }
91
92 return CV_INSTRUMENT_FUN_IPP(llwiCopySplit, src.ptr(), (int)src.step, dstPtrs, (int)dstStep, size, (int)src.elemSize1(), channels, 0) >= 0;
93 }
94 else
95 {
96 const Mat *arrays[5] = {NULL};
97 uchar *ptrs[5] = {NULL};
98 arrays[0] = &src;
99
100 for(int i = 1; i < channels; i++)
101 {
102 arrays[i] = &mv[i-1];
103 }
104
105 NAryMatIterator it(arrays, ptrs);
106 IppiSize size = { (int)it.size, 1 };
107
108 for( size_t i = 0; i < it.nplanes; i++, ++it )
109 {
110 if(CV_INSTRUMENT_FUN_IPP(llwiCopySplit, ptrs[0], 0, (void**)&ptrs[1], 0, size, (int)src.elemSize1(), channels, 0) < 0)
111 return false;
112 }
113 return true;
114 }
115#else
116 CV_UNUSED(src); CV_UNUSED(mv); CV_UNUSED(channels);
117 return false;

122void split(const Mat& src, Mat* mv)
123{
124 CV_INSTRUMENT_REGION();
125
126 int k, depth = src.depth(), cn = src.channels();
127 if( cn == 1 )
128 {
129 src.copyTo(mv[0]);
130 return;
131 }
132
133 for( k = 0; k < cn; k++ )
134 {
135 mv[k].create(src.size, depth);
136 }
137
138 CV_IPP_RUN_FAST(ipp_split(src, mv, cn));
139
140 SplitFunc func = getSplitFunc(depth);
141 CV_Assert( func != 0 );
142
143 size_t esz = src.elemSize(), esz1 = src.elemSize1();
144 size_t blocksize0 = (BLOCK_SIZE + esz-1)/esz;
145 AutoBuffer<uchar> _buf((cn+1)*(sizeof(Mat*) + sizeof(uchar*)) + 16);
146 const Mat** arrays = (const Mat**)_buf.data();
147 uchar** ptrs = (uchar**)alignPtr(arrays + cn + 1, 16);
148
149 arrays[0] = &src;
150 for( k = 0; k < cn; k++ )
151 {
152 arrays[k+1] = &mv[k];
153 }
154
155 NAryMatIterator it(arrays, ptrs, cn+1);
156 size_t total = it.size;
157 size_t blocksize = std::min((size_t)CV_SPLIT_MERGE_MAX_BLOCK_SIZE(cn), cn <= 4 ? total : std::min(total, blocksize0));
158
159 for( size_t i = 0; i < it.nplanes; i++, ++it )
160 {
161 for( size_t j = 0; j < total; j += blocksize )
162 {
163 size_t bsz = std::min(total - j, blocksize);
164 func( ptrs[0], &ptrs[1], (int)bsz, cn );
165
166 if( j + blocksize < total )
167 {
168 ptrs[0] += bsz*esz;
169 for( k = 0; k < cn; k++ )
170 ptrs[k+1] += bsz*esz1;
171 }
172 }
173 }
174}

178static bool ocl_split( InputArray _m, OutputArrayOfArrays _mv )
179{
180 int type = _m.type(), depth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type),
181 rowsPerWI = ocl::Device::getDefault().isIntel() ? 4 : 1;
182
183 String dstargs, processelem, indexdecl;
184 for (int i = 0; i < cn; ++i)
185 {
186 dstargs += format("DECLARE_DST_PARAM(%d)", i);
187 indexdecl += format("DECLARE_INDEX(%d)", i);
188 processelem += format("PROCESS_ELEM(%d)", i);
189 }
190
191 ocl::Kernel k("split", ocl::core::split_merge_oclsrc,
192 format("-D T=%s -D OP_SPLIT -D cn=%d -D DECLARE_DST_PARAMS=%s"
193 " -D PROCESS_ELEMS_N=%s -D DECLARE_INDEX_N=%s",
194 ocl::memopTypeToStr(depth), cn, dstargs.c_str(),
195 processelem.c_str(), indexdecl.c_str()));
196 if (k.empty())
197 return false;
198
199 Size size = _m.size();
200 _mv.create(cn, 1, depth);
201 for (int i = 0; i < cn; ++i)
202 _mv.create(size, depth, i);
203
204 std::vector<UMat> dst;
205 _mv.getUMatVector(dst);
206
207 int argidx = k.set(0, ocl::KernelArg::ReadOnly(_m.getUMat()));
208 for (int i = 0; i < cn; ++i)
209 argidx = k.set(argidx, ocl::KernelArg::WriteOnlyNoSize(dst[i]));
210 k.set(argidx, rowsPerWI);
211
212 size_t globalsize[2] = { (size_t)size.width, ((size_t)size.height + rowsPerWI - 1) / rowsPerWI };
213 return k.run(2, globalsize, NULL, false);
214}

218void split(InputArray _m, OutputArrayOfArrays _mv)
219{
220 CV_INSTRUMENT_REGION();
221
222 CV_OCL_RUN(_m.dims() <= 2 && _mv.isUMatVector(),
223 ocl_split(_m, _mv))
224
225 Mat m = _m.getMat();
226 if( m.empty() )
227 {
228 _mv.release();
229 return;
230 }
231
232 CV_Assert( !_mv.fixedType() || _mv.empty() || _mv.type() == m.depth() );
233
234 int depth = m.depth(), cn = m.channels();
235 _mv.create(cn, 1, depth);
236 for (int i = 0; i < cn; ++i)
237 _mv.create(m.dims, m.size.p, depth, i);
238
239 std::vector<Mat> dst;
240 _mv.getMatVector(dst);
241
242 split(m, &dst[0]);
243}
