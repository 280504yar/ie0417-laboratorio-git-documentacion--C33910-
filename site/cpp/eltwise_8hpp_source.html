61 void forward(
62 const std::vector<cuda::GpuMatND>& inputs,
63 const std::vector<cuda::GpuMatND>& outputs,
64 csl::Workspace& workspace) override
65 {
66 CV_Assert(outputs.size() == 1);
67
68 CV_Assert(coeffs.size() == 0 || op == EltwiseOpType::SUM);
69 CV_Assert(coeffs.size() == 0 || inputs.size() == coeffs.size());
70
71 auto output = csl::spanOf<T>(outputs[0]);
72
73 if (inputs.size() == 2)
74 {
75 auto input_x = csl::viewOf<T>(inputs[0]);
76 auto input_y = csl::viewOf<T>(inputs[1]);
77
78 switch (op)
79 {
80 case EltwiseOpType::MAX: kernels::eltwise_max_2<T>(stream, output, input_x, input_y); break;
81 case EltwiseOpType::MIN: kernels::eltwise_min_2<T>(stream, output, input_x, input_y); break;
82 case EltwiseOpType::PRODUCT: kernels::eltwise_prod_2<T>(stream, output, input_x, input_y); break;
83 case EltwiseOpType::DIV: kernels::eltwise_div_2<T>(stream, output, input_x, input_y); break;
84 case EltwiseOpType::SUM:
85 if (coeffs.empty() || (coeffs[0] == 1 && coeffs[1] == 1))
86 kernels::eltwise_sum_2<T>(stream, output, input_x, input_y);
87 else
88 kernels::eltwise_sum_coeff_2<T>(stream, output, coeffs[0], input_x, coeffs[1], input_y);
89 break;
90 case EltwiseOpType::SUB: kernels::eltwise_sub_2<T>(stream, output, input_x, input_y); break;
91 case EltwiseOpType::MOD: kernels::eltwise_mod_2<T>(stream, output, input_x, input_y); break;
92 case EltwiseOpType::FMOD: kernels::eltwise_fmod_2<T>(stream, output, input_x, input_y); break;
93 case EltwiseOpType::POW: kernels::eltwise_pow_2<T>(stream, output, input_x, input_y); break;
94 }
95 } else if (inputs.size() == 1) {
96 auto input_0 = csl::viewOf<T>(inputs[0]);
97 csl::tensor_ops::copy(stream, output, input_0);
98 } else {
99 auto input_0 = csl::viewOf<T>(inputs[0]);
100
101 /* we first make a copy and then apply EltwiseOp cumulatively */
102 csl::tensor_ops::copy(stream, output, input_0);
103
104 for (int i = 1; i < inputs.size(); i++)
105 {
106 auto input = csl::viewOf<T>(inputs[i]);
107
108 switch (op)
109 {
110 case EltwiseOpType::MAX: kernels::eltwise_max_2<T>(stream, output, output, input); break;
111 case EltwiseOpType::MIN: kernels::eltwise_min_2<T>(stream, output, output, input); break;
112 case EltwiseOpType::PRODUCT: kernels::eltwise_prod_2<T>(stream, output, output, input); break;
113 case EltwiseOpType::DIV: kernels::eltwise_div_2<T>(stream, output, output, input); break;
114 case EltwiseOpType::SUM:
115 if (coeffs.empty() || coeffs[i] == 1)
116 kernels::eltwise_sum_2<T>(stream, output, output, input);
117 else
118 {
119 /* if this is the first op, we must scale output too */
120 T coeff_x = (i == 1) ? coeffs[0] : 1.0;
121 kernels::eltwise_sum_coeff_2<T>(stream, output, coeff_x, output, coeffs[i], input);
122 }
123 break;
124 case EltwiseOpType::SUB: kernels::eltwise_sub_2<T>(stream, output, output, input); break;
125 case EltwiseOpType::MOD: kernels::eltwise_mod_2<T>(stream, output, output, input); break;
126 case EltwiseOpType::FMOD: kernels::eltwise_fmod_2<T>(stream, output, output, input); break;
127 case EltwiseOpType::POW: kernels::eltwise_pow_2<T>(stream, output, output, input); break;
128 }
129 }
130 }
131 }
132 };
