48struct unicode_cpt_flags {
49 enum {
50 UNDEFINED = 0x0001,
51 NUMBER = 0x0002, // regex: \p{N}
52 LETTER = 0x0004, // regex: \p{L}
53 SEPARATOR = 0x0008, // regex: \p{Z}
54 ACCENT_MARK = 0x0010, // regex: \p{M}
55 PUNCTUATION = 0x0020, // regex: \p{P}
56 SYMBOL = 0x0040, // regex: \p{S}
57 CONTROL = 0x0080, // regex: \p{C}
58 MASK_CATEGORIES = 0x00FF,
59 };
60
61 // codepoint type
62 uint16_t is_undefined : 1;
63 uint16_t is_number : 1; // regex: \p{N}
64 uint16_t is_letter : 1; // regex: \p{L}
65 uint16_t is_separator : 1; // regex: \p{Z}
66 uint16_t is_accent_mark : 1; // regex: \p{M}
67 uint16_t is_punctuation : 1; // regex: \p{P}
68 uint16_t is_symbol : 1; // regex: \p{S}
69 uint16_t is_control : 1; // regex: \p{C}
70 // helper flags
71 uint16_t is_whitespace : 1; // regex: \s
72 uint16_t is_lowercase : 1;
73 uint16_t is_uppercase : 1;
74 uint16_t is_nfd : 1;
75
76 // decode from uint16
77 inline unicode_cpt_flags(const uint16_t flags = 0) {
78 *reinterpret_cast<uint16_t*>(this) = flags;
79 }
80
81 inline uint16_t as_uint() const {
82 return *reinterpret_cast<const uint16_t*>(this);
83 }
84
85 inline uint16_t category_flag() const {
86 return this->as_uint() & MASK_CATEGORIES;
87 }
88};
