18/*
19 LpPool layer, as defined in ONNX specification:
20 https://onnx.ai/onnx/operators/onnx__LpPool.html
21 Supported opsets: 1-18
22
23 Computes the Lp-norm pooling (sum(|x|^p))^(1/p) over a sliding window.
24*/
25
26#if CV_SIMD
27// Vectorized per-element policies for the common norms. Each policy supplies
28// the accumulation step (applied per loaded vector) and the final reduction
29// (applied once per output element). This lets a single templated kernel cover
30// both p == 1 and p == 2 instead of duplicating the loop nest for each.
31struct LpPoolL1
32{
33 static inline v_float32 accum(const v_float32& acc, const v_float32& v) { return v_add(acc, v_abs(v)); }
34 static inline v_float32 finalize(const v_float32& acc) { return acc; }
35};
36
37struct LpPoolL2
38{
39 static inline v_float32 accum(const v_float32& acc, const v_float32& v) { return v_add(acc, v_mul(v, v)); }
40 static inline v_float32 finalize(const v_float32& acc) { return v_sqrt(acc); }
41};

43// One output column for NVEC*nlanes channels on the boundary path: every tap is
44// range-checked because the window may stick out of the padded volume. zi_/yi_/xi_
45// are the (already strided, pad-shifted) top-left source coords; inp_c is the input
46// base advanced to the current channel block, out_xc the matching output slot.
47template<class Op, int NVEC>
48static inline void poolBoundaryTile(const float* inp_c, float* out_xc,
49 int zi_, int yi_, int xi_,
50 int Di, int Hi, int Wi, int C0,
51 const int* zyxtab, int ksize, int nlanes, const v_float32& z)
52{
53 constexpr int MAX_POOL_DIMS = ConvState::MAX_CONV_DIMS;
54 v_float32 s[NVEC];
55 for (int j = 0; j < NVEC; j++) s[j] = z;
56 for (int k = 0; k < ksize; k++) {
57 int zi = zi_ + zyxtab[k*MAX_POOL_DIMS];
58 int yi = yi_ + zyxtab[k*MAX_POOL_DIMS + 1];
59 int xi = xi_ + zyxtab[k*MAX_POOL_DIMS + 2];
60 if ((unsigned)zi >= (unsigned)Di ||
61 (unsigned)yi >= (unsigned)Hi ||
62 (unsigned)xi >= (unsigned)Wi)
63 continue;
64 const float* inp_k = inp_c + ((zi*Hi + yi)*Wi + xi)*C0;
65 for (int j = 0; j < NVEC; j++)
66 s[j] = Op::accum(s[j], vx_load(inp_k + j*nlanes));
67 }
68 for (int j = 0; j < NVEC; j++)
69 vx_store(out_xc + j*nlanes, Op::finalize(s[j]));
70}
71
72// Same accumulate-and-store skeleton for the inner path, where the window is fully
73// inside the volume so taps are addressed through the precomputed flat ofstab and
74// no bounds check is needed. inp_xi already points at the column's channel block.
75template<class Op, int NVEC>
76static inline void poolInnerTile(const float* inp_xi, float* out_xc,
77 const int* ofstab, int ksize, int nlanes, const v_float32& z)
78{
79 v_float32 s[NVEC];
80 for (int j = 0; j < NVEC; j++) s[j] = z;
81 for (int k = 0; k < ksize; k++) {
82 int ofs_k = ofstab[k];
83 for (int j = 0; j < NVEC; j++)
84 s[j] = Op::accum(s[j], vx_load(inp_xi + ofs_k + j*nlanes));
85 }
86 for (int j = 0; j < NVEC; j++)
87 vx_store(out_xc + j*nlanes, Op::finalize(s[j]));
88}

90template<class Op>
91static void lpPoolSIMD(const float* inp_, float* out_, const ConvState& cs)
92{
93 int NC1 = cs.inpshape[0]*cs.inpshape[1];
94
95 CV_Assert(cs.inpshape.layout == DATA_LAYOUT_BLOCK);
96 CV_Assert(cs.outshape.layout == DATA_LAYOUT_BLOCK);
97 CV_Assert(cs.inpshape.dims == cs.outshape.dims);
98
99 parallel_for_(Range(0, NC1), [&](const Range& r) {
100 constexpr int MAX_POOL_DIMS = ConvState::MAX_CONV_DIMS;
101
102 CV_Assert(cs.nspatialdims <= MAX_POOL_DIMS && MAX_POOL_DIMS == 3);
103
104 int sdims = cs.nspatialdims;
105 int nc0 = r.start, nc1 = r.end;
106 int C0 = cs.inpshape.back();
107 int Di = sdims > 2 ? cs.inpshape[sdims - 1] : 1;
108 int Hi = sdims > 1 ? cs.inpshape[sdims] : 1;
109 int Wi = cs.inpshape[sdims + 1];
110 int D = sdims > 2 ? cs.outshape[sdims - 1] : 1;
111 int H = sdims > 1 ? cs.outshape[sdims] : 1;
112 int W = cs.outshape[sdims + 1];
113 int iplanesize = Di*Hi*Wi*C0;
114 int planesize = D*H*W*C0;
115 int SZ = cs.strides[0], SY = cs.strides[1], SX = cs.strides[2];
116 int padZ0 = cs.pads[0], padY0 = cs.pads[1], padX0 = cs.pads[2];
117 int inner_z0 = cs.inner[0], inner_z1 = cs.inner[MAX_POOL_DIMS];
118 int inner_y0 = cs.inner[1], inner_y1 = cs.inner[MAX_POOL_DIMS + 1];
119 int inner_x0 = cs.inner[2], inner_x1 = cs.inner[MAX_POOL_DIMS + 2];
120 int ksize = (int)cs.ofstab.size();
121 const int* zyxtab = cs.coordtab.data();
122 const int* ofstab = cs.ofstab.data();
123
124 const float* inp = inp_ + nc0*iplanesize;
125 float* out = out_ + nc0*planesize;
126
127 int nlanes = VTraits<v_float32>::vlanes();
128 CV_Assert(C0 == nlanes || C0 == nlanes*2 || C0 % (nlanes*4) == 0);
129 v_float32 z = vx_setzero_f32();
130
131 for (int nc = nc0; nc < nc1; nc++, inp += iplanesize) {
132 for (int z0 = 0; z0 < D; z0++) {
133 int zi_ = z0*SZ - padZ0;
134 for (int y0 = 0; y0 < H; y0++, out += W*C0) {
135 int x0 = 0;
136 int x1 = z0 >= inner_z0 && z0 < inner_z1 &&
137 y0 >= inner_y0 && y0 < inner_y1 ? inner_x0 : W;
138 int yi_ = y0*SY - padY0;
139
140 for(;;) {
141 // Boundary (outer) path — needs per-element bounds check
142 if (nlanes == C0) {
143 for (; x0 < x1; x0++) {
144 int xi_ = x0*SX - padX0;
145 poolBoundaryTile<Op, 1>(inp, out + x0*C0, zi_, yi_, xi_,
146 Di, Hi, Wi, C0, zyxtab, ksize, nlanes, z);
147 }
148 } else {
149 for (; x0 < x1; x0++) {
150 int xi_ = x0*SX - padX0;
151 for (int c = 0; c < C0; c += nlanes*2)
152 poolBoundaryTile<Op, 2>(inp + c, out + x0*C0 + c, zi_, yi_, xi_,
153 Di, Hi, Wi, C0, zyxtab, ksize, nlanes, z);
154 }
155 }
156
157 if (x0 == W)
158 break;
159 x1 = inner_x1;
160
161 // Inner path — no bounds check needed
162 if (nlanes == C0) {
163 for (; x0 < x1; x0++) {
164 int xi_ = x0*SX - padX0;
165 const float* inp_xi = inp + ((Hi*zi_ + yi_)*Wi + xi_)*C0;
166 poolInnerTile<Op, 1>(inp_xi, out + x0*C0, ofstab, ksize, nlanes, z);
167 }
168 } else if (nlanes*2 == C0) {
169 for (; x0 < x1; x0++) {
170 int xi_ = x0*SX - padX0;
171 const float* inp_xi = inp + ((Hi*zi_ + yi_)*Wi + xi_)*C0;
172 poolInnerTile<Op, 2>(inp_xi, out + x0*C0, ofstab, ksize, nlanes, z);
173 }
174 } else {
175 for (; x0 < x1; x0++) {
176 int xi_ = x0*SX - padX0;
177 for (int c = 0; c < C0; c += nlanes*4) {
178 const float* inp_xi = inp + ((Hi*zi_ + yi_)*Wi + xi_)*C0 + c;
179 poolInnerTile<Op, 4>(inp_xi, out + x0*C0 + c, ofstab, ksize, nlanes, z);
180 }
181 }
182 }
183 x1 = W;
184 }
185 }
186 }
187 }
188 });
189}

192// Scalar implementation for arbitrary p (also used when CV_SIMD is disabled).
193static void lpPoolScalar(const float* inp_, float* out_, const ConvState& cs, int p)
194{
195 int NC1 = cs.inpshape[0]*cs.inpshape[1];
196
197 CV_Assert(cs.inpshape.layout == DATA_LAYOUT_BLOCK);
198 CV_Assert(cs.outshape.layout == DATA_LAYOUT_BLOCK);
199 CV_Assert(cs.inpshape.dims == cs.outshape.dims);
200
201 parallel_for_(Range(0, NC1), [&](const Range& r) {
202 constexpr int MAX_POOL_DIMS = ConvState::MAX_CONV_DIMS;
203
204 CV_Assert(cs.nspatialdims <= MAX_POOL_DIMS && MAX_POOL_DIMS == 3);
205
206 int sdims = cs.nspatialdims;
207 int nc0 = r.start, nc1 = r.end;
208 int C0 = cs.inpshape.back();
209 int Di = sdims > 2 ? cs.inpshape[sdims - 1] : 1;
210 int Hi = sdims > 1 ? cs.inpshape[sdims] : 1;
211 int Wi = cs.inpshape[sdims + 1];
212 int D = sdims > 2 ? cs.outshape[sdims - 1] : 1;
213 int H = sdims > 1 ? cs.outshape[sdims] : 1;
214 int W = cs.outshape[sdims + 1];
215 int iplanesize = Di*Hi*Wi*C0;
216 int planesize = D*H*W*C0;
217 int SZ = cs.strides[0], SY = cs.strides[1], SX = cs.strides[2];
218 int padZ0 = cs.pads[0], padY0 = cs.pads[1], padX0 = cs.pads[2];
219 int inner_z0 = cs.inner[0], inner_z1 = cs.inner[MAX_POOL_DIMS];
220 int inner_y0 = cs.inner[1], inner_y1 = cs.inner[MAX_POOL_DIMS + 1];
221 int inner_x0 = cs.inner[2], inner_x1 = cs.inner[MAX_POOL_DIMS + 2];
222 int ksize = (int)cs.ofstab.size();
223 const int* zyxtab = cs.coordtab.data();
224 const int* ofstab = cs.ofstab.data();
225 float inv_p = 1.f / (float)p;
226
227 const float* inp = inp_ + nc0*iplanesize;
228 float* out = out_ + nc0*planesize;
229
230 for (int nc = nc0; nc < nc1; nc++, inp += iplanesize) {
231 for (int z0 = 0; z0 < D; z0++) {
232 int zi_ = z0*SZ - padZ0;
233 for (int y0 = 0; y0 < H; y0++, out += W*C0) {
234 int x0 = 0;
235 int x1 = z0 >= inner_z0 && z0 < inner_z1 &&
236 y0 >= inner_y0 && y0 < inner_y1 ? inner_x0 : W;
237 int yi_ = y0*SY - padY0;
238
239 for(;;) {
240 // Boundary (outer) path — needs per-element bounds check
241 for (; x0 < x1; x0++) {
242 int xi_ = x0*SX - padX0;
243 for (int c = 0; c < C0; c++)
244 out[x0*C0 + c] = 0.f;
245 for (int k = 0; k < ksize; k++) {
246 int zi = zi_ + zyxtab[k*MAX_POOL_DIMS];
247 int yi = yi_ + zyxtab[k*MAX_POOL_DIMS+1];
248 int xi = xi_ + zyxtab[k*MAX_POOL_DIMS+2];
249 if ((unsigned)zi >= (unsigned)Di ||
250 (unsigned)yi >= (unsigned)Hi ||
251 (unsigned)xi >= (unsigned)Wi)
252 continue;
253 const float* inptr = inp + ((zi*Hi + yi)*Wi + xi)*C0;
254 if (p == 1) {
255 for (int c = 0; c < C0; c++)
256 out[x0*C0 + c] += std::abs(inptr[c]);
257 } else if (p == 2) {
258 for (int c = 0; c < C0; c++)
259 out[x0*C0 + c] += inptr[c] * inptr[c];
260 } else {
261 for (int c = 0; c < C0; c++)
262 out[x0*C0 + c] += std::pow(std::abs(inptr[c]), (float)p);
263 }
264 }
265 if (p == 2) {
266 for (int c = 0; c < C0; c++)
267 out[x0*C0 + c] = std::sqrt(out[x0*C0 + c]);
268 } else if (p != 1) {
269 for (int c = 0; c < C0; c++)
270 out[x0*C0 + c] = std::pow(out[x0*C0 + c], inv_p);
271 }
272 }
273
274 if (x0 == W)
275 break;
276 x1 = inner_x1;
277
278 // Inner path — no bounds check needed
279 for (; x0 < x1; x0++) {
280 int xi_ = x0*SX - padX0;
281 const float* inp_xi = inp + ((Hi*zi_ + yi_)*Wi + xi_)*C0;
282 for (int c = 0; c < C0; c++)
283 out[x0*C0 + c] = 0.f;
284 if (p == 1) {
285 for (int k = 0; k < ksize; k++) {
286 const float* inptr = inp_xi + ofstab[k];
287 for (int c = 0; c < C0; c++)
288 out[x0*C0 + c] += std::abs(inptr[c]);
289 }
290 } else if (p == 2) {
291 for (int k = 0; k < ksize; k++) {
292 const float* inptr = inp_xi + ofstab[k];
293 for (int c = 0; c < C0; c++)
294 out[x0*C0 + c] += inptr[c] * inptr[c];
295 }
296 for (int c = 0; c < C0; c++)
297 out[x0*C0 + c] = std::sqrt(out[x0*C0 + c]);
298 } else {
299 for (int k = 0; k < ksize; k++) {
300 const float* inptr = inp_xi + ofstab[k];
301 for (int c = 0; c < C0; c++)
302 out[x0*C0 + c] += std::pow(std::abs(inptr[c]), (float)p);
303 }
304 for (int c = 0; c < C0; c++)
305 out[x0*C0 + c] = std::pow(out[x0*C0 + c], inv_p);
306 }
307 }
308 x1 = W;
309 }
310 }
311 }
312 }
313 });
314}

316static void lpPool32f(const void* inp_, void* out_, const ConvState& cs, int p)
317{
318 const float* inp = (const float*)inp_;
319 float* out = (float*)out_;
320#if CV_SIMD
321 // SIMD fast paths for the common norms; scalar fallback for any other p.
322 if (p == 1) {
323 lpPoolSIMD<LpPoolL1>(inp, out, cs);
324 return;
325 }
326 if (p == 2) {
327 lpPoolSIMD<LpPoolL2>(inp, out, cs);
328 return;
329 }
330#endif
331 lpPoolScalar(inp, out, cs, p);
332}

430 void forward(InputArrayOfArrays inputs_arr,
431 OutputArrayOfArrays outputs_arr,
432 OutputArrayOfArrays) CV_OVERRIDE
433 {
434 CV_Assert(inputs_arr.total() == 1);
435
436 int inptype = inputs_arr.type(0);
437 MatShape inpshape = inputs_arr.shape(0);
438 MatShape outshape = convInferShape(inpshape, MatShape(),
439 kernel_shape, 0, strides, dilations,
440 pads, auto_pad, ceil_mode);
441 int outKind = outputs_arr.kind();
442 CV_Assert(outKind == _InputArray::STD_VECTOR_MAT ||
443 outKind == _InputArray::STD_VECTOR_UMAT);
444
445 ConvState cs;
446 cs.initPooling(inpshape, outshape, kernel_shape, strides,
447 dilations, pads, auto_pad, ceil_mode);
448
449 if (outKind == _InputArray::STD_VECTOR_MAT) {
450 Mat inp = inputs_arr.getMat(0);
451 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
452 outs.resize(1);
453 outs[0].fit(outshape, inptype);
454 runOp(inp, outs[0], cs);
455 } else {
456 Mat inp = inputs_arr.getMat(0);
457 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
458 outs.resize(1);
459 outs[0].fit(outshape, inptype);
460 Mat temp(outshape, inptype);
461 runOp(inp, temp, cs);
462 temp.copyTo(outs[0]);
463 }
464 }
