53// This method is the same as reconstructPointsFor3View, with only a few numbers adjusted for two-view geometry
54void triangulatePoints( InputArray _P1, InputArray _P2,
55 InputArray _points1, InputArray _points2,
56 OutputArray _points4D )
57{
58 CV_INSTRUMENT_REGION();
59
60 Mat points1 = _points1.getMat(), points2 = _points2.getMat();
61 int depth1 = points1.depth(), depth2 = points2.depth();
62 const float *p1f = depth1 == CV_32F ? points1.ptr<float>() : 0;
63 const float *p2f = depth2 == CV_32F ? points2.ptr<float>() : 0;
64 const double *p1d = depth1 == CV_64F ? points1.ptr<double>() : 0;
65 const double *p2d = depth2 == CV_64F ? points2.ptr<double>() : 0;
66 int pstep1, ystep1, pstep2, ystep2, npoints1, npoints2;
67
68 CV_Assert(depth1 == depth2 && (depth1 == CV_32F || depth1 == CV_64F));
69
70 if ((points1.rows == 1 || points1.cols == 1) && points1.channels() == 2)
71 {
72 npoints1 = points1.rows + points1.cols - 1;
73 ystep1 = 1;
74 pstep1 = 2;
75 }
76 else
77 {
78 npoints1 = points1.cols;
79 ystep1 = (int)(points1.step/points1.elemSize());
80 pstep1 = 1;
81 }
82
83 if ((points2.rows == 1 || points2.cols == 1) && points2.channels() == 2)
84 {
85 npoints2 = points2.rows + points2.cols - 1;
86 ystep2 = 1;
87 pstep2 = 2;
88 }
89 else
90 {
91 npoints2 = points2.cols;
92 ystep2 = (int)(points2.step/points2.elemSize());
93 pstep2 = 1;
94 }
95
96 CV_Assert(npoints1 == npoints2);
97
98 _points4D.create(4, npoints1, depth1);
99 Mat points4D = _points4D.getMat();
100
101 Matx<double, 4, 4> matrA;
102 Matx<double, 4, 4> matrU;
103 Matx<double, 4, 1> matrW;
104 Matx<double, 4, 4> matrV;
105 size_t step4 = 4*sizeof(double);
106 Matx<double, 3, 4> P1;
107 Matx<double, 3, 4> P2;
108 _P1.getMat().convertTo(P1, CV_64F);
109 _P2.getMat().convertTo(P2, CV_64F);
110
111 // Solve system for each point
112 for( int i = 0; i < npoints1; i++ )
113 {
114 // Fill matrix for current point
115 double x1 = p1f ? (double)p1f[pstep1*i] : p1d[pstep1*i];
116 double y1 = p1f ? (double)p1f[pstep1*i + ystep1] : p1d[pstep1*i + ystep1];
117 double x2 = p2f ? (double)p2f[pstep2*i] : p2d[pstep2*i];
118 double y2 = p2f ? (double)p2f[pstep2*i + ystep2] : p2d[pstep2*i + ystep2];
119
120 for(int k = 0; k < 4; k++)
121 {
122 matrA(k, 0) = x1*P1(2, k) - P1(0, k);
123 matrA(k, 1) = y1*P1(2, k) - P1(1, k);
124 matrA(k, 2) = x2*P2(2, k) - P2(0, k);
125 matrA(k, 3) = y2*P2(2, k) - P2(1, k);
126 }
127
128 // Solve system for current point
129 hal::SVD64f(matrA.val, step4, matrW.val, matrU.val, step4, matrV.val, step4, 4, 4, 4);
130
131 // Copy computed point
132 if(depth1 == CV_32F)
133 {
134 points4D.at<float>(0, i) = (float)matrV(3, 0);
135 points4D.at<float>(1, i) = (float)matrV(3, 1);
136 points4D.at<float>(2, i) = (float)matrV(3, 2);
137 points4D.at<float>(3, i) = (float)matrV(3, 3);
138 }
139 else
140 {
141 points4D.at<double>(0, i) = matrV(3, 0);
142 points4D.at<double>(1, i) = matrV(3, 1);
143 points4D.at<double>(2, i) = matrV(3, 2);
144 points4D.at<double>(3, i) = matrV(3, 3);
145 }
146 }
147}

149/*
150 * The Optimal Triangulation Method (see HZ for details)
151 * For each given point correspondence points1[i] <-> points2[i], and a fundamental matrix F,
152 * computes the corrected correspondences new_points1[i] <-> new_points2[i] that minimize the
153 * geometric error d(points1[i],new_points1[i])^2 + d(points2[i],new_points2[i])^2 (where d(a,b)
154 * is the geometric distance between points a and b) subject to the epipolar constraint
155 * new_points2' * F * new_points1 = 0.
156 *
157 * _F : 3x3 fundamental matrix
158 * _points1 : 1xN matrix containing the first set of points
159 * _points2 : 1xN matrix containing the second set of points
160 * _newPoints1 : the optimized _points1.
161 * _newPoints2 : the optimized -points2.
162 */
163void correctMatches( InputArray _F, InputArray _points1, InputArray _points2,
164 OutputArray _newPoints1, OutputArray _newPoints2 )
165{
166 CV_INSTRUMENT_REGION();
167
168 Mat points1 = _points1.getMat(), points2 = _points2.getMat();
169
170 int depth1 = points1.depth(), depth2 = points2.depth();
171 CV_Assert((depth1 == CV_32F || depth1 == CV_64F) && depth1 == depth2);
172
173 CV_Assert(points1.size() == points2.size());
174 CV_Assert(points1.rows == 1 || points1.cols == 1);
175 if (points1.channels() != 2)
176 CV_Error( cv::Error::StsUnmatchedSizes, "The first set of points must contain two channels; one for x and one for y" );
177 if (points2.channels() != 2)
178 CV_Error( cv::Error::StsUnmatchedSizes, "The second set of points must contain two channels; one for x and one for y" );
179
180 _newPoints1.create(points1.size(), points1.type());
181 _newPoints2.create(points2.size(), points2.type());
182 Mat newPoints1 = _newPoints1.getMat(), newPoints2 = _newPoints2.getMat();
183
184 Matx33d F, U, Vt;
185 Matx31d S;
186 int npoints = points1.rows + points1.cols - 1;
187
188 // Make sure F uses double precision
189 _F.getMat().convertTo(F, CV_64F);
190
191 for (int p = 0; p < npoints; ++p) {
192 // Replace F by T2-t * F * T1-t
193 double x1, y1, x2, y2;
194 if (depth1 == CV_32F) {
195 Point2f p1 = points1.at<Point2f>(p);
196 Point2f p2 = points2.at<Point2f>(p);
197 x1 = p1.x; y1 = p1.y;
198 x2 = p2.x; y2 = p2.y;
199 } else {
200 Point2d p1 = points1.at<Point2d>(p);
201 Point2d p2 = points2.at<Point2d>(p);
202 x1 = p1.x; y1 = p1.y;
203 x2 = p2.x; y2 = p2.y;
204 }
205
206 Matx33d T1i(1, 0, x1,
207 0, 1, y1,
208 0, 0, 1);
209 Matx33d T2i(1, 0, x2,
210 0, 1, y2,
211 0, 0, 1);
212 Matx33d TFT = T2i.t()*F*T1i;
213
214 // Compute the right epipole e1 from F * e1 = 0
215 SVDecomp(TFT, S, U, Vt);
216 double scale = sqrt(Vt(2, 0)*Vt(2, 0) + Vt(2, 1)*Vt(2, 1));
217
218 Vec3d e1(Vt(2, 0)/scale, Vt(2, 1)/scale, Vt(2, 2)/scale);
219 if (e1(2) < 0)
220 e1 = -e1;
221
222 // Compute the left epipole e2 from e2' * F = 0 => F' * e2 = 0
223 scale = sqrt(U(0, 2)*U(0, 2) + U(1, 2)*U(1, 2));
224
225 Vec3d e2(U(0, 2)/scale, U(1, 2)/scale, U(2, 2)/scale);
226 if (e2(2) < 0)
227 e2 = -e2;
228
229 // Replace F by R2 * F * R1'
230 Matx33d R1_t(e1(0), -e1(1), 0,
231 e1(1), e1(0), 0,
232 0, 0, 1);
233 Matx33d R2(e2(0), e2(1), 0,
234 -e2(1), e2(0), 0,
235 0, 0, 1);
236 Matx33d RTFTR = R2*TFT*R1_t;
237
238 // Set f1 = e1(3), f2 = e2(3), a = F22, b = F23, c = F32, d = F33
239 double f1 = e1(2);
240 double f2 = e2(2);
241 double a = RTFTR(1,1);
242 double b = RTFTR(1,2);
243 double c = RTFTR(2,1);
244 double d = RTFTR(2,2);
245
246 // Form the polynomial g(t) = k6*t^6 + k5*t^5 + k4*t^4 + k3*t^3 + k2*t^2 + k1*t + k0
247 // from f1, f2, a, b, c and d
248 Vec<double, 7> polynomial(
249 -a*d*d*b+b*b*c*d,
250 +f2*f2*f2*f2*d*d*d*d+b*b*b*b+2*b*b*f2*f2*d*d-a*a*d*d+b*b*c*c,
251 +4*a*b*b*b+4*b*b*f2*f2*c*d+4*f2*f2*f2*f2*c*d*d*d-a*a*d*c+b*c*c*a+4*a*b*f2*f2*d*d-2*a*d*d*f1*f1*b+2*b*b*c*f1*f1*d,
252 +6*a*a*b*b+6*f2*f2*f2*f2*c*c*d*d+2*b*b*f2*f2*c*c+2*a*a*f2*f2*d*d-2*a*a*d*d*f1*f1+2*b*b*c*c*f1*f1+8*a*b*f2*f2*c*d,
253 +4*a*a*a*b+2*b*c*c*f1*f1*a+4*f2*f2*f2*f2*c*c*c*d+4*a*b*f2*f2*c*c+4*a*a*f2*f2*c*d-2*a*a*d*f1*f1*c-a*d*d*f1*f1*f1*f1*b+b*b*c*f1*f1*f1*f1*d,
254 +f2*f2*f2*f2*c*c*c*c+2*a*a*f2*f2*c*c-a*a*d*d*f1*f1*f1*f1+b*b*c*c*f1*f1*f1*f1+a*a*a*a,
255 +b*c*c*f1*f1*f1*f1*a-a*a*d*f1*f1*f1*f1*c);
256
257 // Solve g(t) for t to get 6 roots
258 double rdata[6*2];
259 Mat result(6, 1, CV_64FC2, rdata);
260 solvePoly(polynomial, result);
261
262 // Evaluate the cost function s(t) at the real part of the 6 roots
263 double t_min = DBL_MAX;
264 double s_val = 1./(f1*f1) + (c*c)/(a*a+f2*f2*c*c);
265 for (int ti = 0; ti < 6; ++ti) {
266 Vec2d root_i = result.at<Vec2d>(ti);
267 double t = root_i(0);
268 double s = (t*t)/(1 + f1*f1*t*t) + ((c*t + d)*(c*t + d))/((a*t + b)*(a*t + b) + f2*f2*(c*t + d)*(c*t + d));
269 if (s < s_val) {
270 s_val = s;
271 t_min = t;
272 }
273 }
274
275 // find the optimal x1 and y1 as the points on l1 and l2 closest to the origin
276 scale = t_min*t_min*f1*f1+1;
277 Vec3d tmp31(t_min*t_min*f1/scale, t_min/scale, 1);
278 Vec3d tmp31_2 = T1i*(R1_t*tmp31);
279 x1 = tmp31_2(0);
280 y1 = tmp31_2(1);
281
282 scale = f2*f2*(c*t_min+d)*(c*t_min+d) + (a*t_min+b)*(a*t_min+b);
283 tmp31 = Vec3d(f2*(c*t_min+d)*(c*t_min+d)/scale, -(a*t_min+b)*(c*t_min+d)/scale, 1);
284 tmp31_2 = T2i*(R2.t()*tmp31);
285 x2 = tmp31_2(0);
286 y2 = tmp31_2(1);
287
288 // Return the points in the matrix format that the user wants
289 if (depth1 == CV_32F) {
290 newPoints1.at<Point2f>(p) = Point2f((float)x1, (float)y1);
291 newPoints2.at<Point2f>(p) = Point2f((float)x2, (float)y2);
292 } else {
293 newPoints1.at<Point2d>(p) = Point2d(x1, y1);
294 newPoints2.at<Point2d>(p) = Point2d(x2, y2);
295 }
296 }
297}
