33int numpyTypeToCvDepth(int typenum)
34{
35 // Only canonical NPY_* values may be used as case labels: the fixed-width
36 // aliases (NPY_INT32, NPY_INT64, ...) expand to these and would collide.
37 switch (typenum)
38 {
39 case NPY_UBYTE: return CV_8U;
40 case NPY_BYTE: return CV_8S;
41 case NPY_USHORT: return CV_16U;
42 case NPY_SHORT: return CV_16S;
43 case NPY_UINT: return CV_32U;
44 case NPY_INT: return CV_32S;
45 case NPY_ULONGLONG: return CV_64U;
46 // 'long' is 64-bit on LP64 but 32-bit on LLP64, so decide by size, not by name.
47 case NPY_ULONG: return NPY_SIZEOF_LONG == 8 ? CV_64U : CV_32U;
48 case NPY_HALF: return CV_16F;
49 case NPY_FLOAT: return CV_32F;
50 case NPY_DOUBLE: return CV_64F;
51 case NPY_BOOL: return CV_Bool;
52 default: return -1;
53 }
54}

127void pyRaiseCVOverloadException(const std::string& functionName)
128{
129 const std::vector<std::string>& conversionErrors = conversionErrorsTLS.getRef();
130 const std::size_t conversionErrorsCount = conversionErrors.size();
131 if (conversionErrorsCount > 0)
132 {
133 // In modern std libraries small string optimization is used = no dynamic memory allocations,
134 // but it can be applied only for string with length < 18 symbols (in GCC)
135 const std::string bullet = "\n - ";
136
137 // Estimate required buffer size - save dynamic memory allocations = faster
138 std::size_t requiredBufferSize = bullet.size() * conversionErrorsCount;
139 for (std::size_t i = 0; i < conversionErrorsCount; ++i)
140 {
141 requiredBufferSize += conversionErrors[i].size();
142 }
143
144 // Only string concatenation is required so std::string is way faster than
145 // std::ostringstream
146 std::string errorMessage("Overload resolution failed:");
147 errorMessage.reserve(errorMessage.size() + requiredBufferSize);
148 for (std::size_t i = 0; i < conversionErrorsCount; ++i)
149 {
150 errorMessage += bullet;
151 errorMessage += conversionErrors[i];
152 }
153 cv::Exception exception(Error::StsBadArg, errorMessage, functionName, "", -1);
154 pyRaiseCVException(exception);
155 }
156 else
157 {
158 cv::Exception exception(Error::StsInternal, "Overload resolution failed, but no errors reported",
159 functionName, "", -1);
160 pyRaiseCVException(exception);
161 }
162}

184static int OnError(int status, const char *func_name, const char *err_msg, const char *file_name, int line, void *userdata)
185{
186 PyGILState_STATE gstate;
187 gstate = PyGILState_Ensure();
188
189 PyObject *on_error = (PyObject*)userdata;
190 PyObject *args = Py_BuildValue("isssi", status, func_name, err_msg, file_name, line);
191
192 PyObject *r = PyObject_Call(on_error, args, NULL);
193 if (r == NULL) {
194 PyErr_Print();
195 } else {
196 Py_DECREF(r);
197 }
198
199 Py_DECREF(args);
200 PyGILState_Release(gstate);
201
202 return 0; // The return value isn't used
203}

205PyObject *pycvRedirectError(PyObject*, PyObject *args, PyObject *kw)
206{
207 const char *keywords[] = { "on_error", NULL };
208 PyObject *on_error;
209
210 if (!PyArg_ParseTupleAndKeywords(args, kw, "O", (char**)keywords, &on_error))
211 return NULL;
212
213 if ((on_error != Py_None) && !PyCallable_Check(on_error)) {
214 PyErr_SetString(PyExc_TypeError, "on_error must be callable");
215 return NULL;
216 }
217
218 // Keep track of the previous handler parameter, so we can decref it when no longer used
219 static PyObject* last_on_error = NULL;
220 if (last_on_error) {
221 Py_DECREF(last_on_error);
222 last_on_error = NULL;
223 }
224
225 if (on_error == Py_None) {
226 ERRWRAP2(redirectError(NULL));
227 } else {
228 last_on_error = on_error;
229 Py_INCREF(last_on_error);
230 ERRWRAP2(redirectError(OnError, last_on_error));
231 }
232 Py_RETURN_NONE;
233}
