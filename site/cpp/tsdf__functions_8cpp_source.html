14void preCalculationPixNorm(Size size, const Intr& intrinsics, Mat& pixNorm)
15{
16 CV_TRACE_FUNCTION();
17
18 Point2f fl(intrinsics.fx, intrinsics.fy);
19 Point2f pp(intrinsics.cx, intrinsics.cy);
20 pixNorm = Mat(size.height, size.width, CV_32F);
21 std::vector<float> x(size.width);
22 std::vector<float> y(size.height);
23 for (int i = 0; i < size.width; i++)
24 x[i] = (i - pp.x) / fl.x;
25 for (int i = 0; i < size.height; i++)
26 y[i] = (i - pp.y) / fl.y;
27
28 for (int i = 0; i < size.height; i++)
29 {
30 for (int j = 0; j < size.width; j++)
31 {
32 pixNorm.at<float>(i, j) = sqrtf(x[j] * x[j] + y[i] * y[i] + 1.0f);
33 }
34 }
35
36}

60void integrateTsdfVolumeUnit(const VolumeSettings& settings, const Matx44f& volumePose, const Matx44f& cameraPose,
61 InputArray _depth, InputArray _pixNorms, InputArray _volume)
62{
63 Depth depth = _depth.getMat();
64 Mat volume = _volume.getMat();
65 Mat pixNorms = _pixNorms.getMat();
66
67 TsdfVoxel* volDataStart = volume.ptr<TsdfVoxel>();
68
69 Vec4i volStrides;
70 settings.getVolumeStrides(volStrides);
71
72 Vec3i resolution;
73 settings.getVolumeResolution(resolution);
74 const Point3i volResolution = Point3i(resolution);
75 float voxelSize = settings.getVoxelSize();
76
77 const Affine3f pose = Affine3f(volumePose);
78 const Affine3f vol2cam(Affine3f(cameraPose.inv()) * pose);
79
80 Matx33f intr;
81 settings.getCameraIntegrateIntrinsics(intr);
82 const Intr::Projector proj = Intr(intr).makeProjector();
83 const float dfac(1.f / settings.getDepthFactor());
84 const float truncDist = settings.getTsdfTruncateDistance();
85 const float truncDistInv = 1.f / truncDist;
86 const int maxWeight = settings.getMaxWeight();
87
88 Range integrateRange(0, volResolution.x);
89
90#if USE_INTRINSICS
91 auto IntegrateInvoker = [&](const Range& range)
92 {
93 // zStep == vol2cam*(Point3f(x, y, 1)*voxelSize) - basePt;
94 Point3f zStepPt = Point3f(vol2cam.matrix(0, 2),
95 vol2cam.matrix(1, 2),
96 vol2cam.matrix(2, 2)) * voxelSize;
97
98 v_float32x4 zStep(zStepPt.x, zStepPt.y, zStepPt.z, 0);
99 v_float32x4 vfxy(proj.fx, proj.fy, 0.f, 0.f), vcxy(proj.cx, proj.cy, 0.f, 0.f);
100 const v_float32x4 upLimits = v_cvt_f32(v_int32x4(depth.cols - 1, depth.rows - 1, 0, 0));
101
102 for (int x = range.start; x < range.end; x++)
103 {
104 TsdfVoxel* volDataX = volDataStart + x * volStrides[0];
105 for (int y = 0; y < volResolution.y; y++)
106 {
107 TsdfVoxel* volDataY = volDataX + y * volStrides[1];
108 // optimization of camSpace transformation (vector addition instead of matmul at each z)
109 Point3f basePt = vol2cam * (Point3f((float)x, (float)y, 0) * voxelSize);
110 v_float32x4 camSpacePt(basePt.x, basePt.y, basePt.z, 0);
111
112 int startZ, endZ;
113 if (abs(zStepPt.z) > 1e-5)
114 {
115 int baseZ = (int)(-basePt.z / zStepPt.z);
116 if (zStepPt.z > 0)
117 {
118 startZ = baseZ;
119 endZ = volResolution.z;
120 }
121 else
122 {
123 startZ = 0;
124 endZ = baseZ;
125 }
126 }
127 else
128 {
129 if (basePt.z > 0)
130 {
131 startZ = 0;
132 endZ = volResolution.z;
133 }
134 else
135 {
136 // z loop shouldn't be performed
137 startZ = endZ = 0;
138 }
139 }
140 startZ = max(0, startZ);
141 endZ = min(int(volResolution.z), endZ);
142 for (int z = startZ; z < endZ; z++)
143 {
144 // optimization of the following:
145 //Point3f volPt = Point3f(x, y, z)*voxelSize;
146 //Point3f camSpacePt = vol2cam * volPt;
147 camSpacePt = v_add(camSpacePt, zStep);
148
149 float zCamSpace = v_get0(v_reinterpret_as_f32(v_rotate_right<2>(v_reinterpret_as_u32(camSpacePt))));
150 if (zCamSpace <= 0.f)
151 continue;
152
153 v_float32x4 camPixVec = v_div(camSpacePt, v_setall_f32(zCamSpace));
154 v_float32x4 projected = v_muladd(camPixVec, vfxy, vcxy);
155 // leave only first 2 lanes
156 projected = v_reinterpret_as_f32(v_and(v_reinterpret_as_u32(projected),
157 v_uint32x4(0xFFFFFFFF, 0xFFFFFFFF, 0, 0)));
158
159 depthType v;
160 // bilinearly interpolate depth at projected
161 {
162 const v_float32x4& pt = projected;
163 // check coords >= 0 and < imgSize; NaN passes both, so reject it explicitly
164 v_uint32x4 limits = v_or(v_reinterpret_as_u32(v_lt(pt, v_setzero_f32())),
165 v_reinterpret_as_u32(v_ge(pt, upLimits)));
166 limits = v_or(limits, v_not(v_reinterpret_as_u32(v_not_nan(pt))));
167 limits = v_or(limits, v_rotate_right<1>(limits));
168 if (v_get0(limits))
169 continue;
170
171 // xi, yi = floor(pt)
172 v_int32x4 ip = v_floor(pt);
173 v_int32x4 ipshift = ip;
174 int xi = v_get0(ipshift);
175 ipshift = v_rotate_right<1>(ipshift);
176 int yi = v_get0(ipshift);
177
178 const depthType* row0 = depth[yi + 0];
179 const depthType* row1 = depth[yi + 1];
180
181 // v001 = [v(xi + 0, yi + 0), v(xi + 1, yi + 0)]
182 v_float32x4 v001 = v_load_low(row0 + xi);
183 // v101 = [v(xi + 0, yi + 1), v(xi + 1, yi + 1)]
184 v_float32x4 v101 = v_load_low(row1 + xi);
185
186 v_float32x4 vall = v_combine_low(v001, v101);
187
188 // assume correct depth is positive
189 // don't fix missing data
190 if (v_check_all(v_gt(vall, v_setzero_f32())))
191 {
192 v_float32x4 t = v_sub(pt, v_cvt_f32(ip));
193 float tx = v_get0(t);
194 t = v_reinterpret_as_f32(v_rotate_right<1>(v_reinterpret_as_u32(t)));
195 v_float32x4 ty = v_setall_f32(v_get0(t));
196 // vx is y-interpolated between rows 0 and 1
197 v_float32x4 vx = v_add(v001, v_mul(ty, v_sub(v101, v001)));
198 float v0 = v_get0(vx);
199 vx = v_reinterpret_as_f32(v_rotate_right<1>(v_reinterpret_as_u32(vx)));
200 float v1 = v_get0(vx);
201 v = v0 + tx * (v1 - v0);
202 }
203 else
204 continue;
205 }
206
207 // norm(camPixVec) produces double which is too slow
208 int _u = (int)v_get0(projected);
209 int _v = (int)v_get0(v_rotate_right<1>(projected));
210 if (!(_u >= 0 && _u < depth.cols && _v >= 0 && _v < depth.rows))
211 continue;
212 float pixNorm = pixNorms.at<float>(_v, _u);
213 // float pixNorm = sqrt(v_reduce_sum(camPixVec*camPixVec));
214 // difference between distances of point and of surface to camera
215 float sdf = pixNorm * (v * dfac - zCamSpace);
216 // possible alternative is:
217 // kftype sdf = norm(camSpacePt)*(v*dfac/camSpacePt.z - 1);
218
219 if (sdf >= -truncDist)
220 {
221 TsdfType tsdf = floatToTsdf(fmin(1.f, sdf * truncDistInv));
222
223 TsdfVoxel& voxel = volDataY[z * volStrides[2]];
224 WeightType& weight = voxel.weight;
225 TsdfType& value = voxel.tsdf;
226
227 // update TSDF
228 value = floatToTsdf((tsdfToFloat(value) * weight + tsdfToFloat(tsdf)) / (weight + 1));
229 weight = (weight + 1) < maxWeight ? (weight + 1) : (WeightType)maxWeight;
230 }
231 }
232 }
233 }
234 };
235#else
236 auto IntegrateInvoker = [&](const Range& range)
237 {
238 for (int x = range.start; x < range.end; x++)
239 {
240 TsdfVoxel* volDataX = volDataStart + x * volStrides[0];
241 for (int y = 0; y < volResolution.y; y++)
242 {
243 TsdfVoxel* volDataY = volDataX + y * volStrides[1];
244 // optimization of camSpace transformation (vector addition instead of matmul at each z)
245 Point3f basePt = vol2cam * (Point3f(float(x), float(y), 0.0f) * voxelSize);
246 Point3f camSpacePt = basePt;
247 // zStep == vol2cam*(Point3f(x, y, 1)*voxelSize) - basePt;
248 // zStep == vol2cam*[Point3f(x, y, 1) - Point3f(x, y, 0)]*voxelSize
249 Point3f zStep = Point3f(vol2cam.matrix(0, 2),
250 vol2cam.matrix(1, 2),
251 vol2cam.matrix(2, 2)) * voxelSize;
252 int startZ, endZ;
253 if (abs(zStep.z) > 1e-5)
254 {
255 int baseZ = int(-basePt.z / zStep.z);
256 if (zStep.z > 0)
257 {
258 startZ = baseZ;
259 endZ = volResolution.z;
260 }
261 else
262 {
263 startZ = 0;
264 endZ = baseZ;
265 }
266 }
267 else
268 {
269 if (basePt.z > 0)
270 {
271 startZ = 0;
272 endZ = volResolution.z;
273 }
274 else
275 {
276 // z loop shouldn't be performed
277 startZ = endZ = 0;
278 }
279 }
280 startZ = max(0, startZ);
281 endZ = min(int(volResolution.z), endZ);
282
283 for (int z = startZ; z < endZ; z++)
284 {
285 // optimization of the following:
286 //Point3f volPt = Point3f(x, y, z)*volume.voxelSize;
287 //Point3f camSpacePt = vol2cam * volPt;
288
289 camSpacePt += zStep;
290 if (camSpacePt.z <= 0)
291 continue;
292
293 Point3f camPixVec;
294 Point2f projected = proj(camSpacePt, camPixVec);
295
296 depthType v = bilinearDepth(depth, projected);
297 if (v == 0) {
298 continue;
299 }
300 int _u = (int)projected.x;
301 int _v = (int)projected.y;
302 if (!(_u >= 0 && _u < depth.cols && _v >= 0 && _v < depth.rows))
303 continue;
304
305 float pixNorm = pixNorms.at<float>(_v, _u);
306
307 // difference between distances of point and of surface to camera
308 float sdf = pixNorm * (v * dfac - camSpacePt.z);
309 // possible alternative is:
310 // kftype sdf = norm(camSpacePt)*(v*dfac/camSpacePt.z - 1);
311 if (sdf >= -truncDist)
312 {
313 TsdfType tsdf = floatToTsdf(fmin(1.f, sdf * truncDistInv));
314
315 TsdfVoxel& voxel = volDataY[z * volStrides[2]];
316 WeightType& weight = voxel.weight;
317 TsdfType& value = voxel.tsdf;
318
319 // update TSDF
320 value = floatToTsdf((tsdfToFloat(value) * weight + tsdfToFloat(tsdf)) / (weight + 1));
321 weight = (WeightType)min(weight + 1, maxWeight);
322 }
323 }
324 }
325 }
326 };
327#endif
328 parallel_for_(integrateRange, IntegrateInvoker);
329 //IntegrateInvoker(integrateRange);
330}

406inline float interpolateTsdfVoxel(const Mat& volume,
407 const Vec4i& volDims, const Vec8i& neighbourCoords,
408 const v_float32x4& p)
409{
410 // tx, ty, tz = floor(p)
411 v_int32x4 ip = v_floor(p);
412 v_float32x4 t = v_sub(p, v_cvt_f32(ip));
413 float tx = v_get0(t);
414 t = v_reinterpret_as_f32(v_rotate_right<1>(v_reinterpret_as_u32(t)));
415 float ty = v_get0(t);
416 t = v_reinterpret_as_f32(v_rotate_right<1>(v_reinterpret_as_u32(t)));
417 float tz = v_get0(t);
418
419 int xdim = volDims[0], ydim = volDims[1], zdim = volDims[2];
420 const TsdfVoxel* volData = volume.ptr<TsdfVoxel>();
421
422 int ix = v_get0(ip);
423 ip = v_rotate_right<1>(ip);
424 int iy = v_get0(ip);
425 ip = v_rotate_right<1>(ip);
426 int iz = v_get0(ip);
427
428 int coordBase = ix * xdim + iy * ydim + iz * zdim;
429
430 TsdfType vx[8];
431 for (int i = 0; i < 8; i++)
432 vx[i] = volData[neighbourCoords[i] + coordBase].tsdf;
433
434 v_float32x4 v0246 = tsdfToFloat_INTR(v_int32x4(vx[0], vx[2], vx[4], vx[6]));
435 v_float32x4 v1357 = tsdfToFloat_INTR(v_int32x4(vx[1], vx[3], vx[5], vx[7]));
436 v_float32x4 vxx = v_add(v0246, v_mul(v_setall_f32(tz), v_sub(v1357, v0246)));
437
438 v_float32x4 v00_10 = vxx;
439 v_float32x4 v01_11 = v_reinterpret_as_f32(v_rotate_right<1>(v_reinterpret_as_u32(vxx)));
440
441 v_float32x4 v0_1 = v_add(v00_10, v_mul(v_setall_f32(ty), v_sub(v01_11, v00_10)));
442 float v0 = v_get0(v0_1);
443 v0_1 = v_reinterpret_as_f32(v_rotate_right<2>(v_reinterpret_as_u32(v0_1)));
444 float v1 = v_get0(v0_1);
445
446 return v0 + tx * (v1 - v0);
447}

494//gradientDeltaFactor is fixed at 1.0 of voxel size
495inline v_float32x4 getNormalVoxel( const Mat& volume,
496 const Vec4i& volDims, const Vec8i& neighbourCoords, const Point3i volResolution,
497 const v_float32x4& p)
498{
499 if (v_check_any(v_lt(p, v_float32x4(1.f, 1.f, 1.f, 0.f))) ||
500 v_check_any(v_ge(p, v_float32x4((float)(volResolution.x - 2),
501 (float)(volResolution.y - 2),
502 (float)(volResolution.z - 2), 1.f)))
503 )
504 return nanv;
505
506 v_int32x4 ip = v_floor(p);
507 v_float32x4 t = v_sub(p, v_cvt_f32(ip));
508 float tx = v_get0(t);
509 t = v_reinterpret_as_f32(v_rotate_right<1>(v_reinterpret_as_u32(t)));
510 float ty = v_get0(t);
511 t = v_reinterpret_as_f32(v_rotate_right<1>(v_reinterpret_as_u32(t)));
512 float tz = v_get0(t);
513
514 const int xdim = volDims[0], ydim = volDims[1], zdim = volDims[2];
515 const TsdfVoxel* volData = volume.ptr<TsdfVoxel>();
516
517 int ix = v_get0(ip); ip = v_rotate_right<1>(ip);
518 int iy = v_get0(ip); ip = v_rotate_right<1>(ip);
519 int iz = v_get0(ip);
520
521 int coordBase = ix * xdim + iy * ydim + iz * zdim;
522
523 float CV_DECL_ALIGNED(16) an[4];
524 an[0] = an[1] = an[2] = an[3] = 0.f;
525 for (int c = 0; c < 3; c++)
526 {
527 const int dim = volDims[c];
528 float& nv = an[c];
529
530 float vx[8];
531 for (int i = 0; i < 8; i++)
532 vx[i] = tsdfToFloat(volData[neighbourCoords[i] + coordBase + 1 * dim].tsdf) -
533 tsdfToFloat(volData[neighbourCoords[i] + coordBase - 1 * dim].tsdf);
534
535 v_float32x4 v0246(vx[0], vx[2], vx[4], vx[6]);
536 v_float32x4 v1357(vx[1], vx[3], vx[5], vx[7]);
537 v_float32x4 vxx = v_add(v0246, v_mul(v_setall_f32(tz), v_sub(v1357, v0246)));
538
539 v_float32x4 v00_10 = vxx;
540 v_float32x4 v01_11 = v_reinterpret_as_f32(v_rotate_right<1>(v_reinterpret_as_u32(vxx)));
541
542 v_float32x4 v0_1 = v_add(v00_10, v_mul(v_setall_f32(ty), v_sub(v01_11, v00_10)));
543 float v0 = v_get0(v0_1);
544 v0_1 = v_reinterpret_as_f32(v_rotate_right<2>(v_reinterpret_as_u32(v0_1)));
545 float v1 = v_get0(v0_1);
546
547 nv = v0 + tx * (v1 - v0);
548 }
549
550 v_float32x4 n = v_load_aligned(an);
551 v_float32x4 Norm = v_sqrt(v_setall_f32(v_reduce_sum(v_mul(n, n))));
552
553 return v_get0(Norm) < 0.0001f ? nanv : v_div(n, Norm);
554}

575 p.y < 1 || p.y >= volResolution.y - 2 ||
576 p.z < 1 || p.z >= volResolution.z - 2)
577 return nan3;
578
579 int ix = cvFloor(p.x);
580 int iy = cvFloor(p.y);
581 int iz = cvFloor(p.z);
582
583 float tx = p.x - ix;
584 float ty = p.y - iy;
585 float tz = p.z - iz;
586
587 int coordBase = ix * xdim + iy * ydim + iz * zdim;
588
589 Vec3f an;
590 for (int c = 0; c < 3; c++)
591 {
592 const int dim = volDims[c];
593 float& nv = an[c];
594
595 float vx[8];
596 for (int i = 0; i < 8; i++)
597 vx[i] = tsdfToFloat(volData[neighbourCoords[i] + coordBase + 1 * dim].tsdf) -
598 tsdfToFloat(volData[neighbourCoords[i] + coordBase - 1 * dim].tsdf);
599
600 float v00 = vx[0] + tz * (vx[1] - vx[0]);
601 float v01 = vx[2] + tz * (vx[3] - vx[2]);
602 float v10 = vx[4] + tz * (vx[5] - vx[4]);
603 float v11 = vx[6] + tz * (vx[7] - vx[6]);
604
605 float v0 = v00 + ty * (v01 - v00);
606 float v1 = v10 + ty * (v11 - v10);
607
608 nv = v0 + tx * (v1 - v0);
609 }
610
611 float nv = sqrt(an[0] * an[0] +
612 an[1] * an[1] +
613 an[2] * an[2]);
614 return nv < 0.0001f ? nan3 : an / nv;
615}

675
676 const float(&cm)[16] = cam2vol.matrix.val;
677 const v_float32x4 camRot0(cm[0], cm[4], cm[8], 0);
678 const v_float32x4 camRot1(cm[1], cm[5], cm[9], 0);
679 const v_float32x4 camRot2(cm[2], cm[6], cm[10], 0);
680 const v_float32x4 camTrans(cm[3], cm[7], cm[11], 0);
681
682 const v_float32x4 boxDown(boxMin.x, boxMin.y, boxMin.z, 0.f);
683 const v_float32x4 boxUp(boxMax.x, boxMax.y, boxMax.z, 0.f);
684
685 const v_float32x4 invVoxelSize = v_float32x4(voxelSizeInv, voxelSizeInv, voxelSizeInv, 1.f);
686
687 const float(&vm)[16] = vol2cam.matrix.val;
688 const v_float32x4 volRot0(vm[0], vm[4], vm[8], 0);
689 const v_float32x4 volRot1(vm[1], vm[5], vm[9], 0);
690 const v_float32x4 volRot2(vm[2], vm[6], vm[10], 0);
691 const v_float32x4 volTrans(vm[3], vm[7], vm[11], 0);
692
693 for (int y = range.start; y < range.end; y++)
694 {
695 ptype* ptsRow = points[y];
696 ptype* nrmRow = normals[y];
697
698 for (int x = 0; x < points.cols; x++)
699 {
700 v_float32x4 point = nanv, normal = nanv;
701
702 v_float32x4 orig = camTrans;
703
704 // get direction through pixel in volume space:
705
706 // 1. reproject (x, y) on projecting plane where z = 1.f
707 v_float32x4 planed = v_mul(v_sub(v_float32x4((float)x, (float)y, 0.f, 0.f), vcxy), vfxy);
708 planed = v_combine_low(planed, v_float32x4(1.f, 0.f, 0.f, 0.f));
709
710 // 2. rotate to volume space
711 planed = v_matmuladd(planed, camRot0, camRot1, camRot2, v_setzero_f32());
712
713 // 3. normalize
714 v_float32x4 invNorm = v_invsqrt(v_setall_f32(v_reduce_sum(v_mul(planed, planed))));
715 v_float32x4 dir = v_mul(planed, invNorm);
716
717 // compute intersection of ray with all six bbox planes
718 v_float32x4 rayinv = v_div(v_setall_f32(1.f), dir);
719 // div by zero should be eliminated by these products
720 v_float32x4 tbottom = v_mul(rayinv, v_sub(boxDown, orig));
721 v_float32x4 ttop = v_mul(rayinv, v_sub(boxUp, orig));
722
723 // re-order intersections to find smallest and largest on each axis
724 v_float32x4 minAx = v_min(ttop, tbottom);
725 v_float32x4 maxAx = v_max(ttop, tbottom);
726
727 // near clipping plane
728 const float clip = 0.f;
729 float _minAx[4], _maxAx[4];
730 v_store(_minAx, minAx);
731 v_store(_maxAx, maxAx);
732 float tmin = max({ _minAx[0], _minAx[1], _minAx[2], clip });
733 float tmax = min({ _maxAx[0], _maxAx[1], _maxAx[2] });
734
735 // precautions against getting coordinates out of bounds
736 tmin = tmin + tstep;
737 tmax = tmax - tstep;
738
739 if (tmin < tmax)
740 {
741 // interpolation optimized a little
742 orig = v_mul(orig, invVoxelSize);
743 dir = v_mul(dir, invVoxelSize);
744
745 int xdim = volDims[0];
746 int ydim = volDims[1];
747 int zdim = volDims[2];
748 v_float32x4 rayStep = v_mul(dir, v_setall_f32(tstep));
749 v_float32x4 next = v_add(orig, v_mul(dir, v_setall_f32(tmin)));
750 float f = interpolateTsdfVoxel(volume, volDims, neighbourCoords, next);
751 float fnext = f;
752
753 //raymarch
754 int steps = 0;
755 int nSteps = cvFloor((tmax - tmin) / tstep);
756 for (; steps < nSteps; steps++)
757 {
758 next = v_add(next, rayStep);
759 v_int32x4 ip = v_round(next);
760 int ix = v_get0(ip); ip = v_rotate_right<1>(ip);
761 int iy = v_get0(ip); ip = v_rotate_right<1>(ip);
762 int iz = v_get0(ip);
763 int coord = ix * xdim + iy * ydim + iz * zdim;
764
765 fnext = tsdfToFloat(volume.at<TsdfVoxel>(coord).tsdf);
766 if (fnext != f)
767 {
768 fnext = interpolateTsdfVoxel(volume, volDims, neighbourCoords, next);
769
770 // when ray crosses a surface
771 if (std::signbit(f) != std::signbit(fnext))
772 break;
773
774 f = fnext;
775 }
776 }
777
778 // if ray penetrates a surface from outside
779 // linearly interpolate t between two f values
780 if (f > 0.f && fnext < 0.f)
781 {
782 v_float32x4 tp = v_sub(next, rayStep);
783 float ft = interpolateTsdfVoxel(volume, volDims, neighbourCoords, tp);
784 float ftdt = interpolateTsdfVoxel(volume, volDims, neighbourCoords, next);
785 float ts = tmin + tstep * (steps - ft / (ftdt - ft));
786
787 // avoid division by zero
788 if (!cvIsNaN(ts) && !cvIsInf(ts))
789 {
790 v_float32x4 pv = v_add(orig, v_mul(dir, v_setall_f32(ts)));
791 v_float32x4 nv = getNormalVoxel(volume, volDims, neighbourCoords, volResolution, pv);
792
793 if (!isNaN(nv))
794 {
795 //convert pv and nv to camera space
796 normal = v_matmuladd(nv, volRot0, volRot1, volRot2, v_setzero_f32());
797 // interpolation optimized a little
798 point = v_matmuladd(v_mul(pv, v_float32x4(voxelSize, voxelSize, voxelSize, 1.f)),
799 volRot0, volRot1, volRot2, volTrans);
800 }
801 }
802 }
803 }
804
805 v_store((float*)(&ptsRow[x]), point);
806 v_store((float*)(&nrmRow[x]), normal);
807 }
808 }
809 };

830 // compute intersection of ray with all six bbox planes
831 Vec3f rayinv(1.f / dir.x, 1.f / dir.y, 1.f / dir.z);
832 Point3f tbottom = rayinv.mul(boxMin - orig);
833 Point3f ttop = rayinv.mul(boxMax - orig);
834
835 // re-order intersections to find smallest and largest on each axis
836 Point3f minAx(min(ttop.x, tbottom.x), min(ttop.y, tbottom.y), min(ttop.z, tbottom.z));
837 Point3f maxAx(max(ttop.x, tbottom.x), max(ttop.y, tbottom.y), max(ttop.z, tbottom.z));

839 // near clipping plane
840 const float clip = 0.f;
841 //float tmin = max(max(max(minAx.x, minAx.y), max(minAx.x, minAx.z)), clip);
842 //float tmax = min(min(maxAx.x, maxAx.y), min(maxAx.x, maxAx.z));
843 float tmin = max({ minAx.x, minAx.y, minAx.z, clip });
844 float tmax = min({ maxAx.x, maxAx.y, maxAx.z });

846 // precautions against getting coordinates out of bounds
847 tmin = tmin + tstep;
848 tmax = tmax - tstep;
849
850 if (tmin < tmax)
851 {
852 // interpolation optimized a little
853 orig = orig * voxelSizeInv;
854 dir = dir * voxelSizeInv;
855
856 Point3f rayStep = dir * tstep;
857 Point3f next = (orig + dir * tmin);
858 float f = interpolateTsdfVoxel(volume, volDims, neighbourCoords, next);
859 float fnext = f;
860
861 //raymarch
862 int steps = 0;
863 int nSteps = int(floor((tmax - tmin) / tstep));
864 for (; steps < nSteps; steps++)
865 {
866 next += rayStep;
867 int xdim = volDims[0];
868 int ydim = volDims[1];
869 int zdim = volDims[2];
870 int ix = cvRound(next.x);
871 int iy = cvRound(next.y);
872 int iz = cvRound(next.z);
873 fnext = tsdfToFloat(volume.at<TsdfVoxel>(ix * xdim + iy * ydim + iz * zdim).tsdf);
874 if (fnext != f)
875 {
876 fnext = interpolateTsdfVoxel(volume, volDims, neighbourCoords, next);
877 // when ray crosses a surface
878 if (std::signbit(f) != std::signbit(fnext))
879 break;
880
881 f = fnext;
882 }
883 }
884 // if ray penetrates a surface from outside
885 // linearly interpolate t between two f values
886 if (f > 0.f && fnext < 0.f)
887 {
888 Point3f tp = next - rayStep;
889 float ft = interpolateTsdfVoxel(volume, volDims, neighbourCoords, tp);
890 float ftdt = interpolateTsdfVoxel(volume, volDims, neighbourCoords, next);
891 // float t = tmin + steps*tstep;
892 // float ts = t - tstep*ft/(ftdt - ft);
893 float ts = tmin + tstep * (steps - ft / (ftdt - ft));
894
895 // avoid division by zero
896 if (!cvIsNaN(ts) && !cvIsInf(ts))
897 {
898 Point3f pv = (orig + dir * ts);
899 Point3f nv = getNormalVoxel(volume, volDims, neighbourCoords, volResolution, pv);
900
901 if (!isNaN(nv))
902 {
903 //convert pv and nv to camera space
904 normal = volRot * nv;
905 // interpolation optimized a little
906 point = vol2cam * (pv * voxelSize);
907 }
908 }
909 }
910 }
911 ptsRow[x] = toPtype(point);
912 nrmRow[x] = toPtype(normal);
913 }
914 }
915 };

917#endif
918 parallel_for_(raycastRange, RaycastInvoker);
919}
