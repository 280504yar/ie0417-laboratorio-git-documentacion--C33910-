88double TestUtils::checkRectSimilarity(const Size & sz, std::vector<Rect>& ob1, std::vector<Rect>& ob2)
89{
90 double final_test_result = 0.0;
91 size_t sz1 = ob1.size();
92 size_t sz2 = ob2.size();
93
94 if (sz1 != sz2)
95 return sz1 > sz2 ? (double)(sz1 - sz2) : (double)(sz2 - sz1);
96 else
97 {
98 if (sz1 == 0 && sz2 == 0)
99 return 0;
100 cv::Mat cpu_result(sz, CV_8UC1);
101 cpu_result.setTo(0);
102
103 for (vector<Rect>::const_iterator r = ob1.begin(); r != ob1.end(); ++r)
104 {
105 cv::Mat cpu_result_roi(cpu_result, *r);
106 cpu_result_roi.setTo(1);
107 }
108 int cpu_area = cv::countNonZero(cpu_result > 0);
109
110 cv::Mat gpu_result(sz, CV_8UC1);
111 gpu_result.setTo(0);
112 for(vector<Rect>::const_iterator r2 = ob2.begin(); r2 != ob2.end(); ++r2)
113 {
114 cv::Mat gpu_result_roi(gpu_result, *r2);
115 gpu_result_roi.setTo(1);
116 }
117
118 cv::Mat result_;
119 multiply(cpu_result, gpu_result, result_);
120 int result = cv::countNonZero(result_ > 0);
121 if (cpu_area!=0 && result!=0)
122 final_test_result = 1.0 - (double)result/(double)cpu_area;
123 else if(cpu_area==0 && result!=0)
124 final_test_result = -1;
125 }
126 return final_test_result;
127}

130{
131 Mat src = _src.getMat(), actual = _actual.getMat(), gold = _gold.getMat();
132
133 Mat diff, diff_thresh;
134 absdiff(gold, actual, diff);
135 diff.convertTo(diff, CV_32F);
136 threshold(diff, diff_thresh, eps, 255.0, cv::THRESH_BINARY);
137
