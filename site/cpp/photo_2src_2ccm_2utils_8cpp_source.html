14void gammaCorrection(InputArray _src, OutputArray _dst, double gamma)
15{
16 Mat src = _src.getMat();
17 CV_Assert(gamma > 0);
18
19 double maxVal;
20 int depth = src.depth();
21 switch (depth)
22 {
23 case CV_8U: maxVal = 255.0; break;
24 case CV_16U: maxVal = 65535.0; break;
25 case CV_16S: maxVal = 32767.0; break;
26 case CV_32F: maxVal = 1.0; break;
27 case CV_64F: maxVal = 1.0; break;
28 default:
29 CV_Error(Error::StsUnsupportedFormat,
30 "gammaCorrection: unsupported image depth");
31 }
32
33 // Special‐case for uint8 with a LUT
34 if (depth == CV_8U)
35 {
36 Mat lut(1, 256, CV_8U);
37 uchar* p = lut.ptr<uchar>();
38 for (int i = 0; i < 256; ++i)
39 {
40 double fn = std::pow(i / 255.0, gamma) * 255.0;
41 p[i] = cv::saturate_cast<uchar>(fn + 0.5);
42 }
43 _dst.create(src.size(), src.type());
44 Mat dst = _dst.getMat();
45 cv::LUT(src, lut, dst);
46 return;
47 }
48
49 Mat f;
50 src.convertTo(f, CV_64F, 1.0 / maxVal);
51 cv::pow(f, gamma, f);
52
53 _dst.create(src.size(), src.type());
54 Mat dst = _dst.getMat();
55 f.convertTo(dst, src.type(), maxVal);
56}

59Mat maskCopyTo(const Mat& src, const Mat& mask)
60{
61 Mat fullMasked;
62 src.copyTo(fullMasked, mask);
63
64 std::vector<Point> nonZeroLocations;
65 findNonZero(mask, nonZeroLocations);
66
67 Mat dst(static_cast<int>(nonZeroLocations.size()), 1, src.type());
68
69 int channels = src.channels();
70 if (channels == 1)
71 {
72 for (size_t i = 0; i < nonZeroLocations.size(); i++)
73 {
74 dst.at<double>(static_cast<int>(i), 0) = fullMasked.at<double>(nonZeroLocations[i]);
75 }
76 }
77 else if (channels == 3)
78 {
79 for (size_t i = 0; i < nonZeroLocations.size(); i++)
80 {
81 dst.at<Vec3d>(static_cast<int>(i), 0) = fullMasked.at<Vec3d>(nonZeroLocations[i]);
82 }
83 }
84 else
85 {
86 CV_Error(Error::StsBadArg, "Unsupported number of channels");
87 }
88
89 return dst;
90}

100Mat saturate(Mat& src, double low, double up)
101{
102 CV_Assert(src.type() == CV_64FC3);
103 Scalar lower_bound(low, low, low);
104 Scalar upper_bound(up, up, up);
105
106 Mat mask;
107 inRange(src, lower_bound, upper_bound, mask);
108 mask /= 255;
109
110 return mask;
