108 def resolve_type_nodes(self, root: ASTNode):
109 """Resolves type nodes in all overloads against `root`
110
111 Type resolution errors are postponed until all type nodes are examined.
112
113 Args:
114 root (ASTNode): Root of AST sub-tree used for type nodes resolution.

116 def has_unresolved_type_node(item) -> bool:
117 return item.type_node is not None and not item.type_node.is_resolved

119 errors = []
