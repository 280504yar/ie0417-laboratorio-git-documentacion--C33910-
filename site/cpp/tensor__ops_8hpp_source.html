49 template <class T>
50 void assertGEMMCompatiblity(const TensorSpan<T>& result, bool transa, const TensorView<T>& A, bool transb, const TensorView<T>& B) {
51 /* check dimension requirements for matrix multiplication */
52 if (!transa && !transb) {
53 CV_Assert(A.get_axis_size(-2) == result.get_axis_size(-2));
54 CV_Assert(A.get_axis_size(-1) == B.get_axis_size(-2));
55 CV_Assert(B.get_axis_size(-1) == result.get_axis_size(-1));
56 } else if (!transa && transb) {
57 CV_Assert(A.get_axis_size(-2) == result.get_axis_size(-2));
58 CV_Assert(A.get_axis_size(-1) == B.get_axis_size(-1));
59 CV_Assert(B.get_axis_size(-2) == result.get_axis_size(-1));
60 } else if (transa && !transb) {
61 CV_Assert(A.get_axis_size(-1) == result.get_axis_size(-2));
62 CV_Assert(A.get_axis_size(-2) == B.get_axis_size(-2));
63 CV_Assert(B.get_axis_size(-1) == result.get_axis_size(-1));
64 } else {
65 CV_Assert(A.get_axis_size(-1) == result.get_axis_size(-2));
66 CV_Assert(A.get_axis_size(-2) == B.get_axis_size(-1));
67 CV_Assert(B.get_axis_size(-2) == result.get_axis_size(-1));
68 }

80 template <class T> inline
81 void gemm(const cublas::Handle& handle, T beta, TensorSpan<T> result, T alpha, bool transa, TensorView<T> A, bool transb, TensorView<T> B) {
82 /* matrix operations can be performed only on tensors with rank two or below */
83 CV_Assert(get_effective_rank(A) <= 2);
84 CV_Assert(get_effective_rank(B) <= 2);
85 CV_Assert(get_effective_rank(result) <= 2);
86
87 const auto result_nr = result.get_axis_size(-2);
88 const auto result_nc = result.get_axis_size(-1);
89 const auto common_dim = A.get_axis_size(transa ? -2 : -1);
90 const auto A_nc = A.get_axis_size(-1);
91 const auto B_nc = B.get_axis_size(-1);
92
93 detail::assertGEMMCompatiblity(result, transa, A, transb, B);
94
95 /* tensors are stored in row-major but cublas::gemm operates on column-major matrices
96 * a row-major matrix when read as column-major matrix gives the transpose of the intended matrix
97 *
98 * Required: C = AB
99 * what cuBLAS sees: C^T = A^TB^T = (BA)^T
100 *
101 * By reversing operands, we effectively perform:
102 * C^T = B^TA^T = (AB)^T
103 *
104 * which gives C = AB
105 */
106 cublas::gemm<T>(handle,
107 transb, transa,
108 result_nc, result_nr, common_dim,
109 alpha, B.get(), B_nc,
110 A.get(), A_nc,
111 beta, result.get(), result_nc);
112 }

124 template <class T> inline
125 void gemmStridedBatched(const cublas::Handle& handle, T beta, TensorSpan<T> result, T alpha, bool transa, TensorView<T> A, bool transb, TensorView<T> B) {
126 CV_Assert(A.rank() == 3);
127 CV_Assert(B.rank() == 3);
128 CV_Assert(result.rank() == 3);
129
130 const auto batch_size = result.get_axis_size(0);
131 CV_Assert(batch_size == A.get_axis_size(0));
132 CV_Assert(batch_size == B.get_axis_size(0));
133
134 detail::assertGEMMCompatiblity(result, transa, A, transb, B);
135
136 const auto result_nr = result.get_axis_size(-2);
137 const auto result_nc = result.get_axis_size(-1);
138 const auto common_dim = A.get_axis_size(transa ? -2 : -1);
139 const auto A_nc = A.get_axis_size(-1);
140 const auto B_nc = B.get_axis_size(-1);
141
142 std::size_t strideA = (A.size() / batch_size),
143 strideB = (B.size() / batch_size),
144 strideC = (result.size() / batch_size);
145
146 cublas::gemmStridedBatched<T>(handle,
147 transb, transa,
148 result_nc, result_nr, common_dim,
149 alpha, B.get(), B_nc, strideB,
150 A.get(), A_nc, strideA,
151 beta, result.get(), result_nc, strideC,
152 batch_size);
153 }

164 template <class T> inline
165 void gemmBatched(const cublas::Handle& handle, std::size_t batch,
166 T beta, TensorSpan<T> C, const std::vector<std::size_t> C_offsets, T alpha,
167 bool trans_a, TensorView<T> A, const std::vector<std::size_t> A_offsets,

172 const auto lda = A.get_axis_size(-1),
173 ldb = B.get_axis_size(-1),
174 ldc = N;

187 template <class T> inline
188 void softmax(const cudnn::Handle& handle, TensorSpan<T> output, TensorView<T> input, int channel_axis, bool log) {
189 CV_Assert(is_shape_same(output, input));
190
191 channel_axis = clamp_axis(channel_axis, input.rank());
192
193 std::size_t outer_size = input.size_range(0, channel_axis);
194 auto channel_size = input.get_axis_size(channel_axis);
195 std::size_t inner_size = input.size_range(channel_axis + 1, input.rank());
196
197 std::array<std::size_t, 4> shape = { outer_size, channel_size, 1, inner_size };
198
199 using cudnn::TensorDescriptor;
200 auto inputDesc = TensorDescriptor<T>(shape);
201 auto outputDesc = TensorDescriptor<T>(shape);
202 cudnn::softmax(handle, outputDesc, output.get(), inputDesc, input.get(), log);
203 }

206 template <class T>
207 class Convolution {
208 using TensorDescriptor = cudnn::TensorDescriptor<T>;
209 using FilterDescriptor = cudnn::FilterDescriptor<T>;
210 using ConvolutionDescriptor = cudnn::ConvolutionDescriptor<T>;
211 using ConvolutionAlgorithm = cudnn::ConvolutionAlgorithm<T>;
212 using ActivationDescriptor = cudnn::ActivationDescriptor;
213
214 public:
215 using ActivationType = ActivationDescriptor::ActivationType;
216
217 struct params_type {
218 /* convolution */
219 std::vector<std::size_t> input_shape;
220 std::vector<std::size_t> filter_shape;
221 std::vector<std::size_t> padding;
222 std::vector<std::size_t> stride;
223 std::vector<std::size_t> dilation;
224 std::size_t groups;
225
226 /* bias and activation (only RELU supported) */
227 std::vector<std::size_t> bias_shape;
228 ActivationType activation_type; /* MUST BE identity if there is no bias and ReLU if there is bias */
229 bool eltwise;
230 };
231
232 Convolution() = default;
233 Convolution(const Convolution&) = delete;
234 Convolution(Convolution&&) = default;
235 Convolution(cudnn::Handle handle, const params_type& params) {
236 cudnnHandle = std::move(handle);
237
238 inputTensorDesc = TensorDescriptor(params.input_shape);
239 filterDesc = FilterDescriptor(params.filter_shape);
240 convDesc = ConvolutionDescriptor(params.padding, params.stride, params.dilation, params.groups);
241
242 std::vector<int> output_dims;
243 getConvolutionForwardOutputDim(convDesc, filterDesc, inputTensorDesc, output_dims);
244 outputTensorDesc = TensorDescriptor(output_dims);
245
246 algo = ConvolutionAlgorithm(cudnnHandle, convDesc, filterDesc, inputTensorDesc, outputTensorDesc);
247
248 if (!params.bias_shape.empty()) {
249 CV_Assert(params.activation_type == ActivationType::RELU);
250 biasTensorDesc = TensorDescriptor(params.bias_shape);
251 if (params.eltwise)
252 eltwiseTensorDesc = TensorDescriptor(output_dims);
253 activationDesc = ActivationDescriptor(params.activation_type, 0.0);
254 } else {
255 CV_Assert(params.activation_type == ActivationType::IDENTITY);
256 }
257 }
258
259 Convolution& operator=(const Convolution&) = delete;
260 Convolution& operator=(Convolution&&) = default;
261
262 std::size_t get_workspace_size() const noexcept {
263 return algo.get_workspace_size();
264 }
265
266 void convolve(TensorSpan<T> output, TensorView<T> input, TensorView<T> filters, WorkspaceInstance scratchpad) {
267 cudnn::convolve<T>(
268 cudnnHandle,
269 convDesc, algo, scratchpad,
270 filterDesc, filters.get(),
271 inputTensorDesc, input.get(),
272 1.0, 0.0, outputTensorDesc, output.get()
273 );
274 }

356 }

501 template<class T>
502 class LSTM
503 {
504 using TensorDescriptor = cudnn::TensorDescriptor<T>;
505 using DropoutDescriptor = cudnn::DropoutDescriptor;
506 using RNNDescriptor = cudnn::RNNDescriptor<T>;
507 using FilterDescriptor = cudnn::FilterDescriptor<T>;
508 using TensorDescriptorsArray = cudnn::TensorDescriptorsArray<T>;
509
510 public:
511 using RNNMode = typename RNNDescriptor::RNNMode;
512
513 struct params_type
514 {
515 std::vector<std::size_t> weights_shape;
516
517 int seqLength;
518 int numLayers;
519 int hiddenSize;
520 int inputSize;
521 int miniBatch;
522 bool bidirectional;
523
524 float dropout;
525 RNNMode type;
526 };
527
528 LSTM() = default;
529 LSTM(const LSTM&) = delete;
530 LSTM(LSTM&&) = default;
531
532#if CUDNN_MAJOR >= 9
533 LSTM(cudnn::Handle handle, const params_type &params)
534 : cudnnHandle(std::move(handle)), seqLength(params.seqLength)
535 {
536 std::vector<int> seqLenArr(params.miniBatch, seqLength);
537 cudnnCreateRNNDataDescriptor(&xDesc);
538 cudnnSetRNNDataDescriptor(xDesc, cudnn::detail::get_data_type<T>(),
539 CUDNN_RNN_DATA_LAYOUT_SEQ_MAJOR_PACKED, seqLength,
540 params.miniBatch, params.inputSize, seqLenArr.data(),
541 nullptr);
542 cudnnCreateRNNDataDescriptor(&cyDesc);
543 cudnnSetRNNDataDescriptor(
544 cyDesc, cudnn::detail::get_data_type<T>(),
545 CUDNN_RNN_DATA_LAYOUT_SEQ_MAJOR_PACKED,
546 seqLength, params.miniBatch,
547 params.bidirectional ? params.hiddenSize * 2 : params.hiddenSize,
548 seqLenArr.data(),
549 nullptr);
550
551 dropoutDesc = DropoutDescriptor(cudnnHandle, params.dropout);
552 rnnDesc = RNNDescriptor(cudnnHandle, params.type, params.inputSize, params.hiddenSize,
553 params.numLayers, params.bidirectional, dropoutDesc);
554
555 int num_direction = params.bidirectional ? 2 : 1;
556 h0TensorDesc = TensorDescriptor(num_direction, params.miniBatch, params.hiddenSize);
557 c0TensorDesc = TensorDescriptor(num_direction, params.miniBatch, params.hiddenSize);
558
559 // Get amount of work space required to execute the RNN described by rnnDesc
560 // with input dimensions defined by inputDesc
561 CUDA4DNN_CHECK_CUDNN(cudnnGetRNNTempSpaceSizes(
562 cudnnHandle.get(), rnnDesc.get(), CUDNN_FWD_MODE_INFERENCE,
563 xDesc, &workSpaceSize, &reserveSpaceSize));
564
565 csl::WorkspaceBuilder builder;
566 builder.require<T>(workSpaceSize);
567 builder.require<T>(reserveSpaceSize);
568 scratch_mem_in_bytes = builder.required_workspace_size();
569 }
570#else
571 LSTM(cudnn::Handle handle, const params_type& params)
572 : cudnnHandle(std::move(handle)), seqLength{params.seqLength},
573 inputDesc(seqLength, {params.miniBatch, params.inputSize, 1}),
574 outputDesc(seqLength,
575 {params.miniBatch,
576 params.bidirectional ? params.hiddenSize * 2 : params.hiddenSize,
577 1})
578 {
579 dropoutDesc = DropoutDescriptor(cudnnHandle, params.dropout);
580 filterDesc = FilterDescriptor(params.weights_shape);
581 rnnDesc = RNNDescriptor(cudnnHandle, params.type, params.inputSize, params.hiddenSize,
582 params.numLayers, params.bidirectional, dropoutDesc);
583
584 int num_direction = params.bidirectional ? 2 : 1;
585 h0TensorDesc = TensorDescriptor(
586 {num_direction, params.miniBatch, params.hiddenSize});
587 c0TensorDesc = TensorDescriptor(
588 {num_direction, params.miniBatch, params.hiddenSize});
589
590 // Get amount of work space required to execute the RNN described by rnnDesc
591 // with input dimensions defined by inputDesc
592 csl::WorkspaceBuilder builder;
593 size_t workSize;
594 CUDA4DNN_CHECK_CUDNN(cudnnGetRNNWorkspaceSize(cudnnHandle.get(), rnnDesc.get(), seqLength,
595 inputDesc.get().data(), &workSize));
596 builder.require(workSize);
597 scratch_mem_in_bytes = builder.required_workspace_size();
598 }
599#endif

604 void inference(TensorView<T> input, TensorSpan<T> y_output, TensorSpan<T> yc_output, TensorView<T> filters,
605 TensorView<T> h0, TensorView<T> c0, csl::Workspace& workspace)
606 {
607 auto ws_allocator = csl::WorkspaceAllocator(workspace);
608
609#if CUDNN_MAJOR >= 9
610 size_t weightSpaceSize = sizeof(typename TensorView<T>::value_type) * filters.size();
611 auto workspaceData = ws_allocator.get_span<T>(workSpaceSize);
612 auto reserveSpaceData = ws_allocator.get_span<T>(reserveSpaceSize);
613 cudnn::LSTMForward<T>(cudnnHandle, rnnDesc, xDesc, input.get(), cyDesc,
614 y_output.get(), h0TensorDesc.get(), h0.get(),
615 DevicePtr<T>(nullptr), // hy, final state
616 c0TensorDesc.get(), // maps to cxDesc
617 c0.get(), // maps to cx
618 yc_output.get(), // maps to cy
619 weightSpaceSize,
620 filters.get(), // maps to weightSpace
621 workSpaceSize,
622 workspaceData.data(), // workSpaceSize and workSpace
623 reserveSpaceSize, // reserveSpaceSize
624 reserveSpaceData.data()
625 );
626#else
627 cudnn::LSTMForward<T>(cudnnHandle, rnnDesc, filterDesc, filters.get(), inputDesc,
628 input.get(), h0TensorDesc, h0.get(), c0TensorDesc, c0.get(),
629 seqLength, outputDesc, y_output.get(), yc_output.get(), ws_allocator.get_instance());
630#endif
631 }
