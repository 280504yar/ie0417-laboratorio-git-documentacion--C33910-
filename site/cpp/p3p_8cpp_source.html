49static bool solve_cubic_single_real(double c2, double c1, double c0, double &root) {
50 double a = c1 - c2 * c2 / 3.0;
51 double b = (2.0 * c2 * c2 * c2 - 9.0 * c2 * c1) / 27.0 + c0;
52 double c = b * b / 4.0 + a * a * a / 27.0;
53 if (c != 0) {
54 if (c > 0) {
55 c = std::sqrt(c);
56 b *= -0.5;
57 root = std::cbrt(b + c) + std::cbrt(b - c) - c2 / 3.0;
58 return true;
59 } else {
60 c = 3.0 * b / (2.0 * a) * std::sqrt(-3.0 / a);
61 root = 2.0 * std::sqrt(-a / 3.0) * std::cos(std::acos(c) / 3.0) - c2 / 3.0;
62 }
63 } else {
64 root = -c2 / 3.0 + (a != 0 ? (3.0 * b / a) : 0);
65 }
66 return false;
67}

69static bool root2real(double b, double c, double &r1, double &r2) {
70 const double THRESHOLD = -1.0e-12;
71 double v = b * b - 4.0 * c;
72 if (v < THRESHOLD) {
73 r1 = r2 = -0.5 * b;
74 return v >= 0;
75 }
76 if (v > THRESHOLD && v < 0.0) {
77 r1 = -0.5 * b;
78 r2 = -2;
79 return true;
80 }
81
82 double y = std::sqrt(v);
83 if (b < 0) {
84 r1 = 0.5 * (-b + y);
85 r2 = 0.5 * (-b - y);
86 } else {
87 r1 = 2.0 * c / (-b + y);
88 r2 = 2.0 * c / (-b - y);
89 }
90 return true;
91}

93static std::array<Vec3d, 2> compute_pq(Matx33d C) {
94 std::array<Vec3d, 2> pq;
95 Matx33d C_adj;
96
97 C_adj(0, 0) = C(1, 2) * C(2, 1) - C(1, 1) * C(2, 2);
98 C_adj(1, 1) = C(0, 2) * C(2, 0) - C(0, 0) * C(2, 2);
99 C_adj(2, 2) = C(0, 1) * C(1, 0) - C(0, 0) * C(1, 1);
100 C_adj(0, 1) = C(0, 1) * C(2, 2) - C(0, 2) * C(2, 1);
101 C_adj(0, 2) = C(0, 2) * C(1, 1) - C(0, 1) * C(1, 2);
102 C_adj(1, 0) = C_adj(0, 1);
103 C_adj(1, 2) = C(0, 0) * C(1, 2) - C(0, 2) * C(1, 0);
104 C_adj(2, 0) = C_adj(0, 2);
105 C_adj(2, 1) = C_adj(1, 2);
106
107 Matx31d v;
108 if (C_adj(0, 0) > C_adj(1, 1)) {
109 if (C_adj(0, 0) > C_adj(2, 2)) {
110 v = C_adj.col(0) / std::sqrt(C_adj(0, 0));
111 } else {
112 v = C_adj.col(2) / std::sqrt(C_adj(2, 2));
113 }
114 } else if (C_adj(1, 1) > C_adj(2, 2)) {
115 v = C_adj.col(1) / std::sqrt(C_adj(1, 1));
116 } else {
117 v = C_adj.col(2) / std::sqrt(C_adj(2, 2));
118 }
119
120 C(0, 1) -= v(2);
121 C(0, 2) += v(1);
122 C(1, 2) -= v(0);
123 C(1, 0) += v(2);
124 C(2, 0) -= v(1);
125 C(2, 1) += v(0);
126
127 pq[0](0) = C.col(0)(0);
128 pq[0](1) = C.col(0)(1);
129 pq[0](2) = C.col(0)(2);
130 pq[1](0) = C.row(0)(0);
131 pq[1](1) = C.row(0)(1);
132 pq[1](2) = C.row(0)(2);
133
134 return pq;
135}

137// Performs a few Newton steps on the equations
138static void refine_lambda(double &lambda1, double &lambda2, double &lambda3, const double a12, const double a13,
139 const double a23, const double b12, const double b13, const double b23) {
140 for (int iter = 0; iter < 5; ++iter) {
141 double r1 = (lambda1 * lambda1 - 2.0 * lambda1 * lambda2 * b12 + lambda2 * lambda2 - a12);
142 double r2 = (lambda1 * lambda1 - 2.0 * lambda1 * lambda3 * b13 + lambda3 * lambda3 - a13);
143 double r3 = (lambda2 * lambda2 - 2.0 * lambda2 * lambda3 * b23 + lambda3 * lambda3 - a23);
144
145 if (std::abs(r1) + std::abs(r2) + std::abs(r3) < 1e-10)
146 return;
147
148 double x11 = lambda1 - lambda2 * b12;
149 double x12 = lambda2 - lambda1 * b12;
150 double x21 = lambda1 - lambda3 * b13;
151 double x23 = lambda3 - lambda1 * b13;
152 double x32 = lambda2 - lambda3 * b23;
153 double x33 = lambda3 - lambda2 * b23;
154 double detJ = 0.5 / (x11 * x23 * x32 + x12 * x21 * x33); // half minus inverse determinant
155
156 // This uses the closed form of the inverse for the jacobian.
157 // Due to the zero elements this actually becomes quite nice.
158 lambda1 += (-x23 * x32 * r1 - x12 * x33 * r2 + x12 * x23 * r3) * detJ;
159 lambda2 += (-x21 * x33 * r1 + x11 * x33 * r2 - x11 * x23 * r3) * detJ;
160 lambda3 += (x21 * x32 * r1 - x11 * x32 * r2 - x12 * x21 * r3) * detJ;
161 }
162}

166void p3p::calibrateAndNormalizePointsPnP(const Mat &opoints_, const Mat &ipoints_) {
167 auto convertPoints = [] (const Mat &points_input, Mat &points, int pt_dim) {
168 points_input.convertTo(points, CV_64F); // convert points to have float precision
169 if (points.channels() > 1)
170 points = points.reshape(1, (int)points.total()); // convert point to have 1 channel
171 if (points.rows < points.cols)
172 transpose(points, points); // transpose so points will be in rows
173 CV_CheckGE(points.cols, pt_dim, "Invalid dimension of point");
174 if (points.cols != pt_dim) // in case when image points are 3D convert them to 2D
175 points = points.colRange(0, pt_dim);
176 };
177
178 Mat ipoints;
179 convertPoints(ipoints_, ipoints, 2);
180 for (int i = 0; i < 3; i++) {
181 const double k_inv_u = ipoints.at<double>(i, 0);
182 const double k_inv_v = ipoints.at<double>(i, 1);
183 double x_norm = 1.0 / sqrt(k_inv_u*k_inv_u + k_inv_v*k_inv_v + 1);
184 x_copy[i](0) = k_inv_u * x_norm;
185 x_copy[i](1) = k_inv_v * x_norm;
186 x_copy[i](2) = x_norm;
187 }
188
189 Mat opoints;
190 convertPoints(opoints_, opoints, 3);
191 X_copy[0](0) = opoints.at<double>(0, 0);
192 X_copy[0](1) = opoints.at<double>(0, 1);
193 X_copy[0](2) = opoints.at<double>(0, 2);
194
195 X_copy[1](0) = opoints.at<double>(1, 0);
196 X_copy[1](1) = opoints.at<double>(1, 1);
197 X_copy[1](2) = opoints.at<double>(1, 2);
198
199 X_copy[2](0) = opoints.at<double>(2, 0);
200 X_copy[2](1) = opoints.at<double>(2, 1);
201 X_copy[2](2) = opoints.at<double>(2, 2);
202}

209int p3p::estimate(std::vector<Mat>& Rs, std::vector<Mat>& ts, const cv::Mat& opoints, const cv::Mat& ipoints) {
210 CV_INSTRUMENT_REGION();
211 calibrateAndNormalizePointsPnP(opoints, ipoints);
212
213 Rs.reserve(4);
214 ts.reserve(4);
215
216 Vec3d X01 = X_copy[0] - X_copy[1];
217 Vec3d X02 = X_copy[0] - X_copy[2];
218 Vec3d X12 = X_copy[1] - X_copy[2];
219
220 double a01 = norm(X01, NORM_L2SQR);
221 double a02 = norm(X02, NORM_L2SQR);
222 double a12 = norm(X12, NORM_L2SQR);
223
224 std::array<Vec3d, 3> X = {X_copy[0], X_copy[1], X_copy[2]};
225 std::array<Vec3d, 3> x = {x_copy[0], x_copy[1], x_copy[2]};
226
227 // Switch X,x so that BC is the largest distance among {X01, X02, X12}
228 if (a01 > a02) {
229 if (a01 > a12) {
230 std::swap(x[0], x[2]);
231 std::swap(X[0], X[2]);
232 std::swap(a01, a12);
233 X01 = -X12;
234 X02 = -X02;
235 }
236 } else if (a02 > a12) {
237 std::swap(x[0], x[1]);
238 std::swap(X[0], X[1]);
239 std::swap(a02, a12);
240 X01 = -X01;
241 X02 = X12;
242 }
243
244 const double a12d = 1.0 / a12;
245 const double a = a01 * a12d;
246 const double b = a02 * a12d;
247
248 const double m01 = x[0].dot(x[1]);
249 const double m02 = x[0].dot(x[2]);
250 const double m12 = x[1].dot(x[2]);
251
252 // Ugly parameters to simplify the calculation
253 const double m12sq = -m12 * m12 + 1.0;
254 const double m02sq = -1.0 + m02 * m02;
255 const double m01sq = -1.0 + m01 * m01;
256 const double ab = a * b;
257 const double bsq = b * b;
258 const double asq = a * a;
259 const double m013 = -2.0 + 2.0 * m01 * m02 * m12;
260 const double bsqm12sq = bsq * m12sq;
261 const double asqm12sq = asq * m12sq;
262 const double abm12sq = 2.0 * ab * m12sq;
263
264 const double k3_inv = 1.0 / (bsqm12sq + b * m02sq);
265 const double k2 = k3_inv * ((-1.0 + a) * m02sq + abm12sq + bsqm12sq + b * m013);
266 const double k1 = k3_inv * (asqm12sq + abm12sq + a * m013 + (-1.0 + b) * m01sq);
267 const double k0 = k3_inv * (asqm12sq + a * m01sq);
268
269 double s;
270 bool G = yaqding::solve_cubic_single_real(k2, k1, k0, s);
271
272 Matx33d C;
273 C(0, 0) = -a + s * (1 - b);
274 C(0, 1) = -m02 * s;
275 C(0, 2) = a * m12 + b * m12 * s;
276 C(1, 0) = C(0, 1);
277 C(1, 1) = s + 1;
278 C(1, 2) = -m01;
279 C(2, 0) = C(0, 2);
280 C(2, 1) = C(1, 2);
281 C(2, 2) = -a - b * s + 1;
282
283 std::array<Vec3d, 2> pq = yaqding::compute_pq(C);
284
285 // XX << X01, X02, X01.cross(X02);
286 // XX = XX.inverse().eval();
287 Matx33d XX;
288 XX(0,0) = X01(0); XX(1,0) = X01(1); XX(2,0) = X01(2);
289 XX(0,1) = X02(0); XX(1,1) = X02(1); XX(2,1) = X02(2);
290 Vec3d X01_X02 = X01.cross(X02);
291 XX(0,2) = X01_X02(0); XX(1,2) = X01_X02(1); XX(2,2) = X01_X02(2);
292 XX = XX.inv();
293
294 int n_sols = 0;
295 for (int i = 0; i < 2; ++i) {
296 // [p0 p1 p2] * [1; x; y] = 0, or [p0 p1 p2] * [d2; d0; d1] = 0
297 double p0 = pq[i](0);
298 double p1 = pq[i](1);
299 double p2 = pq[i](2);
300 // here we run into trouble if p0 is zero,
301 // so depending on which is larger, we solve for either d0 or d1
302 // The case p0 = p1 = 0 is degenerate and can be ignored
303 bool switch_12 = std::abs(p0) <= std::abs(p1);
304
305 if (switch_12) {
306 // eliminate d0
307 double w0 = -p0 / p1;
308 double w1 = -p2 / p1;
309 double ca = 1.0 / (w1 * w1 - b);
310 double cb = 2.0 * (b * m12 - m02 * w1 + w0 * w1) * ca;
311 double cc = (w0 * w0 - 2 * m02 * w0 - b + 1.0) * ca;
312 double taus[2];
313
314 if (!yaqding::root2real(cb, cc, taus[0], taus[1]))
315 continue;
316
317 for (double tau : taus) {
318 if (tau <= 0)
319 continue;
320
321 // positive only
322 double d2 = std::sqrt(a12 / (tau * (tau - 2.0 * m12) + 1.0));
323 double d1 = tau * d2;
324 double d0 = (w0 * d2 + w1 * d1);
325 if (d0 < 0)
326 continue;
327
328 yaqding::refine_lambda(d0, d1, d2, a01, a02, a12, m01, m02, m12);
329 Vec3d v1 = d0 * x[0] - d1 * x[1];
330 Vec3d v2 = d0 * x[0] - d2 * x[2];
331 // YY << v1, v2, v1.cross(v2);
332 Matx33d YY;
333 YY(0,0) = v1(0); YY(1,0) = v1(1); YY(2,0) = v1(2);
334 YY(0,1) = v2(0); YY(1,1) = v2(1); YY(2,1) = v2(2);
335 Vec3d v1_v2 = v1.cross(v2);
336 YY(0,2) = v1_v2(0); YY(1,2) = v1_v2(1); YY(2,2) = v1_v2(2);
337
338 // output->emplace_back(R, d0 * x[0] - R * X[0]);
339 Matx33d R = (YY * XX);
340 Rs.push_back(Mat(R));
341 Vec3d trans = (d0 * x[0] - R * X[0]);
342 ts.push_back(Mat(trans));
343 ++n_sols;
344 }
345 } else {
346 double w0 = -p1 / p0;
347 double w1 = -p2 / p0;
348 double ca = 1.0 / (-a * w1 * w1 + 2 * a * m12 * w1 - a + 1);
349 double cb = 2 * (a * m12 * w0 - m01 - a * w0 * w1) * ca;
350 double cc = (1 - a * w0 * w0) * ca;
351
352 double taus[2];
353 if (!yaqding::root2real(cb, cc, taus[0], taus[1]))
354 continue;
355
356 for (double tau : taus) {
357 if (tau <= 0)
358 continue;
359
360 double d0 = std::sqrt(a01 / (tau * (tau - 2.0 * m01) + 1.0));
361 double d1 = tau * d0;
362 double d2 = w0 * d0 + w1 * d1;
363
364 if (d2 < 0)
365 continue;
366
367 yaqding::refine_lambda(d0, d1, d2, a01, a02, a12, m01, m02, m12);
368 Vec3d v1 = d0 * x[0] - d1 * x[1];
369 Vec3d v2 = d0 * x[0] - d2 * x[2];
370 // YY << v1, v2, v1.cross(v2);
371 Matx33d YY;
372 YY(0,0) = v1(0); YY(1,0) = v1(1); YY(2,0) = v1(2);
373 YY(0,1) = v2(0); YY(1,1) = v2(1); YY(2,1) = v2(2);
374 Vec3d v1_v2 = v1.cross(v2);
375 YY(0,2) = v1_v2(0); YY(1,2) = v1_v2(1); YY(2,2) = v1_v2(2);
376
377 // output->emplace_back(R, d0 * x[0] - R * X[0]);
378 Matx33d R = (YY * XX);
379 Rs.push_back(Mat(R));
380 Vec3d trans = (d0 * x[0] - R * X[0]);
381 ts.push_back(Mat(trans));
382 ++n_sols;
383 }
384 }
385
386 if (n_sols > 0 && G)
387 break;
388 }
389
390 return n_sols;
391}
