57// Resolve an engine request to a concrete supported engine (ENGINE_OPENCV or ENGINE_ORT).
58// ENGINE_AUTO resolves to ENGINE_OPENCV. The OPENCV_FORCE_DNN_ENGINE override, when set to a
59// concrete engine, wins over the caller's choice; left unset (ENGINE_AUTO) it does not force.
60static int resolveOnnxEngine(int engine)
61{
62 static const int engine_forced =
63 (int)utils::getConfigurationParameterSizeT("OPENCV_FORCE_DNN_ENGINE", ENGINE_AUTO);
64 if (engine_forced == ENGINE_OPENCV || engine_forced == ENGINE_ORT)
65 engine = engine_forced;
66 if (engine == ENGINE_AUTO)
67 engine = ENGINE_OPENCV;
68 if (engine != ENGINE_OPENCV && engine != ENGINE_ORT)
69 {
70 CV_LOG_WARNING(NULL, "DNN/ONNX: only ENGINE_AUTO, ENGINE_OPENCV and ENGINE_ORT are supported; "
71 "falling back to ENGINE_OPENCV.");
72 engine = ENGINE_OPENCV;
73 }
74 return engine;
75}

77Net readNetFromONNX(const String& onnxFile, int engine)
78{
79 if (resolveOnnxEngine(engine) == ENGINE_ORT)
80 {
81#ifdef HAVE_ONNXRUNTIME
82 Net net = readNetFromONNX2_ORT(onnxFile);
83 if (net.empty())
84 CV_Error(Error::StsError, "DNN/ONNX/ORT: failed to load model");
85 if (!net.getImpl() || net.getImpl()->modelFileName.empty())
86 CV_Error(Error::StsError, "DNN/ONNX/ORT: ONNX Runtime model metadata was not initialized");
87 return net;
88#else
89 CV_LOG_WARNING(NULL, "DNN/ONNX/ORT: OpenCV was built without ONNX Runtime (WITH_ONNXRUNTIME=OFF). Falling back to ENGINE_OPENCV.");
90#endif
91 }
92 return readNetFromONNX2(onnxFile);
93}

121static int onnxDataTypeToCvDepth(int onnxType)
122{
123 switch (onnxType)
124 {
125 case opencv_onnx::TensorProto_DataType_FLOAT: return CV_32F;
126 case opencv_onnx::TensorProto_DataType_UINT8: return CV_8U;
127 case opencv_onnx::TensorProto_DataType_UINT16: return CV_16U;
128 case opencv_onnx::TensorProto_DataType_FLOAT16: return CV_16F;
129 case opencv_onnx::TensorProto_DataType_INT8: return CV_8S;
130 case opencv_onnx::TensorProto_DataType_INT16: return CV_16S;
131 case opencv_onnx::TensorProto_DataType_INT32: return CV_32S;
132 case opencv_onnx::TensorProto_DataType_INT64: return CV_64S;
133 case opencv_onnx::TensorProto_DataType_BOOL: return CV_Bool;
134 case opencv_onnx::TensorProto_DataType_DOUBLE: return CV_64F;
135 case opencv_onnx::TensorProto_DataType_BFLOAT16: return CV_16BF;
136 default: return CV_32F;
137 }
138}

140Mat readTensorFromONNX(const String& path)
141{
142 std::fstream input(path.c_str(), std::ios::in | std::ios::binary);
143 if (!input)
144 {
145 CV_Error(Error::StsBadArg, cv::format("Can't read ONNX file: %s", path.c_str()));
146 }
147
148 opencv_onnx::TensorProto tensor_proto = opencv_onnx::TensorProto();
149 if (!tensor_proto.ParseFromIstream(&input))
150 {
151 CV_Error(Error::StsUnsupportedFormat, cv::format("Failed to parse ONNX data: %s", path.c_str()));
152 }
153 Mat mat = getMatFromTensor(tensor_proto, false);
154 int dims = (int)tensor_proto.dims_size();
155 if (dims > 0 && mat.total() == 0) {
156 int cv_type = onnxDataTypeToCvDepth(tensor_proto.data_type());
157
158 std::vector<int> sizes(dims);
159 for (int i = 0; i < dims; ++i) sizes[i] = (int)tensor_proto.dims(i);
160 mat.create(dims, sizes.data(), cv_type);
161 }
162 releaseONNXTensor(tensor_proto);
163 return mat;
164}
