8static void tanh(const Mat &src, Mat &dst)
9{
10 CV_Assert(src.type() == CV_32F);
11 dst.create(src.size(), src.type());
12 const int nrows = src.rows;
13 const int cols = src.cols;
14 parallel_for_(Range(0, nrows), [&](const Range& range) {
15 for (int row = range.start; row < range.end; ++row)
16 {
17 const float* srcptr = src.ptr<float>(row);
18 float* dstptr = dst.ptr<float>(row);
19 int i = 0;
20#if (CV_SIMD || CV_SIMD_SCALABLE)
21 const int vlanes = VTraits<v_float32>::vlanes();
22 v_float32 one = vx_setall_f32(1.f), two = vx_setall_f32(2.f), minus_two = vx_setall_f32(-2.f);
23 for (; i <= cols - vlanes; i += vlanes)
24 {
25 v_float32 x = vx_load(srcptr + i);
26 v_float32 e = v_exp(v_mul(minus_two, x)); // exp(-2x)
27 v_float32 t = v_sub(v_div(two, v_add(one, e)), one); // 2/(1+exp(-2x)) - 1
28 vx_store(dstptr + i, t);
29 }
30#endif
31 for (; i < cols; ++i)
32 dstptr[i] = std::tanh(srcptr[i]);
33 }
34 });
35}

37static void sigmoid(const Mat &src, Mat &dst)
38{
39 CV_Assert(src.type() == CV_32F);
40 dst.create(src.size(), src.type());
41 const int nrows = src.rows;
42 const int cols = src.cols;
43 parallel_for_(Range(0, nrows), [&](const Range& range) {
44 for (int row = range.start; row < range.end; ++row)
45 {
46 const float* srcptr = src.ptr<float>(row);
47 float* dstptr = dst.ptr<float>(row);
48 int i = 0;
49#if (CV_SIMD || CV_SIMD_SCALABLE)
50 const int vlanes = VTraits<v_float32>::vlanes();
51 v_float32 one = vx_setall_f32(1.f), zero = vx_setzero_f32();
52 for (; i <= cols - vlanes; i += vlanes)
53 {
54 v_float32 x = vx_load(srcptr + i);
55 v_float32 t = v_exp(v_sub(zero, x)); // exp(-x)
56 t = v_div(one, v_add(one, t)); // 1 / (1 + exp(-x))
57 vx_store(dstptr + i, t);
58 }
59#endif
60 for (; i < cols; ++i)
61 dstptr[i] = 1.f / (1.f + std::exp(-srcptr[i]));
62 }
63 });
64}

66// Fused computation of h_t = z (*) h_(t-1) + (1 - z) (*) n_t
67// Single pass over elements instead of 4 separate multiply/subtract/multiply/add calls.
68template<typename Dtype>
69static void gruComputeH(const Mat &z, const Mat &n, Mat &h)
70{
71 CV_Assert(z.rows == n.rows && z.rows == h.rows);
72 CV_Assert(z.cols == n.cols && z.cols == h.cols);
73 for (int row = 0; row < z.rows; ++row)
74 {
75 const Dtype* zPtr = z.ptr<Dtype>(row);
76 const Dtype* nPtr = n.ptr<Dtype>(row);
77 Dtype* hPtr = h.ptr<Dtype>(row);
78 for (int col = 0; col < z.cols; ++col)
79 hPtr[col] = zPtr[col] * hPtr[col] + (Dtype(1) - zPtr[col]) * nPtr[col];
80 }

245 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
246 {
247 CV_TRACE_FUNCTION();
248 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
249
250 if (inputs_arr.depth() == CV_16F)
251 {
252 forward_fallback(inputs_arr, outputs_arr, internals_arr);
253 return;
254 }
255
256 std::vector<Mat> input, output, internals;
257 inputs_arr.getMatVector(input);
258 outputs_arr.getMatVector(output);
259 internals_arr.getMatVector(internals);
260
261 prepareRuntimeState(input);
262 const bool legacyDirectMode = (input.size() == 1);
263 const bool useLinearBeforeReset = linearBeforeReset || legacyDirectMode;
264
265 Mat sequence_input = input[0];
266 Mat sequence_input_time_major = sequence_input;
267 if (layout == BATCH_SEQ_HID)
268 {
269 cv::transposeND(sequence_input, {1, 0, 2}, sequence_input_time_major);
270 }
271
272 const int numDirs = 1 + static_cast<int>(bidirectional);
273 const int numOutGlobal = blobs[0].size[1];
274 int yIndex = -1, yhIndex = -1;
275 for (int oi = 0; oi < (int)output.size(); ++oi)
276 {
277 if (yIndex < 0 && (isYShape3D(output[oi], numDirs, numOutGlobal) || isYShape4D(output[oi], numDirs, numOutGlobal)))
278 yIndex = oi;
279 if (yhIndex < 0 && isYhShape(output[oi], numDirs, numOutGlobal))
280 yhIndex = oi;
281 }
282 bool writeY = false;
283 bool writeYh = false;
284 {
285 const size_t nOut = output.size();
286 if (nOut == 0)
287 {
288 writeY = false;
289 writeYh = false;
290 }
291 else if (nOut == 1)
292 {
293 const bool hasY = (yIndex >= 0);
294 const bool hasYh = (yhIndex >= 0);
295 writeY = hasY && !hasYh;
296 writeYh = !writeY;
297 }
298 else
299 {
300 writeY = true;
301 writeYh = true;
302 }
303 }
304
305 if (!writeY) yIndex = -1;
306 if (!writeYh) yhIndex = -1;
307
308 Mat y = (yIndex >= 0) ? output[yIndex] : Mat();
309 Mat y3d2d;
310 if (!y.empty() && y.dims == 3)
311 y3d2d = y.reshape(1, numTimeStamps * numSamples);
312
313 Mat yh = (yhIndex >= 0) ? output[yhIndex] : Mat();
314 Mat xTs = sequence_input_time_major.reshape(1, numTimeStamps * numSamples);
315
316 for (int i = 0; i < numDirs; ++i)
317 {
318 const Mat &Wh = blobs[0].rowRange(i * blobs[0].rows / numDirs, (i + 1) * blobs[0].rows / numDirs);
319 const Mat &Wx = blobs[1].rowRange(i * blobs[1].rows / numDirs, (i + 1) * blobs[1].rows / numDirs);
320 Mat h_0 = blobs[3].rowRange(i * blobs[3].rows / numDirs, (i + 1) * blobs[3].rows / numDirs);
321
322 int numOut = Wh.size[1];
323 Mat bias;
324 if (blobs.size() > 2 && !blobs[2].empty() && blobs[2].cols >= (i + 1) * 6 * numOut)
325 bias = blobs[2].colRange(i * 6 * numOut, (i + 1) * 6 * numOut);
326 else
327 bias = Mat::zeros(1, 6 * numOut, Wh.type());
328 if (h_0.empty() || h_0.rows != numSamples || h_0.cols != numOut)
329 h_0 = Mat::zeros(numSamples, numOut, Wh.type());
330
331 const Mat &bx = bias.colRange(0, bias.cols / 2);
332 const Mat &bh = bias.colRange(bias.cols / 2, bias.cols);
333
334 Mat hInternal = internals[0], dummyOnes = internals[1], gates = internals[2],
335 b_rz = internals[3], n_t = internals[4];
336 h_0.copyTo(hInternal);
337 dummyOnes.setTo(1.);
338
339 CV_CheckLE(3 * numOut, Wx.rows, "Invalid Wx shape for GRU gates");
340 CV_CheckLE(3 * numOut, Wh.rows, "Invalid Wh shape for GRU gates");
341 CV_CheckLE(3 * numOut, bx.cols, "Invalid input bias shape for GRU gates");
342 CV_CheckLE(3 * numOut, bh.cols, "Invalid recurrent bias shape for GRU gates");
343 const Mat& wx_rz = Wx.rowRange(0, 2 * numOut);
344 const Mat& wh_rz = Wh.rowRange(0, 2 * numOut);
345 b_rz = bx.colRange(0, 2 * numOut) + bh.colRange(0, 2 * numOut);
346 const Mat& wx_n = Wx.rowRange(2 * numOut, 3 * numOut);
347 const Mat& wh_n = Wh.rowRange(2 * numOut, 3 * numOut);
348 const Mat& b_in = bx.colRange(2 * numOut, 3 * numOut);
349 const Mat& b_hn = bh.colRange(2 * numOut, 3 * numOut);
350
351 // Precompute input projections for all timesteps at once (batched GEMM).
352 const int totalRows = numTimeStamps * numSamples;
353 Mat dummyOnesAll = Mat::ones(totalRows, 1, Wh.type());
354
355 // xProj_rz[t] = x[t] * Wx_rz^T + b_rz for all t
356 Mat xProj_rz(totalRows, 2 * numOut, Wh.type());
357 gemm(xTs, wx_rz, 1, xProj_rz, 0, xProj_rz, GEMM_2_T);
358 gemm(dummyOnesAll, b_rz, 1, xProj_rz, 1, xProj_rz);
359
360 // xProj_n[t] = x[t] * Wx_n^T + b_in for all t
361 Mat xProj_n(totalRows, numOut, Wh.type());
362 gemm(xTs, wx_n, 1, xProj_n, 0, xProj_n, GEMM_2_T);
363 gemm(dummyOnesAll, b_in, 1, xProj_n, 1, xProj_n);
364
365 int tsStart, tsEnd, tsInc;
366 if (i == 1) {
367 tsStart = numTimeStamps - 1;
368 tsEnd = -1;
369 tsInc = -1;
370 }
371 else {
372 tsStart = 0;
373 tsEnd = numTimeStamps;
374 tsInc = 1;
375 }
376 for (int ts = tsStart; ts != tsEnd; ts += tsInc)
377 {
378 Range curRowRange(ts * numSamples, (ts + 1) * numSamples);
379
380 // Use precomputed input projection for this timestep
381 Mat xCurrProj_rz = xProj_rz.rowRange(curRowRange);
382 Mat xCurrProj_n = xProj_n.rowRange(curRowRange);
383
384 xCurrProj_rz.copyTo(gates); // x * Wx_rz + b_rz (precomputed)
385 gemm(hInternal, wh_rz, 1, gates, 1, gates, GEMM_2_T); // + h_(t-1) * Wh_rz
386 sigmoid(gates, gates); // sigmoid()
387
388 Mat z = gates.colRange(0, gates.cols / 2);
389 Mat r = gates.colRange(gates.cols / 2, gates.cols);
390
391 if (useLinearBeforeReset)
392 {
393 // n_t = tanh(r (*) (h_(t-1) * Wh_n + b_hn) + x * Wx_n + b_in)
394 gemm(hInternal, wh_n, 1, n_t, 0, n_t, GEMM_2_T); // h_(t-1) * Wh_n
395 gemm(dummyOnes, b_hn, 1, n_t, 1, n_t); // + b_hn
396 multiply(r, n_t, n_t); // r (*) (...)
397 add(n_t, xCurrProj_n, n_t); // + x * Wx_n + b_in (precomputed)
398 }
399 else
400 {
401 // n_t = tanh((r (*) h_(t-1)) * Wh_n + x * Wx_n + b_in + b_hn)
402 multiply(r, hInternal, n_t); // r (*) h_(t-1)
403 gemm(n_t, wh_n, 1, n_t, 0, n_t, GEMM_2_T); // (r*h_(t-1)) * Wh_n
404 add(n_t, xCurrProj_n, n_t); // + x * Wx_n + b_in (precomputed)
405 gemm(dummyOnes, b_hn, 1, n_t, 1, n_t); // + b_hn
406 }
407 tanh(n_t, n_t); // tanh()
408
409 // h_t = z (*) h_(t-1) + (1 - z) (*) n_t (fused single-pass)
410 if (z.type() == CV_32F)
411 gruComputeH<float>(z, n_t, hInternal);
412 else
413 gruComputeH<double>(z, n_t, hInternal);
414
415 writeYStep(y, y3d2d, ts, i, numOut, hInternal);
416 }
417
418 writeYhDir(yh, i, numDirs, hInternal);
419 }

448 void prepareRuntimeState(const std::vector<Mat>& input)
449 {
450 CV_Assert(!input.empty());
451 const Mat& inp0 = input[0];
452
453 int numOut, numInp;
454 if (hasRuntimeWeights(input))
455 {
456 CV_Assert(input.size() >= 3);
457 const Mat& W_orig = input[1]; // [D, 3*H, I]
458 const Mat& R_orig = input[2]; // [D, 3*H, H]
459 CV_Assert(W_orig.dims == 3 && R_orig.dims == 3);
460
461 const int num_directions = W_orig.size[0];
462 numOut = R_orig.size[2];
463 numInp = W_orig.size[2];
464 bidirectional = (num_directions > 1);
465
466 blobs.resize(4);
467 blobs[0] = R_orig.reshape(1, R_orig.size[0] * R_orig.size[1]).clone(); // Wh
468 blobs[1] = W_orig.reshape(1, W_orig.size[0] * W_orig.size[1]).clone(); // Wx
469
470 if (input.size() > 3 && !input[3].empty())
471 blobs[2] = input[3].reshape(1, 1).clone();
472 else if (!blobs[2].empty())
473 blobs[2] = blobs[2].reshape(1, 1).clone();
474 else
475 blobs[2] = Mat::zeros(1, num_directions * 6 * numOut, W_orig.type());
476
477 if (input.size() > 5 && !input[5].empty())
478 blobs[3] = input[5].reshape(1, input[5].size[0] * input[5].size[1]).clone();
479 else if (!blobs[3].empty())
480 blobs[3] = blobs[3].reshape(1, 1).clone();
481 else
482 blobs[3] = Mat::zeros(1, num_directions * numOut, W_orig.type());
483 }
484 else
485 {
486 CV_Assert(blobs.size() >= 2);
487 numOut = blobs[0].size[1];
488 numInp = blobs[1].size[1];
489
490 if (input.size() > 3 && !input[3].empty())
491 blobs[2] = input[3].reshape(1, 1).clone();
492 else if (blobs.size() > 2 && !blobs[2].empty())
493 blobs[2] = blobs[2].reshape(1, 1).clone();
494 else
495 blobs[2] = Mat::zeros(1, (1 + static_cast<int>(bidirectional)) * 6 * numOut, blobs[1].type());
496 }
497
498 if (outTailShape.empty())
499 outTailShape.assign(1, numOut);
500 else
501 CV_Assert(total(outTailShape) == numOut);
502

504 if (layout == BATCH_SEQ_HID)

506 numSamples = inp0.size[0];
507 numTimeStamps = inp0.size[1];

527 return;
528 }
529 if (h0.rows == numDirs && h0.cols == expectedCols)

532 for (int dir = 0; dir < numDirs; ++dir)
533 for (int sample = 0; sample < numSamples; ++sample)
534 h0.row(dir).copyTo(expanded.row(dir * numSamples + sample));
