17bool is_byte_order_swapped(double scale)
18{
19 // ".pfm" format file specifies that:
20 // positive scale means big endianness;
21 // negative scale means little endianness.
22
23 #ifdef WORDS_BIGENDIAN
24 return scale < 0.0;
25 #else
26 return scale >= 0.0;
27 #endif
28}

30void swap_endianness(uint32_t& ui)
31{
32 static const uint32_t A(0x000000ffU);
33 static const uint32_t B(0x0000ff00U);
34 static const uint32_t C(0x00ff0000U);
35 static const uint32_t D(0xff000000U);
36
37 ui = ( (ui & A) << 24 )
38 | ( (ui & B) << 8 )
39 | ( (ui & C) >> 8 )
40 | ( (ui & D) >> 24 );
41}

47template<typename T>
48T read_number(cv::RLByteStream& strm)
49{
50 // should be enough to take string representation of any number
51 const size_t buffer_size = 2048;
52
53 std::vector<char> buffer(buffer_size, 0);
54 for (size_t i = 0; i < buffer_size; ++i) {
55 const int intc = strm.getByte();
56 CV_Assert(intc >= -128 && intc < 128);
57 char c = static_cast<char>(intc);
58 if (std::isspace(c)) {
59 break;
60 }
61 buffer[i] = c;
62 }
63 const std::string str(buffer.begin(), buffer.end());
64 return atoT<T>(str);
65}

87bool PFMDecoder::readHeader()
88{
89 if (!m_buf.empty())
90 m_strm.open(m_buf);
91 else
92 m_strm.open(m_filename);
93
94 if( !m_strm.isOpened()) return false;
95
96 if (m_strm.getByte() != 'P') {
97 CV_Error(Error::StsError, "Unexpected file type (expected P)");
98 }
99
100 switch (m_strm.getByte()) {
101 case 'f':
102 m_type = CV_32FC1;
103 break;
104 case 'F':
105 m_type = CV_32FC3;
106 break;
107 default:
108 CV_Error(Error::StsError, "Unexpected file type (expected `f` or `F`)");
109 }
110
111 if ('\n' != m_strm.getByte()) {
112 CV_Error(Error::StsError, "Unexpected header format (expected line break)");
113 }
114
115
116 m_width = read_number<int>(m_strm);
117 m_height = read_number<int>(m_strm);
118 m_scale_factor = read_number<double>(m_strm);
119 m_swap_byte_order = is_byte_order_swapped(m_scale_factor);
120
121 return true;
122}

124bool PFMDecoder::readData(Mat& mat)
125{
126 if (!m_strm.isOpened()) {
127 CV_Error(Error::StsError, "Unexpected status in data stream");
128 }
129
130 Mat buffer(mat.size(), m_type);
131 for (int y = m_height - 1; y >= 0; --y) {
132 m_strm.getBytes(buffer.ptr(y), static_cast<int>(m_width * buffer.elemSize()));
133 if (is_byte_order_swapped(m_scale_factor)) {
134 for (int i = 0; i < m_width * buffer.channels(); ++i) {
135 static_assert( sizeof(uint32_t) == sizeof(float),
136 "uint32_t and float must have same size." );
137 swap_endianness(buffer.ptr<uint32_t>(y)[i]);
138 }
139 }
140 }
141
142 if (buffer.channels() == 3 && !m_use_rgb) {
143 cv::cvtColor(buffer, buffer, cv::COLOR_BGR2RGB);
144 }
145
146 CV_Assert(fabs(m_scale_factor) > 0.0f);
147 buffer *= 1.f / fabs(m_scale_factor);
148
149 buffer.convertTo(mat, mat.type());
150
151 return true;
152}

191bool PFMEncoder::write(const Mat& img, const std::vector<int>& params)
192{
193 CV_UNUSED(params);
194
195 WLByteStream strm;
196 if (m_buf) {
197 if (!strm.open(*m_buf)) {
198 return false;
199 } else {
200 m_buf->reserve(alignSize(256 + sizeof(float) * img.channels() * img.total(), 256));
201 }
202 } else if (!strm.open(m_filename)) {
203 return false;
204 }
205
206 Mat float_img;
207 CHECK_WRITE(strm.putByte('P'));
208 switch (img.channels()) {
209 case 1:
210 CHECK_WRITE(strm.putByte('f'));
211 img.convertTo(float_img, CV_32FC1);
212 break;
213 case 3:
214 CHECK_WRITE(strm.putByte('F'));
215 img.convertTo(float_img, CV_32FC3);
216 break;
217 default:
218 CV_Error(Error::StsBadArg, "Expected 1 or 3 channel image.");
219 }
220 CHECK_WRITE(strm.putByte('\n'));
221
222
223 CHECK_WRITE(write_anything(strm, float_img.cols));
224 CHECK_WRITE(strm.putByte(' '));
225 CHECK_WRITE(write_anything(strm, float_img.rows));
226 CHECK_WRITE(strm.putByte('\n'));
227#ifdef WORDS_BIGENDIAN
228 CHECK_WRITE(write_anything(strm, 1.0));
229#else
230 CHECK_WRITE(write_anything(strm, -1.0));
231#endif
232
233 CHECK_WRITE(strm.putByte('\n'));
234
235 // Comments are not officially supported in this file format.
236 // write_anything(strm, "# Generated by OpenCV " CV_VERSION "\n");
237
238 for (int y = float_img.rows - 1; y >= 0; --y)
239 {
240 if (float_img.channels() == 3) {
241 const float* bgr_row = float_img.ptr<float>(y);
242 size_t row_size = float_img.cols * float_img.channels();
243 std::vector<float> rgb_row(row_size);
244 for (int x = 0; x < float_img.cols; ++x) {
245 rgb_row[x*3+0] = bgr_row[x*3+2];
246 rgb_row[x*3+1] = bgr_row[x*3+1];
247 rgb_row[x*3+2] = bgr_row[x*3+0];
248 }
249 CHECK_WRITE(strm.putBytes( reinterpret_cast<const uchar*>(rgb_row.data()),
250 static_cast<int>(sizeof(float) * row_size) ));
251 } else if (float_img.channels() == 1) {
252 CHECK_WRITE(strm.putBytes(float_img.ptr(y), sizeof(float) * float_img.cols));
253 }
254 }
255 return true;
256}
