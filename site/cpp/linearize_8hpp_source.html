150 LinearGray(int deg_, Mat src, Color dst, Mat mask, RGBBase_ cs)

152 {
153 dst.getGray();
154 Mat lear_gray_mask = mask & dst.grays;
155

162 calc(src, dst_);

204 LinearColor(int deg_, Mat src_, Color dst, Mat mask, RGBBase_ cs)
205 : deg(deg_)
206 {
207 Mat src = maskCopyTo(src_, mask);
208 Mat dst_ = maskCopyTo(dst.to(*cs.l).colors, mask);
209 calc(src, dst_);
210 }
211
212 void calc(const Mat& src, const Mat& dst)
213 {
214 Mat schannels[3];
215 Mat dchannels[3];
216 split(src, schannels);
217 split(dst, dchannels);
218 pr = T(schannels[0], dchannels[0], deg);
219 pg = T(schannels[1], dchannels[1], deg);
220 pb = T(schannels[2], dchannels[2], deg);
221 }
222
223 Mat linearize(Mat inp) CV_OVERRIDE
224 {
225 Mat channels[3];
226 split(inp, channels);
227 std::vector<Mat> channel;
228 Mat res;
229 merge(std::vector<Mat> { pr(channels[0]), pg(channels[1]), pb(channels[2]) }, res);
230 return res;
231 }
