18Ptr<VisualOdometry> VisualOdometry::create(
19 const Ptr<Feature2D>& detector,
20 const Ptr<DescriptorMatcher>& matcher,
21 InputArray cameraMatrix,
22 InputArray distCoeffs,

68bool VisualOdometryImpl::processFrame(InputArray image)
69{
70 CV_INSTRUMENT_REGION();
71
72 if (image.empty()) return false;
73 lastEvent.clear();
74
75 Frame currentFrame;
76 extractFeatures(image, currentFrame);
77 if (currentFrame.keypoints.empty() || currentFrame.descriptors.empty()) return false;
78
79 currentFrame.mapPoints.assign(currentFrame.keypoints.size(), nullptr);
80 currentFrame.outliers.assign(currentFrame.keypoints.size(), false);
81 currentFrame.buildGrid();
82
83 switch (state)
84 {
85 case NOT_INITIALIZED:
86 refFrame = currentFrame;
87 state = INITIALIZING;
88 return false;
89
90 case INITIALIZING:
91 return bootstrap(currentFrame);
92
93 case TRACKING:
94 return track(currentFrame);
95 }
96 return false;
97}

99// Feature extraction

101void VisualOdometryImpl::extractFeatures(InputArray image, Frame& out) const
102{
103 Mat img = image.getMat();
104 out.imageSize = img.size();
105 out.keypoints.clear();
106
107 // Detect and compute on the original image (color/grey is up to the detector).
108 detector->detectAndCompute(img, noArray(), out.keypoints, out.descriptors);
109
110 // Store a greyscale copy for the optical-flow fallback.
111 if (img.channels() > 1)
112 cvtColor(img, out.image, COLOR_BGR2GRAY);
113 else
114 out.image = img.clone();
115
116 // Pre-compute undistorted pixel coordinates used by every stage.
117 if (!out.keypoints.empty())
118 {
119 std::vector<Point2f> raw;
120 raw.reserve(out.keypoints.size());
121 for (const auto& kp : out.keypoints)
122 raw.push_back(kp.pt);
123
124 if (!dist.empty())
125 undistortPoints(raw, out.undistKpts, K, dist, noArray(), K);
126 else
127 out.undistKpts = raw;
128 }
129}
