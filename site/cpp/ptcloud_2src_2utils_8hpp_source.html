26inline std::string trimSpaces(const std::string &input)
27{
28 size_t start = 0;
29 while (start < input.size() && input[start] == ' ')
30 {
31 start++;
32 }
33 size_t end = input.size();
34 while (end > start && (input[end - 1] == ' ' || input[end - 1] == '\n' || input[end - 1] == '\r'))
35 {
36 end--;
37 }
38 return input.substr(start, end - start);
39}

52void swapEndian(T &val)
53{
54 union U
55 {
56 T val;
57 std::array<std::uint8_t, sizeof(T)> raw;
58 } src, dst;
59
60 src.val = val;

227struct Intr
228{
231 struct Reprojector
232 {
233 Reprojector() {}
234 inline Reprojector(Intr intr)
235 {
236 fxinv = 1.f/intr.fx, fyinv = 1.f/intr.fy;
237 cx = intr.cx, cy = intr.cy;
238 }
239 template<typename T>
240 inline cv::Point3_<T> operator()(cv::Point3_<T> p) const
241 {
242 T x = p.z * (p.x - cx) * fxinv;
243 T y = p.z * (p.y - cy) * fyinv;
244 return cv::Point3_<T>(x, y, p.z);
245 }
246
247 float fxinv, fyinv, cx, cy;
248 };
249
251 struct Projector
252 {
253 inline Projector(Intr intr) : fx(intr.fx), fy(intr.fy), cx(intr.cx), cy(intr.cy) { }
254 template<typename T>
255 inline cv::Point_<T> operator()(cv::Point3_<T> p) const
256 {
257 T invz = T(1)/p.z;
258 T x = fx*(p.x*invz) + cx;
259 T y = fy*(p.y*invz) + cy;
260 return cv::Point_<T>(x, y);
261 }
262 template<typename T>
263 inline cv::Point_<T> operator()(cv::Point3_<T> p, cv::Point3_<T>& pixVec) const
264 {
265 T invz = T(1)/p.z;
266 pixVec = cv::Point3_<T>(p.x*invz, p.y*invz, 1);
267 T x = fx*pixVec.x + cx;
268 T y = fy*pixVec.y + cy;
269 return cv::Point_<T>(x, y);
270 }
271 float fx, fy, cx, cy;
272 };
273 Intr() : fx(), fy(), cx(), cy() { }
274 Intr(float _fx, float _fy, float _cx, float _cy) : fx(_fx), fy(_fy), cx(_cx), cy(_cy) { }
275 Intr(cv::Matx33f m) : fx(m(0, 0)), fy(m(1, 1)), cx(m(0, 2)), cy(m(1, 2)) { }
276 // scale intrinsics to pyramid level
277 inline Intr scale(int pyr) const
278 {
279 float factor = (1.f /(1 << pyr));
280 return Intr(fx*factor, fy*factor, cx*factor, cy*factor);
281 }
282 inline Reprojector makeReprojector() const { return Reprojector(*this); }
283 inline Projector makeProjector() const { return Projector(*this); }
284
285 inline cv::Matx33f getMat() const { return Matx33f(fx, 0, cx, 0, fy, cy, 0, 0, 1); }
286
287 float fx, fy, cx, cy;
288};
