13struct DSU
14{
15 std::vector<int> parent, rank;
16 DSU(int n) : parent(n), rank(n, 0)
17 {
18 for (int i = 0; i < n; ++i)
19 parent[i] = i;
20 }
21
22 int find(int x)
23 {
24 if (parent[x] != x)
25 parent[x] = find(parent[x]);
26 return parent[x];
27 }
28
29 void unite(int x, int y)
30 {
31 int rootX = find(x), rootY = find(y);
32 if (rootX != rootY)
33 {
34 if (rank[rootX] < rank[rootY])
35 {
36 parent[rootX] = rootY;
37 } else if (rank[rootX] > rank[rootY])
38 {
39 parent[rootY] = rootX;
40 } else
41 {
42 parent[rootY] = rootX;
43 ++rank[rootX];
44 }
45 }
46 }
47};

49bool weightComparator(const cv::MSTEdge& a, const cv::MSTEdge& b)
50{
51 if (a.weight != b.weight)
52 return a.weight < b.weight;
53 if (a.source != b.source)
54 return a.source < b.source;
55 return a.target < b.target;
56}

58bool buildMSTKruskal(int numNodes,
59 const std::vector<cv::MSTEdge>& edges,
60 std::vector<cv::MSTEdge>& resultingEdges)
61{
62 std::vector<cv::MSTEdge> sortedEdges = edges;
63 std::sort(sortedEdges.begin(), sortedEdges.end(), weightComparator);
64 DSU dsu(numNodes);
65
66 for (const auto &e : sortedEdges)
67 {
68 int u = e.source, v = e.target;
69 if (u >= numNodes || v >= numNodes)
70 return false;
71 if (u == v)
72 continue;
73 if (dsu.find(u) != dsu.find(v))
74 {
75 resultingEdges.push_back(e);
76 dsu.unite(u, v);
77 }
78 }
79
80 return true;
81}

83bool buildMSTPrim(int numNodes,
84 const std::vector<cv::MSTEdge>& edges,
85 std::vector<cv::MSTEdge>& resultingEdges,
86 int root)
87{
88 std::vector<bool> inMST(numNodes, false);
89 std::vector<std::vector<cv::MSTEdge>> adj(numNodes);
90 for (const auto& e : edges)
91 {
92 int u = e.source, v = e.target;
93 if (u >= numNodes || v >= numNodes)
94 return false;
95 if (u == v)
96 continue;
97 adj[u].push_back({u, v, e.weight});
98 adj[v].push_back({v, u, e.weight});
99 }
100
101 using HeapElem = std::tuple<double, int, int>; // (weight, from, to)
102 std::priority_queue<HeapElem, std::vector<HeapElem>, std::greater<HeapElem>> pq;
103
104 inMST[root] = true;
105 for (const auto& e : adj[root])
106 {
107 pq.emplace(e.weight, root, e.target);
108 }
109
110 while (!pq.empty())
111 {
112 HeapElem he = pq.top();
113 double w = std::get<0>(he);
114 int u = std::get<1>(he);
115 int v = std::get<2>(he);
116 pq.pop();
117
118 if (inMST[v])
119 continue;
120
121 inMST[v] = true;
122 resultingEdges.push_back({u, v, w});
123 for (const auto& e : adj[v])
124 {
125 if (!inMST[e.target])
126 pq.emplace(e.weight, v, e.target);
127 }
128 }
129
130 return true;
131}

138bool buildMST(int numNodes,
139 const std::vector<cv::MSTEdge>& inputEdges,
140 std::vector<cv::MSTEdge>& resultingEdges,
141 MSTAlgorithm algorithm,
142 int root)
143{
144 CV_TRACE_FUNCTION();
145
146 resultingEdges.clear();
147 if (numNodes <= 0 || inputEdges.empty() || root < 0 || root >= numNodes)
148 return false;
149
150 bool result = false;
151 switch (algorithm)
152 {
153 case MST_PRIM:
154 result = buildMSTPrim(numNodes, inputEdges, resultingEdges, root);
155 break;
156 case MST_KRUSKAL:
157 result = buildMSTKruskal(numNodes, inputEdges, resultingEdges);
158 break;
159 default:
160 CV_Error(cv::Error::Code::StsBadArg, "Invalid MST algorithm specified");
161 }
162
163 return (result && resultingEdges.size() == static_cast<size_t>(numNodes - 1));
164}
