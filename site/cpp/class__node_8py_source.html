96 def add_function(self, name: str, arguments: Sequence[FunctionNode.Arg] = (),
97 return_type: Optional[FunctionNode.RetType] = None,
98 is_static: bool = False) -> FunctionNode:
99 """Adds function as a child node of a class.
100
101 Function is classified in 3 categories:
102 1. Instance method.
103 If function is an instance method then `self` argument is
104 inserted at the beginning of its arguments list.
105
106 2. Class method (or factory method)
107 If `is_static` flag is `True` and typename of the function
108 return type matches name of the class then function is treated
109 as class method.
110
111 If function is a class method then `cls` argument is inserted
112 at the beginning of its arguments list.
113
114 3. Static method
115
116 Args:
117 name (str): Name of the function.
118 arguments (Sequence[FunctionNode.Arg], optional): Function arguments.
119 Defaults to ().
120 return_type (Optional[FunctionNode.RetType], optional): Function
121 return type. Defaults to None.
122 is_static (bool, optional): Flag whenever function is static or not.
123 Defaults to False.
124
125 Returns:
126 FunctionNode: created function node.
127 """
128
129 arguments = list(arguments)
130 if return_type is not None:
131 is_classmethod = return_type.typename == self.name
132 else:
133 is_classmethod = False
134 if not is_static:
135 arguments.insert(0, FunctionNode.Arg("self"))
136 elif is_classmethod:
137 is_static = False
138 arguments.insert(0, FunctionNode.Arg("cls"))
139 return self._add_child(FunctionNode, name, arguments=arguments,
140 return_type=return_type, is_static=is_static,
141 is_classmethod=is_classmethod)

152 def resolve_type_nodes(self, root: ASTNode) -> None:
153 """Resolves type nodes for all inner-classes, methods and properties
154 in 2 steps:
155 1. Resolve against `self` as a tree root
156 2. Resolve against `root` as a tree root
157 Type resolution errors are postponed until all children nodes are
158 examined.
159
160 Args:
161 root (Optional[ASTNode], optional): Root of the AST sub-tree.
162 Defaults to None.
163 """
164
165 errors = []
166 for child in itertools.chain(self.properties,
167 self.functions.values(),
168 self.classes.values()):
169 try:
170 try:
171 # Give priority to narrowest scope (class-level scope in this case)
172 child.resolve_type_nodes(self) # type: ignore
173 except TypeResolutionError:
174 child.resolve_type_nodes(root) # type: ignore
175 except TypeResolutionError as e:
176 errors.append(str(e))
177 if len(errors) > 0:
178 raise TypeResolutionError(
179 'Failed to resolve "{}" class against "{}". Errors: {}'.format(
180 self.full_export_name, root.full_export_name, errors
181 )
182 )
