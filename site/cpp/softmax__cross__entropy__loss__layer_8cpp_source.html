22 SoftmaxCrossEntropyLossImpl(const LayerParams& params)
23 {
24 setParamsFrom(params);
25 String red = toLowerCase(params.get<String>("reduction", "mean"));
26 if (red == "none") reduction = LOSS_REDUCTION_NONE;
27 else if (red == "mean") reduction = LOSS_REDUCTION_MEAN;
28 else if (red == "sum") reduction = LOSS_REDUCTION_SUM;
29 else CV_Error(Error::StsBadArg, "Unsupported reduction: " + red);
30 ignoreIndex = params.get<int>("ignore_index", -1);
31 labelSmoothing = params.get<float>("label_smoothing", 0.f);
32 softLabel = params.get<int>("soft_label", 0) != 0;
33 }

40 bool getMemoryShapes(const std::vector<MatShape>& in, const int requiredOutputs,
41 std::vector<MatShape>& out, std::vector<MatShape>& internals) const CV_OVERRIDE
42 {
43 CV_Assert(in.size() >= 2);
44 const MatShape& x = in[0];
45 CV_Assert(x.size() >= 2);
46
47 if (reduction == LOSS_REDUCTION_NONE) {
48 MatShape y; y.push_back(x[0]);
49 for (size_t i = 2; i < x.size(); ++i) y.push_back(x[i]);
50 out.push_back(y);
51 } else {
52 // Reduced (mean/sum) loss is a rank-0 scalar in ONNX, not a [1] tensor.
53 out.push_back(MatShape::scalar());
54 }
55
56 if (requiredOutputs >= 2)
57 out.push_back(x);
58
59 return false;
60 }

70 void forward(InputArrayOfArrays in_arr, OutputArrayOfArrays out_arr, OutputArrayOfArrays) CV_OVERRIDE
71 {
72 std::vector<Mat> inp;
73 in_arr.getMatVector(inp);
74
75 const Mat& logits = inp[0];
76 const Mat& labels = inp[1];
77 const bool hasWeight = inp.size() >= 3;
78 const Mat& w = hasWeight ? inp[2] : Mat();
79
80 CV_Assert(logits.dims >= 2);
81 const int N = logits.size[0], C = logits.size[1];
82 int S = 1; for (int i = 2; i < logits.dims; ++i) S *= logits.size[i];
83 MatShape logitsShape = shape(logits);
84 MatShape lossShape;
85 bool isReduced = (reduction != LOSS_REDUCTION_NONE);
86 if (!isReduced) {
87 lossShape.push_back(N);
88 for (int i = 2; i < logits.dims; ++i) lossShape.push_back(logits.size[i]);
89 }
90
91 auto kind = out_arr.kind();
92 Mat out_loss, out_logprob;
93 bool wantLogProb = false;
94 if (kind == _InputArray::STD_VECTOR_MAT) {
95 std::vector<Mat>& outs = out_arr.getMatVecRef();
96 CV_Assert(outs.size() == 1 || outs.size() == 2);
97 wantLogProb = (outs.size() == 2);
98 if (!isReduced) {
99 outs[0].fit(lossShape, CV_32F);
100 }
101 out_loss = outs[0];
102 if (wantLogProb) {
103 outs[1].fit(logitsShape, CV_32F);
104 out_logprob = outs[1];
105 }
106 } else if (kind == _InputArray::STD_VECTOR_UMAT) {
107 std::vector<UMat>& uouts = out_arr.getUMatVecRef();
108 CV_Assert(uouts.size() == 1 || uouts.size() == 2);
109 wantLogProb = (uouts.size() == 2);
110 if (!isReduced) {
111 uouts[0].fit(lossShape, CV_32F);
112 }
113 out_loss = uouts[0].getMat(ACCESS_WRITE);
114 if (wantLogProb) {
115 uouts[1].fit(logitsShape, CV_32F);
116 out_logprob = uouts[1].getMat(ACCESS_WRITE);
117 }
118 } else {
119 CV_Error(cv::Error::StsBadArg, cv::format("Unsupported output array kind: %d", kind));
120 }
121 CV_Assert(out_loss.type() == CV_32F);
122 if (isReduced) {
123 CV_Assert(out_loss.total() == 1);
124 }
125 const int nstripes = 16;
126 Mat logits32; logits.convertTo(logits32, CV_32F);
127 const size_t sN = logits32.step1(0);
128 const size_t sC = logits32.step1(1);
129
130 std::vector<float> rowLogSumExp(N*S);
131 std::vector<float> meanLogRow(N*S);
132
133 bool writeLogProb = (wantLogProb);
134 size_t sN_out = 0, sC_out = 0;
135 float* outlpBase = nullptr;
136 if (writeLogProb) {
137 CV_Assert(out_logprob.type() == CV_32F);
138 sN_out = out_logprob.step1(0);
139 sC_out = out_logprob.step1(1);
140 outlpBase = out_logprob.ptr<float>();
141 }
142
143 parallel_for_(Range(0, N*S), [&](const Range& rr){
144 const float* base = logits32.ptr<float>();
145 float* rowLSE = rowLogSumExp.data();
146 float* meanLR = meanLogRow.data();
147 const size_t sN_ = sN;
148 const size_t sC_ = sC;
149 const int S_ = S;
150 const int C_ = C;
151 const bool writeLP = writeLogProb;
152 float* outlpPtr = outlpBase;
153 const size_t sN_out_ = sN_out;
154 const size_t sC_out_ = sC_out;
155 AutoBuffer<float> tempBuf_(C_);
156 float* tempbuf = tempBuf_.data();
157 for (int r = rr.start; r < rr.end; ++r) {
158 const int n = r / S_;
159 const int s = r % S_;
160 const float* rowBase = base + n * sN_ + s;
161
162 float maxv = -FLT_MAX;
163 int c = 0;
164#if CV_ENABLE_UNROLLED && defined(_M_ARM64)
165 for (; c + 3 < C_; c += 4) {
166 const float v0 = rowBase[(c + 0) * sC_];
167 const float v1 = rowBase[(c + 1) * sC_];
168 const float v2 = rowBase[(c + 2) * sC_];
169 const float v3 = rowBase[(c + 3) * sC_];
170 tempbuf[c + 0] = v0;
171 tempbuf[c + 1] = v1;
172 tempbuf[c + 2] = v2;
173 tempbuf[c + 3] = v3;
174 maxv = std::max(maxv, std::max(std::max(v0, v1), std::max(v2, v3)));
175 }
176#endif
177 for (; c < C_; ++c) {
178 const float v = rowBase[c * sC_];
179 tempbuf[c] = v;
180 maxv = std::max(maxv, v);
181 }
182
183 float sumExp = 0.f;
184 float sumLogits = 0.f;
185 int i = 0;
186#if (CV_SIMD || CV_SIMD_SCALABLE)
187 const int nlanes = VTraits<v_float32>::vlanes();
188 v_float32 vsExp = vx_setzero_f32();
189 v_float32 vsLog = vx_setzero_f32();
190 v_float32 vmaxv = vx_setall_f32(maxv);
191 for (; i <= C_ - nlanes; i += nlanes) {
192 v_float32 v = vx_load(tempbuf + i);
193 vsLog = v_add(vsLog, v);
194 v_float32 vshift = v_sub(v, vmaxv);
195 v_float32 ev = v_exp(vshift);
196 vsExp = v_add(vsExp, ev);
197 }
198 sumExp += v_reduce_sum(vsExp);
199 sumLogits += v_reduce_sum(vsLog);
200#endif
201 for (; i < C_; ++i) {
202 const float v = tempbuf[i];
203 sumLogits += v;
204 sumExp += expf(v - maxv);
205 }
206 const float lse = logf(sumExp) + maxv;
207 rowLSE[r] = lse;
208 meanLR[r] = sumLogits / static_cast<float>(C_) - lse;
209
210 if (writeLP && outlpPtr) {
211 float* dst = outlpPtr + n * sN_out_ + s;
212 for (int c = 0; c < C_; ++c) dst[c * sC_out_] = tempbuf[c] - lse;
213 }
214 }
215 }, nstripes);
216
217 const bool expanded = (labels.dims == logits.dims && labels.size[1] == C) || softLabel;
218 const float eps = std::max(0.f, std::min(1.f, labelSmoothing));
219
220 Mat per(N*S, 1, CV_32F, Scalar(0));
221 Mat validMask(N*S, 1, CV_8U, Scalar(1));
222 Mat effW(N*S, 1, CV_32F, Scalar(1));
223
224 if (!expanded)
225 {
226 CV_Assert(labels.depth() == CV_32S || labels.depth() == CV_64S);
227 CV_Assert((int)labels.total() == N*S);
228 Mat idx1D = labels.reshape(1, N*S);
229
230 const float* wptr = hasWeight ? w.ptr<float>() : nullptr;
231 if (labels.depth() == CV_32S)
232 {
233 if (hasWeight)
234 reduceSCEPerSample<int32_t, true>(logits32, sN, sC, S, idx1D, C, ignoreIndex, eps, meanLogRow.data(), rowLogSumExp.data(), wptr, per, effW, validMask, nstripes);
235 else
236 reduceSCEPerSample<int32_t, false>(logits32, sN, sC, S, idx1D, C, ignoreIndex, eps, meanLogRow.data(), rowLogSumExp.data(), wptr, per, effW, validMask, nstripes);
237 }
238 else
239 {
240 if (hasWeight)
241 reduceSCEPerSample<int64_t, true>(logits32, sN, sC, S, idx1D, C, ignoreIndex, eps, meanLogRow.data(), rowLogSumExp.data(), wptr, per, effW, validMask, nstripes);
242 else
243 reduceSCEPerSample<int64_t, false>(logits32, sN, sC, S, idx1D, C, ignoreIndex, eps, meanLogRow.data(), rowLogSumExp.data(), wptr, per, effW, validMask, nstripes);
244 }
245 }
246 else
247 {
248 Mat labels32; labels.convertTo(labels32, CV_32F);
249 const size_t sN_lab = labels32.step1(0);
250 const size_t sC_lab = labels32.step1(1);
251
252 const float smoothMul = (!softLabel && eps > 0.f) ? (1.f - eps) : 1.f;
253 const float smoothAdd = (!softLabel && eps > 0.f) ? (eps / C) : 0.f;
254 std::vector<float> onesW;
255 const float* wBase = nullptr;
256 if (hasWeight){
257 wBase = w.ptr<float>();
258 } else {
259 onesW.assign(C, 1.f);
260 wBase = onesW.data();
261 }
262
263 parallel_for_(Range(0, N*S), [&](const Range& rr){
264 const float* baseLogits = logits32.ptr<float>();
265 const float* baseLabels = labels32.ptr<float>();
266 const float* wData = wBase;
267 float* perData = per.ptr<float>();
268 float* effWData = effW.ptr<float>();
269 uchar* validMaskData = validMask.ptr<uchar>();
270 float* rowLSE = rowLogSumExp.data();
271 const size_t sN_ = sN;
272 const size_t sC_ = sC;
273 const size_t sN_lab_ = sN_lab;
274 const size_t sC_lab_ = sC_lab;
275 const int S_ = S;
276 const int C_ = C;
277 const float smoothMul_ = smoothMul;
278 const float smoothAdd_ = smoothAdd;
279
280 const size_t sPer = per.step1(0);
281 const size_t sEff = effW.step1(0);
282 const size_t sVM = validMask.step1(0);
283
284 for (int r = rr.start; r < rr.end; ++r) {
285 const int n = r / S_;
286 const int s = r % S_;
287 const float* arow = baseLabels + n * sN_lab_ + s;
288 const float* lrow = baseLogits + n * sN_ + s;
289
290 double dot = 0.0, wsum = 0.0;
291 for (int c = 0; c < C_; ++c) {
292 float a = arow[c * sC_lab_] * smoothMul_ + smoothAdd_;
293 a *= wData[c];
294 wsum += static_cast<double>(a);
295 dot += static_cast<double>(a) * static_cast<double>(lrow[c * sC_]);
296 }
297
298 const float m = (wsum != 0.0) ? 1.f : 0.f;
299 validMaskData[r * sVM] = static_cast<uchar>(m > 0.f);
300 effWData[r * sEff] = static_cast<float>(wsum * m);
301 perData[r * sPer] = static_cast<float>(-(dot - static_cast<double>(rowLSE[r]) * wsum) * m);
302 }
303 }, nstripes);
304 }
305
306 if (!isReduced)
307 {
308 per.reshape(1, out_loss.size).copyTo(out_loss);
309 }
310 else
311 {
312 double num = 0.0, den = 0.0;
313 const float* perData = per.ptr<float>();
314 const float* effData = effW.ptr<float>();
315 const uchar* validData = validMask.ptr<uchar>();
316 const int R = per.rows;
317
318 for (int r = 0; r < R; ++r) {
319 const float m = validData[r] ? 1.f : 0.f;
320 num += static_cast<double>(perData[r] * m);
321 if (reduction == LOSS_REDUCTION_MEAN)
322 {
323 den += static_cast<double>(std::max(1e-12f, effData[r]) * m);
324 }
325 }
326
327 const float out = (reduction == LOSS_REDUCTION_SUM) ? static_cast<float>(num)
328 : static_cast<float>((den > 0.0) ? (num / den) : 0.0);
329 out_loss.ptr<float>()[0] = out;
330 }
331 }

333private:
334 template<typename T, bool UseWeight>
335 static inline void reduceSCEPerSample(const Mat& logits32,
336 const size_t sN_,
337 const size_t sC_,
338 const int S_,
339 const Mat& idx1D,
340 const int C_,
341 const int ignoreIndex,
342 const float eps,
343 const float* meanLogRowData,
344 const float* rowLogSumExp,
345 const float* weightBase,
346 Mat& per,
347 Mat& effW,
348 Mat& validMask,
349 const int nstripes)
350 {
351 CV_Assert(idx1D.cols == 1);
352 parallel_for_(Range(0, idx1D.rows), [&](const Range& rr){
353 const float* base = logits32.ptr<float>();
354 const T* idx1DData = idx1D.ptr<T>();
355 uchar* validMaskData = validMask.ptr<uchar>();
356 float* effWData = effW.ptr<float>();
357 float* perData = per.ptr<float>();
358
359 const size_t sIdx = idx1D.step1(0);
360 const size_t sVM = validMask.step1(0);
361 const size_t sEff = effW.step1(0);
362 const size_t sPer = per.step1(0);
363
364 const size_t sN = sN_;
365 const size_t sC = sC_;
366 const int S = S_;
367 const int C = C_;
368
369 for (int r = rr.start; r < rr.end; ++r)
370 {
371 const T y_raw = idx1DData[r * sIdx];
372 if ((int64_t)y_raw == (int64_t)ignoreIndex)
373 {
374 validMaskData[r * sVM] = 0;
375 effWData[r * sEff] = 0.f;
376 continue;
377 }
378
379 const int64_t yi64 = (int64_t)y_raw;
380 CV_Assert(yi64 >= 0 && yi64 < (int64_t)C);
381 const int y = static_cast<int>(yi64);
382
383 const int n = r / S;
384 const int s = r % S;
385 const float logits_y = base[n * sN + y * sC + s];
386 const float lp_y = logits_y - rowLogSumExp[r];
387 const float cw = UseWeight ? weightBase[y] : 1.f;
388
389 float loss = -(1.f - eps) * lp_y;
390 if (eps > 0.f) loss += -eps * meanLogRowData[r];
391
392 perData[r * sPer] = loss * cw;
393 effWData[r * sEff] = cw;
394 }
395 }, nstripes);
396 }
