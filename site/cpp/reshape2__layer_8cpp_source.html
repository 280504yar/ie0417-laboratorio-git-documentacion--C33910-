22/*
23 Reshape2 layer, as defined in ONNX specification:
24 https://onnx.ai/onnx/operators/onnx__Reshape.html
25
26 Opset's 1 to 23 are covered.
27
28 The layers Flatten, Reshape2, Squeeze and Unsqueeze all share the same
29 implementation idea:
30 1. calculate shape of the output tensor
31 2. assuming that the input is continuous, just copy all the data to output tensor.
32 reshapeAndCopyFirst() does that.
33 The engine buffer allocator recognizes all these operations and tries to run
34 them in-place. In such a case no copy operation is actually done,
35 so the operations are really cheap.
36*/
37

43 static bool sameDims(const MatShape& a, const MatShape& b)
44 {
45 if (a.dims != b.dims)
46 return false;
47 for (int i = 0; i < a.dims; i++) {
48 if (a[i] != b[i])
49 return false;
50 }
51 return true;
52 }

54 DataLayout getOriginalLayout() const
55 {
56 Net::Impl* netimpl_ = getNetImpl(this);
57 DataLayout layout = netimpl_ ? netimpl_->originalLayout : DATA_LAYOUT_NCHW;
58 CV_Assert(layout == DATA_LAYOUT_NCHW || layout == DATA_LAYOUT_NHWC);
59 return layout;
60 }

62 bool getConstShapeSpec(MatShape& shapeSpec) const
63 {
64 if (!dynamicShapeSpec) {
65 shapeSpec = newShapeDesc;
66 return true;
67 }
68
69 if (this->inputs.size() != 2)
70 return false;
71
72 Net::Impl* netimpl_ = getNetImpl(this);
73 if (!netimpl_ || !netimpl_->isConstArg(this->inputs[1]))
74 return false;
75
76 Mat shapeTensor = netimpl_->argTensor(this->inputs[1]);
77 shapeSpec = tensorToShape(shapeTensor);
78 return true;
79 }
80
81 bool canKeepBlockLayout(const MatShape& inpShape, const MatShape& shapeSpec) const
82 {
83 if (inpShape.layout != DATA_LAYOUT_BLOCK || inpShape.dims != 5)
84 return false;
85
86 DataLayout origLayout = getOriginalLayout();
87 MatShape semanticInp = inpShape.toLayout(origLayout);
88 MatShape semanticOut = getOutShape(semanticInp, shapeSpec);
89 if (sameDims(semanticOut, semanticInp))
90 return true;
91 if (semanticOut.dims != 4 || semanticInp.dims != 4)
92 return false;
93 if (origLayout == DATA_LAYOUT_NCHW)
94 return semanticOut[0] == semanticInp[0] && semanticOut[1] == semanticInp[1];
95 else
96 return semanticOut[0] == semanticInp[0] && semanticOut[3] == semanticInp[3];
97 }

99 Reshape2LayerImpl(const LayerParams& params)
100 {
101 dynamicShapeSpec = true;
102 setParamsFrom(params);
103 if (params.has("shape"))
104 {
105 dynamicShapeSpec = false;
106
107 const DictValue& shapeParam = params.get("shape");
108 int i, ndims = shapeParam.size();
109 newShapeDesc.resize(ndims);
110 for (i = 0; i < ndims; i++) {
111 newShapeDesc[i] = shapeParam.get<int>(i);
112 }
113 }
114 }

118 virtual bool dynamicOutputShapes() const CV_OVERRIDE
119 {
120 if (!dynamicShapeSpec)
121 return false;
122 // When the shape comes from a 2nd input, check if it's a constant tensor.
123 // If so, the output shape can still be inferred.
124 Net::Impl* netimpl_ = getNetImpl(this);
125 if (netimpl_ && this->inputs.size() == 2)
126 return !netimpl_->isConstArg(this->inputs[1]);
127 return true;

139 bool haveShapeSpec() const
140 {
141 return newShapeDesc.dims >= 0;
142 }
143
144 MatShape getOutShape(const MatShape& inpShape, const MatShape& shapeSpec) const
145 {
146 MatShape outShape = shapeSpec;
147 int m1idx = -1;
148 int i, ndims = outShape.dims;
149 int64_t outTotal = 1;
150 for (i = 0; i < ndims; i++) {
151 if (outShape[i] < 0) {
152 CV_Assert(outShape[i] == -1);
153 if (m1idx >= 0) {
154 CV_Error(Error::StsBadArg, "invalid shape spec, there must be at most one '-1'");
155 }
156 m1idx = i;
157 }
158 else {
159 if (outShape[i] == 0) {
160 if (i >= inpShape.dims) {
161 CV_Error(Error::StsBadArg, "cannot copy dimension from the input tensor");
162 }
163 outShape[i] = inpShape[i];
164 }
165 outTotal *= outShape[i];
166 }
167 }
168
169 int64_t inpTotal = (int64_t)inpShape.total();
170 if (m1idx >= 0) {
171 int64_t autoSize = inpTotal/outTotal;
172 CV_Assert(autoSize <= INT_MAX && autoSize*outTotal == inpTotal);
173 outShape[m1idx] = (int)autoSize;
174 }
175 else if (outTotal != inpTotal && ndims == 2 && outShape[0] == 1 && outShape[1] > 0) {
176 outShape[1] = (int)(inpTotal / outShape[0]);
177 }
178 else {
179 CV_Assert(outTotal == inpTotal);
180 }
181
182 return outShape;
183 }

185 bool getMemoryShapes(const std::vector<MatShape> &inputs,
186 const int,
187 std::vector<MatShape> &outputs,
188 std::vector<MatShape> &internals) const CV_OVERRIDE
189 {
190 bool haveShapeSpec_ = haveShapeSpec();
191 CV_Assert((inputs.size() == 1 && haveShapeSpec_) ||
192 (inputs.size() == 2 && !haveShapeSpec_));
193 MatShape shapeSpec = newShapeDesc, outShape;
194
195 if (inputs.size() == 2)
196 {
197 CV_Assert(this->inputs.size() == 2);
198 Net::Impl* netimpl_ = getNetImpl(this);
199 Mat shapeTensor = netimpl_->argTensor(this->inputs[1]);
200 shapeSpec = tensorToShape(shapeTensor);
201 } else {
202 CV_Assert(shapeSpec.dims >= 0);
203 }
204
205 if (inputs[0].layout == DATA_LAYOUT_BLOCK) {
206 if (canKeepBlockLayout(inputs[0], shapeSpec)) {
207 DataLayout origLayout = getOriginalLayout();
208 MatShape semanticInp = inputs[0].toLayout(origLayout);
209 MatShape semanticOut = getOutShape(semanticInp, shapeSpec);
210 if (sameDims(semanticOut, semanticInp)) {
211 outputs.assign(1, inputs[0]);
212 } else {
213 outputs.assign(1, semanticOut.toLayout(DATA_LAYOUT_BLOCK, inputs[0][4]));
214 }
215 } else {
216 DataLayout origLayout = getOriginalLayout();
217 MatShape semanticInp = inputs[0].toLayout(origLayout);
218 outputs.assign(1, getOutShape(semanticInp, shapeSpec));
219 }
220 } else {
221 outputs.assign(1, getOutShape(inputs[0], shapeSpec));
222 }
223 internals.clear();
224 return true;
225 }

227 int getLayouts(const std::vector<DataLayout>& actualInputs,
228 std::vector<DataLayout>& desiredInputs,
229 const int requiredOutputs,
230 std::vector<DataLayout>& outputs) const CV_OVERRIDE
231 {
232 size_t ninputs = actualInputs.size();
233 CV_Assert(ninputs == 1u || ninputs == 2u);
234
235 desiredInputs.assign(ninputs, DATA_LAYOUT_UNKNOWN);
236 desiredInputs[0] = actualInputs[0];
237 outputs.assign(requiredOutputs, actualInputs[0]);
238
239 if (actualInputs[0] == DATA_LAYOUT_BLOCK) {
240 bool canUseBlock = false;
241 MatShape shapeSpec;
242 if (getConstShapeSpec(shapeSpec)) {
243 DataLayout origLayout = getOriginalLayout();
244 int cAxis = (origLayout == DATA_LAYOUT_NCHW) ? 1 : 3;
245 if (shapeSpec.dims == 4 && cAxis < shapeSpec.dims && shapeSpec[cAxis] == 0)
246 canUseBlock = true;
247 }
248 if (!canUseBlock) {
249 desiredInputs[0] = getOriginalLayout();
250 outputs.assign(requiredOutputs, DATA_LAYOUT_UNKNOWN);
251 }
252 }
253
254 if (ninputs > 1)
255 desiredInputs[1] = DATA_LAYOUT_UNKNOWN;
256
257 return outputs[0] == DATA_LAYOUT_BLOCK ? getNetImpl(this)->defaultC0 : 0;
258 }

277 void forward(InputArrayOfArrays inputs_arr,
278 OutputArrayOfArrays outputs_arr,
279 OutputArrayOfArrays) CV_OVERRIDE
280 {
281 CV_TRACE_FUNCTION();
282 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
283
284 Size size = inputs_arr.size();
285 int ninputs = size.area();
286 bool haveShapeSpec_ = haveShapeSpec();
287 CV_Assert((ninputs == 1 && haveShapeSpec_) ||
288 (ninputs == 2 && !haveShapeSpec_));
289
290 MatShape inpShape = inputs_arr.shape(0);
291 CV_Assert(inpShape.layout != DATA_LAYOUT_BLOCK || inpShape.C > 0);
292 MatShape shapeSpec = newShapeDesc;
293 if (!haveShapeSpec_) {
294 Mat shapeTensor = inputs_arr.getMat(1);
295 shapeSpec = tensorToShape(shapeTensor);
296 }
297 MatShape outShape;
298 if (inpShape.layout == DATA_LAYOUT_BLOCK) {
299 if (canKeepBlockLayout(inpShape, shapeSpec)) {
300 DataLayout origLayout = getOriginalLayout();
301 MatShape semanticInp = inpShape.toLayout(origLayout);
302 MatShape semanticOut = getOutShape(semanticInp, shapeSpec);
303 if (sameDims(semanticOut, semanticInp)) {
304 outShape = inpShape;
305 } else {
306 outShape = semanticOut.toLayout(DATA_LAYOUT_BLOCK, inpShape[4]);
307 }
308 } else {
309 DataLayout origLayout = getOriginalLayout();
310 MatShape semanticInp = inpShape.toLayout(origLayout);
311 outShape = getOutShape(semanticInp, shapeSpec);
312 }
313 } else {
314 outShape = getOutShape(inpShape, shapeSpec);
315 }
316
317 reshapeAndCopyFirst(inputs_arr, outputs_arr, outShape);
318 }
