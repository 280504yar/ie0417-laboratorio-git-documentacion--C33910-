96 template <typename T>
97 void forwardImpl(const std::vector<Mat>& inputs, std::vector<Mat>& outputs)
98 {
99 // Get input tensor.
100 const auto& src_mat = inputs[0];
101 const T* src_ptr = src_mat.ptr<T>();
102
103 // Get target axis.
104 int axis = inputs.size() > 1 ? parseAxis(inputs[1]) : axis_raw;
105 axis = normalize_axis(axis, src_mat.dims);
106
107
108 // Get output tensor.
109 auto& dst_mat = outputs[0];
110 T* dst_ptr = dst_mat.ptr<T>();
111
112 // Get flags.
113 const auto exclusive = exclusive_raw == 1;
114 const auto reverse = reverse_raw == 1;
115
116 // Data with [dim_1, .. , dim_k-1, target_dim, dim_k+1, .. , dim_n]
117 // dimensions is represented here as [outer_dim, target_dim, inner_dim]
118 const size_t outer_size = src_mat.total(0, axis);
119 const size_t target_size = src_mat.size[axis];
120 const size_t inner_size = src_mat.total(axis + 1);
121 const size_t outer_step_length = target_size * inner_size;
122
123 // Calculating steps in target dimensions
124 const int target_start = reverse ? target_size - 1 : 0;
125 const int target_stop = reverse ? -1 : target_size;
126 const int target_delta = reverse ? -1 : 1;
127 const int target_step = target_delta * inner_size;
128
129 // If exclusive, the j-th output element would be the sum of the first (j-1) elements.
130 // Otherwise, it would be the sum of the first j elements.
131 const int exclusive_delta = exclusive ? target_step : 0;
132
133 for (size_t outer_idx = 0; outer_idx < outer_size; outer_idx++)
134 {
135 const size_t target_offset = outer_idx * outer_step_length;
136
137 // Handle first element of target dimension.
138 size_t first_inner_offset = target_offset + target_start * inner_size;
139 if (exclusive)
140 for (size_t inner_idx = 0; inner_idx < inner_size; inner_idx++)
141 dst_ptr[first_inner_offset + inner_idx] = 0;
142 else
143 for (size_t inner_idx = 0; inner_idx < inner_size; inner_idx++)
144 dst_ptr[first_inner_offset + inner_idx] = src_ptr[first_inner_offset + inner_idx];
145
146 // Handle remaining elements of target dimension.
147 for (int target_idx = target_start + target_delta; target_idx != target_stop; target_idx += target_delta)
148 {
149 const size_t inner_offset = target_offset + target_idx * inner_size;
150
151 for (size_t inner_idx = 0; inner_idx < inner_size; inner_idx++)
152 {
153 dst_ptr[inner_offset + inner_idx] = dst_ptr[inner_offset - target_step + inner_idx] +
154 src_ptr[inner_offset - exclusive_delta + inner_idx];
155 }
156 }
157 }
158 }

160 int parseAxis(const Mat& axis_mat) {
161 CV_CheckEQ(axis_mat.total(), 1u, "Axis tensor should contain single value");
162 if (axis_mat.type() == CV_32SC1)
163 return axis_mat.at<int32_t>(0);
164 else
165 {
166 Mat axis_mat_int;
167 axis_mat.convertTo(axis_mat_int, CV_32SC1);
168 return axis_mat_int.at<int32_t>(0);
