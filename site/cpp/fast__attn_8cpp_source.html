36template <typename MaskT>
37struct MaskPolicyFloat {
38 static inline v_float32 load_mask(const MaskT* ptr) { return vx_load((const float*)ptr); }
39 static inline v_float32 load_mask_scalar(const MaskT* ptr) { return vx_setall_f32((float)*ptr); }
40 static inline v_float32 apply(v_float32 val, v_float32 mask, v_float32 min_val) { return v_add(val, mask); }
41 static inline float apply_scalar(float val, MaskT mask, float min_val) { return val + mask; }
42};
43
44template <typename MaskT>
45struct MaskPolicyInt {
46 static inline v_float32 load_mask(const MaskT* ptr) { return load_int_mask_as_f32(ptr); }
47 static inline v_float32 load_mask_scalar(const MaskT* ptr) {
48 return (*ptr != 0) ? v_reinterpret_as_f32(vx_setall_u32(0xFFFFFFFF)) : vx_setall_f32(0.f);
49 }
50 static inline v_float32 apply(v_float32 val, v_float32 mask, v_float32 min_val) { return v_select(mask, val, min_val); }
51 static inline float apply_scalar(float val, MaskT mask, float min_val) { return (mask != 0) ? val : min_val; }
52};
53
54struct MaskPolicyNone {
55 static inline v_float32 load_mask(const void* ptr) { return vx_setzero_f32(); }
56 static inline v_float32 load_mask_scalar(const void* ptr) { return vx_setzero_f32(); }
57 static inline v_float32 apply(v_float32 val, v_float32 mask, v_float32 min_val) { return val; }
58 static inline float apply_scalar(float val, int mask, float min_val) { return val; }
59};

61template <typename MaskT, typename Policy>
62void run_fused_softmax(
63 Mat &att_weights, const Mat &att_mask,
64 const float softcap, const bool do_softcap,
65 const float threshold,
66 const float min_val, const bool is_causal, const int past_seq_len,
67 const bool do_softmax)
68{
69 const int batch_size = att_weights.size[0];
70 const int n_heads = att_weights.size[1];
71 const int seq_len_q = att_weights.size[2];
72 const int seq_len_kv = att_weights.size[3];
73
74 float* data = att_weights.ptr<float>();
75 const MaskT* mask_data = att_mask.empty() ? nullptr : att_mask.ptr<MaskT>();
76
77 size_t mask_step_b = 0, mask_step_h = 0, mask_step_q = 0, mask_step_k = 0;
78 if (mask_data) {
79 int dims_m = att_mask.dims;
80 int offset_dim = 4 - dims_m;
81
82 auto get_size = [&](int i) { return (i < offset_dim) ? 1 : att_mask.size[i - offset_dim]; };
83 auto get_step = [&](int i) { return (i < offset_dim) ? 0 : att_mask.step[i - offset_dim] / sizeof(MaskT); };
84
85 if (get_size(0) > 1) mask_step_b = get_step(0);
86 if (get_size(1) > 1) mask_step_h = get_step(1);
87 if (get_size(2) > 1) mask_step_q = get_step(2);
88 if (get_size(3) > 1) mask_step_k = 1;
89 }
90
91 size_t total_tasks = (size_t)batch_size * n_heads;
92
93 parallel_for_(Range(0, (int)total_tasks), [&](const Range &range) {
94 for (int i = range.start; i < range.end; i++) {
95 int b = i / n_heads;
96 int h = i % n_heads;
97
98 size_t offset = (size_t)i * seq_len_q * seq_len_kv;
99 size_t mask_offset_q = b * mask_step_b + h * mask_step_h;
100
101 for (int tq = 0; tq < seq_len_q; tq++){
102 const int tmax = is_causal ? std::min(past_seq_len + tq + 1, seq_len_kv) : seq_len_kv;
103 float maxVal = -FLT_MAX;
104 int tk = 0;
105#if (CV_SIMD || CV_SIMD_SCALABLE)
106 const int w = VTraits<v_float32>::vlanes();
107 v_float32 v_max_val = vx_setall_f32(maxVal);
108 v_float32 v_softcap = vx_setall_f32(softcap);
109 v_float32 v_inv_softcap = vx_setall_f32(1.f / softcap);
110 v_float32 v_threshold = vx_setall_f32(threshold);
111 v_float32 v_minus_threshold = vx_setall_f32(-threshold);
112 v_float32 v_one = vx_setall_f32(1.f);
113 v_float32 v_minus_one = vx_setall_f32(-1.f);
114 v_float32 v_minus_two = vx_setall_f32(-2.f);
115 v_float32 v_min_val = vx_setall_f32(min_val);
116
117 for (; tk <= tmax - w; tk += w) {
118 v_float32 v_val = vx_load(&data[offset + tk]);
119
120 if (mask_data) {
121 v_float32 v_mask_val;
122 if (mask_step_k)
123 v_mask_val = Policy::load_mask(&mask_data[mask_offset_q + tk]);
124 else
125 v_mask_val = Policy::load_mask_scalar(&mask_data[mask_offset_q]);
126 v_val = Policy::apply(v_val, v_mask_val, v_min_val);
127 }
128
129 if (do_softcap) {
130 v_float32 v_scaled = v_mul(v_val, v_inv_softcap);
131
132 v_float32 v_mask_pos = v_gt(v_scaled, v_threshold);
133 v_float32 v_mask_neg = v_lt(v_scaled, v_minus_threshold);
134
135 v_float32 v_scaled_safe = v_select(v_mask_neg, vx_setall_f32(0.f), v_scaled);
136
137 v_float32 v_exp_part = v_exp(v_mul(v_minus_two, v_scaled_safe));
138 v_float32 v_tanh = v_div(v_sub(v_one, v_exp_part), v_add(v_one, v_exp_part));
139
140 v_float32 v_res = v_select(v_mask_pos, v_one,
141 v_select(v_mask_neg, v_minus_one, v_tanh));
142
143 v_val = v_mul(v_res, v_softcap);
144 }
145 vx_store(&data[offset + tk], v_val);
146 v_max_val = v_max(v_max_val, v_val);
147 }
148 maxVal = v_reduce_max(v_max_val);
149#endif
150 for (; tk < tmax; tk++){
151 if (mask_data) {
152 size_t mask_idx = mask_offset_q + (mask_step_k ? tk : 0);
153 data[offset + tk] = Policy::apply_scalar(data[offset + tk], mask_data[mask_idx], min_val);
154 }
155
156 if (do_softcap) {
157 float softcap_val = data[offset + tk] / softcap;
158 if (softcap_val > threshold)
159 data[offset + tk ] = 1.f;
160 else if(softcap_val < -threshold)
161 data[offset + tk ] = -1.f;
162 else
163 data[offset + tk] = (1.f - expf(-2 *softcap_val)) / (1.f + expf(-2 *softcap_val));
164 data[offset + tk] *= softcap;
165 }
166 maxVal = std::max(maxVal, data[offset + tk]);
167 }
168
169 for (; tk < seq_len_kv; tk++) {
170 data[offset + tk] = min_val;
171 }
172
173 if (do_softmax) {
174 float sum = 0.f;
175 tk = 0;
176#if (CV_SIMD || CV_SIMD_SCALABLE)
177 v_float32 v_sum = vx_setzero_f32();
178 v_float32 v_max_val_shift = vx_setall_f32(maxVal);
179 for (; tk <= seq_len_kv - w; tk += w) {
180 v_float32 v_val = vx_load(&data[offset + tk]);
181 v_val = v_sub(v_val, v_max_val_shift);
182 v_val = v_exp(v_val);
183 v_sum = v_add(v_sum, v_val);
184 vx_store(&data[offset + tk], v_val);
185 }
186 sum = v_reduce_sum(v_sum);
187#endif
188 for (; tk < seq_len_kv; tk++){
189 data[offset + tk] = expf(data[offset + tk] - maxVal);
190 sum += data[offset + tk];
191 }
192
193 float inv_sum = 1.f / sum;
194 tk = 0;
195#if (CV_SIMD || CV_SIMD_SCALABLE)
196 v_float32 v_inv_sum = vx_setall_f32(inv_sum);
197 for (; tk <= seq_len_kv - w; tk += w) {
198 v_float32 v_val = vx_load(&data[offset + tk]);
199 v_val = v_mul(v_val, v_inv_sum);
200 vx_store(&data[offset + tk], v_val);
201 }
202#endif
203 for (; tk < seq_len_kv; tk++){
204 data[offset + tk] *= inv_sum;
205 }
206 } // do_softmax
207
208 offset += seq_len_kv;
209 mask_offset_q += mask_step_q;
210 }
211 }
212 });
213}

215void fused_softmax_softcap_mask(
216 Mat &att_weights,const Mat &att_mask,
217 const float softcap, const bool do_softcap,
218 const float threshold,
219 const float min_val, const bool has_mask, const bool is_causal,
220 int past_seq_len, bool do_softmax
221){
222 if (has_mask) {
223 switch(att_mask.depth())
224 {
225 case CV_32F:
226 run_fused_softmax<float, MaskPolicyFloat<float>>(att_weights, att_mask, softcap, do_softcap, threshold, min_val, is_causal, past_seq_len, do_softmax);
227 break;
228 case CV_Bool:
229 case CV_8U:
230 run_fused_softmax<uint8_t, MaskPolicyInt<uint8_t>>(att_weights, att_mask, softcap, do_softcap, threshold, min_val, is_causal, past_seq_len, do_softmax);
231 break;
232 case CV_8S:
233 run_fused_softmax<int8_t, MaskPolicyInt<int8_t>>(att_weights, att_mask, softcap, do_softcap, threshold, min_val, is_causal, past_seq_len, do_softmax);
234 break;
235 case CV_16U:
236 run_fused_softmax<uint16_t, MaskPolicyInt<uint16_t>>(att_weights, att_mask, softcap, do_softcap, threshold, min_val, is_causal, past_seq_len, do_softmax);
237 break;
238 case CV_16S:
239 run_fused_softmax<int16_t, MaskPolicyInt<int16_t>>(att_weights, att_mask, softcap, do_softcap, threshold, min_val, is_causal, past_seq_len, do_softmax);
240 break;
241 case CV_32U:
242 run_fused_softmax<uint32_t, MaskPolicyInt<uint32_t>>(att_weights, att_mask, softcap, do_softcap, threshold, min_val, is_causal, past_seq_len, do_softmax);
243 break;
244 case CV_32S:
245 run_fused_softmax<int32_t, MaskPolicyInt<int32_t>>(att_weights, att_mask, softcap, do_softcap, threshold, min_val, is_causal, past_seq_len, do_softmax);
246 break;
247 case CV_64U:
248 run_fused_softmax<uint64_t, MaskPolicyInt<uint64_t>>(att_weights, att_mask, softcap, do_softcap, threshold, min_val, is_causal, past_seq_len, do_softmax);
249 break;
250 case CV_64S:
251 run_fused_softmax<int64_t, MaskPolicyInt<int64_t>>(att_weights, att_mask, softcap, do_softcap, threshold, min_val, is_causal, past_seq_len, do_softmax);
252 break;
253 default:
254 CV_Error(Error::StsUnsupportedFormat, "Unsupported mask data type in fused_softmax_softcap_mask");
255 }
256 } else {
257 run_fused_softmax<float, MaskPolicyNone>(att_weights, att_mask, softcap, do_softcap, threshold, min_val, is_causal, past_seq_len, do_softmax);
258 }
259}
