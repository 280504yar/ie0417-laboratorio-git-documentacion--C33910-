127bool ExrDecoder::readHeader()
128{
129 bool result = false;
130
131 m_file = new InputFile( m_filename.c_str() );
132
133 m_datawindow = m_file->header().dataWindow();
134 m_width = m_datawindow.max.x - m_datawindow.min.x + 1;
135 m_height = m_datawindow.max.y - m_datawindow.min.y + 1;
136
137 // the type HALF is converted to 32 bit float
138 // and the other types supported by OpenEXR are 32 bit anyway
139 m_bit_depth = 32;
140
141 if( hasChromaticities( m_file->header() ))
142 m_chroma = chromaticities( m_file->header() );
143
144 const ChannelList &channels = m_file->header().channels();
145 m_red = channels.findChannel( "R" );
146 m_green = channels.findChannel( "G" );
147 m_blue = channels.findChannel( "B" );
148 m_alpha = channels.findChannel( "A" );
149 m_multispectral = channels.findChannel( "0" ) != nullptr;
150
151 if( m_alpha ) // alpha channel supported in RGB, Y, and YC scenarios
152 m_hasalpha = true;
153
154 if( m_red || m_green || m_blue )
155 {
156 m_iscolor = true;
157 m_ischroma = false;
158 result = true;
159 }
160 else if( m_multispectral )
161 {
162 m_channels = 0;
163 for( auto it = channels.begin(); it != channels.end(); it++ )
164 m_channels++;
165
166 m_iscolor = true; // ??? false
167 m_ischroma = false;
168 m_hasalpha = false;
169 result = m_channels <= CV_CN_MAX;
170
171 for ( int i = 1; result && i < m_channels; i++ ) // channel 0 was found previously
172 {
173 const Channel *ch = channels.findChannel( std::to_string(i) );
174 result = ch && ch->xSampling == 1 && ch->ySampling == 1; // subsampling is not supported
175 }
176 }
177 else
178 {
179 m_green = channels.findChannel( "Y" );
180 if( !m_green )
181 {
182 m_green = channels.findChannel( "Z" ); // Distance of the front of a sample from the viewer
183 }
184 if( m_green )
185 {
186 m_ischroma = true;
187 m_red = channels.findChannel( "RY" );
188 m_blue = channels.findChannel( "BY" );
189 m_iscolor = (m_blue || m_red);
190 result = true;
191 }
192 else
193 result = false;
194 }
195
196 if( result )
197 {
198 m_type = FLOAT;
199 m_isfloat = ( m_type == FLOAT );
200 }
201
202 if( !result )
203 close();
204
205 return result;
206}

209bool ExrDecoder::readData( Mat& img )
210{
211 m_native_depth = CV_MAT_DEPTH(type()) == img.depth();
212 bool multispectral = img.channels() > 4;
213 bool color = img.channels() > 2; // output mat has 3+ channels; Y or YA are the 1 and 2 channel scenario
214 bool alphasupported = !multispectral && ( img.channels() % 2 == 0 ); // even number of channels indicates alpha
215 int channels = 0;
216 uchar* data = img.ptr();
217 size_t step = img.step;
218 bool justcopy = ( m_native_depth && (color == m_iscolor) );
219 bool chromatorgb = ( m_ischroma && color );
220 bool rgbtogray = ( !m_ischroma && m_iscolor && !color );
221 bool result = true;
222 FrameBuffer frame;
223 const int defaultchannels = 3;
224 int xsample[defaultchannels] = {1, 1, 1};
225 char *buffer;
226 CV_Assert(m_type == FLOAT);
227 const size_t floatsize = sizeof(float);
228 size_t xstep = m_native_depth ? floatsize : 1; // 4 bytes if native depth (FLOAT), otherwise converting to 1 byte U8 depth
229 size_t ystep = 0;
230 const int channelstoread = ( multispectral ? img.channels() : (m_iscolor && alphasupported) ? 4 :
231 ( (m_iscolor && !m_ischroma) || color) ? 3 : alphasupported ? 2 : 1 ); // number of channels to read may exceed channels in output img
232 size_t xStride = floatsize * channelstoread;
233
234 if ( m_multispectral ) // possible gray/RGB conversions
235 {
236 CV_CheckChannelsEQ(img.channels(), CV_MAT_CN(type()), "OpenCV EXR decoder needs more number of channels for multispectral images. Use cv::IMREAD_UNCHANGED mode for imread."); // IMREAD_ANYCOLOR needed
237 CV_CheckDepthEQ(img.depth(), CV_MAT_DEPTH(type()), "OpenCV EXR decoder supports CV_32F depth only for multispectral images. Use cv::IMREAD_UNCHANGED mode for imread."); // IMREAD_ANYDEPTH needed
238 }
239 CV_Assert( multispectral == m_multispectral && (!multispectral || justcopy) ); // should be true after previous checks
240
241 // See https://github.com/opencv/opencv/issues/26705
242 // If ALGO_HINT_ACCURATE is set, read BGR and swap to RGB.
243 // If ALGO_HINT_APPROX is set, read RGB directly.
244 bool doReadRGB = m_use_rgb;
245 bool doPostColorSwap = false; // After decoding, swap BGR to RGB
246 if(m_use_rgb && (getDefaultAlgorithmHint() == ALGO_HINT_ACCURATE) )
247 {
248 doReadRGB = false;
249 doPostColorSwap = true;
250 }
251
252 AutoBuffer<char> copy_buffer;
253
254 if( !justcopy )
255 {
256 copy_buffer.allocate(floatsize * m_width * defaultchannels);
257 buffer = copy_buffer.data();
258 ystep = 0;
259 }
260 else
261 {
262 buffer = (char *)data;
263 ystep = step;
264 }
265
266 if( m_ischroma )
267 {
268 if( color )
269 {
270 if( m_blue )
271 {
272 frame.insert( "BY", Slice( m_type,
273 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep,
274 xStride, ystep, m_blue->xSampling, m_blue->ySampling, 0.0 ));
275 xsample[0] = m_blue->xSampling;
276 }
277 else
278 {
279 frame.insert( "BY", Slice( m_type,
280 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep,
281 xStride, ystep, 1, 1, 0.0 ));
282 }
283 if( m_green )
284 {
285 frame.insert( "Y", Slice( m_type,
286 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep + floatsize,
287 xStride, ystep, m_green->xSampling, m_green->ySampling, 0.0 ));
288 xsample[1] = m_green->xSampling;
289 }
290 else
291 {
292 frame.insert( "Y", Slice( m_type,
293 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep + floatsize,
294 xStride, ystep, 1, 1, 0.0 ));
295 }
296 if( m_red )
297 {
298 frame.insert( "RY", Slice( m_type,
299 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep + (floatsize * 2),
300 xStride, ystep, m_red->xSampling, m_red->ySampling, 0.0 ));
301 xsample[2] = m_red->xSampling;
302 }
303 else
304 {
305 frame.insert( "RY", Slice( m_type,
306 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep + (floatsize * 2),
307 xStride, ystep, 1, 1, 0.0 ));
308 }
309 }
310 else
311 {
312 frame.insert( "Y", Slice( m_type,
313 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep,
314 xStride, ystep, m_green->xSampling, m_green->ySampling, 0.0 ));
315 xsample[0] = m_green->xSampling;
316 }
317 }
318 else if( m_multispectral )
319 {
320 for ( int i = 0; i < m_channels; i++ )
321 {
322 frame.insert( std::to_string(i), Slice( m_type,
323 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep + (floatsize * i),
324 xStride, ystep, 1, 1, 0.0 ));
325 }
326 }
327 else
328 {
329 if( m_blue )
330 {
331 frame.insert( "B", Slice( m_type,
332 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep,
333 xStride, ystep, m_blue->xSampling, m_blue->ySampling, 0.0 ));
334 xsample[0] = m_blue->xSampling;
335 }
336 else
337 {
338 frame.insert( "B", Slice( m_type,
339 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep,
340 xStride, ystep, 1, 1, 0.0 ));
341 }
342 if( m_green )
343 {
344 frame.insert( "G", Slice( m_type,
345 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep + floatsize,
346 xStride, ystep, m_green->xSampling, m_green->ySampling, 0.0 ));
347 xsample[1] = m_green->xSampling;
348 }
349 else
350 {
351 frame.insert( "G", Slice( m_type,
352 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep + floatsize,
353 xStride, ystep, 1, 1, 0.0 ));
354 }
355 if( m_red )
356 {
357 frame.insert( "R", Slice( m_type,
358 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep + (floatsize * 2),
359 xStride, ystep, m_red->xSampling, m_red->ySampling, 0.0 ));
360 xsample[2] = m_red->xSampling;
361 }
362 else
363 {
364 frame.insert( "R", Slice( m_type,
365 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep + (floatsize * 2),
366 xStride, ystep, 1, 1, 0.0 ));
367 }
368 }
369
370 if( justcopy && m_hasalpha && alphasupported )
371 { // alpha preserved only in justcopy scenario where alpha is desired (alphasupported)
372 // and present in original file (m_hasalpha)
373 CV_Assert(channelstoread == img.channels());
374 int offset = (channelstoread - 1) * floatsize;
375 frame.insert( "A", Slice( m_type,
376 buffer - m_datawindow.min.x * xStride - m_datawindow.min.y * ystep + offset,
377 xStride, ystep, m_alpha->xSampling, m_alpha->ySampling, 0.0 ));
378 }
379
380 for (FrameBuffer::Iterator it = frame.begin(); it != frame.end(); it++) {
381 channels++;
382 }
383
384 CV_Assert(channels == channelstoread);
385
386 if( (channels != channelstoread) || (!justcopy && channels > defaultchannels) )
387 { // safety checking what ought to be true here
388 close();
389 return false;
390 }
391
392 m_file->setFrameBuffer( frame );
393 if( justcopy )
394 {
395 m_file->readPixels( m_datawindow.min.y, m_datawindow.max.y );
396
397 if( !m_multispectral )
398 {
399 if( m_iscolor )
400 {
401 if (doReadRGB)
402 {
403 if( m_red && (m_red->xSampling != 1 || m_red->ySampling != 1) )
404 UpSample( data, channelstoread, step / xstep, m_red->xSampling, m_red->ySampling );
405 if( m_green && (m_green->xSampling != 1 || m_green->ySampling != 1) )
406 UpSample( data + xstep, channelstoread, step / xstep, m_green->xSampling, m_green->ySampling );
407 if( m_blue && (m_blue->xSampling != 1 || m_blue->ySampling != 1) )
408 UpSample( data + 2 * xstep, channelstoread, step / xstep, m_blue->xSampling, m_blue->ySampling );
409 }
410 else
411 {
412 if( m_blue && (m_blue->xSampling != 1 || m_blue->ySampling != 1) )
413 UpSample( data, channelstoread, step / xstep, m_blue->xSampling, m_blue->ySampling );
414 if( m_green && (m_green->xSampling != 1 || m_green->ySampling != 1) )
415 UpSample( data + xstep, channelstoread, step / xstep, m_green->xSampling, m_green->ySampling );
416 if( m_red && (m_red->xSampling != 1 || m_red->ySampling != 1) )
417 UpSample( data + 2 * xstep, channelstoread, step / xstep, m_red->xSampling, m_red->ySampling );
418 }
419 }
420 else if( m_green && (m_green->xSampling != 1 || m_green->ySampling != 1) )
421 UpSample( data, channelstoread, step / xstep, m_green->xSampling, m_green->ySampling );
422
423 if( chromatorgb )
424 {
425 if (doReadRGB)
426 ChromaToRGB( (float *)data, m_height, channelstoread, step / xstep );
427 else
428 ChromaToBGR( (float *)data, m_height, channelstoread, step / xstep );
429 }
430 }
431 }
432 else // m_multispectral should be false
433 {
434 uchar *out = data;
435 int x, y;
436 for( y = m_datawindow.min.y; y <= m_datawindow.max.y; y++ )
437 {
438 m_file->readPixels( y, y );
439
440 for( int i = 0; i < channels; i++ )
441 {
442 if( xsample[i] != 1 )
443 UpSampleX( (float *)buffer + i, channels, xsample[i] );
444 }
445 if( rgbtogray )
446 {
447 RGBToGray( (float *)buffer, (float *)out );
448 }
449 else
450 {
451 if( chromatorgb )
452 {
453 if (doReadRGB)
454 ChromaToRGB( (float *)buffer, 1, defaultchannels, step );
455 else
456 ChromaToBGR( (float *)buffer, 1, defaultchannels, step );
457 }
458
459 if( m_type == FLOAT )
460 {
461 float *fi = (float *)buffer;
462 for( x = 0; x < m_width * img.channels(); x++)
463 {
464 out[x] = cv::saturate_cast<uchar>(fi[x]);
465 }
466 }
467 else
468 {
469 unsigned *ui = (unsigned *)buffer;
470 for( x = 0; x < m_width * img.channels(); x++)
471 {
472 out[x] = cv::saturate_cast<uchar>(ui[x]);
473 }
474 }
475 }
476
477 out += step;
478 }
479 if( color )
480 {
481 if (doReadRGB)
482 {
483 if( m_red && (m_red->xSampling != 1 || m_red->ySampling != 1) )
484 UpSampleY( data, defaultchannels, step / xstep, m_red->ySampling );
485 if( m_green && (m_green->xSampling != 1 || m_green->ySampling != 1) )
486 UpSampleY( data + xstep, defaultchannels, step / xstep, m_green->ySampling );
487 if( m_blue && (m_blue->xSampling != 1 || m_blue->ySampling != 1) )
488 UpSampleY( data + 2 * xstep, defaultchannels, step / xstep, m_blue->ySampling );
489 }
490 else
491 {
492 if( m_blue && (m_blue->xSampling != 1 || m_blue->ySampling != 1) )
493 UpSampleY( data, defaultchannels, step / xstep, m_blue->ySampling );
494 if( m_green && (m_green->xSampling != 1 || m_green->ySampling != 1) )
495 UpSampleY( data + xstep, defaultchannels, step / xstep, m_green->ySampling );
496 if( m_red && (m_red->xSampling != 1 || m_red->ySampling != 1) )
497 UpSampleY( data + 2 * xstep, defaultchannels, step / xstep, m_red->ySampling );
498 }
499 }
500 else if( m_green && (m_green->xSampling != 1 || m_green->ySampling != 1) )
501 UpSampleY( data, 1, step / xstep, m_green->ySampling );
502 }
503
504 close();
505
506 if(doPostColorSwap)
507 {
508 cvtColor( img, img, cv::COLOR_BGR2RGB );
509 }
510
511 return result;
512}

518void ExrDecoder::UpSample( uchar *data, int xstep, int ystep, int xsample, int ysample )
519{
520 for( int y = (m_height - 1) / ysample, yre = m_height - ysample; y >= 0; y--, yre -= ysample )
521 {
522 for( int x = (m_width - 1) / xsample, xre = m_width - xsample; x >= 0; x--, xre -= xsample )
523 {
524 for( int i = 0; i < ysample; i++ )
525 {
526 for( int n = 0; n < xsample; n++ )
527 {
528 if( !m_native_depth )
529 data[(yre + i) * ystep + (xre + n) * xstep] = data[y * ystep + x * xstep];
530 else if( m_type == FLOAT )
531 ((float *)data)[(yre + i) * ystep + (xre + n) * xstep] = ((float *)data)[y * ystep + x * xstep];
532 else
533 ((unsigned *)data)[(yre + i) * ystep + (xre + n) * xstep] = ((unsigned *)data)[y * ystep + x * xstep];
534 }
535 }
536 }
537 }

584void ExrDecoder::ChromaToBGR( float *data, int numlines, int xstep, int ystep )
585{
586 for( int y = 0; y < numlines; y++ )
587 {
588 for( int x = 0; x < m_width; x++ )
589 {
590 double b, Y, r;
591 if( m_type == FLOAT )
592 {
593 b = data[y * ystep + x * xstep];
594 Y = data[y * ystep + x * xstep + 1];
595 r = data[y * ystep + x * xstep + 2];
596 }
597 else
598 {
599 b = ((unsigned *)data)[y * ystep + x * xstep];
600 Y = ((unsigned *)data)[y * ystep + x * xstep + 1];
601 r = ((unsigned *)data)[y * ystep + x * xstep + 2];
602 }
603 r = (r + 1) * Y;
604 b = (b + 1) * Y;
605 Y = (Y - b * m_chroma.blue[1] - r * m_chroma.red[1]) / m_chroma.green[1];
606
607 if( m_type == FLOAT )
608 {
609 data[y * ystep + x * xstep] = (float)b;
610 data[y * ystep + x * xstep + 1] = (float)Y;
611 data[y * ystep + x * xstep + 2] = (float)r;
612 }
613 else
614 {
615 int t = cvRound(b);
616 ((unsigned *)data)[y * ystep + x * xstep + 0] = (unsigned)MAX(t, 0);
617 t = cvRound(Y);
618 ((unsigned *)data)[y * ystep + x * xstep + 1] = (unsigned)MAX(t, 0);
619 t = cvRound(r);
620 ((unsigned *)data)[y * ystep + x * xstep + 2] = (unsigned)MAX(t, 0);
621 }
622 }
623 }

626void ExrDecoder::ChromaToRGB(float *data, int numlines, int xstep, int ystep)
627{
628 for( int y = 0; y < numlines; y++ )
629 {
630 for( int x = 0; x < m_width; x++ )
631 {
632 double b, Y, r;
633 if( m_type == FLOAT )
634 {
635 b = data[y * ystep + x * xstep];
636 Y = data[y * ystep + x * xstep + 1];
637 r = data[y * ystep + x * xstep + 2];
638 }
639 else
640 {
641 b = ((unsigned *)data)[y * ystep + x * xstep];
642 Y = ((unsigned *)data)[y * ystep + x * xstep + 1];
643 r = ((unsigned *)data)[y * ystep + x * xstep + 2];
644 }
645 r = (r + 1) * Y;
646 b = (b + 1) * Y;
647 Y = (Y - b * m_chroma.blue[1] - r * m_chroma.red[1]) / m_chroma.green[1];
648
649 if( m_type == FLOAT )
650 {
651 data[y * ystep + x * xstep] = (float)r;
652 data[y * ystep + x * xstep + 1] = (float)Y;
653 data[y * ystep + x * xstep + 2] = (float)b;
654 }
655 else
656 {
657 int t = cvRound(r);
658 ((unsigned *)data)[y * ystep + x * xstep + 0] = (unsigned)MAX(t, 0);
659 t = cvRound(Y);
660 ((unsigned *)data)[y * ystep + x * xstep + 1] = (unsigned)MAX(t, 0);
661 t = cvRound(b);
662 ((unsigned *)data)[y * ystep + x * xstep + 2] = (unsigned)MAX(t, 0);
663 }
664 }

672{

674 {
675 if( m_native_depth )
676 {
677 for( int i = 0, n = 0; i < m_width; i++, n += 3 )
678 out[i] = in[n] * m_chroma.blue[0] + in[n + 1] * m_chroma.green[0] + in[n + 2] * m_chroma.red[0];

685 }

690 {
691 unsigned *ui = (unsigned *)in;
692 for( int i = 0; i < m_width * 3; i++ )
693 ui[i] -= 0x80000000;
694 int *si = (int *)in;
695 for( int i = 0, n = 0; i < m_width; i++, n += 3 )
696 ((int *)out)[i] = int(si[n] * m_chroma.blue[0] + si[n + 1] * m_chroma.green[0] + si[n + 2] * m_chroma.red[0]);
697 }
698 else // how to best convert float to uchar?
699 {
700 unsigned *ui = (unsigned *)in;
701 for( int i = 0, n = 0; i < m_width; i++, n += 3 )
702 ((uchar *)out)[i] = uchar((ui[n] * m_chroma.blue[0] + ui[n + 1] * m_chroma.green[0] + ui[n + 2] * m_chroma.red[0]) * (256.0 / 4294967296.0));

734bool ExrEncoder::write( const Mat& img, const std::vector<int>& params )
735{
736 int width = img.cols, height = img.rows;
737 int depth = img.depth();
738 CV_Assert( depth == CV_32F );
739 int channels = img.channels();
740 bool result = false;
741 Header header( width, height );
742 Imf::PixelType type = FLOAT;
743
744 for( size_t i = 0; i < params.size(); i += 2 )
745 {
746 const int value = params[i+1];
747 if( params[i] == IMWRITE_EXR_TYPE )
748 {
749 switch( value )
750 {
751 case IMWRITE_EXR_TYPE_HALF:
752 type = HALF;
753 break;
754 case IMWRITE_EXR_TYPE_FLOAT:
755 type = FLOAT;
756 break;
757 default:
758 type = FLOAT;
759 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_EXR_TYPE must be one of ImwriteEXRTypeFlags. It is fallbacked to IMWRITE_EXR_TYPE_FLOAT", value));
760 break;
761 }
762 }
763 if ( params[i] == IMWRITE_EXR_COMPRESSION )
764 {
765 switch ( value )
766 {
767 case IMWRITE_EXR_COMPRESSION_NO:
768 header.compression() = NO_COMPRESSION;
769 break;
770 case IMWRITE_EXR_COMPRESSION_RLE:
771 header.compression() = RLE_COMPRESSION;
772 break;
773 case IMWRITE_EXR_COMPRESSION_ZIPS:
774 header.compression() = ZIPS_COMPRESSION;
775 break;
776 case IMWRITE_EXR_COMPRESSION_ZIP:
777 header.compression() = ZIP_COMPRESSION;
778 break;
779 case IMWRITE_EXR_COMPRESSION_PIZ:
780 header.compression() = PIZ_COMPRESSION;
781 break;
782 case IMWRITE_EXR_COMPRESSION_PXR24:
783 header.compression() = PXR24_COMPRESSION;
784 break;
785 case IMWRITE_EXR_COMPRESSION_B44:
786 header.compression() = B44_COMPRESSION;
787 break;
788 case IMWRITE_EXR_COMPRESSION_B44A:
789 header.compression() = B44A_COMPRESSION;
790 break;
791// version macros introduced in openexr 2.0.1.
792// - https://github.com/AcademySoftwareFoundation/openexr/commit/60cdff8a6f5c4e25a374e5f366d6e9b4efd869b3#diff-c4bae0726aebe410e407db9abd406d9cf2684f82dd8a08f46d84e8b7c35cf22aR67
793#if defined(OPENEXR_VERSION_MAJOR) && defined(OPENEXR_VERSION_MINOR) && OPENEXR_VERSION_MAJOR * 1000 + OPENEXR_VERSION_MINOR >= 2 * 1000 + 2
794 // available since version 2.2.0
795 case IMWRITE_EXR_COMPRESSION_DWAA:
796 header.compression() = DWAA_COMPRESSION;
797 break;
798 case IMWRITE_EXR_COMPRESSION_DWAB:
799 header.compression() = DWAB_COMPRESSION;
800 break;
801#endif
802 default:
803 header.compression() = ZIP_COMPRESSION;
804 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_EXR_COMPRESSION must be one of ImwriteEXRCompressionFlags. It is fallbacked to IMWRITE_EXR_COMPRESSION_ZIP", value));
805 break;
806 }
807 }
808 if (params[i] == IMWRITE_EXR_DWA_COMPRESSION_LEVEL)
809 {
810#if !defined(OPENEXR_VERSION_MAJOR)
811 CV_LOG_ONCE_WARNING(NULL, "Setting `IMWRITE_EXR_DWA_COMPRESSION_LEVEL` not supported in unknown OpenEXR version possibly prior to 2.0.1 (version 3 is required)");
812#elif OPENEXR_VERSION_MAJOR < 3
813 CV_LOG_ONCE_WARNING(NULL, "Setting `IMWRITE_EXR_DWA_COMPRESSION_LEVEL` not supported in OpenEXR version " + std::to_string(OPENEXR_VERSION_MAJOR) + " (version 3 is required)");
814#else
815 // See https://github.com/AcademySoftwareFoundation/openexr/blob/v3.3.5/src/lib/OpenEXR/ImfDwaCompressor.cpp#L85
816 header.dwaCompressionLevel() = std::max(params[i + 1], 0);
817 if(value < 0) {
818 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_EXR_DWA_COMPRESSION_LEVEL must be 0 or more. It is fallbacked to 0", value));
819 }
820#endif
821 }
822 }
823
824 if( channels == 3 || channels == 4 )
825 {
826 header.channels().insert( "R", Channel( type ) );
827 header.channels().insert( "G", Channel( type ) );
828 header.channels().insert( "B", Channel( type ) );
829 //printf("bunt\n");
830 }
831 else if( channels == 1 || channels == 2 )
832 {
833 header.channels().insert( "Y", Channel( type ) );
834 //printf("gray\n");
835 }
836 else if( channels > 4 )
837 {
838 for ( int i = 0; i < channels; i++ )
839 header.channels().insert( std::to_string(i), Channel( type ) );
840 //printf("multi-channel\n");
841 }
842
843 if( channels % 2 == 0 && channels <= 4)
844 { // even number of channels indicates Alpha
845 header.channels().insert( "A", Channel( type ) );
846 }
847
848 OutputFile file( m_filename.c_str(), header );
849
850 FrameBuffer frame;
851
852 char *buffer;
853 size_t bufferstep;
854 int size;
855 Mat exrMat;
856 if( type == HALF )
857 {
858 img.convertTo(exrMat, CV_16F);
859 buffer = (char *)const_cast<uchar *>( exrMat.ptr() );
860 bufferstep = exrMat.step;
861 size = 2;
862 }
863 else
864 {
865 buffer = (char *)const_cast<uchar *>( img.ptr() );
866 bufferstep = img.step;
867 size = 4;
868 }
869
870 if( channels == 3 || channels == 4 )
871 {
872 frame.insert( "B", Slice( type, buffer, size * channels, bufferstep ));
873 frame.insert( "G", Slice( type, buffer + size, size * channels, bufferstep ));
874 frame.insert( "R", Slice( type, buffer + size * 2, size * channels, bufferstep ));
875 }
876 else if( channels == 1 || channels == 2 )
877 frame.insert( "Y", Slice( type, buffer, size * channels, bufferstep ));
878 else if( channels > 4 )
879 {
880 for ( int i = 0; i < channels; i++ )
881 frame.insert( std::to_string(i), Slice( type, buffer + size * i, size * channels, bufferstep ));
882 }
883
884 if( channels % 2 == 0 && channels <= 4 )
885 { // even channel count indicates Alpha channel
886 frame.insert( "A", Slice( type, buffer + size * (channels - 1), size * channels, bufferstep ));
887 }
888
889 file.setFrameBuffer( frame );
890
891 result = true;
892 try
893 {
894 file.writePixels( height );
895 }
896 catch(...)
897 {
898 result = false;
899 }
900
901 return result;
902}
