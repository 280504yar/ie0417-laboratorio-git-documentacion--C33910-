56template <typename T> static inline
57void scalarToRawData_(const Scalar& s, T * const buf, const int cn, const int unroll_to)
58{
59 int i = 0;
60 for(; i < cn; i++)
61 buf[i] = saturate_cast<T>(s.val[i]);
62 for(; i < unroll_to; i++)
63 buf[i] = buf[i-cn];
64}

124void convertAndUnrollScalar( const Mat& sc, int buftype, uchar* scbuf, size_t blocksize )
125{
126 int scn = (int)sc.total(), cn = CV_MAT_CN(buftype);
127 size_t esz = CV_ELEM_SIZE(buftype);
128 BinaryFunc cvtFn = getConvertFunc(sc.depth(), buftype);
129 CV_Assert(cvtFn);
130 cvtFn(sc.ptr(), 1, 0, 1, scbuf, 1, Size(std::min(cn, scn), 1), 0);
131 // unroll the scalar
132 if( scn < cn )
133 {
134 CV_Assert( scn == 1 );
135 size_t esz1 = CV_ELEM_SIZE1(buftype);
136 for( size_t i = esz1; i < esz; i++ )
137 scbuf[i] = scbuf[i - esz1];
138 }
139 for( size_t i = esz; i < blocksize*esz; i++ )
140 scbuf[i] = scbuf[i - esz];
141}

144copyMask_(const uchar* _src, size_t sstep, const uchar* mask, size_t mstep, uchar* _dst, size_t dstep, Size size)
145{
146 for( ; size.height--; mask += mstep, _src += sstep, _dst += dstep )
147 {
148 const T* src = (const T*)_src;
149 T* dst = (T*)_dst;
150 int x = 0;
151 #if CV_ENABLE_UNROLLED
152 for( ; x <= size.width - 4; x += 4 )
153 {
154 if( mask[x] )
155 dst[x] = src[x];
156 if( mask[x+1] )
157 dst[x+1] = src[x+1];
158 if( mask[x+2] )
159 dst[x+2] = src[x+2];
160 if( mask[x+3] )
161 dst[x+3] = src[x+3];
162 }
163 #endif
164 for( ; x < size.width; x++ )
165 if( mask[x] )
166 dst[x] = src[x];
167 }

446/* dst = src */
447void Mat::copyTo( OutputArray _dst ) const
448{
449 CV_INSTRUMENT_REGION();
450
451#ifdef HAVE_CUDA
452 if (_dst.isGpuMat())
453 {
454 _dst.getGpuMat().upload(*this);
455 return;
456 }
457#endif
458
459 int stype = type();
460 int dtype = _dst.type();
461 if( _dst.fixedType() && dtype != stype )
462 {
463 CV_Assert( CV_MAT_CN(stype) == CV_MAT_CN(dtype) );
464 convertTo( _dst, dtype );
465 return;
466 }
467
468 if( dims == 0 && empty() )
469 {
470 _dst.release();
471 void* obj = _dst.getObj();
472 if (_dst.isMat())
473 reinterpret_cast<Mat*>(obj)->flags = Mat::MAGIC_VAL | Mat::CONTINUOUS_FLAG | stype;
474 else if (_dst.isUMat())
475 reinterpret_cast<UMat*>(obj)->flags = UMat::MAGIC_VAL | UMat::CONTINUOUS_FLAG | stype;
476 else if (_dst.isGpuMat())
477 reinterpret_cast<cuda::GpuMat*>(obj)->flags = stype;
478 return;
479 }
480
481 bool allowTransposed = dims == 1 ||
482 _dst.kind() == _InputArray::STD_VECTOR ||
483 (_dst.fixedSize() && _dst.dims() == 1);
484 if( _dst.isUMat() )
485 {
486 _dst.create( dims, size.p, stype, -1, allowTransposed );
487 if (empty())
488 return;
489 UMat dst = _dst.getUMat();
490 CV_Assert(dst.u != NULL);
491 size_t i, sz[CV_MAX_DIM] = {1}, dstofs[CV_MAX_DIM] = {0}, esz = CV_ELEM_SIZE(stype);
492 CV_Assert(dims >= 0 && dims < CV_MAX_DIM);
493 for( i = 0; i < (size_t)dims; i++ )
494 sz[i] = size.p[i];
495 int lastdim = dims >= 1 ? dims-1 : 0;
496 sz[lastdim] *= esz;
497 dst.ndoffset(dstofs);
498 dstofs[lastdim] *= esz;
499 dst.u->currAllocator->upload(dst.u, data, std::max(dims, 1), sz, dstofs, dst.step.p, step.p);
500 return;
501 }
502
503 if( dims <= 2 )
504 {
505 _dst.create( dims, size.p, stype, -1, allowTransposed );
506 Mat dst = _dst.getMat();
507 if( data == dst.data )
508 return;
509
510 if( rows > 0 && cols > 0 )
511 {
512 Mat src = *this;
513 Size sz = getContinuousSize2D(src, dst, (int)elemSize());
514 CV_CheckGE(sz.width, 0, "");
515
516 const uchar* sptr = src.data;
517 uchar* dptr = dst.data;
518
519#if IPP_VERSION_X100 >= 201700
520 CV_IPP_RUN_FAST(CV_INSTRUMENT_FUN_IPP(ippiCopy_8u_C1R_L, sptr, (int)src.step, dptr, (int)dst.step, ippiSizeL(sz.width, sz.height)) >= 0)
521#endif
522
523 for (; sz.height--; sptr += src.step, dptr += dst.step)
524 memcpy(dptr, sptr, sz.width);
525 }
526 return;
527 }
528
529 _dst.create( size, stype );
530 Mat dst = _dst.getMat();
531 if( data == dst.data )
532 return;
533
534 if( total() != 0 )
535 {
536 const Mat* arrays[] = { this, &dst };
537 uchar* ptrs[2] = {};
538 NAryMatIterator it(arrays, ptrs, 2);
539 size_t sz = it.size*elemSize();

674static bool can_apply_memset(const Mat &mat, const Scalar &s, int &fill_value)
675{
676 // check if depth is 1 byte.
677 switch (mat.depth())
678 {
679 case CV_8U: fill_value = saturate_cast<uchar>( s.val[0] ); break;
680 case CV_8S: fill_value = saturate_cast<schar>( s.val[0] ); break;
681 default: return false;
682 }
683
684 // check if all element is same.
685 const int64* is = (const int64*)&s.val[0];
686 switch (mat.channels())
687 {
688 case 1: return true;
689 case 2: return (is[0] == is[1]);
690 case 3: return (is[0] == is[1] && is[1] == is[2]);
691 case 4: return (is[0] == is[1] && is[1] == is[2] && is[2] == is[3]);
692 default: return false;
693 }
694}

696Mat& Mat::operator = (const Scalar& s)
697{
698 CV_INSTRUMENT_REGION();
699
700 if (this->empty())
701 return *this;
702
703 const Mat* arrays[] = { this };
704 uchar* dptr;
705 NAryMatIterator it(arrays, &dptr, 1);
706 size_t elsize = it.size*elemSize();
707 const int64* is = (const int64*)&s.val[0];
708
709 if( is[0] == 0 && is[1] == 0 && is[2] == 0 && is[3] == 0 )
710 {
711 for( size_t i = 0; i < it.nplanes; i++, ++it )
712 memset( dptr, 0, elsize );
713 }
714 else
715 {
716 int fill_value = 0;
717 if ( can_apply_memset(*this, s, fill_value) )
718 {
719 for (size_t i = 0; i < it.nplanes; i++, ++it)
720 memset(dptr, fill_value, elsize);
721 return *this;
722 }
723
724 if( it.nplanes > 0 )
725 {
726 double scalar[12];
727 scalarToRawData(s, scalar, type(), 12);
728 size_t blockSize = 12*elemSize1();
729
730 for( size_t j = 0; j < elsize; j += blockSize )
731 {
732 size_t sz = MIN(blockSize, elsize - j);
733 CV_Assert(sz <= sizeof(scalar));
734 memcpy( dptr + j, scalar, sz );
735 }
736 }
737
738 for( size_t i = 1; i < it.nplanes; i++ )
739 {
740 ++it;
741 memcpy( dptr, data, elsize );
742 }
743 }
744 return *this;
745}

807{

845 }

847 memcpy(ptrs[0], scbuf, blockSize);

849 }
850 }

949/*
950 Various border types, image boundaries are denoted with '|'
951
952 * BORDER_REPLICATE: aaaaaa|abcdefgh|hhhhhhh
953 * BORDER_REFLECT: fedcba|abcdefgh|hgfedcb
954 * BORDER_REFLECT_101: gfedcb|abcdefgh|gfedcba
955 * BORDER_WRAP: cdefgh|abcdefgh|abcdefg
956 * BORDER_CONSTANT: iiiiii|abcdefgh|iiiiiii with some specified 'i'
957 */
958int cv::borderInterpolate( int p, int len, int borderType )
959{
960 CV_TRACE_FUNCTION_VERBOSE();
961
962 CV_DbgAssert(len > 0);
963
964#ifdef CV_STATIC_ANALYSIS
965 if(p >= 0 && p < len)
966#else
967 if( (unsigned)p < (unsigned)len )
968#endif
969 ;
970 else if( borderType == BORDER_REPLICATE )
971 p = p < 0 ? 0 : len - 1;
972 else if( borderType == BORDER_REFLECT || borderType == BORDER_REFLECT_101 )
973 {
974 int delta = borderType == BORDER_REFLECT_101;
975 if( len == 1 )
976 return 0;
977 do
978 {
979 if( p < 0 )
980 p = -p - 1 + delta;
981 else
982 p = len - 1 - (p - len) - delta;
983 }
984#ifdef CV_STATIC_ANALYSIS
985 while(p < 0 || p >= len);
986#else
987 while( (unsigned)p >= (unsigned)len );
988#endif
989 }
990 else if( borderType == BORDER_WRAP )
991 {
992 CV_Assert(len > 0);
993 if( p < 0 )
994 p -= ((p-len+1)/len)*len;
995 if( p >= len )
996 p %= len;
997 }
998 else if( borderType == BORDER_CONSTANT )
999 p = -1;
1000 else
1001 CV_Error( cv::Error::StsBadArg, "Unknown/unsupported border type" );
1002 return p;
1003}
