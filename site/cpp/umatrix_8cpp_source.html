157// Do not allow several lock() calls with different UMatData objects.
158struct UMatDataAutoLocker
159{
160 int usage_count;
161 UMatData* locked_objects[2];
162 UMatDataAutoLocker() : usage_count(0) { locked_objects[0] = NULL; locked_objects[1] = NULL; }
163
164 void lock(UMatData*& u1)
165 {
166 bool locked_1 = (u1 == locked_objects[0] || u1 == locked_objects[1]);
167 if (locked_1)
168 {
169 u1 = NULL;
170 return;
171 }
172 CV_Assert(usage_count == 0); // UMatDataAutoLock can't be used multiple times from the same thread
173 usage_count = 1;
174 locked_objects[0] = u1;
175 u1->lock();
176 }
177 void lock(UMatData*& u1, UMatData*& u2)
178 {
179 bool locked_1 = (u1 == locked_objects[0] || u1 == locked_objects[1]);
180 bool locked_2 = (u2 == locked_objects[0] || u2 == locked_objects[1]);
181 if (locked_1)
182 u1 = NULL;
183 if (locked_2)
184 u2 = NULL;
185 if (locked_1 && locked_2)
186 return;
187 CV_Assert(usage_count == 0); // UMatDataAutoLock can't be used multiple times from the same thread
188 usage_count = 1;
189 locked_objects[0] = u1;
190 locked_objects[1] = u2;
191 if (u1)
192 u1->lock();
193 if (u2)
194 u2->lock();
195 }
196 void release(UMatData* u1, UMatData* u2)
197 {
198 if (u1 == NULL && u2 == NULL)
199 return;
200 CV_Assert(usage_count == 1);
201 usage_count = 0;
202 if (u1)
203 u1->unlock();
204 if (u2)
205 u2->unlock();
206 locked_objects[0] = NULL; locked_objects[1] = NULL;
207 }
208};
209static TLSData<UMatDataAutoLocker>& getUMatDataAutoLockerTLS()
210{
211 CV_SINGLETON_LAZY_INIT_REF(TLSData<UMatDataAutoLocker>, new TLSData<UMatDataAutoLocker>());
212}
213static UMatDataAutoLocker& getUMatDataAutoLocker() { return getUMatDataAutoLockerTLS().getRef(); }

471void setSize( UMat& m, int _dims, const int* _sz,
472 const size_t* _steps, bool autoSteps )
473{
474 CV_Assert( 0 <= _dims && _dims <= CV_MAX_DIM && _dims <= MatShape::MAX_DIMS);
475
476 m.dims = _dims;
477 m.size = MatShape(_dims, _sz);
478 m.step[std::max(_dims-1, 0)] = CV_ELEM_SIZE(m.flags);
479 for (int i = _dims-2; i >= 0; i--) {
480 size_t autostep = m.size[i+1]*m.step[i+1];
481 if (_steps) {
482 m.step[i] = _steps[i];
483 //CV_Assert(m.step[i] >= autostep);
484 } else if (autoSteps) {
485 m.step[i] = autostep;
486 } else {
487 m.step[i] = 0;
488 }
489 }
490
491 if( _dims <= 2 )
492 {
493 m.cols = _dims == 0 ? 1 : _sz ? _sz[_dims > 1] : 0;
494 m.rows = _dims < 2 ? 1 : _sz ? _sz[0] : 0;
495 } else {
496 m.cols = m.rows = -1;
497 }
498}

516UMat Mat::getUMat(AccessFlag accessFlags, UMatUsageFlags usageFlags) const
517{
518 UMat hdr;
519 if(!data)
520 return hdr;
521 if (data != datastart)
522 {
523 Size wholeSize;
524 Point ofs;
525 locateROI(wholeSize, ofs);
526 Size sz(cols, rows);
527 if (ofs.x != 0 || ofs.y != 0)
528 {
529 Mat src = *this;
530 int dtop = ofs.y;
531 int dbottom = wholeSize.height - src.rows - ofs.y;
532 int dleft = ofs.x;
533 int dright = wholeSize.width - src.cols - ofs.x;
534 src.adjustROI(dtop, dbottom, dleft, dright);
535 return src.getUMat(accessFlags, usageFlags)(cv::Rect(ofs.x, ofs.y, sz.width, sz.height));
536 }
537 }
538 CV_Assert(data == datastart);
539
540 accessFlags |= ACCESS_RW;
541 UMatData* new_u = NULL;
542 MatStep new_step = step;
543 {
544 MatAllocator *a = allocator, *a0 = getDefaultAllocator();
545 if(!a)
546 a = a0;
547 new_u = a->allocate(dims, size.p, type(), data, new_step.p, accessFlags, usageFlags);
548 new_u->originalUMatData = u;
549 }
550 bool allocated = false;
551 try
552 {
553 allocated = UMat::getStdAllocator()->allocate(new_u, accessFlags, usageFlags);
554 }
555 catch (const cv::Exception& e)
556 {
557 fprintf(stderr, "Exception: %s\n", e.what());
558 }
559 if (!allocated)
560 {
561 allocated = getDefaultAllocator()->allocate(new_u, accessFlags, usageFlags);
562 CV_Assert(allocated);
563 }
564 if (u != NULL)
565 {
566#ifdef HAVE_OPENCL
567 if (ocl::useOpenCL() && new_u->currAllocator == ocl::getOpenCLAllocator())
568 {
569 CV_Assert(new_u->tempUMat());
570 }
571#endif
572 CV_XADD(&(u->refcount), 1);
573 CV_XADD(&(u->urefcount), 1);
574 }
575 try
576 {
577 hdr.flags = flags;
578 hdr.usageFlags = usageFlags;
579 setSize(hdr, dims, size.p, new_step.p);
580 finalizeHdr(hdr);
581 hdr.u = new_u;
582 hdr.offset = 0; //data - datastart;
583 hdr.addref();
584 return hdr;
585 }
586 catch(...)
587 {
588 if (u != NULL)
589 {
590 CV_XADD(&(u->refcount), -1);
591 CV_XADD(&(u->urefcount), -1);
592 }
593 new_u->currAllocator->deallocate(new_u);
594 throw;
595 }

599void UMat::create(int d0, const int* _sizes, int _type, UMatUsageFlags _usageFlags)
600{
601 int sz1 = 1, d = d0;
602 int i;
603 if (d == 0) {
604 d = 1;
605 _sizes = (const int*)&sz1;
606 }
607 CV_Assert(0 <= d && d <= CV_MAX_DIM && _sizes);
608 _type = CV_MAT_TYPE(_type);
609
610 // if param value is USAGE_DEFAULT by implicit default param value -or- explicit value
611 // ...then don't change the existing usageFlags
612 // it is not possible to change usage from non-default to USAGE_DEFAULT through create()
613 // ...instead must construct UMat()
614 if (_usageFlags == cv::USAGE_DEFAULT)
615 {
616 _usageFlags = usageFlags;
617 }
618
619 if( u && d == dims && _type == type() && _usageFlags == usageFlags )
620 {
621 for( i = 0; i < d; i++ )
622 if( size[i] != _sizes[i] )
623 break;
624 if( i == d )
625 return;
626 }
627
628 int _sizes_backup[CV_MAX_DIM]; // #5991
629 if (_sizes == (this->size.p))
630 {
631 for(i = 0; i < d; i++ )
632 _sizes_backup[i] = _sizes[i];
633 _sizes = _sizes_backup;
634 }
635
636 release();
637 usageFlags = _usageFlags;
638 if( d == 0 )
639 return;
640 flags = (_type & CV_MAT_TYPE_MASK) | MAGIC_VAL;
641 setSize(*this, d, _sizes, 0, true);
642 offset = 0;
643
644 if( total() > 0 )
645 {
646 MatAllocator *a = allocator, *a0 = getStdAllocator();
647 if (!a)
648 {
649 a = a0;
650 a0 = Mat::getDefaultAllocator();
651 }
652 try
653 {
654 u = a->allocate(dims, size.p, _type, 0, step.p, ACCESS_RW /* ignored */, usageFlags);
655 CV_Assert(u != 0);
656 }
657 catch(...)
658 {
659 if(a != a0)
660 u = a0->allocate(dims, size.p, _type, 0, step.p, ACCESS_RW /* ignored */, usageFlags);
661 CV_Assert(u != 0);
662 }
663 CV_Assert( step[dims-1] == (size_t)CV_ELEM_SIZE(flags) );
664 }
665
666 finalizeHdr(*this);
667 addref();
668 size.dims = dims = d0;
669}

687void UMat::fit(int _dims, const int* _sizes, int _type, UMatUsageFlags _usageFlags)
688{
689 if (_usageFlags == cv::USAGE_DEFAULT)
690 _usageFlags = usageFlags;
691 size_t oldTotalBytes = u ? u->size : 0;
692 size_t esz = CV_ELEM_SIZE(_type), newTotal = _dims >= 0;
693 for (int i = 0; i < _dims; i++)
694 newTotal *= _sizes[i];
695 size_t newTotalBytes = newTotal*esz;
696 if (newTotalBytes > 0 &&
697 (!isContinuous() ||
698 newTotalBytes > oldTotalBytes ||
699 offset != 0 ||
700 _usageFlags != usageFlags)) {
701 create(_dims, _sizes, _type, _usageFlags);
702 } else {
703 flags = (flags & ~Mat::TYPE_MASK) | CV_MAT_TYPE(_type);
704 int _dummy_size = 0;
705 setSize(*this, (_dims >= 0 ? _dims : 1), (_dims >= 0 ? _sizes : &_dummy_size), nullptr, true);
706 finalizeHdr(*this);
707 }

930void UMat::locateROI( Size& wholeSize, Point& ofs ) const
931{
932 CV_Assert( dims <= 2 && step[0] > 0 );
933 size_t esz = elemSize(), minstep;
934 ptrdiff_t delta1 = (ptrdiff_t)offset, delta2 = (ptrdiff_t)u->size;
935
936 if( delta1 == 0 )
937 ofs.x = ofs.y = 0;
938 else
939 {
940 ofs.y = (int)(delta1/step[0]);
941 ofs.x = (int)((delta1 - step[0]*ofs.y)/esz);
942 CV_DbgAssert( offset == (size_t)(ofs.y*step[0] + ofs.x*esz) );
943 }
944 minstep = (ofs.x + cols)*esz;
945 wholeSize.height = (int)((delta2 - minstep)/step[0] + 1);
946 wholeSize.height = std::max(wholeSize.height, ofs.y + rows);
947 wholeSize.width = (int)((delta2 - step*(wholeSize.height-1))/esz);
948 wholeSize.width = std::max(wholeSize.width, ofs.x + cols);
949}

951
952UMat& UMat::adjustROI( int dtop, int dbottom, int dleft, int dright )
953{
954 CV_Assert( dims == 2 && step[0] > 0 );
955 Size wholeSize; Point ofs;
956 size_t esz = elemSize();
957 locateROI( wholeSize, ofs );
958 int row1 = std::min(std::max(ofs.y - dtop, 0), wholeSize.height), row2 = std::max(0, std::min(ofs.y + rows + dbottom, wholeSize.height));
959 int col1 = std::min(std::max(ofs.x - dleft, 0), wholeSize.width), col2 = std::max(0, std::min(ofs.x + cols + dright, wholeSize.width));
960 if(row1 > row2)
961 std::swap(row1, row2);
962 if(col1 > col2)
963 std::swap(col1, col2);
964
965 offset += (row1 - ofs.y)*step + (col1 - ofs.x)*esz;
966 rows = row2 - row1; cols = col2 - col1;
967 size.p[0] = rows; size.p[1] = cols;
968 updateContinuityFlag();
969 return *this;
970}

973UMat UMat::reshape(int new_cn, int new_rows) const
974{
975 int cn = channels();
976 UMat hdr = *this;
977
978 if( dims > 2 && new_rows == 0 && new_cn != 0 && size[dims-1]*cn % new_cn == 0 )
979 {
980 hdr.flags = (hdr.flags & ~CV_MAT_CN_MASK) | ((new_cn-1) << CV_CN_SHIFT);
981 hdr.step[dims-1] = CV_ELEM_SIZE(hdr.flags);
982 hdr.size[dims-1] = hdr.size[dims-1]*cn / new_cn;
983 return hdr;
984 }
985
986 CV_Assert( dims <= 2 );
987
988 if( new_cn == 0 )
989 new_cn = cn;
990
991 int total_width = cols * cn;
992
993 if( (new_cn > total_width || total_width % new_cn != 0) && new_rows == 0 )
994 new_rows = rows * total_width / new_cn;
995
996 if( new_rows != 0 && new_rows != rows )
997 {
998 int total_size = total_width * rows;
999 if( !isContinuous() )
1000 CV_Error( cv::Error::BadStep,
1001 "The matrix is not continuous, thus its number of rows can not be changed" );
1002
1003 total_width = total_size / new_rows;
1004
1005 if( total_width * new_rows != total_size )
1006 CV_Error( cv::Error::StsBadArg, "The total number of matrix elements "
1007 "is not divisible by the new number of rows" );
1008
1009 hdr.size[0] = hdr.rows = new_rows;
1010 hdr.step[0] = total_width * elemSize1();
1011 } else {
1012 hdr.size[0] = hdr.rows = rows;
1013 if (dims <= 1)
1014 hdr.step[0] = cols * CV_ELEM_SIZE(flags);
1015 }
1016
1017 int new_width = total_width / new_cn;
1018
1019 if( new_width * new_cn != total_width )
1020 CV_Error( cv::Error::BadNumChannels,
1021 "The total width is not divisible by the new number of channels" );
1022
1023 hdr.size.dims = hdr.dims = 2;
1024 hdr.size[1] = hdr.cols = new_width;
1025 hdr.flags = (hdr.flags & ~CV_MAT_CN_MASK) | ((new_cn-1) << CV_CN_SHIFT);
1026 hdr.step[1] = CV_ELEM_SIZE(hdr.flags);
1027 return hdr;
1028}

1098UMat UMat::reshape(int _cn, int _newndims, const int* _newsz) const
1099{
1100 if(_newndims == dims)
1101 {
1102 if(_newsz == 0)
1103 return reshape(_cn);
1104 if(_newndims == 2)
1105 return reshape(_cn, _newsz[0]);
1106 }
1107
1108 if (isContinuous())
1109 {
1110 CV_Assert(_cn >= 0 && _newndims >= 0 && _newndims <= CV_MAX_DIM && (_newndims == 0 || _newsz != 0));
1111
1112 if (_cn == 0)
1113 _cn = this->channels();
1114 else
1115 CV_Assert(_cn <= CV_CN_MAX);
1116
1117 size_t total_elem1_ref = this->total() * this->channels();
1118 size_t total_elem1 = _cn;
1119
1120 AutoBuffer<int, 4> newsz_buf( (size_t)std::max(_newndims, 1) );
1121
1122 int m1_idx = -1;
1123
1124 for (int i = 0; i < _newndims; i++)
1125 {
1126 if (_newsz[i] >= 0) {
1127 if (_newsz[i] == 0 && i < dims)
1128 newsz_buf[i] = size.p[i];
1129 else
1130 newsz_buf[i] = _newsz[i];
1131 total_elem1 *= (size_t)newsz_buf[i];
1132 } else {
1133 if (m1_idx >= 0)
1134 CV_Error(cv::Error::StsBadSize, "More than one '-1' occured in the new shape");
1135 m1_idx = i;
1136 }
1137 }
1138
1139 if (m1_idx >= 0) {
1140 if (total_elem1 == 0) {
1141 CV_Assert(total_elem1_ref == 0);
1142 total_elem1 = 1;
1143 }
1144 CV_Assert(total_elem1_ref % total_elem1 == 0);
1145 newsz_buf[m1_idx] = (int)(total_elem1_ref / total_elem1);
1146 } else if (total_elem1 != total_elem1_ref) {
1147 CV_Error(cv::Error::StsUnmatchedSizes, "Requested and source matrices have different count of elements");
1148 }
1149
1150 UMat hdr = *this;
1151 hdr.flags = (hdr.flags & ~CV_MAT_CN_MASK) | ((_cn-1) << CV_CN_SHIFT);
1152 setSize(hdr, _newndims, newsz_buf.data(), NULL, true);
1153
1154 return hdr;
1155 }
1156
1157 CV_Error(cv::Error::StsNotImplemented, "Reshaping of n-dimensional non-continuous matrices is not supported yet");
1158}

1231void UMat::copyTo(OutputArray _dst) const
1232{
1233 CV_INSTRUMENT_REGION();
1234
1235#ifdef HAVE_CUDA
1236 if (_dst.isGpuMat())
1237 {
1238 _dst.getGpuMat().upload(*this);
1239 return;
1240 }
1241#endif
1242
1243 int stype = type();
1244 int dtype = _dst.type();
1245 if( _dst.fixedType() && dtype != stype )
1246 {
1247 CV_Assert( CV_MAT_CN(stype) == CV_MAT_CN(dtype) );
1248 convertTo( _dst, dtype );
1249 return;
1250 }
1251
1252 if( dims == 0 && empty() )
1253 {
1254 _dst.release();
1255 void* obj = _dst.getObj();
1256 if (_dst.isMat())
1257 reinterpret_cast<Mat*>(obj)->flags = Mat::MAGIC_VAL | Mat::CONTINUOUS_FLAG | stype;
1258 else if (_dst.isUMat())
1259 reinterpret_cast<UMat*>(obj)->flags = UMat::MAGIC_VAL | UMat::CONTINUOUS_FLAG | stype;
1260 else if (_dst.isGpuMat())
1261 reinterpret_cast<cuda::GpuMat*>(obj)->flags = stype;
1262 return;
1263 }
1264
1265 _dst.create( dims, size.p, stype );
1266 if (empty())
1267 return;
1268
1269 size_t sz[CV_MAX_DIM] = {1}, srcofs[CV_MAX_DIM]={0}, dstofs[CV_MAX_DIM]={0};
1270 size_t esz = CV_ELEM_SIZE(stype);
1271 int i, d = std::max(dims, 1);
1272 for( i = 0; i < d; i++ )
1273 sz[i] = size.p[i];
1274 sz[d-1] *= esz;
1275 ndoffset(srcofs);
1276 srcofs[d-1] *= esz;
1277
1278 if( _dst.isUMat() )
1279 {
1280 UMat dst = _dst.getUMat();
1281 CV_Assert(dst.u);
1282 if( u == dst.u && dst.offset == offset )
1283 return;
1284
1285 if (u->currAllocator == dst.u->currAllocator)
1286 {
1287 dst.ndoffset(dstofs);
1288 dstofs[dims-1] *= esz;
1289 u->currAllocator->copy(u, dst.u, d, sz, srcofs, step.p, dstofs, dst.step.p, false);
1290 return;
1291 }
1292 }
1293
1294 Mat dst = _dst.getMat();
1295 u->currAllocator->download(u, dst.ptr(), d, sz, srcofs, step.p, dst.step.p);
1296}
