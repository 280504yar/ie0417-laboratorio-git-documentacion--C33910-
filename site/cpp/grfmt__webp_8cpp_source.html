86bool WebPDecoder::checkSignature(const String & signature) const
87{
88 bool ret = false;
89

91 {
92 WebPBitstreamFeatures features;
93 if(VP8_STATUS_OK == WebPGetFeatures((uint8_t *)signature.c_str(),

96 ret = true;
97 }
98 }
99
100 return ret;

108bool WebPDecoder::readHeader()
109{
110 if (m_has_animation)
111 {
112 return true;
113 }
114
115 uint8_t header[WEBP_HEADER_SIZE] = { 0 };
116 if (m_buf.empty())
117 {
118 fs.open(m_filename.c_str(), std::ios::binary);
119 fs.seekg(0, std::ios::end);
120 fs_size = safeCastToSizeT(fs.tellg(), "File is too large");
121 fs.seekg(0, std::ios::beg);
122 CV_Assert(fs && "File stream error");
123 CV_CheckGE(fs_size, WEBP_HEADER_SIZE, "File is too small");
124 CV_CheckLE(fs_size, param_maxFileSize, "File is too large. Increase OPENCV_IMGCODECS_WEBP_MAX_FILE_SIZE parameter if you want to process large files");
125
126 fs.read((char*)header, sizeof(header));
127 CV_Assert(fs && "Can't read WEBP_HEADER_SIZE bytes");
128 }
129 else
130 {
131 CV_CheckGE(m_buf.total(), WEBP_HEADER_SIZE, "Buffer is too small");
132 memcpy(header, m_buf.ptr(), sizeof(header));
133 data = m_buf;
134 }
135
136 WebPBitstreamFeatures features;
137 if (VP8_STATUS_OK < WebPGetFeatures(header, sizeof(header), &features)) return false;
138
139 m_has_animation = features.has_animation == 1;
140
141 if (m_has_animation)
142 {
143 if (m_buf.empty())
144 {
145 fs.seekg(0, std::ios::beg); CV_Assert(fs && "File stream error");
146 data.create(1, validateToInt(fs_size), CV_8UC1);
147 fs.read((char*)data.ptr(), fs_size);
148 CV_Assert(fs && "Can't read file data");
149 fs.close();
150 }
151
152 CV_Assert(data.type() == CV_8UC1); CV_Assert(data.rows == 1);
153
154 WebPData webp_data;
155 webp_data.bytes = (const uint8_t*)data.ptr();
156 webp_data.size = data.total();
157
158 WebPAnimDecoderOptions dec_options;
159 if (!WebPAnimDecoderOptionsInit(&dec_options))
160 CV_Error(Error::StsInternal, "Failed to initialize animated WebP decoding options");
161
162 dec_options.color_mode = m_use_rgb ? MODE_RGBA : MODE_BGRA;
163 anim_decoder.reset(WebPAnimDecoderNew(&webp_data, &dec_options));
164 CV_Assert(anim_decoder.get() && "Error parsing image");
165
166 WebPAnimInfo anim_info;
167 if (!WebPAnimDecoderGetInfo(anim_decoder.get(), &anim_info))
168 CV_Error(Error::StsInternal, "Failed to get animated WebP information");
169 m_animation.loop_count = anim_info.loop_count;
170
171 m_animation.bgcolor[0] = (anim_info.bgcolor >> 24) & 0xFF;
172 m_animation.bgcolor[1] = (anim_info.bgcolor >> 16) & 0xFF;
173 m_animation.bgcolor[2] = (anim_info.bgcolor >> 8) & 0xFF;
174 m_animation.bgcolor[3] = anim_info.bgcolor & 0xFF;
175 m_frame_count = anim_info.frame_count;
176 }
177 m_width = features.width;
178 m_height = features.height;
179 m_type = features.has_alpha ? CV_8UC4 : CV_8UC3;
180
181 return true;
182}

184bool WebPDecoder::readData(Mat &img)
185{
186 CV_CheckGE(m_width, 0, ""); CV_CheckGE(m_height, 0, "");
187
188 CV_CheckEQ(img.cols, m_width, "");
189 CV_CheckEQ(img.rows, m_height, "");
190
191 if (data.empty())
192 {
193 fs.seekg(0, std::ios::beg); CV_Assert(fs && "File stream error");
194 data.create(1, validateToInt(fs_size), CV_8UC1);
195 fs.read((char*)data.ptr(), fs_size);
196 CV_Assert(fs && "Can't read file data");
197 fs.close();
198 }
199 CV_Assert(data.type() == CV_8UC1); CV_Assert(data.rows == 1);
200
201 if (m_read_options) {
202 WebPData webp_data;
203 webp_data.bytes = (const uint8_t*)data.ptr();
204 webp_data.size = data.total();
205
206 std::vector<uchar> metadata;
207 WebPDemuxer* demux = WebPDemux(&webp_data);
208
209 if (demux)
210 {
211 WebPChunkIterator chunk_iter;
212
213 if (WebPDemuxGetChunk(demux, "EXIF", 1, &chunk_iter))
214 {
215 metadata = std::vector<uchar>(chunk_iter.chunk.bytes,
216 chunk_iter.chunk.bytes + chunk_iter.chunk.size);
217 WebPDemuxReleaseChunkIterator(&chunk_iter);
218 m_exif.parseExif(metadata.data(), metadata.size());
219 }
220
221 if (WebPDemuxGetChunk(demux, "ICCP", 1, &chunk_iter))
222 {
223 metadata = std::vector<uchar>(chunk_iter.chunk.bytes,
224 chunk_iter.chunk.bytes + chunk_iter.chunk.size);
225 WebPDemuxReleaseChunkIterator(&chunk_iter);
226 m_metadata[IMAGE_METADATA_ICCP] = metadata;
227 }
228
229 if (WebPDemuxGetChunk(demux, "XMP ", 1, &chunk_iter)) // note the space in "XMP "
230 {
231 metadata = std::vector<uchar>(chunk_iter.chunk.bytes,
232 chunk_iter.chunk.bytes + chunk_iter.chunk.size);
233 WebPDemuxReleaseChunkIterator(&chunk_iter);
234 m_metadata[IMAGE_METADATA_XMP] = metadata;
235 }
236
237 WebPDemuxDelete(demux);
238 m_read_options = 0;
239 }
240 }
241
242 Mat read_img;
243 CV_CheckType(img.type(), img.type() == CV_8UC1 || img.type() == CV_8UC3 || img.type() == CV_8UC4, "");
244 if (img.type() != m_type || img.cols != m_width || img.rows != m_height)
245 {
246 read_img.create(m_height, m_width, m_type);
247 }
248 else
249 {
250 read_img = img; // copy header
251 }
252
253 uchar* out_data = read_img.ptr();
254 size_t out_data_size = read_img.dataend - out_data;
255
256 uchar* res_ptr = NULL;
257
258 if (m_has_animation)
259 {
260 uint8_t* buf;
261 int timestamp;
262
263 if (!WebPAnimDecoderGetNext(anim_decoder.get(), &buf, &timestamp))
264 CV_Error(Error::StsInternal, "Failed to decode animated WebP frame");
265 Mat tmp(Size(m_width, m_height), CV_8UC4, buf);
266
267 if (img.type() == CV_8UC1)
268 {
269 cvtColor(tmp, img, COLOR_BGR2GRAY);
270 }
271 else
272 if (img.type() == CV_8UC3)
273 {
274 cvtColor(tmp, img, COLOR_BGRA2BGR);
275 }
276 else
277 tmp.copyTo(img);
278
279 m_animation.durations.push_back(timestamp - m_previous_timestamp);
280 m_previous_timestamp = timestamp;
281 return true;
282 }
283
284 if (m_type == CV_8UC3)
285 {
286 CV_CheckTypeEQ(read_img.type(), CV_8UC3, "");
287 if (m_use_rgb)
288 res_ptr = WebPDecodeRGBInto(data.ptr(), data.total(), out_data,
289 (int)out_data_size, (int)read_img.step);
290 else
291 res_ptr = WebPDecodeBGRInto(data.ptr(), data.total(), out_data,
292 (int)out_data_size, (int)read_img.step);
293 }
294 else if (m_type == CV_8UC4)
295 {
296 CV_CheckTypeEQ(read_img.type(), CV_8UC4, "");
297 if (m_use_rgb)
298 res_ptr = WebPDecodeRGBAInto(data.ptr(), data.total(), out_data,
299 (int)out_data_size, (int)read_img.step);
300 else
301 res_ptr = WebPDecodeBGRAInto(data.ptr(), data.total(), out_data,
302 (int)out_data_size, (int)read_img.step);
303 }
304
305 if (res_ptr != out_data)
306 return false;
307
308 if (read_img.data == img.data && img.type() == m_type)
309 {
310 // nothing
311 }
312 else if (img.type() == CV_8UC1)
313 {
314 cvtColor(read_img, img, COLOR_BGR2GRAY);
315 }
316 else if (img.type() == CV_8UC3 && m_type == CV_8UC4)
317 {
318 cvtColor(read_img, img, COLOR_BGRA2BGR);
319 }
320 else
321 {
322 CV_Error(Error::StsInternal, "");
323 }
324 return true;
325}

351// Simple API style
352static size_t cvEncodeLosslessExactBGRA(const uint8_t* rgba, int width, int height, int stride, uint8_t** output)
353{
354 WebPConfig config;
355 WebPPicture pic;
356 WebPMemoryWriter wrt;
357
358 // 6 is the default value for speed/compression balance in lossless mode.
359 // It doesn't affect visual quality, only file size and encoding time.
360 if (!WebPConfigInit(&config) || !WebPConfigLosslessPreset(&config, 6))
361 {
362 return 0;
363 }
364 config.exact = 1;
365
366 if (!WebPPictureInit(&pic))
367 {
368 return 0;
369 }
370 pic.width = width;
371 pic.height = height;
372 pic.use_argb = 1; // BGRA
373 if (!WebPPictureImportBGRA(&pic, rgba, stride))
374 {
375 return 0;
376 }
377
378 WebPMemoryWriterInit(&wrt);
379 pic.writer = WebPMemoryWrite;
380 pic.custom_ptr = &wrt;
381
382 if (!WebPEncode(&config, &pic))
383 {
384 WebPMemoryWriterClear(&wrt);
385 WebPPictureFree(&pic);
386 return 0;
387 }
388
389 *output = wrt.mem;
390 size_t size = wrt.size;
391 WebPPictureFree(&pic);
392 return size;
393}

395bool WebPEncoder::write(const Mat& img, const std::vector<int>& params)
396{
397 CV_CheckDepthEQ(img.depth(), CV_8U, "WebP codec supports 8U images only");
398
399 const int width = img.cols, height = img.rows;
400
401 int lossless_mode = -1; // not specified
402 float quality = 0.0f; // not specified
403
404 for(size_t i = 0; i < params.size(); i += 2)
405 {
406 const int value = params[i+1];
407 if (params[i] == IMWRITE_WEBP_LOSSLESS_MODE)
408 {
409 switch(value)
410 {
411 case IMWRITE_WEBP_LOSSLESS_ON:
412 case IMWRITE_WEBP_LOSSLESS_OFF:
413 case IMWRITE_WEBP_LOSSLESS_PRESERVE_COLOR:
414 lossless_mode = value;
415 break;
416 default:
417 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_WEBP_LOSSLESS_MODE must be one of ImwriteWEBPLosslessMode. It is ignored", value));
418 break;
419 }
420 }
421 if (params[i] == IMWRITE_WEBP_QUALITY)
422 {
423 if (value < 1)
424 {
425 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_WEBP_QUALITY must be between 1 to 100(lossy) or more(lossless). It is fallbacked to 1", value));
426 quality = 1.0f;
427 }
428 else if (value > 100)
429 {
430 quality = 101.0f;
431 }
432 else // value is 1 to 100
433 {
434 quality = static_cast<float>(value);
435 }
436 }
437 }
438
439 switch(lossless_mode)
440 {
441 case -1: // not specified by user
442 case IMWRITE_WEBP_LOSSLESS_OFF:
443 // Fallback to lossless if quality is not specified (-1.0f) or out of lossy range (>100.0f).
444 // This maintains backward compatibility where WebP defaults to lossless.
445 if ((quality < 1.0f) || (quality > 100.0f))
446 {
447 lossless_mode = IMWRITE_WEBP_LOSSLESS_ON;
448 quality = 101.0f;
449 }
450 else
451 {
452 lossless_mode = IMWRITE_WEBP_LOSSLESS_OFF;
453 // Use specified quality for lossy compression.
454 }
455 break;
456
457 case IMWRITE_WEBP_LOSSLESS_ON:
458 case IMWRITE_WEBP_LOSSLESS_PRESERVE_COLOR:
459 // Force quality value to lossless range when explicit lossless mode is selected.
460 quality = 101.0f;
461 break;
462
463 default:
464 CV_Error(Error::StsError, cv::format("Unexpected lossless_mode(%d)", lossless_mode));
465 break;
466 }
467
468 int channels = img.channels();
469 CV_Check(channels, channels == 1 || channels == 3 || channels == 4, "");
470
471 const Mat *image = &img;
472 Mat temp;
473
474 if (channels == 1)
475 {
476 cvtColor(*image, temp, COLOR_GRAY2BGR);
477 image = &temp;
478 channels = 3;
479 }
480
481 uint8_t *encoder_out = NULL;
482 size_t size = 0;
483 if (channels == 3)
484 {
485 switch(lossless_mode)
486 {
487 case IMWRITE_WEBP_LOSSLESS_OFF:
488 size = WebPEncodeBGR(image->ptr(), width, height, (int)image->step, quality, &encoder_out);
489 break;
490 case IMWRITE_WEBP_LOSSLESS_ON:
491 case IMWRITE_WEBP_LOSSLESS_PRESERVE_COLOR:
492 size = WebPEncodeLosslessBGR(image->ptr(), width, height, (int)image->step, &encoder_out);
493 break;
494 default:
495 CV_Error(Error::StsError, cv::format("Unexcepted lossless_mode(%d)", lossless_mode));
496 break;
497 }
498 }
499 else
500 {
501 CV_CheckEQ(channels, 4, "Unexpected channels is used");
502 switch(lossless_mode)
503 {
504 case IMWRITE_WEBP_LOSSLESS_OFF:
505 size = WebPEncodeBGRA(image->ptr(), width, height, (int)image->step, quality, &encoder_out);
506 break;
507 case IMWRITE_WEBP_LOSSLESS_ON:
508 size = WebPEncodeLosslessBGRA(image->ptr(), width, height, (int)image->step, &encoder_out);
509 break;
510 case IMWRITE_WEBP_LOSSLESS_PRESERVE_COLOR:
511 size = cvEncodeLosslessExactBGRA(image->ptr(), width, height, (int)image->step, &encoder_out);
512 break;
513 }
514 }
515
516 if (size == 0)
517 {
518 CV_LOG_ERROR(NULL, cv::format("WebP encoding failed with lossless_mode=%d", lossless_mode));
519 return false;
520 }
521
522#if WEBP_DECODER_ABI_VERSION >= 0x0206
523 Ptr<uint8_t> out_cleaner(encoder_out, WebPFree);
524#else
525 Ptr<uint8_t> out_cleaner(encoder_out, free);
526#endif
527
528 uint8_t *out = encoder_out;
529 uint8_t *muxer_out = nullptr;
530
531 if (!m_metadata.empty())
532 {
533 WebPData muxerData;
534
535 WebPMux* mux = WebPMuxNew();
536 WebPData imageData = { out, size };
537 WebPMuxSetImage(mux, &imageData, 0);
538
539 WebPData metadata;
540 if (m_metadata[IMAGE_METADATA_EXIF].size() > 0)
541 {
542 metadata.bytes = m_metadata[IMAGE_METADATA_EXIF].data();
543 metadata.size = m_metadata[IMAGE_METADATA_EXIF].size();
544 WebPMuxSetChunk(mux, "EXIF", &metadata, 1);
545 }
546 if (m_metadata[IMAGE_METADATA_XMP].size() > 0)
547 {
548 metadata.bytes = m_metadata[IMAGE_METADATA_XMP].data();
549 metadata.size = m_metadata[IMAGE_METADATA_XMP].size();
550 WebPMuxSetChunk(mux, "XMP ", &metadata, 1);
551 }
552
553 if (m_metadata[IMAGE_METADATA_ICCP].size() > 0)
554 {
555 metadata.bytes = m_metadata[IMAGE_METADATA_ICCP].data();
556 metadata.size = m_metadata[IMAGE_METADATA_ICCP].size();
557 WebPMuxSetChunk(mux, "ICCP", &metadata, 1);
558 }
559
560 if (WebPMuxAssemble(mux, &muxerData) == WEBP_MUX_OK) {
561 size = muxerData.size;
562 muxer_out = const_cast<uint8_t*>(muxerData.bytes);
563 out = muxer_out;
564 WebPMuxDelete(mux);
565 }
566 else {
567 WebPMuxDelete(mux);
568 CV_Error(Error::StsError, "Failed to assemble WebP with EXIF");
569 }
570 }
571
572#if WEBP_DECODER_ABI_VERSION >= 0x0206
573 Ptr<const uint8_t> muxer_cleaner(muxer_out, WebPFree);
574#else
575 Ptr<const uint8_t> muxer_cleaner(muxer_out, free);
576#endif
577
578 CV_Assert(size > 0);
579 size_t bytes_written = 0;
580 if (m_buf)
581 {
582 m_buf->resize(size);
583 memcpy(&(*m_buf)[0], out, size);
584 bytes_written = size;
585 }
586 else
587 {
588 FILE *fd = fopen(m_filename.c_str(), "wb");
589 if (fd != NULL)
590 {
591 bytes_written = fwrite(out, sizeof(uint8_t), size, fd);
592 if (size != bytes_written)
593 {
594 CV_LOG_ERROR(NULL, cv::format("Only %zu or %zu bytes are written\n",bytes_written, size));
595 }
596 fclose(fd); fd = NULL;
597 }
598 }
599
600 return (size > 0) && (bytes_written == size);
601}

603bool WebPEncoder::writeanimation(const Animation& animation, const std::vector<int>& params)
604{
605 CV_CheckDepthEQ(animation.frames[0].depth(), CV_8U, "WebP codec supports only 8-bit unsigned images");
606 int ok = 0;
607 int timestamp = 0;
608 const int width = animation.frames[0].cols, height = animation.frames[0].rows;
609
610 WebPAnimEncoderOptions anim_config;
611 WebPConfig config;
612 WebPPicture pic;
613 WebPData webp_data;
614
615 WebPDataInit(&webp_data);
616 if (!WebPAnimEncoderOptionsInit(&anim_config) ||
617 !WebPConfigInit(&config) ||
618 !WebPPictureInit(&pic)) {
619 CV_LOG_ERROR(NULL, "Library version mismatch!\n");
620 WebPDataClear(&webp_data);
621 return false;
622 }
623
624 int bgvalue = (static_cast<int>(animation.bgcolor[0]) & 0xFF) << 24 |
625 (static_cast<int>(animation.bgcolor[1]) & 0xFF) << 16 |
626 (static_cast<int>(animation.bgcolor[2]) & 0xFF) << 8 |
627 (static_cast<int>(animation.bgcolor[3]) & 0xFF);
628
629 anim_config.anim_params.bgcolor = bgvalue;
630 anim_config.anim_params.loop_count = animation.loop_count;
631 anim_config.minimize_size = 0;
632
633 for(size_t i = 0; i < params.size(); i += 2)
634 {
635 const int value = params[i+1];
636 if (params[i] == IMWRITE_WEBP_QUALITY)
637 {
638 config.lossless = 0; // false
639 config.quality = static_cast<float>(value);
640 if (config.quality < 1.0f)
641 {
642 config.quality = 1.0f;
643 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_WEBP_QUALITY must be between 1 to 100(lossy) or more(lossless). It is fallbacked to 1", value));
644 }
645 if (config.quality > 100.0f)
646 {
647 config.quality = 100.0f;
648 config.lossless = 1; // true
649 }
650 }
651 }
652
653 std::unique_ptr<WebPAnimEncoder, void (*)(WebPAnimEncoder*)> anim_encoder(
654 WebPAnimEncoderNew(width, height, &anim_config), WebPAnimEncoderDelete);
655
656 pic.width = width;
657 pic.height = height;
658 pic.use_argb = 1;
659 pic.argb_stride = width;
660
661 bool is_input_rgba = animation.frames[0].channels() == 4;
662 Size canvas_size = Size(animation.frames[0].cols,animation.frames[0].rows);
663
664 for (size_t i = 0; i < animation.frames.size(); i++)
665 {
666 Mat argb;
667 CV_Assert(canvas_size == Size(animation.frames[i].cols,animation.frames[i].rows));
668 if (is_input_rgba)
669 pic.argb = (uint32_t*)animation.frames[i].data;
670 else
671 {
672 cvtColor(animation.frames[i], argb, COLOR_BGR2BGRA);
673 pic.argb = (uint32_t*)argb.data;
674 }
675 ok = WebPAnimEncoderAdd(anim_encoder.get(), &pic, timestamp, &config);
676 timestamp += animation.durations[i];
677 }
678
679 // add a last fake frame to signal the last duration
680 ok = ok & WebPAnimEncoderAdd(anim_encoder.get(), NULL, timestamp, NULL);
681 ok = ok & WebPAnimEncoderAssemble(anim_encoder.get(), &webp_data);
682
683 size_t bytes_written = 0;
684 if (ok)
685 {
686 if (m_buf)
687 {
688 m_buf->resize(webp_data.size);
