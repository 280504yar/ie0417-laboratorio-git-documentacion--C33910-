19void rescaleDepth(InputArray in_in, int type, OutputArray out_out, double depth_factor)
20{
21 cv::Mat in = in_in.getMat();
22 CV_Assert(in.type() == CV_64FC1 || in.type() == CV_32FC1 || in.type() == CV_16UC1 || in.type() == CV_16SC1);
23 CV_Assert(type == CV_64FC1 || type == CV_32FC1);
24
25 int in_depth = in.depth();
26
27 out_out.create(in.size(), type);
28 cv::Mat out = out_out.getMat();
29 if (in_depth == CV_16U)
30 {
31 in.convertTo(out, type, 1 / depth_factor); //convert to float so that it is in meters
32 cv::Mat valid_mask = in == std::numeric_limits<ushort>::min(); // Should we do std::numeric_limits<ushort>::max() too ?
33 out.setTo(std::numeric_limits<float>::quiet_NaN(), valid_mask); //set a$
34 }
35 if (in_depth == CV_16S)
36 {
37 in.convertTo(out, type, 1 / depth_factor); //convert to float so tha$
38 cv::Mat valid_mask = (in == std::numeric_limits<short>::min()) | (in == std::numeric_limits<short>::max()); // Should we do std::numeric_limits<ushort>::max() too ?
39 out.setTo(std::numeric_limits<float>::quiet_NaN(), valid_mask); //set a$
40 }
41 if ((in_depth == CV_32F) || (in_depth == CV_64F))
42 in.convertTo(out, type);
43}

52static void depthTo3d_from_uvz(const cv::Mat& in_K, const cv::Mat& u_mat, const cv::Mat& v_mat, const cv::Mat& z_mat,
53 cv::Mat& points3d)
54{
55 CV_Assert((u_mat.size() == z_mat.size()) && (v_mat.size() == z_mat.size()));
56 if (u_mat.empty())
57 return;
58 CV_Assert((u_mat.type() == z_mat.type()) && (v_mat.type() == z_mat.type()));
59
60 //grab camera params
61 cv::Mat_<float> K;
62
63 if (in_K.depth() == CV_32F)
64 K = in_K;
65 else
66 in_K.convertTo(K, CV_32F);
67
68 float fx = K(0, 0);
69 float fy = K(1, 1);
70 float s = K(0, 1);
71 float cx = K(0, 2);
72 float cy = K(1, 2);
73
74 std::vector<cv::Mat> coordinates(4);
75
76 coordinates[0] = (u_mat - cx) / fx;
77
78 if (s != 0)
79 coordinates[0] = coordinates[0] + (-(s / fy) * v_mat + cy * s / fy) / fx;
80
81 coordinates[0] = coordinates[0].mul(z_mat);
82 coordinates[1] = (v_mat - cy).mul(z_mat) * (1. / fy);
83 coordinates[2] = z_mat;
84 coordinates[3] = Mat(u_mat.size(), CV_32F, Scalar(0));
85 cv::merge(coordinates, points3d);
86}

94static void depthTo3dMask(const cv::Mat& depth, const cv::Mat& K, const cv::Mat& mask, cv::Mat& points3d)
95{
96 // Create 3D points in one go.
97 cv::Mat_<float> u_mat, v_mat, z_mat;
98
99 cv::Mat_<uchar> uchar_mask = mask;
100
101 if (mask.depth() != (CV_8U))
102 mask.convertTo(uchar_mask, CV_8U);
103
104 // Figure out the interesting indices
105 size_t n_points;
106
107 if (depth.depth() == CV_16U)
108 n_points = convertDepthToFloat<ushort>(depth, mask, 1.0f / 1000.0f, u_mat, v_mat, z_mat);
109 else if (depth.depth() == CV_16S)
110 n_points = convertDepthToFloat<short>(depth, mask, 1.0f / 1000.0f, u_mat, v_mat, z_mat);
111 else
112 {
113 CV_Assert(depth.type() == CV_32F);
114 n_points = convertDepthToFloat<float>(depth, mask, 1.0f, u_mat, v_mat, z_mat);
115 }
116
117 if (n_points == 0)
118 return;
119
120 u_mat.resize(n_points);
121 v_mat.resize(n_points);
122 z_mat.resize(n_points);
123
124 depthTo3d_from_uvz(K, u_mat, v_mat, z_mat, points3d);
125 points3d = points3d.reshape(4, 1);
126}

133template<typename T>
134void depthTo3dNoMask(const cv::Mat& in_depth, const cv::Mat_<T>& K, cv::Mat& points3d)
135{
136 const T inv_fx = T(1) / K(0, 0);
137 const T inv_fy = T(1) / K(1, 1);
138 const T ox = K(0, 2);
139 const T oy = K(1, 2);
140
141 // Build z
142 cv::Mat_<T> z_mat;
143 if (z_mat.depth() == in_depth.depth())
144 z_mat = in_depth;
145 else
146 rescaleDepthTemplated<T>(in_depth, z_mat);
147
148 // Pre-compute some constants
149 cv::Mat_<T> x_cache(1, in_depth.cols), y_cache(in_depth.rows, 1);
150 T* x_cache_ptr = x_cache[0], * y_cache_ptr = y_cache[0];
151 for (int x = 0; x < in_depth.cols; ++x, ++x_cache_ptr)
152 *x_cache_ptr = (x - ox) * inv_fx;
153 for (int y = 0; y < in_depth.rows; ++y, ++y_cache_ptr)
154 *y_cache_ptr = (y - oy) * inv_fy;
155
156 y_cache_ptr = y_cache[0];
157 for (int y = 0; y < in_depth.rows; ++y, ++y_cache_ptr)
158 {
159 cv::Vec<T, 4>* point = points3d.ptr<cv::Vec<T, 4> >(y);
160 const T* x_cache_ptr_end = x_cache[0] + in_depth.cols;
161 const T* depth = z_mat[y];
162 for (x_cache_ptr = x_cache[0]; x_cache_ptr != x_cache_ptr_end; ++x_cache_ptr, ++point, ++depth)
163 {
164 T z = *depth;
165 (*point)[0] = (*x_cache_ptr) * z;
166 (*point)[1] = (*y_cache_ptr) * z;
167 (*point)[2] = z;
168 (*point)[3] = 0;
169 }
170 }
171}

182void depthTo3dSparse(InputArray depth_in, InputArray K_in, InputArray points_in, OutputArray points3d_out)
183{
184 // Make sure we use foat types
185 cv::Mat points = points_in.getMat();
186 cv::Mat depth = depth_in.getMat();
187
188 cv::Mat points_float;
189 if (points.depth() != CV_32F)
190 points.convertTo(points_float, CV_32FC2);
191 else
192 points_float = points;
193
194 // Fill the depth matrix
195 cv::Mat_<float> z_mat;
196
197 if (depth.depth() == CV_16U)
198 convertDepthToFloat<ushort>(depth, 1.0f / 1000.0f, points_float, z_mat);
199 else if (depth.depth() == CV_16U)
200 convertDepthToFloat<short>(depth, 1.0f / 1000.0f, points_float, z_mat);
201 else
202 {
203 CV_Assert(depth.type() == CV_32F);
204 convertDepthToFloat<float>(depth, 1.0f, points_float, z_mat);
205 }
206
207 std::vector<cv::Mat> channels(2);
208 cv::split(points_float, channels);
209
210 points3d_out.create(channels[0].rows, channels[0].cols, CV_32FC4);
211 cv::Mat points3d = points3d_out.getMat();
212 depthTo3d_from_uvz(K_in.getMat(), channels[0], channels[1], z_mat, points3d);
213}

223void depthTo3d(InputArray depth_in, InputArray K_in, OutputArray points3d_out, InputArray mask_in)
224{
225 cv::Mat depth = depth_in.getMat();
226 cv::Mat K = K_in.getMat();
227 cv::Mat mask = mask_in.getMat();
228 CV_Assert(K.cols == 3 && K.rows == 3 && (K.depth() == CV_64F || K.depth() == CV_32F));
229 CV_Assert(depth.type() == CV_64FC1 || depth.type() == CV_32FC1 || depth.type() == CV_16UC1 || depth.type() == CV_16SC1);
230 CV_Assert(mask.empty() || mask.channels() == 1);
231
232 cv::Mat K_new;
233 K.convertTo(K_new, depth.depth() == CV_64F ? CV_64F : CV_32F); // issue #1021
234
235 // Create 3D points in one go.
236 if (!mask.empty())
237 {
238 cv::Mat points3d;
239 depthTo3dMask(depth, K_new, mask, points3d);
240 points3d_out.create(points3d.size(), CV_MAKETYPE(K_new.depth(), 4));
241 points3d.copyTo(points3d_out.getMat());
242 }
243 else
244 {
245 points3d_out.create(depth.size(), CV_MAKETYPE(K_new.depth(), 4));
246 cv::Mat points3d = points3d_out.getMat();
247 if (K_new.depth() == CV_64F)
248 depthTo3dNoMask<double>(depth, K_new, points3d);
249 else
250 depthTo3dNoMask<float>(depth, K_new, points3d);
251 }
252}
