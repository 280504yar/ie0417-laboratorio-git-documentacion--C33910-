21// out must be pre-allocated
22static void constantOfShape(const Mat& value, Mat& out)
23{
24 CV_Assert(value.total() == 1);
25 CV_Assert(out.isContinuous());
26 CV_CheckEQ(value.type(), out.type(), "input and output tensor types must be the same");
27
28 size_t esz = value.elemSize();
29 size_t total = out.total();
30 const uchar* inpdata_ = value.data;
31 uchar* outdata_ = out.data;
32
33 #undef IMPL_CONST_OF_SHAPE
34 #define IMPL_CONST_OF_SHAPE(T) \
35 T val = *(const T*)inpdata_; \
36 T* outdata = (T*)outdata_; \
37 for (size_t i = 0; i < total; i++) \
38 outdata[i] = val
39
40 if (esz == 1) {
41 IMPL_CONST_OF_SHAPE(uint8_t);
42 } else if (esz == 2) {
43 IMPL_CONST_OF_SHAPE(uint16_t);
44 } else if (esz == 4) {
45 IMPL_CONST_OF_SHAPE(uint32_t);
46 } else if (esz == 8) {
47 IMPL_CONST_OF_SHAPE(uint64_t);
48 } else {
49 CV_Error_(Error::StsNotImplemented, ("invalid/unsupported tensor type: %s", typeToString(value.type()).c_str()));
50 }
51}

108 void forward(InputArrayOfArrays inputs_arr,
109 OutputArrayOfArrays outputs_arr,
110 OutputArrayOfArrays) CV_OVERRIDE
111 {
112 CV_Assert(blobs.size() == 1);
113 CV_TRACE_FUNCTION();
114 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
115
116 Size size = inputs_arr.size();
117 int ninputs = size.area();
118 CV_Assert(ninputs == 1);
119
120 const Mat& value = blobs[0];
121 Mat shapeTensor = inputs_arr.getMat(0);
122 MatShape shape = tensorToShape(shapeTensor);
123
124 auto kind = outputs_arr.kind();
125 if (kind == _InputArray::STD_VECTOR_MAT) {
126 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
127 outs.resize(1);
128 outs[0].fit(shape, value.type());
129 constantOfShape(value, outs[0]);
130 } else if (kind == _InputArray::STD_VECTOR_UMAT) {
131 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
132 outs.resize(1);
133 outs[0].fit(shape, value.type());
134 Mat temp(shape, value.type());
135 constantOfShape(value, temp);
136 temp.copyTo(outs[0]);
137 } else {
138 CV_Error(Error::StsNotImplemented, "");
139 }
140 }
