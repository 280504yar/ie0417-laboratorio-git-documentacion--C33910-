47 bool getMemoryShapes(const std::vector<MatShape> &inputs,
48 const int requiredOutputs,
49 std::vector<MatShape> &outputs,
50 std::vector<MatShape> &internals) const CV_OVERRIDE
51 {
52 CV_Assert(inputs.size() == 2 || inputs.size() == 3);
53 CV_Assert(total(inputs[0]) == total(inputs[1]));
54
55 MatShape outShape;
56 if (inputs.size() == 2)
57 {
58 outShape = inputs[0];
59 outShape[2] = (outShape[2] - 1) * poolStride.height + poolKernel.height - 2 * poolPad.height;
60 outShape[3] = (outShape[3] - 1) * poolStride.width + poolKernel.width - 2 * poolPad.width;
61 }
62 else
63 outShape = inputs[2];
64
65 outputs.clear();
66 outputs.push_back(outShape);
67
68 return false;
69 }

84 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
85 {
86 CV_TRACE_FUNCTION();
87 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
88
89 std::vector<Mat> inputs, outputs;
90 inputs_arr.getMatVector(inputs);
91 outputs_arr.getMatVector(outputs);
92
93 CV_Assert(inputs.size() == 2 || inputs.size() == 3);
94 Mat& input = inputs[0];
95 Mat& indices = inputs[1];
96
97 if (indices.depth() == CV_32S)
98 typeDispatch<int32_t>(input.type(), input, indices, outputs);
99 else if (indices.depth() == CV_64S)
100 typeDispatch<int64_t>(input.type(), input, indices, outputs);
101 else
102 CV_Error(cv::Error::BadDepth, "Unsupported type.");
103 }

105 template<typename T_INDEX, typename... Args>
106 inline void typeDispatch(const int type, Args&&... args)
107 {
108 switch (type)
109 {
110 case CV_8S:
111 run<int8_t, T_INDEX>(std::forward<Args>(args)...);
112 break;
113 case CV_8U:
114 run<uint8_t, T_INDEX>(std::forward<Args>(args)...);
115 break;
116 case CV_32S:
117 run<int32_t, T_INDEX>(std::forward<Args>(args)...);
118 break;
119 case CV_64S:
120 run<int64_t, T_INDEX>(std::forward<Args>(args)...);
121 break;
122 case CV_32F:
123 run<float, T_INDEX>(std::forward<Args>(args)...);
124 break;
125 case CV_64F:
126 run<double, T_INDEX>(std::forward<Args>(args)...);
127 break;
128 case CV_16F:
129 run<int16_t, T_INDEX>(std::forward<Args>(args)...);
130 break;
131 default:
132 CV_Error(cv::Error::BadDepth, "Unsupported type.");
133 };

136 template<typename T, typename INDEX_TYPE>
137 void run(cv::Mat& input, cv::Mat& indices, std::vector<cv::Mat>& outputs)
138 {
139 CV_Assert(input.total() == indices.total());
140 CV_Assert(input.size[0] == 1);
141 CV_Assert(input.isContinuous());
142
143 for(int i_n = 0; i_n < outputs.size(); i_n++)
144 {
145 Mat& outBlob = outputs[i_n];
146 outBlob.setTo(0);
147 CV_Assert(input.size[1] == outBlob.size[1]);
148 int outPlaneTotal = outBlob.size[2]*outBlob.size[3];
149
150 for (int i_c = 0; i_c < input.size[1]; i_c++)
151 {
152 Mat outPlane = getPlane(outBlob, 0, i_c);
153 int wh_area = input.size[2]*input.size[3];
154 const T* inptr = input.ptr<T>(0, i_c);
155 const INDEX_TYPE* idxptr = indices.ptr<INDEX_TYPE>(0, i_c);
156 T* outptr = outPlane.ptr<T>();
157
158 for(int i_wh = 0; i_wh < wh_area; i_wh++)
159 {
160 int index = idxptr[i_wh];
161 if (!(0 <= index && index < outPlaneTotal))
162 {
163 CV_LOG_ERROR(NULL, cv::format(
164 "i_n=%d\ni_c=%d\ni_wh=%d\nindex=%d\noutPlaneTotal=%d\n",
165 i_n, i_c, i_wh, index, outPlaneTotal));
166 CV_LOG_ERROR(NULL, "input.size=" << input.size);
167 CV_LOG_ERROR(NULL, "indices.size=" << indices.size);
168 CV_LOG_ERROR(NULL, "outBlob=" << outBlob.size);
169 CV_Assert(0 <= index && index < outPlaneTotal);
170 }
171 outptr[index] = inptr[i_wh];
172 }
173 }
174 }

216#ifdef HAVE_DNN_NGRAPH
217 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> >& inputs,
218 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
219 {
220 auto features = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
221 auto indices = nodes[1].dynamicCast<InfEngineNgraphNode>()->node;
222
223 std::vector<MatShape> inpShapes(nodes.size());
224 std::vector<MatShape> outShapes, internals;
225 for (int i = 0; i < nodes.size(); ++i) {
226 std::vector<size_t> shape = nodes[i].dynamicCast<InfEngineNgraphNode>()->node.get_shape();
227 inpShapes[i] = MatShape(shape.begin(), shape.end());
228 }
229 getMemoryShapes(inpShapes, 1, outShapes, internals);
230
231 Mat zeros = Mat::zeros(1, (int)total(outShapes[0]), CV_32F);
232 auto zeroInp = std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{zeros.total()}, zeros.data);
233
234 int newShape = -1;
235 features = std::make_shared<ov::op::v1::Reshape>(
236 features,
237 std::make_shared<ov::op::v0::Constant>(ov::element::i32, ov::Shape{1}, &newShape),
238 true
239 );
240 indices = std::make_shared<ov::op::v1::Reshape>(
241 indices,
242 std::make_shared<ov::op::v0::Constant>(ov::element::i32, ov::Shape{1}, &newShape),
243 true
244 );
245 if (indices.get_element_type() != ov::element::i32 && indices.get_element_type() != ov::element::i64) {
246 indices = std::make_shared<ov::op::v0::Convert>(indices, ov::element::i64);
247 }
248
249 int axis = 0;
250 std::shared_ptr<ov::Node> unpool = std::make_shared<ov::op::v3::ScatterElementsUpdate>(zeroInp, indices, features,
251 std::make_shared<ov::op::v0::Constant>(ov::element::i32, ov::Shape{1}, &axis));
252
253 auto shape = std::make_shared<ov::op::v0::Constant>(ov::element::i32, ov::Shape{outShapes[0].size()}, outShapes[0].data());
254 unpool = std::make_shared<ov::op::v1::Reshape>(unpool, shape, true);
255
256 return Ptr<BackendNode>(new InfEngineNgraphNode(unpool));
257 }
258#endif // HAVE_DNN_NGRAPH
