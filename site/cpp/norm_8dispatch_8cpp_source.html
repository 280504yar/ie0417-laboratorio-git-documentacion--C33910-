58int normHamming(const uchar* a, int n, int cellSize)
59{
60 int output;
61 CALL_HAL_RET(normHamming8u, cv_hal_normHamming8u, output, a, n, cellSize);
62
63 if( cellSize == 1 )
64 return normHamming(a, n);
65 const uchar* tab = 0;
66 if( cellSize == 2 )
67 tab = popCountTable2;
68 else if( cellSize == 4 )
69 tab = popCountTable4;
70 else
71 return -1;
72 int i = 0;
73 int result = 0;
74#if (CV_SIMD || CV_SIMD_SCALABLE)
75 v_uint64 t = vx_setzero_u64();
76 if ( cellSize == 2)
77 {
78 v_uint16 mask = v_reinterpret_as_u16(vx_setall_u8(0x55));
79 for(; i <= n - VTraits<v_uint8>::vlanes(); i += VTraits<v_uint8>::vlanes())
80 {
81 v_uint16 a0 = v_reinterpret_as_u16(vx_load(a + i));
82 t = v_add(t, v_popcount(v_reinterpret_as_u64(v_and(v_or(a0, v_shr<1>(a0)), mask))));
83 }
84 }
85 else // cellSize == 4

88 for(; i <= n - VTraits<v_uint8>::vlanes(); i += VTraits<v_uint8>::vlanes())
89 {
90 v_uint16 a0 = v_reinterpret_as_u16(vx_load(a + i));
91 v_uint16 a1 = v_or(a0, v_shr<2>(a0));
92 t = v_add(t, v_popcount(v_reinterpret_as_u64(v_and(v_or(a1, v_shr<1>(a1)), mask))));
93
94 }
95 }
96 result += (int)v_reduce_sum(t);

98#elif CV_ENABLE_UNROLLED
99 for( ; i <= n - 4; i += 4 )
100 result += tab[a[i]] + tab[a[i+1]] + tab[a[i+2]] + tab[a[i+3]];

156float normL2Sqr_(const float* a, const float* b, int n)
157{
158 int j = 0; float d = 0.f;
159#if (CV_SIMD || CV_SIMD_SCALABLE)
160 const int vl = VTraits<v_float32>::vlanes();
161 v_float32 v_d0 = vx_setzero_f32(), v_d1 = vx_setzero_f32();
162 v_float32 v_d2 = vx_setzero_f32(), v_d3 = vx_setzero_f32();
163 for (; j <= n - 4 * vl; j += 4 * vl)
164 {
165 v_float32 t0 = v_sub(vx_load(a + j), vx_load(b + j));
166 v_float32 t1 = v_sub(vx_load(a + j + vl), vx_load(b + j + vl));
167 v_d0 = v_muladd(t0, t0, v_d0);
168 v_float32 t2 = v_sub(vx_load(a + j + 2 * vl), vx_load(b + j + 2 * vl));
169 v_d1 = v_muladd(t1, t1, v_d1);
170 v_float32 t3 = v_sub(vx_load(a + j + 3 * vl), vx_load(b + j + 3 * vl));
171 v_d2 = v_muladd(t2, t2, v_d2);
172 v_d3 = v_muladd(t3, t3, v_d3);
173 }
174 v_d0 = v_add(v_add(v_d0, v_d1), v_add(v_d2, v_d3));
175 for (; j <= n - vl; j += vl)
176 {
177 v_float32 t0 = v_sub(vx_load(a + j), vx_load(b + j));
178 v_d0 = v_muladd(t0, t0, v_d0);
179 }
180 d = v_reduce_sum(v_d0);
181#endif
182 for( ; j < n; j++ )
183 {
184 float t = a[j] - b[j];
185 d += t*t;
186 }
187 return d;
188}

191float normL1_(const float* a, const float* b, int n)
192{
193 int j = 0; float d = 0.f;
194#if (CV_SIMD || CV_SIMD_SCALABLE)
195 const int vl = VTraits<v_float32>::vlanes();
196 v_float32 v_d0 = vx_setzero_f32(), v_d1 = vx_setzero_f32();
197 v_float32 v_d2 = vx_setzero_f32(), v_d3 = vx_setzero_f32();
198 for (; j <= n - 4 * vl; j += 4 * vl)
199 {
200 v_d0 = v_add(v_d0, v_absdiff(vx_load(a + j), vx_load(b + j)));
201 v_d1 = v_add(v_d1, v_absdiff(vx_load(a + j + vl), vx_load(b + j + vl)));
202 v_d2 = v_add(v_d2, v_absdiff(vx_load(a + j + 2 * vl), vx_load(b + j + 2 * vl)));
203 v_d3 = v_add(v_d3, v_absdiff(vx_load(a + j + 3 * vl), vx_load(b + j + 3 * vl)));
204 }
205 v_d0 = v_add(v_add(v_d0, v_d1), v_add(v_d2, v_d3));
206 for (; j <= n - vl; j += vl)
207 v_d0 = v_add(v_d0, v_absdiff(vx_load(a + j), vx_load(b + j)));
208 d = v_reduce_sum(v_d0);
209#endif
210 for( ; j < n; j++ )
211 d += std::abs(a[j] - b[j]);
212 return d;
213}

215int normL1_(const uchar* a, const uchar* b, int n)
216{
217 int j = 0, d = 0;
218#if (CV_SIMD || CV_SIMD_SCALABLE)
219 const int vl = VTraits<v_uint8>::vlanes();

221 d += v_reduce_sad(vx_load(a + j), vx_load(b + j)) +
222 v_reduce_sad(vx_load(a + j + vl), vx_load(b + j + vl)) +

224 v_reduce_sad(vx_load(a + j + 3 * vl), vx_load(b + j + 3 * vl));
225 for (; j <= n - vl; j += vl)
226 d += v_reduce_sad(vx_load(a + j), vx_load(b + j));
227#endif
228 for( ; j < n; j++ )
229 d += std::abs(a[j] - b[j]);
230 return d;

270 return false;
271 }
272 else if (normType == NORM_L1 || normType == NORM_L2 || normType == NORM_L2SQR)

274 Scalar sc;
275 bool unstype = depth == CV_8U || depth == CV_16U;

279 return false;

283 s += sc[i];
284
285 result = normType == NORM_L1 || normType == NORM_L2SQR ? s : std::sqrt(s);
286 }
287

302double norm( InputArray _src, int normType, InputArray _mask )
303{
304 CV_INSTRUMENT_REGION();
305
306 normType &= NORM_TYPE_MASK;
307 CV_Assert( normType == NORM_INF || normType == NORM_L1 ||
308 normType == NORM_L2 || normType == NORM_L2SQR ||
309 ((normType == NORM_HAMMING || normType == NORM_HAMMING2) && _src.type() == CV_8U) );
310
311#if defined HAVE_OPENCL
312 double _result = 0;
313#endif
314
315#ifdef HAVE_OPENCL
316 CV_OCL_RUN_(OCL_PERFORMANCE_CHECK(_src.isUMat()) && _src.dims() <= 2,
317 ocl_norm(_src, normType, _mask, _result),
318 _result)
319#endif
320
321 Mat src = _src.getMat(), mask = _mask.getMat();
322 int depth = src.depth(), cn = src.channels();
323 if( src.dims <= 2 )
324 {
325 double result;
326 CALL_HAL_RET(norm, cv_hal_norm, result, src.data, src.step, mask.data, mask.step, src.cols, src.rows, src.type(), normType);
327 }
328 else if( src.isContinuous() && mask.isContinuous() )
329 {
330 double result;
331 CALL_HAL_RET(norm, cv_hal_norm, result, src.data, 0, mask.data, 0, (int)src.total(), 1, src.type(), normType);
332 }
333
334 NormFunc func = getNormFunc(normType >> 1, depth);
335 CV_Assert( (normType >> 1) >= 3 || func != 0 );
336
337 if( src.isContinuous() && mask.empty() )
338 {
339 size_t len = src.total()*cn;
340 if( len == (size_t)(int)len )
341 {
342 if( depth == CV_32F )
343 {
344 const uchar* data = src.ptr<const uchar>();
345
346 if( normType == NORM_L2 || normType == NORM_L2SQR || normType == NORM_L1 )
347 {
348 double result = 0;
349 func(data, 0, (uchar*)&result, (int)len, 1);
350 return normType == NORM_L2 ? std::sqrt(result) : result;
351 }
352 if( normType == NORM_INF )
353 {
354 float result = 0;
355 func(data, 0, (uchar*)&result, (int)len, 1);
356 return result;
357 }
358 }
359 if( depth == CV_8U )
360 {
361 const uchar* data = src.ptr<uchar>();
362
363 if( normType == NORM_HAMMING )
364 {
365 return hal::normHamming(data, (int)len, 1);
366 }
367
368 if( normType == NORM_HAMMING2 )
369 {
370 return hal::normHamming(data, (int)len, 2);
371 }
372 }
373 }
374 }
375
376 CV_Assert( mask.empty() || mask.type() == CV_8U || mask.type() == CV_8S || mask.type() == CV_Bool );
377
378 if( normType == NORM_HAMMING || normType == NORM_HAMMING2 )
379 {
380 if( !mask.empty() )
381 {
382 Mat temp;
383 bitwise_and(src, mask, temp);
384 return norm(temp, normType);
385 }
386 int cellSize = normType == NORM_HAMMING ? 1 : 2;
387
388 const Mat* arrays[] = {&src, 0};
389 uchar* ptrs[1] = {};
390 NAryMatIterator it(arrays, ptrs);
391 int total = (int)it.size;
392 int result = 0;
393
394 for( size_t i = 0; i < it.nplanes; i++, ++it )
395 {
396 result += hal::normHamming(ptrs[0], total, cellSize);
397 }
398
399 return result;
400 }
401
402 const Mat* arrays[] = {&src, &mask, 0};
403 uchar* ptrs[2] = {};
404 union
405 {
406 double d;
407 unsigned u;
408 uint64 UL;
409 float f;
410 }
411 result;
412 result.d = 0;
413 NAryMatIterator it(arrays, ptrs);
414 CV_CheckLT((size_t)it.size, (size_t)INT_MAX, "");
415 bool is_fp16 = depth == CV_16F || depth == CV_16BF;
416
417 if ((normType == NORM_L1 && (depth <= CV_16S || depth == CV_Bool || is_fp16)) ||
418 ((normType == NORM_L2 || normType == NORM_L2SQR) && (depth <= CV_8S || depth == CV_Bool || is_fp16)))
419 {
420 // special case to handle "integer" overflow in accumulator
421 const size_t esz = src.elemSize();
422 const int total = (int)it.size;
423 const int blockSize0 = (is_fp16 ? (1 << 10) :
424 depth == CV_Bool ? (1 << 30) :
425 normType == NORM_L1 && depth <= CV_8S ? (1 << 23) : (1 << 15))/cn;
426 const int blockSize = std::min(total, blockSize0);
427 union {
428 int i;
429 float f;
430 } blocksum;
431 blocksum.i = 0;
432 int count = 0;
433
434 for (size_t i = 0; i < it.nplanes; i++, ++it)
435 {
436 for (int j = 0; j < total; j += blockSize)
437 {
438 int bsz = std::min(total - j, blockSize);
439 func(ptrs[0], ptrs[1], (uchar*)&blocksum, bsz, cn);
440 count += bsz;
441 if (count + blockSize >= blockSize0 || (i+1 >= it.nplanes && j+bsz >= total))
442 {
443 result.d += is_fp16 ? (double)blocksum.f : (double)blocksum.i;
444 blocksum.i = 0;
445 count = 0;
446 }
447 ptrs[0] += bsz*esz;
448 if (ptrs[1])
449 ptrs[1] += bsz;
450 }
451 }
452 }
453 else
454 {
455 // generic implementation
456 for (size_t i = 0; i < it.nplanes; i++, ++it)
457 {
458 func(ptrs[0], ptrs[1], (uchar*)&result, (int)it.size, cn);
459 }
460 }
461
462 if( normType == NORM_INF )
463 {
464 if(depth <= CV_32S || depth == CV_32U || depth == CV_Bool)
465 return result.u;
466 if (depth == CV_32F || is_fp16)
467 return result.f;
468 if (depth == CV_64U || depth == CV_64S)
469 return (double)result.UL;
470 }
471 if( normType == NORM_L2 )
472 return std::sqrt(result.d);
473
474 return result.d;
475}

503 if (!ocl_sum(_src1, sc1, normType == NORM_L2 || normType == NORM_L2SQR ?
504 OCL_OP_SUM_SQR : OCL_OP_SUM, _mask, _src2, relative, sc2))
505 return false;
506 }
507 else
508 {
509 if (!ocl_minMaxIdx(_src1, NULL, &sc1[0], NULL, NULL, _mask, std::max(CV_32S, depth),
510 false, _src2, relative ? &sc2[0] : NULL))
511 return false;
512 cn = 1;
513 }
514
515 double s2 = 0;
516 for (int i = 0; i < cn; ++i)
517 {
518 result += sc1[i];
519 if (relative)
520 s2 += sc2[i];
521 }
522
523 if (normType == NORM_L2)
524 {
525 result = std::sqrt(result);
526 if (relative)
527 s2 = std::sqrt(s2);
528 }
529
530 if (relative)
531 result /= (s2 + DBL_EPSILON);
532
533 return true;
534} // ocl_norm()
535#endif // HAVE_OPENCL

537double norm( InputArray _src1, InputArray _src2, int normType, InputArray _mask )
538{
539 CV_INSTRUMENT_REGION();
540
541 CV_CheckTypeEQ(_src1.type(), _src2.type(), "Input type mismatch");
542 CV_Assert(_src1.sameSize(_src2));
543
544#if defined HAVE_OPENCL
545 double _result = 0;
546#endif
547
548#ifdef HAVE_OPENCL
549 CV_OCL_RUN_(OCL_PERFORMANCE_CHECK(_src1.isUMat()),
550 ocl_norm(_src1, _src2, normType, _mask, _result),
551 _result)
552#endif
553
554 Mat src1 = _src1.getMat(), src2 = _src2.getMat(), mask = _mask.getMat();
555 int depth = src1.depth(), cn = src1.channels();
556 if( src1.dims <= 2 )
557 {
558 double result;
559 CALL_HAL_RET(normDiff, cv_hal_normDiff, result, src1.data, src1.step, src2.data, src2.step, mask.data, mask.step, src1.cols, src1.rows, src1.type(), normType);
560 }
561 else if( src1.isContinuous() && src2.isContinuous() && mask.isContinuous() )
562 {
563 double result;
564 CALL_HAL_RET(normDiff, cv_hal_normDiff, result, src1.data, 0, src2.data, 0, mask.data, 0, (int)src1.total(), 1, src1.type(), normType);
565 }
566
567 if( normType & NORM_RELATIVE )
568 {
569 return norm(_src1, _src2, normType & ~NORM_RELATIVE, _mask)/(norm(_src2, normType, _mask) + DBL_EPSILON);
570 }
571
572 normType &= 7;
573 CV_Assert( normType == NORM_INF || normType == NORM_L1 ||
574 normType == NORM_L2 || normType == NORM_L2SQR ||
575 ((normType == NORM_HAMMING || normType == NORM_HAMMING2) && src1.type() == CV_8U) );
576
577 NormDiffFunc func = getNormDiffFunc(normType >> 1, depth);
578 CV_Assert( (normType >> 1) >= 3 || func != 0 );
579
580 if( src1.isContinuous() && src2.isContinuous() && mask.empty() )
581 {
582 size_t len = src1.total()*src1.channels();
583 if( len == (size_t)(int)len )
584 {
585 if( src1.depth() == CV_32F )
586 {
587 const uchar* data1 = src1.ptr<const uchar>();
588 const uchar* data2 = src2.ptr<const uchar>();
589
590 if( normType == NORM_L2 || normType == NORM_L2SQR || normType == NORM_L1 )
591 {
592 double result = 0;
593 func(data1, data2, 0, (uchar*)&result, (int)len, 1);
594 return normType == NORM_L2 ? std::sqrt(result) : result;
595 }
596 if( normType == NORM_INF )
597 {
598 float result = 0;
599 func(data1, data2, 0, (uchar*)&result, (int)len, 1);
600 return result;
601 }
602 }
603 }
604 }
605
606 CV_Assert( mask.empty() || mask.type() == CV_8U || mask.type() == CV_8S || mask.type() == CV_Bool );
607
608 if( normType == NORM_HAMMING || normType == NORM_HAMMING2 )
609 {
610 if( !mask.empty() )
611 {
612 Mat temp;
613 bitwise_xor(src1, src2, temp);
614 bitwise_and(temp, mask, temp);
615 return norm(temp, normType);
616 }
617 int cellSize = normType == NORM_HAMMING ? 1 : 2;
618
619 const Mat* arrays[] = {&src1, &src2, 0};
620 uchar* ptrs[2] = {};
621 NAryMatIterator it(arrays, ptrs);
622 int total = (int)it.size;
623 int result = 0;
624
625 for( size_t i = 0; i < it.nplanes; i++, ++it )
626 {
627 result += hal::normHamming(ptrs[0], ptrs[1], total, cellSize);
628 }
629
630 return result;
631 }
632
633 const Mat* arrays[] = {&src1, &src2, &mask, 0};
634 uchar* ptrs[3] = {};
635 union
636 {
637 double d;
638 float f;
639 unsigned u;
640 uint64 UL;
641 }
642 result;
643 result.d = 0;
644 NAryMatIterator it(arrays, ptrs);
645 CV_CheckLT((size_t)it.size, (size_t)INT_MAX, "");
646
647 bool is_fp16 = depth == CV_16F || depth == CV_16BF;
648
649 if ((normType == NORM_L1 && (depth <= CV_16S || depth == CV_Bool || is_fp16)) ||
650 ((normType == NORM_L2 || normType == NORM_L2SQR) && (depth <= CV_8S || depth == CV_Bool || is_fp16)))
651 {
652 // special case to handle "integer" overflow in accumulator
653 const size_t esz = src1.elemSize();
654 const int total = (int)it.size;
655 const int blockSize0 = (is_fp16 ? (1 << 10) :
656 depth == CV_Bool ? (1 << 30) :
657 normType == NORM_L1 && depth <= CV_8S ? (1 << 23) : (1 << 15))/cn;
658 const int blockSize = std::min(total, blockSize0);
659 union {
660 int i;
661 float f;
662 } blocksum;
663 blocksum.i = 0;
664 int count = 0;
665
666 for (size_t i = 0; i < it.nplanes; i++, ++it)
667 {
668 for (int j = 0; j < total; j += blockSize)
669 {
670 int bsz = std::min(total - j, blockSize);
671 func(ptrs[0], ptrs[1], ptrs[2], (uchar*)&blocksum, bsz, cn);
672 count += bsz;
673 if (count + blockSize >= blockSize0 || (i+1 >= it.nplanes && j+bsz >= total))
674 {
675 result.d += is_fp16 ? (double)blocksum.f : (double)blocksum.i;
676 blocksum.i = 0;
677 count = 0;
678 }
679 ptrs[0] += bsz*esz;
680 ptrs[1] += bsz*esz;
681 if (ptrs[2])
682 ptrs[2] += bsz;
683 }
684 }
685 }
686 else
687 {
688 // generic implementation
689 for (size_t i = 0; i < it.nplanes; i++, ++it)
690 {
691 func(ptrs[0], ptrs[1], ptrs[2], (uchar*)&result, (int)it.size, cn);
692 }
693 }
694
695 if( normType == NORM_INF )
696 {
697 if (depth <= CV_32S || depth == CV_32U || depth == CV_Bool)
698 return result.u;
699 if (depth == CV_32F || is_fp16)
700 return result.f;
701 if (depth == CV_64U || depth == CV_64S)
702 return (double)result.UL;
703 }
704 if( normType == NORM_L2 )
705 return std::sqrt(result.d);
706
707 return result.d;
708}

728static bool ocl_normalize( InputArray _src, InputOutputArray _dst, InputArray _mask, int dtype,
729 double scale, double delta )
730{
731 UMat src = _src.getUMat();
732
733 if( _mask.empty() )
734 src.convertTo( _dst, dtype, scale, delta );
735 else if (src.channels() <= 4)

743 float fscale = static_cast<float>(scale), fdelta = static_cast<float>(delta);
744 bool haveScale = std::fabs(scale - 1) > DBL_EPSILON,

746 haveDelta = std::fabs(delta) > DBL_EPSILON,

765 " -D convertToDT=%s -D workT=%s%s%s%s -D srcT1=%s -D dstT1=%s",

768 rowsPerWI, ocl::convertTypeStr(wdepth, ddepth, cn, cvt[1], sizeof(cvt[1])),

780
781 ocl::KernelArg srcarg = ocl::KernelArg::ReadOnlyNoSize(src),
782 maskarg = ocl::KernelArg::ReadOnlyNoSize(mask),
783 dstarg = ocl::KernelArg::ReadWrite(dst);
784
785 if (haveScale)
786 {
787 if (haveDelta)
788 k.args(srcarg, maskarg, dstarg, fscale, fdelta);
789 else
790 k.args(srcarg, maskarg, dstarg, fscale);
791 }
792 else
793 {
794 if (haveDelta)
795 k.args(srcarg, maskarg, dstarg, fdelta);
796 else
797 k.args(srcarg, maskarg, dstarg);
798 }
799
800 size_t globalsize[2] = { (size_t)src.cols, ((size_t)src.rows + rowsPerWI - 1) / rowsPerWI };
801 return k.run(2, globalsize, NULL, false);
802 }
803 else
804 {
805 UMat temp;
806 src.convertTo( temp, dtype, scale, delta );
807 temp.copyTo( _dst, _mask );
808 }
809
810 return true;
811} // ocl_normalize

814void normalize(InputArray _src, InputOutputArray _dst, double a, double b,
815 int norm_type, int rtype, InputArray _mask)
816{
817 CV_INSTRUMENT_REGION();
818
819 double scale = 1, shift = 0;
820 int type = _src.type(), depth = CV_MAT_DEPTH(type);
821
822 if( rtype < 0 )
823 rtype = _dst.fixedType() ? _dst.depth() : depth;
824
825 if( norm_type == NORM_MINMAX )
826 {
827 double smin = 0, smax = 0;
828 double dmin = MIN( a, b ), dmax = MAX( a, b );
829 minMaxIdx( _src, &smin, &smax, 0, 0, _mask );
830 scale = (dmax - dmin)*(smax - smin > DBL_EPSILON ? 1./(smax - smin) : 0);
831 if( rtype == CV_32F )
832 {
833 scale = (float)scale;
834 shift = (float)dmin - (float)(smin*scale);
835 }
836 else
837 shift = dmin - smin*scale;
838 }
839 else if( norm_type == NORM_L2 || norm_type == NORM_L1 || norm_type == NORM_INF )
840 {
841 scale = norm( _src, norm_type, _mask );
842 scale = scale > DBL_EPSILON ? a/scale : 0.;
843 shift = 0;
844 }
845 else
846 CV_Error( cv::Error::StsBadArg, "Unknown/unsupported norm type" );
847
848 CV_OCL_RUN(_dst.isUMat(),
849 ocl_normalize(_src, _dst, _mask, rtype, scale, shift))
850
851 Mat src = _src.getMat();
852 if( _mask.empty() )
853 src.convertTo( _dst, rtype, scale, shift );
854 else
855 {
856 Mat temp;
857 src.convertTo( temp, rtype, scale, shift );
858 temp.copyTo( _dst, _mask );
859 }
860}
