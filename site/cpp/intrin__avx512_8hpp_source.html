1600 __m512i v_lo = _mm512_mask_blend_epi32(0x5555, magic_i_lo, v.val);

1607 // (v_hi - magic_d_all) + v_lo Do not assume associativity of floating point addition
1608 __m512d result = _mm512_add_pd(v_hi_dbl, _mm512_castsi512_pd(v_lo));
