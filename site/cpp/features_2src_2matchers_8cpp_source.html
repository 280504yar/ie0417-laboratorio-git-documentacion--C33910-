76static bool ocl_matchSingle(InputArray query, InputArray train,
77 UMat &trainIdx, UMat *distance, int distType)
78{
79 if (query.empty() || train.empty())
80 return false;
81
82 const int query_rows = query.rows();
83 const int query_cols = query.cols();
84
85 ensureSizeIsEnough(1, query_rows, CV_32S, trainIdx);
86 if (distance)
87 ensureSizeIsEnough(1, query_rows, CV_32F, *distance);
88
89 ocl::Device devDef = ocl::Device::getDefault();
90
91 UMat uquery = query.getUMat(), utrain = train.getUMat();
92 int kercn = 1;
93 if (devDef.isIntel() &&
94 (0 == (uquery.step % 4)) && (0 == (uquery.cols % 4)) && (0 == (uquery.offset % 4)) &&
95 (0 == (utrain.step % 4)) && (0 == (utrain.cols % 4)) && (0 == (utrain.offset % 4)))
96 kercn = 4;
97
98 int block_size = 16;
99 int max_desc_len = 0;
100 bool is_cpu = devDef.type() == ocl::Device::TYPE_CPU;
101 if (query_cols <= 64)
102 max_desc_len = 64 / kercn;
103 else if (query_cols <= 128 && !is_cpu)
104 max_desc_len = 128 / kercn;
105
106 int depth = query.depth();
107 cv::String opts;
108 opts = cv::format("-D T=%s -D TN=%s -D kercn=%d %s -D DIST_TYPE=%d -D BLOCK_SIZE=%d -D MAX_DESC_LEN=%d",
109 ocl::typeToStr(depth), ocl::typeToStr(CV_MAKETYPE(depth, kercn)), kercn, depth == CV_32F ? "-D T_FLOAT" : "", distType, block_size, max_desc_len);
110 ocl::Kernel k("BruteForceMatch_Match", ocl::features::brute_force_match_oclsrc, opts);
111 if(k.empty())
112 return false;
113
114 size_t globalSize[] = {((size_t)query.size().height + block_size - 1) / block_size * block_size, (size_t)block_size};
115 size_t localSize[] = {(size_t)block_size, (size_t)block_size};
116
117 int idx = 0;
118 idx = k.set(idx, ocl::KernelArg::PtrReadOnly(uquery));
119 idx = k.set(idx, ocl::KernelArg::PtrReadOnly(utrain));
120 idx = k.set(idx, ocl::KernelArg::PtrWriteOnly(trainIdx));
121 UMat dummyDist;
122 if (!distance)
123 ensureSizeIsEnough(1, query_rows, CV_32F, dummyDist);
124 idx = k.set(idx, ocl::KernelArg::PtrWriteOnly(distance ? *distance : dummyDist));
125 idx = k.set(idx, uquery.rows);
126 idx = k.set(idx, uquery.cols);
127 idx = k.set(idx, utrain.rows);
128 idx = k.set(idx, utrain.cols);
129 idx = k.set(idx, (int)(uquery.step / sizeof(float)));
130
131 return k.run(2, globalSize, localSize, false);
132}

134static bool ocl_matchConvert(const Mat &trainIdx, const Mat &distance, std::vector< std::vector<DMatch> > &matches)
135{
136 if (trainIdx.empty() || distance.empty())
137 return false;
138
139 if( (trainIdx.type() != CV_32SC1) || (distance.type() != CV_32FC1 || distance.cols != trainIdx.cols) )
140 return false;

143
144 matches.clear();
145 matches.reserve(nQuery);
146
147 const int *trainIdx_ptr = trainIdx.ptr<int>();
148 const float *distance_ptr = distance.ptr<float>();

151 int trainIndex = *trainIdx_ptr;
152
153 if (trainIndex == -1)
154 continue;
155
156 float dst = *distance_ptr;
157
158 DMatch m(queryIdx, trainIndex, 0, dst);
159
160 std::vector<DMatch> temp;
161 temp.push_back(m);
162 matches.push_back(temp);
163 }
164 return true;
165}

234static bool ocl_knnMatchConvert(const Mat &trainIdx, const Mat &distance, std::vector< std::vector<DMatch> > &matches, bool compactResult)
235{
236 if (trainIdx.empty() || distance.empty())
237 return false;
238
239 if(trainIdx.type() != CV_32SC2 && trainIdx.type() != CV_32SC1) return false;
240 if(distance.type() != CV_32FC2 && distance.type() != CV_32FC1)return false;
241 if(distance.size() != trainIdx.size()) return false;
242 if(!trainIdx.isContinuous() || !distance.isContinuous()) return false;
243
244 const int nQuery = trainIdx.type() == CV_32SC2 ? trainIdx.cols : trainIdx.rows;
245 const int k = trainIdx.type() == CV_32SC2 ? 2 : trainIdx.cols;
246
247 matches.clear();
248 matches.reserve(nQuery);
249
250 const int *trainIdx_ptr = trainIdx.ptr<int>();
251 const float *distance_ptr = distance.ptr<float>();
252
253 for (int queryIdx = 0; queryIdx < nQuery; ++queryIdx)
254 {
255 matches.push_back(std::vector<DMatch>());
256 std::vector<DMatch> &curMatches = matches.back();
257 curMatches.reserve(k);
258
259 for (int i = 0; i < k; ++i, ++trainIdx_ptr, ++distance_ptr)
260 {
261 int trainIndex = *trainIdx_ptr;
262
263 if (trainIndex != -1)
264 {
265 float dst = *distance_ptr;
266
267 DMatch m(queryIdx, trainIndex, 0, dst);
268
269 curMatches.push_back(m);
270 }
271 }
272
273 if (compactResult && curMatches.empty())
274 matches.pop_back();
275 }
276 return true;
277}

291 UMat &trainIdx, UMat &distance, UMat &nMatches, float maxDistance, int distType)
292{
293 if (query.empty() || train.empty())

303 ensureSizeIsEnough(query_rows, std::max((train_rows / 100), 10), CV_32SC1, trainIdx);
304 ensureSizeIsEnough(query_rows, std::max((train_rows / 100), 10), CV_32FC1, distance);
305 }
306
307 nMatches.setTo(Scalar::all(0));
308
309 ocl::Device devDef = ocl::Device::getDefault();
310 UMat uquery = query.getUMat(), utrain = train.getUMat();
311 int kercn = 1;

314 (0 == (utrain.step % 4)) && (0 == (utrain.cols % 4)) && (0 == (utrain.offset % 4)))
315 kercn = 4;
316

318 int depth = query.depth();
319 cv::String opts;
320 opts = cv::format("-D T=%s -D TN=%s -D kercn=%d %s -D DIST_TYPE=%d -D BLOCK_SIZE=%d",
321 ocl::typeToStr(depth), ocl::typeToStr(CV_MAKETYPE(depth, kercn)), kercn, depth == CV_32F ? "-D T_FLOAT" : "", distType, block_size);
322 ocl::Kernel k("BruteForceMatch_RadiusMatch", ocl::features::brute_force_match_oclsrc, opts);
323 if (k.empty())
324 return false;

327 size_t localSize[] = {(size_t)block_size, (size_t)block_size};

332 idx = k.set(idx, maxDistance);
333 idx = k.set(idx, ocl::KernelArg::PtrWriteOnly(trainIdx));
334 idx = k.set(idx, ocl::KernelArg::PtrWriteOnly(distance));
335 idx = k.set(idx, ocl::KernelArg::PtrWriteOnly(nMatches));
336 idx = k.set(idx, uquery.rows);
337 idx = k.set(idx, uquery.cols);

339 idx = k.set(idx, utrain.cols);

342 idx = k.set(idx, (int)(trainIdx.step / sizeof(int)));

347static bool ocl_radiusMatchConvert(const Mat &trainIdx, const Mat &distance, const Mat &_nMatches,
348 std::vector< std::vector<DMatch> > &matches, bool compactResult)
349{
350 if (trainIdx.empty() || distance.empty() || _nMatches.empty())
351 return false;
352
353 if( (trainIdx.type() != CV_32SC1) ||
354 (distance.type() != CV_32FC1 || distance.size() != trainIdx.size()) ||
355 (_nMatches.type() != CV_32SC1 || _nMatches.cols != trainIdx.rows) )
356 return false;
357
358 const int nQuery = trainIdx.rows;
359
360 matches.clear();
361 matches.reserve(nQuery);
362
363 const int *nMatches_ptr = _nMatches.ptr<int>();
364
365 for (int queryIdx = 0; queryIdx < nQuery; ++queryIdx)
366 {
367 const int *trainIdx_ptr = trainIdx.ptr<int>(queryIdx);
368 const float *distance_ptr = distance.ptr<float>(queryIdx);
369
370 const int nMatches = std::min(nMatches_ptr[queryIdx], trainIdx.cols);
371
372 if (nMatches == 0)
373 {
374 if (!compactResult)
375 matches.push_back(std::vector<DMatch>());
376 continue;
377 }
378
379 matches.push_back(std::vector<DMatch>(nMatches));
380 std::vector<DMatch> &curMatches = matches.back();
381
382 for (int i = 0; i < nMatches; ++i, ++trainIdx_ptr, ++distance_ptr)
383 {
384 int trainIndex = *trainIdx_ptr;
385
386 float dst = *distance_ptr;
387
388 DMatch m(queryIdx, trainIndex, 0, dst);
389
390 curMatches[i] = m;
391 }
392
393 std::sort(curMatches.begin(), curMatches.end());
394 }
395 return true;
396}

427void DescriptorMatcher::DescriptorCollection::set( const std::vector<Mat>& descriptors )
428{
429 clear();
430
431 size_t imageCount = descriptors.size();
432 CV_Assert( imageCount > 0 );
433
434 startIdxs.resize( imageCount );
435
436 int dim = -1;
437 int type = -1;
438 startIdxs[0] = 0;
439 for( size_t i = 1; i < imageCount; i++ )
440 {
441 int s = 0;
442 if( !descriptors[i-1].empty() )
443 {
444 dim = descriptors[i-1].cols;
445 type = descriptors[i-1].type();
446 s = descriptors[i-1].rows;
447 }
448 startIdxs[i] = startIdxs[i-1] + s;
449 }
450 if( imageCount == 1 )
451 {
452 if( descriptors[0].empty() ) return;
453
454 dim = descriptors[0].cols;
455 type = descriptors[0].type();
456 }
457 CV_Assert( dim > 0 );
458
459 int count = startIdxs[imageCount-1] + descriptors[imageCount-1].rows;
460
461 if( count > 0 )
462 {
463 mergedDescriptors.create( count, dim, type );
464 for( size_t i = 0; i < imageCount; i++ )
465 {
466 if( !descriptors[i].empty() )
467 {
468 CV_Assert( descriptors[i].cols == dim && descriptors[i].type() == type );
469 Mat m = mergedDescriptors.rowRange( startIdxs[i], startIdxs[i] + descriptors[i].rows );
470 descriptors[i].copyTo(m);
471 }
472 }
473 }

764static bool ocl_matchWithCrossCheckSinglePass(InputArray query, InputArray train,
765 std::vector< std::vector<DMatch> >& matches, int dstType)
766{
767 if (query.empty() || train.empty())
768 return false;
769
770 const int query_rows = query.rows();
771 const int train_rows = train.rows();
772
773 UMat fwdIdx, fwdDist;
774 ensureSizeIsEnough(1, query_rows, CV_32S, fwdIdx);
775 ensureSizeIsEnough(1, query_rows, CV_32F, fwdDist);
776
777 UMat revBest;
778 ensureSizeIsEnough(1, train_rows, CV_32SC2, revBest);
779 revBest.setTo(Scalar::all(-1));
780
781 ocl::Device devDef = ocl::Device::getDefault();
782
783 UMat uquery = query.getUMat(), utrain = train.getUMat();
784 int kercn = 1;
785 if (devDef.isIntel() &&
786 (0 == (uquery.step % 4)) && (0 == (uquery.cols % 4)) && (0 == (uquery.offset % 4)) &&
787 (0 == (utrain.step % 4)) && (0 == (utrain.cols % 4)) && (0 == (utrain.offset % 4)))
788 kercn = 4;
789
790 int block_size = 16;
791 int max_desc_len = 0;
792 bool is_cpu = devDef.type() == ocl::Device::TYPE_CPU;
793 if (uquery.cols <= 64)
794 max_desc_len = 64 / kercn;
795 else if (uquery.cols <= 128 && !is_cpu)
796 max_desc_len = 128 / kercn;
797
798 int depth = uquery.depth();
799 cv::String opts;
800 opts = cv::format("-D T=%s -D TN=%s -D kercn=%d %s -D DIST_TYPE=%d -D BLOCK_SIZE=%d -D MAX_DESC_LEN=%d -D HAVE_INT64_ATOMICS",
801 ocl::typeToStr(depth), ocl::typeToStr(CV_MAKETYPE(depth, kercn)), kercn, depth == CV_32F ? "-D T_FLOAT" : "", dstType, block_size, max_desc_len);
802 ocl::Kernel k("BruteForceMatch_CrossCheckMatch", ocl::features::brute_force_match_oclsrc, opts);
803 if(k.empty())
804 return false;
805
806 size_t globalSize[] = {((size_t)uquery.size().height + block_size - 1) / block_size * block_size, (size_t)block_size};
807 size_t localSize[] = {(size_t)block_size, (size_t)block_size};
808
809 int idx = 0;
810 idx = k.set(idx, ocl::KernelArg::PtrReadOnly(uquery));
811 idx = k.set(idx, ocl::KernelArg::PtrReadOnly(utrain));
812 idx = k.set(idx, ocl::KernelArg::PtrWriteOnly(fwdIdx));
813 idx = k.set(idx, ocl::KernelArg::PtrWriteOnly(fwdDist));
814 idx = k.set(idx, ocl::KernelArg::PtrWriteOnly(revBest));
815 idx = k.set(idx, uquery.rows);
816 idx = k.set(idx, uquery.cols);
817 idx = k.set(idx, utrain.rows);
818 idx = k.set(idx, utrain.cols);
819 idx = k.set(idx, (int)(uquery.step / sizeof(float)));
820
821 if (!k.run(2, globalSize, localSize, false))
822 return false;
823
824 Mat fwdIdxCPU = fwdIdx.getMat(ACCESS_READ);
825 Mat fwdDistCPU = fwdDist.getMat(ACCESS_READ);
826 Mat revBestCPU = revBest.getMat(ACCESS_READ);
827
828 if (fwdIdxCPU.empty() || fwdDistCPU.empty() || revBestCPU.empty())
829 return false;
830
831 const int* fwd = fwdIdxCPU.ptr<int>();
832 const float* dist = fwdDistCPU.ptr<float>();
833 const int* revBestData = revBestCPU.ptr<int>();
834
835 matches.clear();
836 matches.reserve(query_rows);
837
838 for (int q = 0; q < query_rows; ++q)
839 {
840 int t = fwd[q];
841 if (t >= 0 && t < train_rows)
842 {
843 uint64_t packed_val;
844 memcpy(&packed_val, &revBestData[2 * t], sizeof(uint64_t));
845 int revQueryIdx = (int)(packed_val & 0xFFFFFFFF);
846 if (revQueryIdx == q)
847 {
848 matches.push_back(std::vector<DMatch>(1, DMatch(q, t, 0, dist[q])));
849 }
850 }
851 }
852 return true;
853}

855static bool ocl_matchWithCrossCheck(InputArray query, InputArray train,
856 std::vector< std::vector<DMatch> >& matches, int dstType)
857{
858 if (query.empty() || train.empty())
859 return false;
860
861 ocl::Device devDef = ocl::Device::getDefault();
862
863 if (devDef.isExtensionSupported("cl_khr_int64_base_atomics"))
864 {
865 if (ocl_matchWithCrossCheckSinglePass(query, train, matches, dstType))
866 return true;
867 }
868
869 UMat fwdIdx, fwdDist;
870 if (!ocl_matchSingle(query, train, fwdIdx, &fwdDist, dstType))
871 return false;
872
873 UMat revIdx;
874 if (!ocl_matchSingle(train, query, revIdx, nullptr, dstType))
875 return false;
876
877 Mat fwdIdxCPU = fwdIdx.getMat(ACCESS_READ);
878 Mat revIdxCPU = revIdx.getMat(ACCESS_READ);
879 Mat fwdDistCPU = fwdDist.getMat(ACCESS_READ);
880
881 if (fwdIdxCPU.empty() || revIdxCPU.empty() || fwdDistCPU.empty())
882 return false;
883
884 const int nQuery = fwdIdxCPU.cols;
885 const int nTrain = revIdxCPU.cols;
886 const int* fwd = fwdIdxCPU.ptr<int>();
887 const int* rev = revIdxCPU.ptr<int>();
888 const float* dist = fwdDistCPU.ptr<float>();
889
890 matches.clear();
891 matches.reserve(nQuery);
892
893 for (int q = 0; q < nQuery; ++q)
894 {
895 int t = fwd[q];
896 if (t >= 0 && t < nTrain && rev[t] == q)
897 {
898 matches.push_back(std::vector<DMatch>(1, DMatch(q, t, 0, dist[q])));
899 }
900 }
901 return true;

905void BFMatcher::knnMatchImpl( InputArray _queryDescriptors, std::vector<std::vector<DMatch> >& matches, int knn,
906 InputArrayOfArrays _masks, bool compactResult )
907{
908 int trainDescType = trainDescCollection.empty() ? utrainDescCollection[0].type() : trainDescCollection[0].type();
909 CV_Assert( _queryDescriptors.type() == trainDescType );
910
911 const int IMGIDX_SHIFT = 18;
912 const int IMGIDX_ONE = (1 << IMGIDX_SHIFT);
913
914 if( _queryDescriptors.empty() || (trainDescCollection.empty() && utrainDescCollection.empty()))
915 {
916 matches.clear();
917 return;
918 }
919
920 std::vector<Mat> masks;
921 _masks.getMatVector(masks);
922
923 if(!trainDescCollection.empty() && !utrainDescCollection.empty())
924 {
925 for(int i = 0; i < (int)utrainDescCollection.size(); i++)
926 {
927 Mat tempMat;
928 utrainDescCollection[i].copyTo(tempMat);
929 trainDescCollection.push_back(tempMat);
930 }
931 utrainDescCollection.clear();
932 }
933
934#ifdef HAVE_OPENCL
935 int trainDescVectorSize = trainDescCollection.empty() ? (int)utrainDescCollection.size() : (int)trainDescCollection.size();
936 Size trainDescSize = trainDescCollection.empty() ? utrainDescCollection[0].size() : trainDescCollection[0].size();
937 int trainDescOffset = trainDescCollection.empty() ? (int)utrainDescCollection[0].offset : 0;
938
939 if ( ocl::isOpenCLActivated() && _queryDescriptors.isUMat() && _queryDescriptors.dims()<=2 && trainDescVectorSize == 1 &&
940 _queryDescriptors.type() == CV_32FC1 && _queryDescriptors.offset() == 0 && trainDescOffset == 0 &&
941 trainDescSize.width == _queryDescriptors.size().width && masks.size() == 1 && masks[0].total() == 0 )
942 {
943 if(knn == 1)
944 {
945 InputArray train = trainDescCollection.empty() ?
946 (InputArray)utrainDescCollection[0] : (InputArray)trainDescCollection[0];
947 bool oclOk = crossCheck ?
948 ocl_matchWithCrossCheck(_queryDescriptors, train, matches, normType) :
949 ocl_match(_queryDescriptors, train, matches, normType);
950 if (oclOk)
951 {
952 CV_IMPL_ADD(CV_IMPL_OCL);
953 return;
954 }
955 }
956 else
957 {
958 if(trainDescCollection.empty())
959 {
960 if(ocl_knnMatch(_queryDescriptors, utrainDescCollection[0], matches, knn, normType, compactResult) )
961 {
962 CV_IMPL_ADD(CV_IMPL_OCL);
963 return;
964 }
965 }
966 else
967 {
968 if(ocl_knnMatch(_queryDescriptors, trainDescCollection[0], matches, knn, normType, compactResult) )
969 {
970 CV_IMPL_ADD(CV_IMPL_OCL);
971 return;
972 }
973 }
974 }
975 }
976#endif
977
978 Mat queryDescriptors = _queryDescriptors.getMat();
979 if(trainDescCollection.empty() && !utrainDescCollection.empty())
980 {
981 for(int i = 0; i < (int)utrainDescCollection.size(); i++)
982 {
983 Mat tempMat;
984 utrainDescCollection[i].copyTo(tempMat);
985 trainDescCollection.push_back(tempMat);
986 }
987 utrainDescCollection.clear();
988 }
989
990 matches.reserve(queryDescriptors.rows);
991
992 Mat dist, nidx;
993
994 int iIdx, imgCount = (int)trainDescCollection.size(), update = 0;
995 int dtype = normType == NORM_HAMMING || normType == NORM_HAMMING2 ||
996 (normType == NORM_L1 && queryDescriptors.type() == CV_8U) ? CV_32S : CV_32F;
997
998 CV_Assert( (int64)imgCount*IMGIDX_ONE < INT_MAX );
999
1000 for( iIdx = 0; iIdx < imgCount; iIdx++ )
1001 {
1002 CV_Assert( trainDescCollection[iIdx].rows < IMGIDX_ONE );
1003 Mat mask = (crossCheck || masks.empty()) ? Mat() : masks[iIdx];
1004 batchDistance(queryDescriptors, trainDescCollection[iIdx], dist, dtype, nidx,
1005 normType, knn, mask, update, crossCheck);
1006 update += IMGIDX_ONE;
1007 }
1008
1009 if( dtype == CV_32S )
1010 {
1011 Mat temp;
1012 dist.convertTo(temp, CV_32F);
1013 dist = temp;
1014 }
1015
1016 for( int qIdx = 0; qIdx < queryDescriptors.rows; qIdx++ )
1017 {
1018 const float* distptr = dist.ptr<float>(qIdx);
1019 const int* nidxptr = nidx.ptr<int>(qIdx);
1020
1021 matches.push_back( std::vector<DMatch>() );
1022 std::vector<DMatch>& mq = matches.back();
1023 mq.reserve(knn);
1024
1025 for( int k = 0; k < nidx.cols; k++ )
1026 {
1027 if( nidxptr[k] < 0 )
1028 break;
1029 mq.push_back( DMatch(qIdx, nidxptr[k] & (IMGIDX_ONE - 1),
1030 nidxptr[k] >> IMGIDX_SHIFT, distptr[k]) );
1031 }
1032
1033 if( mq.empty() && compactResult )
1034 matches.pop_back();
1035 }
1036}
