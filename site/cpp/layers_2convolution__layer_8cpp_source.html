78class BaseConvolutionLayerImpl : public ConvolutionLayer
79{
80public:
81 bool fusedWeights, fusedBias;
82 std::vector<double> weightsMultipliers;
83 int groups;
84 BaseConvolutionLayerImpl(const LayerParams &params)
85 {
86 setParamsFrom(params);
87 getConvolutionKernelParams(params, kernel_size, pads_begin, pads_end, strides, dilations,
88 padMode, adjust_pads, useWinograd);
89
90 numOutput = -1;
91 groups = params.get<int>("group", 1);
92
93 if (kernel_size.size() == 2) {
94 kernel = Size(kernel_size[1], kernel_size[0]);
95 stride = Size(strides[1], strides[0]);
96 pad = Size(pads_begin[1], pads_begin[0]);
97 dilation = Size(dilations[1], dilations[0]);
98
99 adjustPad.height = adjust_pads[0];
100 adjustPad.width = adjust_pads[1];
101 }
102
103 for (int i = 0; i < adjust_pads.size(); i++) {
104 CV_Assert(adjust_pads[i] < strides[i]);
105 }
106
107 fusedWeights = false;
108 fusedBias = false;
109 }

111 virtual void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr) CV_OVERRIDE
112 {
113 std::vector<Mat> inputs, outputs;
114 inputs_arr.getMatVector(inputs);
115 outputs_arr.getMatVector(outputs);
116
117 CV_Assert((inputs.size() > outputs.size() && blobs.empty()) ||
118 (!inputs.empty() && (blobs.size() == 1 || blobs.size() == 2)));
119 MatShape weightShape = blobs.empty() ? inputs[1].shape() : blobs[0].shape();
120 numOutput = weightShape[0];
121
122 CV_Assert(inputs[0].dims == outputs[0].dims);
123 if (weightShape.dims == 3)
124 {
125 kernel_size.resize(1, kernel_size[0]);
126 strides.resize(1, strides[0]);
127 dilations.resize(1, dilations[0]);
128 pads_begin.resize(1, pads_begin[0]);
129 pads_end.resize(1, pads_end[0]);
130 }
131 CV_Assert(weightShape.dims == kernel_size.size() + 2);
132 for (int i = 0; i < kernel_size.size(); i++) {
133 CV_Assert(weightShape[i + 2] == kernel_size[i]);
134 }
135
136 const Mat &input = inputs[0];
137 CV_Assert(((input.dims == 3 && kernel_size.size() == 1) || input.dims == 4 || input.dims == 5) && (input.type() == CV_32F || input.type() == CV_16F));
138 for (size_t i = 0; i < outputs.size(); i++)
139 {
140 CV_Assert(inputs[i].type() == input.type());
141 CV_Assert(((input.dims == 3 && kernel_size.size() == 1) || inputs[i].dims == 4 || inputs[i].dims == 5) && inputs[i].size[1] == input.size[1]);
142 for (int j = 0; j < inputs[i].dims; j++) {
143 CV_Assert(inputs[i].size[j] == input.size[j]);
144 }
145 }
146
147 std::vector<int> inpShape;
148 std::vector<int> outShape;
149 for (int i = 2; i < inputs[0].dims; i++) {
150 inpShape.push_back(inputs[0].size[i]);
151 outShape.push_back(outputs[0].size[i]);
152 }
153 getConvPoolPaddings(inpShape, kernel_size, strides, padMode, pads_begin, pads_end);
154 if (pads_begin.size() == 2) {
155 pad = Size(pads_begin[1], pads_begin[0]);
156 }
157 fusedWeights = false;
158 fusedBias = false;
159 }
160
161 bool hasBias() const
162 {
163 return blobs.size() >= 2;
164 }

166 virtual MatShape computeColRowShape(const MatShape &inpShape, const MatShape &outShape) const = 0;
167 bool is1x1() const
168 {
169 return (kernel.height == 1 && kernel.width == 1) &&
170 (stride.height == 1 && stride.width == 1) &&
171 (dilation.height == 1 && dilation.width == 1);
172 }
173
174 virtual bool tryFuse(Ptr<Layer>& top) CV_OVERRIDE
175 {
176 if (fusedAdd) // If the Conv layer has fused Add layer, it cannot fuse other layers.
177 return false;
178
179 Ptr<BlankLayer> blank_layer = top.dynamicCast<BlankLayer>();
180 if (blank_layer)
181 return true;
182
183 Mat w, b;
184 top->getScaleShift(w, b);
185 if (!w.empty() || !b.empty())
186 {
187 fuseWeights(w, b);
188 fusedWeights = fusedWeights || !w.empty();
189 fusedBias = fusedBias || (hasBias() && !w.empty()) || !b.empty();
190 return true;
191 }
192 return false;
193 }

195 virtual void fuseWeights(const Mat& w_, const Mat& b_) = 0;
196};

314 bool getMemoryShapes(const std::vector<MatShape> &inputs,
315 const int requiredOutputs,
316 std::vector<MatShape> &outputs,
317 std::vector<MatShape> &internals) const CV_OVERRIDE
318 {
319 CV_Assert(!blobs.empty() || inputs.size() > 1);
320 const int* weightShape = blobs.empty() ? &inputs[1][0] : blobs[0].size.p;
321 CV_Assert(!hasBias() || blobs[1].total() == (size_t)weightShape[0]);
322
323 internals.clear();
324
325 CV_Assert(!inputs.empty());
326 CV_Assert(inputs[0].size() > 2);
327 std::vector<int> inpShape(inputs[0].begin() + 2, inputs[0].end());
328
329 int outCn = weightShape[0];
330 std::vector<int> outShape;
331 outShape.push_back(inputs[0][0]);
332 outShape.push_back(outCn);
333
334 int inpCn = inputs[0][1];
335 if (padMode.empty())
336 {
337 for (int i = 0; i < inpShape.size(); i++)
338 outShape.push_back((inpShape[i] + pads_begin[i] + pads_end[i] -
339 dilations[i] * (kernel_size[i] - 1) - 1) / strides[i] + 1);
340 }
341 else
342 {
343 getConvPoolOutParams(inpShape, kernel_size, strides, padMode, dilations, outShape);
344 }
345
346 int ngroups = inpCn / weightShape[1];
347 if (ngroups == 0 || ngroups * weightShape[1] != inpCn)
348 CV_Error(Error::StsError, format("Number of input channels should "
349 "be multiple of %d but got %d", weightShape[1], inpCn));
350 CV_Assert(ngroups > 0 && inpCn % ngroups == 0 && outCn % ngroups == 0);
351
352 outputs.resize(1, MatShape(outShape));
353
354 return false;
355 }

357 virtual void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr) CV_OVERRIDE
358 {
359 BaseConvolutionLayerImpl::finalize(inputs_arr, outputs_arr);
360 std::vector<Mat> inputs;
361 inputs_arr.getMatVector(inputs);
362 // prepare weightsMat where each row is aligned and has enough zero padding on the right to
363 // use vectorized (i.e. with intrinsics) loops without tail processing
364 if (!blobs.empty())
365 {
366 Mat wm = blobs[0].reshape(1, numOutput);
367 if ((wm.step1() % VEC_ALIGN != 0) ||
368 !isAligned<VEC_ALIGN * sizeof(float)>(wm.data)
369 )
370 {
371 int newcols = (int)alignSize(wm.step1(), VEC_ALIGN);
372 Mat wm_buffer = Mat(numOutput, newcols, wm.type());
373 Mat wm_padding = wm_buffer.colRange(wm.cols, newcols);
374 wm_padding.setTo(Scalar::all(0.));
375 Mat wm_aligned = wm_buffer.colRange(0, wm.cols);
376 wm.copyTo(wm_aligned);
377 wm = wm_aligned;
378 }
379 weightsMat = wm;
380 }
381 else
382 {
383 // initialized in .forward()
384 weightsMat.release();
385 }
386
387 weightsMultipliers.assign(numOutput, 1.0);
388
389 Mat biasMat = hasBias() ? blobs[1].reshape(1, numOutput) : Mat();
390 biasvec.resize(numOutput+2);
391 if( biasMat.empty() )
392 {
393 for(int i = 0; i < numOutput; i++ )
394 biasvec[i] = 0.f;
395 }
396 else
397 {
398 for(int i = 0; i < numOutput; i++ )
399 biasvec[i] = biasMat.at<float>(i);
400 }
401#ifdef HAVE_OPENCL
402 convolutionOp.release();
403#endif
404
405 // Winograd only works when input h and w >= 12.
406 canUseWinograd = useWinograd && inputs[0].dims == 4 && inputs[0].size[2] >= 12 && inputs[0].size[3] >= 12;
407 if (fastConvImpl && (fastConvImpl->conv_type == CONV_TYPE_WINOGRAD3X3) ^ canUseWinograd)
408 {
409 fastConvImpl.reset();
410 }
411 }

413 bool setActivation(const Ptr<ActivationLayer>& layer) CV_OVERRIDE
414 {
415 if ((!activ.empty() && !layer.empty()) || blobs.empty())
416 return false;
417
418 activ = layer;
419 if (activ.empty())
420 reluslope.clear();
421#ifdef HAVE_OPENCL
422 newActiv = true;
423 activType = OCL4DNN_CONV_FUSED_ACTIV_NONE;
424
425 if (IS_DNN_OPENCL_TARGET(preferableTarget))
426 {
427 Ptr<PowerLayer> activ_power = activ.dynamicCast<PowerLayer>();
428 if (!activ_power.empty())
429 {
430 if (activ_power->scale != 1.0f) // not supported well by implementation, #17964
431 {
432 // FIXIT no way to check number of blobs (like, eltwise input)
433 CV_LOG_DEBUG(NULL, "DNN/OpenCL: can't configure Power activation (scale != 1.0f)");
434 activ.release();
435 newActiv = false;
436 return false;
437 }
438 if (activ_power->scale != 1.f || activ_power->shift != 0.f)
439 {
440 const int outCh = blobs[0].size[0];
441 fuseWeights(Mat(1, outCh, CV_32F, Scalar(activ_power->scale)),
442 Mat(1, outCh, CV_32F, Scalar(activ_power->shift)));
443 }
444
445 power = activ_power->power;
446 activType = OCL4DNN_CONV_FUSED_ACTIV_POWER;
447 }
448 Ptr<TanHLayer> activ_tanh = activ.dynamicCast<TanHLayer>();
449 if (!activ_tanh.empty())
450 {
451 activType = OCL4DNN_CONV_FUSED_ACTIV_TANH;
452 }
453 }
454#endif
455
456#ifdef HAVE_CUDA
457 if (activ.empty())
458 {
459 /* setActivation was called with empty argument => reset all fusions */
460 cudaFusionMode = cuda4dnn::ConvolutionConfiguration::FusionMode::NONE;
461 cudaActType = cuda4dnn::ConvolutionConfiguration::ActivationType::IDENTITY;
462 }
463
464 if(IS_DNN_CUDA_TARGET(preferableTarget))
465 {
466 CV_Assert(cudaFusionMode == ConvolutionConfiguration::FusionMode::NONE ||
467 cudaFusionMode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM);
468
469 Ptr<ReLULayer> activ_relu = activ.dynamicCast<ReLULayer>();
470 if(!activ_relu.empty())
471 {
472 cudaActType = cuda4dnn::ConvolutionConfiguration::ActivationType::RELU;
473 cuda_relu_slope = activ_relu->negativeSlope;
474 }
475
476 Ptr<ReLU6Layer> activ_relu6 = activ.dynamicCast<ReLU6Layer>();
477 if(!activ_relu6.empty())
478 {
479 cudaActType = cuda4dnn::ConvolutionConfiguration::ActivationType::CLIPPED_RELU;
480 cuda_crelu_floor = activ_relu6->minValue;
481 cuda_crelu_ceil = activ_relu6->maxValue;
482 }
483
484 Ptr<PowerLayer> activ_power = activ.dynamicCast<PowerLayer>();
485 if (!activ_power.empty())
486 {
487 cuda_power_scale = activ_power->scale;
488 cuda_power_shift = activ_power->shift;
489 cuda_power_exp = activ_power->power;
490 cudaActType = cuda4dnn::ConvolutionConfiguration::ActivationType::POWER;
491 }
492
493 Ptr<TanHLayer> activ_tanh = activ.dynamicCast<TanHLayer>();
494 if(!activ_tanh.empty())
495 cudaActType = cuda4dnn::ConvolutionConfiguration::ActivationType::TANH;
496
497 Ptr<SigmoidLayer> activ_sigmoid = activ.dynamicCast<SigmoidLayer>();
498 if(!activ_sigmoid.empty())
499 cudaActType = cuda4dnn::ConvolutionConfiguration::ActivationType::SIGMOID;
500
501 Ptr<SwishLayer> activ_swish = activ.dynamicCast<SwishLayer>();
502 if(!activ_swish.empty())
503 cudaActType = cuda4dnn::ConvolutionConfiguration::ActivationType::SWISH;
504
505 Ptr<MishLayer> activ_mish = activ.dynamicCast<MishLayer>();
506 if(!activ_mish.empty())
507 cudaActType = cuda4dnn::ConvolutionConfiguration::ActivationType::MISH;
508
509 if (cudaActType == cuda4dnn::ConvolutionConfiguration::ActivationType::IDENTITY)
510 {
511 /* no activation fused */
512 activ.reset();
513 }
514 else
515 {
516 /* activation was fused */
517 if (cudaFusionMode == ConvolutionConfiguration::FusionMode::NONE) /* no previous fusion */
518 cudaFusionMode = ConvolutionConfiguration::FusionMode::ACTIVATION; /* now activation */
519 else if (cudaFusionMode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM) /* previously eltwise was fused */
520 cudaFusionMode = ConvolutionConfiguration::FusionMode::ELTWISE_SUM_THEN_ACTIVATION; /* now activation on eltwise output */
521 }
522 }
523#endif
524 fusedActivation = !activ.empty();
525 return fusedActivation;
526 }

528 virtual bool tryFuse(Ptr<Layer>& top) CV_OVERRIDE
529 {
530 if (fusedAdd) // If the Conv layer has fused Add layer, it cannot fuse other layers.
531 return false;
532
533#ifdef HAVE_CUDA

537 Ptr<NaryEltwiseLayer> naryEltwise = top.dynamicCast<NaryEltwiseLayer>();

539 {

541 * it's difficult to verify it here but we hope that `fuseLayers` has done the check already
542 */

544 {
545 /* no previous fusion */
546 cudaFusionMode = ConvolutionConfiguration::FusionMode::ELTWISE_SUM; /* now eltwise */
547 return true;
548 }
549 else if(cudaFusionMode == ConvolutionConfiguration::FusionMode::ACTIVATION)
550 {
551 /* previously an activation was fused */
552 cudaFusionMode = ConvolutionConfiguration::FusionMode::ACTIVATION_THEN_ELTWISE_SUM;
553 return true;
554 }
555 return false;
556 }
557 }
558#endif
559 return BaseConvolutionLayerImpl::tryFuse(top);

562 void fuseWeights(const Mat& w_, const Mat& b_) CV_OVERRIDE
563 {
564 // Convolution weights have OIHW data layout. Parameters fusion in case of
565 // (conv(I) + b1 ) * w + b2
566 // means to replace convolution's weights to [w*conv(I)] and bias to [b1 * w + b2]
567 const int outCn = weightsMat.size[0];
568 Mat w = w_.total() == 1 ? Mat(1, outCn, CV_32F, Scalar(w_.at<float>(0))) : w_;
569 Mat b = b_.total() == 1 ? Mat(1, outCn, CV_32F, Scalar(b_.at<float>(0))) : b_;
570 CV_Assert_N(!weightsMat.empty(), biasvec.size() == outCn + 2,
571 w.empty() || outCn == w.total(), b.empty() || outCn == b.total());
572
573 if (!w.empty())
574 {
575 // Keep origin weights unchanged.
576 if (weightsMat.data == blobs[0].data)
577 weightsMat = weightsMat.clone();
578
579 Mat originWeights = blobs[0].reshape(1, outCn);
580 for (int i = 0; i < outCn; ++i)
581 {
582 double wi = w.at<float>(i);
583 weightsMultipliers[i] *= wi;
584 cv::multiply(originWeights.row(i), weightsMultipliers[i], weightsMat.row(i));
585 biasvec[i] *= wi;
586 }
587 }
588
589 if (!b.empty())
590 {
591 for (int i = 0; i < outCn; ++i)
592 biasvec[i] += b.at<float>(i);
593 }
594 biasvec[outCn] = biasvec[outCn+1] = biasvec[outCn-1];
595 }

597 virtual Ptr<BackendNode> initVkCom(const std::vector<Ptr<BackendWrapper> > &inputs, std::vector<Ptr<BackendWrapper> > &outputs) CV_OVERRIDE
598 {
599#ifdef HAVE_VULKAN
600 int activationType = transFusedActivType(activ);
601
602 CV_Assert(inputs.size() == 1 && outputs.size() == 1);
603 Ptr<VkComBackendWrapper> inputWrap = inputs[0].dynamicCast<VkComBackendWrapper>();
604 Ptr<VkComBackendWrapper> outputWrap = outputs[0].dynamicCast<VkComBackendWrapper>();
605 CV_Assert(inputWrap && outputWrap);
606
607 MatShape inpShape = shape(*inputWrap->getMat());
608 MatShape outShape = shape(*outputWrap->getMat());
609
610 CV_Assert(inpShape.size() == 4 && inpShape.size() == outShape.size());
611
612 if (activationType == -1)
613 {
614 CV_LOG_WARNING(NULL, "Unsupported fused Active type in Conv layer!!!");
615 return Ptr<BackendNode>();
616 }
617
618 const int inpGroupCn = blobs[0].size[1];
619 int ngroups = inpShape[1] / inpGroupCn;
620 CV_Assert(outShape[1] % ngroups == 0);
621 if (ngroups != 1)
622 return Ptr<BackendNode>();
623
624 Mat weightVK;
625 if (fusedWeights)
626 {
627 weightsMat.copyTo(weightVK); // to handle the case of isContinuous() == false
628 weightVK = weightVK.reshape(1, blobs[0].size);
629 }
630 else
631 weightVK = blobs[0];
632
633 CV_Assert(weightVK.isContinuous());
634 CV_Assert(pads_begin.size() == 2);
635 CV_Assert(fusedAdd == false && "Vulkan Backend can not support the Conv_Add optimization.");
636 Ptr<vkcom::OpBase> op(new vkcom::OpConv(weightVK, biasvec, activationType, ngroups, outShape[1], inpShape[1],

684 }

692 op->set_attr_groups(groups);

714 op->set_input_bias(*(op_const_bias->getOp()));
