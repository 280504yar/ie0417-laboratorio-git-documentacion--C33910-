51 bool getMemoryShapes(const std::vector<MatShape> &inputs,
52 const int requiredOutputs,
53 std::vector<MatShape> &outputs,
54 std::vector<MatShape> &internals) const CV_OVERRIDE {
55 const auto &input = inputs[0];
56 const auto &scale = inputs[1];
57 const auto &bias = inputs[2];
58 CV_CheckGE(input.size(), static_cast<size_t>(3), "DNN/InstanceNorm: input dimension >= 3 is required");
59 if (input.layout == DATA_LAYOUT_BLOCK)
60 CV_CheckEQ(input.dims, 5, "DNN/InstanceNorm: only 5D block layout is supported");
61
62 int C = input.layout == DATA_LAYOUT_BLOCK ? input.C : input[1];
63 int scale_dim = std::accumulate(scale.begin(), scale.end(), 1, std::multiplies<int>());
64 CV_CheckEQ(scale_dim, C, "DNN/InstanceNorm: scale must be a 1d tensor and match the channel of input");
65 int bias_dim = std::accumulate(bias.begin(), bias.end(), 1, std::multiplies<int>());
66 CV_CheckEQ(bias_dim, C, "DNN/InstanceNorm: bias must be a 1d tensor and match the channel of input");
67
68 outputs.assign(1, inputs[0]);
69 return false;
70 }

82 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE {
83 CV_TRACE_FUNCTION();
84 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
85
86 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
87 forward_ocl(inputs_arr, outputs_arr, internals_arr))
88
89 if (inputs_arr.depth() == CV_16F)
90 {
91 forward_fallback(inputs_arr, outputs_arr, internals_arr);
92 return;
93 }

95 std::vector<Mat> inputs, outputs;
96 inputs_arr.getMatVector(inputs);
97 outputs_arr.getMatVector(outputs);
98
99 const auto &input = inputs[0];
100 const auto &scale = inputs[1];
101 const auto &bias = inputs[2];
102
103 fastNormChannel(input, scale, bias, outputs[0], epsilon);
104 }

106#ifdef HAVE_OPENCL
107 bool forward_ocl(InputArrayOfArrays inputs_, OutputArrayOfArrays outputs_, OutputArrayOfArrays internals_) {
108 std::vector<UMat> inputs;
109 std::vector<UMat> outputs;
110
111 inputs_.getUMatVector(inputs);
112 outputs_.getUMatVector(outputs);
113
114 const auto &input = inputs[0], &scale = inputs[1], &bias = inputs[2];
115 auto &output = outputs[0];
116
117 const auto input_shape = shape(input);
118 size_t N = input_shape[0], C = input_shape[1],
119 loops = N * C, norm_size = static_cast<size_t>(total(input_shape, 2));
120 float inv_norm_size = 1.f / norm_size;
121
122 // no fp16 support
123 if (input.depth() == CV_16F) {
124 return false;
125 }
126
127 String base_opts = format(" -DT=float -DT4=float4 -Dconvert_T=convert_float4");
128
129 // Calculate mean
130 UMat one = UMat::ones(norm_size, 1, CV_32F);
131 UMat mean = UMat(loops, 1, CV_32F);
132 UMat mean_square = UMat(loops, 1, CV_32F);
133 UMat tmp = UMat(loops, norm_size, CV_32F);
134 bool ret = ocl4dnn::ocl4dnnGEMV<float>(ocl4dnn::CblasNoTrans, loops, norm_size, inv_norm_size,
135 input, 0, one, 0, 0.f, mean, 0);
136 if (!ret) {
137 return false;
138 }
139 // Calculate mean_square
140 int num_vector = (norm_size % 8 == 0) ? 8 : ((norm_size % 4 == 0) ? 4 : 1);
141 size_t global[] = {loops, static_cast<size_t>(norm_size / num_vector)};
142 String build_opt = format(" -DNUM=%d", num_vector) + base_opts;
143 String mean_square_kernel_name = format("calc_mean%d", num_vector);
144 ocl::Kernel mean_square_kernel(mean_square_kernel_name.c_str(), ocl::dnn::mvn_oclsrc, build_opt + " -DKERNEL_MEAN");
145 if (mean_square_kernel.empty()) {
146 return false;
147 }
148 mean_square_kernel.set(0, ocl::KernelArg::PtrReadOnly(input));
149 mean_square_kernel.set(1, (int)loops);
150 mean_square_kernel.set(2, (int)norm_size);
151 mean_square_kernel.set(3, ocl::KernelArg::PtrReadOnly(mean));
152 mean_square_kernel.set(4, ocl::KernelArg::PtrWriteOnly(tmp));
153 ret = mean_square_kernel.run(2, global, NULL, false);
154 if (!ret) {
155 return false;
156 }
157 ret = ocl4dnn::ocl4dnnGEMV<float>(ocl4dnn::CblasNoTrans, loops, norm_size, inv_norm_size,
158 tmp, 0, one, 0, 0.f, mean_square, 0);
159 if (!ret) {
160 return false;
161 }
162 // Calculate instance norm: output = scale * (x - mean) / sqrt(var + eps) + bias
163 String mvn_kernel_name = format("mvn%d", num_vector);
164 build_opt += " -DNORM_VARIANCE -DFUSE_BATCH_NORM -DKERNEL_MVN";
165 ocl::Kernel mvn_kernel(mvn_kernel_name.c_str(), ocl::dnn::mvn_oclsrc, build_opt);
166 if (mvn_kernel.empty()) {
167 return false;
168 }
169 mvn_kernel.set(0, ocl::KernelArg::PtrReadOnly(input));
170 mvn_kernel.set(1, (int)loops);
171 mvn_kernel.set(2, (int)norm_size);
172 mvn_kernel.set(3, (float)epsilon);
173 mvn_kernel.set(4, ocl::KernelArg::PtrReadOnly(mean));
174 mvn_kernel.set(5, ocl::KernelArg::PtrReadOnly(mean_square));
175 mvn_kernel.set(6, ocl::KernelArg::PtrReadOnly(scale));
176 mvn_kernel.set(7, ocl::KernelArg::PtrReadOnly(bias));
177 mvn_kernel.set(8, (int)C);
178 mvn_kernel.set(9, (float)0.f);
179 mvn_kernel.set(10, ocl::KernelArg::PtrWriteOnly(output));
180 ret = mvn_kernel.run(2, global, NULL, false);
181 if (!ret) {
182 return false;
183 }
184
185 return true;
186 }
187#endif

190 virtual Ptr<BackendNode> initCann(const std::vector<Ptr<BackendWrapper> > &inputs,
191 const std::vector<Ptr<BackendWrapper> > &outputs,
192 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE {
193 auto input_tensor_wrapper = inputs[0].dynamicCast<CannBackendWrapper>();
194 auto input_tensor_desc = input_tensor_wrapper->getTensorDesc();
195
196 auto scale_tensor_wrapper = inputs[1].dynamicCast<CannBackendWrapper>();
197 auto scale_tensor_desc = scale_tensor_wrapper->getTensorDesc();
198
199 auto bias_tensor_wrapper = inputs[2].dynamicCast<CannBackendWrapper>();
200 auto bias_tensor_desc = bias_tensor_wrapper->getTensorDesc();
201
202 auto last_node = nodes[0].dynamicCast<CannBackendNode>()->getOp();
203 auto scale_node = nodes[1].dynamicCast<CannBackendNode>()->getOp();
204 auto bias_node = nodes[2].dynamicCast<CannBackendNode>()->getOp();
205
206 auto op = std::make_shared<ge::op::InstanceNorm>(name);
207
208 // set attrs
209 op->set_attr_epsilon(epsilon);
210
211 // set inputs
212 // set inputs : x
213 op->set_input_x_by_name(*last_node, input_tensor_wrapper->name.c_str());
214 op->update_input_desc_x(*input_tensor_desc);
215 // set inputs : gamma
216 op->set_input_gamma_by_name((*scale_node), scale_tensor_wrapper->name.c_str());
217 op->update_input_desc_gamma(*scale_tensor_desc);
218 // set inputs : beta
219 op->set_input_beta_by_name(*bias_node, bias_tensor_wrapper->name.c_str());
220 op->update_input_desc_beta(*bias_tensor_desc);
221
222 // set outputs
223 auto output_desc_y = std::make_shared<ge::TensorDesc>(ge::Shape(), ge::FORMAT_NCHW, ge::DT_FLOAT);
224 op->update_output_desc_y(*output_desc_y);
225 auto output_desc_mean = std::make_shared<ge::TensorDesc>(ge::Shape(), ge::FORMAT_NCHW, ge::DT_FLOAT);
226 op->update_output_desc_mean(*output_desc_mean);
227 auto output_desc_var = std::make_shared<ge::TensorDesc>(ge::Shape(), ge::FORMAT_NCHW, ge::DT_FLOAT);
228 op->update_output_desc_variance(*output_desc_var);
229
230 return Ptr<BackendNode>(new CannBackendNode(op));

234#ifdef HAVE_DNN_NGRAPH
235 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> >& inputs,
236 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE {
237 // onnx to openvino conversion: https://github.com/openvinotoolkit/openvino/blob/2023.1.0/src/frontends/onnx/frontend/src/op/instance_norm.cpp
238
239 auto ieInpNode = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
240 const auto &input_shape = ieInpNode.get_shape();
241 std::shared_ptr<ov::Node> mvn, result;
242
243 // mvn
244 // https://docs.openvino.ai/2023.1/openvino_docs_ops_normalization_MVN_6.html
245 std::vector<int64_t> axes_v(input_shape.size() - 2);
246 std::iota(axes_v.begin(), axes_v.end(), 2); // {2, 3, ...} for nd input tensor, n>=3
247 auto axes = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{axes_v.size()}, axes_v.data());
248 bool normalize_variance = true;
249 mvn = std::make_shared<ov::op::v6::MVN>(ieInpNode, axes, normalize_variance, epsilon, ov::op::MVNEpsMode::INSIDE_SQRT);
250
251 // instance norm = scale * mvn + bias
252 auto scale = nodes[1].dynamicCast<InfEngineNgraphNode>()->node;
253 std::vector<int64_t> shared_shape_v(input_shape.size(), 1);
254 shared_shape_v[1] = -1;
255 auto shared_shape = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{shared_shape_v.size()}, shared_shape_v.data());
256 scale = std::make_shared<ov::op::v1::Reshape>(scale, shared_shape, true);
257 result = std::make_shared<ov::op::v1::Multiply>(mvn, scale);
258 auto bias = nodes[2].dynamicCast<InfEngineNgraphNode>()->node;
259 bias = std::make_shared<ov::op::v1::Reshape>(bias, shared_shape, true);
260 result = std::make_shared<ov::op::v1::Add>(result, bias);
261
262 return Ptr<BackendNode>(new InfEngineNgraphNode(result));
263 }
264#endif // HAVE_DNN_NGRAPH
