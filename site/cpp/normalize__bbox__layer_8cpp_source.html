59 NormalizeBBoxLayerImpl(const LayerParams& params)
60 {
61 setParamsFrom(params);
62 pnorm = params.get<float>("p", 2);
63 epsilon = params.get<float>("eps", 1e-10f);
64 acrossSpatial = params.get<bool>("across_spatial", true);
65
66 if (!params.has("start_axis") && !params.has("across_spatial"))
67 {
68 int axis = params.get<int>("axis", -1);
69 startAxis = axis;
70 endAxis = axis;
71 }
72 else
73 {
74 startAxis = params.get<int>("start_axis", 1);
75 CV_Assert(!params.has("across_spatial") || !params.has("end_axis"));
76 endAxis = params.get<int>("end_axis", acrossSpatial ? -1 : startAxis);
77 }
78 CV_Assert(pnorm > 0);
79 }

109 {

118#ifdef HAVE_OPENCL
119 bool forward_ocl(InputArrayOfArrays inputs_, OutputArrayOfArrays outputs_, OutputArrayOfArrays internals_)
120 {
121 std::vector<UMat> inputs;
122 std::vector<UMat> outputs;
123 std::vector<UMat> internals;
124
125 if (inputs_.depth() == CV_16F)
126 return false;
127
128 inputs_.getUMatVector(inputs);
129 outputs_.getUMatVector(outputs);
130 internals_.getUMatVector(internals);
131
132 CV_Assert(inputs.size() == 1 && outputs.size() == 1);
133 CV_Assert(inputs[0].total() == outputs[0].total());
134
135 const UMat& inp0 = inputs[0];
136 UMat& buffer = internals[0];
137 startAxis = normalize_axis(startAxis, inp0.dims);
138 endAxis = normalize_axis(endAxis, inp0.dims);
139
140 size_t num = total(inp0.shape(), 0, startAxis);
141 size_t numPlanes = total(inp0.shape(), startAxis, endAxis + 1);
142 size_t planeSize = inp0.total() / (num * numPlanes);
143 MatShape s = shape(1, inputs[0].total());
144 UMat inp = inputs[0].reshape(1, s.size(), &s[0]).reshape(1, num);
145 UMat out = outputs[0].reshape(1, s.size(), &s[0]).reshape(1, num);
146 for (size_t i = 0; i < num; ++i)
147 {
148 s = shape(numPlanes, planeSize);
149 UMat src = inp.row(i).reshape(1, s.size(), &s[0]);
150 UMat dst = out.row(i).reshape(1, s.size(), &s[0]);
151
152 UMat abs_mat;
153 absdiff(src, cv::Scalar::all(0), abs_mat);
154 pow(abs_mat, pnorm, buffer);
155
156 if (planeSize == 1)
157 {
158 // add eps to avoid overflow
159 float absSum = sum(buffer)[0] + epsilon;
160 float norm = std::pow(absSum, 1.0f / pnorm);
161 multiply(src, 1.0f / norm, dst);
162 }
163 else
164 {
165 Mat norm;
166 reduce(buffer, norm, 0, REDUCE_SUM);
167 norm += epsilon;
168
169 // compute inverted norm to call multiply instead divide
170 cv::pow(norm, -1.0f / pnorm, norm);
171
172 repeat(norm, numPlanes, 1, buffer);
173 multiply(src, buffer, dst);
174 }
175
176 if (!blobs.empty())
177 {
178 // scale the output
179 Mat scale = blobs[0];
180 if (scale.total() == 1)
181 {
182 // _scale: 1 x 1
183 multiply(dst, scale.at<float>(0, 0), dst);
184 }
185 else
186 {
187 // _scale: _channels x 1
188 CV_Assert(scale.total() == numPlanes);
189 repeat(scale, 1, dst.cols, buffer);
190 multiply(dst, buffer, dst);
191 }
192 }
193 }
194 return true;
195 }
196#endif

198 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
199 {
200 CV_TRACE_FUNCTION();
201 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
202
203 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
204 forward_ocl(inputs_arr, outputs_arr, internals_arr))
205
206 if (inputs_arr.depth() == CV_16F)
207 {
208 forward_fallback(inputs_arr, outputs_arr, internals_arr);
209 return;
210 }
211
212 std::vector<Mat> inputs, outputs, internals;
213 inputs_arr.getMatVector(inputs);
214 outputs_arr.getMatVector(outputs);
215 internals_arr.getMatVector(internals);
216
217 CV_Assert(inputs.size() == 1 && outputs.size() == 1);
218 CV_Assert(inputs[0].total() == outputs[0].total());
219
220 const Mat& inp0 = inputs[0];
221 Mat& buffer = internals[0];
222 startAxis = normalize_axis(startAxis, inp0.dims);
223 endAxis = normalize_axis(endAxis, inp0.dims);
224
225 const float* inpData = inp0.ptr<float>();
226 float* outData = outputs[0].ptr<float>();
227
228 size_t num = total(inp0, 0, startAxis);
229 size_t numPlanes = total(inp0, startAxis, endAxis + 1);
230 CV_Assert(num * numPlanes != 0);
231 size_t planeSize = inp0.total() / (num * numPlanes);
232 for (size_t n = 0; n < num; ++n)
233 {
234 Mat src = Mat(numPlanes, planeSize, CV_32F, (void*)inpData);
235 Mat dst = Mat(numPlanes, planeSize, CV_32F, (void*)outData);
236 cv::pow(abs(src), pnorm, buffer);
237
238 if (planeSize == 1)
239 {
240 // add eps to avoid overflow
241 float absSum = sum(buffer)[0] + epsilon;
242 float norm = std::pow(absSum, 1.0f / pnorm);
243 multiply(src, 1.0f / norm, dst);
244 }
245 else
246 {
247 Mat norm;
248 reduce(buffer, norm, 0, REDUCE_SUM);
249 norm += epsilon;
250
251 // compute inverted norm to call multiply instead divide
252 cv::pow(norm, -1.0f / pnorm, norm);
253
254 repeat(norm, numPlanes, 1, buffer);
255 multiply(src, buffer, dst);
256 }
257
258 if (!blobs.empty())
259 {
260 // scale the output
261 Mat scale = blobs[0];
262 if (scale.total() == 1)
263 {
264 // _scale: 1 x 1
265 dst *= scale.at<float>(0, 0);
266 }
267 else
268 {
269 // _scale: _channels x 1
270 CV_Assert(scale.total() == numPlanes);
271 scale = scale.reshape(1, (int)scale.total());
272 repeat(scale, 1, dst.cols, buffer);
273 multiply(dst, buffer, dst);
274 }
275 }
276 inpData += numPlanes * planeSize;
277 outData += numPlanes * planeSize;
278 }
279 }

283 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> >& inputs,
284 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
285 {
286 auto& ieInpNode = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
287 const size_t batch = ieInpNode.get_shape()[0];
288 const size_t numChannels = ieInpNode.get_shape()[1];
289
290 std::vector<int64_t> axes_data;
291 if (!acrossSpatial) {
292 axes_data.push_back(1);
293 } else {
294 axes_data.resize(ieInpNode.get_shape().size() - 1);
295 std::iota(axes_data.begin(), axes_data.end(), 1);
296 }
297 auto axes = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{axes_data.size()}, axes_data);
298 auto norm = std::make_shared<ov::op::v0::NormalizeL2>(ieInpNode, axes, epsilon, ov::op::EpsMode::ADD);
299
300 CV_Assert(blobs.empty() || numChannels == blobs[0].total());
301 std::vector<size_t> shape(ieInpNode.get_shape().size(), 1);
302 shape[0] = blobs.empty() ? 1 : batch;
303 shape[1] = numChannels;
304 if (!blobs.empty())
305 {
306 auto weight = std::make_shared<ov::op::v0::Constant>(
307 ov::element::f32, ov::Shape(shape), blobs[0].data);
308 auto mul = std::make_shared<ov::op::v1::Multiply>(norm, weight, ov::op::AutoBroadcastType::NUMPY);
309 return Ptr<BackendNode>(new InfEngineNgraphNode(mul));
310 }
311 return Ptr<BackendNode>(new InfEngineNgraphNode(norm));
312 }
313#endif // HAVE_DNN_NGRAPH

316#ifdef HAVE_CUDA
317 Ptr<BackendNode> initCUDA(
318 void *context_,
319 const std::vector<Ptr<BackendWrapper>>& inputs,
320 const std::vector<Ptr<BackendWrapper>>& outputs
321 ) override
322 {
323 auto context = reinterpret_cast<csl::CSLContext*>(context_);
324
325 if(pnorm != 1 && pnorm != 2)
326 CV_Error(Error::StsNotImplemented, "Unsupported normalization mode");
327
328 auto input_wrapper = inputs[0].dynamicCast<CUDABackendWrapper>();
329 auto input_shape = input_wrapper->getShape();
330
331 NormalizeConfiguration<float> config;
332 config.input_shape.assign(std::begin(input_shape), std::end(input_shape));
333 config.axis_start = normalize_axis(startAxis, input_shape.size());
334 config.axis_end = normalize_axis(endAxis, input_shape.size()) + 1; /* +1 because NormalizeOp follows [start, end) convention */
335 config.norm = pnorm;
336 config.eps = epsilon;
337
338 const auto& weightsMat = blobs.empty() ? Mat() : blobs[0];
339 return make_cuda_node<cuda4dnn::NormalizeOp>(preferableTarget, std::move(context->stream), weightsMat, config);
340 }
341#endif
