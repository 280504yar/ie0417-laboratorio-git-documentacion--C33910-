49void checkImageDimensions(const std::vector<Mat>& images)
50{
51 CV_Assert(!images.empty());
52 int width = images[0].cols;
53 int height = images[0].rows;
54 int type = images[0].type();
55
56 for(size_t i = 0; i < images.size(); i++) {
57 CV_Assert(images[i].cols == width && images[i].rows == height);
58 CV_Assert(images[i].type() == type);
59 }
60}

62Mat triangleWeights(int length)
63{
64 // hat function
65 CV_Assert(length >= 2);
66 Mat w(length, 1, CV_32F);
67 int half = length / 2;
68 int maxVal = length - 1;
69 float epsilon = 1e-6f;
70 w.at<float>(0) = epsilon;
71 w.at<float>(length - 1) = epsilon;
72 for (int i = 1; i < length - 1; i++){
73 w.at<float>(i) = (i < half)
74 ? static_cast<float>(i)
75 : static_cast<float>(maxVal - i);
76 }
77 return w;
78}

80Mat RobertsonWeights(int length)
81{
82 CV_Assert(length >= 1);
83 Mat weight(length, 1, CV_32FC3);
84 float q = (length - 1) / 4.0f;
85 float e4 = exp(4.f);
86 float scale = e4/(e4 - 1.f);
87 float shift = 1 / (1.f - e4);
88
89 for(int i = 0; i < length; i++) {
90 float value = i / q - 2.0f;
91 value = scale*exp(-value * value) + shift;
92 weight.at<Vec3f>(i) = Vec3f::all(value);
93 }
94 return weight;
95}

97void mapLuminance(Mat src, Mat dst, Mat lum, Mat new_lum, float saturation)
98{
99 std::vector<Mat> channels(3);
100 split(src, channels);
101 for(int i = 0; i < 3; i++) {
102 channels[i] = channels[i].mul(1.0f / lum);
103 pow(channels[i], saturation, channels[i]);
104 channels[i] = channels[i].mul(new_lum);
105 }
106 merge(channels, dst);
107}

109Mat linearResponse(int channels, int length)
110{
111 CV_Assert(length >= 1);
112 Mat response = Mat(length, 1, CV_MAKETYPE(CV_32F, channels));
113
114 if (channels == 1)
115 {
116 for (int i = 0; i < length; i++)
117 {
118 response.at<float>(i) = static_cast<float>(i);
119 }
120 }
121 else if (channels == 3)
122 {
123 for (int i = 0; i < length; i++)
124 {
125 response.at<Vec3f>(i) = Vec3f::all(static_cast<float>(i));
126 }
127 }
128 else
129 {
130 CV_Error(Error::StsBadArg, "Unsupported number of channels in linearResponse");
131 }
132 return response;
133}
