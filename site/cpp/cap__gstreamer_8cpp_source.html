257 isFailed = true;
258 return;
259 }
260

460{

474 {
475 CV_LOG_ERROR(NULL, "VIDEOIO/GStreamer: Invalid usage of CAP_PROP_HW_DEVICE with 'ANY' H/W acceleration. Bailout");

477 }
478 if (hw_device != -1)
479 {
480 CV_LOG_ERROR(NULL, "VIDEOIO/GStreamer: CAP_PROP_HW_DEVICE is not supported. Specify -1 (auto) value. Bailout");
481 return false;
482 }
483 }
484 return true;

555bool GStreamerCapture::grabFrame()
556{
557 if (!pipeline || !GST_IS_ELEMENT(pipeline.get()))
558 return false;
559
560 // start the pipeline if it was not in playing state yet
561 if (!this->isPipelinePlaying())
562 this->startPipeline();
563 // bail out if EOS
564
565 if (vEOS)
566 {
567 return false;
568 }
569
570 bool returnFlag = true;
571 if (videoStream >= 0)
572 {
573 if (!vEOS)
574 returnFlag &= grabVideoFrame();
575 if (!returnFlag)
576 {
577 return false;
578 }
579 }
580
581 if (audioStream >= 0)
582 {
583 bufferedAudioDurationNS = (gint64)(1e9*(((double)bufferAudioData.size()/((audioBitPerSample/8)*nAudioChannels))/audioSamplesPerSecond));
584 audioFrame.release();
585 if (!aEOS)
586 returnFlag &= grabAudioFrame();
587 }
588
589 return returnFlag;
590}

592bool GStreamerCapture::grabVideoFrame()
593{
594 usedVideoSample.release();
595
596 bool returnFlag = false;
597 bool stopFlag = false;
598
599 if (audioStream != -1)
600 {
601 usedVideoSample.swap(impendingVideoSample);
602 std::swap(usedVideoSampleTimeNS, impendingVideoSampleTimeNS);
603 }
604
605 while (!stopFlag)
606 {
607 if (gst_app_sink_is_eos(GST_APP_SINK(sink.get())))
608 {
609 vEOS = true;
610 lastFrame = true;
611 stopFlag = true;
612 if (audioStream == -1)
613 {
614 returnFlag = false;
615 }
616 else if (usedVideoSample)
617 {
618 gst_element_query_position(sink.get(), CV_GST_FORMAT(GST_FORMAT_TIME), &impendingVideoSampleTimeNS);
619 videoSampleDurationNS = impendingVideoSampleTimeNS - usedVideoSampleTimeNS;
620 requiredAudioTimeNS = impendingVideoSampleTimeNS - givenAudioTimeNS;
621 givenAudioTimeNS += requiredAudioTimeNS;
622 returnFlag = true;
623 }
624 }
625 else
626 {
627 #if FULL_GST_VERSION >= VERSION_NUM(1,10,0)
628 impendingVideoSample.attach(gst_app_sink_try_pull_sample(GST_APP_SINK(sink.get()), readTimeout));
629 #else
630 impendingVideoSample.attach(gst_app_sink_pull_sample(GST_APP_SINK(sink.get())));
631 #endif
632 if (!impendingVideoSample)
633 {
634 if (!gst_app_sink_is_eos(GST_APP_SINK(sink.get())))
635 {
636 CV_LOG_DEBUG(NULL, "videoio(Gstreamer): gst_app_sink_pull_sample() method is not succeeded");
637 }
638 else
639 {
640 vEOS = true;
641 lastFrame = true;
642 stopFlag = true;
643 if (audioStream == -1)
644 {
645 return false;
646 }
647 else if (usedVideoSample)
648 {
649 gst_element_query_position(sink.get(), CV_GST_FORMAT(GST_FORMAT_TIME), &impendingVideoSampleTimeNS);
650 videoSampleDurationNS = impendingVideoSampleTimeNS - usedVideoSampleTimeNS;
651 requiredAudioTimeNS = impendingVideoSampleTimeNS - givenAudioTimeNS;
652 givenAudioTimeNS += requiredAudioTimeNS;
653 returnFlag = true;
654 }
655 }
656 }
657 gst_element_query_position(sink.get(), CV_GST_FORMAT(GST_FORMAT_TIME), &impendingVideoSampleTimeNS);
658
659 if (audioStream != -1)
660 {
661 if (!usedVideoSample)
662 {
663 usedVideoSample.swap(impendingVideoSample);
664 std::swap(usedVideoSampleTimeNS, impendingVideoSampleTimeNS);
665 }
666 else
667 {
668 stopFlag = true;
669 }
670 if (impendingVideoSample)
671 {
672 emulatedFrameNumber++;
673 }
674 videoSampleDurationNS = impendingVideoSampleTimeNS - usedVideoSampleTimeNS;
675 requiredAudioTimeNS = impendingVideoSampleTimeNS - givenAudioTimeNS;
676 givenAudioTimeNS += requiredAudioTimeNS;
677 }
678 else
679 {
680 usedVideoSample.swap(impendingVideoSample);
681 std::swap(usedVideoSampleTimeNS, impendingVideoSampleTimeNS);
682 stopFlag = true;
683 emulatedFrameNumber++;
684 }
685 if (usedVideoSample)
686 {
687 auto *buffer = gst_sample_get_buffer((GstSample*)usedVideoSample);
688 timestamp = GST_BUFFER_PTS(buffer);
689 }
690 returnFlag = true;
691 }
692 }
693 return returnFlag;
694}

696bool GStreamerCapture::grabAudioFrame()
697{
698 audioSample.reset(NULL);
699
700 bool returnFlag = false;
701 gint64 audioTimeNS = bufferedAudioDurationNS;
702
703 if (bufferedAudioDurationNS > requiredAudioTimeNS)
704 {
705 return true;
706 }
707
708 while ((!vEOS) ? audioTimeNS <= requiredAudioTimeNS : !aEOS)
709 {
710 if (gst_app_sink_is_eos(GST_APP_SINK(audiosink.get())))
711 {
712 aEOS = true;
713 if (videoStream != -1 && !vEOS)
714 returnFlag = true;
715 if (videoStream == -1)
716 audioSamplePosInSamples += chunkLengthOfBytes/((audioBitPerSample/8)*nAudioChannels);
717 break;
718 }
719 else
720 {
721 audioSample.attach(gst_app_sink_pull_sample(GST_APP_SINK(audiosink.get())));
722 if (!audioSample)
723 {
724 if (!gst_app_sink_is_eos(GST_APP_SINK(audiosink.get())))
725 {
726 CV_LOG_ERROR(NULL, "videoio(Gstreamer): gst_app_sink_pull_sample() method is not succeeded");
727 return false;
728 }
729 else
730 {
731 aEOS = true;
732 break;
733 }
734 }
735 gst_element_query_position(audiosink.get(), CV_GST_FORMAT(GST_FORMAT_TIME), &audioSampleTimeNS);
736
737 GstMapInfo map_info = {};
738 GstCaps* frame_caps = gst_sample_get_caps(audioSample.get()); // no lifetime transfer
739 if (!frame_caps)
740 {
741 CV_LOG_ERROR(NULL, "GStreamer: gst_sample_get_caps() returns NULL");
742 return false;
743 }
744 if (!GST_CAPS_IS_SIMPLE(frame_caps))
745 {
746 // bail out in no caps
747 CV_LOG_ERROR(NULL, "GStreamer: GST_CAPS_IS_SIMPLE(frame_caps) check is failed");
748 return false;
749 }
750 GstAudioInfo info = {};
751 gboolean audio_info_res = gst_audio_info_from_caps(&info, frame_caps);
752 if (!audio_info_res)
753 {
754 CV_Error(Error::StsError, "GStreamer: gst_audio_info_from_caps() is failed. Can't handle unknown layout");
755 }
756 audioBitPerFrame = GST_AUDIO_INFO_BPF(&info);
757 CV_CheckGT(audioBitPerFrame, 0, "");
758
759 GstStructure* structure = gst_caps_get_structure(frame_caps, 0); // no lifetime transfer
760 if (!structure)
761 {
762 CV_LOG_ERROR(NULL, "GStreamer: Can't query 'structure'-0 from GStreamer sample");
763 return false;
764 }
765
766 const gchar* name_ = gst_structure_get_name(structure);
767 if (!name_)
768 {
769 CV_LOG_ERROR(NULL, "GStreamer: Can't query 'name' from GStreamer sample");
770 return false;
771 }
772 std::string name = toLowerCase(std::string(name_));
773 if (name != "audio/x-raw")
774 {
775 CV_Error_(Error::StsNotImplemented, ("Unsupported GStreamer layer type: %s", name.c_str()));
776 return false;
777 }
778
779 const gchar* format_ = gst_structure_get_string(structure, "format");
780 if (!format_)
781 {
782 CV_LOG_ERROR(NULL, "GStreamer: Can't query 'format' of 'audio/x-raw'");
783 return false;
784 }
785 std::string format = toUpperCase(std::string(format_));
786 if (format != "S8" && format != "S16LE" && format != "S32LE" && format != "F32LE")
787 {
788 CV_Error_(Error::StsNotImplemented, ("Unsupported GStreamer audio format: %s", format.c_str()));
789 return false;
790 }
791 if (format != audioFormat)
792 {
793 return false;
794 }
795
796 GstBuffer* buf = gst_sample_get_buffer(audioSample);
797 if (!buf)
798 {
799 CV_LOG_ERROR(NULL, "GStreamer: Failed. Buffer is empty");
800 return false;
801 }
802 timestamp = GST_BUFFER_PTS(buf);
803 if (!gst_buffer_map(buf, &map_info, GST_MAP_READ))
804 {
805 CV_LOG_ERROR(NULL, "GStreamer: Failed to map GStreamer buffer to system memory");
806 return false;
807 }
808 ScopeGuardGstMapInfo map_guard(buf, &map_info);
809 gsize lastSize = bufferAudioData.size();
810 audioSampleSize = map_info.size;
811 bufferAudioData.resize(lastSize+audioSampleSize);
812 for (gint j = 0; j < audioSampleSize; j++)
813 {
814 bufferAudioData[lastSize+j]=*(map_info.data + j);
815 }
816 audioSampleDurationNS = 1e9*(((double)audioSampleSize/((audioBitPerSample/8)*nAudioChannels))/audioSamplesPerSecond);
817
818 CV_LOG_DEBUG(NULL, "videoio(Gstreamer): got audio frame with timestamp=" << audioSampleTimeNS << " duration=" << audioSampleDurationNS);
819 audioTimeNS += (int64_t)(audioSampleDurationNS);
820
821 returnFlag = true;
822 }
823 }
824
825 returnFlag &= configureAudioFrame();
826 return returnFlag;
827}

829bool GStreamerCapture::configureAudioFrame()
830{
831 if (bufferAudioData.empty() && aEOS)
832 {
833 return false;
834 }
835 std::vector<uint8_t> audioDataInUse;
836
837 audioSamplePosInSamples += chunkLengthOfBytes/((audioBitPerSample/8)*nAudioChannels);
838 chunkLengthOfBytes = (videoStream != -1) ? (int64_t)((requiredAudioTimeNS * 1e-9 * audioSamplesPerSecond*nAudioChannels*(audioBitPerSample)/8)) : audioSampleSize;
839 if ((videoStream != -1) && (chunkLengthOfBytes % ((int)(audioBitPerSample)/8* (int)nAudioChannels) != 0))
840 {
841 if ( (double)audioSamplePosInSamples/audioSamplesPerSecond - usedVideoSampleTimeNS * 1e-9 >= 0 )
842 chunkLengthOfBytes -= numberOfAdditionalAudioBytes;
843 numberOfAdditionalAudioBytes = ((int)(audioBitPerSample)/8* (int)nAudioChannels)
844 - chunkLengthOfBytes % ((int)(audioBitPerSample)/8* (int)nAudioChannels);
845 chunkLengthOfBytes += numberOfAdditionalAudioBytes;
846 }
847 if ((lastFrame && !syncLastFrame) || (aEOS && !vEOS))
848 {
849 chunkLengthOfBytes = bufferAudioData.size();
850 audioSamplePosInSamples += chunkLengthOfBytes/((audioBitPerSample/8)*nAudioChannels);
851 }
852 CV_Check((double)chunkLengthOfBytes, chunkLengthOfBytes >= INT_MIN || chunkLengthOfBytes <= INT_MAX, "GSTREAMER: The chunkLengthOfBytes is out of the allowed range");
853 copy(bufferAudioData.begin(), bufferAudioData.begin() + (int)chunkLengthOfBytes, std::back_inserter(audioDataInUse));
854 bufferAudioData.erase(bufferAudioData.begin(), bufferAudioData.begin() + (int)chunkLengthOfBytes);
855
856 cv::Mat data;
857
858 if (audioFormat == "S8")
859 {
860 Mat((int)chunkLengthOfBytes/audioBitPerFrame, nAudioChannels, CV_8S, audioDataInUse.data()).copyTo(audioFrame);
861 return true;
862 }
863 if (audioFormat == "S16LE")
864 {
865 Mat((int)chunkLengthOfBytes/audioBitPerFrame, nAudioChannels, CV_16S, audioDataInUse.data()).copyTo(audioFrame);
866 return true;
867 }
868 if (audioFormat == "S32LE")
869 {
870 Mat((int)chunkLengthOfBytes/audioBitPerFrame, nAudioChannels, CV_32S, audioDataInUse.data()).copyTo(audioFrame);
871 return true;
872 }
873 if (audioFormat == "F32LE")
874 {
875 Mat((int)chunkLengthOfBytes/audioBitPerFrame, nAudioChannels, CV_32F, audioDataInUse.data()).copyTo(audioFrame);
876 return true;
877 }
878
879 audioDataInUse.clear();
880 audioDataInUse.shrink_to_fit();
881 return true;
882}

884bool GStreamerCapture::retrieveAudioFrame(int index, OutputArray dst)
885{
886 if (audioFrame.empty())
887 {
888 dst.release();
889 if (aEOS)
890 return true;
891 }
892 CV_Check(index, index >= audioBaseIndex && index < audioBaseIndex + nAudioChannels, "");
893 index -= audioBaseIndex;
894
895 CV_CheckType(outputAudioFormat,
896 outputAudioFormat == CV_8S ||
897 outputAudioFormat == CV_16S ||
898 outputAudioFormat == CV_32S ||
899 outputAudioFormat == CV_32F,
900 "");
901
902 dst.create(1, audioFrame.rows, outputAudioFormat);
903 Mat data = dst.getMat();
904 switch (outputAudioFormat)
905 {
906 case CV_8S:
907 for (int i = 0; i < audioFrame.rows; i++)
908 data.at<char>(i) = audioFrame.at<char>(i, index);
909 return true;
910 case CV_16S:
911 for (int i = 0; i < audioFrame.rows; i++)
912 data.at<short>(i) = audioFrame.at<short>(i, index);
913 return true;
914 case CV_32S:
915 for (int i = 0; i < audioFrame.rows; i++)
916 data.at<int>(i) = audioFrame.at<int>(i, index);
917 return true;
918 case CV_32F:
919 for (int i = 0; i < audioFrame.rows; i++)
920 data.at<float>(i) = audioFrame.at<float>(i, index);
921 return true;
922 }
923
924 dst.release();
925 return false;
926}

928bool GStreamerCapture::retrieveVideoFrame(int, OutputArray dst)
929{
930 GstCaps* frame_caps = gst_sample_get_caps(usedVideoSample); // no lifetime transfer
931 if (!frame_caps)
932 {
933 CV_LOG_ERROR(NULL, "GStreamer: gst_sample_get_caps() returns NULL");
934 return false;
935 }
936
937 if (!GST_CAPS_IS_SIMPLE(frame_caps))
938 {
939 // bail out in no caps
940 CV_LOG_ERROR(NULL, "GStreamer: GST_CAPS_IS_SIMPLE(frame_caps) check is failed");
941 return false;
942 }
943
944 GstVideoInfo info = {};
945 gboolean video_info_res = gst_video_info_from_caps(&info, frame_caps);
946 if (!video_info_res)
947 {
948 CV_Error(Error::StsError, "GStreamer: gst_video_info_from_caps() is failed. Can't handle unknown layout");
949 }
950
951 // gstreamer expects us to handle the memory at this point
952 // so we can just wrap the raw buffer and be done with it
953 GstBuffer* buf = gst_sample_get_buffer(usedVideoSample); // no lifetime transfer
954 if (!buf)
955 return false;
956
957 // at this point, the gstreamer buffer may contain a video meta with special
958 // stride and plane locations. We __must__ consider in order to correctly parse
959 // the data. The gst_video_frame_map will parse the meta for us, or default to
960 // regular strides/offsets if no meta is present.
961 GstVideoFrame frame = {};
962#if FULL_GST_VERSION >= VERSION_NUM(1,6,0)
963 GstMapFlags flags = static_cast<GstMapFlags>(+GST_MAP_READ | GST_VIDEO_FRAME_MAP_FLAG_NO_REF);
964#else
965 GstMapFlags flags = static_cast<GstMapFlags>(GST_MAP_READ);
966#endif
967
968 if (!gst_video_frame_map(&frame, &info, buf, flags))
969 {
970 CV_LOG_ERROR(NULL, "GStreamer: Failed to map GStreamer buffer to system memory");
971 return false;
972 }
973
974 ScopeGuardGstVideoFrame frame_guard(&frame); // call gst_video_frame_unmap(&frame) on scope leave
975
976 int frame_width = GST_VIDEO_FRAME_COMP_WIDTH(&frame, 0);
977 int frame_height = GST_VIDEO_FRAME_COMP_HEIGHT(&frame, 0);
978 if (frame_width <= 0 || frame_height <= 0)
979 {
980 CV_LOG_ERROR(NULL, "GStreamer: Can't query frame size from GStreamer sample");
981 return false;
982 }
983
984 GstStructure* structure = gst_caps_get_structure(frame_caps, 0); // no lifetime transfer
985 if (!structure)
986 {
987 CV_LOG_ERROR(NULL, "GStreamer: Can't query 'structure'-0 from GStreamer sample");
988 return false;
989 }
990
991 const gchar* name_ = gst_structure_get_name(structure);
992 if (!name_)
993 {
994 CV_LOG_ERROR(NULL, "GStreamer: Can't query 'name' from GStreamer sample");
995 return false;
996 }
997 std::string name = toLowerCase(std::string(name_));
998
999 // we support these types of data:
1000 // video/x-raw, format=BGR -> 8bit, 3 channels
1001 // video/x-raw, format=GRAY8 -> 8bit, 1 channel
1002 // video/x-raw, format=UYVY -> 8bit, 2 channel
1003 // video/x-raw, format=YUY2 -> 8bit, 2 channel
1004 // video/x-raw, format=YVYU -> 8bit, 2 channel
1005 // video/x-raw, format=NV12 -> 8bit, 1 channel (height is 1.5x larger than true height)
1006 // video/x-raw, format=NV21 -> 8bit, 1 channel (height is 1.5x larger than true height)
1007 // video/x-raw, format=YV12 -> 8bit, 1 channel (height is 1.5x larger than true height)
1008 // video/x-raw, format=I420 -> 8bit, 1 channel (height is 1.5x larger than true height)
1009 // video/x-bayer -> 8bit, 1 channel
1010 // image/jpeg -> 8bit, mjpeg: buffer_size x 1 x 1
1011 // video/x-raw, format=GRAY16_LE (BE) -> 16 bit, 1 channel
1012 // video/x-raw, format={BGRA, RGBA, BGRx, RGBx} -> 8bit, 4 channels
1013 // bayer data is never decoded, the user is responsible for that
1014 Size sz = Size(frame_width, frame_height);
1015 guint n_planes = GST_VIDEO_FRAME_N_PLANES(&frame);
1016
1017 if (name == "video/x-raw")
1018 {
1019 const gchar* format_ = frame.info.finfo->name;
1020 if (!format_)
1021 {
1022 CV_LOG_ERROR(NULL, "GStreamer: Can't query 'format' of 'video/x-raw'");
1023 return false;
1024 }
1025 std::string format = toUpperCase(std::string(format_));
1026
1027 if (format == "BGR")
1028 {
1029 CV_CheckEQ((int)n_planes, 1, "");
1030 size_t step = GST_VIDEO_FRAME_PLANE_STRIDE(&frame, 0);
1031 CV_CheckGE(step, (size_t)frame_width * 3, "");
1032 Mat src(sz, CV_8UC3, GST_VIDEO_FRAME_PLANE_DATA(&frame, 0), step);
1033 src.copyTo(dst);
1034 return true;
1035 }
1036 else if (format == "GRAY8")
1037 {
1038 CV_CheckEQ((int)n_planes, 1, "");
1039 size_t step = GST_VIDEO_FRAME_PLANE_STRIDE(&frame, 0);
1040 CV_CheckGE(step, (size_t)frame_width, "");
1041 Mat src(sz, CV_8UC1, GST_VIDEO_FRAME_PLANE_DATA(&frame, 0), step);
1042 src.copyTo(dst);
1043 return true;
1044 }
1045 else if (format == "GRAY16_LE" || format == "GRAY16_BE")
1046 {
1047 CV_CheckEQ((int)n_planes, 1, "");
1048 size_t step = GST_VIDEO_FRAME_PLANE_STRIDE(&frame, 0);
1049 CV_CheckGE(step, (size_t)frame_width, "");
1050 Mat src(sz, CV_16UC1, GST_VIDEO_FRAME_PLANE_DATA(&frame, 0), step);
1051 src.copyTo(dst);
1052 return true;
1053 }
1054 else if (format == "BGRA" || format == "RGBA" || format == "BGRX" || format == "RGBX")
1055 {
1056 CV_CheckEQ((int)n_planes, 1, "");
1057 size_t step = GST_VIDEO_FRAME_PLANE_STRIDE(&frame, 0);
1058 CV_CheckGE(step, (size_t)frame_width, "");
1059 Mat src(sz, CV_8UC4, GST_VIDEO_FRAME_PLANE_DATA(&frame, 0), step);
1060 src.copyTo(dst);
1061 return true;
1062 }
1063 else if (format == "UYVY" || format == "YUY2" || format == "YVYU")
1064 {
1065 CV_CheckEQ((int)n_planes, 1, "");
1066 size_t step = GST_VIDEO_FRAME_PLANE_STRIDE(&frame, 0);
1067 CV_CheckGE(step, (size_t)frame_width * 2, "");
1068 Mat src(sz, CV_8UC2, GST_VIDEO_FRAME_PLANE_DATA(&frame, 0), step);
1069 src.copyTo(dst);
1070 return true;
1071 }
1072 else if (format == "NV12" || format == "NV21")
1073 {
1074 CV_CheckEQ((int)n_planes, 2, "");
1075 size_t stepY = GST_VIDEO_FRAME_PLANE_STRIDE(&frame, 0);
1076 CV_CheckGE(stepY, (size_t)frame_width, "");
1077 size_t stepUV = GST_VIDEO_FRAME_PLANE_STRIDE(&frame, 1);
1078 CV_CheckGE(stepUV, (size_t)frame_width, "");
1079
1080 dst.create(Size(frame_width, frame_height * 3 / 2), CV_8UC1);
1081 Mat dst_ = dst.getMat();
1082 Mat srcY(sz, CV_8UC1, GST_VIDEO_FRAME_PLANE_DATA(&frame,0), stepY);
1083 Mat srcUV(Size(frame_width, frame_height / 2), CV_8UC1, GST_VIDEO_FRAME_PLANE_DATA(&frame,1), stepUV);
1084 srcY.copyTo(dst_(Rect(0, 0, frame_width, frame_height)));
1085 srcUV.copyTo(dst_(Rect(0, frame_height, frame_width, frame_height / 2)));
1086 return true;
1087 }
1088 else if (format == "YV12" || format == "I420")
1089 {
1090 CV_CheckEQ((int)n_planes, 3, "");
1091 size_t step0 = GST_VIDEO_FRAME_PLANE_STRIDE(&frame, 0);
1092 CV_CheckGE(step0, (size_t)frame_width, "");
1093 size_t step1 = GST_VIDEO_FRAME_PLANE_STRIDE(&frame, 1);
1094 CV_CheckGE(step1, (size_t)frame_width / 2, "");
1095 size_t step2 = GST_VIDEO_FRAME_PLANE_STRIDE(&frame, 2);
1096 CV_CheckGE(step2, (size_t)frame_width / 2, "");
1097
1098 dst.create(Size(frame_width, frame_height * 3 / 2), CV_8UC1);
1099 Mat dst_ = dst.getMat();
1100 Mat srcY(sz, CV_8UC1, GST_VIDEO_FRAME_PLANE_DATA(&frame,0), step0);
1101 Size sz2(frame_width / 2, frame_height / 2);
1102 Mat src1(sz2, CV_8UC1, GST_VIDEO_FRAME_PLANE_DATA(&frame,1), step1);
1103 Mat src2(sz2, CV_8UC1, GST_VIDEO_FRAME_PLANE_DATA(&frame,2), step2);
1104 srcY.copyTo(dst_(Rect(0, 0, frame_width, frame_height)));
1105 src1.copyTo(Mat(sz2, CV_8UC1, dst_.ptr<uchar>(frame_height)));
1106 src2.copyTo(Mat(sz2, CV_8UC1, dst_.ptr<uchar>(frame_height) + src1.total()));
1107 return true;
1108 }
1109 else
1110 {
1111 CV_Error_(Error::StsNotImplemented, ("Unsupported GStreamer 'video/x-raw' format: %s", format.c_str()));
1112 }
1113 }
1114 else if (name == "video/x-bayer")
1115 {
1116 CV_CheckEQ((int)n_planes, 0, "");
1117 Mat src(sz, CV_8UC1, frame.map[0].data);
1118 src.copyTo(dst);
1119 return true;
1120 }
1121 else if (name == "image/jpeg")
1122 {
1123 CV_CheckEQ((int)n_planes, 0, "");
1124 Mat src(Size(frame.map[0].size, 1), CV_8UC1, frame.map[0].data);
1125 src.copyTo(dst);
1126 return true;
1127 }
1128
1129 CV_Error_(Error::StsNotImplemented, ("Unsupported GStreamer layer type: %s", name.c_str()));
1130}

1473 return false;

1513 v4l2src.attach(GST_ELEMENT(gst_object_ref(element)));

1518 }

1520 break;

1522 gst_iterator_resync (it);
1523 break;

1525 case GST_ITERATOR_DONE:

1527 break;

1533 {
1534 CV_WARN("cannot find appsink in manual pipeline");
1535 return false;

1543 CV_Assert(pipeline);

1547 queue.reset(gst_element_factory_make("queue", NULL));
1548 CV_Assert(queue);
1549 sink.reset(gst_element_factory_make("appsink", NULL));
1550 CV_Assert(sink);

1554 CV_Assert(color);

1558 if (element_from_uri)
1559 {
1560 if(!gst_element_link(uridecodebin, color.get()))
1561 {
1562 CV_WARN("GStreamer(video): cannot link color -> sink");
1563 pipeline.release();
1564 return false;
1565 }
1566 }
1567 else
1568 {
1569 g_signal_connect(uridecodebin, "pad-added", G_CALLBACK(newPad), color.get());

1571
1572 if (!gst_element_link(color.get(), sink.get()))
1573 {
1574 CV_WARN("GStreamer(video): cannot link color -> sink");
1575 pipeline.release();
1576 return false;
1577 }
1578 }
1579 if (audioStream >= 0)
1580 {
1581 queue.reset(gst_element_factory_make("queue", NULL));
1582 CV_Assert(queue);
1583 convert.reset(gst_element_factory_make("audioconvert", NULL));
1584 resample.reset(gst_element_factory_make("audioresample", NULL));

1587 CV_Assert(resample);
1588 CV_Assert(audiosink);

1591 if (!gst_element_link_many (convert.get(), resample.get(), audiosink.get(), NULL))
1592 {
1593 CV_WARN("GStreamer(audio): cannot link convert -> resample -> sink");
1594 pipeline.release();
1595 return false;
1596 }
1597 g_signal_connect (uridecodebin, "pad-added", G_CALLBACK (newPad), convert.get());
1598 }
1599 }
1600
1601 if (!manualpipeline || strstr(filename, " max-buffers=") == NULL)
1602 {
1603 //TODO: is 1 single buffer really high enough?
1604 if (videoStream >= 0)
1605 gst_app_sink_set_max_buffers(GST_APP_SINK(sink.get()), 1);
1606 if (audioStream >= 0)
1607 gst_app_sink_set_max_buffers(GST_APP_SINK(audiosink.get()), 1);
1608 }
1609 if (!manualpipeline)
1610 {
1611 if (videoStream >= 0)
1612 gst_base_sink_set_sync(GST_BASE_SINK(sink.get()), FALSE);
1613 if (audioStream >= 0)
1614 gst_base_sink_set_sync(GST_BASE_SINK(audiosink.get()), FALSE);
1615 }
1616 if (videoStream >= 0)
1617 {
1618 //do not emit signals: all calls will be synchronous and blocking
1619 gst_app_sink_set_emit_signals (GST_APP_SINK(sink.get()), FALSE);
1620 caps.attach(gst_caps_from_string("video/x-raw, format=(string){BGR}; video/x-raw, format=(string){BGRx, BGRA}; video/x-bayer,format=(string){rggb,bggr,grbg,gbrg}; video/x-raw, format=(string){GRAY8}; image/jpeg"));
1621 }
1622 if (audioStream >= 0)
1623 {
1624 gst_app_sink_set_emit_signals(GST_APP_SINK(audiosink.get()), FALSE);
1625 switch (outputAudioFormat)
1626 {
1627 case CV_8S:
1628 {
1629 audioBitPerSample = 8;
1630 audioFormat = "S8";
1631 break;
1632 }
1633 case CV_16S:
1634 {
1635 audioBitPerSample = 16;
1636 audioFormat = "S16LE";
1637 break;
1638 }
1639 case CV_32S:
1640 {
1641 audioBitPerSample = 32;
1642 audioFormat = "S32LE";
1643 break;
1644 }
1645 case CV_32F:
1646 {
1647 audioBitPerSample = 32;
1648 audioFormat = "F32LE";
1649 break;
1650 }
1651 default:
1652 audioFormat = "S16LE";
1653 break;
1654 }
1655 std::string stringCaps = "audio/x-raw, format=(string)" + audioFormat + ", rate=(int)" + std::to_string(audioSamplesPerSecond) + ", channels=(int){1, 2}, layout=(string)interleaved";
1656 audiocaps.attach(gst_caps_from_string(stringCaps.c_str()));
1657
1658 gst_app_sink_set_caps(GST_APP_SINK(audiosink.get()), audiocaps);
1659 audiocaps.release();
1660 }
1661 if (manualpipeline)
1662 {
1663 if (videoStream >= 0)
1664 {
1665 GSafePtr<GstCaps> peer_caps;
1666 GSafePtr<GstPad> sink_pad;
1667 sink_pad.attach(gst_element_get_static_pad(sink, "sink"));
1668 peer_caps.attach(gst_pad_peer_query_caps(sink_pad, NULL));

1672 CV_Assert(caps);
1673 }
1674 }
1675 }
1676 if (videoStream >= 0)
1677 {
1678 gst_app_sink_set_caps(GST_APP_SINK(sink.get()), caps);

1686 file ? GST_STATE_PAUSED : GST_STATE_PLAYING);
1687 if (status == GST_STATE_CHANGE_ASYNC)
1688 {
1689 // wait for status update
1690 status = gst_element_get_state(pipeline, NULL, NULL, openTimeout);
1691 }

1693 {

1698 return false;
1699 }

1701 if (videoStream >= 0)
1702 {
1703 GSafePtr<GstPad> pad;
1704 pad.attach(gst_element_get_static_pad(sink, "sink"));
1705
1706 GSafePtr<GstCaps> buffer_caps;
1707 buffer_caps.attach(gst_pad_get_current_caps(pad));

1710 if (!gst_structure_get_int (structure, "width", &width) ||
1711 !gst_structure_get_int (structure, "height", &height))
1712 {
1713 CV_WARN("cannot query video width/height");
1714 }

1717 bool fps_query_success = true;
1718
1719 if (!gst_structure_get_fraction(structure, "framerate", &num, &denom))
1720 {
1721 CV_WARN("cannot query video fps");
1722 fps_query_success = false;
1723 }
1724
1725 fps = (double)num/(double)denom;
1726
1727 // If num == 0 and denom == 1 -> variable frame rate video.
1728 if (fps_query_success && !(num == 0 && denom == 1))
1729 {
1730 GSafePtr<GstQuery> query;
1731 query.attach(gst_query_new_duration(GST_FORMAT_DEFAULT));
1732
1733 gboolean res = gst_element_query(pipeline.get(), query);
1734
1735 if (res)
1736 {
1737 gst_query_parse_duration(query, NULL, &duration);
1738 }
1739 else if (fps != 0)
1740 {
1741 GSafePtr<GstQuery> query2;
1742 query2.attach(gst_query_new_duration(GST_FORMAT_TIME));
1743 gboolean res2 = gst_element_query(pipeline.get(), query2);
1744
1745 if (res2)
1746 {
1747 gst_query_parse_duration(query2, NULL, &duration);
1748 duration = static_cast<gint64>((float)duration / GST_SECOND * fps);
1749 CV_WARN("frame count is estimated by duration and fps");
1750 }
1751 else
1752 {
1753 CV_WARN("unable to query duration of stream");
1754 duration = -1;
1755 }
1756 }
1757 else
1758 {
1759 CV_WARN("unable to query frame count of stream and fps are not available to estimate it");
1760 duration = -1;
1761 }
1762 }
1763
1764 handleMessage(pipeline);
1765
1766 {
1767 GstFormat format_;
1768 gint64 value_ = -1;
1769 gboolean status_;
1770
1771 format_ = GST_FORMAT_DEFAULT;
1772
1773 status_ = gst_element_query_position(sink, CV_GST_FORMAT(format_), &value_);
1774 if (!status_ || value_ != 0 || duration < 0)
1775 {
1776 CV_WARN("Cannot query video position: status=" << status_ << ", value=" << value_ << ", duration=" << duration);
1777 isPosFramesSupported = false;
1778 isPosFramesEmulated = true;
1779 emulatedFrameNumber = 0;
1780 }
1781 else
1782 isPosFramesSupported = true;
1783 }
1784 }
1785
1786 if (audioStream >= 0)
1787 {
1788 GSafePtr<GstPad> pad;
1789 pad.attach(gst_element_get_static_pad(audiosink, "sink"));
1790
1791 GSafePtr<GstCaps> buffer_caps;
1792 buffer_caps.attach(gst_pad_get_current_caps(pad));
1793
1794 GstAudioInfo info = {};
1795 if (gst_audio_info_from_caps(&info, buffer_caps))
1796 {
1797 nAudioChannels = GST_AUDIO_INFO_CHANNELS(&info);
1798 audioSamplesPerSecond = GST_AUDIO_INFO_RATE(&info);
1799 }
1800 else
1801 {
1802 CV_WARN("cannot query audio nChannels and SamplesPerSecond");
1803 }
1804 }
1805 GST_DEBUG_BIN_TO_DOT_FILE(GST_BIN(pipeline.get()), GST_DEBUG_GRAPH_SHOW_ALL, "pipeline");
1806 }
1807
1808 std::vector<int> unused_params = params.getUnused();
1809 for (int key : unused_params)
1810 {
1811 if (!setProperty(key, params.get<double>(key)))
1812 {
1813 CV_LOG_ERROR(NULL, "VIDEOIO/GStreamer: can't set property " << key);
1814 return false;
1815 }
1816 }
1817
1818 if (pipeline)
1819 {
1820 VideoAccelerationType actual_va_type = VIDEO_ACCELERATION_NONE;
1821 GstIterator *iter = gst_bin_iterate_recurse(GST_BIN (pipeline.get()));
1822 gst_iterator_foreach(iter, find_hw_element, (gpointer)&actual_va_type);
1823 gst_iterator_free(iter);
1824 if (va_type != VIDEO_ACCELERATION_NONE && va_type != VIDEO_ACCELERATION_ANY)
1825 {
1826 if (va_type != actual_va_type)
1827 {
1828 CV_LOG_ERROR(NULL, "VIDEOIO/GStreamer: Can't select requested video acceleration through CAP_PROP_HW_ACCELERATION: "
1829 << va_type << " (actual is " << actual_va_type << "). Bailout");
1830 return false;
1831 }
1832 }
1833 else
1834 {
1835 va_type = actual_va_type;
1836 }
1837 }
1838
1839 return true;

1851double GStreamerCapture::getProperty(int propId) const
1852{
1853 GstFormat format;
1854 gint64 value;
1855 gboolean status;
1856
1857 if(!pipeline) {
1858 CV_WARN("GStreamer: no pipeline");
1859 return CAP_PROP_UNKNOWN;
1860 }
1861
1862 switch(propId)
1863 {
1864 case CAP_PROP_POS_MSEC:
1865 return double(timestamp) / GST_MSECOND;
1866 case CAP_PROP_POS_FRAMES:
1867 if (!isPosFramesSupported)
1868 {
1869 if (isPosFramesEmulated)
1870 return emulatedFrameNumber;
1871 return CAP_PROP_UNKNOWN;
1872 }
1873 format = GST_FORMAT_DEFAULT;
1874 status = gst_element_query_position(sink.get(), CV_GST_FORMAT(format), &value);
1875 if(!status) {
1876 handleMessage(pipeline);
1877 CV_WARN("GStreamer: unable to query position of stream");
1878 return CAP_PROP_UNKNOWN;
1879 }
1880 return value;
1881 case CAP_PROP_POS_AVI_RATIO:
1882 format = GST_FORMAT_PERCENT;
1883 status = gst_element_query_position(sink.get(), CV_GST_FORMAT(format), &value);
1884 if(!status) {
1885 handleMessage(pipeline);
1886 CV_WARN("GStreamer: unable to query position of stream");
1887 return CAP_PROP_UNKNOWN;
1888 }
1889 return ((double) value) / GST_FORMAT_PERCENT_MAX;
1890 case CAP_PROP_FRAME_WIDTH:
1891 return width;
1892 case CAP_PROP_FRAME_HEIGHT:
1893 return height;
1894 case CAP_PROP_FPS:
1895 return fps;
1896 case CAP_PROP_FRAME_COUNT:
1897 return (double)duration;
1898 case CAP_PROP_BRIGHTNESS:
1899 case CAP_PROP_CONTRAST:
1900 case CAP_PROP_SATURATION:
1901 case CAP_PROP_HUE:
1902 if (v4l2src)
1903 {
1904 std::string propName = get_gst_propname(propId);
1905 if (!propName.empty())
1906 {
1907 gint32 val = 0;
1908 g_object_get(G_OBJECT(v4l2src.get()), propName.c_str(), &val, NULL);
1909 return static_cast<double>(val);
1910 }
1911 }
1912 break;
1913 case CAP_PROP_HW_ACCELERATION:
1914 return static_cast<double>(va_type);
1915 case CAP_PROP_HW_DEVICE:
1916 return static_cast<double>(hw_device);
1917 case CAP_PROP_GSTREAMER_QUEUE_LENGTH:
1918 if(!sink)
1919 {
1920 CV_WARN("there is no sink yet");
1921 return CAP_PROP_UNKNOWN;
1922 }
1923 return gst_app_sink_get_max_buffers(GST_APP_SINK(sink.get()));
1924 case CAP_PROP_AUDIO_TOTAL_CHANNELS:
1925 return nAudioChannels;
1926 case CAP_PROP_AUDIO_SAMPLES_PER_SECOND:
1927 return audioSamplesPerSecond;
1928 case CAP_PROP_AUDIO_DATA_DEPTH:
1929 return outputAudioFormat;
1930 case CAP_PROP_AUDIO_BASE_INDEX:
1931 return audioBaseIndex;
1932 case CAP_PROP_AUDIO_TOTAL_STREAMS:
1933 CV_LOG_ONCE_WARNING(NULL, "OpenCV | GStreamer: CAP_PROP_AUDIO_TOTAL_STREAMS property is not supported");
1934 return CAP_PROP_UNKNOWN;
1935 case CAP_PROP_AUDIO_POS:
1936 return audioSamplePosInSamples;
1937 case CAP_PROP_AUDIO_SHIFT_NSEC:
1938 CV_LOG_ONCE_WARNING(NULL, "OpenCV | GStreamer: CAP_PROP_AUDIO_SHIFT_NSEC property is not supported");
1939 return CAP_PROP_UNKNOWN;
1940 case CAP_PROP_OPEN_TIMEOUT_MSEC:
1941 return GST_TIME_AS_MSECONDS(openTimeout);
1942 case CAP_PROP_READ_TIMEOUT_MSEC:
1943#if FULL_GST_VERSION >= VERSION_NUM(1,10,0)
1944 return GST_TIME_AS_MSECONDS(readTimeout);
1945#else
1946 return CAP_PROP_UNKNOWN;
1947#endif
1948 default:
1949 CV_WARN("unhandled property: " << propId);
1950 break;
1951 }
1952
1953 return CAP_PROP_UNKNOWN;
1954}

1964bool GStreamerCapture::setProperty(int propId, double value)
1965{
1966 const GstSeekFlags flags = (GstSeekFlags)(GST_SEEK_FLAG_FLUSH | GST_SEEK_FLAG_ACCURATE);
1967
1968 if(!pipeline)
1969 {
1970 CV_WARN("no pipeline");
1971 return false;
1972 }
1973
1974 bool needRestart = this->isPipelinePlaying() && (propId == CAP_PROP_FRAME_WIDTH || propId == CAP_PROP_FRAME_HEIGHT || propId == CAP_PROP_FPS);
1975 if (needRestart) {
1976 this->stopPipeline();
1977 }
1978
1979 switch(propId)
1980 {
1981 case CAP_PROP_POS_MSEC:
1982 {
1983 if(!gst_element_seek_simple(GST_ELEMENT(pipeline.get()), GST_FORMAT_TIME,
1984 flags, (gint64) (value * GST_MSECOND))) {
1985 handleMessage(pipeline);
1986 CV_WARN("GStreamer: unable to seek");
1987 }
1988 else
1989 {
1990 // Optimistically caching the target timestamp before reading the first frame from the new position since
1991 // the timestamp in GStreamer can be reliable extracted from the read frames.
1992 timestamp = (gint64)value;
1993 if (isPosFramesEmulated)
1994 {
1995 if (value == 0)
1996 {
1997 emulatedFrameNumber = 0;
1998 return true;
1999 }
2000 else
2001 {
2002 isPosFramesEmulated = false; // reset frame counter emulation
2003 }
2004 }
2005 }
2006 return true;
2007 }
2008 case CAP_PROP_POS_FRAMES:
2009 {
2010 if (!isPosFramesSupported)
2011 {
2012 if (isPosFramesEmulated)
2013 {
2014 if (value == 0)
2015 {
2016 restartPipeline();
2017 emulatedFrameNumber = 0;
2018 return true;
2019 }
2020 }
2021 CV_WARN("unable to seek");
2022 return false;
2023 }
2024 // Certain mov and mp4 files seek incorrectly if the pipeline is not stopped before.
2025 if (this->isPipelinePlaying()) {
2026 this->stopPipeline();
2027 }
2028
2029 if(!gst_element_seek_simple(GST_ELEMENT(pipeline.get()), GST_FORMAT_DEFAULT,
2030 flags, (gint64) value)) {
2031 handleMessage(pipeline);
2032 CV_WARN("GStreamer: unable to seek");
2033 return false;
2034 }
2035 // wait for status update
2036 gst_element_get_state(pipeline, NULL, NULL, GST_CLOCK_TIME_NONE);
2037 return true;
2038 }
2039 case CAP_PROP_POS_AVI_RATIO:
2040 {
2041 // https://stackoverflow.com/questions/31290315
2042 // GStreamer docs: GST_FORMAT_PERCENT (5) – percentage of stream (few, if any, elements implement this as of May 2009)
2043 CV_WARN("GStreamer: seeking by file percent are not supported by most GStreamer elements");
2044 if(!gst_element_seek_simple(GST_ELEMENT(pipeline.get()), GST_FORMAT_PERCENT,
2045 flags, (gint64) (value * GST_FORMAT_PERCENT_MAX))) {

2700 if (image.type() != CV_8UC1 || image.size().height != 1) {
2701 CV_WARN("write frame skipped - expected CV_8UC1, height==1");

2703 }
2704 }

2707 if (image.type() != CV_8UC3) {
2708 CV_WARN("write frame skipped - expected CV_8UC3");

2718 else if (input_pix_fmt == GST_VIDEO_FORMAT_GRAY16_LE) {
2719 if (image.type() != CV_16UC1) {
2720 CV_WARN("write frame skipped - expected CV_16UC1");
2721 return false;

2724 else {
2725 CV_WARN("write frame skipped - unsupported format");
2726 return false;

2732 timestamp = gst_util_uint64_scale_int(num_frames, GST_SECOND, framerate);
2733
2734 //gst_app_src_push_buffer takes ownership of the buffer, so we need to supply it a copy
2735 GstBuffer *buffer = gst_buffer_new_allocate(NULL, buf_size, NULL);

2739 gst_buffer_unmap(buffer, &info);

2841 gst_message_parse_state_changed(msg, &oldstate, &newstate, &pendstate);
