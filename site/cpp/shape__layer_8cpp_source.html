42 Range getShapeRange(const MatShape& inpShape) const
43 {
44 int outDims = inpShape.layout == DATA_LAYOUT_BLOCK ? inpShape.dims - 1 : inpShape.dims;
45 int start_ = start < 0 ? start + outDims : start;
46 int end_ = end >= outDims ? outDims : end < 0 ? end + outDims : end;
47
48 start_ = std::max(0, std::min(start_, outDims));
49 end_ = std::max(0, std::min(end_, outDims));
50
51 CV_Assert(0 <= start_);
52 CV_Assert(start_ <= end_);
53 CV_Assert(end_ <= outDims);
54
55 return Range(start_, end_);
56 }

58 MatShape getOutShape(const MatShape& inpShape) const
59 {
60 MatShape outShape;
61 outShape.dims = 1;
62
63 Range r = getShapeRange(inpShape);
64
65 outShape[0] = r.end - r.start;
66 return outShape;

109 void forward(InputArrayOfArrays inputs_arr,
110 OutputArrayOfArrays outputs_arr,
111 OutputArrayOfArrays) CV_OVERRIDE
112 {
113 CV_TRACE_FUNCTION();
114 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
115
116 Size size = inputs_arr.size();
117 int ninputs = size.area();
118 CV_Assert(ninputs == 1);
119
120 MatShape inpShape = inputs_arr.shape(0);
121 Range r = getShapeRange(inpShape);
122
123 shape_type_t shapeData[CV_MAX_DIM];
124 if (inpShape.layout != DATA_LAYOUT_BLOCK)
125 {
126 for (int i = r.start; i < r.end; i++)
127 shapeData[i - r.start] = (shape_type_t)inpShape[i];
128 }
129 else
130 {
131 CV_Assert(inpShape.dims >= 3);
132 CV_Assert(inpShape.C > 0);
133 Net::Impl* netimpl_ = getNetImpl(this);
134 DataLayout origLayout = netimpl_ ? netimpl_->originalLayout : DATA_LAYOUT_NCHW;
135 CV_Assert(origLayout == DATA_LAYOUT_NCHW || origLayout == DATA_LAYOUT_NHWC);
136
137 const int semanticNdims = inpShape.dims - 1;
138 std::vector<shape_type_t> semanticShape(semanticNdims);
139 semanticShape[0] = (shape_type_t)inpShape[0];
140
141 if (origLayout == DATA_LAYOUT_NCHW)
142 {
143 CV_Assert(semanticNdims >= 2);
144 semanticShape[1] = (shape_type_t)inpShape.C;
145 for (int i = 2; i < semanticNdims; i++)
146 semanticShape[i] = (shape_type_t)inpShape[i];
147 }
148 else
149 {
150 CV_Assert(semanticNdims >= 2);
151 semanticShape[semanticNdims - 1] = (shape_type_t)inpShape.C;
152 for (int i = 1; i < semanticNdims - 1; i++)
153 semanticShape[i] = (shape_type_t)inpShape[i + 1];
154 }
155
156 for (int i = r.start; i < r.end; i++)
157 shapeData[i - r.start] = semanticShape[i];
158 }
159
160 Mat shape({r.end - r.start}, shapeType, shapeData);
161
162 int outKind = outputs_arr.kind();
163
164 if (outKind == _InputArray::STD_VECTOR_MAT) {
165 std::vector<Mat>& out = outputs_arr.getMatVecRef();
166 CV_Assert(out.size() == 1);
167 shape.copyTo(out[0]);
168 } else if (outKind == _InputArray::STD_VECTOR_UMAT) {
169 std::vector<UMat>& out = outputs_arr.getUMatVecRef();
170 CV_Assert(out.size() == 1);
171 shape.copyTo(out[0]);
172 } else {
173 CV_Error_(Error::StsBadArg, ("invalid/unsupported outputs_arr kind: %d", outKind));
174 }
175 }
