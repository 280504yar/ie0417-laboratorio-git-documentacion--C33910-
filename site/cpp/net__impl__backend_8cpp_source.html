30OrtPathString toOrtPath(const std::string& utf8Path)
31{
32#ifdef _WIN32
33 if (utf8Path.empty())
34 return std::wstring();
35
36 const int len = MultiByteToWideChar(CP_UTF8, MB_ERR_INVALID_CHARS,
37 utf8Path.c_str(), (int)utf8Path.size(), NULL, 0);
38 if (len <= 0)
39 {
40 CV_LOG_WARNING(NULL, "DNN/ONNX/ORT: path is not valid UTF-8, cannot pass it to ONNX Runtime: "
41 << utf8Path);
42 return std::wstring();
43 }
44
45 std::wstring wide((size_t)len, L'\0');
46 MultiByteToWideChar(CP_UTF8, MB_ERR_INVALID_CHARS,
47 utf8Path.c_str(), (int)utf8Path.size(), &wide[0], len);
48 return wide;
49#else
50 return utf8Path;
51#endif
52}

54void Net::Impl::refreshOrtMainGraphOutputs()
55{
56 CV_Assert(mainGraph && ort_session);
57
58 Ort::Session& session = *ort_session;
59 Ort::AllocatorWithDefaultOptions allocator;
60 const size_t noutputs = session.GetOutputCount();
61 CV_Assert(noutputs > 0);
62
63 std::vector<Arg> outs;
64 outs.reserve(noutputs);
65 for (size_t i = 0; i < noutputs; ++i)
66 {
67 Ort::AllocatedStringPtr out = session.GetOutputNameAllocated(i, allocator);
68 std::string outName = out ? std::string(out.get()) : std::string();
69 if (outName.empty())
70 outName = format("output_%zu", i);
71
72 if (haveArg(outName))
73 {
74 const int existingIdx = (int)argnames[outName];
75 if (args[(size_t)existingIdx].kind == DNN_ARG_OUTPUT)
76 {
77 outs.push_back(Arg(existingIdx));
78 continue;
79 }
80
81 std::string uniqueOutName = format("%s_%zu", outName.c_str(), i);
82 while (haveArg(uniqueOutName))
83 uniqueOutName += "_";
84 outName = uniqueOutName;
85 }
86
87 outs.push_back(newArg(outName, DNN_ARG_OUTPUT));
88 }
89 mainGraph->setOutputs(outs);
90}

92void Net::Impl::finalizeOrt()
93{
94 if (ort_session && !ortNeedsReinit)
95 return;
96 CV_Assert(!modelFileName.empty());
97
98 int target = IS_DNN_CUDA_TARGET(preferableTarget) ? preferableTarget : DNN_TARGET_CPU;
99 ort_session.reset();
100 ort_names_cache.reset();
101
102 static auto s_env = std::make_shared<Ort::Env>(ORT_LOGGING_LEVEL_WARNING, "OpenCV_DNN_ORT");
103 ort_env = s_env;
104
105 Ort::SessionOptions opts;
106 if (IS_DNN_CUDA_TARGET(target))
107 {
108 OrtCUDAProviderOptions cuda{};
109 cuda.device_id = 0;
110 try { opts.AppendExecutionProvider_CUDA(cuda); }
111 catch (const std::exception& e)
112 {
113 CV_LOG_WARNING(NULL, "DNN/ONNX/ORT: CUDA EP unavailable (" << e.what() << "), using CPU");
114 target = DNN_TARGET_CPU;
115 opts = Ort::SessionOptions();
116 }
117 }
118
119 // If the user set DNN_PROFILE_*, turn on ORT's session profiler. The JSON
120 // file is parsed lazily by collectOrtProfileData() inside getPerfProfile()/printPerfProfile().
121 ort_profile_path_prefix.clear();
122 ort_profile_collected = false;
123 ort_profile_runs = 0;
124 ort_profile_data.clear();
125 if (profilingMode != DNN_PROFILE_NONE) {
126 ort_profile_path_prefix = cv::tempfile("opencv_ort_profile_");
127 const OrtPathString profilePath = toOrtPath(ort_profile_path_prefix);
128 opts.EnableProfiling(profilePath.c_str());
129 }
130
131 const OrtPathString modelPath = toOrtPath(modelFileName);
132 ort_session = std::make_shared<Ort::Session>(*ort_env, modelPath.c_str(), opts);
133 preferableTarget = target;
134 ortNeedsReinit = false;
135
136 refreshOrtMainGraphOutputs();
137 applyStagedOrtInputs();
138}

141Ptr<BackendWrapper> Net::Impl::wrap(Mat& host)
142{
143 if (preferableBackend == DNN_BACKEND_OPENCV &&
144 (preferableTarget == DNN_TARGET_CPU || preferableTarget == DNN_TARGET_CPU_FP16))
145 return Ptr<BackendWrapper>();
146
147 MatShape shape(host.dims);
148 for (int i = 0; i < host.dims; ++i)
149 shape[i] = host.size[i];
150
151 void* data = host.data;
152 if (backendWrappers.find(data) != backendWrappers.end())
153 {
154 Ptr<BackendWrapper> baseBuffer = backendWrappers[data];
155 if (preferableBackend == DNN_BACKEND_OPENCV)
156 {
157#ifdef HAVE_OPENCL
158 CV_Assert(IS_DNN_OPENCL_TARGET(preferableTarget));
159 return OpenCLBackendWrapper::create(baseBuffer, host);
160#else
161 CV_Error(Error::StsInternal, "");
162#endif
163 }
164 else if (preferableBackend == DNN_BACKEND_INFERENCE_ENGINE_NN_BUILDER_2019)
165 {
166 CV_ERROR_DNN_BACKEND_INFERENCE_ENGINE_NN_BUILDER_2019;
167 }
168 else if (preferableBackend == DNN_BACKEND_INFERENCE_ENGINE_NGRAPH)
169 {
170 return wrapMat(preferableBackend, preferableTarget, host);
171 }
172 else if (preferableBackend == DNN_BACKEND_WEBNN)
173 {
174#ifdef HAVE_WEBNN
175 return wrapMat(preferableBackend, preferableTarget, host);
176#endif
177 }
178 else if (preferableBackend == DNN_BACKEND_VKCOM)
179 {
180#ifdef HAVE_VULKAN
181 return Ptr<BackendWrapper>(new VkComBackendWrapper(baseBuffer, host));
182#endif
183 }
184 else if (preferableBackend == DNN_BACKEND_CUDA)
185 {
186 CV_Assert(haveCUDA());
187#ifdef HAVE_CUDA
188 CV_CheckType(host.depth(), host.depth() == CV_32F || host.depth() == CV_8S || host.depth() == CV_8U || host.depth() == CV_32S || host.depth() == CV_64S || host.depth() == CV_Bool, "Unsupported type for CUDA");
189 CV_Assert(IS_DNN_CUDA_TARGET(preferableTarget));
190 switch (host.depth())
191 {
192 case CV_32F:
193 if (preferableTarget == DNN_TARGET_CUDA_FP16)
194 return CUDABackendWrapperFP16::create(baseBuffer, shape);
195 else
196 return CUDABackendWrapperFP32::create(baseBuffer, shape);
197 case CV_8S:
198 return CUDABackendWrapperINT8::create(baseBuffer, shape);
199 case CV_8U:
200 return CUDABackendWrapperUINT8::create(baseBuffer, shape);
201 case CV_32S:
202 return CUDABackendWrapperINT32::create(baseBuffer, shape);
203 case CV_64S:
204 return CUDABackendWrapperINT64::create(baseBuffer, shape);
205 case CV_Bool:
206 return CUDABackendWrapperBOOL::create(baseBuffer, shape);
207 default:
208 CV_Error(Error::BadDepth, "Unsupported mat type for CUDA");
209 }
210#endif
211 }
212 else if (preferableBackend == DNN_BACKEND_TIMVX)
213 {
214#ifdef HAVE_TIMVX
215 return Ptr<BackendWrapper>(new TimVXBackendWrapper(baseBuffer, host));
216#endif
217 }
218 else if (preferableBackend == DNN_BACKEND_CANN)
219 {
220 CV_Assert(0 && "Internal error: DNN_BACKEND_CANN must be implemented through inheritance");
221 }
222 else
223 CV_Error(Error::StsNotImplemented, "Unknown backend identifier");
224 }
225
226 Ptr<BackendWrapper> wrapper = wrapMat(preferableBackend, preferableTarget, host);
227 backendWrappers[data] = wrapper;
228 return wrapper;
229}

232void Net::Impl::initBackend(const std::vector<LayerPin>& blobsToKeep_)
233{
234 CV_TRACE_FUNCTION();
235 if (preferableBackend == DNN_BACKEND_OPENCV)
236 {
237 CV_Assert(preferableTarget == DNN_TARGET_CPU || preferableTarget == DNN_TARGET_CPU_FP16 || IS_DNN_OPENCL_TARGET(preferableTarget));
238 }
239 else if (preferableBackend == DNN_BACKEND_INFERENCE_ENGINE_NGRAPH)
240 {
241 CV_Assert(0 && "Inheritance must be used with OpenVINO backend");
242 }
243 else if (preferableBackend == DNN_BACKEND_WEBNN)
244 {
245#ifdef HAVE_WEBNN
246 initWebnnBackend(blobsToKeep_);
247#else
248 CV_Error(Error::StsNotImplemented, "This OpenCV version is built without support of WebNN");
249#endif
250 }
251 else if (preferableBackend == DNN_BACKEND_VKCOM)
252 {
253#ifdef HAVE_VULKAN
254 initVkComBackend();
255#else
256 CV_Error(Error::StsNotImplemented, "This OpenCV version is built without support of Vulkan");
257#endif
258 }
259 else if (preferableBackend == DNN_BACKEND_CUDA)
260 {
261#ifdef HAVE_CUDA
262 initCUDABackend(blobsToKeep_);
263#else
264 CV_Error(Error::StsNotImplemented, "This OpenCV version is built without support of CUDA/CUDNN");
265#endif
266 }
267 else if (preferableBackend == DNN_BACKEND_TIMVX)
268 {
269#ifdef HAVE_TIMVX
270 initTimVXBackend();
271#else
272 CV_Error(Error::StsNotImplemented, "This OpenCV version is built without support of TimVX");
273#endif
274 }
275 else if (preferableBackend == DNN_BACKEND_CANN)
276 {
277 CV_Assert(0 && "Internal error: DNN_BACKEND_CANN must be implemented through inheritance");
278 }
279 else
280 {
281 CV_Error(Error::StsNotImplemented, cv::format("Unknown backend identifier: %d", preferableBackend));
282 }

286void Net::Impl::setPreferableBackend(Net& net, int backendId)
287{
288 if (backendId == DNN_BACKEND_DEFAULT)
289 backendId = (Backend)getParam_DNN_BACKEND_DEFAULT();
290
291 if (backendId == DNN_BACKEND_INFERENCE_ENGINE)
292 backendId = DNN_BACKEND_INFERENCE_ENGINE_NGRAPH;
293
294 if (netWasQuantized && backendId != DNN_BACKEND_OPENCV && backendId != DNN_BACKEND_TIMVX &&
295 backendId != DNN_BACKEND_INFERENCE_ENGINE_NGRAPH)
296 {
297 CV_LOG_WARNING(NULL, "DNN: Only default, TIMVX and OpenVINO backends support quantized networks");
298 backendId = DNN_BACKEND_OPENCV;
299 }
300#ifdef HAVE_DNN_NGRAPH
301 if (netWasQuantized && backendId == DNN_BACKEND_INFERENCE_ENGINE_NGRAPH && INF_ENGINE_VER_MAJOR_LT(INF_ENGINE_RELEASE_2023_0))
302 {
303 CV_LOG_WARNING(NULL, "DNN: OpenVINO 2023.0 and higher is required to supports quantized networks");
304 backendId = DNN_BACKEND_OPENCV;
305 }
306#endif
307
308 if (preferableBackend == backendId)
309 return;
310
311#ifdef HAVE_ONNXRUNTIME
312 if (useOrtEngine && mainGraph && modelFormat == DNN_MODEL_ONNX && !modelFileName.empty())
313 {
314 preferableBackend = backendId;
315 ortNeedsReinit = true; // will be applied on finalizeNet()
316 return;
317 }
318#endif
319
320 if (mainGraph)
321 {
322 if (backendId == DNN_BACKEND_OPENCV
323#ifdef HAVE_CUDA
324 || backendId == DNN_BACKEND_CUDA
325#endif
326 )
327 {
328 if (preferableBackend != backendId) {
329 preferableBackend = backendId;
330 finalized = false; // re-select per-op executors on next finalize()
331 }
332 return;
333 }
334 CV_LOG_WARNING(NULL, "Back-ends are not supported by the new graph engine for now");
335 preferableBackend = backendId;
336 return;
337 }
338
339 clear();
340 if (backendId == DNN_BACKEND_INFERENCE_ENGINE_NGRAPH)
341 {
342#if defined(HAVE_INF_ENGINE)
343 switchToOpenVINOBackend(net);
344#elif defined(ENABLE_PLUGINS)
345 auto& networkBackend = dnn_backend::createPluginDNNNetworkBackend("openvino");
346 networkBackend.switchBackend(net);
347#else
348 CV_Error(Error::StsNotImplemented, "OpenVINO backend is not available in the current OpenCV build");
349#endif
350 }
351 else if (backendId == DNN_BACKEND_CANN)
352 {
353#ifdef HAVE_CANN
354 switchToCannBackend(net);
355#else
356 CV_Error(Error::StsNotImplemented, "CANN backend is not availlable in the current OpenCV build");
357#endif
358 }
359 else
360 {
361 preferableBackend = backendId;
362 }

365void Net::Impl::setPreferableTarget(int targetId)
366{
367#ifdef HAVE_ONNXRUNTIME
368 if (useOrtEngine && mainGraph && modelFormat == DNN_MODEL_ONNX && !modelFileName.empty())
369 {
370 int resolved = IS_DNN_CUDA_TARGET(targetId) ? targetId : DNN_TARGET_CPU;
371 if (preferableTarget != resolved)
372 {
373 preferableTarget = resolved;
374 ortNeedsReinit = true; // will be applied on finalizeNet()
375 }
376 return;
377 }
378#endif
379
380 if (mainGraph)
381 {
382#ifdef HAVE_CUDA
383 if (targetId == DNN_TARGET_CPU || IS_DNN_CUDA_TARGET(targetId))
384 {
385 if (preferableTarget != targetId) {
386 preferableTarget = targetId;
387 finalized = false; // re-select per-op executors on next finalize()
388 }
389 return;
390 }
391#endif
392 CV_LOG_WARNING(NULL, "Targets are not supported by the new graph engine for now");
393 return;
394 }
395 if (netWasQuantized && targetId != DNN_TARGET_CPU &&
396 targetId != DNN_TARGET_OPENCL && targetId != DNN_TARGET_OPENCL_FP16 && targetId != DNN_TARGET_NPU)
397 {
398 CV_LOG_WARNING(NULL, "DNN: Only CPU, OpenCL/OpenCL FP16 and NPU targets are supported by quantized networks");
399 targetId = DNN_TARGET_CPU;
400 }
401
402 if (preferableTarget != targetId)
403 {
404 preferableTarget = targetId;
405 if (IS_DNN_OPENCL_TARGET(targetId))
406 {
407#ifndef HAVE_OPENCL
408#ifdef HAVE_INF_ENGINE
409 if (preferableBackend == DNN_BACKEND_OPENCV)
410#else
411 if (preferableBackend == DNN_BACKEND_DEFAULT ||
412 preferableBackend == DNN_BACKEND_OPENCV)
413#endif // HAVE_INF_ENGINE
414 preferableTarget = DNN_TARGET_CPU;
415#else
416 bool fp16 = ocl::Device::getDefault().isExtensionSupported("cl_khr_fp16");
417 if (!fp16 && targetId == DNN_TARGET_OPENCL_FP16)
418 preferableTarget = DNN_TARGET_OPENCL;
419#endif
420 }
421
422 if (IS_DNN_CUDA_TARGET(targetId))
423 {
424 preferableTarget = DNN_TARGET_CPU;
425#ifdef HAVE_CUDA
426 if (cuda4dnn::doesDeviceSupportFP16() && targetId == DNN_TARGET_CUDA_FP16)
427 preferableTarget = DNN_TARGET_CUDA_FP16;
428 else
429 preferableTarget = DNN_TARGET_CUDA;
430#endif
431 }
432#if !defined(__arm64__) || !__arm64__
433 if (targetId == DNN_TARGET_CPU_FP16)
434 {
435 CV_LOG_WARNING(NULL, "DNN: fall back to DNN_TARGET_CPU. Only ARM v8 CPU is supported by DNN_TARGET_CPU_FP16.");
436 targetId = DNN_TARGET_CPU;
437 }
438#endif
439
440 clear();
441
442 if (targetId == DNN_TARGET_CPU_FP16)
443 {
444 if (useWinograd) {
445 CV_LOG_INFO(NULL, "DNN: DNN_TARGET_CPU_FP16 is set => Winograd convolution is disabled by default to preserve accuracy. If needed, enable it explicitly using enableWinograd(true).");
446 enableWinograd(false);
447 }
448 }
449 }
