46 std::vector<MatShape>& outputs, std::vector<MatShape>&) const CV_OVERRIDE

57 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays) CV_OVERRIDE
58 {
59 std::vector<Mat> rawIn, rawOut;
60 inputs_arr.getMatVector(rawIn);
61 outputs_arr.getMatVector(rawOut);
62
63 const bool fp16 = rawIn[0].depth() == CV_16F;
64 std::vector<Mat> in32, out32;
65 if (fp16)
66 {
67 in32.resize(rawIn.size());
68 for (size_t i = 0; i < rawIn.size(); ++i)
69 if (!rawIn[i].empty()) rawIn[i].convertTo(in32[i], CV_32F);
70 out32.resize(rawOut.size());
71 for (size_t i = 0; i < rawOut.size(); ++i)
72 out32[i].create(rawOut[i].dims, rawOut[i].size.p, CV_32F);
73 }
74 std::vector<Mat>& inputs = fp16 ? in32 : rawIn;
75 std::vector<Mat>& outputs = fp16 ? out32 : rawOut;
76
77 const Mat& input = inputs[0];
78 const Mat& weight = inputs[1];
79 const bool has_bias = present(inputs, 2);
80 const bool has_past = present(inputs, 3);
81
82 const int B = input.size[0], C = input.size[1], T = input.size[2];
83 const int K = weight.size[2];
84 const int P = K - 1; // state / left-pad width
85
86 const float* Ip = input.ptr<float>();
87 const float* Wp = weight.ptr<float>();
88 const float* Bp = has_bias ? inputs[2].ptr<float>() : nullptr;
89 const float* Sp = has_past ? inputs[3].ptr<float>() : nullptr;
90 float* Op = outputs[0].ptr<float>();
91 float* PSp = outputs[1].ptr<float>();
92
93 parallel_for_(Range(0, B * C), [&](const Range& r)
94 {
95 std::vector<float> pad(P + T);
96 for (int bc = r.start; bc < r.end; ++bc)
97 {
98 const int c = bc % C;
99 const float* x = Ip + (size_t)bc * T;
100 const float* w = Wp + (size_t)c * K;
101
102 for (int j = 0; j < P; ++j)
103 pad[j] = has_past ? Sp[(size_t)bc * P + j] : 0.f;
104 for (int t = 0; t < T; ++t)
105 pad[P + t] = x[t];
106
107 const float bias = has_bias ? Bp[c] : 0.f;
108 float* o = Op + (size_t)bc * T;
109 for (int t = 0; t < T; ++t)
110 {
111 float acc = bias;
112 for (int k = 0; k < K; ++k)
113 acc += w[k] * pad[t + k];
114 o[t] = acc;
115 }
116 if (silu)
117 for (int t = 0; t < T; ++t)
118 o[t] = o[t] / (1.f + std::exp(-o[t]));
119
120 float* ps = PSp + (size_t)bc * P;
121 for (int j = 0; j < P; ++j)
122 ps[j] = pad[T + j];
123 }
124 });
125
126 if (fp16)
127 for (size_t i = 0; i < rawOut.size(); ++i)
128 out32[i].convertTo(rawOut[i], CV_16F);
129 }

131 int64 getFLOPS(const std::vector<MatShape>& inputs, const std::vector<MatShape>&) const CV_OVERRIDE
132 {
133 const int64 B = inputs[0][0], C = inputs[0][1], T = inputs[0][2], K = inputs[1][2];
134 return B * C * T * (2 * K + 4);
135 }

137private:
138 bool silu = false;
139};
