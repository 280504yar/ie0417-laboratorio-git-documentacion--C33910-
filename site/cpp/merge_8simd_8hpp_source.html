19/*
20 The trick with STORE_UNALIGNED/STORE_ALIGNED_NOCACHE is the following:
21 on IA there are instructions movntps and such to which
22 v_store_interleave(...., STORE_ALIGNED_NOCACHE) is mapped.
23 Those instructions write directly into memory w/o touching cache
24 that results in dramatic speed improvements, especially on
25 large arrays (FullHD, 4K etc.).
26
27 Those intrinsics require the destination address to be aligned
28 by 16/32 bits (with SSE2 and AVX2, respectively).
29 So we potentially split the processing into 3 stages:
30 1) the optional prefix part [0:i0), where we use simple unaligned stores.
31 2) the optional main part [i0:len - VECSZ], where we use "nocache" mode.
32 But in some cases we have to use unaligned stores in this part.
33 3) the optional suffix part (the tail) (len - VECSZ:len) where we switch back to "unaligned" mode
34 to process the remaining len - VECSZ elements.
35 In principle there can be very poorly aligned data where there is no main part.
36 For that we set i0=0 and use unaligned stores for the whole array.
37*/
38template<typename T, typename VecT> static void
39vecmerge_( const T** src, T* dst, int len, int cn )
40{
41 const int VECSZ = VTraits<VecT>::vlanes();
42 int i, i0 = 0;
43 const T* src0 = src[0];
44 const T* src1 = src[1];
45
46 const int dstElemSize = cn * sizeof(T);
47 int r = (int)((size_t)(void*)dst % (VECSZ*sizeof(T)));
48 hal::StoreMode mode = hal::STORE_ALIGNED_NOCACHE;
49 if( r != 0 )
50 {
51 mode = hal::STORE_UNALIGNED;
52 if (r % dstElemSize == 0 && len > VECSZ*2)
53 i0 = VECSZ - (r / dstElemSize);
54 }
55
56 if( cn == 2 )
57 {
58 for( i = 0; i < len; i += VECSZ )
59 {
60 if( i > len - VECSZ )
61 {
62 i = len - VECSZ;
63 mode = hal::STORE_UNALIGNED;
64 }
65 VecT a = vx_load(src0 + i), b = vx_load(src1 + i);
66 v_store_interleave(dst + i*cn, a, b, mode);
67 if( i < i0 )
68 {
69 i = i0 - VECSZ;
70 mode = hal::STORE_ALIGNED_NOCACHE;
71 }
72 }
73 }
74 else if( cn == 3 )
75 {
76 const T* src2 = src[2];
77 for( i = 0; i < len; i += VECSZ )
78 {
79 if( i > len - VECSZ )
80 {
81 i = len - VECSZ;
82 mode = hal::STORE_UNALIGNED;
83 }
84 VecT a = vx_load(src0 + i), b = vx_load(src1 + i), c = vx_load(src2 + i);
85 v_store_interleave(dst + i*cn, a, b, c, mode);
86 if( i < i0 )
87 {
88 i = i0 - VECSZ;
89 mode = hal::STORE_ALIGNED_NOCACHE;
90 }
91 }
92 }
93 else
94 {
95 CV_Assert( cn == 4 );
96 const T* src2 = src[2];
97 const T* src3 = src[3];
98 for( i = 0; i < len; i += VECSZ )
99 {
100 if( i > len - VECSZ )
101 {
102 i = len - VECSZ;
103 mode = hal::STORE_UNALIGNED;
104 }
105 VecT a = vx_load(src0 + i), b = vx_load(src1 + i);
106 VecT c = vx_load(src2 + i), d = vx_load(src3 + i);
107 v_store_interleave(dst + i*cn, a, b, c, d, mode);
108 if( i < i0 )
109 {
110 i = i0 - VECSZ;
111 mode = hal::STORE_ALIGNED_NOCACHE;
112 }
113 }
114 }
115 vx_cleanup();
116}

119template<typename T> static void
120merge_( const T** src, T* dst, int len, int cn )
121{
122 int k = cn % 4 ? cn % 4 : 4;
123 int i, j;
124 if( k == 1 )
125 {
126 const T* src0 = src[0];
127 for( i = j = 0; i < len; i++, j += cn )
128 dst[j] = src0[i];
129 }
130 else if( k == 2 )
131 {
132 const T *src0 = src[0], *src1 = src[1];
133 i = j = 0;
134 for( ; i < len; i++, j += cn )
135 {
136 dst[j] = src0[i];
137 dst[j+1] = src1[i];
138 }
139 }
140 else if( k == 3 )
141 {
142 const T *src0 = src[0], *src1 = src[1], *src2 = src[2];
143 i = j = 0;
144 for( ; i < len; i++, j += cn )
145 {
146 dst[j] = src0[i];
147 dst[j+1] = src1[i];
148 dst[j+2] = src2[i];
149 }
150 }
151 else
152 {
153 const T *src0 = src[0], *src1 = src[1], *src2 = src[2], *src3 = src[3];
154 i = j = 0;
155 for( ; i < len; i++, j += cn )
156 {
157 dst[j] = src0[i]; dst[j+1] = src1[i];
158 dst[j+2] = src2[i]; dst[j+3] = src3[i];
159 }
160 }
161
162 for( ; k < cn; k += 4 )
163 {
164 const T *src0 = src[k], *src1 = src[k+1], *src2 = src[k+2], *src3 = src[k+3];
165 for( i = 0, j = k; i < len; i++, j += cn )
166 {
167 dst[j] = src0[i]; dst[j+1] = src1[i];
168 dst[j+2] = src2[i]; dst[j+3] = src3[i];
169 }
170 }
171}
