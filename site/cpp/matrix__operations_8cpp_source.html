130static bool ocl_setIdentity( InputOutputArray _m, const Scalar& s )
131{
132 int type = _m.type(), depth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type), kercn = cn, rowsPerWI = 1;
133 int sctype = CV_MAKE_TYPE(depth, cn == 3 ? 4 : cn);
134 if (ocl::Device::getDefault().isIntel())
135 {
136 rowsPerWI = 4;
137 if (cn == 1)
138 {
139 kercn = std::min(ocl::predictOptimalVectorWidth(_m), 4);
140 if (kercn != 4)
141 kercn = 1;
142 }
143 }
144
145 ocl::Kernel k("setIdentity", ocl::core::set_identity_oclsrc,
146 format("-D T=%s -D T1=%s -D cn=%d -D ST=%s -D kercn=%d -D rowsPerWI=%d",
147 ocl::memopTypeToStr(CV_MAKE_TYPE(depth, kercn)),
148 ocl::memopTypeToStr(depth), cn,
149 ocl::memopTypeToStr(sctype),
150 kercn, rowsPerWI));
151 if (k.empty())
152 return false;
153
154 UMat m = _m.getUMat();
155 k.args(ocl::KernelArg::WriteOnly(m, cn, kercn),
156 ocl::KernelArg::Constant(Mat(1, 1, sctype, s)));
157
158 size_t globalsize[2] = { (size_t)m.cols * cn / kercn, ((size_t)m.rows + rowsPerWI - 1) / rowsPerWI };
159 return k.run(2, globalsize, nullptr, false);
160}

166void cv::setIdentity( InputOutputArray _m, const Scalar& s )
167{
168 CV_INSTRUMENT_REGION();
169
170 CV_Assert( _m.dims() <= 2 );
171
172 CV_OCL_RUN(_m.isUMat(),
173 ocl_setIdentity(_m, s))
174
175 Mat m = _m.getMat();
176 int rows = m.rows, cols = m.cols, type = m.type();
177
178 if( type == CV_32FC1 )
179 {
180 float* data = m.ptr<float>();
181 float val = (float)s[0];
182 size_t step = m.step/sizeof(data[0]);
183
184 for( int i = 0; i < rows; i++, data += step )
185 {
186 for( int j = 0; j < cols; j++ )
187 data[j] = 0;
188 if( i < cols )
189 data[i] = val;
190 }
191 }
192 else if( type == CV_64FC1 )
193 {
194 double* data = m.ptr<double>();
195 double val = s[0];
196 size_t step = m.step/sizeof(data[0]);
197
198 for( int i = 0; i < rows; i++, data += step )
199 {
200 std::fill(data, data + cols, 0.0);
201 if (i < cols)
202 data[i] = val;
203 }
204 }
205 else
206 {
207 m = Scalar(0);
208 m.diag() = s;
209 }
210}

231cv::Scalar cv::trace( InputArray _m )
232{
233 CV_INSTRUMENT_REGION();
234
235 Mat m = _m.getMat();
236 CV_Assert( m.dims <= 2 );
237 int type = m.type();
238 int nm = std::min(m.rows, m.cols);
239
240 if( type == CV_32FC1 )
241 {
242 const float* ptr = m.ptr<float>();
243 size_t step = m.step/sizeof(ptr[0]) + 1;
244 double _s = 0;
245 for( int i = 0; i < nm; i++ )
246 _s += ptr[i*step];
247 return _s;
248 }
249
250 if( type == CV_64FC1 )
251 {
252 const double* ptr = m.ptr<double>();
253 size_t step = m.step/sizeof(ptr[0]) + 1;
254 double _s = 0;
255 for( int i = 0; i < nm; i++ )
256 _s += ptr[i*step];
257 return _s;
258 }
259
260 return cv::sum(m.diag());
261}

266void cv::completeSymm( InputOutputArray _m, bool LtoR )
267{
268 CV_INSTRUMENT_REGION();
269
270 Mat m = _m.getMat();
271 size_t step = m.step, esz = m.elemSize();
272 CV_Assert( m.dims <= 2 && m.rows == m.cols );
273
274 int rows = m.rows;
275 int j0 = 0, j1 = rows;
276
277 uchar* data = m.ptr();
278 for( int i = 0; i < rows; i++ )
279 {
280 if( !LtoR ) j1 = i; else j0 = i+1;
281 for( int j = j0; j < j1; j++ )
282 memcpy(data + (i*step + j*esz), data + (j*step + i*esz), esz);
283 }

287cv::Mat cv::Mat::cross(InputArray _m) const
288{
289 Mat m = _m.getMat();
290 int tp = type(), d = CV_MAT_DEPTH(tp);
291 CV_Assert( dims <= 2 && m.dims <= 2 && size() == m.size() && tp == m.type() &&
292 ((rows == 3 && cols == 1) || (cols*channels() == 3 && rows == 1)));
293 Mat result(rows, cols, tp);
294
295 if( d == CV_32F )
296 {
297 const float *a = (const float*)data, *b = (const float*)m.data;
298 float* c = (float*)result.data;
299 size_t lda = rows > 1 ? step/sizeof(a[0]) : 1;
300 size_t ldb = rows > 1 ? m.step/sizeof(b[0]) : 1;
301
302 c[0] = a[lda] * b[ldb*2] - a[lda*2] * b[ldb];
303 c[1] = a[lda*2] * b[0] - a[0] * b[ldb*2];
304 c[2] = a[0] * b[ldb] - a[lda] * b[0];
305 }
306 else if( d == CV_64F )
307 {
308 const double *a = (const double*)data, *b = (const double*)m.data;
309 double* c = (double*)result.data;
310 size_t lda = rows > 1 ? step/sizeof(a[0]) : 1;
311 size_t ldb = rows > 1 ? m.step/sizeof(b[0]) : 1;
312
313 c[0] = a[lda] * b[ldb*2] - a[lda*2] * b[ldb];
314 c[1] = a[lda*2] * b[0] - a[0] * b[ldb*2];
315 c[2] = a[0] * b[ldb] - a[lda] * b[0];
316 }
317
318 return result;
319}

341template<typename T, typename ST, typename WT, class Op, class OpInit>
342class ReduceR_Invoker : public ParallelLoopBody
343{
344public:
345 ReduceR_Invoker(const Mat& aSrcmat, Mat& aDstmat, Op& aOp, OpInit& aOpInit)
346 :srcmat(aSrcmat),dstmat(aDstmat),op(aOp),opInit(aOpInit),buffer(srcmat.size().width*srcmat.channels())
347 {
348 }
349 void operator()(const Range& range) const CV_OVERRIDE
350 {
351 const T* src = srcmat.ptr<T>();
352 const size_t srcstep = srcmat.step/sizeof(src[0]);
353 WT* buf = buffer.data();
354 ST* dst = dstmat.ptr<ST>();
355 int i = 0;
356
357 for( i = range.start ; i < range.end; i++ )
358 buf[i] = opInit(src[i]);
359
360 int height = srcmat.size().height;
361 for( ; --height; )
362 {
363 src += srcstep;
364 ReduceR_SIMD<T, WT, Op> simd_op;
365 i = simd_op(src, range.start, range.end, buf, op);
366 #if CV_ENABLE_UNROLLED
367 for(; i <= range.end - 4; i += 4 )
368 {
369 WT s0, s1;
370 s0 = op(buf[i], (WT)src[i]);
371 s1 = op(buf[i+1], (WT)src[i+1]);
372 buf[i] = s0; buf[i+1] = s1;
373
374 s0 = op(buf[i+2], (WT)src[i+2]);
375 s1 = op(buf[i+3], (WT)src[i+3]);
376 buf[i+2] = s0; buf[i+3] = s1;
377 }
378 #endif
379 for( ; i < range.end; i++ )
380 buf[i] = op(buf[i], (WT)src[i]);
381 }
382
383 for( i = range.start ; i < range.end; i++ )
384 dst[i] = (ST)buf[i];
385 }
386private:
387 const Mat& srcmat;
388 Mat& dstmat;
389 Op& op;
390 OpInit& opInit;
391 mutable AutoBuffer<WT> buffer;
392};

406template<typename T, typename ST, typename WT, class Op, class OpInit>
407class ReduceC_Invoker : public ParallelLoopBody
408{
409public:
410 ReduceC_Invoker(const Mat& aSrcmat, Mat& aDstmat, Op& aOp, OpInit& aOpInit)
411 :srcmat(aSrcmat),dstmat(aDstmat),op(aOp),opInit(aOpInit)
412 {
413 }
414 void operator()(const Range& range) const CV_OVERRIDE
415 {
416 const int cn = srcmat.channels();
417 const int width = srcmat.size().width*cn;
418 AutoBuffer<WT> cumul(cn);
419 for( int y = range.start; y < range.end; y++ )
420 {
421 const T* src = srcmat.ptr<T>(y);
422 ST* dst = dstmat.ptr<ST>(y);
423 if( width == cn )
424 {
425 for( int k = 0; k < cn; k++ )
426 dst[k] = (ST)opInit(src[k]);
427 }
428 else
429 {
430 for(int k = 0; k < cn ; ++k )
431 cumul[k] = opInit(src[k]);
432 for(int k = cn ; k < width ; k += cn )
433 {
434 for (int c = 0 ; c < cn ; ++c)
435 cumul[c] = op(cumul[c], src[k+c]);
436 }
437 for(int k = 0 ; k < cn ; ++k )
438 dst[k] = (ST)cumul[k];
439 }
440 }
441 }
442private:
443 const Mat& srcmat;
444 Mat& dstmat;
445 Op& op;
446 OpInit& opInit;
447};

986{

1012 memcpy(dptr, sptr, sizeof(T) * len);

1014 ptr = dptr;

1027 }
1028

1081 if(ippsSortRadixGetBufferSize(len, type, &bufferSize) < 0)
1082 return false;

1085 // IPP scratch (buffer) and per-column temporaries are thread-local; a shared flag
1086 // records IPP failures since the lambda cannot return from ipp_sort.

1098 if( !ok )
1099 break;
1100 void* ptr;
1101 if( sortRows )
1102 {
1103 uchar* dptr = dst.ptr(i);
1104 if( !inplace )
1105 memcpy(dptr, src.ptr(i), src.cols * src.elemSize());
1106 ptr = dptr;
1107 }
1108 else

1110 subRect.x = i;
1111 Mat(src, subRect).copyTo(row);

1122 {
1123 Mat dstSub(dst, subRect);
1124 row = row.reshape(1, dstSub.rows);
1125 row.copyTo(dstSub);
1126 row = row.reshape(1, 1);
1127 }
1128 }
1129 });
