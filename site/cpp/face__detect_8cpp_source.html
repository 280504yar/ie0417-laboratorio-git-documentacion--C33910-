134 int detect(InputArray input_image, OutputArray faces) override
135 {
136 // TODO: more checkings should be done?
137 if (input_image.empty())
138 {
139 return 0;
140 }
141 CV_CheckEQ(input_image.size(), Size(inputW, inputH), "Size does not match. Call setInputSize(size) if input size does not match the preset size");
142
143 Mat input_blob;
144 if(input_image.kind() == _InputArray::UMAT) {
145 // Pad input_image with divisor 32
146 UMat pad_image;
147 padWithDivisor(input_image, pad_image);
148 // Build blob from input image
149 input_blob = dnn::blobFromImage(pad_image);
150 } else {
151 // Pad input_image with divisor 32
152 Mat pad_image;
153 padWithDivisor(input_image, pad_image);
154 // Build blob from input image
155 input_blob = dnn::blobFromImage(pad_image);
156 }
157 // Forward
158 std::vector<String> output_names = { "cls_8", "cls_16", "cls_32", "obj_8", "obj_16", "obj_32", "bbox_8", "bbox_16", "bbox_32", "kps_8", "kps_16", "kps_32" };
159 std::vector<Mat> output_blobs;
160 net.setInput(input_blob);
161 net.forward(output_blobs, output_names);
162
163 // Post process
164 Mat results = postProcess(output_blobs);
165 results.convertTo(faces, CV_32FC1);
166 return 1;
167 }
168private:
169 Mat postProcess(const std::vector<Mat>& output_blobs)
170 {
171 Mat faces;
172 for (size_t i = 0; i < strides.size(); ++i) {
173 int cols = int(padW / strides[i]);
174 int rows = int(padH / strides[i]);
175
176 // Extract from output_blobs
177 Mat cls = output_blobs[i];
178 Mat obj = output_blobs[i + strides.size() * 1];
179 Mat bbox = output_blobs[i + strides.size() * 2];
180 Mat kps = output_blobs[i + strides.size() * 3];
181
182 // Decode from predictions
183 float* cls_v = (float*)(cls.data);
184 float* obj_v = (float*)(obj.data);
185 float* bbox_v = (float*)(bbox.data);
186 float* kps_v = (float*)(kps.data);
187
188 // (tl_x, tl_y, w, h, re_x, re_y, le_x, le_y, nt_x, nt_y, rcm_x, rcm_y, lcm_x, lcm_y, score)
189 // 'tl': top left point of the bounding box
190 // 're': right eye, 'le': left eye
191 // 'nt': nose tip
192 // 'rcm': right corner of mouth, 'lcm': left corner of mouth
193 Mat face(1, 15, CV_32FC1);
194
195 for(int r = 0; r < rows; ++r) {
196 for(int c = 0; c < cols; ++c) {
197 size_t idx = r * cols + c;
198
199 // Get score
200 float cls_score = cls_v[idx];
201 float obj_score = obj_v[idx];
202
203 // Clamp
204 cls_score = MIN(cls_score, 1.f);
205 cls_score = MAX(cls_score, 0.f);
206 obj_score = MIN(obj_score, 1.f);
207 obj_score = MAX(obj_score, 0.f);
208 float score = std::sqrt(cls_score * obj_score);
209 face.at<float>(0, 14) = score;
210
211 // Checking if the score meets the threshold before adding the face
212 if (score < scoreThreshold)
213 continue;
214 // Get bounding box
215 float cx = ((c + bbox_v[idx * 4 + 0]) * strides[i]);
216 float cy = ((r + bbox_v[idx * 4 + 1]) * strides[i]);
217 float w = exp(bbox_v[idx * 4 + 2]) * strides[i];
218 float h = exp(bbox_v[idx * 4 + 3]) * strides[i];
219
220 float x1 = cx - w / 2.f;
221 float y1 = cy - h / 2.f;
222
223 face.at<float>(0, 0) = x1;
224 face.at<float>(0, 1) = y1;
225 face.at<float>(0, 2) = w;
226 face.at<float>(0, 3) = h;
227
228 // Get landmarks
229 for(int n = 0; n < 5; ++n) {
230 face.at<float>(0, 4 + 2 * n) = (kps_v[idx * 10 + 2 * n] + c) * strides[i];
231 face.at<float>(0, 4 + 2 * n + 1) = (kps_v[idx * 10 + 2 * n + 1]+ r) * strides[i];
232 }
233 faces.push_back(face);
234 }
235 }
236 }
237
238 if (faces.rows > 1)
239 {
240 // Retrieve boxes and scores
241 std::vector<Rect2i> faceBoxes;
242 std::vector<float> faceScores;
243 for (int rIdx = 0; rIdx < faces.rows; rIdx++)
244 {
245 faceBoxes.push_back(Rect2i(int(faces.at<float>(rIdx, 0)),
246 int(faces.at<float>(rIdx, 1)),
247 int(faces.at<float>(rIdx, 2)),
248 int(faces.at<float>(rIdx, 3))));
249 faceScores.push_back(faces.at<float>(rIdx, 14));
250 }
251
252 std::vector<int> keepIdx;
253 dnn::NMSBoxes(faceBoxes, faceScores, scoreThreshold, nmsThreshold, keepIdx, 1.f, topK);
254
255 // Get NMS results
256 Mat nms_faces;
257 for (int idx: keepIdx)
258 {
259 nms_faces.push_back(faces.row(idx));
260 }
261 return nms_faces;
262 }
263 else
264 {
265 return faces;
266 }
267 }
