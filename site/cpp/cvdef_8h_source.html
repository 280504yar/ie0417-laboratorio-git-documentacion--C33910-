988// round-half-up (ties away from zero) — deliberately NOT round-to-nearest-even / OCP-ONNX spec
989inline unsigned roundHalfUp(unsigned full, int shift)
990{
991 if (shift <= 0) return full << (-shift);
992 unsigned q = full >> shift, rem = full & ((1u << shift) - 1), half = 1u << (shift - 1);
993 if (rem >= half) q++;
994 return q;
995}

998// E4M3 encode; no inf, bias/fnuz select E4M3FN vs E4M3FNUZ
999inline uchar encodeE4M3(float x, int bias, bool fnuz)
1000{
1001 Cv32suf in; in.f = x;
1002 unsigned u = in.u, sign = (u >> 31) & 1, e = (u >> 23) & 0xFF, m = u & 0x7FFFFF;
1003 const unsigned sbit = sign << 7;
1004 const unsigned nanc = fnuz ? 0x80u : (sbit | 0x7Fu);
1005 if (e == 0xFF) return (uchar)nanc;
1006 if (e == 0 && m == 0) return (uchar)(fnuz ? 0u : sbit);
1007 int newexp = (int)e - 127 + bias;
1008 const unsigned full = (1u << 23) | m;
1009 if (newexp <= 0) // subnormal
1010 {
1011 const int shift = 20 + (1 - newexp);
1012 unsigned mant = (shift >= 32) ? 0u : roundHalfUp(full, shift);
1013 return (uchar)(mant == 0 ? (fnuz ? 0u : sbit) : (sbit | mant));
1014 }
1015 unsigned rounded = roundHalfUp(full, 20);
1016 if (rounded & 16u) { rounded >>= 1; newexp++; } // carry into exponent
1017 const unsigned mant = rounded & 7u;
1018 const bool ov = fnuz ? ((unsigned)newexp > 15)
1019 : ((unsigned)newexp > 15 || ((unsigned)newexp == 15 && mant == 7));
1020 if (ov) return (uchar)nanc;
1021 return (uchar)(sbit | ((unsigned)newexp << 3) | mant);
1022}

1024inline float decodeE4M3(uchar b, int bias, bool fnuz)
1025{
1026 const unsigned sign = ((unsigned)b >> 7) & 1, exp = ((unsigned)b >> 3) & 15, man = (unsigned)b & 7;
1027 const float s = sign ? -1.f : 1.f;
1028 Cv32suf qn; qn.u = sign ? 0xFFC00000u : 0x7FC00000u;
1029 if (fnuz) { if ((unsigned)b == 0x80u) return qn.f; }
1030 else if (exp == 15) { if (man == 7) return qn.f; }
1031 if (exp == 0) return s * (float)man * pow2(1 - bias - 3);
1032 return s * (1.0f + (float)man / 8.0f) * pow2((int)exp - bias);
1033}
