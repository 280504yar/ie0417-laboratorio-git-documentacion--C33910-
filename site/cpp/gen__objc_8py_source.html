178 def parseName(self, name, namespaces):
179 '''
180 input: full name and available namespaces
181 returns: (namespace, classpath, classname, name)
182 '''
183 name = name[name.find(" ")+1:].strip() # remove struct/class/const prefix
184 spaceName = ""
185 localName = name # <classes>.<name>
186 for namespace in sorted(namespaces, key=len, reverse=True):
187 if name.startswith(namespace + "."):
188 spaceName = namespace
189 localName = name.replace(namespace + ".", "")
190 break
191 pieces = localName.split(".")
192 if len(pieces) > 2: # <class>.<class>.<class>.<name>
193 return name, spaceName, ".".join(pieces[:-1]), pieces[-2], pieces[-1]
194 elif len(pieces) == 2: # <class>.<name>
195 return name, spaceName, pieces[0], pieces[0], pieces[1]
196 elif len(pieces) == 1: # <name>
197 return name, spaceName, "", "", pieces[0]
198 else:
199 return name, spaceName, "", "" # error?!
