89bool HdrDecoder::readData(Mat& _img)
90{
91 Mat img(m_height, m_width, CV_32FC3);
92 if(!file) {
93 if(!readHeader()) {
94 return false;
95 }
96 }
97 RGBE_ReadPixels_RLE(file, const_cast<float*>(img.ptr<float>()), img.cols, img.rows);
98 fclose(file); file = NULL;
99
100 // NOTE: 'img' has type CV32FC3
101 switch (_img.depth())
102 {
103 case CV_8U: img.convertTo(img, _img.depth(), 255); break;
104 case CV_32F: break;
105 default: CV_Error(Error::StsError, "Wrong expected image depth, allowed: CV_8U and CV_32F");
106 }
107 switch (_img.channels())
108 {
109 case 1: cvtColor(img, _img, COLOR_BGR2GRAY); break;
110 case 3:
111 // TODO, try to modify RGBE_ReadPixels_RLE to load rgb data directly.
112 if (m_use_rgb)
113 cv::cvtColor(img, _img, cv::COLOR_BGR2RGB);
114 else
115 img.copyTo(_img);
116 break;
117 default: CV_Error(Error::StsError, "Wrong expected image channels, allowed: 1 and 3");
118 }
119 return true;
120}

150bool HdrEncoder::write( const Mat& input_img, const std::vector<int>& params )
151{
152 Mat img;
153 CV_Assert(input_img.channels() == 3 || input_img.channels() == 1);
154 if(input_img.channels() == 1) {
155 std::vector<Mat> splitted(3, input_img);
156 merge(splitted, img);
157 } else {
158 input_img.copyTo(img);
159 }
160 if(img.depth() != CV_32F) {
161 img.convertTo(img, CV_32FC3, 1/255.0f);
162 }
163
164 int compression = IMWRITE_HDR_COMPRESSION_RLE;
165 for (size_t i = 0; i + 1 < params.size(); i += 2)
166 {
167 const int value = params[i+1];
168 switch (params[i])
169 {
170 case IMWRITE_HDR_COMPRESSION:
171 switch(value)
172 {
173 case IMWRITE_HDR_COMPRESSION_NONE:
174 case IMWRITE_HDR_COMPRESSION_RLE:
175 compression = value;
176 break;
177 default:
178 compression = IMWRITE_HDR_COMPRESSION_RLE;
179 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_HDR_COMPRESSION must be one of ImwriteHDRCompressionFlags. It is fallbacked to IMWRITE_HDR_COMPRESSION_RLE", value));
180 break;
181 }
182 break;
183 default:
184 break;
185 }
186 }
187 CV_Check(compression, compression == IMWRITE_HDR_COMPRESSION_NONE || compression == IMWRITE_HDR_COMPRESSION_RLE, "");
188
189 FILE *fout = fopen(m_filename.c_str(), "wb");
190 if(!fout) {
191 return false;
192 }
193
194 RGBE_WriteHeader(fout, img.cols, img.rows, NULL);
195 if (compression == IMWRITE_HDR_COMPRESSION_RLE) {
196 RGBE_WritePixels_RLE(fout, const_cast<float*>(img.ptr<float>()), img.cols, img.rows);
197 } else {
198 RGBE_WritePixels(fout, const_cast<float*>(img.ptr<float>()), img.cols * img.rows);
199 }
200
201 fclose(fout);
202 return true;
203}
