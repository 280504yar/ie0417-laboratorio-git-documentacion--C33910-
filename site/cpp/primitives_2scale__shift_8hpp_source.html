35 template <class T>
36 class ScaleShiftOp final : public CUDABackendNode {
37 public:
38 using wrapper_type = GetCUDABackendWrapperType<T>;
39
40 ScaleShiftOp(csl::Stream stream_, const ScaleShiftConfiguration& config, const cv::Mat& weights, const cv::Mat& bias)
41 : stream(std::move(stream_)), axis{ config.axis }
42 {
43 scaleMode = config.scaleMode;
44 if (scaleMode == ScaleShiftConfiguration::OpMode::TRAINABLE)
45 {
46 CV_Assert(!weights.empty());
47 weightsTensor = csl::makeTensorHeader<T>(weights);
48 csl::copyMatToTensor<T>(weights, weightsTensor, stream);
49 }
50
51 shiftMode = config.shiftMode;
52 if (shiftMode == ScaleShiftConfiguration::OpMode::TRAINABLE)
53 {
54 CV_Assert(!bias.empty());
55 biasTensor = csl::makeTensorHeader<T>(bias);
56 csl::copyMatToTensor<T>(bias, biasTensor, stream);
57 }
58
59 CV_Assert(scaleMode != ScaleShiftConfiguration::OpMode::NONE ||
60 shiftMode != ScaleShiftConfiguration::OpMode::NONE);
61
62 if (scaleMode == ScaleShiftConfiguration::OpMode::UNTRAINABLE &&
63 shiftMode == ScaleShiftConfiguration::OpMode::UNTRAINABLE)
64 {
65 CV_Error(cv::Error::StsNotImplemented, "scale and shift both in untrainable mode is not supported");
66 }
67 }
68
69 void forward(
70 const std::vector<cv::Ptr<BackendWrapper>>& inputs,
71 const std::vector<cv::Ptr<BackendWrapper>>& outputs,
72 csl::Workspace& workspace) override
73 {
74 CV_Assert(outputs.size() == 1);
75
76 auto input_wrapper = inputs[0].dynamicCast<wrapper_type>();
77 auto input = input_wrapper->getView();
78
79 auto output_wrapper = outputs[0].dynamicCast<wrapper_type>();
80 auto output = output_wrapper->getSpan();
81
82 /* number of batches in the weights/bias
83 * trainable mode: same for all batches
84 * untrainable mode: could be different for different batch samples
85 */
86 std::size_t parameter_batch_size = 1;
87
88 csl::TensorView<T> weights;
89 if (scaleMode == ScaleShiftConfiguration::OpMode::TRAINABLE)
90 {
91 CV_Assert(!weightsTensor.empty());
92 weights = csl::TensorView<T>(weightsTensor);
93 }
94 else if (scaleMode == ScaleShiftConfiguration::OpMode::UNTRAINABLE)
95 {
96 CV_Assert(inputs.size() == 2);
97 auto wrapper = inputs[1].dynamicCast<wrapper_type>();
98 weights = wrapper->getView();
99
100 parameter_batch_size = weights.get_axis_size(0);
101 CV_Assert(parameter_batch_size == input.get_axis_size(0));
102 }
103
104 csl::TensorView<T> bias;
105 if (shiftMode == ScaleShiftConfiguration::OpMode::TRAINABLE)
106 {
107 CV_Assert(!biasTensor.empty());
108 bias = csl::TensorView<T>(biasTensor);
109 }
110 else if (shiftMode == ScaleShiftConfiguration::OpMode::UNTRAINABLE)
111 {
112 CV_Assert(inputs.size() == 2);
113 auto wrapper = inputs[1].dynamicCast<wrapper_type>();
114 bias = wrapper->getView();
115
116 parameter_batch_size = bias.get_axis_size(0);
117 CV_Assert(parameter_batch_size == input.get_axis_size(0));
118 }
119
120 CV_Assert(!weights.empty() || !bias.empty());
121 if (!weights.empty() && !bias.empty())
122 {
123 CV_CheckEQ(weights.size(), bias.size(), "different broadcasting options for weights and bias is not supported");
124 }
125
126 const auto num_parameters = !weights.empty() ? weights.size() : bias.size();
127 const auto mid_size = num_parameters / parameter_batch_size;
128
129 /* the scale shift operation might require broadcasting */
130 const int end_axis = [&] {
131 if (num_parameters == 1) {
132 return static_cast<int>(axis + 1);
133 }
134 for (int endAxis = axis + 1; endAxis <= input.rank(); endAxis++) {
135 if (input.size_range(axis, endAxis) == mid_size)
136 return endAxis;
137 }
138 CV_Assert(0 /* failed to find a broadcast config */);
139 }();
140
141 std::size_t inner_size = input.size_range(end_axis, input.rank());
142
143 if (!weights.empty() && !bias.empty())
144 kernels::scaleN_with_biasN<T>(stream, output, input, inner_size, weights, bias);
145 else if (!weights.empty())
146 kernels::scaleN<T>(stream, output, input, inner_size, weights);
147 else
148 kernels::biasN<T>(stream, output, input, inner_size, bias);
149 }
150
151 private:
152 csl::Stream stream;
153 csl::Tensor<T> weightsTensor, biasTensor;
154 std::size_t axis;
155
156 ScaleShiftConfiguration::OpMode scaleMode, shiftMode;
157 };
