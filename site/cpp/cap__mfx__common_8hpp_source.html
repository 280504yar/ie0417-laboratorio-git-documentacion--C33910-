77struct FourCC
78{
79 FourCC(uint val) : val32(val) {}
80 FourCC(char a, char b, char c, char d) { val8[0] = a; val8[1] = b; val8[2] = c; val8[3] = d; }
81 union {
82 uint val32;
83 int vali32;

85 };
