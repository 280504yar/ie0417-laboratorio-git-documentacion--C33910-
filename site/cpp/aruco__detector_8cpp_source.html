131static void _findMarkerContours(const Mat &in, vector<vector<Point2f> > &candidates,
132 vector<vector<Point> > &contoursOut, double minPerimeterRate,
133 double maxPerimeterRate, double accuracyRate,
134 double minCornerDistanceRate, int minSize) {
135
136 CV_Assert(minPerimeterRate > 0 && maxPerimeterRate > 0 && accuracyRate > 0 &&
137 minCornerDistanceRate >= 0);
138
139 // calculate maximum and minimum sizes in pixels
140 unsigned int minPerimeterPixels =
141 (unsigned int)(minPerimeterRate * max(in.cols, in.rows));
142 unsigned int maxPerimeterPixels =
143 (unsigned int)(maxPerimeterRate * max(in.cols, in.rows));
144
145 // for aruco3 functionality
146 if (minSize != 0) {
147 minPerimeterPixels = 4*minSize;
148 }
149
150 vector<vector<Point> > contours;
151 findContours(in, contours, RETR_LIST, CHAIN_APPROX_NONE);
152 // now filter list of contours
153 for(unsigned int i = 0; i < contours.size(); i++) {
154 // check perimeter
155 if(contours[i].size() < minPerimeterPixels || contours[i].size() > maxPerimeterPixels)
156 continue;
157
158 // check is square and is convex
159 vector<Point> approxCurve;
160 approxPolyDP(contours[i], approxCurve, double(contours[i].size()) * accuracyRate, true);
161 if(approxCurve.size() != 4 || !isContourConvex(approxCurve)) continue;
162
163 // check min distance between corners
164 double minDistSq = max(in.cols, in.rows) * max(in.cols, in.rows);
165 for(int j = 0; j < 4; j++) {
166 double d = (double)(approxCurve[j].x - approxCurve[(j + 1) % 4].x) *
167 (double)(approxCurve[j].x - approxCurve[(j + 1) % 4].x) +
168 (double)(approxCurve[j].y - approxCurve[(j + 1) % 4].y) *
169 (double)(approxCurve[j].y - approxCurve[(j + 1) % 4].y);
170 minDistSq = min(minDistSq, d);
171 }
172 double minCornerDistancePixels = double(contours[i].size()) * minCornerDistanceRate;
173 if(minDistSq < minCornerDistancePixels * minCornerDistancePixels) continue;
174
175 // if it passes all the test, add to candidates vector
176 vector<Point2f> currentCandidate;
177 currentCandidate.resize(4);
178 for(int j = 0; j < 4; j++) {
179 currentCandidate[j] = Point2f((float)approxCurve[j].x, (float)approxCurve[j].y);
180 }
181 candidates.push_back(currentCandidate);
182 contoursOut.push_back(contours[i]);
183 }
184}

190static void _reorderCandidatesCorners(vector<vector<Point2f> > &candidates) {
191
192 for(unsigned int i = 0; i < candidates.size(); i++) {
193 double dx1 = candidates[i][1].x - candidates[i][0].x;
194 double dy1 = candidates[i][1].y - candidates[i][0].y;
195 double dx2 = candidates[i][2].x - candidates[i][0].x;
196 double dy2 = candidates[i][2].y - candidates[i][0].y;
197 double crossProduct = (dx1 * dy2) - (dy1 * dx2);
198
199 if(crossProduct < 0.0) { // not clockwise direction
200 swap(candidates[i][1], candidates[i][3]);
201 }
202 }

205static float getAverageModuleSize(const vector<Point2f>& markerCorners, int markerSize, int markerBorderBits) {
206 float averageArucoModuleSize = 0.f;
207 for (size_t i = 0ull; i < 4ull; i++) {

249// returns the average distance between the marker points
250float static inline getAverageDistance(const std::vector<Point2f>& marker1, const std::vector<Point2f>& marker2) {
251 float minDistSq = std::numeric_limits<float>::max();
252 // fc is the first corner considered on one of the markers, 4 combinations are possible
253 for(int fc = 0; fc < 4; fc++) {
254 float distSq = 0;
255 for(int c = 0; c < 4; c++) {
256 // modC is the corner considering first corner is fc
257 int modC = (c + fc) % 4;
258 distSq += normL2Sqr<float>(marker1[modC] - marker2[c]);
259 }
260 distSq /= 4.f;
261 minDistSq = min(minDistSq, distSq);
262 }
263 return sqrt(minDistSq);
264}

269static void _detectInitialCandidates(const Mat &grey, vector<vector<Point2f> > &candidates,
270 vector<vector<Point> > &contours,
271 const DetectorParameters &params) {
272
273 CV_Assert(params.adaptiveThreshWinSizeMin >= 3 && params.adaptiveThreshWinSizeMax >= 3);
274 CV_Assert(params.adaptiveThreshWinSizeMax >= params.adaptiveThreshWinSizeMin);
275 CV_Assert(params.adaptiveThreshWinSizeStep > 0);
276
277 // number of window sizes (scales) to apply adaptive thresholding
278 int nScales = (params.adaptiveThreshWinSizeMax - params.adaptiveThreshWinSizeMin) /
279 params.adaptiveThreshWinSizeStep + 1;
280
281 vector<vector<vector<Point2f> > > candidatesArrays((size_t) nScales);
282 vector<vector<vector<Point> > > contoursArrays((size_t) nScales);
283
285 parallel_for_(Range(0, nScales), [&](const Range& range) {
286 const int begin = range.start;
287 const int end = range.end;
288
289 for (int i = begin; i < end; i++) {
290 int currScale = params.adaptiveThreshWinSizeMin + i * params.adaptiveThreshWinSizeStep;
291 // threshold
292 Mat thresh;
293 _threshold(grey, thresh, currScale, params.adaptiveThreshConstant);
294
295 // detect rectangles
296 _findMarkerContours(thresh, candidatesArrays[i], contoursArrays[i],
297 params.minMarkerPerimeterRate, params.maxMarkerPerimeterRate,
298 params.polygonalApproxAccuracyRate, params.minCornerDistanceRate,
299 params.minSideLengthCanonicalImg);
300 }
301 });
302 // join candidates
303 for(int i = 0; i < nScales; i++) {
304 for(unsigned int j = 0; j < candidatesArrays[i].size(); j++) {
305 candidates.push_back(candidatesArrays[i][j]);
306 contours.push_back(contoursArrays[i][j]);
307 }
308 }
309}

316static Mat _extractCellPixelRatio(InputArray _image, const vector<Point2f>& corners, int markerSize,
317 int markerBorderBits, int cellSize, double cellMarginRate, double minStdDevOtsu) {
318 CV_Assert(_image.getMat().channels() == 1);
319 CV_Assert(corners.size() == 4ull);
320 CV_Assert(markerBorderBits > 0 && cellSize > 0 && cellMarginRate >= 0 && cellMarginRate <= 0.5);
321 CV_Assert(minStdDevOtsu >= 0);
322
323 // number of bits in the marker
324 int markerSizeWithBorders = markerSize + 2 * markerBorderBits;
325 int cellMarginPixels = int(cellMarginRate * cellSize);
326
327 Mat resultImg; // marker image after removing perspective
328 int resultImgSize = markerSizeWithBorders * cellSize;
329 Mat resultImgCorners(4, 1, CV_32FC2);
330 resultImgCorners.ptr<Point2f>(0)[0] = Point2f(0, 0);
331 resultImgCorners.ptr<Point2f>(0)[1] = Point2f((float)resultImgSize - 1, 0);
332 resultImgCorners.ptr<Point2f>(0)[2] =
333 Point2f((float)resultImgSize - 1, (float)resultImgSize - 1);
334 resultImgCorners.ptr<Point2f>(0)[3] = Point2f(0, (float)resultImgSize - 1);
335
336 // remove perspective
337 Mat transformation = getPerspectiveTransform(corners, resultImgCorners);
338 warpPerspective(_image, resultImg, transformation, Size(resultImgSize, resultImgSize),
339 INTER_NEAREST);
340
341 // output image containing the ratio of white pixels in each cell
342 Mat cellPixelRatio(markerSizeWithBorders, markerSizeWithBorders, CV_32FC1, Scalar::all(0));
343
344 // check if standard deviation is enough to apply Otsu
345 // if not enough, it probably means all pixels are the same color (black or white)
346 Mat mean, stddev;
347 // Remove some border just to avoid border noise from perspective transformation
348 Mat innerRegion = resultImg.colRange(cellSize / 2, resultImg.cols - cellSize / 2)
349 .rowRange(cellSize / 2, resultImg.rows - cellSize / 2);
350 meanStdDev(innerRegion, mean, stddev);
351 if(stddev.ptr< double >(0)[0] < minStdDevOtsu) {
352 // all black or all white, depending on mean value
353 if(mean.ptr< double >(0)[0] > 127){
354 cellPixelRatio.setTo(1);
355 } else {
356 cellPixelRatio.setTo(0);
357 }
358
359 return cellPixelRatio;
360 }
361
362 // now extract code, first threshold using Otsu
363 threshold(resultImg, resultImg, 125, 255, THRESH_BINARY | THRESH_OTSU);
364
365 // for each cell
366 for(int y = 0; y < markerSizeWithBorders; y++) {
367 for(int x = 0; x < markerSizeWithBorders; x++) {
368 int Xstart = x * (cellSize) + cellMarginPixels;
369 int Ystart = y * (cellSize) + cellMarginPixels;
370 Mat square = resultImg(Rect(Xstart, Ystart, cellSize - 2 * cellMarginPixels,
371 cellSize - 2 * cellMarginPixels));
372 // count white pixels on each cell to assign its value
373 size_t nZ = (size_t) countNonZero(square);
374
375 // define the cell pixel ratio as the ratio of the white pixels. For inverted markers, the ratio will be inverted.
376 cellPixelRatio.at<float>(y, x) = (nZ / (float)square.total());
377 }
378 }
379
380 return cellPixelRatio;
381}

392static void _getBorderErrors(const Mat &cellPixelRatio, int markerSize, int borderSize, float validBitIdThreshold,
393 int &borderErrors, int &invBorderErrors) {
394
395 int sizeWithBorders = markerSize + 2 * borderSize;
396
397 CV_Assert(markerSize > 0 && cellPixelRatio.cols == sizeWithBorders && cellPixelRatio.rows == sizeWithBorders);
398
399 // Get border error. cellPixelRatio has the opposite color as the borders.
400 const float invThreshold = 1.f - validBitIdThreshold;
401 borderErrors = invBorderErrors = 0;
402 const auto countCell = [&](float ratio) {
403 if(ratio > validBitIdThreshold) borderErrors++;
404 if(ratio < invThreshold) invBorderErrors++;
405 };
406 for(int y = 0; y < sizeWithBorders; y++) {
407 const float* row = cellPixelRatio.ptr<float>(y);
408 for(int k = 0; k < borderSize; k++) {
409 // Left and right vertical sides
410 countCell(row[k]);
411 countCell(row[sizeWithBorders - 1 - k]);
412 }
413 }
414 for(int x = borderSize; x < sizeWithBorders - borderSize; x++) {
415 for(int k = 0; k < borderSize; k++) {
416 // Top and bottom horizontal sides
417 countCell(cellPixelRatio.ptr<float>(k)[x]);
418 countCell(cellPixelRatio.ptr<float>(sizeWithBorders - 1 - k)[x]);
419 }
420 }
421}

429static float _getMarkerConfidence(const Mat& groundTruthbits, const Mat &cellPixelRatio, const int markerSize, const int borderSize) {
430
431 CV_Assert(markerSize == groundTruthbits.cols && markerSize == groundTruthbits.rows);
432
433 const int sizeWithBorders = markerSize + 2 * borderSize;
434 CV_Assert(markerSize > 0 && cellPixelRatio.cols == sizeWithBorders && cellPixelRatio.rows == sizeWithBorders);
435
436 // Get border uncertainty. cellPixelRatio has the opposite color as the borders --> it is the uncertainty.
437 float tempBorderUnc = 0.f;
438 for(int y = 0; y < sizeWithBorders; y++) {
439 for(int k = 0; k < borderSize; k++) {
440 // Left and right vertical sides
441 tempBorderUnc += cellPixelRatio.ptr<float>(y)[k];
442 tempBorderUnc += cellPixelRatio.ptr<float>(y)[sizeWithBorders - 1 - k];
443 }
444 }
445 for(int x = borderSize; x < sizeWithBorders - borderSize; x++) {
446 for(int k = 0; k < borderSize; k++) {
447 // Top and bottom horizontal sides
448 tempBorderUnc += cellPixelRatio.ptr<float>(k)[x];
449 tempBorderUnc += cellPixelRatio.ptr<float>(sizeWithBorders - 1 - k)[x];
450 }
451 }
452
453 // Get the inner marker uncertainty. For a white or black cell, the uncertainty is the ratio of black or white pixels respectively.
454 float tempInnerUnc = 0.f;
455 for(int y = borderSize; y < markerSize + borderSize; y++) {
456 for(int x = borderSize; x < markerSize + borderSize; x++) {
457 tempInnerUnc += std::abs(groundTruthbits.ptr<float>(y - borderSize)[x - borderSize] - cellPixelRatio.ptr<float>(y)[x]);
458 }
459 }
460
461 // Compute the overall normalized marker uncertainty and convert it to confidence
462 const float area = static_cast<float>(sizeWithBorders) * sizeWithBorders;
463 const float normalizedMarkerUnc = (tempInnerUnc + tempBorderUnc) / area;
464 const float normalizedMarkerConfidence = 1.f - normalizedMarkerUnc;
465
466 return std::max(0.f, std::min(1.f, normalizedMarkerConfidence));
467}

475static uint8_t _identifyOneCandidate(const Dictionary& dictionary, const Mat& _image,
476 const vector<Point2f>& _corners, int& idx,
477 const DetectorParameters& params, int& rotation,
478 float &markerConfidence, bool confidenceNeeded,
479 const float scale = 1.f) {
480 CV_DbgAssert(params.markerBorderBits > 0);
481 uint8_t typ=1;
482 // get bits
483 // scale corners to the correct size to search on the corresponding image pyramid
484 vector<Point2f> scaled_corners(4);
485 for (int i = 0; i < 4; ++i) {
486 scaled_corners[i].x = _corners[i].x * scale;
487 scaled_corners[i].y = _corners[i].y * scale;
488 }
489
490 Mat cellPixelRatio =
491 _extractCellPixelRatio(_image, scaled_corners, dictionary.markerSize, params.markerBorderBits,
492 params.perspectiveRemovePixelPerCell,
493 params.perspectiveRemoveIgnoredMarginPerCell, params.minOtsuStdDev);
494
495 // analyze border bits
496 int maximumErrorsInBorder =
497 int(dictionary.markerSize * dictionary.markerSize * params.maxErroneousBitsInBorderRate);
498 int borderErrors = 0, invBorderErrors = 0;
499 _getBorderErrors(cellPixelRatio, dictionary.markerSize, params.markerBorderBits, params.validBitIdThreshold,
500 borderErrors, invBorderErrors);
501
502 // check if it is a white marker
503 if(params.detectInvertedMarker && invBorderErrors < borderErrors) {
504 // white marker: invert the observed ratios in place and continue as a normal marker
505 borderErrors = invBorderErrors;
506 subtract(Scalar::all(1), cellPixelRatio, cellPixelRatio);
507 typ=2;
508 }
509 if(borderErrors > maximumErrorsInBorder) return 0; // border is wrong
510
511 // take only inner bits
512 Mat onlyCellPixelRatio = cellPixelRatio(
513 Rect(params.markerBorderBits, params.markerBorderBits,
514 cellPixelRatio.cols - 2 * params.markerBorderBits,
515 cellPixelRatio.rows - 2 * params.markerBorderBits));
516
517 // try to identify the marker
518 if(!dictionary.identify(onlyCellPixelRatio, idx, rotation, params.errorCorrectionRate, params.validBitIdThreshold))
519 return 0;
520
521 // compute the candidate's confidence
522 if(confidenceNeeded) {
523 Mat groundTruthbits = dictionary.getMarkerBits(idx, rotation);
524 markerConfidence = _getMarkerConfidence(groundTruthbits, cellPixelRatio, dictionary.markerSize, params.markerBorderBits);
525 }
526
527 return typ;
528}

537static size_t _findOptPyrImageForCanonicalImg(
538 const vector<Mat>& img_pyr,
539 const int scaled_width,
540 const int cur_perimeter,
541 const int min_perimeter) {
542 CV_Assert(scaled_width > 0);
543 size_t optLevel = 0;
544 float dist = std::numeric_limits<float>::max();
545 for (size_t i = 0; i < img_pyr.size(); ++i) {
546 const float scale = img_pyr[i].cols / static_cast<float>(scaled_width);
547 const float perimeter_scaled = cur_perimeter * scale;
548 // instead of std::abs() favor the larger pyramid level by checking if the distance is postive
549 // will slow down the algorithm but find more corners in the end
550 const float new_dist = perimeter_scaled - min_perimeter;
551 if (new_dist < dist && new_dist > 0.f) {
552 dist = new_dist;
553 optLevel = i;
554 }
555 }
556 return optLevel;
557}

564static Point3f _interpolate2Dline(const vector<Point2f>& nContours){
565 CV_Assert(nContours.size() >= 2);
566 float minX, minY, maxX, maxY;
567 minX = maxX = nContours[0].x;
568 minY = maxY = nContours[0].y;
569
570 for(unsigned int i = 0; i< nContours.size(); i++){
571 minX = nContours[i].x < minX ? nContours[i].x : minX;
572 minY = nContours[i].y < minY ? nContours[i].y : minY;
573 maxX = nContours[i].x > maxX ? nContours[i].x : maxX;
574 maxY = nContours[i].y > maxY ? nContours[i].y : maxY;
575 }
576
577 Mat A = Mat::ones((int)nContours.size(), 2, CV_32F); // Coefficient Matrix (N x 2)
578 Mat B((int)nContours.size(), 1, CV_32F); // Variables Matrix (N x 1)
579 Mat C; // Constant
580
581 if(maxX - minX > maxY - minY){
582 for(unsigned int i =0; i < nContours.size(); i++){
583 A.at<float>(i,0)= nContours[i].x;
584 B.at<float>(i,0)= nContours[i].y;
585 }
586
587 solve(A, B, C, DECOMP_NORMAL);
588
589 return Point3f(C.at<float>(0, 0), -1., C.at<float>(1, 0));
590 }
591 else{
592 for(unsigned int i =0; i < nContours.size(); i++){
593 A.at<float>(i,0)= nContours[i].y;
594 B.at<float>(i,0)= nContours[i].x;
595 }
596
597 solve(A, B, C, DECOMP_NORMAL);
598
599 return Point3f(-1., C.at<float>(0, 0), C.at<float>(1, 0));
600 }

621static void _refineCandidateLines(vector<Point>& nContours, vector<Point2f>& nCorners){
622 vector<Point2f> contour2f(nContours.begin(), nContours.end());
623 /* 5 groups :: to group the edges
624 * 4 - classified by its corner
625 * extra group - (temporary) if contours do not begin with a corner
626 */
627 vector<Point2f> cntPts[5];
628 int cornerIndex[4]={-1};
629 int group=4;
630
631 for ( unsigned int i =0; i < nContours.size(); i++ ) {
632 for(unsigned int j=0; j<4; j++){
633 if ( nCorners[j] == contour2f[i] ){
634 cornerIndex[j] = i;
635 group=j;
636 }
637 }
638 cntPts[group].push_back(contour2f[i]);
639 }
640 for (int i = 0; i < 4; i++)
641 {
642 CV_Assert(cornerIndex[i] != -1);
643 }
644 // saves extra group into corresponding
645 if( !cntPts[4].empty() ){
646 for( unsigned int i=0; i < cntPts[4].size() ; i++ )
647 cntPts[group].push_back(cntPts[4].at(i));
648 cntPts[4].clear();
649 }
650
651 //Evaluate contour direction :: using the position of the detected corners
652 int inc=1;
653
654 inc = ( (cornerIndex[0] > cornerIndex[1]) && (cornerIndex[3] > cornerIndex[0]) ) ? -1:inc;
655 inc = ( (cornerIndex[2] > cornerIndex[3]) && (cornerIndex[1] > cornerIndex[2]) ) ? -1:inc;
656
657 // calculate the line :: who passes through the grouped points
658 Point3f lines[4];
659 for(int i=0; i<4; i++){
660 lines[i]=_interpolate2Dline(cntPts[i]);
661 }
662
663 /*
664 * calculate the corner :: where the lines crosses to each other
665 * clockwise direction no clockwise direction
666 * 0 1
667 * .---. 1 .---. 2
668 * | | | |
669 * 3 .___. 0 .___.
670 * 2 3
671 */
672 for(int i=0; i < 4; i++){
673 if(inc<0)
674 nCorners[i] = _getCrossPoint(lines[ i ], lines[ (i+1)%4 ]); // 01 12 23 30
675 else
676 nCorners[i] = _getCrossPoint(lines[ i ], lines[ (i+3)%4 ]); // 30 01 12 23
677 }
678}

680static inline void findCornerInPyrImage(const float scale_init, const int closest_pyr_image_idx,
681 const vector<Mat>& grey_pyramid, Mat corners,
682 const DetectorParameters& params) {
683 // scale them to the closest pyramid level
684 if (scale_init != 1.f)
685 corners *= scale_init; // scale_init * scale_pyr
686 for (int idx = closest_pyr_image_idx - 1; idx >= 0; --idx) {
687 // scale them to new pyramid level
688 corners *= 2.f; // *= scale_pyr;
689 // use larger win size for larger images
690 const int subpix_win_size = std::max(grey_pyramid[idx].cols, grey_pyramid[idx].rows) > 1080 ? 5 : 3;
691 cornerSubPix(grey_pyramid[idx], corners,
692 Size(subpix_win_size, subpix_win_size),
693 Size(-1, -1),
694 TermCriteria(TermCriteria::MAX_ITER | TermCriteria::EPS,
695 params.cornerRefinementMaxIterations,
696 params.cornerRefinementMinAccuracy));
697 }

727 CV_Assert(!_image.empty());
728
729 CV_Assert(detectorParams.markerBorderBits > 0);

731 CV_Assert(!(detectorParams.useAruco3Detection == true &&
732 detectorParams.minSideLengthCanonicalImg == 0 &&
733 detectorParams.minMarkerLengthRatioOriginalImg == 0.0));
734
735 Mat grey;
736 _convertToGrey(_image, grey);
737
738 // Aruco3 functionality is the extension of Aruco.
739 // The description can be found in:
740 // [1] Speeded up detection of squared fiducial markers, 2018, FJ Romera-Ramirez et al.
741 // if Aruco3 functionality if not wanted
742 // change some parameters to be sure to turn it off
743 if (!detectorParams.useAruco3Detection) {
744 detectorParams.minMarkerLengthRatioOriginalImg = 0.0;
745 detectorParams.minSideLengthCanonicalImg = 0;
746 }
747 else {
748 // always turn on corner refinement in case of Aruco3, due to upsampling
749 detectorParams.cornerRefinementMethod = (int)CORNER_REFINE_SUBPIX;
750 // only CORNER_REFINE_SUBPIX implement correctly for useAruco3Detection
751 // Todo: update other CORNER_REFINE methods
752 }
753
755 const float fxfy = (!detectorParams.useAruco3Detection ? 1.f : detectorParams.minSideLengthCanonicalImg /
756 (detectorParams.minSideLengthCanonicalImg + std::max(grey.cols, grey.rows)*
757 detectorParams.minMarkerLengthRatioOriginalImg));
758
760 vector<Mat> grey_pyramid;
761 int closest_pyr_image_idx = 0, num_levels = 0;
763 if (detectorParams.useAruco3Detection) {
764 const float scale_pyr = 2.f;
765 const float img_area = static_cast<float>(grey.rows*grey.cols);
766 const float min_area_marker = static_cast<float>(detectorParams.minSideLengthCanonicalImg*
767 detectorParams.minSideLengthCanonicalImg);
768 // find max level
769 num_levels = static_cast<int>(log2(img_area / min_area_marker)/scale_pyr);
770 // the closest pyramid image to the downsampled segmentation image
771 // will later be used as start index for corner upsampling
772 const float scale_img_area = img_area * fxfy * fxfy;
773 closest_pyr_image_idx = cvRound(log2(img_area / scale_img_area)/scale_pyr);
774 }
775 buildPyramid(grey, grey_pyramid, num_levels);
776
777 // resize to segmentation image
778 // in this reduces size the contours will be detected
779 if (fxfy != 1.f)
780 resize(grey, grey, Size(cvRound(fxfy * grey.cols), cvRound(fxfy * grey.rows)));
781
783 vector<vector<Point2f> > candidates;
784 vector<vector<Point> > contours;
785 vector<int> ids;
786 vector<float> markersConfidence;
787
789 if(detectorParams.cornerRefinementMethod == (int)CORNER_REFINE_APRILTAG){
790 _apriltag(grey, detectorParams, candidates, contours);
791 }
793 else {
794 detectCandidates(grey, candidates, contours);
795 }
796
798 vector<int> dictIndices;
799 vector<vector<Point2f>> rejectedImgPoints;
800 if (DictionaryMode::Single == dictMode) {
801 Dictionary& dictionary = dictionaries.at(0);
802 auto selectedCandidates = filterTooCloseCandidates(grey.size(), candidates, contours, dictionary.markerSize);
803 candidates.clear();
804 contours.clear();
805
807 identifyCandidates(grey, grey_pyramid, selectedCandidates, candidates, contours,
808 ids, dictionary, rejectedImgPoints, markersConfidence, _markersConfidence.needed());
809
811 if (detectorParams.cornerRefinementMethod == (int)CORNER_REFINE_SUBPIX) {
812 performCornerSubpixRefinement(grey, grey_pyramid, closest_pyr_image_idx, candidates, dictionary);
813 }
814 } else if (DictionaryMode::Multi == dictMode) {
815 map<int, vector<MarkerCandidateTree>> candidatesPerDictionarySize;
816 for (const Dictionary& dictionary : dictionaries) {
817 candidatesPerDictionarySize.emplace(dictionary.markerSize, vector<MarkerCandidateTree>());
818 }
819
820 // create candidate trees for each dictionary size
821 for (auto& candidatesTreeEntry : candidatesPerDictionarySize) {
822 // copy candidates
823 vector<vector<Point2f>> candidatesCopy = candidates;
824 vector<vector<Point> > contoursCopy = contours;
825 candidatesTreeEntry.second = filterTooCloseCandidates(grey.size(), candidatesCopy, contoursCopy, candidatesTreeEntry.first);
826 }
827 candidates.clear();
828 contours.clear();
829
831 int dictIndex = 0;
832 for (const Dictionary& currentDictionary : dictionaries) {
833 // temporary variable to store the current candidates
834 vector<vector<Point2f>> currentCandidates;
835 identifyCandidates(grey, grey_pyramid, candidatesPerDictionarySize.at(currentDictionary.markerSize), currentCandidates, contours,
836 ids, currentDictionary, rejectedImgPoints, markersConfidence, _markersConfidence.needed());
837 if (_dictIndices.needed()) {
838 dictIndices.insert(dictIndices.end(), currentCandidates.size(), dictIndex);
839 }
840
842 if (detectorParams.cornerRefinementMethod == (int)CORNER_REFINE_SUBPIX) {
843 performCornerSubpixRefinement(grey, grey_pyramid, closest_pyr_image_idx, currentCandidates, currentDictionary);
844 }
845 candidates.insert(candidates.end(), currentCandidates.begin(), currentCandidates.end());
846 dictIndex++;
847 }
848
849 // Clean up rejectedImgPoints by comparing to itself and all candidates
850 const float epsilon = 0.000001f;
851 auto compareCandidates = [epsilon](vector<Point2f> a, vector<Point2f> b) {
852 for (int i = 0; i < 4; i++) {
853 if (std::abs(a[i].x - b[i].x) > epsilon || std::abs(a[i].y - b[i].y) > epsilon) {
854 return false;
855 }
856 }
857 return true;
858 };
859 std::sort(rejectedImgPoints.begin(), rejectedImgPoints.end(), [](const vector<Point2f>& a, const vector<Point2f>&b){
860 float avgX = (a[0].x + a[1].x + a[2].x + a[3].x)*.25f;
861 float avgY = (a[0].y + a[1].y + a[2].y + a[3].y)*.25f;
862 float aDist = avgX*avgX + avgY*avgY;
863 avgX = (b[0].x + b[1].x + b[2].x + b[3].x)*.25f;
864 avgY = (b[0].y + b[1].y + b[2].y + b[3].y)*.25f;
865 float bDist = avgX*avgX + avgY*avgY;
866 return aDist < bDist;
867 });
868 auto last = std::unique(rejectedImgPoints.begin(), rejectedImgPoints.end(), compareCandidates);
869 rejectedImgPoints.erase(last, rejectedImgPoints.end());
870
871 for (auto it = rejectedImgPoints.begin(); it != rejectedImgPoints.end();) {
872 bool erased = false;
873 for (const auto& candidate : candidates) {
874 if (compareCandidates(candidate, *it)) {
875 it = rejectedImgPoints.erase(it);
876 erased = true;
877 break;
878 }
879 }
880 if (!erased) {
881 it++;
882 }
883 }
884 }
885
887 if (detectorParams.cornerRefinementMethod == (int)CORNER_REFINE_CONTOUR){
888
889 if (!ids.empty()) {
890
891 // do corner refinement using the contours for each detected markers
892 parallel_for_(Range(0, (int)candidates.size()), [&](const Range& range) {
893 for (int i = range.start; i < range.end; i++) {
894 _refineCandidateLines(contours[i], candidates[i]);
895 }
896 });
897 }
898 }
899
900 if (detectorParams.cornerRefinementMethod != (int)CORNER_REFINE_SUBPIX && fxfy != 1.f) {
901 // only CORNER_REFINE_SUBPIX implement correctly for useAruco3Detection
902 // Todo: update other CORNER_REFINE methods
903
904 // scale to original size, this however will lead to inaccurate detections!
905 for (auto &vecPoints : candidates)
906 for (auto &point : vecPoints)
907 point *= 1.f/fxfy;
908 }
909
910 // copy to output arrays
911 _copyVector2Output(candidates, _corners);
912 Mat(ids).copyTo(_ids);
913 if(_rejectedImgPoints.needed()) {
914 _copyVector2Output(rejectedImgPoints, _rejectedImgPoints);
915 }
916 if (_dictIndices.needed()) {
917 Mat(dictIndices).copyTo(_dictIndices);
918 }
919 if (_markersConfidence.needed()) {
920 Mat(markersConfidence).copyTo(_markersConfidence);
921 }

940 vector<MarkerCandidateTree>
941 filterTooCloseCandidates(const Size &imageSize, vector<vector<Point2f> > &candidates, vector<vector<Point> > &contours, int markerSize) {
942 CV_Assert(detectorParams.minMarkerDistanceRate >= 0. && detectorParams.minDistanceToBorder >= 0);
943 vector<MarkerCandidateTree> candidateTree(candidates.size());
944 for(size_t i = 0ull; i < candidates.size(); i++) {
945 candidateTree[i] = MarkerCandidateTree(std::move(candidates[i]), std::move(contours[i]));
946 }
947
948 // sort candidates from big to small
949 std::stable_sort(candidateTree.begin(), candidateTree.end());
950 // group index for each candidate
951 vector<int> groupId(candidateTree.size(), -1);
952 vector<vector<size_t> > groupedCandidates;
953 vector<bool> isSelectedContours(candidateTree.size(), true);
954
955 for (size_t i = 0ull; i < candidateTree.size(); i++) {
956 for (size_t j = i + 1ull; j < candidateTree.size(); j++) {
957 float minDist = getAverageDistance(candidateTree[i].corners, candidateTree[j].corners);
958 // if mean distance is too low, group markers
959 // the distance between the points of two independent markers should be more than half the side of the marker
960 // half the side of the marker = (perimeter / 4) * 0.5 = perimeter * 0.125
961 if(minDist < candidateTree[j].perimeter*(float)detectorParams.minMarkerDistanceRate) {
962 isSelectedContours[i] = false;
963 isSelectedContours[j] = false;
964 // i and j are not related to a group
965 if(groupId[i] < 0 && groupId[j] < 0){
966 // mark candidates with their corresponding group number
967 groupId[i] = groupId[j] = (int)groupedCandidates.size();
968 // create group
969 groupedCandidates.push_back({i, j});
970 }
971 // i is related to a group
972 else if(groupId[i] > -1 && groupId[j] == -1) {
973 int group = groupId[i];
974 groupId[j] = group;
975 // add to group
976 groupedCandidates[group].push_back(j);
977 }
978 // j is related to a group
979 else if(groupId[j] > -1 && groupId[i] == -1) {
980 int group = groupId[j];
981 groupId[i] = group;
982 // add to group
983 groupedCandidates[group].push_back(i);
984 }
985 }
986 }
987 // group of one candidate
988 if(isSelectedContours[i]) {
989 isSelectedContours[i] = false;
990 groupId[i] = (int)groupedCandidates.size();
991 groupedCandidates.push_back({i});
992 }
993 }
994
995 for (vector<size_t>& grouped : groupedCandidates) {
996 if (detectorParams.detectInvertedMarker) // if detectInvertedMarker choose smallest contours
997 std::stable_sort(grouped.begin(), grouped.end(), [](const size_t &a, const size_t &b) {
998 return a > b;
999 });
1000 else // if detectInvertedMarker==false choose largest contours
1001 std::stable_sort(grouped.begin(), grouped.end());
1002 size_t currId = grouped[0];
1003 // check if it is too near to the image border
1004 bool tooNearBorder = false;
1005 for (const auto& corner : candidateTree[currId].corners) {
1006 if (corner.x < detectorParams.minDistanceToBorder ||
1007 corner.y < detectorParams.minDistanceToBorder ||
1008 corner.x > imageSize.width - 1 - detectorParams.minDistanceToBorder ||
1009 corner.y > imageSize.height - 1 - detectorParams.minDistanceToBorder) {
1010 tooNearBorder = true;
1011 break;
1012 }
1013 }
1014 if (tooNearBorder) {
1015 continue;
1016 }
1017 isSelectedContours[currId] = true;
1018 for (size_t i = 1ull; i < grouped.size(); i++) {
1019 size_t id = grouped[i];
1020 float dist = getAverageDistance(candidateTree[id].corners, candidateTree[currId].corners);
1021 float moduleSize = getAverageModuleSize(candidateTree[id].corners, markerSize, detectorParams.markerBorderBits);
1022 if (dist > detectorParams.minGroupDistance*moduleSize) {
1023 currId = id;
1024 candidateTree[grouped[0]].closeContours.push_back(candidateTree[id]);
1025 }
1026 }
1027 }
1028
1029 vector<MarkerCandidateTree> selectedCandidates;
1030 selectedCandidates.reserve(groupedCandidates.size());
1031 for (size_t i = 0ull; i < candidateTree.size(); i++) {
1032 if (isSelectedContours[i]) {
1033 selectedCandidates.push_back(std::move(candidateTree[i]));
1034 }
1035 }
1036
1037 // find hierarchy in the candidate tree
1038 for (int i = (int)selectedCandidates.size()-1; i >= 0; i--) {
1039 for (int j = i - 1; j >= 0; j--) {
1040 if (checkMarker1InMarker2(selectedCandidates[i].corners, selectedCandidates[j].corners)) {
1041 selectedCandidates[i].parent = j;
1042 selectedCandidates[j].depth = max(selectedCandidates[j].depth, selectedCandidates[i].depth + 1);
1043 break;
1044 }
1045 }
1046 }
1047 return selectedCandidates;
1048 }

1053 void identifyCandidates(const Mat& grey, const vector<Mat>& image_pyr, vector<MarkerCandidateTree>& selectedContours,
1054 vector<vector<Point2f> >& accepted, vector<vector<Point> >& contours,
1055 vector<int>& ids, const Dictionary& currentDictionary, vector<vector<Point2f>>& rejected, vector<float>& markersConfidence, bool confidenceNeeded) const {
1056 size_t ncandidates = selectedContours.size();
1057
1058 vector<float> markersConfidenceTmp(ncandidates, 0.f);
1059 vector<int> idsTmp(ncandidates, -1);
1060 vector<int> rotated(ncandidates, 0);
1061 vector<uint8_t> validCandidates(ncandidates, 0);
1062 vector<uint8_t> was(ncandidates, false);
1063 bool checkCloseContours = true;
1064
1065 int maxDepth = 0;
1066 for (size_t i = 0ull; i < selectedContours.size(); i++)
1067 maxDepth = max(selectedContours[i].depth, maxDepth);
1068 vector<vector<size_t>> depths(maxDepth+1);
1069 for (size_t i = 0ull; i < selectedContours.size(); i++) {
1070 depths[selectedContours[i].depth].push_back(i);
1071 }
1072
1074 int depth = 0;
1075 size_t counter = 0;
1076 while (counter < ncandidates) {
1077 parallel_for_(Range(0, (int)depths[depth].size()), [&](const Range& range) {
1078 const int begin = range.start;
1079 const int end = range.end;
1080 for (int i = begin; i < end; i++) {
1081 size_t v = depths[depth][i];
1082 was[v] = true;
1083 Mat img = grey;
1084 // implements equation (4)
1085 if (detectorParams.useAruco3Detection) {
1086 const int minPerimeter = detectorParams.minSideLengthCanonicalImg * 4;
1087 const size_t nearestImgId = _findOptPyrImageForCanonicalImg(image_pyr, grey.cols, static_cast<int>(selectedContours[v].contour.size()), minPerimeter);
1088 img = image_pyr[nearestImgId];
1089 }
1090 const float scale = detectorParams.useAruco3Detection ? img.cols / static_cast<float>(grey.cols) : 1.f;
1091
1092 validCandidates[v] = _identifyOneCandidate(currentDictionary, img, selectedContours[v].corners, idsTmp[v], detectorParams, rotated[v], markersConfidenceTmp[v], confidenceNeeded, scale);
1093
1094 if (validCandidates[v] == 0 && checkCloseContours) {
1095 for (const MarkerCandidate& closeMarkerCandidate: selectedContours[v].closeContours) {
1096 validCandidates[v] = _identifyOneCandidate(currentDictionary, img, closeMarkerCandidate.corners, idsTmp[v], detectorParams, rotated[v], markersConfidenceTmp[v], confidenceNeeded, scale);
1097 if (validCandidates[v] > 0) {
1098 selectedContours[v].corners = closeMarkerCandidate.corners;
1099 selectedContours[v].contour = closeMarkerCandidate.contour;
1100 break;
1101 }
1102 }
1103 }
1104 }
1105 });
1106
1107 // visit the parent vertices of the detected markers to skip identify parent contours
1108 for(size_t v : depths[depth]) {
1109 if(validCandidates[v] > 0) {
1110 int parent = selectedContours[v].parent;
1111 while (parent != -1) {
1112 if (!was[parent]) {
1113 was[parent] = true;
1114 counter++;
1115 }
1116 parent = selectedContours[parent].parent;
1117 }
1118 }
1119 counter++;
1120 }
1121 depth++;
1122 }
1123
1124 for (size_t i = 0ull; i < selectedContours.size(); i++) {
1125 if (validCandidates[i] > 0) {
1126 // shift corner positions to the correct rotation
1127 correctCornerPosition(selectedContours[i].corners, rotated[i]);
1128
1129 accepted.push_back(selectedContours[i].corners);
1130 contours.push_back(selectedContours[i].contour);
1131 ids.push_back(idsTmp[i]);
1132 } else {
1133 rejected.push_back(selectedContours[i].corners);
1134 }
1135 }
1136
1137 if(confidenceNeeded) {
1138 for (size_t i = 0ull; i < selectedContours.size(); i++) {
1139 if (validCandidates[i] > 0) {
1140 markersConfidence.push_back(markersConfidenceTmp[i]);
1141 }
1142 }
1143 }
1144 }

1146 void performCornerSubpixRefinement(const Mat& grey, const vector<Mat>& grey_pyramid, int closest_pyr_image_idx, const vector<vector<Point2f>>& candidates, const Dictionary& dictionary) const {
1147 CV_Assert(detectorParams.cornerRefinementWinSize > 0 && detectorParams.cornerRefinementMaxIterations > 0 &&
1148 detectorParams.cornerRefinementMinAccuracy > 0);
1149 // Do subpixel estimation. In Aruco3 start on the lowest pyramid level and upscale the corners
1150 parallel_for_(Range(0, (int)candidates.size()), [&](const Range& range) {
1151 const int begin = range.start;
1152 const int end = range.end;
1153
1154 for (int i = begin; i < end; i++) {
1155 if (detectorParams.useAruco3Detection) {
1156 const float scale_init = (float) grey_pyramid[closest_pyr_image_idx].cols / grey.cols;
1157 findCornerInPyrImage(scale_init, closest_pyr_image_idx, grey_pyramid, Mat(candidates[i]), detectorParams);
1158 } else {
1159 int cornerRefinementWinSize = std::max(1, cvRound(detectorParams.relativeCornerRefinmentWinSize*
1160 getAverageModuleSize(candidates[i], dictionary.markerSize, detectorParams.markerBorderBits)));
1161 cornerRefinementWinSize = min(cornerRefinementWinSize, detectorParams.cornerRefinementWinSize);
1162 cornerSubPix(grey, Mat(candidates[i]), Size(cornerRefinementWinSize, cornerRefinementWinSize), Size(-1, -1),
1163 TermCriteria(TermCriteria::MAX_ITER | TermCriteria::EPS,
1164 detectorParams.cornerRefinementMaxIterations,
1165 detectorParams.cornerRefinementMinAccuracy));
1166 }
1167 }
1168 });
1169 }

1208 // To refine corners of ArUco markers the function refineDetectedMarkers() find an aruco markers pose from 3D-2D point correspondences.
1209 // To find 3D-2D point correspondences uses matchImagePoints().
1210 // The method matchImagePoints() works with ArUco corners (in Board/GridBoard cases) or with ChArUco corners (in CharucoBoard case).
1211 // To refine corners of ArUco markers we need work with ArUco corners only in all boards.
1212 // To call matchImagePoints() with ArUco corners for all boards we need to call matchImagePoints() from base class Board.
1213 // The method matchImagePoints() implemented in Pimpl and we need to create temp Board object to call the base method.
1214 Board(board.getObjPoints(), board.getDictionary(), board.getIds()).matchImagePoints(detectedCorners, detectedIds, objPoints, imgPoints);
1215 if (objPoints.total() < 4ull) // at least one marker from board so rvec and tvec are valid
1216 return;
1217 solvePnP(objPoints, imgPoints, cameraMatrix, distCoeffs, rvec, tvec);
1218
1219 // search undetected markers and project them using the previous pose
1220 vector<vector<Point2f> > undetectedCorners;
1221 const std::vector<int>& ids = board.getIds();
1222 vector<int> undetectedIds;
1223 for(unsigned int i = 0; i < ids.size(); i++) {
1224 int foundIdx = -1;
1225 for(unsigned int j = 0; j < detectedIds.total(); j++) {
1226 if(ids[i] == detectedIds.getMat().ptr<int>()[j]) {
1227 foundIdx = j;
1228 break;
1229 }
1230 }
1231
1232 // not detected
1233 if(foundIdx == -1) {
1234 undetectedCorners.push_back(vector<Point2f>());
1235 undetectedIds.push_back(ids[i]);
1236 projectPoints(board.getObjPoints()[i], rvec, tvec, cameraMatrix, distCoeffs,
1237 undetectedCorners.back());
1238 }
1239 }
1240 // parse output
1241 Mat(undetectedIds).copyTo(undetectedMarkersIds);
1242 undetectedMarkersProjectedCorners = undetectedCorners;

1249static void _projectUndetectedMarkers(const Board &_board, InputOutputArrayOfArrays _detectedCorners,
1250 InputOutputArray _detectedIds,
1251 vector<vector<Point2f> >& _undetectedMarkersProjectedCorners,
1252 OutputArray _undetectedMarkersIds) {
1253 // check board points are in the same plane, if not, global homography cannot be applied
1254 CV_Assert(_board.getObjPoints().size() > 0);
1255 CV_Assert(_board.getObjPoints()[0].size() > 0);
1256 float boardZ = _board.getObjPoints()[0][0].z;
1257 for(unsigned int i = 0; i < _board.getObjPoints().size(); i++) {
1258 for(unsigned int j = 0; j < _board.getObjPoints()[i].size(); j++)
1259 CV_Assert(boardZ == _board.getObjPoints()[i][j].z);
1260 }
1261
1262 vector<Point2f> detectedMarkersObj2DAll; // Object coordinates (without Z) of all the detected
1263 // marker corners in a single vector
1264 vector<Point2f> imageCornersAll; // Image corners of all detected markers in a single vector
1265 vector<vector<Point2f> > undetectedMarkersObj2D; // Object coordinates (without Z) of all
1266 // missing markers in different vectors
1267 vector<int> undetectedMarkersIds; // ids of missing markers
1268 // find markers included in board, and missing markers from board. Fill the previous vectors
1269 for(unsigned int j = 0; j < _board.getIds().size(); j++) {
1270 bool found = false;
1271 for(unsigned int i = 0; i < _detectedIds.total(); i++) {
1272 if(_detectedIds.getMat().ptr<int>()[i] == _board.getIds()[j]) {
1273 for(int c = 0; c < 4; c++) {
1274 imageCornersAll.push_back(_detectedCorners.getMat(i).ptr<Point2f>()[c]);
1275 detectedMarkersObj2DAll.push_back(
1276 Point2f(_board.getObjPoints()[j][c].x, _board.getObjPoints()[j][c].y));
1277 }
1278 found = true;
1279 break;
1280 }
1281 }
1282 if(!found) {
1283 undetectedMarkersObj2D.push_back(vector<Point2f>());
1284 for(int c = 0; c < 4; c++) {
1285 undetectedMarkersObj2D.back().push_back(
1286 Point2f(_board.getObjPoints()[j][c].x, _board.getObjPoints()[j][c].y));
1287 }
1288 undetectedMarkersIds.push_back(_board.getIds()[j]);
1289 }
1290 }
1291 if(imageCornersAll.size() == 0) return;
1292
1293 // get homography from detected markers
1294 Mat transformation = findHomography(detectedMarkersObj2DAll, imageCornersAll);
1295
1296 _undetectedMarkersProjectedCorners.resize(undetectedMarkersIds.size());
1297
1298 // for each undetected marker, apply transformation
1299 for(unsigned int i = 0; i < undetectedMarkersObj2D.size(); i++) {
1300 perspectiveTransform(undetectedMarkersObj2D[i], _undetectedMarkersProjectedCorners[i], transformation);
1301 }
1302 Mat(undetectedMarkersIds).copyTo(_undetectedMarkersIds);

1305void ArucoDetector::refineDetectedMarkers(InputArray _image, const Board& _board,
1306 InputOutputArrayOfArrays _detectedCorners, InputOutputArray _detectedIds,
1307 InputOutputArrayOfArrays _rejectedCorners, InputArray _cameraMatrix,
1308 InputArray _distCoeffs, OutputArray _recoveredIdxs) const {
1309 DetectorParameters& detectorParams = arucoDetectorImpl->detectorParams;
1310 const Dictionary& dictionary = arucoDetectorImpl->dictionaries.at(0);
1311 RefineParameters& refineParams = arucoDetectorImpl->refineParams;
1312 CV_Assert(refineParams.minRepDistance > 0);
1313
1314 if(_detectedIds.total() == 0 || _rejectedCorners.total() == 0) return;
1315
1316 // get projections of missing markers in the board
1317 vector<vector<Point2f> > undetectedMarkersCorners;
1318 vector<int> undetectedMarkersIds;
1319 if(_cameraMatrix.total() != 0) {
1320 // reproject based on camera projection model
1321 _projectUndetectedMarkers(_board, _detectedCorners, _detectedIds, _cameraMatrix, _distCoeffs,
1322 undetectedMarkersCorners, undetectedMarkersIds);
1323
1324 } else {
1325 // reproject based on global homography
1326 _projectUndetectedMarkers(_board, _detectedCorners, _detectedIds, undetectedMarkersCorners,
1327 undetectedMarkersIds);
1328 }
1329
1330 // list of missing markers indicating if they have been assigned to a candidate
1331 vector<bool > alreadyIdentified(_rejectedCorners.total(), false);
1332
1333 // maximum bits that can be corrected
1334 int maxCorrectionRecalculated =
1335 int(double(dictionary.maxCorrectionBits) * refineParams.errorCorrectionRate);
1336
1337 Mat grey;
1338 _convertToGrey(_image, grey);
1339
1340 // vector of final detected marker corners and ids
1341 vector<vector<Point2f> > finalAcceptedCorners;
1342 vector<int> finalAcceptedIds;
1343 // fill with the current markers
1344 finalAcceptedCorners.resize(_detectedCorners.total());
1345 finalAcceptedIds.resize(_detectedIds.total());
1346 for(unsigned int i = 0; i < _detectedIds.total(); i++) {
1347 finalAcceptedCorners[i] = _detectedCorners.getMat(i).clone();
1348 finalAcceptedIds[i] = _detectedIds.getMat().ptr<int>()[i];
1349 }
1350 vector<int> recoveredIdxs; // original indexes of accepted markers in _rejectedCorners
1351
1352 // for each missing marker, try to find a correspondence
1353 for(unsigned int i = 0; i < undetectedMarkersIds.size(); i++) {
1354
1355 // best match at the moment
1356 int closestCandidateIdx = -1;
1357 double closestCandidateDistance = refineParams.minRepDistance * refineParams.minRepDistance + 1;
1358 Mat closestRotatedMarker;
1359
1360 for(unsigned int j = 0; j < _rejectedCorners.total(); j++) {
1361 if(alreadyIdentified[j]) continue;
1362
1363 // check distance
1364 double minDistance = closestCandidateDistance + 1;
1365 bool valid = false;
1366 int validRot = 0;
1367 for(int c = 0; c < 4; c++) { // first corner in rejected candidate
1368 double currentMaxDistance = 0;
1369 for(int k = 0; k < 4; k++) {
1370 Point2f rejCorner = _rejectedCorners.getMat(j).ptr<Point2f>()[(c + k) % 4];
1371 Point2f distVector = undetectedMarkersCorners[i][k] - rejCorner;
1372 double cornerDist = distVector.x * distVector.x + distVector.y * distVector.y;
1373 currentMaxDistance = max(currentMaxDistance, cornerDist);
1374 }
1375 // if distance is better than current best distance
1376 if(currentMaxDistance < closestCandidateDistance) {
1377 valid = true;
1378 validRot = c;
1379 minDistance = currentMaxDistance;
1380 }
1381 if(!refineParams.checkAllOrders) break;
1382 }
1383
1384 if(!valid) continue;
1385
1386 // apply rotation
1387 Mat rotatedMarker;
1388 if(refineParams.checkAllOrders) {
1389 rotatedMarker = Mat(4, 1, CV_32FC2);
1390 for(int c = 0; c < 4; c++)
1391 rotatedMarker.ptr<Point2f>()[c] =
1392 _rejectedCorners.getMat(j).ptr<Point2f>()[(c + 4 + validRot) % 4];
1393 }
1394 else rotatedMarker = _rejectedCorners.getMat(j);
1395
1396 // last filter, check if inner code is close enough to the assigned marker code
1397 int codeDistance = 0;
1398 // if errorCorrectionRate, don't check code
1399 if(refineParams.errorCorrectionRate >= 0) {
1400
1401 // extract bits
1402 Mat cellPixelRatio = _extractCellPixelRatio(
1403 grey, rotatedMarker, dictionary.markerSize, detectorParams.markerBorderBits,
1404 detectorParams.perspectiveRemovePixelPerCell,
1405 detectorParams.perspectiveRemoveIgnoredMarginPerCell, detectorParams.minOtsuStdDev);
1406
1407 Mat onlyCellPixelRatio = cellPixelRatio(
1408 Rect(detectorParams.markerBorderBits, detectorParams.markerBorderBits,
1409 cellPixelRatio.cols - 2 * detectorParams.markerBorderBits,
1410 cellPixelRatio.rows - 2 * detectorParams.markerBorderBits));
1411
1412 codeDistance = dictionary.getDistanceToId(onlyCellPixelRatio, undetectedMarkersIds[i],
1413 false, detectorParams.validBitIdThreshold);
1414 }
1415
1416 // if everythin is ok, assign values to current best match
1417 if(refineParams.errorCorrectionRate < 0 || codeDistance < maxCorrectionRecalculated) {
1418 closestCandidateIdx = j;
1419 closestCandidateDistance = minDistance;
1420 closestRotatedMarker = rotatedMarker;
1421 }
1422 }
1423
1424 // if at least one good match, we have rescue the missing marker
1425 if(closestCandidateIdx >= 0) {
1426
1427 // subpixel refinement
1428 if(detectorParams.cornerRefinementMethod == (int)CORNER_REFINE_SUBPIX) {
1429 CV_Assert(detectorParams.cornerRefinementWinSize > 0 &&
1430 detectorParams.cornerRefinementMaxIterations > 0 &&
1431 detectorParams.cornerRefinementMinAccuracy > 0);
1432
1433 std::vector<Point2f> marker(closestRotatedMarker.begin<Point2f>(), closestRotatedMarker.end<Point2f>());
1434 int cornerRefinementWinSize = std::max(1, cvRound(detectorParams.relativeCornerRefinmentWinSize*
1435 getAverageModuleSize(marker, dictionary.markerSize, detectorParams.markerBorderBits)));
1436 cornerRefinementWinSize = min(cornerRefinementWinSize, detectorParams.cornerRefinementWinSize);
1437 cornerSubPix(grey, closestRotatedMarker,
1438 Size(cornerRefinementWinSize, cornerRefinementWinSize),
1439 Size(-1, -1), TermCriteria(TermCriteria::MAX_ITER | TermCriteria::EPS,
1440 detectorParams.cornerRefinementMaxIterations,
1441 detectorParams.cornerRefinementMinAccuracy));
1442 }
1443
1444 // remove from rejected
1445 alreadyIdentified[closestCandidateIdx] = true;
1446
1447 // add to detected
1448 finalAcceptedCorners.push_back(closestRotatedMarker);
1449 finalAcceptedIds.push_back(undetectedMarkersIds[i]);
1450
1451 // add the original index of the candidate
1452 recoveredIdxs.push_back(closestCandidateIdx);
1453 }
1454 }
1455
1456 // parse output
1457 if(finalAcceptedIds.size() != _detectedIds.total()) {
1458 // parse output
1459 Mat(finalAcceptedIds).copyTo(_detectedIds);
1460 _copyVector2Output(finalAcceptedCorners, _detectedCorners);
1461
1462 vector<vector<Point2f> > rejectedCorners;
1463 _copyInput2Vector(_rejectedCorners, rejectedCorners);
1464
1465 // recalculate _rejectedCorners based on alreadyIdentified
1466 vector<vector<Point2f> > finalRejected;
1467 for(size_t i = 0; i < alreadyIdentified.size(); i++) {
1468 if(!alreadyIdentified[i]) {
1469 finalRejected.push_back(rejectedCorners[i]);
1470 }
1471 }
1472 rejectedCorners.clear();
1473 for (size_t i = 0; i < finalRejected.size(); i++)
1474 rejectedCorners.push_back(finalRejected[i]);
1475 _copyVector2Output(rejectedCorners, _rejectedCorners);
1476
1477 if(_recoveredIdxs.needed()) {
1478 Mat(recoveredIdxs).copyTo(_recoveredIdxs);
1479 }
1480 }
1481}
