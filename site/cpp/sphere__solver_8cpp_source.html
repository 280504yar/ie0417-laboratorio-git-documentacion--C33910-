33 int estimate(const std::vector<int> &sample, std::vector<Mat> &models) const override
34 {
35 models.clear();
36
37 // Get point data
38 const int p1_idx = sample[0], p2_idx = sample[1], p3_idx = sample[2], p4_idx = sample[3];
39 float x1 = pts_ptr_x[p1_idx], y1 = pts_ptr_y[p1_idx], z1 = pts_ptr_z[p1_idx];
40 float x2 = pts_ptr_x[p2_idx], y2 = pts_ptr_y[p2_idx], z2 = pts_ptr_z[p2_idx];
41 float x3 = pts_ptr_x[p3_idx], y3 = pts_ptr_y[p3_idx], z3 = pts_ptr_z[p3_idx];
42 float x4 = pts_ptr_x[p4_idx], y4 = pts_ptr_y[p4_idx], z4 = pts_ptr_z[p4_idx];
43
44 double center_x, center_y, center_z, radius; // Cramer's Rule
45 {
46 double a11, a12, a13, a21, a22, a23, a31, a32, a33, t1, t2, t3, t4, d, d1, d2, d3;
47 a11 = x2 - x1;
48 a12 = y2 - y1;
49 a13 = z2 - z1;
50 a21 = x3 - x2;
51 a22 = y3 - y2;
52 a23 = z3 - z2;
53 a31 = x4 - x3;
54 a32 = y4 - y3;
55 a33 = z4 - z3;
56 t1 = x1 * x1 + y1 * y1 + z1 * z1;
57 t2 = x2 * x2 + y2 * y2 + z2 * z2;
58 t3 = x3 * x3 + y3 * y3 + z3 * z3;
59 t4 = x4 * x4 + y4 * y4 + z4 * z4;
60
61 double b1 = t2 - t1,
62 b2 = t3 - t2,
63 b3 = t4 - t3;
64
65 // d = a11 * a22 * a33 + a12 * a23 * a31 + a13 * a21 * a32 - a11 * a23 * a32 -
66 // a12 * a21 * a33 - a13 * a22 * a31;
67 // d1 = b1 * a22 * a33 + a12 * a23 * b3 + a13 * b2 * a32 - b1 * a23 * a32 -
68 // a12 * b2 * a33 - a13 * a22 * b3;
69 // d2 = a11 * b2 * a33 + b1 * a23 * a31 + a13 * a21 * b3 - a11 * a23 * b3 -
70 // b1 * a21 * a33 - a13 * b2 * a31;
71 // d3 = a11 * a22 * b3 + a12 * b2 * a31 + b1 * a21 * a32 - a11 * b2 * a32 -
72 // a12 * a21 * b3 - b1 * a22 * a31;
73
74 double tmp1 = a22 * a33 - a23 * a32,
75 tmp2 = a23 * a31 - a21 * a33,
76 tmp3 = a21 * a32 - a22 * a31;
77 d = a11 * tmp1 + a12 * tmp2 + a13 * tmp3;
78 d1 = b1 * tmp1 + a12 * (a23 * b3 - b2 * a33) + a13 * (b2 * a32 - a22 * b3);
79 d2 = a11 * (b2 * a33 - a23 * b3) + b1 * tmp2 + a13 * (a21 * b3 - b2 * a31);
80 d3 = a11 * (a22 * b3 - b2 * a32) + a12 * (b2 * a31 - a21 * b3) + b1 * tmp3;
81
82 if (d == 0) {
83 return 0;
84 }
85
86 d = 0.5 / d;
87 center_x = d1 * d;
88 center_y = d2 * d;
89 center_z = d3 * d;
90
91 tmp1 = center_x - x1;
92 tmp2 = center_y - y1;
93 tmp3 = center_z - z1;
94 radius = sqrt(tmp1 * tmp1 + tmp2 * tmp2 + tmp3 * tmp3);
95 }
96
97 double sphere_coeff[4] = {center_x, center_y, center_z, radius};
98 models.emplace_back(cv::Mat(1, 4, CV_64F, sphere_coeff).clone());
99
100 return 1;
101 }

134 int estimate(const std::vector<int> &sample, int sample_size, std::vector<Mat> &models,
135 const std::vector<double> &/*weights*/) const override
136 {
137 const double inv_sample_size = 1.0 / (double) sample_size;
138
139 // Compute the average of the data points.
140 // Vec3d A(0, 0, 0);
141 double A0 = 0, A1 = 0, A2 = 0;
142
143 for (int i = 0; i < sample_size; ++i)
144 {
145 int sample_idx = sample[i];
146 float x = pts_ptr_x[sample_idx];
147 float y = pts_ptr_y[sample_idx];
148 float z = pts_ptr_z[sample_idx];
149 // A += Vec3d(x, y, z);
150 A0 += x;
151 A1 += y;
152 A2 += z;
153 }
154
155 // A *= inv_sample_size;
156 A0 *= inv_sample_size;
157 A1 *= inv_sample_size;
158 A2 *= inv_sample_size;
159
160 // Compute the covariance matrix M of the Y[i] = X[i]-A and the
161 // right-hand side R of the linear system M*(C-A) = R.
162 double M00 = 0, M01 = 0, M02 = 0, M11 = 0, M12 = 0, M22 = 0;
163 // Vec3d R(0, 0, 0);
164 double R0 = 0, R1 = 0, R2 = 0;
165 for (int i = 0; i < sample_size; ++i)
166 {
167 int sample_idx = sample[i];
168 float x = pts_ptr_x[sample_idx];
169 float y = pts_ptr_y[sample_idx];
170 float z = pts_ptr_z[sample_idx];
171
172 // Vec3d Y = Vec3d(x, y, z) - A;
173 // double Y0Y0 = Y[0] * Y[0];
174 // double Y0Y1 = Y[0] * Y[1];
175 // double Y0Y2 = Y[0] * Y[2];
176 // double Y1Y1 = Y[1] * Y[1];
177 // double Y1Y2 = Y[1] * Y[2];
178 // double Y2Y2 = Y[2] * Y[2];
179 // M00 += Y0Y0;
180 // M01 += Y0Y1;
181 // M02 += Y0Y2;
182 // M11 += Y1Y1;
183 // M12 += Y1Y2;
184 // M22 += Y2Y2;
185 // R += (Y0Y0 + Y1Y1 + Y2Y2) * Y;
186
187
188 double Y0 = x - A0, Y1 = y - A1, Y2 = z - A2;
189 double Y0Y0 = Y0 * Y0;
190 double Y1Y1 = Y1 * Y1;
191 double Y2Y2 = Y2 * Y2;
192 M00 += Y0Y0;
193 M01 += Y0 * Y1;
194 M02 += Y0 * Y2;
195 M11 += Y1Y1;
196 M12 += Y1 * Y2;
197 M22 += Y2Y2;
198 double sum_diag = Y0Y0 + Y1Y1 + Y2Y2;
199 R0 += sum_diag * Y0;
200 R1 += sum_diag * Y1;
201 R2 += sum_diag * Y2;
202 }
203 // R *= 0.5;
204 R0 *= 0.5;
205 R1 *= 0.5;
206 R2 *= 0.5;
207
208 double center_x, center_y, center_z, radius;
209
210 // Solve the linear system M*(C-A) = R for the center C.
211 double cof00 = M11 * M22 - M12 * M12;
212 double cof01 = M02 * M12 - M01 * M22;
213 double cof02 = M01 * M12 - M02 * M11;
214 double det = M00 * cof00 + M01 * cof01 + M02 * cof02;
215 if (det != 0)
216 {
217 double cof11 = M00 * M22 - M02 * M02;
218 double cof12 = M01 * M02 - M00 * M12;
219 double cof22 = M00 * M11 - M01 * M01;
220 // center_x = A[0] + (cof00 * R[0] + cof01 * R[1] + cof02 * R[2]) / det;
221 // center_y = A[1] + (cof01 * R[0] + cof11 * R[1] + cof12 * R[2]) / det;
222 // center_z = A[2] + (cof02 * R[0] + cof12 * R[1] + cof22 * R[2]) / det;
223
224 center_x = A0 + (cof00 * R0 + cof01 * R1 + cof02 * R2) / det;
225 center_y = A1 + (cof01 * R0 + cof11 * R1 + cof12 * R2) / det;
226 center_z = A2 + (cof02 * R0 + cof12 * R1 + cof22 * R2) / det;
227
228 double rsqr = 0;
229 for (int i = 0; i < sample_size; ++i)
230 {
231 int sample_idx = sample[i];
232 float x = pts_ptr_x[sample_idx];
233 float y = pts_ptr_y[sample_idx];
234 float z = pts_ptr_z[sample_idx];
235
236 double diff_x = x - center_x, diff_y = y - center_y, diff_z = z - center_z;
237 rsqr += diff_x * diff_x + diff_y * diff_y + diff_z * diff_z;
238 }
239
240 rsqr *= inv_sample_size;
241 radius = std::sqrt(rsqr);
242
243 double sphere_coeff[4] = {center_x, center_y, center_z, radius};
244 models.emplace_back(cv::Mat(1, 4, CV_64F, sphere_coeff).clone());
245
246 return 1;
247 }
248 else
249 {
250 return 0;
251 }
252
