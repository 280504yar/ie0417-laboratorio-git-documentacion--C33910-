232void SimpleBlobDetectorImpl::findBlobs(InputArray _image, InputArray _binaryImage, std::vector<Center> &centers,
233 std::vector<std::vector<Point> > &contoursOut, std::vector<Moments> &momentss) const
234{
235 CV_INSTRUMENT_REGION();
236
237 Mat image = _image.getMat(), binaryImage = _binaryImage.getMat();
238 CV_UNUSED(image);
239 centers.clear();
240 contoursOut.clear();
241 momentss.clear();
242
243 std::vector < std::vector<Point> > contours;
244 findContours(binaryImage, contours, RETR_LIST, CHAIN_APPROX_NONE);
245
246#ifdef DEBUG_BLOB_DETECTOR
247 Mat keypointsImage;
248 cvtColor(binaryImage, keypointsImage, COLOR_GRAY2RGB);
249
250 Mat contoursImage;
251 cvtColor(binaryImage, contoursImage, COLOR_GRAY2RGB);
252 drawContours( contoursImage, contours, -1, Scalar(0,255,0) );
253 imshow("contours", contoursImage );
254#endif
255
256 std::vector<double> dists;
257
258 for (size_t contourIdx = 0; contourIdx < contours.size(); contourIdx++)
259 {
260 Center center;
261 center.confidence = 1;
262 Moments moms = moments(contours[contourIdx]);
263 if (params.filterByArea)
264 {
265 double area = moms.m00;
266 if (area < params.minArea || area >= params.maxArea)
267 continue;
268 }
269
270 if (params.filterByCircularity)
271 {
272 double area = moms.m00;
273 double perimeter = arcLength(contours[contourIdx], true);
274 double ratio = 4 * CV_PI * area / (perimeter * perimeter);
275 if (ratio < params.minCircularity || ratio >= params.maxCircularity)
276 continue;
277 }
278
279 if (params.filterByInertia)
280 {
281 double denominator = std::sqrt(std::pow(2 * moms.mu11, 2) + std::pow(moms.mu20 - moms.mu02, 2));
282 const double eps = 1e-2;
283 double ratio;
284 if (denominator > eps)
285 {
286 double cosmin = (moms.mu20 - moms.mu02) / denominator;
287 double sinmin = 2 * moms.mu11 / denominator;
288 double cosmax = -cosmin;
289 double sinmax = -sinmin;
290
291 double imin = 0.5 * (moms.mu20 + moms.mu02) - 0.5 * (moms.mu20 - moms.mu02) * cosmin - moms.mu11 * sinmin;
292 double imax = 0.5 * (moms.mu20 + moms.mu02) - 0.5 * (moms.mu20 - moms.mu02) * cosmax - moms.mu11 * sinmax;
293 ratio = imin / imax;
294 }
295 else
296 {
297 ratio = 1;
298 }
299
300 if (ratio < params.minInertiaRatio || ratio >= params.maxInertiaRatio)
301 continue;
302
303 center.confidence = ratio * ratio;
304 }
305
306 if (params.filterByConvexity)
307 {
308 std::vector < Point > hull;
309 convexHull(contours[contourIdx], hull);
310 double area = moms.m00;
311 double hullArea = contourArea(hull);
312 if (fabs(hullArea) < DBL_EPSILON)
313 continue;
314 double ratio = area / hullArea;
315 if (ratio < params.minConvexity || ratio >= params.maxConvexity)
316 continue;
317 }
318
319 if(moms.m00 == 0.0)
320 continue;
321 center.location = Point2d(moms.m10 / moms.m00, moms.m01 / moms.m00);
322 center.radius = 0.;
323
324 if (params.filterByColor)
325 {
326 if (binaryImage.at<uchar> (cvRound(center.location.y), cvRound(center.location.x)) != params.blobColor)
327 continue;
328 }
329
330 //compute blob radius
331 {
332 const std::vector<cv::Point>& contour = contours[contourIdx];
333 const size_t contourSize = contour.size();
334
335 if (contourSize > 0u) {
336 dists.resize(contourSize);
337 for (size_t pointIdx = 0; pointIdx < contourSize; pointIdx++)
338 {
339 const Point2d& pt = contour[pointIdx];
340 dists[pointIdx] = norm(center.location - pt);
341 }
342 std::sort(dists.begin(), dists.end());
343 center.radius = (dists[(contourSize - 1) / 2] + dists[contourSize / 2]) / 2.;
344 }
345 }
346
347 centers.push_back(center);
348 if (params.collectContours)
349 {
350 contoursOut.push_back(contours[contourIdx]);
351 momentss.push_back(moms);
352 }
353
354#ifdef DEBUG_BLOB_DETECTOR
355 circle( keypointsImage, center.location, 1, Scalar(0,0,255), 1 );
356#endif
357 }
358#ifdef DEBUG_BLOB_DETECTOR
359 imshow("bk", keypointsImage );
360 waitKey();
361#endif
362}

364void SimpleBlobDetectorImpl::detect(InputArray image, std::vector<cv::KeyPoint>& keypoints, InputArray mask)
365{
366 CV_INSTRUMENT_REGION();
367
368 keypoints.clear();
369 blobContours.clear();
370
371 CV_Assert(params.minRepeatability != 0);
372 Mat grayscaleImage;
373 if (image.channels() == 3 || image.channels() == 4)
374 cvtColor(image, grayscaleImage, COLOR_BGR2GRAY);
375 else
376 grayscaleImage = image.getMat();
377
378 if (grayscaleImage.type() != CV_8UC1) {
379 CV_Error(Error::StsUnsupportedFormat, "Blob detector only supports 8-bit images!");
380 }
381
382 CV_CheckGT(params.thresholdStep, 0.0f, "");
383 if (params.minThreshold + params.thresholdStep >= params.maxThreshold)
384 {
385 // https://github.com/opencv/opencv/issues/6667
386 CV_LOG_ONCE_INFO(NULL, "SimpleBlobDetector: params.minDistBetweenBlobs is ignored for case with single threshold");
387 CV_CheckEQ(params.minRepeatability, 1u, "Incompatible parameters for case with single threshold");
388 }
389
390 std::vector < std::vector<Center> > centers;
391 std::vector<Moments> momentss;
392 for (double thresh = params.minThreshold; thresh < params.maxThreshold; thresh += params.thresholdStep)
393 {
394 Mat binarizedImage;
395 threshold(grayscaleImage, binarizedImage, thresh, 255, THRESH_BINARY);
396
397 std::vector < Center > curCenters;
398 std::vector<std::vector<Point> > curContours;
399 std::vector<Moments> curMomentss;
400 findBlobs(grayscaleImage, binarizedImage, curCenters, curContours, curMomentss);
401 std::vector < std::vector<Center> > newCenters;
402 std::vector<std::vector<Point> > newContours;
403 std::vector<Moments> newMomentss;
404 for (size_t i = 0; i < curCenters.size(); i++)
405 {
406 bool isNew = true;
407 for (size_t j = 0; j < centers.size(); j++)
408 {
409 double dist = norm(centers[j][ centers[j].size() / 2 ].location - curCenters[i].location);
410 isNew = dist >= params.minDistBetweenBlobs && dist >= centers[j][ centers[j].size() / 2 ].radius && dist >= curCenters[i].radius;
411 if (!isNew)
412 {
413 centers[j].push_back(curCenters[i]);
414
415 size_t k = centers[j].size() - 1;
416 while( k > 0 && curCenters[i].radius < centers[j][k-1].radius )
417 {
418 centers[j][k] = centers[j][k-1];
419 k--;
420 }
421
422 if (params.collectContours)
423 {
424 if (curCenters[i].confidence > centers[j][k].confidence
425 || (curCenters[i].confidence == centers[j][k].confidence && curMomentss[i].m00 > momentss[j].m00))
426 {
427 blobContours[j] = curContours[i];
428 momentss[j] = curMomentss[i];
429 }
430 }
431 centers[j][k] = curCenters[i];
432
433 break;
434 }
435 }
436 if (isNew)
437 {
438 newCenters.push_back(std::vector<Center> (1, curCenters[i]));
439 if (params.collectContours)
440 {
441 newContours.push_back(curContours[i]);
442 newMomentss.push_back(curMomentss[i]);
443 }
444 }
445 }
446 std::copy(newCenters.begin(), newCenters.end(), std::back_inserter(centers));
447 if (params.collectContours)
448 {
449 std::copy(newContours.begin(), newContours.end(), std::back_inserter(blobContours));
450 std::copy(newMomentss.begin(), newMomentss.end(), std::back_inserter(momentss));
451 }
452 }
453
454 for (size_t i = 0; i < centers.size(); i++)
455 {
456 if (centers[i].size() < params.minRepeatability)
457 continue;
458 Point2d sumPoint(0, 0);
459 double normalizer = 0;
460 for (size_t j = 0; j < centers[i].size(); j++)
461 {
462 sumPoint += centers[i][j].confidence * centers[i][j].location;
463 normalizer += centers[i][j].confidence;
464 }
465 sumPoint *= (1. / normalizer);
466 KeyPoint kpt(sumPoint, (float)(centers[i][centers[i].size() / 2].radius) * 2.0f);
467 keypoints.push_back(kpt);
468 }
469
470 if (!mask.empty())
471 {
472 if (params.collectContours)
473 {
474 KeyPointsFilter::runByPixelsMask2VectorPoint(keypoints, blobContours, mask.getMat());
475 }
476 else
477 {
478 KeyPointsFilter::runByPixelsMask(keypoints, mask.getMat());
479 }
480 }
