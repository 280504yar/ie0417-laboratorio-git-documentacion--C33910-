24// The function expects 1-based indexing for ofs
25// Zero is treated as invalid offset (not found)
26static void ofs2idx(const Mat& a, size_t ofs, int* idx)
27{
28 int i, d = a.dims;
29 if( ofs > 0 )
30 {
31 ofs--;
32 for( i = d-1; i >= 0; i-- )
33 {
34 int sz = a.size[i];
35 idx[i] = (int)(ofs % sz);
36 ofs /= sz;
37 }
38 }
39 else
40 {
41 for( i = d-1; i >= 0; i-- )
42 idx[i] = -1;
43 }
44}

50template <typename T>
51void getMinMaxRes(const Mat & db, double * minVal, double * maxVal,
52 int* minLoc, int* maxLoc,
53 int groupnum, int cols, double * maxVal2)
54{
55 uint index_max = std::numeric_limits<uint>::max();
56 T minval = std::numeric_limits<T>::max();
57 T maxval = std::numeric_limits<T>::min() > 0 ? -std::numeric_limits<T>::max() : std::numeric_limits<T>::min(), maxval2 = maxval;
58 uint minloc = index_max, maxloc = index_max;
59
60 size_t index = 0;
61 const T * minptr = NULL, * maxptr = NULL, * maxptr2 = NULL;
62 const uint * minlocptr = NULL, * maxlocptr = NULL;
63 if (minVal || minLoc)
64 {
65 minptr = db.ptr<T>();
66 index += sizeof(T) * groupnum;
67 index = alignSize(index, MINMAX_STRUCT_ALIGNMENT);
68 }
69 if (maxVal || maxLoc)
70 {
71 maxptr = (const T *)(db.ptr() + index);
72 index += sizeof(T) * groupnum;
73 index = alignSize(index, MINMAX_STRUCT_ALIGNMENT);
74 }
75 if (minLoc)
76 {
77 minlocptr = (const uint *)(db.ptr() + index);
78 index += sizeof(uint) * groupnum;
79 index = alignSize(index, MINMAX_STRUCT_ALIGNMENT);
80 }
81 if (maxLoc)
82 {
83 maxlocptr = (const uint *)(db.ptr() + index);
84 index += sizeof(uint) * groupnum;
85 index = alignSize(index, MINMAX_STRUCT_ALIGNMENT);
86 }
87 if (maxVal2)
88 maxptr2 = (const T *)(db.ptr() + index);
89
90 for (int i = 0; i < groupnum; i++)
91 {
92 if (minptr && minptr[i] <= minval)
93 {
94 if (minptr[i] == minval)
95 {
96 if (minlocptr)
97 minloc = std::min(minlocptr[i], minloc);
98 }
99 else
100 {
101 if (minlocptr)
102 minloc = minlocptr[i];
103 minval = minptr[i];
104 }
105 }
106 if (maxptr && maxptr[i] >= maxval)
107 {
108 if (maxptr[i] == maxval)
109 {
110 if (maxlocptr)
111 maxloc = std::min(maxlocptr[i], maxloc);
112 }
113 else
114 {
115 if (maxlocptr)
116 maxloc = maxlocptr[i];
117 maxval = maxptr[i];
118 }
119 }
120 if (maxptr2 && maxptr2[i] > maxval2)
121 maxval2 = maxptr2[i];
122 }
123 bool zero_mask = (minLoc && minloc == index_max) ||
124 (maxLoc && maxloc == index_max);
125
126 if (minVal)
127 *minVal = zero_mask ? 0 : (double)minval;
128 if (maxVal)
129 *maxVal = zero_mask ? 0 : (double)maxval;
130 if (maxVal2)
131 *maxVal2 = zero_mask ? 0 : (double)maxval2;
132
133 if (minLoc)
134 {
135 minLoc[0] = zero_mask ? -1 : minloc / cols;
136 minLoc[1] = zero_mask ? -1 : minloc % cols;
137 }
138 if (maxLoc)
139 {
140 maxLoc[0] = zero_mask ? -1 : maxloc / cols;
141 maxLoc[1] = zero_mask ? -1 : maxloc % cols;
142 }
143}

148bool ocl_minMaxIdx( InputArray _src, double* minVal, double* maxVal, int* minLoc, int* maxLoc, InputArray _mask,
149 int ddepth, bool absValues, InputArray _src2, double * maxVal2)
150{
151 const ocl::Device & dev = ocl::Device::getDefault();
152
153#ifdef __ANDROID__
154 if (dev.isNVidia())
155 return false;
156#endif
157
158 if (dev.deviceVersionMajor() == 1 && dev.deviceVersionMinor() < 2)
159 {
160 // 'static' storage class specifier used by "minmaxloc" is available from OpenCL 1.2+ only
161 return false;
162 }
163
164 bool doubleSupport = dev.doubleFPConfig() > 0, haveMask = !_mask.empty(),
165 haveSrc2 = _src2.kind() != _InputArray::NONE;
166 int type = _src.type(), depth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type),
167 kercn = haveMask ? cn : std::min(4, ocl::predictOptimalVectorWidth(_src, _src2));
168
169 if (depth >= CV_16F)
170 return false;
171
172 // disabled following modes since it occasionally fails on AMD devices (e.g. A10-6800K, sep. 2014)
173 if ((haveMask || type == CV_32FC1) && dev.isAMD())
174 return false;
175
176 CV_Assert( (cn == 1 && (!haveMask || _mask.type() == CV_8U || _mask.type() == CV_8S || _mask.type() == CV_Bool)) ||
177 (cn >= 1 && !minLoc && !maxLoc) );
178
179 if (ddepth < 0)
180 ddepth = depth;
181
182 CV_Assert(!haveSrc2 || _src2.type() == type);
183
184 if (depth == CV_32S || depth == CV_8S || depth == CV_32U || depth == CV_64U ||
185 depth == CV_64S || depth == CV_16F || depth == CV_16BF)
186 return false;
187
188 if ((depth == CV_64F || ddepth == CV_64F) && !doubleSupport)
189 return false;
190
191 int groupnum = dev.maxComputeUnits();
192 size_t wgs = dev.maxWorkGroupSize();
193
194 int wgs2_aligned = 1;
195 while (wgs2_aligned < (int)wgs)
196 wgs2_aligned <<= 1;
197 wgs2_aligned >>= 1;
198
199 bool needMinVal = minVal || minLoc, needMinLoc = minLoc != NULL,
200 needMaxVal = maxVal || maxLoc, needMaxLoc = maxLoc != NULL;
201
202 // in case of mask we must know whether mask is filled with zeros or not
203 // so let's calculate min or max location, if it's undefined, so mask is zeros
204 if (!(needMaxLoc || needMinLoc) && haveMask)
205 {
206 if (needMinVal)
207 needMinLoc = true;
208 else
209 needMaxLoc = true;
210 }
211
212 char cvt[2][50];
213 String opts = format("-D DEPTH_%d -D srcT1=%s%s -D WGS=%d -D srcT=%s"
214 " -D WGS2_ALIGNED=%d%s%s%s -D kercn=%d%s%s%s%s"
215 " -D dstT1=%s -D dstT=%s -D convertToDT=%s%s%s%s%s -D wdepth=%d -D convertFromU=%s"
216 " -D MINMAX_STRUCT_ALIGNMENT=%d",
217 depth, ocl::typeToStr(depth), haveMask ? " -D HAVE_MASK" : "", (int)wgs,
218 ocl::typeToStr(CV_MAKE_TYPE(depth, kercn)), wgs2_aligned,
219 doubleSupport ? " -D DOUBLE_SUPPORT" : "",
220 _src.isContinuous() ? " -D HAVE_SRC_CONT" : "",
221 _mask.isContinuous() ? " -D HAVE_MASK_CONT" : "", kercn,
222 needMinVal ? " -D NEED_MINVAL" : "", needMaxVal ? " -D NEED_MAXVAL" : "",
223 needMinLoc ? " -D NEED_MINLOC" : "", needMaxLoc ? " -D NEED_MAXLOC" : "",
224 ocl::typeToStr(ddepth), ocl::typeToStr(CV_MAKE_TYPE(ddepth, kercn)),
225 ocl::convertTypeStr(depth, ddepth, kercn, cvt[0], sizeof(cvt[0])),
226 absValues ? " -D OP_ABS" : "",
227 haveSrc2 ? " -D HAVE_SRC2" : "", maxVal2 ? " -D OP_CALC2" : "",
228 haveSrc2 && _src2.isContinuous() ? " -D HAVE_SRC2_CONT" : "", ddepth,
229 depth <= CV_32S && ddepth == CV_32S ? ocl::convertTypeStr(CV_8U, ddepth, kercn, cvt[1], sizeof(cvt[1])) : "noconvert",
230 MINMAX_STRUCT_ALIGNMENT);
231
232 ocl::Kernel k("minmaxloc", ocl::core::minmaxloc_oclsrc, opts);
233 if (k.empty())
234 return false;
235
236 int esz = CV_ELEM_SIZE(ddepth), esz32s = CV_ELEM_SIZE1(CV_32S),
237 dbsize = groupnum * ((needMinVal ? esz : 0) + (needMaxVal ? esz : 0) +
238 (needMinLoc ? esz32s : 0) + (needMaxLoc ? esz32s : 0) +
239 (maxVal2 ? esz : 0))
240 + 5 * MINMAX_STRUCT_ALIGNMENT;
241 UMat src = _src.getUMat(), src2 = _src2.getUMat(), db(1, dbsize, CV_8UC1), mask = _mask.getUMat();
242
243 if (cn > 1 && !haveMask)
244 {
245 src = src.reshape(1);
246 src2 = src2.reshape(1);
247 }
248
249 if (haveSrc2)
250 {
251 if (!haveMask)
252 k.args(ocl::KernelArg::ReadOnlyNoSize(src), src.cols, (int)src.total(),
253 groupnum, ocl::KernelArg::PtrWriteOnly(db), ocl::KernelArg::ReadOnlyNoSize(src2));
254 else
255 k.args(ocl::KernelArg::ReadOnlyNoSize(src), src.cols, (int)src.total(),
256 groupnum, ocl::KernelArg::PtrWriteOnly(db), ocl::KernelArg::ReadOnlyNoSize(mask),
257 ocl::KernelArg::ReadOnlyNoSize(src2));
258 }
259 else
260 {
261 if (!haveMask)
262 k.args(ocl::KernelArg::ReadOnlyNoSize(src), src.cols, (int)src.total(),
263 groupnum, ocl::KernelArg::PtrWriteOnly(db));
264 else
265 k.args(ocl::KernelArg::ReadOnlyNoSize(src), src.cols, (int)src.total(),
266 groupnum, ocl::KernelArg::PtrWriteOnly(db), ocl::KernelArg::ReadOnlyNoSize(mask));
267 }
268
269 size_t globalsize = groupnum * wgs;
270 if (!k.run(1, &globalsize, &wgs, true))
271 return false;

273 static const getMinMaxResFunc functab[7] =
274 {
275 getMinMaxRes<uchar>,
276 getMinMaxRes<char>,
277 getMinMaxRes<ushort>,
278 getMinMaxRes<short>,
279 getMinMaxRes<int>,
280 getMinMaxRes<float>,
281 getMinMaxRes<double>
282 };
283
284 CV_Assert(ddepth <= CV_64F);
285 getMinMaxResFunc func = functab[ddepth];
286
287 int locTemp[2];
288 func(db.getMat(ACCESS_READ), minVal, maxVal,
289 needMinLoc ? minLoc ? minLoc : locTemp : minLoc,
290 needMaxLoc ? maxLoc ? maxLoc : locTemp : maxLoc,
291 groupnum, src.cols, maxVal2);
292
293 return true;
294}

300void cv::minMaxIdx(InputArray _src, double* minVal,
301 double* maxVal, int* minIdx, int* maxIdx,
302 InputArray _mask)
303{
304 CV_INSTRUMENT_REGION();
305
306 int type = _src.type(), depth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type);
307 CV_Assert( (cn == 1 && (_mask.empty() || _mask.type() == CV_8U || _mask.type() == CV_8S || _mask.type() == CV_Bool)) ||
308 (cn > 1 && _mask.empty() && !minIdx && !maxIdx) );
309
310 CV_OCL_RUN(OCL_PERFORMANCE_CHECK(_src.isUMat()) && _src.dims() <= 2 && (_mask.empty() || _src.size() == _mask.size()),
311 ocl_minMaxIdx(_src, minVal, maxVal, minIdx, maxIdx, _mask))
312
313 Mat src = _src.getMat(), mask = _mask.getMat();
314
315 if (src.dims <= 2)
316 {
317 if ((size_t)src.step == (size_t)mask.step || mask.empty())
318 {
319 CALL_HAL(minMaxIdx, cv_hal_minMaxIdx, src.data, src.step, src.cols*cn, src.rows,
320 src.depth(), minVal, maxVal, minIdx, maxIdx, mask.data);
321 }
322 CALL_HAL(minMaxIdxMaskStep, cv_hal_minMaxIdxMaskStep, src.data, src.step, src.cols*cn, src.rows,
323 src.depth(), minVal, maxVal, minIdx, maxIdx, mask.data, mask.step);
324 }
325 else if (src.isContinuous() && (mask.isContinuous() || mask.empty()))
326 {
327 int res = cv_hal_minMaxIdx(src.data, 0, (int)src.total()*cn, 1, src.depth(),
328 minVal, maxVal, minIdx, maxIdx, mask.data);
329
330 if (res == CV_HAL_ERROR_OK)
331 {
332 // minIdx[0] and minIdx[0] are always 0 for "flatten" version
333 if (minIdx)
334 ofs2idx(src, minIdx[1]+1, minIdx);
335 if (maxIdx)
336 ofs2idx(src, maxIdx[1]+1, maxIdx);
337 return;
338 }
339 else if (res != CV_HAL_ERROR_NOT_IMPLEMENTED)
340 {
341 CV_Error_(cv::Error::StsInternal,
342 ("HAL implementation minMaxIdx ==> " CVAUX_STR(cv_hal_minMaxIdx) " returned %d (0x%08x)", res, res));
343 }
344 }
345
346 MinMaxIdxFunc func = getMinMaxIdxFunc(depth);
347 CV_Assert( func != 0 );
348
349 const Mat* arrays[] = {&src, &mask, 0};
350 uchar* ptrs[2] = {};
351 NAryMatIterator it(arrays, ptrs);
352
353 size_t minidx = 0, maxidx = 0;
354 size_t startidx = 1;
355 union {
356 int i;
357 float f;
358 double d;
359 int64 L;
360 uint64 UL;
361 } minval, maxval;
362 int planeSize = (int)it.size*cn;
363 minval.L = maxval.L = 0;
364
365 for( size_t i = 0; i < it.nplanes; i++, ++it, startidx += planeSize )
366 func( ptrs[0], ptrs[1], &minval.L, &maxval.L, &minidx, &maxidx, planeSize, startidx );
367
368 double dminval, dmaxval;
369 if( depth <= CV_32S || depth == CV_Bool )
370 dminval = minval.i, dmaxval = maxval.i;
371 else if( depth == CV_32F || depth == CV_16F || depth == CV_16BF )
372 dminval = minval.f, dmaxval = maxval.f;
373 else if( depth == CV_64F )
374 dminval = minval.d, dmaxval = maxval.d;
375 else if( depth == CV_64S || depth == CV_32U )
376 dminval = (double)minval.L, dmaxval = (double)maxval.L;
377 else {
378 CV_Assert(depth == CV_64U);
379 dminval = (double)minval.UL, dmaxval = (double)maxval.UL;
380 }
381
382 if( minVal )
383 *minVal = dminval;
384 if( maxVal )
385 *maxVal = dmaxval;
386
387 if( minIdx )
388 ofs2idx(src, minidx, minIdx);
389 if( maxIdx )
390 ofs2idx(src, maxidx, maxIdx);
391}

418enum class ReduceMode
419{
420 FIRST_MIN = 0,
421 LAST_MIN = 1,
422 FIRST_MAX = 2,
423 LAST_MAX = 3,

426template <typename T>
427struct reduceMinMaxImpl
428{
429 void operator()(const cv::Mat& src, cv::Mat& dst, ReduceMode mode, const int axis) const
430 {
431 switch(mode)
432 {
433 case ReduceMode::FIRST_MIN:
434 reduceMinMaxApply<std::less>(src, dst, axis);
435 break;
436 case ReduceMode::LAST_MIN:
437 reduceMinMaxApply<std::less_equal>(src, dst, axis);
438 break;
439 case ReduceMode::FIRST_MAX:
440 reduceMinMaxApply<std::greater>(src, dst, axis);
441 break;
442 case ReduceMode::LAST_MAX:
443 reduceMinMaxApply<std::greater_equal>(src, dst, axis);
444 break;
445 }
446 }

448 template <template<class> class Cmp>
449 static void reduceMinMaxApply(const cv::Mat& src, cv::Mat& dst, const int axis)
450 {
451 Cmp<T> cmp;
452
453 const auto *src_ptr = src.ptr<T>();
454 auto *dst_ptr = dst.ptr<int32_t>();
455
456 const size_t outer_size = src.total(0, axis);
457 const auto mid_size = static_cast<size_t>(src.size[axis]);
458
459 const size_t outer_step = src.total(axis);
460 const size_t dst_step = dst.total(axis);
461
462 const size_t mid_step = src.total(axis + 1);
463
464 for (size_t outer = 0; outer < outer_size; ++outer)
465 {
466 const size_t outer_offset = outer * outer_step;
467 const size_t dst_offset = outer * dst_step;
468 for (size_t mid = 0; mid != mid_size; ++mid)
469 {
470 const size_t src_offset = outer_offset + mid * mid_step;
471 for (size_t inner = 0; inner < mid_step; inner++)
472 {
473 int32_t& index = dst_ptr[dst_offset + inner];
474
475 const size_t prev = outer_offset + index * mid_step + inner;
476 const size_t curr = src_offset + inner;
477
478 if (cmp(src_ptr[curr], src_ptr[prev]))
479 {
480 index = static_cast<int32_t>(mid);
481 }
482 }
483 }
484 }
485 }
486};

488static void reduceMinMax(cv::InputArray src, cv::OutputArray dst, ReduceMode mode, int axis)
489{
490 CV_INSTRUMENT_REGION();
491
492 cv::Mat srcMat = src.getMat();
493 int dims = std::max(1, srcMat.dims);
494 axis = (axis + dims) % dims;
495 CV_Assert(srcMat.channels() == 1 && axis >= 0 && axis <= srcMat.dims);
496
497 std::vector<int> sizes(srcMat.dims);
498 std::copy(srcMat.size.p, srcMat.size.p + srcMat.dims, sizes.begin());
499 if(!sizes.empty())
500 sizes[axis] = 1;
501
502 dst.create(srcMat.dims, sizes.data(), CV_32SC1); // indices
503 cv::Mat dstMat = dst.getMat();
504 dstMat.setTo(cv::Scalar::all(0));
505
506 if (!srcMat.isContinuous())
507 {
508 srcMat = srcMat.clone();
509 }
510
511 bool needs_copy = !dstMat.isContinuous();
512 if (needs_copy)
513 {
514 dstMat = dstMat.clone();
515 }
516
517 cv::detail::depthDispatch<reduceMinMaxImpl>(srcMat.depth(), srcMat, dstMat, mode, axis);
518
519 if (needs_copy)
520 {
521 dstMat.copyTo(dst);
522 }
523}
