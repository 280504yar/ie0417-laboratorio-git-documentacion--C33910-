315void warpFrame(InputArray depth, InputArray image, InputArray mask,
316 InputArray Rt, InputArray cameraMatrix,
317 OutputArray warpedDepth, OutputArray warpedImage, OutputArray warpedMask)
318{
319 CV_Assert(cameraMatrix.size() == Size(3, 3));
320 CV_Assert(cameraMatrix.depth() == CV_32F || cameraMatrix.depth() == CV_64F);
321 Matx33d K, Kinv;
322 cameraMatrix.getMat().convertTo(K, CV_64F);
323 std::vector<bool> camPlaces { /* fx */ true, false, /* cx */ true, false, /* fy */ true, /* cy */ true, false, false, /* 1 */ true};
324 for (int i = 0; i < 9; i++)
325 {
326 CV_Assert(camPlaces[i] == (K.val[i] > DBL_EPSILON));
327 }
328 Kinv = K.inv();
329
330 CV_Assert((Rt.cols() == 4) && (Rt.rows() == 3 || Rt.rows() == 4));
331 CV_Assert(Rt.depth() == CV_32F || Rt.depth() == CV_64F);
332 Mat rtmat;
333 Rt.getMat().convertTo(rtmat, CV_64F);
334 Affine3d rt(rtmat);
335
336 CV_Assert(!depth.empty());
337 CV_Assert(depth.channels() == 1);
338 double maxDepth = 0;
339 int depthDepth = depth.depth();
340 switch (depthDepth)
341 {
342 case CV_16U:
343 maxDepth = std::numeric_limits<unsigned short>::max();
344 break;
345 case CV_32F:
346 maxDepth = std::numeric_limits<float>::max();
347 break;
348 case CV_64F:
349 maxDepth = std::numeric_limits<double>::max();
350 break;
351 default:
352 CV_Error(Error::StsBadArg, "Unsupported depth data type");
353 }
354 Mat_<double> depthDbl;
355 depth.getMat().convertTo(depthDbl, CV_64F);
356 Size sz = depth.size();
357
358 Mat_<uchar> maskMat;
359 if (!mask.empty())
360 {
361 CV_Assert(mask.type() == CV_8UC1 || mask.type() == CV_8SC1 || mask.type() == CV_BoolC1);
362 CV_Assert(mask.size() == sz);
363 maskMat = mask.getMat();
364 }
365
366 int imageType = -1;
367 Mat imageMat;
368 if (!image.empty())
369 {
370 imageType = image.type();
371 CV_Assert(imageType == CV_8UC1 || imageType == CV_8UC3 || imageType == CV_8UC4);
372 CV_Assert(image.size() == sz);
373 CV_Assert(warpedImage.needed());
374 imageMat = image.getMat();
375 }
376
377 CV_Assert(warpedDepth.needed() || warpedImage.needed() || warpedMask.needed());
378
379 // Getting new coords for depth point
380
381 // see the explanation in the loop below
382 Matx33d krki = K * rt.rotation() * Kinv;
383 Matx32d krki_cols01 = krki.get_minor<3, 2>(0, 0);
384 Vec3d krki_col2(krki.col(2).val);
385
386 Vec3d ktmat = K * rt.translation();
387 Mat_<Vec3d> reprojBack(depth.size());
388 for (int y = 0; y < sz.height; y++)
389 {
390 const uchar* maskRow = maskMat.empty() ? nullptr : maskMat[y];
391 const double* depthRow = depthDbl[y];
392 Vec3d* reprojRow = reprojBack[y];
393 for (int x = 0; x < sz.width; x++)
394 {
395 double z = depthRow[x];
396 bool badz = cvIsNaN(z) || cvIsInf(z) || z <= 0 || z >= maxDepth || (maskRow && !maskRow[x]);
397 Vec3d v;
398 if (!badz)
399 {
400 // Reproject pixel (x, y) using known z, rotate+translate and project back
401 // getting new pixel in projective coordinates:
402 // v = K * Rt * K^-1 * ([x, y, 1] * z) = [new_x*new_z, new_y*new_z, new_z]
403 // v = K * (R * K^-1 * ([x, y, 1] * z) + t) =
404 // v = krki * [x, y, 1] * z + ktmat =
405 // v = (krki_cols01 * [x, y] + krki_col2) * z + K * t
406 v = (krki_cols01 * Vec2d(x, y) + krki_col2) * z + ktmat;
407 }
408 else
409 {
410 v = Vec3d();
411 }
412 reprojRow[x] = v;
413 }
414 }
415
416 // Draw new depth in z-buffer manner
417
418 Mat warpedImageMat;
419 if (warpedImage.needed())
420 {
421 warpedImage.create(sz, imageType);
422 warpedImage.setZero();
423 warpedImageMat = warpedImage.getMat();
424 }
425
426 const double infinity = std::numeric_limits<double>::max();
427
428 Mat zBuffer(sz, CV_32FC1, infinity);
429
430 const Rect rect = Rect(Point(), sz);
431
432 for (int y = 0; y < sz.height; y++)
433 {
434 uchar* imageRow1ch = nullptr;
435 Vec3b* imageRow3ch = nullptr;
436 Vec4b* imageRow4ch = nullptr;
437 switch (imageType)
438 {
439 case -1:
440 break;
441 case CV_8UC1:
442 imageRow1ch = imageMat.ptr<uchar>(y);
443 break;
444 case CV_8UC3:
445 imageRow3ch = imageMat.ptr<Vec3b>(y);
446 break;
447 case CV_8UC4:
448 imageRow4ch = imageMat.ptr<Vec4b>(y);
449 break;
450 default:
451 break;
452 }
453
454 const Vec3d* reprojRow = reprojBack[y];
455 for (int x = 0; x < sz.width; x++)
456 {
457 Vec3d v = reprojRow[x];
458 double z = v[2];
459
460 if (z > 0)
461 {
462 Point uv(cvFloor(v[0] / z), cvFloor(v[1] / z));
463 if (rect.contains(uv))
464 {
465 float oldz = zBuffer.at<float>(uv);
466
467 if (z < oldz)
468 {
469 zBuffer.at<float>(uv) = (float)z;
470
471 switch (imageType)
472 {
473 case -1:
474 break;
475 case CV_8UC1:
476 warpedImageMat.at<uchar>(uv) = imageRow1ch[x];
477 break;
478 case CV_8UC3:
479 warpedImageMat.at<Vec3b>(uv) = imageRow3ch[x];
480 break;
481 case CV_8UC4:
482 warpedImageMat.at<Vec4b>(uv) = imageRow4ch[x];
483 break;
484 default:
485 break;
486 }
487 }
488 }
489 }
490 }
491 }
492
493 if (warpedDepth.needed() || warpedMask.needed())
494 {
495 Mat goodMask = (zBuffer < infinity);
496
497 if (warpedDepth.needed())
498 {
499 warpedDepth.create(sz, depthDepth);
500
501 double badVal;
502 switch (depthDepth)
503 {
504 case CV_16U:
505 badVal = 0;
506 break;
507 case CV_32F:
508 badVal = std::numeric_limits<float>::quiet_NaN();
509 break;
510 case CV_64F:
511 badVal = std::numeric_limits<double>::quiet_NaN();
512 break;
513 default:
514 break;
515 }
516
517 zBuffer.convertTo(warpedDepth, depthDepth);
518 warpedDepth.setTo(badVal, ~goodMask);
519 }
520
521 if (warpedMask.needed())
522 {
523 warpedMask.create(sz, CV_8UC1);
524 goodMask.copyTo(warpedMask);
525 }
526 }
527}
