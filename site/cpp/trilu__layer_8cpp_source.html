26 std::vector<Mat> inputs, outputs;
27 inputs_arr.getMatVector(inputs);
28 outputs_arr.getMatVector(outputs);
29 inputs[0].copyTo(outputs[0]);
30 if (inputs[0].empty())
31 return;
32
33 int k = inputs.size() > 1 ? inputs[1].at<int64>(0,0) : 0;
34 const auto shape_input = shape(inputs[0]);
35 const int cdims = std::max(int(shape_input.size()) - 2, 0);
36 const int w = inputs[0].size[shape_input.size() - 1];
37 const int h = shape_input.size() >= 2 ? inputs[0].size[shape_input.size() - 2] : 1;
38
39 const int m = std::min(h,w);
40 int loops = 1;
41 for (int i = 0; i < cdims; ++i)
42 loops *= shape_input[i];
43
44 const size_t elemSize = outputs[0].elemSize();
45 uchar *dst = outputs[0].ptr<uchar>();
46 auto fn = [&](const Range &r) {
47 for (int i = r.start; i < r.end; i++) {
48 for(int l=0; l < m; l+=1) {
49 int cmin = upperTri ? 0 : (l + k + 1);
50 cmin = std::max(cmin, 0);
51 const int cmax = upperTri ? min(l + k -1, w-1) : w-1;
52 const int num_zeros = cmax - cmin + 1;
53 size_t offset = (static_cast<size_t>(w) * h * i + (static_cast<size_t>(w) * l + cmin)) * elemSize;
54 auto *cur_dst = dst + offset;
55 if (cmin < w && num_zeros > 0)
56 std::memset(cur_dst, 0, elemSize * num_zeros);
57 }
58 }
59 };
60
61 double nstripes = loops * h * w / 1024.0;
62 parallel_for_(Range(0, loops), fn, nstripes);
63 }
