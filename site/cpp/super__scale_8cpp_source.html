29void SuperScale::processImageScale(const Mat &src, Mat &dst, float scale, const bool &use_sr, int sr_max_size)
30{
31 scale = min(scale, MAX_SCALE);
32 if (scale > .0 && scale < 1.0)
33 { // down sample
34 resize(src, dst, Size(), scale, scale, INTER_AREA);
35 }
36 else if (scale > 1.5 && scale < 2.0)
37 {
38 resize(src, dst, Size(), scale, scale, INTER_CUBIC);
39 }
40 else if (scale >= 2.0)
41 {
42 int width = src.cols;
43 int height = src.rows;
44 if (use_sr && (int) sqrt(width * height * 1.0) < sr_max_size && net_loaded_)
45 {
46 superResolutionScale(src, dst);
47 if (scale > 2.0)
48 {
49 processImageScale(dst, dst, scale / 2.0f, use_sr);
50 }
51 }
52 else
53 { resize(src, dst, Size(), scale, scale, INTER_CUBIC); }
54 }
55}

57int SuperScale::superResolutionScale(const Mat &src, Mat &dst)
58{
59 Mat blob;
60 dnn::blobFromImage(src, blob, 1.0 / 255, Size(src.cols, src.rows), {0.0f}, false, false);
61
62 srnet_.setInput(blob);
63 auto prob = srnet_.forward();
64
65 dst = Mat(prob.size[2], prob.size[3], CV_8UC1);
66
67 for (int row = 0; row < prob.size[2]; row++)
68 {
69 const float *prob_score = prob.ptr<float>(0, 0, row);
70 auto *dst_row = dst.ptr<uchar>(row);
71 for (int col = 0; col < prob.size[3]; col++)
72 {
73 dst_row[col] = saturate_cast<uchar>(prob_score[col] * 255.0f);
74 }
75 }
76 return 0;
77}
