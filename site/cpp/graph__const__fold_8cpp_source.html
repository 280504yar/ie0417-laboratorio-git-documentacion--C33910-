24 void process()

27 netimpl->__tensors__.resize(nargs);

36 }

38 void unuse(Arg inp)
39 {
40 CV_Assert(usecounts[inp.idx] > 0);
41 if (--usecounts[inp.idx] == 0 && netimpl->isConstArg(inp)) {
42 netimpl->__tensors__[inp.idx] = Mat(); // deallocate unused tensor
43 }
44 }

46 bool processGraph(Ptr<Graph>& graph)
47 {
48 netimpl->scratchBufs.clear();
49 bool modified = false;
50 const std::vector<Ptr<LayerInfo> >& prog = graph->prog();
51 size_t i, nops = prog.size();
52 std::vector<Ptr<LayerInfo> > newprog;
53 std::vector<Arg> removed_args;
54 std::vector<Mat> inpMats, tempMats;
55 std::vector<int> inpTypes, outTypes, tempTypes;
56 std::vector<MatShape> inpShapes, outShapes, tempShapes;
57
58 for (i = 0; i < nops; i++) {
59 const Ptr<LayerInfo>& layer = prog[i];
60 std::vector<Ptr<Graph> >* subgraphs = layer->subgraphs();
61 if (subgraphs) {
62 for (Ptr<Graph>& g: *subgraphs) {
63 if (processGraph(g))
64 modified = true;
65 }
66 newprog.push_back(layer);
67 continue;
68 }
69 const std::vector<Arg>& inputs = layer->inputs;
70 const std::vector<Arg>& outputs = layer->outputs;
71 size_t j, ninputs = inputs.size(), noutputs = outputs.size();
72 bool all_const = true;
73 inpMats.assign(ninputs, Mat());
74 inpTypes.resize(ninputs);
75 inpShapes.resize(ninputs);
76 for (j = 0; j < ninputs; j++) {
77 Arg inp = inputs[j];
78 bool const_arg = netimpl->isConstArg(inp);
79 if (!const_arg)
80 all_const = false;
81 if (all_const) {
82 const Mat& m = netimpl->argTensor(inp);
83 inpMats[j] = m;
84 inpTypes[j] = m.type();
85 inpShapes[j] = m.shape();
86 }
87 }
88
89 if (all_const /*&&
90 op->supportBlockLayout(0, (int)ninputs) <= 0 // we don't currently support constant folding
91 // for block-layout operations (Convolution, MaxPool, AveragePool)
92 */) {
93 // Use a fresh vector of Mat's for outputs since we want to make these outputs the new constant tensors.
94 // So, they must be unique and don't interfere with other tensors.
95 std::vector<Mat> outMats(noutputs);
96 std::vector<std::pair<uchar*, size_t> > outOrigData;
97 if (!layer->dynamicOutputShapes())
98 netimpl->allocateLayerOutputs(layer, inpTypes, inpShapes, outTypes,
99 outShapes, outOrigData, outMats, tempTypes, tempShapes, tempMats,
100 netimpl->scratchBufs, false);
101 Ptr<Layer> execLayer = layer.dynamicCast<Layer>();
102 CV_Assert(execLayer); // const-folded ops are CPU-executable (monolithic) layers
103 execLayer->finalize(inpMats, outMats);
104 execLayer->forward(inpMats, outMats, tempMats);
105 CV_Assert(outMats.size() == noutputs);
106 for (j = 0; j < noutputs; j++) {
107 Arg out = outputs[j];
108 ArgData& out_data = netimpl->args.at(out.idx);
109 const Mat& m = outMats[j];
110 out_data.type = m.type();
111 out_data.shape = m.shape();
112 out_data.kind = DNN_ARG_CONST; // re-classify each output as constant
113 netimpl->__tensors__.at(out.idx) = m;
114 }
115
116 modified = true;
117 for (size_t i = 0; i < ninputs; i++)
118 unuse(inputs[i]);
119 //printf("folded %s: %s\n", op->name().data(), node->name().data());
120 // we don't add operation into the new program,
121 // because the output of the all-const inputs operation is now a constant,
122 // stored in a separate tensor
123 } else {
124 newprog.push_back(layer);
125 }
126 }
127
128 if (modified) {
129 graph->setProg(newprog);
130 }
131
132 return modified;
133 }
134};
