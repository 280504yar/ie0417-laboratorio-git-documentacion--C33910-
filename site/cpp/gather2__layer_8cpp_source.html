30static void gather(const Mat& data, const Mat& ind, Mat& out, int axis)
31{
32 CV_Assert_N(data.isContinuous(), ind.isContinuous(), out.isContinuous());
33 int indType = ind.type();
34 CV_Assert(indType == CV_32S || indType == CV_64S);
35
36 MatShape dataShape = data.shape();
37 MatShape indShape = ind.shape();
38 MatShape outShape = out.shape();
39 int dataDims = dataShape.dims;
40 int indDims = indShape.dims;
41 int outDims = outShape.dims;
42
43 CV_Assert(outDims == dataDims + indDims - 1);
44 size_t indTotal = indShape.total(), nslices = 1;
45 size_t elemSize = data.elemSize();
46 size_t sliceSize = elemSize;
47
48 for(int j = 0; j < dataDims; j++) {
49 int szj = dataShape[j];
50 if (j < axis)
51 nslices *= szj;
52 else if (j > axis)
53 sliceSize *= szj;
54 }
55 size_t dataStep = sliceSize * dataShape[axis];
56 size_t outStep = sliceSize * indTotal;
57 volatile bool globOutOfRangeIdx = false;
58
59 parallel_for_(Range(0, (int)indTotal), [&](const Range& r) {
60 int shape_a = dataShape[axis];
61 const uchar* dataptr0 = data.data;
62 uchar* outptr0 = out.data;
63 const int32_t* ind32 = indType == CV_32S ? ind.ptr<int32_t>() : nullptr;
64 const int64_t* ind64 = indType == CV_64S ? ind.ptr<int64_t>() : nullptr;
65 bool outOfRangeIdx = globOutOfRangeIdx;
66 for (int j = r.start; j < r.end && !outOfRangeIdx; j++) {
67 int k = ind32 ? (int)ind32[j] : (int)ind64[j];
68 uchar* outptr = outptr0 + j*sliceSize;
69 const uchar* dataptr = dataptr0;
70 for (size_t i = 0; i < nslices; i++, dataptr += dataStep, outptr += outStep) {
71 k += k < 0 ? shape_a : 0;
72 if (k < 0 || k >= shape_a) {
73 outOfRangeIdx = true;
74 break;
75 }
76 memcpy(outptr, dataptr + k*sliceSize, sliceSize);
77 }
78 }
79 if (outOfRangeIdx)
80 globOutOfRangeIdx = true;
81 }, std::min((double)indTotal, (double)sliceSize*nslices*indTotal/1e6));
82
83 if (globOutOfRangeIdx) {
84 CV_Error(Error::StsOutOfRange, "some of indices are outside of range");
85 }
86}

102 MatShape getOutShape(const MatShape& dataShape, const MatShape& indShape) const
103 {
104 int dataDims = dataShape.dims;
105 int indDims = indShape.dims;
106
107 int axis_ = normalize_axis(axis, dataDims);
108 CV_Assert(0 <= axis_ && axis_ < dataDims);
109 MatShape outShape(dataDims + indDims - 1);
110
111 for (int i = 0; i < outShape.dims; i++) {
112 if (i < axis_) {
113 outShape[i] = dataShape[i];
114 } else {
115 int j = i - axis_;
116 outShape[i] = j < indDims ? indShape[j] : dataShape[i - indDims + 1];
117 }
118 }
119 return outShape;
120 }

153 void forward(InputArrayOfArrays inputs_arr,
154 OutputArrayOfArrays outputs_arr,
155 OutputArrayOfArrays) CV_OVERRIDE
156 {
157 CV_TRACE_FUNCTION();
158 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
159
160 Size size = inputs_arr.size();
161 int ninputs = size.area();
162
163 CV_Assert(ninputs == 2);
164
165 MatShape dataShape = inputs_arr.shape(0);
166 MatShape indShape = inputs_arr.shape(1);
167 int dataType = inputs_arr.type(0);
168 int indType = inputs_arr.type(1);
169 CV_Assert(indType == CV_32S || indType == CV_64S);
170
171 MatShape outShape = getOutShape(dataShape, indShape);
172 int outKind = outputs_arr.kind();
173 int axis_ = normalize_axis(axis, dataShape.dims);
174
175 CV_Assert(outKind == _InputArray::STD_VECTOR_MAT ||
176 outKind == _InputArray::STD_VECTOR_UMAT);
177
178 if (outKind == _InputArray::STD_VECTOR_MAT) {
179 Mat data = inputs_arr.getMat(0);
180 Mat ind = inputs_arr.getMat(1);
181 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
182 outs.resize(1);
183 outs[0].fit(outShape, dataType);
184 runOp(data, ind, outs[0], axis_);
185 } else {
186 // [TODO] more efficient OpenCL implementation
187 Mat data = inputs_arr.getMat(0);
188 Mat ind = inputs_arr.getMat(1);
189 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
190 outs.resize(1);
191 outs[0].fit(outShape, dataType);
192 Mat temp(outShape, dataType);
193 runOp(data, ind, temp, axis_);
194 temp.copyTo(outs[0]);
195 }
196 }
