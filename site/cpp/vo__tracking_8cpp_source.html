15bool projectPoint(const Matx44d& T_cw, const Mat& K,
16 double Xw, double Yw, double Zw,
17 double& u, double& v)
18{
19 const double Xc = T_cw(0,0)*Xw + T_cw(0,1)*Yw + T_cw(0,2)*Zw + T_cw(0,3);
20 const double Yc = T_cw(1,0)*Xw + T_cw(1,1)*Yw + T_cw(1,2)*Zw + T_cw(1,3);
21 const double Zc = T_cw(2,0)*Xw + T_cw(2,1)*Yw + T_cw(2,2)*Zw + T_cw(2,3);
22 if (Zc <= 0.0) return false;
23 u = K.at<double>(0,0) * Xc / Zc + K.at<double>(0,2);
24 v = K.at<double>(1,1) * Yc / Zc + K.at<double>(1,2);
25 return true;
26}

28inline double descDist(const Mat& a, const Mat& b)
29{
30 if (a.empty() || b.empty()) return std::numeric_limits<double>::max();
31 int normType = (a.type() == CV_8U) ? NORM_HAMMING : NORM_L2;
32 return norm(a, b, normType);
33}

35void buildLocalMapPoints(const KeyFrame* kf, int topK,
36 std::set<MapPoint*>& localMps)
37{
38 if (!kf) return;
39 for (MapPoint* mp : kf->mapPoints)
40 if (mp && !mp->bad) localMps.insert(mp);
41
42 int k = 0;
43 for (const auto& [nbKf, cnt] : kf->orderedCovisibility)
44 {
45 if (k++ >= topK) break;

51int runPnP(const std::vector<Point3f>& obj, const std::vector<Point2f>& img,
52 Frame& frame,
53 const Mat& K, double reprojThresh, int maxIters, double confidence,
54 int minInliers)
55{
56 if ((int)obj.size() < minInliers) return -1;
57
58 Mat rvec, tvec, inlierIdx;
59 bool ok = solvePnPRansac(obj, img, K, Mat(),
60 rvec, tvec, false,
61 maxIters, (float)reprojThresh, confidence,
62 inlierIdx, SOLVEPNP_AP3P);
63 if (!ok || inlierIdx.rows < minInliers) return -1;
64
65 std::vector<Point3f> objIn;
66 std::vector<Point2f> imgIn;
67 objIn.reserve(inlierIdx.rows); imgIn.reserve(inlierIdx.rows);
68 for (int i = 0; i < inlierIdx.rows; ++i)
69 {
70 int idx = inlierIdx.at<int>(i);
71 objIn.push_back(obj[idx]);
72 imgIn.push_back(img[idx]);
73 }
74 solvePnPRefineLM(objIn, imgIn, K, Mat(), rvec, tvec);
75
76 Mat R;
77 Rodrigues(rvec, R);
78 frame.poseCw = detail::makePose(R, tvec);
79 return inlierIdx.rows;
80}

84// Motion model: constant-velocity pose prediction + projected map point search
85bool VisualOdometryImpl::trackWithMotionModel(Frame& cur)
86{
87 if (!lastKf) return false;
88
89 cur.poseCw = velocity * lastPoseCw;
90
91 std::set<MapPoint*> localMps;
92 buildLocalMapPoints(lastKf, params.localMapTopK, localMps);
93
94 std::fill(cur.mapPoints.begin(), cur.mapPoints.end(), nullptr);
95 std::fill(cur.outliers.begin(), cur.outliers.end(), false);
96
97 auto doSearch = [&](float radius) -> int {
98 int n = 0;
99 for (MapPoint* mp : localMps)
100 {
101 double u, v;
102 if (!projectPoint(cur.poseCw, K, mp->pos.x, mp->pos.y, mp->pos.z, u, v))
103 continue;
104 if (u < 0 || u >= cur.imageSize.width ||
105 v < 0 || v >= cur.imageSize.height) continue;
106
107 mp->visibleCount++;
108
109 auto cands = cur.getKeypointsInRadius((float)u, (float)v, radius);
110 double bestD = std::numeric_limits<double>::max();
111 size_t bestI = std::numeric_limits<size_t>::max();
112 for (size_t idx : cands)
113 {
114 if (cur.mapPoints[idx]) continue;
115 double d = descDist(mp->refDesc, cur.descriptors.row((int)idx));
116 if (d < bestD) { bestD = d; bestI = idx; }
117 }
118 if (bestI != std::numeric_limits<size_t>::max() &&
119 bestD < params.descProjThresh)
120 {
121 cur.mapPoints[bestI] = mp;
122 mp->foundCount++;
123 ++n;
124 }
125 }
126 return n;
127 };
128
129 int n = doSearch((float)params.motionModelRadius);
130 if (n < params.motionModelMinMatches)
131 {
132 std::fill(cur.mapPoints.begin(), cur.mapPoints.end(), nullptr);
133 n = doSearch((float)params.motionModelRadiusWide);
134 }
135
136 if (n < params.pnpMinInliers) return false;
137
138 std::vector<Point3f> obj; std::vector<Point2f> img;
139 obj.reserve(n); img.reserve(n);
140 for (size_t i = 0; i < cur.mapPoints.size(); ++i)
141 {
142 if (!cur.mapPoints[i]) continue;
143 const MapPoint* mp = cur.mapPoints[i];
144 obj.push_back(Point3f((float)mp->pos.x,(float)mp->pos.y,(float)mp->pos.z));
145 img.push_back(cur.undistKpts[i]);
146 }
147
148 int nInliers = runPnP(obj, img, cur, K,
149 params.pnpReprojThresh,
150 params.pnpRansacIters,
151 params.pnpConfidence,
152 params.pnpMinInliers);
153 if (nInliers < 0) return false;
154
155 int nOpt = poseInlierCheck(cur, K, params.pnpReprojThresh);
156 return nOpt >= params.pnpMinInliers;
157}

159// Fallback 1: descriptor match against the reference keyframe
160bool VisualOdometryImpl::trackWithReferenceKF(Frame& cur)
161{
162 if (!lastKf) return false;
163
164 std::fill(cur.mapPoints.begin(), cur.mapPoints.end(), nullptr);
165 std::fill(cur.outliers.begin(), cur.outliers.end(), false);
166
167 std::vector<DMatch> matches;
168 matchFrames(lastKf->keypoints, lastKf->descriptors, lastKf->imageSize,
169 cur.keypoints, cur.descriptors, cur.imageSize, matches);
170
171 std::vector<Point3f> obj;
172 std::vector<Point2f> img;
173 std::vector<MapPoint*> corrMps;
174 std::vector<int> corrKp;
175 obj.reserve(matches.size()); img.reserve(matches.size());
176 corrMps.reserve(matches.size()); corrKp.reserve(matches.size());
177
178 for (const auto& m : matches)
179 {
180 if ((size_t)m.queryIdx >= lastKf->mapPoints.size()) continue;
181 MapPoint* mp = lastKf->mapPoints[m.queryIdx];
182 if (!mp || mp->bad) continue;
183 obj.push_back(Point3f((float)mp->pos.x,(float)mp->pos.y,(float)mp->pos.z));
184 img.push_back(cur.undistKpts[m.trainIdx]);
185 corrMps.push_back(mp);
186 corrKp.push_back(m.trainIdx);
187 }
188
189 if ((int)obj.size() < params.pnpMinInliers)
190 {
191 lastEvent = format("refKF: 2d3d=%d < %d", (int)obj.size(), params.pnpMinInliers);
192 return false;
193 }
194
195 int nInliers = runPnP(obj, img, cur, K,
196 params.pnpReprojThresh, params.pnpRansacIters,
197 params.pnpConfidence, params.pnpMinInliers);
198 if (nInliers < 0)
199 {
200 lastEvent = "refKF: PnP failed";
201 return false;
202 }
203
204 for (size_t k = 0; k < corrMps.size(); ++k)
205 {
206 int kpIdx = corrKp[k];
207 if ((size_t)kpIdx < cur.mapPoints.size() && !cur.mapPoints[kpIdx])
208 cur.mapPoints[kpIdx] = corrMps[k];
209 }
210
211 int nOpt = poseInlierCheck(cur, K, params.pnpReprojThresh);
212 return nOpt >= params.pnpMinInliers;
213}

215// Fallback 2: optical flow when descriptor match also fails
216bool VisualOdometryImpl::trackWithOpticalFlow(Frame& cur)
217{
218 if (!hasPrevFrame || prevFrame.image.empty()) return false;
219
220 const Frame& prev = prevFrame;
221 if (prev.keypoints.empty()) return false;
222
223 // LK runs on the raw imgs
224 std::vector<Point2f> prevPts;
225 prevPts.reserve(prev.keypoints.size());
226 for (const auto& kp : prev.keypoints) prevPts.push_back(kp.pt);
227
228 std::vector<Point2f> curPts;
229 std::vector<uchar> status;
230 std::vector<float> err;
231
232 calcOpticalFlowPyrLK(prev.image, cur.image, prevPts, curPts,
233 status, err, Size(21, 21), 3,
234 TermCriteria(TermCriteria::COUNT | TermCriteria::EPS, 30, 0.01));
235
236 // PnP uses K with no distortion
237 std::vector<Point2f> curUndist;
238 if (!dist.empty())
239 undistortPoints(curPts, curUndist, K, dist, noArray(), K);
240 else
241 curUndist = curPts;
242
243 std::vector<Point3f> obj; std::vector<Point2f> img;
244 obj.reserve(prevPts.size()); img.reserve(prevPts.size());
245
246 for (size_t i = 0; i < prevPts.size(); ++i)
247 {
248 if (!status[i]) continue;
249 if (i >= prev.mapPoints.size()) continue;
250 MapPoint* mp = prev.mapPoints[i];
251 if (!mp || mp->bad) continue;
252 obj.push_back(Point3f((float)mp->pos.x,(float)mp->pos.y,(float)mp->pos.z));
253 img.push_back(curUndist[i]);
254 }
255
256 if ((int)obj.size() < params.opticalFlowMinInliers)
257 {
258 lastEvent = format("optflow: corr=%d < %d", (int)obj.size(), params.opticalFlowMinInliers);
259 return false;
260 }
261
262 int nInliers = runPnP(obj, img, cur, K,
263 params.pnpReprojThresh, params.pnpRansacIters,
264 params.pnpConfidence, params.opticalFlowMinInliers);
265 if (nInliers < 0)
266 {
267 lastEvent = "optflow: PnP failed";
268 return false;
269 }
270
271 return true;
272}

274// Local map refinement: expand coverage + recheck inliers
275void VisualOdometryImpl::trackLocalMap(Frame& cur)
276{
277 if (!lastKf) return;
278
279 std::set<MapPoint*> localMps;
280 buildLocalMapPoints(lastKf, params.localMapTopK, localMps);
281
282 int nbK = 0;
283 for (const auto& [nbKf, cnt] : lastKf->orderedCovisibility)
284 {
285 if (nbK++ >= params.localMapTopK) break;
286 int nb2K = 0;
287 for (const auto& [nb2Kf, cnt2] : nbKf->orderedCovisibility)
288 {
289 if (nb2K++ >= params.localMapNeighborK) break;
290 for (MapPoint* mp : nb2Kf->mapPoints)
291 if (mp && !mp->bad) localMps.insert(mp);
292 }
293 }
294
295 std::set<MapPoint*> alreadyMatched;
296 for (MapPoint* mp : cur.mapPoints)
297 if (mp) alreadyMatched.insert(mp);
298
299 bool anyNew = false;
300 const float r = (float)params.localMapRadius;
301
302 for (MapPoint* mp : localMps)
303 {
304 if (alreadyMatched.count(mp)) continue;
305
306 double u, v;
307 if (!projectPoint(cur.poseCw, K, mp->pos.x, mp->pos.y, mp->pos.z, u, v))
308 continue;
309 if (u < 0 || u >= cur.imageSize.width ||
310 v < 0 || v >= cur.imageSize.height) continue;
311
312 mp->visibleCount++;
313
314 auto cands = cur.getKeypointsInRadius((float)u, (float)v, r);
315 double bestD = std::numeric_limits<double>::max();
316 size_t bestI = std::numeric_limits<size_t>::max();
317 for (size_t idx : cands)
318 {
319 if (cur.mapPoints[idx]) continue;
320 double d = descDist(mp->refDesc, cur.descriptors.row((int)idx));
321 if (d < bestD) { bestD = d; bestI = idx; }
322 }
323 if (bestI != std::numeric_limits<size_t>::max() &&
324 bestD < params.descProjThresh)
325 {
326 cur.mapPoints[bestI] = mp;
327 cur.outliers[bestI] = false;
328 mp->foundCount++;
329 alreadyMatched.insert(mp);
330 anyNew = true;
331 }
332 }
333
334 if (anyNew)
335 poseInlierCheck(cur, K, params.pnpReprojThresh);
336}

338bool VisualOdometryImpl::track(Frame& cur)
339{
340 if (!lastKf)
341 {
342 refFrame = cur;
343 state = INITIALIZING;
344 return false;
345 }
346
347 // motion model tracking
348 bool ok = false;
349 if (hasVelocity)
350 ok = trackWithMotionModel(cur);
351
352 // fallback 1: descriptor match against reference keyframe
353 if (!ok)
354 ok = trackWithReferenceKF(cur);
355
356 // fallback 2: optical flow
357 if (!ok)
358 ok = trackWithOpticalFlow(cur);
359
360 if (!ok)
361 {
362 lastEvent = "track lost: all stages failed";
363 refFrame = cur;
364 state = INITIALIZING;
365 return false;
366 }
367
368 trackLocalMap(cur);
369
370 int nInliers = 0;
371 for (size_t i = 0; i < cur.mapPoints.size(); ++i)
372 if (cur.mapPoints[i] && !cur.outliers[i]) ++nInliers;
373
374 {
375 Matx44d Tcw_last_inv = lastPoseCw.inv();
376 velocity = cur.poseCw * Tcw_last_inv;
377 hasVelocity = true;
378 }
379
380 lastPoseCw = cur.poseCw;
381 map.appendPose(cur.poseCw);
382 ++framesSinceKf;
383
384 prevFrame = cur;
385 hasPrevFrame = true;
386
387 String kf_reason;
388 if (shouldPromoteKeyframe(nInliers, cur.poseCw, kf_reason))
389 {
390 int mpBefore = map.numMapPoints();
391 promoteKeyframeAndGrowMap(cur);
392 lastPoseCw = lastKf->poseCw;
393 lastEvent = format("keyframe: %s, +%d mp",
394 kf_reason.c_str(),
395 map.numMapPoints() - mpBefore);
396 }
397
398 return true;
399}
