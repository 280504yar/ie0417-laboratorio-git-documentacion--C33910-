19static inline bool isNonZeroF16(uint16_t val)
20{
21 return (val & 0x7fff) != 0;
22}

24template <typename T, typename PT>
25static size_t computeNonZeroCountsPerStripe(const T* data, size_t total, int nstripes, std::vector<size_t>& nzcounts, PT isNonZero)
26{
27 nzcounts.assign(nstripes, 0);
28 if (total == 0)
29 return 0;
30 cv::parallel_for_(cv::Range(0, nstripes), [&](const cv::Range& range) {
31 const size_t i0 = total * (size_t)range.start / (size_t)nstripes;
32 const size_t i1 = total * (size_t)range.end / (size_t)nstripes;
33 size_t nz = 0;
34 for (size_t i = i0; i < i1; ++i)
35 nz += isNonZero(data[i]);
36 nzcounts[(size_t)range.start] = nz;
37 });
38
39 size_t total_nz = 0;
40 for (size_t c : nzcounts) total_nz += c;
41 return total_nz;
42}

44template <typename T>
45static inline size_t computeNonZeroCountsPerStripe(const T* data, size_t total, int nstripes, std::vector<size_t>& nzcounts)
46{
47 return computeNonZeroCountsPerStripe<T>(data, total, nstripes, nzcounts, [](T v){ return v != (T)0; });
48}

50template <typename T, typename PT>
51static void emitIndicesStripes(const T* data,
52 size_t total,
53 int nstripes,
54 int rank,
55 const std::vector<int>& dims,
56 const std::vector<int>& strides,
57 const std::vector<size_t>& nzstart,
58 int64* y,
59 PT isNonZero)
60{
61 const size_t nnz_total = nzstart.back();
62 if (rank == 0) {
63 return;
64 }
65 const int lastDimSize = dims[rank - 1];
66
67 cv::parallel_for_(cv::Range(0, nstripes), [&](const cv::Range& range) {
68 size_t i = total * (size_t)range.start / (size_t)nstripes;
69 const size_t i1 = total * (size_t)range.end / (size_t)nstripes;
70 size_t col = nzstart[(size_t)range.start];
71
72 std::array<int, CV_MAX_DIM> coord;
73 std::fill_n(coord.begin(), rank, 0);
74 if (rank > 0) coord[rank - 1] = lastDimSize - 1;
75
76 for (; i < i1; ++i)
77 {
78 if (rank > 0 && ++coord[rank - 1] >= lastDimSize)
79 {
80 size_t idxLinear = i;
81 for (int d = rank - 1; d >= 0; --d)
82 {
83 coord[d] = static_cast<int>(idxLinear % (size_t)dims[d]);
84 idxLinear /= (size_t)dims[d];
85 }
86 }
87 if (isNonZero(data[i]))
88 {
89 for (int d = 0; d < rank; ++d)
90 y[(size_t)d * nnz_total + col] = (int64)coord[d];
91 ++col;
92 }
93 }
94 });
95}

146 void forward(InputArrayOfArrays in_arr, OutputArrayOfArrays out_arr, OutputArrayOfArrays) CV_OVERRIDE
147 {
148 CV_Assert(in_arr.size().area() == 1);
149 const Mat X = in_arr.getMat(0);
150 CV_Assert(X.total() == 0 || X.data != nullptr);
151
152 const int rank = X.dims;
153 std::vector<int> dims(rank), strides(rank);
154 for (int i = 0; i < rank; ++i) dims[i] = X.size[i];
155
156 if (rank > 0) {
157 strides.back() = 1;
158 for (int i = rank - 2; i >= 0; --i)
159 strides[i] = strides[i + 1] * dims[i + 1];
160 }
161
162 size_t total = X.total();
163 const int nstripes = 16;
164
165 const int depth = CV_MAT_DEPTH(X.type());
166 std::vector<size_t> nzcounts;
167 std::vector<size_t> nzstart;
168 size_t nnz = 0;
169 switch (depth)
170 {
171 case CV_Bool:
172 case CV_8U:
173 case CV_8S: { nnz = computeNonZeroCountsPerStripe<uchar>(X.ptr<uchar>(), total, nstripes, nzcounts); break; }
174 case CV_16U:
175 case CV_16S: { nnz = computeNonZeroCountsPerStripe<uint16_t>(X.ptr<uint16_t>(), total, nstripes, nzcounts); break; }
176 case CV_32U:
177 case CV_32S: { nnz = computeNonZeroCountsPerStripe<uint32_t>(X.ptr<uint32_t>(), total, nstripes, nzcounts); break; }
178 case CV_64U:
179 case CV_64S: { nnz = computeNonZeroCountsPerStripe<uint64_t>(X.ptr<uint64_t>(), total, nstripes, nzcounts); break; }
180 case CV_32F: { nnz = computeNonZeroCountsPerStripe<float>(X.ptr<float>(), total, nstripes, nzcounts); break; }
181 case CV_64F: { nnz = computeNonZeroCountsPerStripe<double>(X.ptr<double>(), total, nstripes, nzcounts); break; }
182 case CV_16F:
183 case CV_16BF:{ nnz = computeNonZeroCountsPerStripe<uint16_t>(X.ptr<uint16_t>(), total, nstripes, nzcounts, [](uint16_t v){ return isNonZeroF16(v); }); break; }
184 default:
185 CV_Error_(Error::StsError, ("NonZero: Unsupported input depth=%d", depth));
186 }
187
188 nzstart.assign(nzcounts.size() + 1, 0);
189 for (size_t i = 1; i <= nzcounts.size(); ++i)
190 nzstart[i] = nzstart[i - 1] + nzcounts[i - 1];
191
192 MatShape outShape = shape(rank, (int)nnz);
193 auto kind = out_arr.kind();
194 std::vector<Mat>* out_mats = nullptr;
195 std::vector<UMat>* out_umats = nullptr;
196 Mat Y;
197 if (kind == _InputArray::STD_VECTOR_MAT) {
198 out_mats = &out_arr.getMatVecRef();
199 out_mats->resize(1);
200 out_mats->at(0).fit(outShape, CV_64S);
201 Y = out_mats->at(0);
202 } else {
203 CV_Assert(kind == _InputArray::STD_VECTOR_UMAT);
204 out_umats = &out_arr.getUMatVecRef();
205 out_umats->resize(1);
206 out_umats->at(0).fit(outShape, CV_64S);
207 Y = Mat(outShape, CV_64S);
208 }
209
210 if (nnz == 0) return;
211
212 int64* y = Y.ptr<int64>();
213
214 switch (depth)
215 {
216 case CV_Bool:
217 case CV_8U:
218 case CV_8S: { emitIndicesStripes<uchar>(X.ptr<uchar>(), total, nstripes, rank, dims, strides, nzstart, y); break; }
219 case CV_16U:
220 case CV_16S: { emitIndicesStripes<uint16_t>(X.ptr<uint16_t>(), total, nstripes, rank, dims, strides, nzstart, y); break; }
221 case CV_32U:
222 case CV_32S: { emitIndicesStripes<uint32_t>(X.ptr<uint32_t>(), total, nstripes, rank, dims, strides, nzstart, y); break; }
223 case CV_64U:
224 case CV_64S: { emitIndicesStripes<uint64_t>(X.ptr<uint64_t>(), total, nstripes, rank, dims, strides, nzstart, y); break; }
225 case CV_32F: { emitIndicesStripes<float>(X.ptr<float>(), total, nstripes, rank, dims, strides, nzstart, y); break; }
226 case CV_64F: { emitIndicesStripes<double>(X.ptr<double>(), total, nstripes, rank, dims, strides, nzstart, y); break; }
227 case CV_16F:
228 case CV_16BF:{ emitIndicesStripes<uint16_t>(X.ptr<uint16_t>(), total, nstripes, rank, dims, strides, nzstart, y, [](uint16_t v){ return isNonZeroF16(v); }); break; }
229 default: break;
230 }
231
232 if (kind == _InputArray::STD_VECTOR_UMAT) {
233 Y.copyTo(out_umats->at(0));
234 }
235 }
