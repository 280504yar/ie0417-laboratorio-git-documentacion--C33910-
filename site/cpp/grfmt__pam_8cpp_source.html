128static bool
129rgb_convert (void *src, void *target, int width, int target_channels, int target_depth, bool use_rgb)
130{
131 bool ret = false;
132 if (target_channels == 3) {
133 switch (target_depth) {
134 case CV_8U:
135 if (use_rgb)
136 memcpy(target, src, sizeof(uchar) * width);
137 else
138 icvCvt_RGB2BGR_8u_C3R( (uchar*) src, 0, (uchar*) target, 0,
139 Size(width,1) );
140 ret = true;
141 break;
142 case CV_16U:
143 if (use_rgb)
144 memcpy(target, src, sizeof(ushort) * width);
145 else
146 icvCvt_RGB2BGR_16u_C3R( (ushort *)src, 0, (ushort *)target, 0,
147 Size(width,1) );
148 ret = true;
149 break;
150 default:
151 break;
152 }
153 } else if (target_channels == 1) {
154 switch (target_depth) {
155 case CV_8U:
156 icvCvt_BGR2Gray_8u_C3C1R( (uchar*) src, 0, (uchar*) target, 0,
157 Size(width,1), 2 );
158 ret = true;
159 break;
160 case CV_16U:
161 icvCvt_BGRA2Gray_16u_CnC1R( (ushort *)src, 0, (ushort *)target, 0,
162 Size(width,1), 3, 2 );
163 ret = true;
164 break;
165 default:
166 break;
167 }
168 }
169 return ret;
170}

177static void
178basic_conversion (void *src, const struct channel_layout *layout, int src_sample_size,
179 int src_width, void *target, int target_channels, int target_depth, bool use_rgb)
180{
181 switch (target_depth) {
182 case CV_8U:
183 {
184 uchar *d = (uchar *)target, *s = (uchar *)src,
185 *end = ((uchar *)src) + src_width * src_sample_size;
186 switch (target_channels) {
187 case 1:
188 for( ; s < end; d += 1, s += src_sample_size )
189 d[0] = s[layout->graychan];
190 break;
191 case 3:
192 if (use_rgb)
193 for( ; s < end; d += 3, s += src_sample_size ) {
194 d[0] = s[layout->rchan];
195 d[1] = s[layout->gchan];
196 d[2] = s[layout->bchan];
197 }
198 else
199 for( ; s < end; d += 3, s += src_sample_size ) {
200 d[0] = s[layout->bchan];
201 d[1] = s[layout->gchan];
202 d[2] = s[layout->rchan];
203 }
204 break;
205 default:
206 CV_Error(Error::StsInternal, "");
207 }
208 break;
209 }
210 case CV_16U:
211 {
212 ushort *d = (ushort *)target, *s = (ushort *)src,
213 *end = ((ushort *)src) + src_width * src_sample_size;
214 switch (target_channels) {
215 case 1:
216 for( ; s < end; d += 1, s += src_sample_size )
217 d[0] = s[layout->graychan];
218 break;
219 case 3:
220 if (use_rgb)
221 for( ; s < end; d += 3, s += src_sample_size ) {
222 d[0] = s[layout->rchan];
223 d[1] = s[layout->gchan];
224 d[2] = s[layout->bchan];
225 }
226 else
227 for( ; s < end; d += 3, s += src_sample_size ) {
228 d[0] = s[layout->bchan];
229 d[1] = s[layout->gchan];
230 d[2] = s[layout->rchan];
231 }
232 break;
233 default:
234 CV_Error(Error::StsInternal, "");
235 }
236 break;
237 }
238 default:
239 CV_Error(Error::StsInternal, "");
240 }

244static
245bool ReadPAMHeaderLine(
246 cv::RLByteStream& strm,
247 CV_OUT PamHeaderFieldType &fieldtype,
248 CV_OUT char value[MAX_PAM_HEADER_VALUE_LENGTH+1])
249{
250 int code;
251 char ident[MAX_PAM_HEADER_IDENITFIER_LENGTH+1] = {};
252
253 do {
254 code = strm.getByte();
255 } while ( isspace(code) );
256
257 if (code == '#') {
258 /* we are in a comment, eat characters until linebreak */
259 do
260 {
261 code = strm.getByte();
262 } while( code != '\n' && code != '\r' );
263 fieldtype = PAM_HEADER_COMMENT;
264 return true;
265 } else if (code == '\n' || code == '\r' ) {
266 fieldtype = PAM_HEADER_NONE;
267 return true;
268 }
269
270 int ident_sz = 0;
271 for (; ident_sz < MAX_PAM_HEADER_IDENITFIER_LENGTH; ident_sz++)
272 {
273 if (isspace(code))
274 break;
275 ident[ident_sz] = (char)code;
276 code = strm.getByte();
277 }
278 CV_DbgAssert(ident_sz <= MAX_PAM_HEADER_IDENITFIER_LENGTH);
279 ident[ident_sz] = 0;
280
281 /* we may have filled the buffer and still have data */
282 if (!isspace(code))
283 return false;
284
285 bool ident_found = false;
286 for (uint i = 0; i < PAM_FIELDS_NO; i++)
287 {
288 if (0 == strncmp(fields[i].identifier, ident, std::min(ident_sz, MAX_PAM_HEADER_IDENITFIER_LENGTH) + 1))
289 {
290 fieldtype = fields[i].type;
291 ident_found = true;
292 break;
293 }
294 }
295
296 if (!ident_found)
297 return false;
298
299 memset(value, 0, sizeof(char) * (MAX_PAM_HEADER_VALUE_LENGTH + 1));
300 /* we may have an identifier that has no value */
301 if (code == '\n' || code == '\r')
302 return true;
303
304 do {
305 code = strm.getByte();
306 } while (isspace(code));
307
308 /* read identifier value */
309 int value_sz = 0;
310 for (; value_sz < MAX_PAM_HEADER_VALUE_LENGTH; value_sz++)
311 {
312 if (code == '\n' || code == '\r')
313 break;
314 value[value_sz] = (char)code;
315 code = strm.getByte();
316 }
317 CV_DbgAssert(value_sz <= MAX_PAM_HEADER_VALUE_LENGTH);
318 value[value_sz] = 0;
319
320 int pos = value_sz;
321
322 /* should be terminated */
323 if (code != '\n' && code != '\r')
324 return false;
325
326 /* remove trailing white spaces */
327 while (--pos >= 0 && isspace(value[pos]))
328 value[pos] = 0;
329
330 return true;
331}

333static int ParseInt(const char *str, int len)
334{
335 CV_Assert(len > 0);
336
337 int pos = 0;
338 bool is_negative = false;
339 if (str[0] == '-')
340 {
341 is_negative = true;
342 pos++;
343 CV_Assert(isdigit(str[pos]));
344 }
345 uint64_t number = 0;
346 while (pos < len && isdigit(str[pos]))
347 {
348 char ch = str[pos];
349 number = (number * 10) + (uint64_t)((int)ch - (int)'0');
350 CV_Assert(number < INT_MAX);
351 pos++;
352 }
353 if (pos < len)
354 CV_Assert(str[pos] == 0);
355 return (is_negative) ? -(int)number : (int)number;
356}

394bool PAMDecoder::readHeader()
395{
396 PamHeaderFieldType fieldtype = PAM_HEADER_NONE;
397 char value[MAX_PAM_HEADER_VALUE_LENGTH+1];
398 int byte;
399
400 if( !m_buf.empty() )
401 {
402 if( !m_strm.open(m_buf) )
403 return false;
404 }
405 else if( !m_strm.open( m_filename ))
406 return false;
407
408 try
409 {
410 byte = m_strm.getByte();
411 if( byte != 'P' )
412 throw RBS_BAD_HEADER;
413
414 byte = m_strm.getByte();
415 if (byte != '7')
416 throw RBS_BAD_HEADER;
417
418 byte = m_strm.getByte();
419 if (byte != '\n' && byte != '\r')
420 throw RBS_BAD_HEADER;
421
422 bool flds_endhdr = false, flds_height = false, flds_width = false, flds_depth = false, flds_maxval = false;
423
424 do {
425 if (!ReadPAMHeaderLine(m_strm, fieldtype, value))
426 throw RBS_BAD_HEADER;
427 switch (fieldtype)
428 {
429 case PAM_HEADER_NONE:
430 case PAM_HEADER_COMMENT:
431 continue;
432 case PAM_HEADER_ENDHDR:
433 flds_endhdr = true;
434 break;
435 case PAM_HEADER_HEIGHT:
436 if (flds_height)
437 throw RBS_BAD_HEADER;
438 m_height = ParseInt(value, MAX_PAM_HEADER_VALUE_LENGTH);
439 flds_height = true;
440 break;
441 case PAM_HEADER_WIDTH:
442 if (flds_width)
443 throw RBS_BAD_HEADER;
444 m_width = ParseInt(value, MAX_PAM_HEADER_VALUE_LENGTH);
445 flds_width = true;
446 break;
447 case PAM_HEADER_DEPTH:
448 if (flds_depth)
449 throw RBS_BAD_HEADER;
450 m_channels = ParseInt(value, MAX_PAM_HEADER_VALUE_LENGTH);
451 flds_depth = true;
452 break;
453 case PAM_HEADER_MAXVAL:
454 if (flds_maxval)
455 throw RBS_BAD_HEADER;
456 m_maxval = ParseInt(value, MAX_PAM_HEADER_VALUE_LENGTH);
457 if ( m_maxval > 65535 )
458 throw RBS_BAD_HEADER;
459 m_sampledepth = (m_maxval > 255) ? CV_16U : CV_8U;
460 if (m_maxval == 1)
461 bit_mode = true;
462 flds_maxval = true;
463 break;
464 case PAM_HEADER_TUPLTYPE:
465 {
466 bool format_found = false;
467 for (uint i=0; i<PAM_FORMATS_NO; i++)
468 {
469 if (0 == strncmp(formats[i].name, value, MAX_PAM_HEADER_VALUE_LENGTH+1))
470 {
471 selected_fmt = formats[i].fmt;
472 format_found = true;
473 break;
474 }
475 }
476 CV_Assert(format_found);
477 break;
478 }
479 default:
480 throw RBS_BAD_HEADER;
481 }
482 } while (fieldtype != PAM_HEADER_ENDHDR);
483
484 if (selected_fmt != IMWRITE_PAM_FORMAT_NULL && flds_depth) {
485 if (selected_fmt == IMWRITE_PAM_FORMAT_BLACKANDWHITE && m_channels != 1) {
486 CV_Error(Error::StsError, "fmt is IMWRITE_PAM_FORMAT_BLACKANDWHITE but number of channels is not 1");
487 }
488 if (selected_fmt == IMWRITE_PAM_FORMAT_GRAYSCALE && m_channels != 1) {
489 CV_Error(Error::StsError, "fmt is IMWRITE_PAM_FORMAT_GRAYSCALE but number of channels is not 1");
490 }
491 if (selected_fmt == IMWRITE_PAM_FORMAT_GRAYSCALE_ALPHA && m_channels != 2) {
492 CV_Error(Error::StsError, "fmt is IMWRITE_PAM_FORMAT_GRAYSCALE_ALPHA but number of channels is not 2");
493 }
494 if (selected_fmt == IMWRITE_PAM_FORMAT_RGB && m_channels != 3) {
495 CV_Error(Error::StsError, "fmt is IMWRITE_PAM_FORMAT_RGB but number of channels is not 3");
496 }
497 if (selected_fmt == IMWRITE_PAM_FORMAT_RGB_ALPHA && m_channels != 4) {
498 CV_Error(Error::StsError, "fmt is IMWRITE_PAM_FORMAT_RGB_ALPHA but number of channels is not 4");
499 }
500 }
501 if (flds_endhdr && flds_height && flds_width && flds_depth && flds_maxval)
502 {
503 if (selected_fmt == IMWRITE_PAM_FORMAT_NULL)
504 {
505 if (m_channels == 1 && m_maxval == 1)
506 selected_fmt = IMWRITE_PAM_FORMAT_BLACKANDWHITE;
507 else if (m_channels == 1 && m_maxval < 256)
508 selected_fmt = IMWRITE_PAM_FORMAT_GRAYSCALE;
509 else if (m_channels == 3 && m_maxval < 256)
510 selected_fmt = IMWRITE_PAM_FORMAT_RGB;
511 else
512 CV_Error(Error::StsError, "Can't determine selected_fmt (IMWRITE_PAM_FORMAT_NULL)");
513 }
514 CV_CheckDepth(m_sampledepth, m_sampledepth == CV_8U || m_sampledepth == CV_16U, "");
515 CV_Check(m_channels, m_channels >= 1 && m_channels <= 4, "Unsupported number of channels");
516 m_type = CV_MAKETYPE(m_sampledepth, m_channels);
517 m_offset = m_strm.getPos();
518
519 return true;
520 }
521
522 // failed
523 m_offset = -1;
524 m_width = m_height = -1;
525 m_strm.close();
526 return false;
527 }
528 catch (...)
529 {
530 m_offset = -1;
531 m_width = m_height = -1;
532 m_strm.close();
533 throw;
534 }
535}

538bool PAMDecoder::readData(Mat& img)
539{
540 uchar* data = img.ptr();
541 const int target_channels = img.channels();
542 size_t imp_stride = img.step;
543 const int sample_depth = CV_ELEM_SIZE1(m_type);
544 const int src_elems_per_row = m_width*m_channels;
545 const int src_stride = src_elems_per_row*sample_depth;
546 PaletteEntry palette[256] = {};
547 const struct pam_format *fmt = NULL;
548 struct channel_layout layout = { 0, 0, 0, 0 }; // normalized to 1-channel grey format
549
550 /* setting buffer to max data size so scaling up is possible */
551 AutoBuffer<uchar> _src(src_elems_per_row * 2);
552 uchar* src = _src.data();
553
554 if( m_offset < 0 || !m_strm.isOpened())
555 return false;
556
557 if (selected_fmt != IMWRITE_PAM_FORMAT_NULL)
558 fmt = &formats[selected_fmt];
559 else {
560 /* default layout handling */
561 if (m_channels >= 3) {
562 layout.bchan = 0;
563 layout.gchan = 1;
564 layout.rchan = 2;
565 }
566 }
567
568 {
569 m_strm.setPos( m_offset );
570
571 /* the case where data fits the opencv matrix */
572 if (m_sampledepth == img.depth() && target_channels == m_channels && !bit_mode) {
573 /* special case for 16bit images with wrong endianness */
574 if (m_sampledepth == CV_16U && !isBigEndian())
575 {
576 for (int y = 0; y < m_height; y++, data += imp_stride)
577 {
578 m_strm.getBytes( src, src_stride );
579 for (int x = 0; x < src_elems_per_row; x++)
580 {
581 uchar v = src[x * 2];
582 data[x * 2] = src[x * 2 + 1];
583 data[x * 2 + 1] = v;
584 }
585 }
586 }
587 else {
588 m_strm.getBytes( data, src_stride * m_height );
589 }
590
591 }
592 else {
593 /* black and white mode */
594 if (bit_mode) {
595 if( target_channels == 1 )
596 {
597 uchar gray_palette[2] = {0, 255};
598 for (int y = 0; y < m_height; y++, data += imp_stride)
599 {
600 m_strm.getBytes( src, src_stride );
601 FillGrayRow1( data, src, m_width, gray_palette );
602 }
603 } else if ( target_channels == 3 )
604 {
605 FillGrayPalette( palette, 1 , false );
606 for (int y = 0; y < m_height; y++, data += imp_stride)
607 {
608 m_strm.getBytes( src, src_stride );
609 FillColorRow1( data, src, m_width, palette );
610 }
611 }
612 else
613 {
614 CV_Error(Error::StsError, cv::format("Unsupported value of target_channels: %d", target_channels));
615 }
616 } else {
617 for (int y = 0; y < m_height; y++, data += imp_stride)
618 {
619 m_strm.getBytes( src, src_stride );
620
621 /* endianness correction */
622 if( m_sampledepth == CV_16U && !isBigEndian() )
623 {
624 for (int x = 0; x < src_elems_per_row; x++)
625 {
626 uchar v = src[x * 2];
627 src[x * 2] = src[x * 2 + 1];
628 src[x * 2 + 1] = v;
629 }
630 }
631
632 /* scale down */
633 if( img.depth() == CV_8U && m_sampledepth == CV_16U )
634 {
635 for (int x = 0; x < src_elems_per_row; x++)
636 {
637 int v = ((ushort *)src)[x];
638 src[x] = (uchar)(v >> 8);
639 }
640 }
641
642 /* if we are only scaling up/down then we can then copy the data */
643 if (target_channels == m_channels) {
644 memcpy (data, src, imp_stride);
645 }
646 /* perform correct conversion based on format */
647 else if (fmt) {
648 bool funcout = false;
649 if (fmt->cvt_func)
650 funcout = fmt->cvt_func (src, data, m_width, target_channels,
651 img.depth(), m_use_rgb);
652 /* fall back to default if there is no conversion function or it
653 * can't handle the specified characteristics
654 */
655 if (!funcout)
656 basic_conversion (src, &fmt->layout, m_channels,
657 m_width, data, target_channels, img.depth(), m_use_rgb);
658
659 /* default to selecting the first available channels */
660 } else {
661 basic_conversion (src, &layout, m_channels,
662 m_width, data, target_channels, img.depth(), m_use_rgb);
663 }
664 }
665 }
666 }
667 }
668 return true;
669}

699bool PAMEncoder::write( const Mat& img, const std::vector<int>& params )
700{
701
702 WLByteStream strm;
703
704 int width = img.cols, height = img.rows;
705 int stride = width*(int)img.elemSize();
706 const uchar* data = img.ptr();
707 const struct pam_format *fmt = NULL;
708 int x, y, tmp, bufsize = 256;
709
710 /* parse save file type */
711 for( size_t i = 0; i < params.size(); i += 2 ) {
712 const int value = params[i+1];
713 if( params[i] == IMWRITE_PAM_TUPLETYPE ) {
714 switch(value) {
715 case IMWRITE_PAM_FORMAT_NULL:
716 case IMWRITE_PAM_FORMAT_BLACKANDWHITE:
717 case IMWRITE_PAM_FORMAT_GRAYSCALE:
718 case IMWRITE_PAM_FORMAT_GRAYSCALE_ALPHA:
719 case IMWRITE_PAM_FORMAT_RGB:
720 case IMWRITE_PAM_FORMAT_RGB_ALPHA:
721 fmt = &formats[value];
722 break;
723 default:
