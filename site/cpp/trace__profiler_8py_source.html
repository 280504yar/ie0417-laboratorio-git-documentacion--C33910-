78def getCXXFunctionName(spec):
79 def dropParams(spec):
80 pos = len(spec) - 1
81 depth = 0
82 while pos >= 0:
83 if spec[pos] == ')':
84 depth = depth + 1
85 elif spec[pos] == '(':
86 depth = depth - 1
87 if depth == 0:
88 if pos == 0 or spec[pos - 1] in ['#', ':']:
89 res = dropParams(spec[pos+1:-1])
90 return (spec[:pos] + res[0], res[1])
91 return (spec[:pos], spec[pos:])
92 pos = pos - 1

95 def extractName(spec):
96 pos = len(spec) - 1
97 inName = False
98 while pos >= 0:
99 if spec[pos] == ' ':
100 if inName:
101 return spec[pos+1:]
102 elif spec[pos].isalnum():
103 inName = True
104 pos = pos - 1
105 return spec
106
107 if spec.startswith('IPP') or spec.startswith('OpenCL'):
108 prefix_size = len('IPP') if spec.startswith('IPP') else len('OpenCL')
109 prefix = spec[:prefix_size]
110 if prefix_size < len(spec) and spec[prefix_size] in ['#', ':']:
111 prefix = prefix + spec[prefix_size]
112 prefix_size = prefix_size + 1
113 begin = prefix_size
114 while begin < len(spec):
115 if spec[begin].isalnum() or spec[begin] in ['_', ':']:
116 break
117 begin = begin + 1
118 if begin == len(spec):
119 return spec
120 end = begin
121 while end < len(spec):
122 if not (spec[end].isalnum() or spec[end] in ['_', ':']):
123 break
124 end = end + 1
125 return prefix + spec[begin:end]
126
127 spec = spec.replace(') const', ')') # const methods
128 (ret_type_name, params) = dropParams(spec)
129 name = extractName(ret_type_name)
130 if 'operator' in name:
131 return name + params
132 if name.startswith('&'):
133 return name[1:]
134 return name
