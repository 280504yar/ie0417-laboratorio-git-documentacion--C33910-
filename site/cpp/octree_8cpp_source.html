52bool OctreeNode::isPointInBound(const Point3f& _point) const
53{
54 Point3f eps;
55 eps.x = std::max(std::abs(_point.x), std::abs(this->origin.x));
56 eps.y = std::max(std::abs(_point.y), std::abs(this->origin.y));
57 eps.z = std::max(std::abs(_point.z), std::abs(this->origin.z));
58 eps *= std::numeric_limits<float>::epsilon();
59 Point3f ptEps = _point + eps;
60 Point3f upPt = this->origin + eps + Point3f {(float)this->size, (float)this->size, (float)this->size};
61
62 return (ptEps.x >= this->origin.x) &&
63 (ptEps.y >= this->origin.y) &&
64 (ptEps.z >= this->origin.z) &&
65 (_point.x <= upPt.x) &&
66 (_point.y <= upPt.y) &&
67 (_point.z <= upPt.z);
68}

154bool Octree::Impl::insertPoint(const Point3f& point, const Point3f &color)
155{
156 size_t depthMask = (size_t)(1ULL << (this->maxDepth - 1));
157
158 if(this->rootNode.empty())
159 {
160 this->rootNode = new OctreeNode( 0, this->size, this->origin, -1);
161 }
162
163 bool pointInBoundFlag = this->rootNode->isPointInBound(point);
164 if(this->rootNode->depth == 0 && !pointInBoundFlag)
165 {
166 return false;
167 }
168
169 OctreeKey key((size_t)floor((point.x - this->origin.x) / this->resolution),
170 (size_t)floor((point.y - this->origin.y) / this->resolution),
171 (size_t)floor((point.z - this->origin.z) / this->resolution));
172
173 Ptr<OctreeNode> node = this->rootNode;
174 while (node->depth != maxDepth)
175 {
176 double childSize = node->size * 0.5;
177
178 // calculate the index and the origin of child.
179 size_t childIndex = key.findChildIdxByMask(depthMask);
180 size_t xIndex = (childIndex & 1) ? 1 : 0;
181 size_t yIndex = (childIndex & 2) ? 1 : 0;
182 size_t zIndex = (childIndex & 4) ? 1 : 0;
183
184 Point3f childOrigin = node->origin + Point3f(float(xIndex), float(yIndex), float(zIndex)) * float(childSize);
185
186 Ptr<OctreeNode> &childPtr = node->children[childIndex];
187 if (!childPtr)
188 {
189 childPtr = new OctreeNode(node->depth + 1, childSize, childOrigin, int(childIndex));
190 childPtr->parent = node;
191 }
192
193 node = childPtr;
194 depthMask = depthMask >> 1;
195 }
196
197 node->isLeaf = true;
198 node->pointList.push_back(point);
199 node->colorList.push_back(color);
200
201 return true;
202}

205static Vec6f getBoundingBox(const Mat& points)
206{
207 const float mval = std::numeric_limits<float>::max();
208 Vec6f bb(mval, mval, mval, -mval, -mval, -mval);
209
210 for (int i = 0; i < (int)points.total(); i++)
211 {
212 Point3f pt = points.at<Point3f>(i);
213 bb[0] = min(bb[0], pt.x);
214 bb[1] = min(bb[1], pt.y);
215 bb[2] = min(bb[2], pt.z);
216 bb[3] = max(bb[3], pt.x);
217 bb[4] = max(bb[4], pt.y);
218 bb[5] = max(bb[5], pt.z);
219 }
220
221 return bb;
222}

224void Octree::Impl::fill(bool useResolution, InputArray _points, InputArray _colors)
225{
226 CV_CheckFalse(_points.empty(), "No points provided");
227
228 Mat points, colors;
229 int nPoints = 0, nColors = 0;
230
231 int pointType = _points.type();
232 CV_Assert(pointType == CV_32FC1 || pointType == CV_32FC3);
233 points = _points.getMat();
234 // transform 3xN matrix to Nx3, except 3x3
235 if ((_points.channels() == 1) && (_points.rows() == 3) && (_points.cols() != 3))
236 {
237 points = points.t();
238 }
239 // This transposition is performed on 1xN matrix so it's almost free in terms of performance
240 points = points.reshape(3, 1).t();
241 nPoints = (int)points.total();
242
243 if (!_colors.empty())
244 {
245 int colorType = _colors.type();
246 CV_Assert(colorType == CV_32FC1 || colorType == CV_32FC3);
247 colors = _colors.getMat();
248 // transform 3xN matrix to Nx3, except 3x3
249 if ((_colors.channels() == 1) && (_colors.rows() == 3) && (_colors.cols() != 3))
250 {
251 colors = colors.t();
252 }
253 colors = colors.reshape(3, 1).t();
254 nColors = (int)colors.total();
255
256 CV_Assert(nColors == nPoints);
257 this->hasColor = true;
258 }
259
260 Vec6f bbox = getBoundingBox(points);
261 Point3f minBound(bbox[0], bbox[1], bbox[2]);
262 Point3f maxBound(bbox[3], bbox[4], bbox[5]);
263
264 double maxSize = max(max(maxBound.x - minBound.x, maxBound.y - minBound.y), maxBound.z - minBound.z);
265
266 // Extend maxSize to the closest power of 2 that exceeds it for bit operations
267 maxSize = double(1 << int(ceil(log2(maxSize))));
268
269 // to calculate maxDepth from resolution or vice versa
270 if (useResolution)
271 {
272 this->maxDepth = (int)ceil(log2(maxSize / this->resolution));
273 }
274 else
275 {
276 this->resolution = (maxSize / (1 << (this->maxDepth + 1)));
277 }
278
279 this->size = (1 << this->maxDepth) * this->resolution;
280 this->origin = Point3f(float(floor(minBound.x / this->resolution) * this->resolution),
281 float(floor(minBound.y / this->resolution) * this->resolution),
282 float(floor(minBound.z / this->resolution) * this->resolution));
283
284 // Insert every point in PointCloud data.
285 for (int idx = 0; idx < nPoints; idx++)
286 {
287 Point3f pt = points.at<Point3f>(idx);
288 Point3f insertColor = this->hasColor ? colors.at<Point3f>(idx) : Point3f { };
289 if (!this->insertPoint(pt, insertColor))
290 {
291 CV_Error(Error::StsBadArg, "The point is out of boundary!");
292 }
293 }

313bool Octree::deletePoint(const Point3f& point)
314{
315 OctreeKey key = OctreeKey((size_t)floor((point.x - this->p->origin.x) / p->resolution),
316 (size_t)floor((point.y - this->p->origin.y) / p->resolution),
317 (size_t)floor((point.z - this->p->origin.z) / p->resolution));
318 size_t depthMask = (size_t)1 << (p->maxDepth - 1);
319
320 Ptr<OctreeNode> node = p->rootNode;
321 while(node)
322 {
323 if (node->empty())
324 {
325 node = nullptr;
326 }
327 else if (node->isLeaf)
328 {
329 const float eps = 1e-9f;
330 bool found = std::any_of(node->pointList.begin(), node->pointList.end(),
331 [point, eps](const Point3f& pt) -> bool
332 {
333 return abs(point.x - pt.x) < eps &&
334 abs(point.y - pt.y) < eps &&
335 abs(point.z - pt.z) < eps;
336 });
337 if (!found)
338 node = nullptr;
339 break;
340 }
341 else
342 {
343 node = node->children[key.findChildIdxByMask(depthMask)];
344 depthMask = depthMask >> 1;
345 }
346 }
347
348 if(!node)
349 return false;
350
351 const float eps = 1e-9f;
352
353 // we've found a leaf node and delete all verts equal to given one
354 size_t ctr = 0;
355 while (!node->pointList.empty() && ctr < node->pointList.size())
356 {
357 if (abs(point.x - node->pointList[ctr].x) < eps &&
358 abs(point.y - node->pointList[ctr].y) < eps &&
359 abs(point.z - node->pointList[ctr].z) < eps)
360 {
361 node->pointList.erase(node->pointList.begin() + ctr);
362 }
363 else
364 {
365 ctr++;
366 }
367 }
368
369 if (node->pointList.empty())
370 {
371 // empty node and its empty parents should be removed
372 OctreeNode *parentPtr = node->parent;
373 int parentdIdx = node->parentIndex;
374
375 while (parentPtr)
376 {
377 parentPtr->children[parentdIdx].release();
378
379 // check if all children were deleted
380 bool deleteFlag = true;
381 for (size_t i = 0; i < 8; i++)
382 {
383 if (!parentPtr->children[i].empty())
384 {
385 deleteFlag = false;
386 break;
387 }
388 }
389
390 if (deleteFlag)
391 {
392 // we're at empty node, going up
393 parentdIdx = parentPtr->parentIndex;
394 parentPtr = parentPtr->parent;
395 }
396 else
397 {
398 // reached first non-empty node, stopping
399 parentPtr = nullptr;
400 }
401 }
402 }
403 return true;
404}

407void Octree::getPointCloudByOctree(OutputArray restorePointCloud, OutputArray restoreColor)
408{
409 Ptr<OctreeNode> root = p->rootNode;
410 double resolution = p->resolution;
411 std::vector<Point3f> outPts, outColors;
412
413 typedef std::tuple<Ptr<OctreeNode>, size_t, size_t, size_t> stack_element;
414 std::stack<stack_element> toCheck;
415 toCheck.push(stack_element(root, 0, 0, 0));
416 while (!toCheck.empty())
417 {
418 auto top = toCheck.top();
419 toCheck.pop();
420 Ptr<OctreeNode> node = std::get<0>(top);
421 size_t x_key = std::get<1>(top);
422 size_t y_key = std::get<2>(top);
423 size_t z_key = std::get<3>(top);
424
425 if (node->isLeaf)
426 {
427 outPts.emplace_back(
428 (float) (resolution * x_key) + (float) (resolution * 0.5) + p->origin.x,
429 (float) (resolution * y_key) + (float) (resolution * 0.5) + p->origin.y,
430 (float) (resolution * z_key) + (float) (resolution * 0.5) + p->origin.z);
431 if (p->hasColor)
432 {
433 Point3f avgColor { };
434 for (const auto& c : node->colorList)
435 {
436 avgColor += c;
437 }
438 avgColor *= (1.f/(float)node->colorList.size());
439 outColors.emplace_back(avgColor);
440 }
441 }
442 else
443 {
444 unsigned char x_mask = 1;
445 unsigned char y_mask = 2;
446 unsigned char z_mask = 4;
447 for (unsigned char i = 0; i < 8; i++)
448 {
449 size_t x_copy = x_key;
450 size_t y_copy = y_key;
451 size_t z_copy = z_key;
452 if (!node->children[i].empty())
453 {
454 size_t x_offSet = !!(x_mask & i);
455 size_t y_offSet = !!(y_mask & i);
456 size_t z_offSet = !!(z_mask & i);
457 x_copy = (x_copy << 1) | x_offSet;
458 y_copy = (y_copy << 1) | y_offSet;
459 z_copy = (z_copy << 1) | z_offSet;
460 toCheck.push(stack_element(node->children[i], x_copy, y_copy, z_copy));
461 }
462 }
463 }
464 }
465
466 if (restorePointCloud.needed())
467 {
468 Mat(outPts).copyTo(restorePointCloud);
469 }
470 if (restoreColor.needed())
471 {
472 Mat(outColors).copyTo(restoreColor);
473 }
474}

483bool OctreeNode::overlap(const Point3f& query, float squareRadius) const
484{
485 float halfSize = float(this->size * 0.5);
486 Point3f center = this->origin + Point3f( halfSize, halfSize, halfSize );
487
488 float dist = SquaredDistance(center, query);
489 float temp = float(this->size) * float(this->size) * 3.0f;
490
491 return ( dist + dist * std::numeric_limits<float>::epsilon() ) <= float(temp * 0.25f + squareRadius + sqrt(temp * squareRadius)) ;
492}

500int Octree::radiusNNSearch(const Point3f& query, float radius, OutputArray points, OutputArray colors, OutputArray squareDists) const
501{
502 std::vector<Point3f> outPoints, outColors;
503 std::vector<float> outSqDists;
504
505 if (!p->rootNode.empty())
506 {
507 float squareRadius = radius * radius;
508
509 std::vector<std::tuple<float, Point3f, Point3f>> candidatePoints;
510
511 std::stack<Ptr<OctreeNode>> toCheck;
512 toCheck.push(p->rootNode);
513
514 while (!toCheck.empty())
515 {
516 Ptr<OctreeNode> node = toCheck.top();
517 toCheck.pop();
518 for(size_t i = 0; i < 8; i++)
519 {
520 Ptr<OctreeNode> child = node->children[i];
521 if( child && child->overlap(query, squareRadius))
522 {
523 if(child->isLeaf)
524 {
525 for(size_t j = 0; j < child->pointList.size(); j++)
526 {
527 Point3f pt = child->pointList[j];
528 Point3f col;
529 if (!child->colorList.empty())
530 {
531 col = child->colorList[j];
532 }
533 float dist = SquaredDistance(pt, query);
534 if(dist + dist * std::numeric_limits<float>::epsilon() <= squareRadius)
535 {
536 candidatePoints.emplace_back(dist, pt, col);
537 }
538 }
539 }
540 else
541 {
542 toCheck.push(child);
543 }
544 }
545 }
546 }
547
548 for (size_t i = 0; i < candidatePoints.size(); i++)
549 {
550 auto cp = candidatePoints[i];
551 outSqDists.push_back(std::get<0>(cp));
552 outPoints.push_back(std::get<1>(cp));
553 outColors.push_back(std::get<2>(cp));
554 }
555 }
556
557 if (points.needed())
558 {
559 Mat(outPoints).copyTo(points);
560 }
561 if (colors.needed())
562 {
563 CV_Assert(this->p->hasColor);
564 Mat(outColors).copyTo(colors);
565 }
566 if (squareDists.needed())
567 {
568 Mat(outSqDists).copyTo(squareDists);
569 }
570
571 return int(outPoints.size());
572}

575void OctreeNode::KNNSearchRecurse(const Point3f& query, const int K,
576 float& smallestDist, std::vector<std::tuple<float, Point3f, Point3f>>& candidatePoint) const
577{
578 std::vector<std::pair<float, int>> priorityQue;
579
580 // Add the non-empty OctreeNode to priorityQue
581 for(size_t i = 0; i < 8; i++)
582 {
583 Ptr<OctreeNode> child = this->children[i];
584 if(child)
585 {
586 float halfSize = float(child->size * 0.5);
587
588 Point3f center = child->origin + Point3f(halfSize, halfSize, halfSize);
589
590 float dist = SquaredDistance(query, center);
591 priorityQue.emplace_back(dist, int(i));
592 }
593 }
594
595 std::sort(priorityQue.rbegin(), priorityQue.rend(),
596 [](const std::pair<float, int>& a, const std::pair<float, int>& b) -> bool
597 {
598 return std::get<0>(a) < std::get<0>(b);
599 });
600 Ptr<OctreeNode> child = this->children[std::get<1>(priorityQue.back())];
601
602 while (!priorityQue.empty() && child->overlap(query, smallestDist))
603 {
604 if (!child->isLeaf)
605 {
606 child->KNNSearchRecurse(query, K, smallestDist, candidatePoint);
607 }
608 else
609 {
610 for (size_t i = 0; i < child->pointList.size(); i++)
611 {
612 float dist = SquaredDistance(child->pointList[i], query);
613
614 if ( dist + dist * std::numeric_limits<float>::epsilon() <= smallestDist )
615 {
616 Point3f pt = child->pointList[i];
617 Point3f col { };
618 if (child->colorList.empty())
619 {
620 col = child->colorList[i];
621 }
622 candidatePoint.emplace_back(dist, pt, col);
623 }
624 }
625
626 std::sort(candidatePoint.begin(), candidatePoint.end(),
627 [](const std::tuple<float, Point3f, Point3f>& a, const std::tuple<float, Point3f, Point3f>& b) -> bool
628 {
629 return std::get<0>(a) < std::get<0>(b);
630 }
631 );
632
633 if (int(candidatePoint.size()) > K)
634 {
635 candidatePoint.resize(K);
636 }
637
638 if (int(candidatePoint.size()) == K)
639 {
640 smallestDist = std::get<0>(candidatePoint.back());
641 }
642 }
643
644 priorityQue.pop_back();
645
646 // To next child
647 if(!priorityQue.empty())
648 {
649 child = this->children[std::get<1>(priorityQue.back())];
650 }
651 }
652}
