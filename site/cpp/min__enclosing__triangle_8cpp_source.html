318static void findMinEnclosingTriangle(cv::InputArray points,
319 CV_OUT cv::OutputArray triangle, CV_OUT double &area) {
320 CV_Assert(!points.empty());
321 std::vector<cv::Point2f> resultingTriangle;
322 cv::Mat polygon;
323 convexHull(points, polygon, true, true);
324 findMinEnclosingTriangle(polygon, resultingTriangle, area);
325 cv::Mat(resultingTriangle).copyTo(triangle);
326}

337static void findMinEnclosingTriangle(const std::vector<cv::Point2f> &polygon,
338 std::vector<cv::Point2f> &triangle, double &area) {
339 initialise(triangle, area);
340
341 if (polygon.size() > 3) {
342 findMinimumAreaEnclosingTriangle(polygon, triangle, area);
343 } else {
344 returnMinimumAreaEnclosingTriangle(polygon, triangle, area);
345 }

353static void initialise(std::vector<cv::Point2f> &triangle, double &area) {

357 triangle.clear();

366static void findMinimumAreaEnclosingTriangle(const std::vector<cv::Point2f> &polygon,
367 std::vector<cv::Point2f> &triangle, double &area) {
368 // Algorithm specific variables
369
370 unsigned int validationFlag;
371
372 cv::Point2f vertexA, vertexB, vertexC;
373
374 cv::Point2f sideAStartVertex, sideAEndVertex;
375 cv::Point2f sideBStartVertex, sideBEndVertex;
376 cv::Point2f sideCStartVertex, sideCEndVertex;
377
378 unsigned int a, b, c;
379 unsigned int nrOfPoints;
380
381 // Variables initialisation
382
383 nrOfPoints = static_cast<unsigned int>(polygon.size());
384
385 a = 1;
386 b = 2;
387
388 // Main algorithm steps
389
390 for (c = 0; c < nrOfPoints; c++) {
391 advanceBToRightChain(polygon, nrOfPoints, b, c);
392 moveAIfLowAndBIfHigh(polygon, nrOfPoints, a, b, c);
393 searchForBTangency(polygon, nrOfPoints, a ,b, c);
394
395 updateSidesCA(polygon, nrOfPoints, a, c, sideAStartVertex, sideAEndVertex,
396 sideCStartVertex, sideCEndVertex);
397
398 if (isNotBTangency(polygon, nrOfPoints, a, b, c)) {
399 updateSidesBA(polygon, nrOfPoints, a, b, c, validationFlag, sideAStartVertex,
400 sideAEndVertex, sideBStartVertex, sideBEndVertex,
401 sideCStartVertex, sideCEndVertex);
402 } else {
403 updateSideB(polygon, nrOfPoints, a, b, c, validationFlag,
404 sideBStartVertex, sideBEndVertex);
405 }
406
407 if (isLocalMinimalTriangle(vertexA, vertexB, vertexC, polygon, nrOfPoints, a, b,
408 validationFlag, sideAStartVertex, sideAEndVertex,
409 sideBStartVertex, sideBEndVertex, sideCStartVertex,
410 sideCEndVertex)) {
411 updateMinimumAreaEnclosingTriangle(triangle, area, vertexA, vertexB, vertexC);
412 }
413 }
414}

422static void returnMinimumAreaEnclosingTriangle(const std::vector<cv::Point2f> &polygon,
423 std::vector<cv::Point2f> &triangle, double &area) {
424 unsigned int nrOfPoints = static_cast<unsigned int>(polygon.size());
425
426 for (int i = 0; i < 3; i++) {
427 triangle.push_back(polygon[i % nrOfPoints]);
428 }
429
430 area = areaOfTriangle(triangle[0], triangle[1], triangle[2]);
431}

442static void advanceBToRightChain(const std::vector<cv::Point2f> &polygon,
443 unsigned int nrOfPoints, unsigned int &b,
444 unsigned int c) {
445 while (greaterOrEqual(height(successor(b, nrOfPoints), polygon, nrOfPoints, c),
446 height(b, polygon, nrOfPoints, c))) {
447 advance(b, nrOfPoints);
448 }
449}

461static void moveAIfLowAndBIfHigh(const std::vector<cv::Point2f> &polygon,
462 unsigned int nrOfPoints, unsigned int &a, unsigned int &b,
463 unsigned int c) {
464 cv::Point2f gammaOfA;
465
466 while(height(b, polygon, nrOfPoints, c) > height(a, polygon, nrOfPoints, c)) {
467 if ((gamma(a, gammaOfA, polygon, nrOfPoints, a, c)) && (intersectsBelow(gammaOfA, b, polygon, nrOfPoints, c))) {
468 advance(b, nrOfPoints);
469 } else {
470 advance(a, nrOfPoints);
471 }
472 }
473}

485static void searchForBTangency(const std::vector<cv::Point2f> &polygon,
486 unsigned int nrOfPoints, unsigned int a, unsigned int &b,
487 unsigned int c) {
488 cv::Point2f gammaOfB;
489
490 while (((gamma(b, gammaOfB, polygon, nrOfPoints, a, c)) &&
491 (intersectsBelow(gammaOfB, b, polygon, nrOfPoints, c))) &&
492 (greaterOrEqual(height(b, polygon, nrOfPoints, c),
493 height(predecessor(a, nrOfPoints), polygon, nrOfPoints, c)))
494 ) {
495 advance(b, nrOfPoints);
496 }
497}

509static bool isNotBTangency(const std::vector<cv::Point2f> &polygon,
510 unsigned int nrOfPoints, unsigned int a, unsigned int b,
511 unsigned int c) {
512 cv::Point2f gammaOfB;
513
514 if (((gamma(b, gammaOfB, polygon, nrOfPoints, a, c)) &&
515 (intersectsAbove(gammaOfB, b, polygon, nrOfPoints, c))) ||
516 (height(b, polygon, nrOfPoints, c) < height(predecessor(a, nrOfPoints), polygon, nrOfPoints, c))) {
517 return true;
518 }
519
520 return false;
521}

538 unsigned int nrOfPoints, unsigned int a, unsigned int c,

565static void updateSidesBA(const std::vector<cv::Point2f> &polygon,
566 unsigned int nrOfPoints, unsigned int a, unsigned int b,
567 unsigned int c, unsigned int &validationFlag,
568 cv::Point2f &sideAStartVertex, cv::Point2f &sideAEndVertex,
569 cv::Point2f &sideBStartVertex, cv::Point2f &sideBEndVertex,
570 const cv::Point2f &sideCStartVertex, const cv::Point2f &sideCEndVertex) {
571 // Side B is flush with edge [b, b-1]
572 sideBStartVertex = polygon[predecessor(b, nrOfPoints)];
573 sideBEndVertex = polygon[b];
574
575 // Find middle point of side B
576 cv::Point2f sideBMiddlePoint;
577
578 if ((middlePointOfSideB(sideBMiddlePoint, sideAStartVertex, sideAEndVertex, sideBStartVertex,
579 sideBEndVertex, sideCStartVertex, sideCEndVertex)) &&
580 (height(sideBMiddlePoint, polygon, nrOfPoints, c) <
581 height(predecessor(a, nrOfPoints), polygon, nrOfPoints, c))) {
582 sideAStartVertex = polygon[predecessor(a, nrOfPoints)];
583 sideAEndVertex = findVertexCOnSideB(polygon, nrOfPoints, a, c,
584 sideBStartVertex, sideBEndVertex,
585 sideCStartVertex, sideCEndVertex);
586
587 validationFlag = VALIDATION_SIDE_A_TANGENT;
588 } else {
589 validationFlag = VALIDATION_SIDES_FLUSH;
590 }
591}

606static void updateSideB(const std::vector<cv::Point2f> &polygon,
607 unsigned int nrOfPoints, unsigned int a, unsigned int b,
608 unsigned int c, unsigned int &validationFlag,
609 cv::Point2f &sideBStartVertex, cv::Point2f &sideBEndVertex) {
610 if (!gamma(b, sideBStartVertex, polygon, nrOfPoints, a, c)) {
611 CV_Error(cv::Error::StsInternal, ERR_SIDE_B_GAMMA);
612 }
613
614 sideBEndVertex = polygon[b];
615
616 validationFlag = VALIDATION_SIDE_B_TANGENT;
617}

638static bool isLocalMinimalTriangle(cv::Point2f &vertexA, cv::Point2f &vertexB,
639 cv::Point2f &vertexC, const std::vector<cv::Point2f> &polygon,
640 unsigned int nrOfPoints, unsigned int a, unsigned int b,
641 unsigned int validationFlag, const cv::Point2f &sideAStartVertex,
642 const cv::Point2f &sideAEndVertex, const cv::Point2f &sideBStartVertex,
643 const cv::Point2f &sideBEndVertex, const cv::Point2f &sideCStartVertex,
644 const cv::Point2f &sideCEndVertex) {
645 if ((!lineIntersection(sideAStartVertex, sideAEndVertex,
646 sideBStartVertex, sideBEndVertex, vertexC)) ||
647 (!lineIntersection(sideAStartVertex, sideAEndVertex,
648 sideCStartVertex, sideCEndVertex, vertexB)) ||
649 (!lineIntersection(sideBStartVertex, sideBEndVertex,
650 sideCStartVertex, sideCEndVertex, vertexA))) {
651 return false;
652 }
653
654 return isValidMinimalTriangle(vertexA, vertexB, vertexC, polygon, nrOfPoints, a, b,
655 validationFlag, sideAStartVertex, sideAEndVertex,
656 sideBStartVertex, sideBEndVertex, sideCStartVertex,
657 sideCEndVertex);
658}

681static bool isValidMinimalTriangle(const cv::Point2f &vertexA, const cv::Point2f &vertexB,
682 const cv::Point2f &vertexC, const std::vector<cv::Point2f> &polygon,
683 unsigned int nrOfPoints, unsigned int a, unsigned int b,
684 unsigned int validationFlag, const cv::Point2f &sideAStartVertex,
685 const cv::Point2f &sideAEndVertex, const cv::Point2f &sideBStartVertex,
686 const cv::Point2f &sideBEndVertex, const cv::Point2f &sideCStartVertex,
687 const cv::Point2f &sideCEndVertex) {
688 cv::Point2f midpointSideA = middlePoint(vertexB, vertexC);
689 cv::Point2f midpointSideB = middlePoint(vertexA, vertexC);
690 cv::Point2f midpointSideC = middlePoint(vertexA, vertexB);
691
692 bool sideAValid = (validationFlag == VALIDATION_SIDE_A_TANGENT)
693 ? (areEqualPoints(midpointSideA, polygon[predecessor(a, nrOfPoints)]))
694 : (isPointOnLineSegment(midpointSideA, sideAStartVertex, sideAEndVertex));
695
696 bool sideBValid = (validationFlag == VALIDATION_SIDE_B_TANGENT)
697 ? (areEqualPoints(midpointSideB, polygon[b]))
698 : (isPointOnLineSegment(midpointSideB, sideBStartVertex, sideBEndVertex));
699
700 bool sideCValid = (validationFlag == VALIDATION_SIDES_FLUSH) || isPointOnLineSegment(midpointSideC, sideCStartVertex, sideCEndVertex);
701
702 return (sideAValid && sideBValid && sideCValid);
703}

713static void updateMinimumAreaEnclosingTriangle(std::vector<cv::Point2f> &triangle, double &area,
714 const cv::Point2f &vertexA, const cv::Point2f &vertexB,
715 const cv::Point2f &vertexC) {
716 double triangleArea = areaOfTriangle(vertexA, vertexB, vertexC);
717
718 if (triangleArea < area) {
719 triangle.clear();
720
721 triangle.push_back(vertexA);
722 triangle.push_back(vertexB);
723 triangle.push_back(vertexC);
724
725 area = triangleArea;
726 }
727}

739static bool middlePointOfSideB(cv::Point2f &middlePointOfSideB, const cv::Point2f &sideAStartVertex,
740 const cv::Point2f &sideAEndVertex, const cv::Point2f &sideBStartVertex,
741 const cv::Point2f &sideBEndVertex, const cv::Point2f &sideCStartVertex,
742 const cv::Point2f &sideCEndVertex) {
743 cv::Point2f vertexA, vertexC;

747 return false;
748 }
749
750 middlePointOfSideB = middlePoint(vertexA, vertexC);
751
752 return true;

802 const std::vector<cv::Point2f> &polygon, unsigned int nrOfPoints,
803 unsigned int c) {

813 (almostEqual(angleGammaAndPoint, anglePointPredecessor))) {
814 return intersectsAboveOrBelow(predecessor(polygonPointIndex, nrOfPoints),
815 polygonPointIndex, polygon, nrOfPoints, c);

817 (almostEqual(angleGammaAndPoint, anglePointSuccessor))) {
818 return intersectsAboveOrBelow(successor(polygonPointIndex, nrOfPoints),
819 polygonPointIndex, polygon, nrOfPoints, c);

823 (isGammaAngleBtw(angleGammaAndPoint, anglePointPredecessor, anglePointSuccessor)) ||

826 (!isGammaAngleEqualTo(angleGammaAndPoint, angleFlushEdge))

830 (!isGammaAngleEqualTo(angleGammaAndPoint, angleFlushEdge))
