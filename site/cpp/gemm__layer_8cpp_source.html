99 LayerGemmOpMode getOpMode(size_t n_inputs, size_t n_blobs) const {
100 if (n_blobs == 0) return LayerGemmOpMode::noblob;
101 if (n_inputs == 3) return LayerGemmOpMode::noblob ; // if all inputs are given, then no blobs are used
102 if (n_inputs == 2) {
103 if (have_bias) {
104 // check where the input comes from
105 if(const_B)
106 return LayerGemmOpMode::blobB;
107 if(const_C)
108 return LayerGemmOpMode::blobC;
109 return LayerGemmOpMode::blobC;
110 } else {
111 if (n_blobs == 1)
112 // 2 inputs, no bias => input[1] is B
113 return LayerGemmOpMode::noblob;
114 if (n_blobs == 2)
115 return LayerGemmOpMode::blobC;
116 }
117 }
118 if (n_inputs == 1) {
119 // only A is given per input
120 if (n_blobs == 2)
121 return LayerGemmOpMode::blobBC;
122 else
123 return LayerGemmOpMode::blobB;
124 }
125 CV_Error(Error::StsError, "DNN/Gemm: could not derive OP mode");
126 }

129 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
130 const int requiredOutputs,
131 std::vector<MatShape> &outputs,
132 std::vector<MatShape> &internals) const CV_OVERRIDE {
133 int num_inputs = static_cast<int>(inputs.size() + blobs.size());
134
135 CV_CheckGE(num_inputs, 2, "DNN/Gemm: Gemm takes at least two inputs");
136 CV_CheckLE(num_inputs, 3, "DNN/Gemm: Gemm takes at most three inputs");
137
138 LayerGemmOpMode mode = getOpMode(inputs.size(), blobs.size());
139
140 // Check whether A and B are two dimensional
141 const auto shape_A = inputs[0];
142 const auto shape_B = constB(mode) ? shape(blobs[0]) : inputs[1];
143 CV_CheckGE(shape_A.size(), static_cast<size_t>(2), "DNN/Gemm: Tensor A must be n-dimensional (n >= 2)");
144 CV_CheckEQ(shape_B.size(), static_cast<size_t>(2), "DNN/Gemm: Tensor B must be two dimensional");
145
146 // Check legal matrix multiplication
147 size_t dims_A = shape_A.size();
148 int ma = shape_A[dims_A - 2], na = shape_A[dims_A - 1];
149 int mb = shape_B[0], nb = shape_B[1];
150 int M = trans_a ? na : ma;
151 int N = trans_b ? mb : nb;
152 int K_a = trans_a ? ma : na;
153 int K_b = trans_b ? nb : mb;
154
155
156 CV_CheckEQ(K_a, K_b, "DNN/Gemm: Invalid dimension of dim K");
157
158 // Check whether C can be unidirectional broadcast to (M, N). Handle carefully with 1D Mat.
159 if (have_bias) {
160 const auto shape_C = constC(mode) ? shape(blobs.back()) : inputs.back();
161
162 auto ndims_C = shape_C.size();
163 CV_CheckLE(ndims_C, static_cast<size_t>(2), "DNN/Gemm: C can only be 0d (scalar) / 1d / 2d tensor");
164
165 int real_ndims_C_ = real_ndims_C >= 0 ? real_ndims_C : ndims_C;
166
167 if (real_ndims_C_ == 1) { // (1,) or (N,)
168 CV_Check(shape_C[0], shape_C[0] == 1 || shape_C[0] == N, "DNN/Gemm: invalid dimension of C");
169 } else if (real_ndims_C_ == 2) { // (1, 1) or (1, N) or (M, 1) or (M, N)
170 // printf("shape_C=[%d, %d]\n", shape_C[0], shape_C[1]);
171 CV_Check(shape_C[0], (shape_C[0] == 1 && shape_C[1] == 1) ||
172 (shape_C[0] == 1 && shape_C[1] == N) ||
173 (shape_C[0] == M && shape_C[1] == 1) ||
174 (shape_C[0] == M && shape_C[1] == N),
175 "DNN/Gemm: C must be of shape (1, 1) or (1, N) or (M, 1) or (M, N)");
176 if (shape_C[0] == 1) {
177 CV_Check(shape_C[1], shape_C[1] == 1 || shape_C[1] == N, "DNN/Gemm: invalid dimension of C");
178 } else if (shape_C[0] == M) {
179 CV_Check(shape_C[1], shape_C[1] == 1 || shape_C[1] == N, "DNN/Gemm: invalid dimension of C");
180 } else {
181 CV_Error(Error::StsBadSize, "DNN/Gemm: invalid dimension of C");
182 }
183 }
184 }
185
186 if (flatten_a) {
187 int batches = std::accumulate(shape_A.begin(), shape_A.end() - 2, 1, std::multiplies<int>());
188 MatShape shape_y{M * batches, N};
189 outputs.assign(1, shape_y);
190 } else {
191 // Preserve A's leading dims; only the trailing axis changes from K to N.
192 // (trans_a is rejected upstream for this mode, so M corresponds to
193 // shape_A[-2] and we just rewrite the last axis.)
194 CV_CheckFalse(trans_a, "DNN/Gemm: flatten_a=false requires trans_a=false");
195 MatShape shape_y = shape_A;
196 shape_y[shape_y.size() - 1] = N;
197 outputs.assign(1, shape_y);
198 }
199 return false;
200 }

202 virtual int64 getFLOPS(const std::vector<MatShape> &inputs,
203 const std::vector<MatShape> &outputs) const CV_OVERRIDE
204 {
205 CV_Assert(!inputs.empty());
206 LayerGemmOpMode mode_ = getOpMode(inputs.size(), blobs.size());
207 const auto shape_A = inputs[0];
208 const auto shape_B = constB(mode_) ? shape(blobs[0]) : inputs[1];
209 int M = trans_a ? shape_A.back() : shape_A[shape_A.size() - 2];
210 int K = trans_a ? shape_A[shape_A.size() - 2] : shape_A.back();
211 int N = trans_b ? shape_B[shape_B.size() - 2] : shape_B.back();
212
213 int64 batches = std::accumulate(shape_A.begin(), shape_A.end() - 2,
214 CV_BIG_INT(1), std::multiplies<int64>());
215
216 // 2*M*N*K multiply-adds, +M*N for bias
217 int64 flops = batches * (CV_BIG_INT(2) * M * N * K);
218 if (have_bias)
219 flops += batches * M * N;
220 return flops;
221 }

223 // TODO: replace with cv::broadcast() once 1d mat is supported
224 // FIXME: fix if conditions if 1d mat is supported properly
225 void broadcastCWtihBeta(int M, int N, const Mat &C) {
226 broadcast_C.clear();
227 broadcast_C.resize(M * N, 0.f);
228 if (beta != 0 && !C.empty()) {
229 int real_ndims_C_ = real_ndims_C >= 0 ? real_ndims_C : C.dims;
230
231 const float *ptr_c = C.ptr<const float>();
232 const auto shape_C = shape(C);
233 if ((real_ndims_C_ == 0) || (real_ndims_C_ == 1 && shape_C[0] == 1) ||
234 (real_ndims_C_ == 2 && shape_C[0] == 1 && shape_C[1] == 1)) {
235 // (), (1,), (1, 1)
236 float c = *ptr_c;
237 int total = M * N;
238 for (int i = 0; i < total; ++i) {
239 broadcast_C[i] = beta * c;
240 }
241 } else if ((real_ndims_C_ == 1 && shape_C[0] == N) ||
242 (real_ndims_C_ == 2 && shape_C[0] == 1 && shape_C[1] == N)) {
243 // (N,), (1, N)
244 for (int i = 0; i < M; ++i) {
245 int step = i * N;
246 for (int j = 0; j < N; ++j) {
247 broadcast_C[step + j] = beta * ptr_c[j];
248 }
249 }
250 } else if (real_ndims_C_ == 2 && shape_C[0] == M && shape_C[1] == 1) {
251 // (M, 1)
252 for (int i = 0; i < M; ++i) {
253 int step = i * N;
254 for (int j = 0; j < N; ++j) {
255 broadcast_C[step + j] = beta * ptr_c[i];
256 }
257 }
258 } else {
259 // (M, N)
260 std::transform(ptr_c, ptr_c + M * N, broadcast_C.begin(), [this] (const float &c) {
261 return this->beta * c; });
262 }
263 }
264 }

268 virtual void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr) CV_OVERRIDE {
269 opt.init();
270 std::vector<Mat> inputs;
271 inputs_arr.getMatVector(inputs);
272
273 // CV_64F skips the float-only packed-B/MLAS caching below; see forwardDouble().
274 if (inputs[0].depth() == CV_64F)
275 return;
276
277 LayerGemmOpMode mode = getOpMode(inputs.size(), blobs.size());
278
279 // pack B if it is const
280 if (constB(mode) && blobs[0].data != last_packed_blob_data) {
281 packed_B.clear();
282 packed_B.shrink_to_fit();
283 thin_packed_B.clear();
284 thin_packed_B.shrink_to_fit();
285
286 bool mlas_packed = false;
287#ifdef HAVE_MLAS
288 packed_B_mlas.release();
289 packed_B_mlas_M = packed_B_mlas_N = packed_B_mlas_K = 0;
290 if (mlasAvailable() && !opt.use_rvv) {
291 std::vector<Mat> outputs;
292 outputs_arr.getMatVector(outputs);
293 const auto shape_A = shape(inputs[0]);
294 const auto shape_Y = shape(outputs[0]);
295 const int na = shape_A[shape_A.size() - 1];
296 const int ma = shape_A[shape_A.size() - 2];
297 const int N = shape_Y[shape_Y.size() - 1];
298 const int M = shape_Y[shape_Y.size() - 2];
299 const int K = trans_a ? ma : na;
300 const Mat& Bmat = blobs[0];
301 const int ldb = Bmat.size[Bmat.dims - 1];
302 const size_t packed_bytes = mlasSgemmPackBSize(trans_a, trans_b, N, K);
303 if (packed_bytes > 0 && packed_bytes <= static_cast<size_t>(INT_MAX)) {
304 packed_B_mlas.create(1, static_cast<int>(packed_bytes), CV_8U);
305 if (mlasSgemmPackB(trans_a, trans_b, N, K,
306 Bmat.ptr<const float>(), ldb,
307 packed_B_mlas.data)) {
308 packed_B_mlas_M = M;
309 packed_B_mlas_N = N;
310 packed_B_mlas_K = K;
311 mlas_packed = true;
312 } else {
313 packed_B_mlas.release();
314 }
315 }
316 }
317#endif
318 if (!mlas_packed) {
319 fastGemmPackB(blobs[0], packed_B, trans_b, opt);
320
321 if (!trans_a && blobs[0].type() == CV_32F) {
322 std::vector<Mat> outputs;
323 outputs_arr.getMatVector(outputs);
324 if (!outputs.empty()) {
325 const auto &Y = outputs[0];
326 const auto shape_Y = shape(Y);
327 const int N = shape_Y.back();
328 const int K = trans_b ? blobs[0].size[1] : blobs[0].size[0];
329 const int rows_thin = flatten_a ? shape_Y[shape_Y.size() - 2]
330 : (int)(Y.total() / (size_t)N);
331 if (fastGemmThinEligible(rows_thin, N, K)) {
332 thin_packed_B.resize(fastGemmThinPackBSize(N, K));
333 const size_t ldb_K = trans_b ? 1 : N;
334 const size_t ldb_N = trans_b ? K : 1;
335 fastGemmThinPackB(N, K, blobs[0].ptr<const float>(),
336 ldb_K, ldb_N, thin_packed_B.data());
337 }
338 }
339 }
340 }
341
342 last_packed_blob_data = blobs[0].data;
343 }
344
345 if (constC(mode) && flatten_a) {
346 const auto &C = blobs.back();
347
348 std::vector<Mat> outputs;
349 outputs_arr.getMatVector(outputs);
350 const auto &Y = outputs[0];
351 const auto shape_Y = shape(Y);
352 size_t dims_Y = shape_Y.size();
353 int M = shape_Y[dims_Y - 2], N = shape_Y[dims_Y - 1];
354
355 // broadcast
356 broadcastCWtihBeta(M, N, C);
357 }
358 }

360 // Y = A * B + C, note that C is unidirectionaly broadcastable to (A * B).
361 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE {
362 CV_TRACE_FUNCTION();
363 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
364
365 if (inputs_arr.depth() == CV_16F)
366 {
367 forward_fallback(inputs_arr, outputs_arr, internals_arr);
368 return;
369 }
370
371 std::vector<Mat> inputs, outputs;
372 inputs_arr.getMatVector(inputs);
373 outputs_arr.getMatVector(outputs);
374
375 LayerGemmOpMode mode = getOpMode(inputs.size(), blobs.size());
376
377 if (inputs[0].depth() == CV_64F) {
378 forwardDouble(inputs, outputs, mode);
379 return;
380 }
381
382 const auto &A = inputs[0];
383 auto &Y = outputs[0];
384
385 const auto shape_A = shape(A), shape_Y = shape(Y);
386 size_t dims_A = shape_A.size();
387 int ma = shape_A[dims_A - 2], na = shape_A[dims_A - 1];
388 size_t dims_Y = shape_Y.size();
389 int M = shape_Y[dims_Y - 2], N = shape_Y[dims_Y - 1];
390 int K = trans_a ? ma : na;
391
392 // In flatten_a=false mode the output keeps A's leading dims, so the
393 // GEMM row count spans those dims as well: rows = total(Y)/N.
394 const int rows = flatten_a ? M : (int)(Y.total() / (size_t)N);
395
396 // broadcast C and copy C to output
397 if (constC(mode) || inputs.size() >= 3) {
398 float *ptr_y = Y.ptr<float>();
399 if (flatten_a) {
400 if (!constC(mode) || broadcast_C.empty()) {
401 broadcastCWtihBeta(M, N, (inputs.size() >= 3 ? inputs.back() : blobs.back()));
402 }
403 int step = M * N;
404 CV_CheckEQ(broadcast_C.size(), static_cast<size_t>(step), "DNN/Gemm: C is not broadcast properly");
405 const float* src = broadcast_C.data();
406 constexpr size_t PAR_THRESHOLD = 1u << 14;
407 if ((size_t)step < PAR_THRESHOLD) {
408 std::memcpy(ptr_y, src, (size_t)step * sizeof(float));
409 } else {
410 int nthreads = std::max(1, getNumThreads());
411 parallel_for_(Range(0, nthreads), [=](const Range& r) {
412 size_t st = (size_t)r.start * (size_t)step / (size_t)nthreads;
413 size_t en = (size_t)r.end * (size_t)step / (size_t)nthreads;
414 std::memcpy(ptr_y + st, src + st, (en - st) * sizeof(float));
415 }, nthreads);
416 }
417 } else {
418 // ND output: tile the (1D / scalar) bias across all `rows`
419 // rows, scaled by beta. The rewriter restricts bias to scalar
420 // or 1D length-N; assert here.
421 const Mat& C = (inputs.size() >= 3) ? inputs.back() : blobs.back();
422 const float* c = C.ptr<const float>();
423 if (C.total() == 1) {
424 float val = beta * (*c);
425 size_t total = (size_t)rows * (size_t)N;
426 constexpr size_t PAR_THRESHOLD = 1u << 14;
427 if (total < PAR_THRESHOLD) {
428 std::fill_n(ptr_y, total, val);
429 } else {
430 int nthreads = std::max(1, getNumThreads());
431 parallel_for_(Range(0, nthreads), [=](const Range& r) {
432 size_t st = (size_t)r.start * total / (size_t)nthreads;
433 size_t en = (size_t)r.end * total / (size_t)nthreads;
434 std::fill_n(ptr_y + st, en - st, val);
435 }, nthreads);
436 }
437 } else {
438 CV_CheckEQ((int)C.total(), N, "DNN/Gemm: bias must be scalar or length-N in flatten_a=false mode");
439 // Compute first row, then broadcast.
440 for (int j = 0; j < N; j++) ptr_y[j] = beta * c[j];
441 if (rows > 1) {
442 size_t row_bytes = (size_t)N * sizeof(float);
443 constexpr int PAR_THRESHOLD_ROWS = 32;
444 if (rows <= PAR_THRESHOLD_ROWS) {
445 for (int i = 1; i < rows; i++) {
446 std::memcpy(ptr_y + (size_t)i * N, ptr_y, row_bytes);
447 }
448 } else {
449 int nthreads = std::max(1, getNumThreads());
450 double nstripes = std::min((double)(rows - 1), (double)nthreads);
451 parallel_for_(Range(1, rows), [=](const Range& r) {
452 for (int i = r.start; i < r.end; i++) {
453 std::memcpy(ptr_y + (size_t)i * N, ptr_y, row_bytes);
454 }
455 }, nstripes);
456 }
457 }
458 }
459 }
460 } else { // initialization
461 float *ptr_y = Y.ptr<float>();
462 size_t total = Y.total();
463 constexpr size_t PAR_THRESHOLD = 1u << 14;
464 if (total < PAR_THRESHOLD) {
465 std::memset(ptr_y, 0, total * sizeof(float));
466 } else {
467 int nthreads = std::max(1, getNumThreads());
468 parallel_for_(Range(0, nthreads), [=](const Range& r) {
469 size_t st = (size_t)r.start * total / (size_t)nthreads;
470 size_t en = (size_t)r.end * total / (size_t)nthreads;
471 std::memset(ptr_y + st, 0, (en - st) * sizeof(float));
472 }, nthreads);
473 }
474 }
475
476 if (constB(mode)) {
477#ifdef HAVE_MLAS
478 if (!opt.use_rvv && !packed_B_mlas.empty() &&
479 packed_B_mlas_N == N && packed_B_mlas_K == K)
480 {
481 if (mlasSgemmPacked(trans_a, trans_b, rows, N, K,
482 alpha,
483 A.ptr<const float>(), na,
484 packed_B_mlas.data,
485 1.f,
486 Y.ptr<float>(), N)) {
487 return;
488 }
489 }
490#endif
491 CV_CheckGT(packed_B.size(), static_cast<size_t>(0), "DNN/Gemm: constant B is not pre-packed");
492 if (!thin_packed_B.empty()) {
493 fastGemmThin(rows, N, K, alpha, A.ptr<const float>(), na, 1,
494 thin_packed_B.data(), 1.f, Y.ptr<float>(), N, opt.multi_thread);
495 } else {
496 fastGemm(trans_a, rows, N, K, alpha, A.ptr<const float>(), na, packed_B.data(), 1.f, Y.ptr<float>(), N, opt);
497 }
498 } else {
499 fastGemmBatch(trans_a, trans_b, alpha, A, inputs[1], 1.f, Y, opt);
500 }
501 }

503 // Double-precision analogue of broadcastCWtihBeta() above; not cached (see finalize()).
504 static void fillBroadcastCDouble(int M, int N, const Mat& C, double beta, Mat& dst)
505 {
506 CV_Assert(dst.rows == M && dst.cols == N);
507 double* out = dst.ptr<double>();
508 size_t total = (size_t)M * (size_t)N;
509
510 if (beta == 0 || C.empty()) {
511 std::fill_n(out, total, 0.0);
512 return;
513 }
514
515 const double* c = C.ptr<const double>();
516 const auto shape_C = shape(C);
517 int ndims_C = (int)shape_C.size();
518
519 if (ndims_C == 0 || (ndims_C == 1 && shape_C[0] == 1) ||
520 (ndims_C == 2 && shape_C[0] == 1 && shape_C[1] == 1)) {
521 // (), (1,), (1, 1): single scalar broadcast to every element.
522 std::fill_n(out, total, beta * c[0]);
523 } else if ((ndims_C == 1 && shape_C[0] == N) ||
524 (ndims_C == 2 && shape_C[0] == 1 && shape_C[1] == N)) {
525 // (N,), (1, N): one row broadcast down every row.
526 for (int i = 0; i < M; i++)
527 for (int j = 0; j < N; j++)
528 out[(size_t)i * N + j] = beta * c[j];
529 } else if (ndims_C == 2 && shape_C[0] == M && shape_C[1] == 1) {
530 // (M, 1): one value per row broadcast across every column.
531 for (int i = 0; i < M; i++)
532 std::fill_n(out + (size_t)i * N, N, beta * c[i]);
533 } else {
534 // (M, N): no broadcast, just scale.
535 CV_CheckEQ(shape_C[0], M, "DNN/Gemm: C is not broadcast properly");
536 CV_CheckEQ(shape_C[1], N, "DNN/Gemm: C is not broadcast properly");
537 for (size_t i = 0; i < total; i++)
538 out[i] = beta * c[i];
539 }
540 }

542 // CV_64F via cv::gemm, not the float-only fastGemm/MLAS kernels above.
543 void forwardDouble(const std::vector<Mat>& inputs, std::vector<Mat>& outputs, LayerGemmOpMode mode)
544 {
545 const Mat &A = inputs[0];
546 Mat &Y = outputs[0];
547 const Mat &B = constB(mode) ? blobs[0] : inputs[1];
548
549 CV_CheckTypeEQ(B.depth(), CV_64F, "DNN/Gemm: B must be CV_64F to match A");
550 CV_Assert(A.isContinuous() && Y.isContinuous());
551
552 const auto shape_A = shape(A), shape_Y = shape(Y);
553 size_t dims_A = shape_A.size();
554 int na = shape_A[dims_A - 1];
555 size_t dims_Y = shape_Y.size();
556 int N = shape_Y[dims_Y - 1];
557 const int rows = (int)(Y.total() / (size_t)N);
558
559 // 2D views onto existing data, no copy; trans_a/trans_b via cv::gemm's own flags.
560 Mat Aview = A.reshape(1, (int)(A.total() / (size_t)na));
561 Mat Yview = Y.reshape(1, rows);
562 const int flags = (trans_a ? GEMM_1_T : 0) | (trans_b ? GEMM_2_T : 0);
563
564 const bool haveC = constC(mode) || inputs.size() >= 3;
565 if (haveC) {
566 const Mat& C = (inputs.size() >= 3) ? inputs.back() : blobs.back();
567 CV_CheckTypeEQ(C.depth(), CV_64F, "DNN/Gemm: C must be CV_64F to match A");
568 fillBroadcastCDouble(rows, N, C, (double)beta, Yview);
569 cv::gemm(Aview, B, (double)alpha, Yview, 1.0, Yview, flags);
570 } else {
571 cv::gemm(Aview, B, (double)alpha, Mat(), 0.0, Yview, flags);
572 }
573 }

648 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> >& inputs,
649 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
650 {
651 ov::Output<ov::Node> nodeA = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
652 ov::Output<ov::Node> nodeB;
653 if (const_B)
654 nodeB = std::make_shared<ov::op::v0::Constant>(ov::element::f32, getShape(blobs[0]), blobs[0].data);

656 nodeB = nodes[1].dynamicCast<InfEngineNgraphNode>()->node;

659 if (flatten_axis > 0) {
660 std::vector<int> shape(1 + flatten_axis, 0);
661 shape[shape.size() - 1] = -1;
662 nodeA = std::make_shared<ov::op::v1::Reshape>(
663 nodeA,
664 std::make_shared<ov::op::v0::Constant>(ov::element::i32, ov::Shape{shape.size()}, shape.data()),
665 true);
666 }
667
668 std::shared_ptr<ov::Node> nodeAB = std::make_shared<ov::op::v0::MatMul>(nodeA, nodeB, trans_a, trans_b);
669 if (alpha != 1.0f)
670 {
671 nodeAB = std::make_shared<ov::op::v1::Multiply>(
672 nodeAB,
673 std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{1}, &alpha));
674 }
675
676 if (!have_bias)
677 return Ptr<BackendNode>(new InfEngineNgraphNode(nodeAB));
678
679 ov::Output<ov::Node> nodeC;
680 if (const_C)
681 {
682 auto shape_C = blobs.back().total() == blobs.back().size[0] ? ov::Shape{blobs.back().total()} : getShape(blobs.back());
683 nodeC = std::make_shared<ov::op::v0::Constant>(ov::element::f32, shape_C, blobs.back().data);
684 }
685 else
686 {
687 nodeC = nodes.back().dynamicCast<InfEngineNgraphNode>()->node;
688 }
689
690 if (beta != 1.0f)
691 {
692 nodeC = std::make_shared<ov::op::v1::Multiply>(
693 nodeC,
694 std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{1}, &beta));
695 }
696
697 auto nodeGemm = std::make_shared<ov::op::v1::Add>(nodeAB, nodeC, ov::op::AutoBroadcastType::NUMPY);
698 return Ptr<BackendNode>(new InfEngineNgraphNode(nodeGemm));
699 }
