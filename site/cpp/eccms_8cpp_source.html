114template<> struct MotionTraits<MOTION_HOMOGRAPHY> {
115 enum { paramAmount = 8};
116 static inline void tailHandlerGetCoord(float& sx,
117 float& sy,
118 float& denominator,
119 int col,
120 float numeratorX0,
121 float numeratorY0,
122 float denominator0,
123 float a00,
124 float a10,
125 float a20)

132 template<typename elemtype>
133 static constexpr std::array<float, paramAmount> fillJacobian(int col, int row, float sx, float sy, float fVal,
134 elemtype gx, elemtype gy, float/*a00*/, float/*a10*/,
135 float denominator) {
136#define GX (fVal * float(gx) * denominator)
137#define GY (fVal * float(gy) * denominator)
138#define GZ (-(GX * sx + GY * sy))
139 return std::array<float, paramAmount>{GX * col, GY * col, GZ * col, GX * row, GY * row, GZ * row, GX, GY};
140#undef GX
141#undef GY
142#undef GZ
143 }

215template<int motionType, typename elemtype>
216static double imageHessianProjECC(const Mat& map,
217 const Mat& sampleWithGrad,
218 const Mat& ref,
219 double& sampSum,
220 double& sampSqSum,
221 double& refSum,
222 double& refSqSum,
223 int& nz,
224 Mat& hessian,
225 Mat& sampleProj,
226 Mat& refProj,
227 int deltaY,
228 int interpolation) {
229 static_assert(std::is_same<float, elemtype>::value, "imageHessianProjECC: f16 is not supported yet");
230#define HESSIAN_PARAMS (MotionTraits<motionType>::paramAmount)
231
232 CV_Assert(map.type() == CV_64F);
233 CV_Assert(interpolation == INTER_NEAREST || interpolation == INTER_LINEAR);
234 CV_Assert(hessian.type() == CV_64F && sampleProj.type() == CV_64F && refProj.type() == CV_64F);
235 if (sampleProj.size() != Size(1, HESSIAN_PARAMS) || refProj.size() != Size(1, HESSIAN_PARAMS)) {
236 CV_Error(Error::BadImageSize, format("imageHessianProjECC: Wrong sample projection/reference projection size. 1x%d expected", HESSIAN_PARAMS));
237 }
238 if (hessian.size() != Size(HESSIAN_PARAMS, HESSIAN_PARAMS)) {
239 CV_Error(Error::BadImageSize, format("imageHessianProjECC: Wrong hessian size. %dx%d expected", HESSIAN_PARAMS, HESSIAN_PARAMS));
240 }
241 if (!map.isContinuous()) {
242 CV_Error(Error::BadStep, "imageHessianProjECC: Map should be continuous");
243 }
244 if (std::is_same<float, elemtype>::value) {
245 CV_Assert(sampleWithGrad.type() == CV_32FC4 && ref.type() == CV_32FC2);
246 }
247
248 int hr = ref.rows;
249 int wr = ref.cols;
250 int hs = sampleWithGrad.rows;
251 int ws = sampleWithGrad.cols;
252 unsigned int ycond = hs - (INTER_LINEAR ? 1 : 0);
253 unsigned int xcond = ws - (INTER_LINEAR ? 1 : 0);
254
255 hessian = Mat::zeros(hessian.size(), hessian.type());
256 sampleProj = Mat::zeros(sampleProj.size(), sampleProj.type());
257 refProj = Mat::zeros(refProj.size(), refProj.type());
258
259 const int MAX_STRIPES = 128;
260 int stripesAmount = std::min(MAX_STRIPES, hr / deltaY);
261 std::vector<Matx<double, MotionTraits<motionType>::paramAmount, MotionTraits<motionType>::paramAmount> > hessPs(stripesAmount);
262 std::vector<Vec<double, MotionTraits<motionType>::paramAmount> > iprojs(stripesAmount);
263 std::vector<Vec<double, MotionTraits<motionType>::paramAmount> > tprojs(stripesAmount);
264 std::vector<Vec<double, MotionTraits<motionType>::paramAmount> > projSubs(stripesAmount);
265 std::vector<double> correlations(stripesAmount, 0.);
266
267 std::vector<double> sampSums(stripesAmount, 0);
268 std::vector<double> sampSqSums(stripesAmount, 0);
269 std::vector<double> refSums(stripesAmount, 0);
270 std::vector<double> refSqSums(stripesAmount, 0);
271 std::vector<int> nzs(stripesAmount, 0);
272 std::vector<double> sampMaskedSums(stripesAmount, 0);
273 std::vector<double> refMaskedSums(stripesAmount, 0);
274
275 double a00 = map.at<double>(0, 0);
276 double a01 = map.at<double>(0, 1);
277 double a02 = map.at<double>(0, 2);
278 double a10 = map.at<double>(1, 0);
279 double a11 = map.at<double>(1, 1);
280 double a12 = map.at<double>(1, 2);
281 double a20 = 0;
282 double a21 = 0;
283 double a22 = 0;
284 if (motionType == MOTION_HOMOGRAPHY) {
285 a20 = map.at<double>(2, 0);
286 a21 = map.at<double>(2, 1);
287 a22 = map.at<double>(2, 2);
288 }
289
290 const elemtype* samplePtr0 = sampleWithGrad.ptr<elemtype>(0);
291
292 parallel_for_(Range(0, stripesAmount), [&](const Range& range) {
293 int stripeIdx = range.start;
294 int ystart = (hr * stripeIdx) / stripesAmount;
295 ystart = roundUp(ystart, deltaY);
296 int yend = (hr * (range.end)) / stripesAmount;
297 // we don't store intermediate jacobian; instead, we iteratively update Hessian, sampleProj and refProj
298 for (int y = ystart; y < yend; y += deltaY) {
299 const elemtype* refPtr = ref.ptr<elemtype>(y);
300
301 std::array<float, (HESSIAN_PARAMS * HESSIAN_PARAMS + HESSIAN_PARAMS) / 2> hessPcache{};
302 std::array<float, HESSIAN_PARAMS> iprojCache{};
303 std::array<float, HESSIAN_PARAMS> tprojCache{};
304 std::array<float, HESSIAN_PARAMS> projSubCache{};
305
306 const float numeratorX0 = y * (float)a01 + (float)a02;
307 const float numeratorY0 = y * (float)a11 + (float)a12;
308 const float denominator0 = y * (float)a21 + (float)a22;
309 int x = 0;
310 for (; x < wr; x++) { //Tail handler
311 float sx, sy, denominator;
312 MotionTraits<motionType>::tailHandlerGetCoord(sx, sy, denominator, x, numeratorX0, numeratorY0,
313 denominator0, (float)a00, (float)a10, (float)a20);
314 const unsigned int x0 = (interpolation == INTER_LINEAR) ? static_cast<int>(std::floor(sx)) : saturate_cast<unsigned>(sx);
315 const unsigned int y0 = (interpolation == INTER_LINEAR) ? static_cast<int>(std::floor(sy)) : saturate_cast<unsigned>(sy);
316 if(interpolation == INTER_LINEAR && (static_cast<int>(x0 < xcond) & static_cast<int>(y0 < ycond)) == 0)
317 continue;
318 if (interpolation == INTER_NEAREST && (static_cast<int>(x0 < xcond) & static_cast<int>(y0 < ycond)) == 0)
319 continue;
320 float sampleVal = 0;
321 float gx = 0;
322 float gy = 0;
323 float fVal = 0;
324 if(interpolation == INTER_LINEAR) {
325 const int x1 = x0 + 1;
326 const int y1 = y0 + 1;
327
328 const float dx = sx - x0;
329 const float dy = sy - y0;
330
331 const float p00_val = samplePtr0[4 * y0 * ws + 4 * x0];
332 const float p01_val = samplePtr0[4 * y0 * ws + 4 * x1];
333 const float p10_val = samplePtr0[4 * y1 * ws + 4 * x0];
334 const float p11_val = samplePtr0[4 * y1 * ws + 4 * x1];
335 const float p0_val = p00_val * (1.0f - dx) + p01_val * dx;
336 const float p1_val = p10_val * (1.0f - dx) + p11_val * dx;
337
338 const float p00_gx = samplePtr0[4 * y0 * ws + 4 * x0 + 1];
339 const float p01_gx = samplePtr0[4 * y0 * ws + 4 * x1 + 1];
340 const float p10_gx = samplePtr0[4 * y1 * ws + 4 * x0 + 1];
341 const float p11_gx = samplePtr0[4 * y1 * ws + 4 * x1 + 1];
342 const float p0_gx = p00_gx * (1.0f - dx) + p01_gx * dx;
343 const float p1_gx = p10_gx * (1.0f - dx) + p11_gx * dx;
344
345 const float p00_gy = samplePtr0[4 * y0 * ws + 4 * x0 + 2];
346 const float p01_gy = samplePtr0[4 * y0 * ws + 4 * x1 + 2];
347 const float p10_gy = samplePtr0[4 * y1 * ws + 4 * x0 + 2];
348 const float p11_gy = samplePtr0[4 * y1 * ws + 4 * x1 + 2];
349 const float p0_gy = p00_gy * (1.0f - dx) + p01_gy * dx;
350 const float p1_gy = p10_gy * (1.0f - dx) + p11_gy * dx;
351
352 const float p00_mask = samplePtr0[4 * y0 * ws + 4 * x0 + 2] == 0.f ? 0.f : 1.f;
353 const float p01_mask = samplePtr0[4 * y0 * ws + 4 * x1 + 2] == 0.f ? 0.f : 1.f;
354 const float p10_mask = samplePtr0[4 * y1 * ws + 4 * x0 + 2] == 0.f ? 0.f : 1.f;
355 const float p11_mask = samplePtr0[4 * y1 * ws + 4 * x1 + 2] == 0.f ? 0.f : 1.f;
356
357 sampleVal = p0_val * (1.0f - dy) + p1_val * dy;
358 gx = p0_gx * (1.0f - dy) + p1_gx * dy;
359 gy = p0_gy * (1.0f - dy) + p1_gy * dy;
360 fVal = p00_mask * p01_mask * p10_mask * p11_mask;
361 }
362 else { // if(interpolation == INTER_NEAREST)
363 const elemtype* samplePtr = samplePtr0 + y0 * (ws * 4) + x0 * 4;
364 sampleVal = samplePtr[0];
365 gx = samplePtr[1];
366 gy = samplePtr[2];
367 fVal = float(samplePtr[3]) == 0.f ? 0.f : 1.f;
368 }
369
370 float refVal = refPtr[2 * x];
371 fVal *= float(refPtr[2 * x + 1]) == 0.f ? 0.f : 1.f;
372 sampleVal *= fVal;
373 refVal *= fVal;
374 sampSums[stripeIdx] += sampleVal;
375 sampSqSums[stripeIdx] += sampleVal * sampleVal;
376 refSums[stripeIdx] += refVal;
377 refSqSums[stripeIdx] += refVal * refVal;
378 nzs[stripeIdx] += (int)fVal;
379 sampMaskedSums[stripeIdx] += sampleVal;
380 refMaskedSums[stripeIdx] += refVal;
381 std::array<float, HESSIAN_PARAMS> jac = MotionTraits<motionType>::fillJacobian(x, y, sx, sy,
382 fVal, gx,
383 gy, (float)a00,
384 (float)a10, denominator);
385 constexprForUpperTriangle<HESSIAN_PARAMS>([&](int row_i, int col_i) {
386 hessPcache[hessianRowStart<motionType>(row_i) + (col_i - row_i)] += jac[row_i] * jac[col_i];
387 });
388 constexprFor<HESSIAN_PARAMS>([&](int elem) {
389 iprojCache[elem] += jac[elem] * sampleVal;
390 tprojCache[elem] += jac[elem] * refVal;
391 projSubCache[elem] += jac[elem] * fVal;
392 });
393 correlations[stripeIdx] += sampleVal * refVal;
394 }
395
396 constexprForUpperTriangle<HESSIAN_PARAMS>([&](int row, int col) {
397 hessPs[stripeIdx](row, col) += hessPcache[hessianRowStart<motionType>(row) + (col - row)];
398 });
399 constexprFor<HESSIAN_PARAMS>([&](int elem) {
400 iprojs[stripeIdx][elem] += iprojCache[elem];
401 tprojs[stripeIdx][elem] += tprojCache[elem];
402 projSubs[stripeIdx][elem] += projSubCache[elem];
403 });
404 }
405 });
406 double sampMaskedSum = 0;
407 double refMaskedSum = 0;
408 double correlation = 0;
409 sampSum = sampSqSum = refSum = refSqSum = nz = 0;
410
411 for (int stripeIdx = 0; stripeIdx < stripesAmount; stripeIdx++) {
412 correlation += correlations[stripeIdx];
413 sampSum += sampSums[stripeIdx];
414 sampSqSum += sampSqSums[stripeIdx];
415 refSum += refSums[stripeIdx];
416 refSqSum += refSqSums[stripeIdx];
417 sampMaskedSum += sampMaskedSums[stripeIdx];
418 refMaskedSum += refMaskedSums[stripeIdx];
419 nz += nzs[stripeIdx];
420 }
421 double scale = nz == 0 ? 0. : 1. / nz;
422 double sampMean = sampSum * scale;
423 double refMean = refSum * scale;
424 correlation += nz * sampMean * refMean - sampMaskedSum * refMean - refMaskedSum * sampMean;
425 double* hessPtr = hessian.ptr<double>(0);
426 double* sampleProjPtr = sampleProj.ptr<double>(0);
427 double* refProjPtr = refProj.ptr<double>(0);
428 for (int stripeIdx = 0; stripeIdx < stripesAmount; stripeIdx++) {
429 for (int hessNum = 0; hessNum < HESSIAN_PARAMS * HESSIAN_PARAMS; hessNum++) {
430 hessPtr[hessNum] += hessPs[stripeIdx].val[hessNum];
431 }
432 for (int projNum = 0; projNum < HESSIAN_PARAMS; projNum++) {
433 sampleProjPtr[projNum] += iprojs[stripeIdx][projNum] - projSubs[stripeIdx][projNum] * sampMean;
434 refProjPtr[projNum] += tprojs[stripeIdx][projNum] - projSubs[stripeIdx][projNum] * refMean;
435 }
436 }
437 constexprForUpperTriangle<HESSIAN_PARAMS>([&](int row, int col) {
438 hessPtr[col * HESSIAN_PARAMS + row] = hessPtr[row * HESSIAN_PARAMS + col];
439 });
440 return correlation;
441#undef HESSIAN_PARAMS
442}

444static void updateWarpingMatrixECC(Mat& map_matrix, const Mat& update, const int motionType) {
445 CV_Assert(map_matrix.type() == CV_64FC1);
446 CV_Assert(update.type() == CV_64FC1);
447
448 CV_Assert(motionType == MOTION_TRANSLATION || motionType == MOTION_EUCLIDEAN || motionType == MOTION_AFFINE ||
449 motionType == MOTION_HOMOGRAPHY);
450
451 if (motionType == MOTION_HOMOGRAPHY)
452 CV_Assert(map_matrix.rows == 3 && update.rows == 8);
453 else if (motionType == MOTION_AFFINE)
454 CV_Assert(map_matrix.rows == 2 && update.rows == 6);
455 else if (motionType == MOTION_EUCLIDEAN)
456 CV_Assert(map_matrix.rows == 2 && update.rows == 3);
457 else
458 CV_Assert(map_matrix.rows == 2 && update.rows == 2);
459
460 CV_Assert(update.cols == 1);
461
462 CV_Assert(map_matrix.isContinuous());
463 CV_Assert(update.isContinuous());
464
465 double* mapPtr = map_matrix.ptr<double>(0);
466 const double* updatePtr = update.ptr<double>(0);
467
468 if (motionType == MOTION_TRANSLATION) {
469 mapPtr[2] += updatePtr[0];
470 mapPtr[5] += updatePtr[1];
471 }
472 if (motionType == MOTION_AFFINE) {
473 mapPtr[0] += updatePtr[0];
474 mapPtr[3] += updatePtr[1];
475 mapPtr[1] += updatePtr[2];
476 mapPtr[4] += updatePtr[3];
477 mapPtr[2] += updatePtr[4];
478 mapPtr[5] += updatePtr[5];
479 }
480 if (motionType == MOTION_HOMOGRAPHY) {
481 mapPtr[0] += updatePtr[0];
482 mapPtr[3] += updatePtr[1];
483 mapPtr[6] += updatePtr[2];
484 mapPtr[1] += updatePtr[3];
485 mapPtr[4] += updatePtr[4];
486 mapPtr[7] += updatePtr[5];
487 mapPtr[2] += updatePtr[6];
488 mapPtr[5] += updatePtr[7];
489 }
490 if (motionType == MOTION_EUCLIDEAN) {
491 double new_theta = updatePtr[0];
492 new_theta += asin(mapPtr[3]);
493
494 mapPtr[2] += updatePtr[1];
495 mapPtr[5] += updatePtr[2];
496 mapPtr[0] = mapPtr[4] = cos(new_theta);
497 mapPtr[3] = sin(new_theta);
498 mapPtr[1] = -mapPtr[3];
499 }
500}

502static void optimizeECC(Mat& sampleWithGrad,
503 const Mat& reference,
504 Mat& map,
505 int motionType,
506 double* rho,
507 double* lastRho,
508 int deltaY,
509 int nparams,
510 int interpolation) {
511 CV_Assert(interpolation == INTER_NEAREST || interpolation == INTER_LINEAR);
512
513 // warp-back portion of the inputImage and gradients to the coordinate space of the referenceFloat
514 double correlation = 0;
515
516 // matrices needed for solving linear equation system for maximizing ECC
517 Mat hessian = Mat(nparams, nparams, CV_64F);
518 Mat hessianInv = Mat(nparams, nparams, CV_64F);
519 Mat sampleProjection = Mat(nparams, 1, CV_64F);
520 Mat referenceProjection = Mat(nparams, 1, CV_64F);
521 Mat sampleProjectionHessian = Mat(nparams, 1, CV_64F);
522 Mat errorProjection = Mat(nparams, 1, CV_64F);
523 Mat deltaP = Mat(nparams, 1, CV_64F);
524
525 double sampSum;
526 double sampSqSum;
527 double referenceSum;
528 double referenceSqSum;
529 int nz;
530
531 { // if(imageWithGrad.type() == CV_32FC4)
532 if (motionType == MOTION_TRANSLATION) {
533 correlation = imageHessianProjECC<MOTION_TRANSLATION, float>(map,
534 sampleWithGrad,
535 reference,
536 sampSum,
537 sampSqSum,
538 referenceSum,
539 referenceSqSum,
540 nz,
541 hessian,
542 sampleProjection,
543 referenceProjection,
544 deltaY,
545 interpolation);
546 } else if (motionType == MOTION_EUCLIDEAN) {
547 correlation = imageHessianProjECC<MOTION_EUCLIDEAN, float>(map,
548 sampleWithGrad,
549 reference,
550 sampSum,
551 sampSqSum,
552 referenceSum,
553 referenceSqSum,
554 nz,
555 hessian,
556 sampleProjection,
557 referenceProjection,
558 deltaY,
559 interpolation);
560 } else if (motionType == MOTION_AFFINE) {
561 correlation = imageHessianProjECC<MOTION_AFFINE, float>(map,
562 sampleWithGrad,
563 reference,
564 sampSum,
565 sampSqSum,
566 referenceSum,
567 referenceSqSum,
568 nz,
569 hessian,
570 sampleProjection,
571 referenceProjection,
572 deltaY,
573 interpolation);
574 } else {
575 correlation = imageHessianProjECC<MOTION_HOMOGRAPHY, float>(map,
576 sampleWithGrad,
577 reference,
578 sampSum,
579 sampSqSum,
580 referenceSum,
581 referenceSqSum,
582 nz,
583 hessian,
584 sampleProjection,
585 referenceProjection,
586 deltaY,
587 interpolation);
588 }
589 }
590 double scale = nz == 0 ? 0. : 1. / nz;
591 double sampMean = sampSum * scale;
592 double refMean = referenceSum * scale;
593 double sampStd = std::sqrt(std::max(sampSqSum * scale - sampMean * sampMean, 0.));
594 double refStd = std::sqrt(std::max(referenceSqSum * scale - refMean * refMean, 0.));
595
596 // inverse of Hessian
597 hessianInv = hessian.inv();
598 // calculate enhanced correlation coefficient (ECC)->rho
599 *lastRho = *rho;
600 double refNorm = std::sqrt(nz * refStd * refStd);
601 double sampNorm = std::sqrt(nz * sampStd * sampStd);
602
603 *rho = correlation / (sampNorm * refNorm);
604 if ((bool)cvIsNaN(*rho)) {
605 CV_Error(Error::StsNoConv, "NaN encountered.");
606 }
607
608 // calculate the parameter lambda to account for illumination variation
609 sampleProjectionHessian = hessianInv * sampleProjection;
610 const double lambdaN = (sampNorm * sampNorm) - sampleProjection.dot(sampleProjectionHessian);
611 const double lambdaD = correlation - referenceProjection.dot(sampleProjectionHessian);
612
613 if (lambdaD <= 0.0) {
614 CV_Error(Error::StsNoConv, "The algorithm stopped before its convergence. The correlation is going to be minimized. "
615 "Images may be uncorrelated or non-overlapped");
616 }
617 const double lambda = (lambdaN / lambdaD);
618
619 // estimate the update step delta_p
620 errorProjection = lambda * referenceProjection - sampleProjection;
621 gemm(hessianInv, errorProjection, 1., noArray(), 0., deltaP);
622
623 // update warping matrix
624 updateWarpingMatrixECC(map, deltaP, motionType);
625}

627static Mat prepareGradients(const Mat& sample) {
628 CV_Assert(sample.type() == CV_32FC2 || sample.type() == CV_16FC2);
629
630 const int ws = sample.cols;
631 const int hs = sample.rows;
632
633 Mat sampleWithGrad;
634 int ntasks = std::min(4, hs);
635
636 {
637 sampleWithGrad = Mat(hs, ws, CV_32FC4);
638 float* dstPtr = sampleWithGrad.ptr<float>();
639 parallel_for_(Range(0, ntasks), [&](const Range& range) {
640 int rowstart = range.start * hs / ntasks;
641 int rowend = range.end * hs / ntasks;
642 for (int row = rowstart; row < rowend; row++) {
643 const float* sampleCurLine = sample.ptr<float>(row);
644 const float* samplePrevLine = sample.ptr<float>(std::max(row - 1, 0));
645 const float* sampleNextLine = sample.ptr<float>(std::min(row + 1, hs - 1));
646 float gradDivY = (row > 0 && row + 1 < hs) ? 0.5f : 0.25f;
647 int col = 0;
648 for (; col < ws; col++) {
649 int prevCol = std::max(col - 1, 0);
650 int nextCol = std::min(col + 1, ws - 1);
651 float gradDivX = (col > 0 && col + 1 < ws) ? 0.5f : 0.25f;
652 dstPtr[row * ws * 4 + col * 4] = sampleCurLine[2 * col];
653 dstPtr[row * ws * 4 + col * 4 + 1] =
654 gradDivX * (sampleCurLine[2 * nextCol] - sampleCurLine[2 * prevCol]);
655 dstPtr[row * ws * 4 + col * 4 + 2] = gradDivY * (sampleNextLine[2 * col] - samplePrevLine[2 * col]);
656 dstPtr[row * ws * 4 + col * 4 + 3] = sampleCurLine[2 * col + 1];
657 }
658 }
659 }, ntasks);
660 }
661 return sampleWithGrad;
662}

685static Mat spliceWithMask(const Mat& image, const Mat& mask) {
686 CV_Assert(image.type() == CV_32F && (mask.empty() || mask.type() == CV_8U));
687 if (!mask.empty() && image.size() != mask.size()) {
688 CV_Error(Error::BadImageSize, "spliceWithMask: Mask and image have to be of same size.");
689 }
690 const int hs = image.rows;
691 const int ws = image.cols;
692
693 Mat result;
694 int ntasks = std::min(4, hs);
695 {
696 union conv_ {
697 uint32_t valU;
698 float val;
699 conv_() : valU(0xffffffff) {}
700 } conv;
701 result = Mat(hs, ws, CV_32FC2);
702 parallel_for_(Range(0, ntasks), [&](const Range& range) {
703 int rowstart = range.start * hs / ntasks;
704 int rowend = range.end * hs / ntasks;
705 for (int row = rowstart; row < rowend; row++) {
706 float* dstPtr = result.ptr<float>(row);
707 const float* srcPtr = image.ptr<float>(row);
708 const uint8_t* maskPtr = !mask.empty() ? mask.ptr<uint8_t>(row) : nullptr;
709 int col = 0;
710 for (; col < ws; col++) {
711 dstPtr[col * 2] = srcPtr[col];
712 dstPtr[col * 2 + 1] = (!maskPtr || maskPtr[col]) ? conv.val : 0;
713 }
714 }
715 }, ntasks);
716 }
717 return result;
718}

720static void scaleWarpMatrix(Mat& warpMatrix, float scale) {
721 if (warpMatrix.rows == 3) {
722 Mat invertScaleMat = Mat(3, 3, CV_64F, 0.f);
723 invertScaleMat.at<double>(0, 0) = 1.f / scale;
724 invertScaleMat.at<double>(1, 1) = 1.f / scale;
725 invertScaleMat.at<double>(2, 2) = 1.f;
726 Mat scaleMatrix = invertScaleMat.clone();
727 scaleMatrix.at<double>(0, 0) = scale;
728 scaleMatrix.at<double>(1, 1) = scale;
729 gemm(warpMatrix, invertScaleMat, 1., noArray(), 0., warpMatrix);
730 gemm(scaleMatrix, warpMatrix, 1., noArray(), 0., warpMatrix);
731 // Normalization, internal algorithms assumes, that a22 = 1.0f
732 for (int mel = 0; mel < 8; mel++) {
733 (reinterpret_cast<double*>(warpMatrix.data))[mel] /=
734 (reinterpret_cast<double*>(warpMatrix.data))[8];
735 }
736 (reinterpret_cast<double*>(warpMatrix.data))[8] = 1.f;
737 } else {
738 warpMatrix.at<double>(0, 2) *= scale;
739 warpMatrix.at<double>(1, 2) *= scale;
740 }
741}

743static void checkParams(const MatPyramid& referencePyramid,
744 const MatPyramid& samplePyramid,
745 Mat& map,
746 std::vector<int>& itersPerLevel,
747 const ECCParameters& eccParams) {
748 if (itersPerLevel.empty()) {
749 itersPerLevel.resize(eccParams.nlevels, eccParams.criteria.maxCount);
750 }
751 CV_Assert(eccParams.interpolation == INTER_NEAREST || eccParams.interpolation == INTER_LINEAR);
752 CV_Assert(static_cast<int>(itersPerLevel.size()) == eccParams.nlevels);
753 for (const auto& lvl : referencePyramid) {
754 CV_Assert(!lvl.empty() && lvl.type() == referencePyramid[0].type());
755 }
756 CV_Assert(!samplePyramid.empty());
757 for (const auto& lvl : samplePyramid) {
758 CV_Assert(!lvl.empty() && lvl.type() == samplePyramid[0].type());
759 }
760 CV_Assert(samplePyramid.size() == referencePyramid.size() && samplePyramid.size() == itersPerLevel.size());
761 CV_Assert(referencePyramid.back().rows > 1 && referencePyramid.back().cols > 1 &&
762 samplePyramid.back().rows > 1 && samplePyramid.back().cols > 1);
763 // If the user passed an un-initialized warpMatrix, initialize to identity
764 if (referencePyramid[0].type() != CV_32FC2 && referencePyramid[0].type() != CV_16FC2) {
765 CV_Error(Error::StsError, "Reference pyramid have to be prepared via prepareReferencePyramid function");
766 }
767 // accept only 1-channel images
768 CV_Assert(samplePyramid[0].type() == CV_32FC2 || samplePyramid[0].type() != CV_16FC2);
769 CV_Assert(map.type() == CV_64FC1);
770 if (map.cols != 3 || (map.rows != 2 && map.rows != 3)) {
771 CV_Error(Error::BadImageSize, "warpMatrix has incorrect size");
772 }
773
774 if (eccParams.motionType != MOTION_TRANSLATION && eccParams.motionType != MOTION_EUCLIDEAN &&
775 eccParams.motionType != MOTION_AFFINE && eccParams.motionType != MOTION_HOMOGRAPHY) {
776 CV_Error(Error::StsError, "Incorrect motion type");
777 }
778
779 if (eccParams.motionType == MOTION_HOMOGRAPHY && map.rows != 3) {
780 CV_Error(Error::BadImageSize, "warpMatrix has incorrect size");
781 }
782
783 if (!((bool)(eccParams.criteria.type & TermCriteria::COUNT) || (bool)(eccParams.criteria.type & TermCriteria::EPS))) {
784 CV_Error(Error::StsError, "Incorrect stop eccParams.criteria");
785 }
786}

789 InputArray imageMask, // Can be empty

791 int nlevels) {
792 MatPyramid imagePyramid, maskPyramid;

796 imagePyramid[lvl].convertTo(imgFloat, CV_32F, 1. / 255.);
797 if (gaussFiltSize != 0) {
798 GaussianBlur(imgFloat, imgFloat, Size(gaussFiltSize, gaussFiltSize), 0, 0);
799 }
800 imagePyramid[lvl] = spliceWithMask(
801 imgFloat,
802 (static_cast<int>(maskPyramid.size()) > lvl && !maskPyramid[lvl].empty()) ? maskPyramid[lvl] : Mat());
803 }
804 return imagePyramid;

807double findTransformECCMultiScale(InputArray reference,
808 InputArray sample,
809 InputOutputArray warpMatrixA,
810 const ECCParameters& eccParams,
811 InputArray referenceMask,
812 InputArray sampleMask) {
813 MatPyramid referencePyramid = prepareECCPyramid(reference, referenceMask, eccParams.gaussFiltSize, eccParams.nlevels);
814 MatPyramid samplePyramid = prepareECCPyramid(sample, sampleMask, eccParams.gaussFiltSize, eccParams.nlevels);
815 Mat& warpMatrix = warpMatrixA.getMatRef();
816 std::vector<int> itersPerLevelCopy = eccParams.itersPerLevel;
817 // If the user passed an un-initialized warpMatrix, initialize to identity
818 if (warpMatrix.empty())
819 {
820 int rowCount = eccParams.motionType == MOTION_HOMOGRAPHY ? 3 : 2;
821 warpMatrix = Mat::eye(rowCount, 3, CV_64FC1);
822 }
823 int warpMatrixType = warpMatrix.type();
824 if (warpMatrixType != CV_64FC1)
825 {
826 warpMatrix.convertTo(warpMatrix, CV_64FC1);
827 }
828
829 checkParams(referencePyramid,
830 samplePyramid,
831 warpMatrix,
832 itersPerLevelCopy,
833 eccParams);
834
835 int nparams = 0;
836 switch (eccParams.motionType) {
837 case MOTION_TRANSLATION:
838 nparams = MotionTraits<MOTION_TRANSLATION>::paramAmount;
839 break;
840 case MOTION_EUCLIDEAN:
841 nparams = MotionTraits<MOTION_EUCLIDEAN>::paramAmount;
842 break;
843 case MOTION_AFFINE:
844 nparams = MotionTraits<MOTION_AFFINE>::paramAmount;
845 break;
846 case MOTION_HOMOGRAPHY:
847 nparams = MotionTraits<MOTION_HOMOGRAPHY>::paramAmount;
848 break;
849 default:
850 CV_Error(Error::StsBadArg, "Incorrect motion type");
851 }
852
853 const std::vector<int> numberOfIterations = ((eccParams.criteria.type & TermCriteria::COUNT) != 0)
854 ? itersPerLevelCopy
855 : std::vector<int>(eccParams.nlevels, 200);
856 const double terminationEPS = (bool)(eccParams.criteria.type & TermCriteria::EPS) ? eccParams.criteria.epsilon : -1;
857
858 // Scale warp matrix multiple times to lower pyramid level
859 for (int pyrLevel = 0; pyrLevel < eccParams.nlevels - 1; pyrLevel++) {
860 scaleWarpMatrix(warpMatrix, 0.5);
861 }
862 double rho = -1;
863 for (int pyrLevel = eccParams.nlevels - 1; pyrLevel >= 0; --pyrLevel) {
864 const int hr = referencePyramid[pyrLevel].rows;
865
866 Mat sampleWithGrad = prepareGradients(samplePyramid[pyrLevel]);
867
868 const int LOW_SIZE = 200;
869 int deltaY = hr < LOW_SIZE ? 1 : 2;
870
871 // iteratively update mapMatrix
872 double lastRho = -terminationEPS;
873 for (int i = 1; (i <= numberOfIterations[pyrLevel]) && (fabs(rho - lastRho) >= terminationEPS); i++) {
874 optimizeECC(sampleWithGrad, referencePyramid[pyrLevel], warpMatrix, eccParams.motionType, &rho, &lastRho, deltaY, nparams, eccParams.interpolation);
875 }
876 if (pyrLevel > 0) {
877 scaleWarpMatrix(warpMatrix, 2);
878 }
879 }
880 if(warpMatrixType != CV_64FC1)
881 {
882 warpMatrix.convertTo(warpMatrix, warpMatrixType);
883 }
884 // return final correlation coefficient
885 return rho;
886}
