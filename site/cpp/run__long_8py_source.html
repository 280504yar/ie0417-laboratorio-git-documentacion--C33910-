77def longTestFilter(data, module=None):
78 res = ['*', '-'] + [v for m, v, _time in data if module is None or m == module]
79 return '--gtest_filter={}'.format(':'.join(res))

82# Parse one xml file, filter out tests which took less than 'timeLimit' seconds
83# Returns tuple: ( <module_name>, [ (<module_name>, <test_name>, <test_time>), ... ] )
84def parseOneFile(filename, timeLimit):
85 tree = ET.parse(filename)
86 root = tree.getroot()
87
88 def guess(s, delims):
89 for delim in delims:
90 tmp = s.partition(delim)
91 if len(tmp[1]) != 0:
92 return tmp[0]
93 return None
94 module = guess(filename, ['_posix_', '_nt_', '__']) or root.get('cv_module_name')
95 if not module:
96 return (None, None)
97 res = []
98 for elem in root.findall('.//testcase'):
99 key = '{}.{}'.format(elem.get('classname'), elem.get('name'))
100 val = elem.get('time')
101 if float(val) >= timeLimit:
102 res.append((module, key, float(val)))
103 return (module, res)
