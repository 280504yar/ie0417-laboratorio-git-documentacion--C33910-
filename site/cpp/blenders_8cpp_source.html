96void Blender::feed(InputArray _img, InputArray _mask, Point tl)
97{
98 Mat img = _img.getMat();
99 Mat mask = _mask.getMat();
100 Mat dst = dst_.getMat(ACCESS_RW);
101 Mat dst_mask = dst_mask_.getMat(ACCESS_RW);
102
103 CV_Assert(img.type() == CV_16SC3);
104 CV_Assert(mask.type() == CV_8U);
105 int dx = tl.x - dst_roi_.x;
106 int dy = tl.y - dst_roi_.y;
107
108 for (int y = 0; y < img.rows; ++y)
109 {
110 const Point3_<short> *src_row = img.ptr<Point3_<short> >(y);
111 Point3_<short> *dst_row = dst.ptr<Point3_<short> >(dy + y);
112 const uchar *mask_row = mask.ptr<uchar>(y);
113 uchar *dst_mask_row = dst_mask.ptr<uchar>(dy + y);
114
115 for (int x = 0; x < img.cols; ++x)
116 {
117 if (mask_row[x])
118 dst_row[dx + x] = src_row[x];
119 dst_mask_row[dx + x] |= mask_row[x];
120 }
121 }
122}

125void Blender::blend(InputOutputArray dst, InputOutputArray dst_mask)
126{
127 UMat mask;
128 compare(dst_mask_, 0, mask, CMP_EQ);
129 dst_.setTo(Scalar::all(0), mask);
130 dst.assign(dst_);
131 dst_mask.assign(dst_mask_);
132 dst_.release();
133 dst_mask_.release();
134}

144void FeatherBlender::feed(InputArray _img, InputArray mask, Point tl)
145{
146 Mat img = _img.getMat();
147 Mat dst = dst_.getMat(ACCESS_RW);
148
149 CV_Assert(img.type() == CV_16SC3);
150 CV_Assert(mask.type() == CV_8U);
151
152 createWeightMap(mask, sharpness_, weight_map_);
153 Mat weight_map = weight_map_.getMat(ACCESS_READ);
154 Mat dst_weight_map = dst_weight_map_.getMat(ACCESS_RW);
155
156 int dx = tl.x - dst_roi_.x;
157 int dy = tl.y - dst_roi_.y;
158
159 for (int y = 0; y < img.rows; ++y)
160 {
161 const Point3_<short>* src_row = img.ptr<Point3_<short> >(y);
162 Point3_<short>* dst_row = dst.ptr<Point3_<short> >(dy + y);
163 const float* weight_row = weight_map.ptr<float>(y);
164 float* dst_weight_row = dst_weight_map.ptr<float>(dy + y);
165
166 for (int x = 0; x < img.cols; ++x)
167 {
168 dst_row[dx + x].x += static_cast<short>(src_row[x].x * weight_row[x]);
169 dst_row[dx + x].y += static_cast<short>(src_row[x].y * weight_row[x]);
170 dst_row[dx + x].z += static_cast<short>(src_row[x].z * weight_row[x]);
171 dst_weight_row[dx + x] += weight_row[x];
172 }
173 }
174}

178{
179 normalizeUsingWeightMap(dst_weight_map_, dst_);
180 compare(dst_weight_map_, WEIGHT_EPS, dst_mask_, CMP_GT);
181 Blender::blend(dst, dst_mask);

185Rect FeatherBlender::createWeightMaps(const std::vector<UMat> &masks, const std::vector<Point> &corners,
186 std::vector<UMat> &weight_maps)
187{
188 weight_maps.resize(masks.size());
189 for (size_t i = 0; i < masks.size(); ++i)
190 createWeightMap(masks[i], sharpness_, weight_maps[i]);
191
192 Rect dst_roi = resultRoi(corners, masks);
193 Mat weights_sum(dst_roi.size(), CV_32F);
194 weights_sum.setTo(0);
195
196 for (size_t i = 0; i < weight_maps.size(); ++i)
197 {
198 Rect roi(corners[i].x - dst_roi.x, corners[i].y - dst_roi.y,
199 weight_maps[i].cols, weight_maps[i].rows);
200 add(weights_sum(roi), weight_maps[i], weights_sum(roi));
201 }
202
203 for (size_t i = 0; i < weight_maps.size(); ++i)
204 {
205 Rect roi(corners[i].x - dst_roi.x, corners[i].y - dst_roi.y,
206 weight_maps[i].cols, weight_maps[i].rows);
207 Mat tmp = weights_sum(roi);
208 tmp.setTo(1, tmp < std::numeric_limits<float>::epsilon());
209 divide(weight_maps[i], tmp, weight_maps[i]);
210 }
211
212 return dst_roi;
213}

233void MultiBandBlender::prepare(Rect dst_roi)
234{
235 dst_roi_final_ = dst_roi;
236
237 // Crop unnecessary bands
238 double max_len = static_cast<double>(std::max(dst_roi.width, dst_roi.height));
239 num_bands_ = std::min(actual_num_bands_, static_cast<int>(ceil(std::log(max_len) / std::log(2.0))));
240
241 // Add border to the final image, to ensure sizes are divided by (1 << num_bands_)
242 dst_roi.width += ((1 << num_bands_) - dst_roi.width % (1 << num_bands_)) % (1 << num_bands_);
243 dst_roi.height += ((1 << num_bands_) - dst_roi.height % (1 << num_bands_)) % (1 << num_bands_);
244
245 Blender::prepare(dst_roi);
246
247#if defined(HAVE_CUDA) && defined(HAVE_OPENCV_CUDAARITHM) && defined(HAVE_OPENCV_CUDAWARPING)
248 if (can_use_gpu_)
249 {
250 gpu_initialized_ = false;
251 gpu_feed_idx_ = 0;
252
253 gpu_tl_points_.clear();
254 gpu_weight_pyr_gauss_vec_.clear();
255 gpu_src_pyr_laplace_vec_.clear();
256 gpu_ups_.clear();
257 gpu_imgs_with_border_.clear();
258
259 gpu_dst_pyr_laplace_.resize(num_bands_ + 1);
260 gpu_dst_pyr_laplace_[0].create(dst_roi.size(), CV_16SC3);
261 gpu_dst_pyr_laplace_[0].setTo(Scalar::all(0));
262
263 gpu_dst_band_weights_.resize(num_bands_ + 1);
264 gpu_dst_band_weights_[0].create(dst_roi.size(), weight_type_);
265 gpu_dst_band_weights_[0].setTo(0);
266
267 for (int i = 1; i <= num_bands_; ++i)
268 {
269 gpu_dst_pyr_laplace_[i].create((gpu_dst_pyr_laplace_[i - 1].rows + 1) / 2,
270 (gpu_dst_pyr_laplace_[i - 1].cols + 1) / 2, CV_16SC3);
271 gpu_dst_band_weights_[i].create((gpu_dst_band_weights_[i - 1].rows + 1) / 2,
272 (gpu_dst_band_weights_[i - 1].cols + 1) / 2, weight_type_);
273 gpu_dst_pyr_laplace_[i].setTo(Scalar::all(0));
274 gpu_dst_band_weights_[i].setTo(0);
275 }
276 }
277 else
278#endif
279 {
280 dst_pyr_laplace_.clear();
281 dst_band_weights_.clear();
282
283 dst_pyr_laplace_.resize(num_bands_ + 1);
284 dst_pyr_laplace_[0] = dst_;
285
286 dst_band_weights_.resize(num_bands_ + 1);
287 dst_band_weights_[0].create(dst_roi.size(), weight_type_);
288 dst_band_weights_[0].setTo(0);
289
290 for (int i = 1; i <= num_bands_; ++i)
291 {
292 dst_pyr_laplace_[i].create((dst_pyr_laplace_[i - 1].rows + 1) / 2,
293 (dst_pyr_laplace_[i - 1].cols + 1) / 2, CV_16SC3);
294 dst_band_weights_[i].create((dst_band_weights_[i - 1].rows + 1) / 2,
295 (dst_band_weights_[i - 1].cols + 1) / 2, weight_type_);
296 dst_pyr_laplace_[i].setTo(Scalar::all(0));
297 dst_band_weights_[i].setTo(0);
298 }
299 }
300}

303static bool ocl_MultiBandBlender_feed(InputArray _src, InputArray _weight,
304 InputOutputArray _dst, InputOutputArray _dst_weight)
305{
306 String buildOptions = "-D DEFINE_feed";
307 ocl::buildOptionsAddMatrixDescription(buildOptions, "src", _src);
308 ocl::buildOptionsAddMatrixDescription(buildOptions, "weight", _weight);
309 ocl::buildOptionsAddMatrixDescription(buildOptions, "dst", _dst);
310 ocl::buildOptionsAddMatrixDescription(buildOptions, "dstWeight", _dst_weight);

316
317 k.args(ocl::KernelArg::ReadOnly(src),
318 ocl::KernelArg::ReadOnly(_weight.getUMat()),
319 ocl::KernelArg::ReadWrite(_dst.getUMat()),
320 ocl::KernelArg::ReadWrite(_dst_weight.getUMat())
321 );
322
323 size_t globalsize[2] = {(size_t)src.cols, (size_t)src.rows };
324 return k.run(2, globalsize, NULL, false);
325}

328void MultiBandBlender::feed(InputArray _img, InputArray mask, Point tl)
329{
330#if ENABLE_LOG
331 int64 t = getTickCount();
332#endif
333
334 UMat img;
335
336#if defined(HAVE_CUDA) && defined(HAVE_OPENCV_CUDAARITHM) && defined(HAVE_OPENCV_CUDAWARPING)
337 // If using gpu save the top left coordinate when running first time after prepare
338 if (can_use_gpu_)
339 {
340 if (!gpu_initialized_)
341 {
342 gpu_tl_points_.push_back(tl);
343 }
344 else
345 {
346 tl = gpu_tl_points_[gpu_feed_idx_];
347 }
348 }
349 // If _img is not a GpuMat get it as UMat from the InputArray object.
350 // If it is GpuMat make a dummy object with right dimensions but no data and
351 // get _img as a GpuMat
352 if (!_img.isGpuMat())
353#endif
354 {
355 img = _img.getUMat();
356 }
357#if defined(HAVE_CUDA) && defined(HAVE_OPENCV_CUDAARITHM) && defined(HAVE_OPENCV_CUDAWARPING)
358 else
359 {
360 gpu_img_ = _img.getGpuMat();
361 img = UMat(gpu_img_.rows, gpu_img_.cols, gpu_img_.type());
362 }
363#endif
364
365 CV_Assert(img.type() == CV_16SC3 || img.type() == CV_8UC3);
366 CV_Assert(mask.type() == CV_8U);
367
368 // Keep source image in memory with small border
369 int gap = 3 * (1 << num_bands_);
370 Point tl_new(std::max(dst_roi_.x, tl.x - gap),
371 std::max(dst_roi_.y, tl.y - gap));
372 Point br_new(std::min(dst_roi_.br().x, tl.x + img.cols + gap),
373 std::min(dst_roi_.br().y, tl.y + img.rows + gap));
374
375 // Ensure coordinates of top-left, bottom-right corners are divided by (1 << num_bands_).
376 // After that scale between layers is exactly 2.
377 //
378 // We do it to avoid interpolation problems when keeping sub-images only. There is no such problem when
379 // image is bordered to have size equal to the final image size, but this is too memory hungry approach.
380 tl_new.x = dst_roi_.x + (((tl_new.x - dst_roi_.x) >> num_bands_) << num_bands_);
381 tl_new.y = dst_roi_.y + (((tl_new.y - dst_roi_.y) >> num_bands_) << num_bands_);
382 int width = br_new.x - tl_new.x;
383 int height = br_new.y - tl_new.y;
384 width += ((1 << num_bands_) - width % (1 << num_bands_)) % (1 << num_bands_);
385 height += ((1 << num_bands_) - height % (1 << num_bands_)) % (1 << num_bands_);
386 br_new.x = tl_new.x + width;
387 br_new.y = tl_new.y + height;
388 int dy = std::max(br_new.y - dst_roi_.br().y, 0);
389 int dx = std::max(br_new.x - dst_roi_.br().x, 0);
390 tl_new.x -= dx; br_new.x -= dx;
391 tl_new.y -= dy; br_new.y -= dy;
392
393 int top = tl.y - tl_new.y;
394 int left = tl.x - tl_new.x;
395 int bottom = br_new.y - tl.y - img.rows;
396 int right = br_new.x - tl.x - img.cols;
397
398#if defined(HAVE_CUDA) && defined(HAVE_OPENCV_CUDAARITHM) && defined(HAVE_OPENCV_CUDAWARPING)
399 if (can_use_gpu_)
400 {
401 if (!gpu_initialized_)
402 {
403 gpu_imgs_with_border_.push_back(cuda::GpuMat());
404 gpu_weight_pyr_gauss_vec_.push_back(std::vector<cuda::GpuMat>(num_bands_+1));
405 gpu_src_pyr_laplace_vec_.push_back(std::vector<cuda::GpuMat>(num_bands_+1));
406 gpu_ups_.push_back(std::vector<cuda::GpuMat>(num_bands_));
407 }
408
409 // If _img is not GpuMat upload it to gpu else gpu_img_ was set already
410 if (!_img.isGpuMat())
411 {
412 gpu_img_.upload(img);
413 }
414
415 // Create the source image Laplacian pyramid
416 cuda::copyMakeBorder(gpu_img_, gpu_imgs_with_border_[gpu_feed_idx_], top, bottom,
417 left, right, BORDER_REFLECT);
418 gpu_imgs_with_border_[gpu_feed_idx_].convertTo(gpu_src_pyr_laplace_vec_[gpu_feed_idx_][0], CV_16S);
419 for (int i = 0; i < num_bands_; ++i)
420 cuda::pyrDown(gpu_src_pyr_laplace_vec_[gpu_feed_idx_][i],
421 gpu_src_pyr_laplace_vec_[gpu_feed_idx_][i + 1]);
422 for (int i = 0; i < num_bands_; ++i)
423 {
424 cuda::pyrUp(gpu_src_pyr_laplace_vec_[gpu_feed_idx_][i + 1], gpu_ups_[gpu_feed_idx_][i]);
425 cuda::subtract(gpu_src_pyr_laplace_vec_[gpu_feed_idx_][i],
426 gpu_ups_[gpu_feed_idx_][i],
427 gpu_src_pyr_laplace_vec_[gpu_feed_idx_][i]);
428 }
429
430 // Create the weight map Gaussian pyramid only if not yet initialized
431 if (!gpu_initialized_)
432 {
433 if (mask.isGpuMat())
434 {
435 gpu_mask_ = mask.getGpuMat();
436 }
437 else
438 {
439 gpu_mask_.upload(mask);
440 }
441
442 if (weight_type_ == CV_32F)
443 {
444 gpu_mask_.convertTo(gpu_weight_map_, CV_32F, 1. / 255.);
445 }
446 else // weight_type_ == CV_16S
447 {
448 gpu_mask_.convertTo(gpu_weight_map_, CV_16S);
449 cuda::compare(gpu_mask_, 0, gpu_add_mask_, CMP_NE);
450 cuda::add(gpu_weight_map_, Scalar::all(1), gpu_weight_map_, gpu_add_mask_);
451 }
452 cuda::copyMakeBorder(gpu_weight_map_, gpu_weight_pyr_gauss_vec_[gpu_feed_idx_][0], top,
453 bottom, left, right, BORDER_CONSTANT);
454 for (int i = 0; i < num_bands_; ++i)
455 cuda::pyrDown(gpu_weight_pyr_gauss_vec_[gpu_feed_idx_][i],
456 gpu_weight_pyr_gauss_vec_[gpu_feed_idx_][i + 1]);
457 }
458
459 int y_tl = tl_new.y - dst_roi_.y;
460 int y_br = br_new.y - dst_roi_.y;
461 int x_tl = tl_new.x - dst_roi_.x;
462 int x_br = br_new.x - dst_roi_.x;
463
464 // Add weighted layer of the source image to the final Laplacian pyramid layer
465 for (int i = 0; i <= num_bands_; ++i)
466 {
467 Rect rc(x_tl, y_tl, x_br - x_tl, y_br - y_tl);
468 cuda::GpuMat &_src_pyr_laplace = gpu_src_pyr_laplace_vec_[gpu_feed_idx_][i];
469 cuda::GpuMat _dst_pyr_laplace = gpu_dst_pyr_laplace_[i](rc);
470 cuda::GpuMat &_weight_pyr_gauss = gpu_weight_pyr_gauss_vec_[gpu_feed_idx_][i];
471 cuda::GpuMat _dst_band_weights = gpu_dst_band_weights_[i](rc);
472
473 using namespace cv::cuda::device::blend;
474 if (weight_type_ == CV_32F)
475 {
476 addSrcWeightGpu32F(_src_pyr_laplace, _weight_pyr_gauss, _dst_pyr_laplace, _dst_band_weights, rc);
477 }
478 else
479 {
480 addSrcWeightGpu16S(_src_pyr_laplace, _weight_pyr_gauss, _dst_pyr_laplace, _dst_band_weights, rc);
481 }
482 x_tl /= 2; y_tl /= 2;
483 x_br /= 2; y_br /= 2;
484 }
485 ++gpu_feed_idx_;
486 return;
487 }
488#endif
489
490 // Create the source image Laplacian pyramid
491 UMat img_with_border;
492 copyMakeBorder(_img, img_with_border, top, bottom, left, right,
493 BORDER_REFLECT);
494 LOGLN(" Add border to the source image, time: " << ((getTickCount() - t) / getTickFrequency()) << " sec");
495#if ENABLE_LOG
496 t = getTickCount();
497#endif
498
499 std::vector<UMat> src_pyr_laplace;
500 createLaplacePyr(img_with_border, num_bands_, src_pyr_laplace);
501
502 LOGLN(" Create the source image Laplacian pyramid, time: " << ((getTickCount() - t) / getTickFrequency()) << " sec");
503#if ENABLE_LOG
504 t = getTickCount();
505#endif
506
507 // Create the weight map Gaussian pyramid
508 UMat weight_map;
509 std::vector<UMat> weight_pyr_gauss(num_bands_ + 1);
510
511 if (weight_type_ == CV_32F)
512 {
513 mask.getUMat().convertTo(weight_map, CV_32F, 1./255.);
514 }
515 else // weight_type_ == CV_16S
516 {
517 mask.getUMat().convertTo(weight_map, CV_16S);
518 UMat add_mask;
519 compare(mask, 0, add_mask, CMP_NE);
520 add(weight_map, Scalar::all(1), weight_map, add_mask);
521 }
522
523 copyMakeBorder(weight_map, weight_pyr_gauss[0], top, bottom, left, right, BORDER_CONSTANT);
524
525 for (int i = 0; i < num_bands_; ++i)
526 pyrDown(weight_pyr_gauss[i], weight_pyr_gauss[i + 1]);
527
528 LOGLN(" Create the weight map Gaussian pyramid, time: " << ((getTickCount() - t) / getTickFrequency()) << " sec");
529#if ENABLE_LOG
530 t = getTickCount();
531#endif
532
533 int y_tl = tl_new.y - dst_roi_.y;
534 int y_br = br_new.y - dst_roi_.y;
535 int x_tl = tl_new.x - dst_roi_.x;
536 int x_br = br_new.x - dst_roi_.x;
537
538 // Add weighted layer of the source image to the final Laplacian pyramid layer
539 for (int i = 0; i <= num_bands_; ++i)
540 {
541 Rect rc(x_tl, y_tl, x_br - x_tl, y_br - y_tl);
542#ifdef HAVE_OPENCL
543 if ( !cv::ocl::isOpenCLActivated() ||
544 !ocl_MultiBandBlender_feed(src_pyr_laplace[i], weight_pyr_gauss[i],
545 dst_pyr_laplace_[i](rc), dst_band_weights_[i](rc)) )
546#endif
547 {
548 Mat _src_pyr_laplace = src_pyr_laplace[i].getMat(ACCESS_READ);
549 Mat _dst_pyr_laplace = dst_pyr_laplace_[i](rc).getMat(ACCESS_RW);
550 Mat _weight_pyr_gauss = weight_pyr_gauss[i].getMat(ACCESS_READ);
551 Mat _dst_band_weights = dst_band_weights_[i](rc).getMat(ACCESS_RW);
552 if (weight_type_ == CV_32F)
553 {
554 for (int y = 0; y < rc.height; ++y)
555 {
556 const Point3_<short>* src_row = _src_pyr_laplace.ptr<Point3_<short> >(y);
557 Point3_<short>* dst_row = _dst_pyr_laplace.ptr<Point3_<short> >(y);
558 const float* weight_row = _weight_pyr_gauss.ptr<float>(y);
559 float* dst_weight_row = _dst_band_weights.ptr<float>(y);
560
561 for (int x = 0; x < rc.width; ++x)
562 {
563 dst_row[x].x += static_cast<short>(src_row[x].x * weight_row[x]);
564 dst_row[x].y += static_cast<short>(src_row[x].y * weight_row[x]);
565 dst_row[x].z += static_cast<short>(src_row[x].z * weight_row[x]);
566 dst_weight_row[x] += weight_row[x];
567 }
568 }
569 }
570 else // weight_type_ == CV_16S
571 {
572 for (int y = 0; y < y_br - y_tl; ++y)
573 {
574 const Point3_<short>* src_row = _src_pyr_laplace.ptr<Point3_<short> >(y);
575 Point3_<short>* dst_row = _dst_pyr_laplace.ptr<Point3_<short> >(y);
576 const short* weight_row = _weight_pyr_gauss.ptr<short>(y);
577 short* dst_weight_row = _dst_band_weights.ptr<short>(y);
578
579 for (int x = 0; x < x_br - x_tl; ++x)
580 {
581 dst_row[x].x += short((src_row[x].x * weight_row[x]) >> 8);
582 dst_row[x].y += short((src_row[x].y * weight_row[x]) >> 8);
583 dst_row[x].z += short((src_row[x].z * weight_row[x]) >> 8);
584 dst_weight_row[x] += weight_row[x];
585 }
586 }
587 }
588 }
589#ifdef HAVE_OPENCL
590 else
591 {
592 CV_IMPL_ADD(CV_IMPL_OCL);
593 }
594#endif
595
596 x_tl /= 2; y_tl /= 2;
597 x_br /= 2; y_br /= 2;
598 }
599
600 LOGLN(" Add weighted layer of the source image to the final Laplacian pyramid layer, time: " << ((getTickCount() - t) / getTickFrequency()) << " sec");
601}

604void MultiBandBlender::blend(InputOutputArray dst, InputOutputArray dst_mask)
605{
606 Rect dst_rc(0, 0, dst_roi_final_.width, dst_roi_final_.height);
607#if defined(HAVE_CUDA) && defined(HAVE_OPENCV_CUDAARITHM) && defined(HAVE_OPENCV_CUDAWARPING)
608 if (can_use_gpu_)
609 {
610 if (!gpu_initialized_)
611 {
612 gpu_ups_.push_back(std::vector<cuda::GpuMat>(num_bands_+1));
613 }
614
615 for (int i = 0; i <= num_bands_; ++i)
616 {
617 cuda::GpuMat dst_i = gpu_dst_pyr_laplace_[i];
618 cuda::GpuMat weight_i = gpu_dst_band_weights_[i];
619
620 using namespace ::cv::cuda::device::blend;
621 if (weight_type_ == CV_32F)
622 {
623 normalizeUsingWeightMapGpu32F(weight_i, dst_i, weight_i.cols, weight_i.rows);
624 }
625 else
626 {
627 normalizeUsingWeightMapGpu16S(weight_i, dst_i, weight_i.cols, weight_i.rows);
628 }
629 }
630
631 // Restore image from Laplacian pyramid
632 for (size_t i = num_bands_; i > 0; --i)
633 {
634 cuda::pyrUp(gpu_dst_pyr_laplace_[i], gpu_ups_[gpu_ups_.size()-1][num_bands_-i]);
635 cuda::add(gpu_ups_[gpu_ups_.size()-1][num_bands_-i],
636 gpu_dst_pyr_laplace_[i - 1],
637 gpu_dst_pyr_laplace_[i - 1]);
638 }
639
640 // If dst is GpuMat do masking on gpu and return dst as a GpuMat
641 // else download the image to cpu and return it as an ordinary Mat
642 if (dst.isGpuMat())
643 {
644 cuda::GpuMat &gpu_dst = dst.getGpuMatRef();
645
646 cuda::compare(gpu_dst_band_weights_[0](dst_rc), WEIGHT_EPS, gpu_dst_mask_, CMP_GT);
647
648 cuda::compare(gpu_dst_mask_, 0, gpu_mask_, CMP_EQ);
649
650 gpu_dst_pyr_laplace_[0](dst_rc).setTo(Scalar::all(0), gpu_mask_);
651 gpu_dst_pyr_laplace_[0](dst_rc).convertTo(gpu_dst, CV_16S);
652
653 }
654 else
655 {
656 gpu_dst_pyr_laplace_[0](dst_rc).download(dst_);
657 Mat dst_band_weights_0;
658 gpu_dst_band_weights_[0].download(dst_band_weights_0);
659
660 compare(dst_band_weights_0(dst_rc), WEIGHT_EPS, dst_mask_, CMP_GT);
661 Blender::blend(dst, dst_mask);
662 }
663
664 // Set destination Mats to 0 so new image can be blended
665 for (size_t i = 0; i < (size_t)(num_bands_ + 1); ++i)
666 {
667 gpu_dst_band_weights_[i].setTo(0);
668 gpu_dst_pyr_laplace_[i].setTo(Scalar::all(0));
669 }
670 gpu_feed_idx_ = 0;
671 gpu_initialized_ = true;
672 }
673 else
674#endif
675 {
676 cv::UMat dst_band_weights_0;
677
678 for (int i = 0; i <= num_bands_; ++i)
679 normalizeUsingWeightMap(dst_band_weights_[i], dst_pyr_laplace_[i]);
680
681 restoreImageFromLaplacePyr(dst_pyr_laplace_);
682
683 dst_ = dst_pyr_laplace_[0](dst_rc);
684 dst_band_weights_0 = dst_band_weights_[0];
685
686 dst_pyr_laplace_.clear();
687 dst_band_weights_.clear();
688
689 compare(dst_band_weights_0(dst_rc), WEIGHT_EPS, dst_mask_, CMP_GT);
690
691 Blender::blend(dst, dst_mask);
692 }
693}

720void normalizeUsingWeightMap(InputArray _weight, InputOutputArray _src)
721{
722 Mat src;
723 Mat weight;
724
725#ifdef HAVE_OPENCL
726 if ( !cv::ocl::isOpenCLActivated() ||
727 !ocl_normalizeUsingWeightMap(_weight, _src) )
728#endif
729 {
730 src = _src.getMat();
731 weight = _weight.getMat();
732
733 CV_Assert(src.type() == CV_16SC3);
734
735 if (weight.type() == CV_32FC1)
736 {
737 for (int y = 0; y < src.rows; ++y)
738 {
739 Point3_<short> *row = src.ptr<Point3_<short> >(y);
740 const float *weight_row = weight.ptr<float>(y);
741
742 for (int x = 0; x < src.cols; ++x)
743 {
744 row[x].x = static_cast<short>(row[x].x / (weight_row[x] + WEIGHT_EPS));
745 row[x].y = static_cast<short>(row[x].y / (weight_row[x] + WEIGHT_EPS));
746 row[x].z = static_cast<short>(row[x].z / (weight_row[x] + WEIGHT_EPS));
747 }
748 }
749 }
750 else
751 {
752 CV_Assert(weight.type() == CV_16SC1);
753
754 for (int y = 0; y < src.rows; ++y)
755 {
756 const short *weight_row = weight.ptr<short>(y);
757 Point3_<short> *row = src.ptr<Point3_<short> >(y);
758
759 for (int x = 0; x < src.cols; ++x)
760 {
761 int w = weight_row[x] + 1;
762 row[x].x = static_cast<short>((row[x].x << 8) / w);
763 row[x].y = static_cast<short>((row[x].y << 8) / w);
764 row[x].z = static_cast<short>((row[x].z << 8) / w);
765 }
766 }
767 }
768 }
769#ifdef HAVE_OPENCL
770 else
771 {
772 CV_IMPL_ADD(CV_IMPL_OCL);
773 }
774#endif
775}

788void createLaplacePyr(InputArray img, int num_levels, std::vector<UMat> &pyr)
789{
790 pyr.resize(num_levels + 1);
791
792 if(img.depth() == CV_8U)
793 {
794 if(num_levels == 0)
795 {
796 img.getUMat().convertTo(pyr[0], CV_16S);
797 return;
798 }
799
800 UMat downNext;
801 UMat current = img.getUMat();
802 pyrDown(img, downNext);
803
804 for(int i = 1; i < num_levels; ++i)
805 {
806 UMat lvl_up;
807 UMat lvl_down;
808
809 pyrDown(downNext, lvl_down);
810 pyrUp(downNext, lvl_up, current.size());
811 subtract(current, lvl_up, pyr[i-1], noArray(), CV_16S);
812
813 current = downNext;
814 downNext = lvl_down;
815 }
816
817 {
818 UMat lvl_up;
819 pyrUp(downNext, lvl_up, current.size());
820 subtract(current, lvl_up, pyr[num_levels-1], noArray(), CV_16S);
821
822 downNext.convertTo(pyr[num_levels], CV_16S);
823 }
824 }
825 else
826 {
827 pyr[0] = img.getUMat();
828 for (int i = 0; i < num_levels; ++i)
829 pyrDown(pyr[i], pyr[i + 1]);
830 UMat tmp;
831 for (int i = 0; i < num_levels; ++i)
832 {
833 pyrUp(pyr[i + 1], tmp, pyr[i].size());
834 subtract(pyr[i], tmp, pyr[i]);
835 }
836 }
837}
