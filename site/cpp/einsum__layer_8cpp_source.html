17static bool IsTransposeReshapeForEinsum(const std::vector<size_t>& perm,
18 const MatShape& input_dims,
19 MatShape& new_shape) {
20 // As long as the dims with values > 1 stay in the same order, it's a reshape.
21 // Example: Shape=(1,1,1024,4096) -> perm=(2,0,3,1).
22 size_t last_permuted_axis = 0;
23 for (size_t i = 0; i < perm.size(); ++i) {
24 if (input_dims[perm[i]] == 1)
25 continue;
26 if (perm[i] < last_permuted_axis)
27 return false;
28 last_permuted_axis = perm[i];
29 }
30 new_shape.assign(input_dims.begin(), input_dims.end());
31 for (size_t i = 0; i < perm.size(); ++i) {
32 new_shape[i] = input_dims[perm[i]];
33 }
34 return true;
35}

38static Mat Transpose(
39 const Mat& input,
40 const MatShape& input_shape_override,
41 const std::vector<size_t> permutation)
42{
43
44 int input_rank = input_shape_override.size();
45 CV_Assert(input_rank == permutation.size());
46
47 bool reshape = input.dims != input_rank;
48
49 Mat input_reshaped;
50 if(reshape){
51 input_reshaped = input.reshape(1, input_shape_override.size(), input_shape_override.data());
52 }
53
54 MatShape outputDims;
55 outputDims.reserve(input_rank);
56 for (const auto& dim : permutation)
57 outputDims.emplace_back(input_shape_override[dim]);
58
59 Mat output;
60 MatShape order(permutation.begin(), permutation.end());
61
62 std::vector<int> order_(order.begin(), order.end());
63 cv::transposeND((reshape ? input_reshaped : input), order_, output);
64 return output;
65}

68bool IsTransposeRequired(size_t input_rank, const std::vector<size_t>& permutation) {
69
70 // No transpose required for scalars
71 if (input_rank == 0 || permutation.size() == 0){
72 return false;
73 }
74 CV_Assert(input_rank == permutation.size());
75
76 // Weeds out cases where permutation is something like [0, 1, 2] for a 3D input and so on
77 bool transpose_required = false;
78 for (size_t i = 0; i < input_rank; ++i) {
79 if (permutation[i] != i) {
80 transpose_required = true;
81 break;
82 }
83 }
84
85 return transpose_required;
86}

89bool IsTransposeRequiredForDiagonal(int dim1, int dim2, int rank) {
90 // If the input is 2D, we don't need a transpose
91 if (rank == 2)
92 return false;
93
94 // If the two dims are the innermost dims, no transpose is required
95 if ((dim1 == rank - 1 && dim2 == rank - 2) ||
96 (dim1 == rank - 2 && dim2 == rank - 1))
97 return false;
98
99 // Transpose is required
100 return true;
101}

103template <typename T>
104Mat DiagonalDataAssignment(Mat input) {
105
106 int rank = input.dims;
107 CV_Assert(rank >= 2);
108 CV_Assert(input.size[rank - 1] == input.size[rank - 2]);
109 MatShape original_dims = shape(input);
110
111 if (rank > 3){
112 //reshape to 3D mat
113 int collapsed_size = 1;
114 for (int i = 0; i < rank - 2; ++i) {
115 collapsed_size *= input.size[i];
116 }
117 std::vector<int> reshaped_dims = {collapsed_size, input.size[rank - 2], input.size[rank - 1]};
118 input = input.reshape(1, reshaped_dims);
119 }
120
121 // Compute total number of higher-dimensional slices
122 int total_slices = input.size[0];
123
124 original_dims[rank - 1] = 1; // Set the last dimension to 1, as we have extracted the diagonal
125 Mat output = Mat(original_dims, input.type());
126
127 int inner_stride = input.size[input.dims - 1];
128 auto inputPtr = input.ptr<T>();
129 auto outputPtr = output.ptr<T>();
130 for (int slice = 0; slice < total_slices; ++slice) {
131 for (int j = 0; j < inner_stride; ++j) {
132 // Direct memory access using raw pointers
133 outputPtr[slice * inner_stride + j] = inputPtr[slice * inner_stride * inner_stride + j * inner_stride + j];
134 }
135 }
136 return output;
137}

139/* Extract the diagonal elements from the last two dimensions of the tensor.
140For instance, given an input_shape of [1, 2, 3, 3]:
141
142The flexibility in this implementation allows one to choose which of the two
143last dimensions retains its value, determined by the `preserve_innermost_dim_val` parameter.
144
145When preserve_innermost_dim_val == true:
146 The resulting shape is [1, 2, 1, 3], indicating the diagonal has 3 elements,
147 and it keeps the dimension value of the innermost dimension.
148
149When preserve_innermost_dim_val == false:
150 The resulting shape is [1, 2, 3, 1], indicating the diagonal also has 3 elements,
151 but it retains the dimension value of the penultimate dimension. */
152Mat DiagonalInnermostDims(const Mat& input, bool preserve_innermost_dim_val) {
153 const MatShape input_dims = shape(input);
154 int rank = input_dims.size();
155
156 // This is an internal method and we already have finished all validations in the calling method.
157 // We proceed without duplicating all validations again here.
158
159 // We have a minimalistic check here to make sure the innermost dims have the same dim value
160 // as the calling method may have done a transpose before calling this method
161 CV_CheckEQ(input.size[rank - 1], input.size[rank - 2],
162 "innermost dims should have the same dim value to parse the diagonal elements");
163
164 MatShape output_dims = input_dims; // Copy the original dims
165 if (preserve_innermost_dim_val) {
166 output_dims[rank - 2] = 1;
167 } else {
168 output_dims[rank - 1] = 1;
169 }
170
171 Mat output;
172 switch (input.depth())
173 {
174 case CV_32F:
175 output = DiagonalDataAssignment<float>(input);
176 break;
177 case CV_64F:
178 output = DiagonalDataAssignment<double>(input);
179 break;
180 case CV_16F:
181 {
182 Mat tmp32;
183 input.convertTo(tmp32, CV_32F);
184 Mat out32 = DiagonalDataAssignment<float>(tmp32);
185 out32.convertTo(output, input.type());
186 break;
187 }
188 default:
189 {
190 Mat tmp32;
191 input.convertTo(tmp32, CV_32F);
192 Mat out32 = DiagonalDataAssignment<float>(tmp32);
193 out32.convertTo(output, input.type());
194 break;
195 }
196 }
197
198 if (output_dims != shape(output)){
199 CV_Error(Error::StsError, "Output shape does not match with calculated shape");
200 }
201 return output;
202}

204Mat Diagonal(const Mat& input, int dim1, int dim2)
205{
206 const MatShape input_dims = shape(input);
207 int rank = input_dims.size();
208
209 if (!(rank >= 2 && dim1 != dim2 && input_dims[dim1] == input_dims[dim2])){
210 std::string input_dims_str = std::accumulate(std::next(input_dims.begin()), input_dims.end(), std::to_string(input_dims[0]),
211 [](const std::string& a, int b) {
212 return a + ' ' + std::to_string(b);
213 });
214 CV_Error(Error::StsError, cv::format("Cannot parse the diagonal elements along dims %d and %d for input shape %s",dim1, dim2, input_dims_str.c_str()));
215 }
216
217 int first_dim = std::min(dim1, dim2);
218 int second_dim = std::max(dim1, dim2);
219
220 Mat output;
221 bool preserve_innermost_dim_val = false;
222
223 bool is_transpose_required = IsTransposeRequiredForDiagonal(dim1, dim2, rank);
224 if (is_transpose_required)
225 {
226 std::vector<size_t> permutation(rank, 0);
227 int first_dim_axis = -1; // This is the axis eventually occupied by the first_dim
228
229 // If one of the diagonal dimensions is one of the 2 innermost dims, then leave it as such
230 // so as to avoid transpose overhead
231 if (first_dim == rank - 2) { // If rank - 2 is occupied by first_dim, keep it there
232 permutation[rank - 2] = first_dim;
233 first_dim_axis = rank - 2;
234 } else {
235 if (second_dim != rank - 2) { // If rank - 2 is not occupied by second_dim, then put first_dim there
236 permutation[rank - 2] = first_dim;
237 first_dim_axis = rank - 2;
238 } else { // If rank - 2 is occupied by second_dim, then put first_dim in rank - 1
239 permutation[rank - 1] = first_dim;
240 first_dim_axis = rank - 1;
241 preserve_innermost_dim_val = true; // We always want to preserve the dim value of the first_dim
242 }
243 }
244
245 // Put the second_dim in the dim not occupied by the first_dim
246 if (first_dim_axis != rank - 1) {
247 permutation[rank - 1] = second_dim;
248 } else {
249 permutation[rank - 2] = second_dim;
250 }
251
252 size_t iter = 0;
253 for (int i = 0; i < rank; ++i) {
254 if (i != first_dim && i != second_dim) {
255 permutation[iter++] = i;
256 }
257 }
258
259 // Permute the input so that the dims from which we need the diagonal form the innermost dims
260 Mat transposed = Transpose(input, input_dims, permutation);
261
262 // Parse the diagonal from the innermost dims
263 output = DiagonalInnermostDims(transposed, preserve_innermost_dim_val);
264
265 // Swap back the dimensions to the original axes ordering using a "reverse permutation"
266 // Find the "reverse" permutation
267 iter = 0;
268 std::vector<size_t> reverse_permutation(rank, 0);
269 for (const auto& perm : permutation) {
270 reverse_permutation[perm] = iter++;
271 }
272
273 // Permute using the reverse permutation to get back the original axes ordering
274 // (Pass in CPU Transpose function here as this Diagonal method will only be used for CPU based diagonal parsing)
275 output = Transpose(output, shape(output), reverse_permutation);
276 } else {
277 // No transposing required
278 output = DiagonalInnermostDims(input, preserve_innermost_dim_val);
279 }
280
281 // Make copy of the output dims
282 MatShape output_dims = shape(output);
283
284 // Unsqueeze the reduced dim
285 auto iter = output_dims.begin() + second_dim;
286 output_dims.erase(iter);
287 output = output.reshape(1, output_dims);
288 return output;
289}

306 return -1;

510 // forward
511 void forward(InputArrayOfArrays inputs_arr,
512 OutputArrayOfArrays outputs_arr,
513 OutputArrayOfArrays internals_arr) CV_OVERRIDE
514 {
515 CV_TRACE_FUNCTION();
516 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
517 CV_CheckEQ((size_t)inputs_arr.total(), (size_t)numInputs, "Number of inputs in forward and inputs during graph construction do not match");
518
519 if (inputs_arr.depth() == CV_16F)
520 {
521 forward_fallback(inputs_arr, outputs_arr, internals_arr);
522 return;
523 }
524
525 // homogenize inputs
526 preProcessInputs(inputs_arr);
527
528 std::vector<cv::Mat> rawInputs, outputs;
529 inputs_arr.getMatVector(rawInputs);
530 outputs_arr.getMatVector(outputs);
531 Mat result;
532
533 // Pre-process the first input so as to reduce any dims that only it has
534 {
535 MatShape reducedDims;
536 MatShape preservedDims;
537 MatShape preservedShape;
538
539 reducedDims.reserve(numLetterIndices); // num_subscript_labels is the upper bound. No harm in over-reserving.
540 preservedDims.reserve(numLetterIndices); // num_subscript_labels is the upper bound. No harm in over-reserving.
541
542 for (size_t i = 0; i < numLetterIndices; ++i) {
543 if (subscriptIndicesToLastInput[i] == 0) {
544 reducedDims.push_back(i);
545 } else {
546 preservedDims.push_back(i);
547 }
548 }
549
550 // Reduce the dims that are last seen in the first input alone
551 if (reducedDims.size() != 0)
552 {
553 result = reduceSum((!preProcessedInputs[0].empty() ? preProcessedInputs[0] : rawInputs[0]), reducedDims);
554 } else {
555 // Check if there is a pre-processed version of this input
556 // If so assign it to result
557 if (!preProcessedInputs.empty() && !preProcessedInputs[0].empty())
558 {
559 result = preProcessedInputs[0];
560 }
561 }
562
563 // Finalize the output at this stage if num_inputs == 1
564 if (numInputs == 1) {
565 // Finalize the output by applying any transpose required to get
566 // it to the required output ordering and move it to the op's output
567 result = FinalizeOutput(!result.empty() ? result : rawInputs[0], preservedDims);
568 }
569 }
570
571
572 // Process the operands in a pair-wise fashion
573 {
574 bool isFinalPair = false;
575 // Keep processing each input pair-wise
576 for (int input = 1; input < numInputs; ++input) {
577 MatShape reducedDims;
578 reducedDims.reserve(numLetterIndices); // num_subscript_labels is the upper bound. No harm in over-reserving by a small margin.
579 for (int dim = 0; dim < numLetterIndices; ++dim)
580 {
581 if (subscriptIndicesToLastInput[dim] == input)
582 {
583 // This is the last input we are seeing this dimension (and it doesn't occur in the output), so reduce along the dimension
584 reducedDims.push_back(dim);
585 }
586 }
587
588 if (input == numInputs - 1)
589 isFinalPair = true;
590
591 // create temporary variable
592 MatShape tmpResult;
593 for (int i = 0; i < result.size.dims; i++)
594 tmpResult.emplace_back(result.size[i]);
595
596
597 // Use either the preprocessed inputs (if it is available) or the corresponding raw inputs
598 result = pairwiseOperandProcess(!result.empty() ? result : rawInputs[0],
599 !result.empty() ? tmpResult : homogenizedInputDims[0],
600 (!preProcessedInputs[input].empty()) ? preProcessedInputs[input] : rawInputs[input],
601 homogenizedInputDims[input],
602 reducedDims,
603 isFinalPair);
604 }
605 }
606
607 // check that product of output dimensions and computed output dimensions match
608 size_t reqProd = std::accumulate(einsumOutDims.begin(), einsumOutDims.end(), 1, std::multiplies<int>());
609 MatShape realOutputDims = shape(result);
610 size_t realProd = std::accumulate(realOutputDims.begin(), realOutputDims.end(), 1, std::multiplies<int>());
611
612 CV_CheckEQ(reqProd, realProd, "Real output can not be shaped in to required output");
613
614 // reduce dimensions
615 result = result.reshape(1, einsumOutDims.size(), einsumOutDims.data());
616 result.copyTo(outputs[0]);
617 } // forward

633Mat LayerEinsumImpl::reduceSum(Mat& src, MatShape& reduceAxis)
634{
635 // initialize ReduceLayer
636 LayerParams lp;
637 lp.set("reduce", "SUM");
638 int num_axes = reduceAxis.size();
639 lp.set("axes", DictValue::arrayInt(&reduceAxis[0] , num_axes));
640 reduce = ReduceLayer::create(lp);
641
642 // Compute output shapes
643 std::vector<MatShape> inputShapes{shape(src)};
644 std::vector<MatShape> outputShapes, internalShapes;
645 reduce->getMemoryShapes(inputShapes, 1, outputShapes, internalShapes);
646
647 int origType = src.type();
648 Mat src32 = src;
649 if (src.depth() != CV_32F)
650 src.convertTo(src32, CV_32F);
651 Mat output(outputShapes[0], CV_32F);
652
653 std::vector<Mat> inputs;
654 std::vector<Mat> outputs;
655 std::vector<Mat> internals;
656 inputs.emplace_back(src32);
657 outputs.emplace_back(output);
658
659 reduce->forward(inputs, outputs, internals);
660 Mat out = outputs[0];
661 if (CV_MAT_TYPE(origType) != CV_32F)
662 {
663 Mat converted;
664 out.convertTo(converted, origType);
665 return converted;
666 }
667 return out;
668}

670void LayerEinsumImpl::preProcessInputs(InputArrayOfArrays& inputs_arr)
671{
672 std::vector<cv::Mat> inputs;
673 inputs_arr.getMatVector(inputs);
674
675 preProcessedInputs.resize(inputs.size());
676 homogenizedInputDims.resize(inputs.size());
677
678 int inputIter = 0;
679 for(const Mat& input : inputs)
680 {
681 Mat preprocessed;
682
683 // variable to hold processed version of the original input
684 MatShape input_dims = shape(input);
685 const auto& currSubscriptIndices = inputSubscriptIndices[inputIter];
686
687 if (input_dims.empty() || currSubscriptIndices.empty()){
688 CV_CheckEQ(total(input_dims), (size_t)1, "Einsum: input with no subscript labels must be a scalar");
689 homogenizedInputDims[inputIter] = MatShape(numLetterIndices, 1);
690 ++inputIter;
691 continue;
692 }
693
694 // There should be subscript index (subscript label) for each dim of the input
695 CV_CheckEQ(input_dims.size(), currSubscriptIndices.size(),
696 "Rank of the input must match number of subscript labels corresponding to the input");
697
698 std::vector<int> subscriptIndicesToInputIndex(numLetterIndices, -1);
699 // this will hold input dims after reordering so that all inputs have
700 // same axes order
701 MatShape homogenizedInputDims_(numLetterIndices, 1);
702
703 int dimIndexInPreprocessedInput = 0;
704 int dimIndexInOriginalInput = 0;
705
706 for (const auto& subscriptIndex : currSubscriptIndices)
707 {
708 if(subscriptIndicesToInputIndex[subscriptIndex] == -1){
709 subscriptIndicesToInputIndex[subscriptIndex] = dimIndexInPreprocessedInput++;
710 homogenizedInputDims_[subscriptIndex] = input_dims[dimIndexInOriginalInput];
711 } else {
712 // Call diagonal
713 preprocessed = Diagonal(
714 !preprocessed.empty() ? preprocessed : inputs[inputIter],
715 subscriptIndicesToInputIndex[subscriptIndex],
716 dimIndexInPreprocessedInput);
717 }
718 ++dimIndexInOriginalInput;
719 }
720
721 std::vector<size_t> permutation;
722 for(auto& d : subscriptIndicesToInputIndex)
723 {
724 if (d != -1)
725 permutation.emplace_back(d);
726 }
727
728 if (IsTransposeRequired(
729 !preprocessed.empty() ? preprocessed.size.dims : inputs[inputIter].size.dims,
730 permutation))
731 {
732 // call transpose
733 preprocessed = Transpose(
734 !preprocessed.empty() ? preprocessed : inputs[inputIter],
735 !preprocessed.empty() ? shape(preprocessed) : shape(inputs[inputIter]),
736 permutation);
737 }
738
739 if (!preprocessed.empty())
740 {
741 preprocessed = preprocessed.reshape(1, homogenizedInputDims_.size(), homogenizedInputDims_.data());
742 }
743 preProcessedInputs[inputIter] = preprocessed;
744 homogenizedInputDims[inputIter] = homogenizedInputDims_;
745
746 ++inputIter;
747 }
748}

750void LayerEinsumImpl::parseEquation(String equation)
751{
752 // remove white spaces in the copy
753 equation.erase(std::remove_if(equation.begin(), equation.end(), ::isspace), equation.end());
754
755 // check if '->' - the output subscript label is present in the equation;
756 std::size_t arrow_idx = equation.find("->");
757 if (arrow_idx != std::string::npos)
758 {
759 // split left- and right-hand sides of the equation
760 lhs_eq = equation.substr(0, arrow_idx);
761 rhs_eq = equation.substr(arrow_idx + 2);
762 explicitEquation = true;
763 } else {
764 lhs_eq = equation;
765 }
766
767 // split lhs_eq by ',' - comma and put all created token - splits
768 // into lhs_eq_tokens vector
769 // the implementation does not ignore empty tokens and trailing comma
770 size_t start = 0;
771 while(start < lhs_eq.size())
772 {
773 size_t comma = lhs_eq.find(',', start);
774 if (comma != std::string::npos)
775 {
776 std::string token = lhs_eq.substr(start, comma-start);
777 lhs_eq_tokens.push_back(token);
778 start = comma+1;
779 }
780 else
781 {
782 std::string token = lhs_eq.substr(start);
783 lhs_eq_tokens.push_back(token);
784 start = lhs_eq.size()+1;
785 }
786 }
787
788 // trailing comma without token
789 if (lhs_eq[lhs_eq.size()-1] == ',')
790 lhs_eq_tokens.push_back(std::string());
791
792}

795void LayerEinsumImpl::calculateOutputShape()
796{
797 // Traverse through each of the subscript labels within the output subscript.
798 bool middleOfEllipsis = false;
799 int ellipsisCharCount = 0;
800
801 subscriptIndicesToOutputIndices.resize(numLetterIndices, -1);
802
803 std::array<int, numOfLetters> outputLetterToCount;
804 outputLetterToCount.fill(0);
805
806 int outputDimCounter = 0;
807 for (auto letter : rhs_eq)
808 {
809 if(letter == '.')
810 {
811 middleOfEllipsis = true;
812 // Make sure there aren't more than 3 '.'s in the current subscript
813 if (++ellipsisCharCount > 3) {
814 CV_Error(Error::StsError, "Found a '.' not part of an ellipsis in the output subscript provided");
815 }
816
817 if (ellipsisCharCount == 3) { // Ellipsis is complete. Process it.
818 middleOfEllipsis = false;
819 for (size_t i = 0; i < numOfEllipsisDims; ++i) {
820 einsumOutDims.emplace_back(subscriptIndicesToDimValue[i]);
821 // The ellipsis is seen in the output and hence the corresponding dims are to not be reduced
822 subscriptIndicesToLastInput[i] = -1;
823 subscriptIndicesToOutputIndices[i] = outputDimCounter++;
824 }
825 }
826 } else {
827 CV_CheckEQ(middleOfEllipsis, false,
828 "Encountered '.' character that is not part of output subscript");
829
830 auto letterIndex = letterToIndex(letter);
831
832 CV_CheckNE(letterIndex, -1,
833 "The only permissible subscript labels are lowercase letters (a-z) and uppercase letters (A-Z).");
834 CV_CheckEQ(outputLetterToCount[letterIndex], 0,
835 "Output subscript contains repeated letters");
836
837 ++outputLetterToCount[letterIndex];
838 auto mappedIndex = letter2index[letterIndex];
839
840 CV_CheckNE(mappedIndex, -1,
841 "Output subscript has letters that were not encountered in the inputs");
842
843 // Push output dimension
844 // Einsum layer only has one output vector
845 einsumOutDims.emplace_back(subscriptIndicesToDimValue[mappedIndex]);
846
847 // Reset the last input index for this subscript label
848 // given that it is seen in the output and hence can't be reduced
849 subscriptIndicesToLastInput[mappedIndex] = -1;
850 subscriptIndicesToOutputIndices[mappedIndex] = outputDimCounter++;
851 }
852 }
853 if (rhs_eq.empty()) {
854 einsumOutDims = MatShape(0, 0); // handle scalar output case
855 }
856}

876void LayerEinsumImpl::processBroadcastedDims()
877{
878 // Only compute this function if ellipsis "..." was found in the equation
879 if (numOfEllipsisDims > 0)
880 {
881 // extend the number of subscript labels to include each ellipsis dim as
882 // theoretically each ellipsis dim does correspond to a "virtual" subscript label
883 numLetterIndices += numOfEllipsisDims;
884
885 // We are going to assign the broadcasted dims outermost subscript indices (i.e.) 0 -> numOfEllipsisDims - 1
886 // as most likely bradcasted dims will be batch dimensions (i.e.) outermost dimensions and hence we don't have to pay
887 // transposing while "homogenizing" the input
888
889 // Hence offset all subscript indices by numOfEllipsisDims
890 for (size_t i = 0; i < numOfLetters; ++i){
891 if (letter2count[i] != -1){
892 letter2index[i] += numOfEllipsisDims;
893 }
894 }
895
896 std::vector<int> tempIndex2LastInput(numLetterIndices, -1);
897 for (int i = 0; i < subscriptIndicesToLastInput.size(); ++i){
898 tempIndex2LastInput[i + numOfEllipsisDims] = subscriptIndicesToLastInput[i];
899 }
900 subscriptIndicesToLastInput = std::move(tempIndex2LastInput);
901
902 std::vector<int> tempIndexToDimValue(numLetterIndices, -1);
903 for (int i = 0; i < subscriptIndicesToDimValue.size(); ++i){
904 tempIndexToDimValue[i + numOfEllipsisDims] = subscriptIndicesToDimValue[i];
905 }
906 subscriptIndicesToDimValue = std::move(tempIndexToDimValue);
907
908 for (size_t i = 0; i < inputSubscriptIndices.size(); ++i)
909 {
910 auto& currentInputDimIndicesToSubscriptIndices = inputSubscriptIndices[i];
911 std::vector<int> tempCurrentInputDimIndicesToSubscriptIndices;
912 tempCurrentInputDimIndicesToSubscriptIndices.reserve(currentInputDimIndicesToSubscriptIndices.size());
913
914 // make sure it is correct
915 const auto& dims = einsumInpShapes[i];
916 auto rank = dims.size();
917
918 size_t dimIter = 0;
919 size_t numBroadcastedIndices = 0;
920 while (dimIter < currentInputDimIndicesToSubscriptIndices.size())
921 {
922 auto value = currentInputDimIndicesToSubscriptIndices[dimIter];
923 if (value == numOfLetters)
924 { // This is a broadcasted dim
925 // Shouldn't hit this error - just a sanity check
926 CV_Assert(numBroadcastedIndices < numOfEllipsisDims);
927 tempCurrentInputDimIndicesToSubscriptIndices.push_back(static_cast<int>(numBroadcastedIndices));
928 subscriptIndicesToLastInput[numBroadcastedIndices] = i;
929
930 // This is the first time we are seeing this broadcasted dim
931 if (subscriptIndicesToDimValue[numBroadcastedIndices] == -1)
932 {
933 subscriptIndicesToDimValue[numBroadcastedIndices] = dims[dimIter];
934 } else { // We have seen this broadcasted dim before
935 // Check if the previous value is equal to the current value
936 if (subscriptIndicesToDimValue[numBroadcastedIndices] != dims[dimIter])
937 {
938 // If they are not equal, one of them needs to be 1
939 if (subscriptIndicesToDimValue[numBroadcastedIndices] == 1)
940 {
941 subscriptIndicesToDimValue[numBroadcastedIndices] = dims[dimIter];
942 } else {
943 CV_CheckEQ(dims[dimIter], 1, "The broadcasted dimensions of the inputs are incompatible");
944 }
945 }
946 }
947 ++numBroadcastedIndices;
948 } else { // This is a regular dim - offset it by number of broadcasted dims
949 tempCurrentInputDimIndicesToSubscriptIndices.push_back(value + static_cast<int>(numOfEllipsisDims));
950 }
951 ++dimIter;
952 }
953 // Shouldn't hit this error - just a sanity check
954 CV_Assert(dimIter == rank);
955 currentInputDimIndicesToSubscriptIndices = std::move(tempCurrentInputDimIndicesToSubscriptIndices);
956 }
957 }
958}

961
962void LayerEinsumImpl::processEquation(const std::vector<MatShape>& inputs)
963{
964
965 // fill in the einsumInpShapes
966 for (const auto& input : inputs) {
967 einsumInpShapes.emplace_back(input);
968 }
969
970
971 numInputs = inputs.size();
972 inputSubscriptIndices.reserve(numInputs);
973 // Check if number of tokens in equal to number of inputs.
974 // For install "ij, jk -> ik" needs to have 2 inputs tensors
975 int num_input_tensors = inputs.size();
976 if (lhs_eq_tokens.empty() || (lhs_eq == ",") ) {
977 inputSubscriptIndices.resize(numInputs);
978 return;
979 }
980 // if we have only one token and two inputs lets skip the check
981 if (lhs_eq_tokens.size() > 1)
982 CV_CheckEQ(static_cast<int>(lhs_eq_tokens.size()), num_input_tensors,
983 "Number of input tensors does not match the number of subscripts in the input equation");
984
985 int inputIdx = 0;
986 for (const auto& token : lhs_eq_tokens)
987 {
988 const MatShape shape = inputs[inputIdx];
989 size_t rank = shape.size();
990 size_t dim_count = 0;
991
992 std::vector<int> currTokenIndices;
993 currTokenIndices.reserve(rank);
994
995 // Variable to deal with "ellipsis" - '...' in the input
996 bool middleOfellipsis = false;
997 int ellipsisCharCount = 0;
998 for (auto letter : token)
999 {
1000 if (letter == '.')
1001 {
1002 middleOfellipsis = true;
1003
1004 // there should not be more than 3 '.'s in the current subscript
1005 if (++ellipsisCharCount > 3)
1006 {
1007 CV_Error(Error::StsError, cv::format("Found a '.' not part of an ellipsis in input: %d", inputIdx));
1008 }
1009
1010 // We have seen all 3 '.'s. We can safely process the ellipsis now.
1011 if (ellipsisCharCount == 3)
1012 {
1013 middleOfellipsis = false;
1014
1015 // Example for the following line of code
1016 // Subscript "...ij" for an input of rank 6
1017 // numOfEllipsisDims = 6 - 5 + 3 = 4
1018 int currentNumOfEllipsisDims = static_cast<int>(rank) - token.length() + 3;
1019 CV_CheckGE(currentNumOfEllipsisDims, 0,
1020 "Einsum subscripts string contains too many subscript labels when compared to the rank of the input");
1021
1022 // Theoretically, currentNumOfEllipsisDims could be 0
1023 // Example: For an input of rank 2 paired with a subscript "...ij"
1024 if (currentNumOfEllipsisDims != 0)
1025 {
1026 // We have seen a ellipsis before - make sure ranks align as per the ONNX spec -
1027 // "Ellipsis must indicate a fixed number of dimensions."
1028 if (numOfEllipsisDims != 0){
1029 CV_CheckEQ(numOfEllipsisDims, static_cast<size_t>(currentNumOfEllipsisDims),
1030 "Ellipsis must indicate a fixed number of dimensions across all inputs");
1031 } else {
1032 numOfEllipsisDims = static_cast<size_t>(currentNumOfEllipsisDims);
1033 }
1034
1035 // We reserve 'numOfLetters' for broadcasted dims as we only allow 'a' - 'z'
1036 // and 'A' - 'Z' (0 - 51) for non-broadcasted dims.
1037 // We will assign appropriate indices (based on number of dimensions the ellipsis corresponds to)
1038 // during broadcasting related post-processing.
1039 for (size_t i = 0; i < numOfEllipsisDims; ++i){
1040 currTokenIndices.push_back(numOfLetters);
1041 }
1042
1043 // Offset 'dim_count' by number of dimensions the ellipsis corresponds to
1044 dim_count += numOfEllipsisDims;
1045 }
1046 }
1047 } else {
1048 if (middleOfellipsis){
1049 CV_Error(Error::StsAssert,
1050 cv::format(
1051 "Encountered '.' character that is not part of an ellipsis in the input: [%d]",
1052 inputIdx));
1053 }
1054
1055 int letterIdx = letterToIndex(letter);
1056 CV_CheckNE(letterIdx, -1,
1057 "The only permissible subscript labels are lowercase letters (a-z) and uppercase letters (A-Z).");
1058
1059 CV_CheckLT(dim_count, rank,
1060 "The Einsum subscripts string has an excessive number of subscript labels compared to the rank of the input.");
1061 int dimValue = shape[dim_count];
1062
1063 // The subscript label was not found in the global subscript label array
1064 // Therefore, it is added to both the local and global subscript arrays
1065 if(letter2count[letterIdx] == 0){
1066 letter2index[letterIdx] = numLetterIndices++;
1067 subscriptIndicesToDimValue.push_back(dimValue);
1068 subscriptIndicesToLastInput.push_back(inputIdx);
1069
1070 } else {
1071 // This letter has been seen in at least one other operand's subscript
1072 // It must be equal unless one of them is a 1 (Numpy allows this)
1073 auto mappedIndx = letter2index[letterIdx];
1074 subscriptIndicesToLastInput[mappedIndx] = inputIdx;
1075
1076 if (subscriptIndicesToDimValue[mappedIndx] != dimValue) {
1077 if (dimValue != 1) {
1078 CV_Error(Error::StsError, cv::format("Einsum operands can not be broadcasted."
1079 "Check input shapes/equation passed."
1080 "Input shape of operand [%d]", inputIdx) +
1081 cv::format(" is incompatible in the dimension [%zu].", static_cast<size_t>(dim_count)));
1082 }
1083 }
1084 }
1085 ++letter2count[letterIdx];
1086 currTokenIndices.push_back(letter2index[letterIdx]);
1087
1088 ++dim_count;
1089 }
1090 }
1091
1092 // When no broadcasting is requested, the number of subscript labels (dim_counter) should match the input's rank.
1093 CV_Assert(!(numOfEllipsisDims == 0 && dim_count != rank)
1094 && "The Einsum subscripts string does not contain required amount of subscript labels and no ellipsis is provided in the input.");
1095
1096 inputSubscriptIndices.emplace_back(std::move(currTokenIndices));
1097 ++inputIdx;
1098 }
1099}

1101Mat LayerEinsumImpl::FinalizeOutput(
1102 const Mat& candidateOutput,
1103 const MatShape& ordered_subscript_indices_in_candidate)
1104{
1105 const std::vector<int>& subscript_indices_to_output_indices = subscriptIndicesToOutputIndices;
1106 const auto output_dims = einsumOutDims;
1107
1108 const auto output_rank = output_dims.size();
1109
1110 // MatShape output_shape = output_dims;
1111 // CV_CheckEQ((int) candidateOutput.dims, (int) output_shape.size(),
1112 // "Einsum op: The candidate output cannot be reshaped into the op's output");
1113
1114 const MatShape candidate_output_dims = MatShape(candidateOutput.size.p, candidateOutput.size.p + candidateOutput.dims);
1115 const int candidate_output_rank = candidate_output_dims.size();
1116
1117 // This vector holds the shape of the candidate_output after removing the dims that have
1118 // been reduced in the final output
1119 MatShape candidate_output_shape_without_reduced_dims;
1120 candidate_output_shape_without_reduced_dims.reserve(candidate_output_rank); // reserve upper bound
1121
1122 // Identify the permutation required by the op's output
1123 std::vector<size_t> output_permutation;
1124 output_permutation.resize(output_rank, 0);
1125 size_t output_iter = 0;
1126
1127
1128 for (size_t iter = 0, end = ordered_subscript_indices_in_candidate.size(); iter < end; ++iter)
1129 {
1130 auto output_index = subscript_indices_to_output_indices[ordered_subscript_indices_in_candidate[iter]];
1131
1132 // If output_index is -1, then this dimension does not show up in the op's output and has been reduced along the way
1133 if (output_index != -1)
1134 {
1135 output_permutation[output_index] = output_iter++;
1136 candidate_output_shape_without_reduced_dims.push_back(candidate_output_dims[iter]);
1137 } else {
1138 // This dim doesn't show up in the op's output and hence we check if the dim has been reduced in the candidate output
1139 CV_CheckEQ(candidate_output_dims[iter], 1,
1140 "Not all dimensions to be reduced have been reduced in the candidate output. Candidate output dims: "); //%d", candidateOutput.size));
1141 }
1142 }
1143
1144 // Transpose to the required final output order
1145 // (Identify no-op transposes and prevent triggering the transpose)
1146
1147 if (IsTransposeRequired(candidate_output_shape_without_reduced_dims.size(), output_permutation))
1148 {
1149 auto candidate_output_transposed = Transpose(
1150 candidateOutput,
1151 candidate_output_shape_without_reduced_dims,
1152 output_permutation);
1153 return candidate_output_transposed;
1154 }
1155 return candidateOutput;
1156}

1158Mat LayerEinsumImpl::pairwiseOperandProcess(
1159 const Mat& left,
1160 const MatShape& leftShapeOverride,
1161 const Mat& right,
1162 const MatShape& rightShapeOverride,
1163 const MatShape& reduceDims,
1164 bool isFinalPair
1165)
1166{
1167 size_t matDimSize = left.total();
1168 size_t overrideDimSize = total(leftShapeOverride);
1169
1170 CV_CheckEQ(matDimSize, overrideDimSize, "Override dims are not compatible with left tensor shape");
1171
1172 matDimSize = right.total();
1173 overrideDimSize = total(rightShapeOverride);
1174
1175 CV_CheckEQ(matDimSize, overrideDimSize, "Override dims are not compatible with right tensor shape");
1176
1177 // Make copy as this may be overridden downstream
1178 const auto& leftDims = leftShapeOverride;
1179 const auto& rightDims = rightShapeOverride;
1180
1181 int leftRank = static_cast<int>(leftDims.size());
1182 int rightRank = static_cast<int>(rightDims.size());
1183
1184 Mat currentLeft;
1185 Mat currentRight;
1186
1187 CV_CheckEQ(leftRank, rightRank, "Ranks of pair-wise operands must be equal");
1188
1189 // Following vectors hold:
1190 // lro: dim indices that are present in left, right, and reduce_dims
1191 // lo: dim indices that are present in left and reduce_dims
1192 // ro: dim indices that are present in right and reduce_dims
1193 std::vector<size_t> lro;
1194 lro.reserve(5); // Reserve an arbitrary amount of space for this vector (not bound to see a tensor of rank > kTensorShapeSmallBufferElementsSize)
1195
1196 std::vector<size_t> lo;
1197 lo.reserve(5); // Reserve an arbitrary amount of space for this vector (not bound to see a tensor of rank > kTensorShapeSmallBufferElementsSize)
1198
1199 std::vector<size_t> ro;
1200 ro.reserve(5); // Reserve an arbitrary amount of space for this vector (not bound to see a tensor of rank > kTensorShapeSmallBufferElementsSize)
1201
1202 // Maintain sizes to create reshaped "views"
1203 int lro_size = 1;
1204 int lo_size = 1;
1205 int ro_size = 1;
1206 int reduced_size = 1;
1207
1208 size_t reduceDimsIter = 0;
1209 size_t reduceDimsSize = reduceDims.size();
1210
1211 for (int i = 0; i < leftRank; ++i)
1212 {
1213 int leftDim = leftDims[i];
1214 int rightDim = rightDims[i];
1215
1216 bool hasLeftDim = leftDim > 1; // non-trivial dimension (dim_value != 1)
1217 bool hasRightDim = rightDim > 1; // non-trivial dimension (dim_value != 1)
1218
1219 if (reduceDimsIter < reduceDimsSize && reduceDims[reduceDimsIter] == i)
1220 {
1221 // This dimension is to be reduced after this pair-wise operation
1222 ++reduceDimsIter;
1223 if (hasLeftDim && hasRightDim){
1224 // Both inputs have non-trivial dim values along this dimension
1225 // Both the left and right operands have non-trivial dimension value along this axis
1226 CV_CheckEQ(leftDim, rightDim, "Einsum op: Input dimensions must be equal along an axis to be reduced across all inputs");
1227 reduced_size *= leftDim;
1228
1229 } else if (hasLeftDim){
1230 // if the dim to be reduced is only in one of left and right, we can reduce right away
1231 Mat tensorToReduce = !currentLeft.empty() ? currentLeft : left;
1232 MatShape shapeToReduce = !currentLeft.empty() ? shape(currentLeft) : leftDims;
1233 currentLeft = reduceSum(tensorToReduce, shapeToReduce);
1234
1235 } else if (hasRightDim){
1236 Mat tensorToReduce = !currentRight.empty() ? currentRight : right;
1237 MatShape shapeToReduce = !currentRight.empty() ? shape(currentRight) : rightDims;
1238 currentLeft = reduceSum(tensorToReduce, shapeToReduce);
1239 }
1240
1241 } else {
1242 // This dimension is not reduced (i.e.) it appears in the output after processing these 2 operands
1243 // Both the left and right operands have non-trivial dimension value along this axis
1244 // They must be equal
1245 if (hasLeftDim && hasRightDim){
1246 CV_CheckEQ(leftDim, rightDim, "Input shapes do not align");
1247 lro.push_back(i);
1248 lro_size *= leftDim;
1249
1250 } else if (hasLeftDim) {
1251 // The left operand has non-trivial dimension value
1252 lo.push_back(i);
1253 lo_size *= leftDim;
1254
1255 } else {
1256 // The right operand may or may not have non-trivial dim value
1257 // If it has trivial dim value (1),
1258 // it will just form a trailing dimension for the right operand
1259 ro.push_back(i);
1260 ro_size *= rightDim;
1261 }
1262 }
1263 }
1264
1265
1266 // Permute the left operand so that the axes order go like this: [lro, lo, reduce_dims, ro]
1267 MatShape reshaped_dims;
1268 std::vector<size_t> left_permutation;
1269 left_permutation.reserve(lro.size() + lo.size() + reduceDims.size() + ro.size());
1270 left_permutation.insert(left_permutation.end(), lro.begin(), lro.end());
1271 left_permutation.insert(left_permutation.end(), lo.begin(), lo.end());
1272 // left_permutation.insert(left_permutation.end(), reduce_dims.begin(), reduce_dims.end());
1273
1274 for (auto& a : reduceDims)
1275 {
1276 left_permutation.push_back(a);
1277 }
1278 left_permutation.insert(left_permutation.end(), ro.begin(), ro.end());
1279
1280 if (IsTransposeRequired(!currentLeft.empty() ? currentLeft.dims : leftDims.size(),
1281 left_permutation))
1282 {
1283 if (!currentLeft.empty() && IsTransposeReshapeForEinsum(left_permutation,
1284 shape(currentLeft),
1285 reshaped_dims))
1286 {
1287 // This can be done because current_* tensors (if they exist) and output tensors are
1288 // intermediate tensors and cannot be input tensors to the Einsum node itself
1289 // (which are immutable).
1290 currentLeft = currentLeft.reshape(1, reshaped_dims.size(), reshaped_dims.data());
1291 } else {
1292 // Covered by ExplicitEinsumAsTensorContraction, DiagonalWithMatmul, ...
1293 currentLeft = Transpose(!currentLeft.empty() ? currentLeft: left,
1294 !currentLeft.empty() ? shape(currentLeft) : leftDims,
1295 left_permutation);
1296 }
1297 }
1298
1299 // Permute the right operand so that the axes order go like this: [lro, reduce_dims, ro, lo]
1300 std::vector<size_t> right_permutation;
1301 right_permutation.reserve(lro.size() + lo.size() + reduceDims.size() + ro.size());
1302 right_permutation.insert(right_permutation.end(), lro.begin(), lro.end());
1303 // right_permutation.insert(right_permutation.end(), reduce_dims.begin(), reduce_dims.end());
1304 for (auto& a : reduceDims) {
1305 right_permutation.push_back(a);
1306 }
1307 right_permutation.insert(right_permutation.end(), ro.begin(), ro.end());
1308 right_permutation.insert(right_permutation.end(), lo.begin(), lo.end());
1309
1310 if (IsTransposeRequired(!currentRight.empty() ? currentRight.dims: rightDims.size(),
1311 right_permutation))
1312 {
1313 if (!currentRight.empty() && IsTransposeReshapeForEinsum(right_permutation,
1314 shape(currentRight),
1315 reshaped_dims))
1316 {
1317 currentRight = currentRight.reshape(1, reshaped_dims.size(), reshaped_dims.data());
1318 } else {
1319 currentRight = Transpose(!currentRight.empty() ? currentRight : right,
1320 !currentRight.empty() ? shape(currentRight) : rightDims,
1321 right_permutation);
1322 }
1323 }
1324
1325 // Calculate output size
1326 // Output shape will be determined by rules of MatMul:
1327 // because we are multiplying two tensors of shapes [lro, lo, reduce_dims] , [lro, reduce_dims, ro]
1328 // [dim_value of `lro` dims,
1329 // dim_value of `lo` dims,
1330 // `1` for each of the `reduce_dims`,
1331 // dim_value of `ro` dims]
1332 MatShape outputDims;
1333 outputDims.reserve(lro.size() + lo.size() + reduceDims.size() + ro.size());
1334 for (size_t i = 0; i < lro.size(); ++i)
1335 {
1336 outputDims.emplace_back(leftDims[lro[i]]);
1337 }
1338
1339 for (size_t i = 0; i < lo.size(); ++i)
1340 {
1341 outputDims.emplace_back(leftDims[lo[i]]);
1342 }
1343
1344 for (size_t i = 0; i < reduceDims.size(); ++i)
1345 {
1346 outputDims.emplace_back(1); // reduced dimensions will have a value 1 in it
1347 }
1348
1349 for (size_t i = 0; i < ro.size(); ++i) {
1350 outputDims.emplace_back(rightDims[ro[i]]);
1351 }
1352
1353 MatShape currentSubscriptOrder;
1354 // Calculate output permutation
1355 // After the MatMul op, the because the two operands have been permutated,
1356 // the output is permutated as well with respect to the original ordering of the axes.
1357 // The permutated order will be the dims in: [lro, lo, reduced_dims, ro]
1358 // Hence invert the permutation by a permutation that puts the axes in the same ordering
1359 std::vector<size_t> outputPermutation;
1360 if (!isFinalPair) { // If this is not the final pair, we need to permutate the result to match the pre-fixed order for the next iteration
1361 outputPermutation.resize(lro.size() + lo.size() + reduceDims.size() + ro.size(), 0);
1362 size_t iter = 0;
1363 for (size_t i = 0; i < lro.size(); ++i)
1364 {
1365 outputPermutation[lro[i]] = iter++;
1366 }
1367
1368 for (size_t i = 0; i < lo.size(); ++i)
1369 {
1370 outputPermutation[lo[i]] = iter++;
1371 }
1372
1373 for (size_t i = 0; i < reduceDims.size(); ++i)
1374 {
1375 outputPermutation[reduceDims[i]] = iter++;
1376 }
1377
1378 for (size_t i = 0; i < ro.size(); ++i)
1379 {
1380 outputPermutation[ro[i]] = iter++;
1381 }
1382
1383 } else {
1384 currentSubscriptOrder.reserve(lro.size() + lo.size() + reduceDims.size() + ro.size());
1385 currentSubscriptOrder.insert(currentSubscriptOrder.end(), lro.begin(), lro.end());
1386 currentSubscriptOrder.insert(currentSubscriptOrder.end(), lo.begin(), lo.end());
1387 currentSubscriptOrder.insert(currentSubscriptOrder.end(), reduceDims.begin(), reduceDims.end());
1388 currentSubscriptOrder.insert(currentSubscriptOrder.end(), ro.begin(), ro.end());
1389 }
1390
1391 Mat output = batchwiseMatMul(
1392 !currentLeft.empty() ? currentLeft : left,
1393 MatShape({static_cast<int>(lro_size), static_cast<int>(lo_size), static_cast<int>(reduced_size)}),
1394 !currentRight.empty() ? currentRight : right,
1395 MatShape({static_cast<int>(lro_size), static_cast<int>(reduced_size), static_cast<int>(ro_size)})
1396 );
1397
1398 //reshape
1399 output = output.reshape(1, outputDims.size(), outputDims.data());
1400
1401 if (!isFinalPair)
1402 { // This is not the final pair - so bring the axes order to what the inputs conformed to
1403 if (IsTransposeRequired(outputDims.size(), outputPermutation))
1404 {
1405 if (IsTransposeReshapeForEinsum(outputPermutation,
1406 outputDims,
1407 reshaped_dims))
1408 {
1409 // See note following the previous call of function IsTransposeReshapeForEinsum.
1410 // Covered by ExplicitEinsumAsTensorContractionReshapeFinal.
1411 output = output.reshape(1, reshaped_dims.size(), reshaped_dims.data());
1412 }
1413 else {
1414 output = Transpose(
1415 output,
1416 outputDims,
1417 outputPermutation);
1418 }
1419 }
1420 } else { // This is the final pair - Transpose directly to the output ordering required and copy the contents to the op's output
1421 // not sure if this finalize shape is needed at all
1422 output = FinalizeOutput(output, currentSubscriptOrder);
1423 }
1424 return output;
1425};
