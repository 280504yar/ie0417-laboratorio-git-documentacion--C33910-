13int solve_deg2(double a, double b, double c, double & x1, double & x2)
14{
15 double delta = b * b - 4 * a * c;
16
17 if (delta < 0) return 0;
18
19 double inv_2a = 0.5 / a;
20
21 if (delta == 0) {
22 x1 = -b * inv_2a;
23 x2 = x1;
24 return 1;
25 }
26
27 double sqrt_delta = sqrt(delta);
28 x1 = (-b + sqrt_delta) * inv_2a;
29 x2 = (-b - sqrt_delta) * inv_2a;
30 return 2;
31}

37int solve_deg3(double a, double b, double c, double d,
38 double & x0, double & x1, double & x2)
39{
40 if (a == 0) {
41 // Solve second order system
42 if (b == 0) {
43 // Solve first order system
44 if (c == 0)
45 return 0;
46
47 x0 = -d / c;
48 return 1;
49 }
50
51 x2 = 0;
52 return solve_deg2(b, c, d, x0, x1);
53 }
54
55 // Calculate the normalized form x^3 + a2 * x^2 + a1 * x + a0 = 0
56 double inv_a = 1. / a;
57 double b_a = inv_a * b, b_a2 = b_a * b_a;
58 double c_a = inv_a * c;
59 double d_a = inv_a * d;
60
61 // Solve the cubic equation
62 double Q = (3 * c_a - b_a2) / 9;
63 double R = (9 * b_a * c_a - 27 * d_a - 2 * b_a * b_a2) / 54;
64 double Q3 = Q * Q * Q;
65 double D = Q3 + R * R;
66 double b_a_3 = (1. / 3.) * b_a;
67
68 if (Q == 0) {
69 if(R == 0) {
70 x0 = x1 = x2 = - b_a_3;
71 return 3;
72 }
73 else {
74 double cube_root = cv::cubeRoot(2 * R);
75 x0 = cube_root - b_a_3;
76 return 1;
77 }
78 }
79
80 if (D <= 0) {
81 // Three real roots
82 double theta = std::acos(R / std::sqrt(-Q3));
83 double sqrt_Q = std::sqrt(-Q);
84 x0 = 2 * sqrt_Q * std::cos(theta / 3.0) - b_a_3;
85 x1 = 2 * sqrt_Q * std::cos((theta + 2 * CV_PI)/ 3.0) - b_a_3;
86 x2 = 2 * sqrt_Q * std::cos((theta + 4 * CV_PI)/ 3.0) - b_a_3;
87
88 return 3;
89 }
90
91 // D > 0, only one real root
92 double AD = 0.;
93 double BD = 0.;
94 double R_abs = fabs(R);
95 if (R_abs > DBL_EPSILON)
96 {
97 AD = cv::cubeRoot(R_abs + sqrt(D));
98 AD = (R >= 0) ? AD : -AD;
99 BD = -Q / AD;
100 }
101
102 // Calculate the only real root
103 x0 = AD + BD - b_a_3;
104
105 return 1;
106}

111int solve_deg4(double a, double b, double c, double d, double e,
112 double & x0, double & x1, double & x2, double & x3)
113{
114 if (a == 0) {
115 x3 = 0;
116 return solve_deg3(b, c, d, e, x0, x1, x2);
117 }
118
119 // Normalize coefficients
120 double inv_a = 1. / a;
121 b *= inv_a; c *= inv_a; d *= inv_a; e *= inv_a;
122 double b2 = b * b, bc = b * c, b3 = b2 * b;
123
124 // Solve resultant cubic
125 double r0, r1, r2;
126 int n = solve_deg3(1, -c, d * b - 4 * e, 4 * c * e - d * d - b2 * e, r0, r1, r2);
127 if (n == 0) return 0;
128
129 // Calculate R^2
130 double R2 = 0.25 * b2 - c + r0, R;
131 if (R2 < 0)
132 return 0;
133
134 R = sqrt(R2);
135 double inv_R = 1. / R;
136
137 int nb_real_roots = 0;
138
139 // Calculate D^2 and E^2
140 double D2, E2;
141 if (R < 10E-12) {
142 double temp = r0 * r0 - 4 * e;
143 if (temp < 0)
144 D2 = E2 = -1;
145 else {
146 double sqrt_temp = sqrt(temp);
147 D2 = 0.75 * b2 - 2 * c + 2 * sqrt_temp;
148 E2 = D2 - 4 * sqrt_temp;
149 }
150 }
151 else {
152 double u = 0.75 * b2 - 2 * c - R2,
153 v = 0.25 * inv_R * (4 * bc - 8 * d - b3);
154 D2 = u + v;
155 E2 = u - v;
156 }
157
158 double b_4 = 0.25 * b, R_2 = 0.5 * R;
159 if (D2 >= 0) {
160 double D = sqrt(D2);
161 nb_real_roots = 2;
162 double D_2 = 0.5 * D;
163 x0 = R_2 + D_2 - b_4;
164 x1 = x0 - D;
165 }
166
167 // Calculate E^2
168 if (E2 >= 0) {
169 double E = sqrt(E2);
170 double E_2 = 0.5 * E;
171 if (nb_real_roots == 0) {
172 x0 = - R_2 + E_2 - b_4;
173 x1 = x0 - E;
174 nb_real_roots = 2;
175 }
176 else {
177 x2 = - R_2 + E_2 - b_4;
178 x3 = x2 - E;
179 nb_real_roots = 4;
180 }
181 }
182
183 return nb_real_roots;
184}
