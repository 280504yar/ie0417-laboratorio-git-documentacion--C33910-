18 TileLayerImpl(const LayerParams& params)

43 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
44 const int requiredOutputs,
45 std::vector<MatShape> &outputs,
46 std::vector<MatShape> &internals) const CV_OVERRIDE
47 {
48 CV_CheckEQ(inputs.size(), 1ull, "Tile: one input is expected");
49 int nrepeats = (int)repeats.size();
50
51 // repeats must have the same length as input's dimension number
52 if (inputs[0].size() > 1) {
53 CV_CheckEQ(inputs[0].size(), repeats.size(), "Tile: repeats must be a 1D tensor of the same length as input's dimension number");
54 outputs.assign(1, inputs[0]);
55 for (int i = 0; i < nrepeats; i++)
56 {
57 outputs[0][i] *= repeats[i];
58 }
59 } else {
60 CV_CheckGE(nrepeats, 1, "Tile: Provide at least one repeat along any dimension");
61 outputs.assign(1, MatShape(repeats));
62 if (inputs[0].size() == 1)
63 outputs[0][nrepeats - 1] *= inputs[0][0];
64 }
65
66 return false;
67 }

78 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
79 {
80 CV_TRACE_FUNCTION();
81 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
82
83 std::vector<Mat> inputs, outputs;
84 inputs_arr.getMatVector(inputs);
85 outputs_arr.getMatVector(outputs);
86
87 const Mat& data = inputs[0];
88 Mat& out = outputs[0];
89
90 Mat tmp = data.clone();
91 MatShape out_shape = shape(out);
92 int rep_i, ndims = data.dims;
93 int dims = 1;
94 if (ndims > 1){
95 for (int i = 0; i < ndims; i++)
96 {
97 rep_i = repeats[i];
98 if (rep_i != 1)
99 {
100 tmp = tmp.reshape(0, dims);
101 tmp = cv::repeat(tmp, 1, rep_i);
102 }
103 dims *= out_shape[i];
104 }
105 tmp = tmp.reshape(0, out_shape);
106 } else {
107 for (int i = 0; i < repeats.size(); i++){
108 tmp = tmp.reshape(0, dims);
109 tmp = cv::repeat(tmp, repeats[i], 1);
110 dims *= out_shape[i];
111 }
112 tmp = tmp.reshape(0, out_shape);
113 }
114 tmp.copyTo(out);
115 }
