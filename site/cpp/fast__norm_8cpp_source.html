13// Sum and squared sum of n floats with single-precision accumulators, tail processed in scalar.
14static inline void normAccumSumSqSum(const float* x, size_t n, float& sum, float& sqsum) {
15 const size_t VEC_SZ = (size_t)VTraits<v_float32>::vlanes();
16 v_float32 vsum0 = vx_setzero_f32(), vsum1 = vx_setzero_f32(),
17 vsqsum0 = vx_setzero_f32(), vsqsum1 = vx_setzero_f32();
18 size_t j = 0;
19 for (; j + 2 * VEC_SZ <= n; j += 2 * VEC_SZ) {
20 v_float32 v0 = vx_load(x + j), v1 = vx_load(x + j + VEC_SZ);
21 vsum0 = v_add(vsum0, v0);
22 vsum1 = v_add(vsum1, v1);
23 vsqsum0 = v_fma(v0, v0, vsqsum0);
24 vsqsum1 = v_fma(v1, v1, vsqsum1);
25 }
26 if (j + VEC_SZ <= n) {
27 v_float32 v0 = vx_load(x + j);
28 vsum0 = v_add(vsum0, v0);
29 vsqsum0 = v_fma(v0, v0, vsqsum0);
30 j += VEC_SZ;
31 }
32 float s = v_reduce_sum(v_add(vsum0, vsum1));
33 float sq = v_reduce_sum(v_add(vsqsum0, vsqsum1));
34 for (; j < n; j++) {
35 float v = x[j];
36 s += v;
37 sq += v * v;
38 }
39 sum = s;
40 sqsum = sq;
41}
42#endif

45// Same as normAccumSumSqSum but with double-precision accumulators (used where the scalar
46// reference accumulates in double).
47static inline void normAccumSumSqSum64f(const float* x, size_t n, double& sum, double& sqsum) {
48 const size_t VEC_SZ = (size_t)VTraits<v_float32>::vlanes();
49 v_float64 vsum_lo = vx_setzero_f64(), vsum_hi = vx_setzero_f64(),
50 vsqsum_lo = vx_setzero_f64(), vsqsum_hi = vx_setzero_f64();
51 size_t j = 0;
52 for (; j + VEC_SZ <= n; j += VEC_SZ) {
53 v_float32 v = vx_load(x + j);
54 v_float64 vlo = v_cvt_f64(v), vhi = v_cvt_f64_high(v);
55 vsum_lo = v_add(vsum_lo, vlo);
56 vsum_hi = v_add(vsum_hi, vhi);
57 vsqsum_lo = v_fma(vlo, vlo, vsqsum_lo);
58 vsqsum_hi = v_fma(vhi, vhi, vsqsum_hi);
59 }
60 double s = v_reduce_sum(v_add(vsum_lo, vsum_hi));
61 double sq = v_reduce_sum(v_add(vsqsum_lo, vsqsum_hi));
62 for (; j < n; j++) {
63 double v = (double)x[j];
64 s += v;
65 sq += v * v;
66 }
67 sum = s;
68 sqsum = sq;
69}

72// Unchanged: only mvn_layer.cpp uses this overload, out of scope here.
73void fastNorm(const Mat &input, Mat &output, float epsilon, size_t normalized_axis, bool normalize_variance) {
74 const auto input_shape = shape(input);
75 CV_CheckLT(normalized_axis, input_shape.size(), "fastNorm: axis out of range");
76
77 size_t loops = static_cast<size_t>(total(input_shape, 0, static_cast<int>(normalized_axis))),
78 norm_size = static_cast<size_t>(total(input_shape, static_cast<int>(normalized_axis)));
79 float inv_norm_size = 1.0 / norm_size;
80
81 auto fn = [&](const Range &r) {
82 const auto *input_data = input.ptr<const float>();
83 auto *output_data = output.ptr<float>();
84 for (int i = r.start; i < r.end; i++) {
85 const auto *x = input_data + norm_size * i;
86 auto *y = output_data + norm_size * i;
87
88 float mean = 0.f, mean_square = 0.f;
89#if (CV_SIMD || CV_SIMD_SCALABLE)
90 normAccumSumSqSum(x, norm_size, mean, mean_square);
91#else
92 for (int j = 0; j < norm_size; j++) {
93 float v = x[j];
94 mean += v;
95 mean_square += v * v;
96 }
97#endif
98
99 mean *= inv_norm_size;
100 mean_square = std::sqrt(std::max(0.f, mean_square * inv_norm_size - mean * mean) + epsilon);
101 float inv_stdev = normalize_variance ? 1.f / mean_square : 1.f;
102
103 size_t j = 0;
104#if (CV_SIMD || CV_SIMD_SCALABLE)
105 const size_t VEC_SZ = (size_t)VTraits<v_float32>::vlanes();
106 v_float32 vmean = vx_setall_f32(mean), vinv_stdev = vx_setall_f32(inv_stdev);
107 for (; j + VEC_SZ <= norm_size; j += VEC_SZ)
108 vx_store(y + j, v_mul(v_sub(vx_load(x + j), vmean), vinv_stdev));
109#endif
110 for (; j < norm_size; j++) {
111 y[j] = (x[j] - mean) * inv_stdev;
112 }
113 }
114 };
115 double nstripes = loops * norm_size * (1 / 1024.0);
116 parallel_for_(Range(0, loops), fn, nstripes);
117}

119// Templated on T so CV_64F gets a genuine double accumulator, not a narrowed float one.
120template<typename T>
121static void fastNormMeanInvStdDevImpl(const Mat& input, Mat& mean, Mat& invStdDev, T epsilon, size_t normalized_axis)
122{
123 CV_Assert(input.isContinuous() && mean.isContinuous() && invStdDev.isContinuous());
124
125 const auto input_shape = shape(input);
126 CV_CheckLT(normalized_axis, input_shape.size(), "fastNormMeanInvStdDev: axis out of range");
127
128 const size_t loops = static_cast<size_t>(total(input_shape, 0, static_cast<int>(normalized_axis)));
129 const size_t norm_size = static_cast<size_t>(total(input_shape, static_cast<int>(normalized_axis)));
130 const T inv_norm_size = (T)1 / (T)norm_size;
131
132 CV_CheckEQ((size_t)mean.total(), loops, "fastNormMeanInvStdDev: mean output size mismatch");
133 CV_CheckEQ((size_t)invStdDev.total(), loops, "fastNormMeanInvStdDev: invStdDev output size mismatch");
134
135 auto fn = [&](const Range& r) {
136 const T* input_data = input.ptr<T>();
137 T* mean_data = mean.ptr<T>();
138 T* invstd_data = invStdDev.ptr<T>();
139 for (int i = r.start; i < r.end; ++i)
140 {
141 const T* x = input_data + norm_size * (size_t)i;
142 T m = 0, mean_square = 0;
143 bool simd_done = false;
144 if constexpr (std::is_same<T, float>::value) {
145#if (CV_SIMD || CV_SIMD_SCALABLE)
146 normAccumSumSqSum(x, norm_size, m, mean_square);
147 simd_done = true;
148#endif
149 }
150 if (!simd_done) {
151 for (size_t j = 0; j < norm_size; ++j)
152 {
153 T v = x[j];
154 m += v;
155 mean_square += v * v;
156 }
157 }
158 m *= inv_norm_size;
159 const T var = std::max((T)0, mean_square * inv_norm_size - m * m);
160 const T stdev = std::sqrt(var + epsilon);
161 mean_data[i] = m;
162 invstd_data[i] = (T)1 / stdev;
163 }
164 };
165
166 const double nstripes = loops * norm_size * (1 / 1024.0);
167 parallel_for_(Range(0, (int)loops), fn, nstripes);
168}

184template<typename T>
185static void fastNormImpl(const Mat &input, const Mat &scale, Mat &output, T epsilon, size_t normalized_axis, bool recenter) {
186 const auto input_shape = shape(input);
187 CV_CheckLT(normalized_axis, input_shape.size(), "fastNorm: axis out of range");
188
189 size_t loops = static_cast<size_t>(total(input_shape, 0, static_cast<int>(normalized_axis))),
190 norm_size = static_cast<size_t>(total(input_shape, static_cast<int>(normalized_axis)));
191 T inv_norm_size = (T)1 / (T)norm_size;
192
193 auto fn = [&](const Range &r) {
194 const T *input_data = input.ptr<const T>();
195 const T *scale_data = scale.ptr<const T>();
196 T *output_data = output.ptr<T>();
197 for (int i = r.start; i < r.end; i++) {
198 const T *x = input_data + norm_size * i;
199 T *y = output_data + norm_size * i;
200
201 T mean = 0, mean_square = 0;
202 bool simd_done = false;
203 if constexpr (std::is_same<T, float>::value) {
204#if (CV_SIMD || CV_SIMD_SCALABLE)
205 normAccumSumSqSum(x, norm_size, mean, mean_square);
206 if (!recenter)
207 mean = 0.f;
208 simd_done = true;
209#endif
210 }
211 if (!simd_done) {
212 for (int j = 0; j < norm_size; j++) {
213 T v = x[j];
214 if (recenter)
215 mean += v;
216 mean_square += v * v;
217 }
218 }
219
220 mean *= inv_norm_size;
221 mean_square = std::sqrt(std::max((T)0, mean_square * inv_norm_size - mean * mean) + epsilon);
222 T inv_stdev = (T)1 / mean_square;
223
224 size_t j = 0;
225 if constexpr (std::is_same<T, float>::value) {
226#if (CV_SIMD || CV_SIMD_SCALABLE)
227 const size_t VEC_SZ = (size_t)VTraits<v_float32>::vlanes();
228 v_float32 vmean = vx_setall_f32(mean), vinv_stdev = vx_setall_f32(inv_stdev);
229 for (; j + VEC_SZ <= norm_size; j += VEC_SZ) {
230 v_float32 vs = vx_load(scale_data + j);
231 vx_store(y + j, v_mul(v_mul(vs, v_sub(vx_load(x + j), vmean)), vinv_stdev));
232 }
233#endif
234 }
235 for (; j < norm_size; j++) {
236 y[j] = scale_data[j] * (x[j] - mean) * inv_stdev;
237 }
238 }
239 };
240 double nstripes = loops * norm_size * (1 / 1024.0);
241 parallel_for_(Range(0, loops), fn, nstripes);
242}

256// Full LayerNorm (scale + bias) -- LayerNorm and LayerNorm2's with-bias path.
257template<typename T>
258static void fastNormImpl(const Mat &input, const Mat &scale, const Mat &bias, Mat &output, T epsilon, size_t normalized_axis) {
259 const auto input_shape = shape(input);
260 CV_CheckLT(normalized_axis, input_shape.size(), "fastNorm: axis out of range");
261 CV_CheckEQ(scale.total(), bias.total(), "fastNorm: scale and bias should have the same shape");
262
263 size_t loops = static_cast<size_t>(total(input_shape, 0, static_cast<int>(normalized_axis))),
264 norm_size = static_cast<size_t>(total(input_shape, static_cast<int>(normalized_axis)));
265 T inv_norm_size = (T)1 / (T)norm_size;
266
267 auto fn = [&](const Range &r) {
268 const T *input_data = input.ptr<const T>();
269 const T *scale_data = scale.ptr<const T>();
270 const T *bias_data = bias.ptr<const T>();
271 T *output_data = output.ptr<T>();
272 for (int i = r.start; i < r.end; i++) {
273 const T *x = input_data + norm_size * i;
274 T *y = output_data + norm_size * i;
275
276 T mean = 0, mean_square = 0;
277 bool simd_done = false;
278 if constexpr (std::is_same<T, float>::value) {
279#if (CV_SIMD || CV_SIMD_SCALABLE)
280 normAccumSumSqSum(x, norm_size, mean, mean_square);
281 simd_done = true;
282#endif
283 }
284 if (!simd_done) {
285 for (int j = 0; j < norm_size; j++) {
286 T v = x[j];
287 mean += v;
288 mean_square += v * v;
289 }
290 }
291
292 mean *= inv_norm_size;
293 mean_square = std::sqrt(std::max((T)0, mean_square * inv_norm_size - mean * mean) + epsilon);
294 T inv_stdev = (T)1 / mean_square;
295
296 size_t j = 0;
297 if constexpr (std::is_same<T, float>::value) {
298#if (CV_SIMD || CV_SIMD_SCALABLE)
299 const size_t VEC_SZ = (size_t)VTraits<v_float32>::vlanes();
300 v_float32 vmean = vx_setall_f32(mean), vinv_stdev = vx_setall_f32(inv_stdev);
301 for (; j + VEC_SZ <= norm_size; j += VEC_SZ) {
302 v_float32 vs = vx_load(scale_data + j);
303 v_float32 vb = vx_load(bias_data + j);
304 vx_store(y + j, v_fma(v_mul(vs, v_sub(vx_load(x + j), vmean)), vinv_stdev, vb));
305 }
306#endif
307 }
308 for (; j < norm_size; j++) {
309 y[j] = scale_data[j] * (x[j] - mean) * inv_stdev + bias_data[j];
310 }
311 }
312 };
313 double nstripes = loops * norm_size * (1 / 1024.0);
314 parallel_for_(Range(0, loops), fn, nstripes);
315}

330// InstanceNorm, BLOCK layout, CV_32F -- extracted verbatim, unchanged.
331static void fastNormChannelBlockF32(const Mat &input, const Mat &scale, const Mat &bias, Mat &output, float epsilon) {
332 const auto input_shape = shape(input);
333 size_t C = (size_t)input_shape.C;
334
335 CV_Assert(input.dims == 5 && output.dims == 5);
336 CV_Assert(input.isContinuous() && output.isContinuous());
337
338 const int N = input.size[0];
339 const int C1 = input.size[1];
340 const int H = input.size[2];
341 const int W = input.size[3];
342 const int C0 = input.size[4];
343 const int Ci = (int)C;
344
345 const float* scale_data = scale.ptr<float>();
346 const float* bias_data = bias.ptr<float>();
347
348 const size_t inStep0 = input.step.p[0] / sizeof(float);
349 const size_t inStep1 = input.step.p[1] / sizeof(float);
350 const size_t inStep2 = input.step.p[2] / sizeof(float);
351 const size_t inStep3 = input.step.p[3] / sizeof(float);
352
353 const size_t outStep0 = output.step.p[0] / sizeof(float);
354 const size_t outStep1 = output.step.p[1] / sizeof(float);
355 const size_t outStep2 = output.step.p[2] / sizeof(float);
356 const size_t outStep3 = output.step.p[3] / sizeof(float);
357
358 const size_t norm_size = (size_t)H * (size_t)W;
359
360#if (CV_SIMD || CV_SIMD_SCALABLE)
361 const int VEC_SZ = VTraits<v_float32>::vlanes();
362#endif
363
364 // Accumulators are double-precision
365 const double inv_norm_size_d = 1.0 / (double)norm_size;
366
367 parallel_for_(Range(0, N * C1), [&](const Range& r) {
368 const float* inptr0 = (const float*)input.data;
369 float* outptr0 = (float*)output.data;
370
371 AutoBuffer<double> sumBuf(C0 * 2);
372 double* sum = sumBuf.data();
373 double* sqsum = sum + C0;
374 AutoBuffer<float> abBuf(C0 * 2);
375 float* alpha = abBuf.data();
376 float* beta = alpha + C0;
377
378 for (int i = r.start; i < r.end; ++i) {
379 int n = i / C1;
380 int c1 = i - n * C1;
381 int cbase = c1 * C0;
382 int validC0 = std::min(C0, std::max(0, Ci - cbase));
383
384 const float* inbase = inptr0 + n * inStep0 + c1 * inStep1;
385 float* outbase = outptr0 + n * outStep0 + c1 * outStep1;
386
387 int c0 = 0;
388#if CV_SIMD_64F || CV_SIMD_SCALABLE_64F
389 const int VEC_SZ_D = VTraits<v_float64>::vlanes();
390 CV_DbgAssert(VEC_SZ == 2 * VEC_SZ_D);
391 for (; c0 <= validC0 - VEC_SZ; c0 += VEC_SZ) {
392 v_float64 vsum_lo = vx_setzero_f64(), vsum_hi = vx_setzero_f64();
393 v_float64 vsqsum_lo = vx_setzero_f64(), vsqsum_hi = vx_setzero_f64();
394 for (int h = 0; h < H; ++h) {
395 const float* inrow = inbase + h * inStep2;
396 for (int w = 0; w < W; ++w) {
397 v_float32 v = vx_load(inrow + w * inStep3 + c0);
398 v_float64 vlo = v_cvt_f64(v);
399 v_float64 vhi = v_cvt_f64_high(v);
400 vsum_lo = v_add(vsum_lo, vlo);
401 vsum_hi = v_add(vsum_hi, vhi);
402 vsqsum_lo = v_fma(vlo, vlo, vsqsum_lo);
403 vsqsum_hi = v_fma(vhi, vhi, vsqsum_hi);
404 }
405 }
406 vx_store(sum + c0, vsum_lo);
407 vx_store(sum + c0 + VEC_SZ_D, vsum_hi);
408 vx_store(sqsum + c0, vsqsum_lo);
409 vx_store(sqsum + c0 + VEC_SZ_D, vsqsum_hi);
410 }
411#endif
412 for (; c0 < validC0; ++c0) {
413 double s = 0., sq = 0.;
414 for (int h = 0; h < H; ++h) {
415 const float* inrow = inbase + h * inStep2;
416 for (int w = 0; w < W; ++w) {
417 double v = (double)inrow[w * inStep3 + c0];
418 s += v;
419 sq += v * v;
420 }
421 }
422 sum[c0] = s;
423 sqsum[c0] = sq;
424 }
425
426 for (int c = 0; c < validC0; ++c) {
427 double mean = sum[c] * inv_norm_size_d;
428 double var = std::max(0., sqsum[c] * inv_norm_size_d - mean * mean);
429 float inv_stdev = 1.f / std::sqrt((float)var + epsilon);
430 alpha[c] = scale_data[cbase + c] * inv_stdev;
431 beta[c] = bias_data[cbase + c] - alpha[c] * (float)mean;
432 }
433
434 c0 = 0;
435#if (CV_SIMD || CV_SIMD_SCALABLE)
436 for (; c0 <= validC0 - VEC_SZ; c0 += VEC_SZ) {
437 v_float32 va = vx_load(alpha + c0);
438 v_float32 vb = vx_load(beta + c0);
439 for (int h = 0; h < H; ++h) {
440 const float* inrow = inbase + h * inStep2;
441 float* outrow = outbase + h * outStep2;
442 for (int w = 0; w < W; ++w) {
443 v_float32 vin = vx_load(inrow + w * inStep3 + c0);
444 vx_store(outrow + w * outStep3 + c0, v_fma(vin, va, vb));
445 }
446 }
447 }
448#endif
449 for (; c0 < validC0; ++c0) {
450 float a = alpha[c0], b = beta[c0];
451 for (int h = 0; h < H; ++h) {
452 const float* inrow = inbase + h * inStep2;
453 float* outrow = outbase + h * outStep2;
454 for (int w = 0; w < W; ++w)
455 outrow[w * outStep3 + c0] = inrow[w * inStep3 + c0] * a + b;
456 }
457 }
458
459 int c0_pad = validC0;
460#if (CV_SIMD || CV_SIMD_SCALABLE)
461 for (; c0_pad <= C0 - VEC_SZ; c0_pad += VEC_SZ) {
462 v_float32 vzero = vx_setzero_f32();
463 for (int h = 0; h < H; ++h) {
464 float* outrow = outbase + h * outStep2;
465 for (int w = 0; w < W; ++w)
466 vx_store(outrow + w * outStep3 + c0_pad, vzero);
467 }
468 }
469#endif
470 for (; c0_pad < C0; ++c0_pad)
471 for (int h = 0; h < H; ++h) {
472 float* outrow = outbase + h * outStep2;
473 for (int w = 0; w < W; ++w)
474 outrow[w * outStep3 + c0_pad] = 0.f;
475 }
476 }
477 });
478}

480// InstanceNorm, BLOCK layout, CV_64F -- scalar counterpart to fastNormChannelBlockF32.
481template<typename T>
482static void fastNormChannelBlockT(const Mat &input, const Mat &scale, const Mat &bias, Mat &output, T epsilon) {
483 const auto input_shape = shape(input);
484 size_t C = (size_t)input_shape.C;
485
486 CV_Assert(input.dims == 5 && output.dims == 5);
487 CV_Assert(input.isContinuous() && output.isContinuous());
488
489 const int N = input.size[0];
490 const int C1 = input.size[1];
491 const int H = input.size[2];
492 const int W = input.size[3];
493 const int C0 = input.size[4];
494 const int Ci = (int)C;
495
496 const T* scale_data = scale.ptr<T>();
497 const T* bias_data = bias.ptr<T>();
498
499 const size_t inStep0 = input.step.p[0] / sizeof(T);
500 const size_t inStep1 = input.step.p[1] / sizeof(T);
501 const size_t inStep2 = input.step.p[2] / sizeof(T);
502 const size_t inStep3 = input.step.p[3] / sizeof(T);
503
504 const size_t outStep0 = output.step.p[0] / sizeof(T);
505 const size_t outStep1 = output.step.p[1] / sizeof(T);
506 const size_t outStep2 = output.step.p[2] / sizeof(T);
507 const size_t outStep3 = output.step.p[3] / sizeof(T);
508
509 const size_t norm_size = (size_t)H * (size_t)W;

576static void fastNormChannelImpl(const Mat &input, const Mat &scale, const Mat &bias, Mat &output, T epsilon) {
577 const auto input_shape = shape(input);
578 size_t C = input_shape[1];
579
580 size_t N = input_shape[0];
581 CV_CheckGE(input.dims, 3, "fastNormChannel: input dimension >= 3");

583 size_t loops = N * C,
584 norm_size = static_cast<size_t>(total(input_shape, 2));
585
586 auto fn = [&](const Range &r) {

589 const T *bias_data = bias.ptr<const T>();
590 T *output_data = output.ptr<T>();
591 for (int i = r.start; i < r.end; i++) {

593 T *y = output_data + norm_size * i;

595 double dmean = 0., dmean_sq = 0.;
596 bool simd_done = false;
597 if constexpr (std::is_same<T, float>::value) {

600 simd_done = true;
601#endif
602 }
603 if (!simd_done) {

607 dmean_sq += v * v;
608 }
609 }

891 size_t loops = N * num_groups;

902 const T *x = input_data + norm_size * i;
903 T *y = output_data + norm_size * i;
904
905 double dmean = 0., dmean_sq = 0.;

916 dmean += v;
917 dmean_sq += v * v;

942 }
943 }
