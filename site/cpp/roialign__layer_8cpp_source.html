32template<typename T, bool MaxMode>
33class RoiAlignForwardInvoker CV_FINAL : public ParallelLoopBody
34{
35public:
36 RoiAlignForwardInvoker(const Mat& X_, Mat& Y_,
37 const Mat& rois_, const Mat& batch_indices_,
38 const int* Xsize,
39 int pooled_h_, int pooled_w_, int sampling_ratio_,
40 float spatial_scale_, float offset_,
41 bool clampMalformedRoi_)
42 : X(X_), Y(Y_), rois(rois_), batch_indices(batch_indices_),
43 output_height(pooled_h_), output_width(pooled_w_), sampling_ratio(sampling_ratio_),
44 spatial_scale(spatial_scale_), offset(offset_),
45 clampMalformedRoi(clampMalformedRoi_)
46 {
47 CV_Assert(X.isContinuous() && Y.isContinuous());
48 CV_Assert(rois.isContinuous() && batch_indices.isContinuous());
49 CV_Assert(Xsize);
50 N = Xsize[0]; C = Xsize[1]; H = Xsize[2]; W = Xsize[3];
51
52 Xdata = X.ptr<T>();
53 Ydata = Y.ptr<T>();
54
55 spatialSize = static_cast<size_t>(H) * W;
56 channelStride = spatialSize;
57 batchStride = static_cast<size_t>(C) * spatialSize;
58 outSpatial = output_height * output_width;
59
60 Rdepth = rois.depth();
61
62 const int bdepth = batch_indices.depth();
63 if (bdepth == CV_32S)
64 {
65 batch32 = batch_indices.ptr<int>();
66 }
67 else if (bdepth == CV_64S)
68 {
69 batch64 = batch_indices.ptr<int64>();
70 }
71 else
72 {
73 CV_Error(Error::StsUnsupportedFormat, "Unsupported batch_indices depth (expected int32 or int64)");
74 }
75 }

77 void operator()(const Range& range) const CV_OVERRIDE
78 {
79 for (int r = range.start; r < range.end; ++r)
80 {
81 const int b = batch64 ? static_cast<int>(batch64[r]) : batch32[r];
82 CV_Assert(0 <= b && b < N);
83
84 float x1, y1, x2, y2;
85 if (Rdepth == CV_32F)
86 readScaledRoiCoords<float>(rois, r, spatial_scale, offset, x1, y1, x2, y2);
87 else if (Rdepth == CV_64F)
88 readScaledRoiCoords<double>(rois, r, spatial_scale, offset, x1, y1, x2, y2);
89 else if (Rdepth == CV_16F)
90 readScaledRoiCoords<hfloat>(rois, r, spatial_scale, offset, x1, y1, x2, y2);
91 else if (Rdepth == CV_16BF)
92 readScaledRoiCoords<bfloat>(rois, r, spatial_scale, offset, x1, y1, x2, y2);
93 else
94 CV_Error(Error::StsUnsupportedFormat, "Unsupported rois depth");
95
96 float roi_width = x2 - x1;
97 float roi_height = y2 - y1;
98
99 if (clampMalformedRoi)
100 {
101 roi_width = std::max(roi_width, 1.f);
102 roi_height = std::max(roi_height, 1.f);
103 }
104
105 const float bin_size_w = roi_width / output_width;
106 const float bin_size_h = roi_height / output_height;
107
108 const int gw = std::max((sampling_ratio > 0) ? sampling_ratio
109 : cvCeil(roi_width / output_width), 1);
110 const int gh = std::max((sampling_ratio > 0) ? sampling_ratio
111 : cvCeil(roi_height / output_height), 1);
112 const int sample_count_i = gw * gh;
113 const float inv_sample_count = 1.f / static_cast<float>(sample_count_i);
114 const float step_y = bin_size_h / gh;
115 const float step_x = bin_size_w / gw;
116
117 AutoBuffer<BilinearSample> samples(static_cast<size_t>(outSpatial) * sample_count_i);
118 for (int outIdx = 0; outIdx < outSpatial; ++outIdx)
119 {
120 const int ph = outIdx / output_width;
121 const int pw = outIdx - ph * output_width;
122 const float ph_off = ph * bin_size_h;
123 const float pw_off = pw * bin_size_w;
124
125 BilinearSample* dst = samples.data() + static_cast<size_t>(outIdx) * sample_count_i;
126 for (int s = 0; s < sample_count_i; ++s)
127 {
128 const int iy = s / gw;
129 const int ix = s - iy * gw;
130 const float yy = y1 + ph_off + (iy + 0.5f) * step_y;
131 const float xx = x1 + pw_off + (ix + 0.5f) * step_x;
132 dst[s] = precomputeBilinearSample(H, W, yy, xx);
133 }
134 }
135
136 const size_t y_roi_base = static_cast<size_t>(r) * static_cast<size_t>(C) * outSpatial;
137 const size_t x_batch_base = static_cast<size_t>(b) * batchStride;
138
139 for (int outPlane = 0; outPlane < C * outSpatial; ++outPlane)
140 {
141 const int c = outPlane / outSpatial;
142 const int outIdx = outPlane - c * outSpatial;
143
144 const T* img = Xdata + x_batch_base + static_cast<size_t>(c) * channelStride;
145 const BilinearSample* src = samples.data() + static_cast<size_t>(outIdx) * sample_count_i;
146
147 float outv = MaxMode ? -FLT_MAX : 0.f;
148 for (int s = 0; s < sample_count_i; ++s)
149 {
150 const BilinearSample& bs = src[s];
151 const float v00 = static_cast<float>(img[bs.idx00]);
152 const float v01 = static_cast<float>(img[bs.idx01]);
153 const float v10 = static_cast<float>(img[bs.idx10]);
154 const float v11 = static_cast<float>(img[bs.idx11]);
155
156 const float v = MaxMode
157 ? std::max(std::max(bs.w1 * v00, bs.w2 * v01), std::max(bs.w3 * v10, bs.w4 * v11))
158 : (bs.w1 * v00 + bs.w2 * v01 + bs.w3 * v10 + bs.w4 * v11);
159
160 outv = MaxMode ? std::max(outv, v) : (outv + v);
161 }
162
163 outv = MaxMode ? ((outv == -FLT_MAX) ? 0.f : outv) : (outv * inv_sample_count);
164
165 Ydata[y_roi_base + static_cast<size_t>(outPlane)] = saturate_cast<T>(outv);
166 }
167 }
168 }

171 struct BilinearSample
172 {
173 size_t idx00 = 0, idx01 = 0, idx10 = 0, idx11 = 0;
174 float w1 = 0.f, w2 = 0.f, w3 = 0.f, w4 = 0.f;
175 };

177 template<typename TRoi>
178 static inline void readScaledRoiCoords(const Mat& rois, int r, float spatial_scale, float offset,
179 float& x1, float& y1, float& x2, float& y2)
180 {
181 const TRoi* p = rois.ptr<TRoi>(r);
182 x1 = static_cast<float>(p[0]) * spatial_scale - offset;
183 y1 = static_cast<float>(p[1]) * spatial_scale - offset;
184 x2 = static_cast<float>(p[2]) * spatial_scale - offset;
185 y2 = static_cast<float>(p[3]) * spatial_scale - offset;
186 }
187
188 static inline BilinearSample precomputeBilinearSample(int height, int width, float y, float x)
189 {
190 BilinearSample s;
191 if (y < -1.f || y > height || x < -1.f || x > width)
192 return s;
193
194 y = std::min(std::max(y, 0.f), static_cast<float>(height - 1));
195 x = std::min(std::max(x, 0.f), static_cast<float>(width - 1));
196
197 const int y_low = static_cast<int>(y);
198 const int x_low = static_cast<int>(x);
199 const int y_high = std::min(y_low + 1, height - 1);
200 const int x_high = std::min(x_low + 1, width - 1);
201
202 const float ly = y - y_low;
203 const float lx = x - x_low;
204 const float hy = 1.f - ly;
205 const float hx = 1.f - lx;
206
207 s.w1 = hy * hx;
208 s.w2 = hy * lx;
209 s.w3 = ly * hx;
210 s.w4 = ly * lx;
211
212 s.idx00 = static_cast<size_t>(y_low) * width + x_low;
213 s.idx01 = static_cast<size_t>(y_low) * width + x_high;
214 s.idx10 = static_cast<size_t>(y_high) * width + x_low;
215 s.idx11 = static_cast<size_t>(y_high) * width + x_high;
216
217 return s;
218 }

241 const int64* batch64 = nullptr;

319 OutputArrayOfArrays outputs_arr,

321 {
322 CV_TRACE_FUNCTION();

324
325 std::vector<Mat> inputs, outputs;
326 inputs_arr.getMatVector(inputs);
327 outputs_arr.getMatVector(outputs);
328
329 CV_Assert(inputs.size() == 3);
330 CV_Assert(outputs.size() == 1);
331
332 const Mat& X = inputs[0];
333 Mat rois = inputs[1];
334 Mat batch_indices = inputs[2];
335 Mat& Y = outputs[0];
336
337 CV_Assert(X.dims == 4);
338 const int C = X.size[1];
339
340 CV_Assert(X.depth() == CV_32F || X.depth() == CV_64F || X.depth() == CV_16F || X.depth() == CV_16BF);
341 CV_Assert(rois.depth() == CV_32F || rois.depth() == CV_64F || rois.depth() == CV_16F || rois.depth() == CV_16BF);
342 CV_Assert(batch_indices.depth() == CV_32S || batch_indices.depth() == CV_64S);
343
344 CV_Assert(rois.total() % 4 == 0);
345 const int num_rois = static_cast<int>(rois.total() / 4);
346 rois = rois.reshape(1, num_rois);
347 batch_indices = batch_indices.reshape(1, static_cast<int>(batch_indices.total()));
348
349 CV_Assert(rois.rows == num_rois && rois.cols == 4);
350 CV_Assert(static_cast<int>(batch_indices.total()) == num_rois);
351
352 CV_Assert(Y.dims == 4);
353 CV_Assert(Y.size[0] == num_rois && Y.size[1] == C && Y.size[2] == output_height && Y.size[3] == output_width);
354
355 const bool isMaxMode = (mode_ == RoiAlignMode::MAX);
356 const float offset = (coord_mode_ == CoordTransformMode::HALF_PIXEL) ? 0.5f : 0.0f;
357 const bool clampMalformedRoi = (coord_mode_ != CoordTransformMode::HALF_PIXEL);

361 {
362 if (isMaxMode)
363 parallel_for_(Range(0, num_rois), RoiAlignForwardInvoker<float, true>(X, Y, rois, batch_indices, Xsize, output_height, output_width, sampling_ratio, spatial_scale, offset, clampMalformedRoi));
