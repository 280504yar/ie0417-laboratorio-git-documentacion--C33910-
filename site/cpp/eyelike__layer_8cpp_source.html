51 std::vector<MatType>& outputs,
52 std::vector<MatType>& internals) const CV_OVERRIDE
53 {
54 CV_Assert(!inputs.empty());

60 void forward(InputArrayOfArrays inputs_arr,
61 OutputArrayOfArrays outputs_arr,
62 OutputArrayOfArrays /*internals_arr*/) CV_OVERRIDE
63 {
64 std::vector<Mat> inputs, outputs;
65 inputs_arr.getMatVector(inputs);
66 outputs_arr.getMatVector(outputs);
67
68 CV_Assert(inputs.size() == 1);
69 const Mat& X = inputs[0];
70 CV_Assert(X.dims == 2);
71
72 int rows = X.size[0];
73 int cols = X.size[1];
74
75 int outType = (outputDtype >= 0) ? outputDtype : X.type();
76 outputs[0].create(rows, cols, outType);
77 Mat& Y = outputs[0];
78 Y.setTo(Scalar::all(0));
79
80 // Set ones on the k-th diagonal: Y[i, i+k] = 1
81 int iStart = (k >= 0) ? 0 : -k;
82 int jStart = (k >= 0) ? k : 0;
83 int diagLen = std::min(rows - iStart, cols - jStart);
84
85 switch (outType)
86 {
87 case CV_32F: fillDiag<float> (Y, iStart, jStart, diagLen, 1.0f); break;
88 case CV_64F: fillDiag<double> (Y, iStart, jStart, diagLen, 1.0); break;
89 case CV_32S: fillDiag<int32_t> (Y, iStart, jStart, diagLen, 1); break;
90 case CV_64S: fillDiag<int64_t> (Y, iStart, jStart, diagLen, 1); break;
91 case CV_8U: fillDiag<uint8_t> (Y, iStart, jStart, diagLen, 1); break;
92 case CV_8S: fillDiag<int8_t> (Y, iStart, jStart, diagLen, 1); break;
93 case CV_16U: fillDiag<uint16_t>(Y, iStart, jStart, diagLen, 1); break;
94 case CV_16S: fillDiag<int16_t> (Y, iStart, jStart, diagLen, 1); break;
95 case CV_16F: fillDiag<hfloat> (Y, iStart, jStart, diagLen, hfloat(1.0f)); break;
96 case CV_16BF: fillDiag<bfloat> (Y, iStart, jStart, diagLen, bfloat(1.0f)); break;
97 case CV_Bool: fillDiag<bool> (Y, iStart, jStart, diagLen, true); break;
98 default:
99 CV_Error(Error::BadDepth, "Unsupported output depth for EyeLikeLayer");
100 }
101 }
102
103private:
104 template<typename T>
105 static void fillDiag(Mat& Y, int iStart, int jStart, int diagLen, T one)
106 {
107 for (int d = 0; d < diagLen; ++d)
108 Y.at<T>(iStart + d, jStart + d) = one;
109 }
110};
