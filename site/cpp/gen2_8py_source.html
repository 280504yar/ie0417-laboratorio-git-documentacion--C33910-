543def find_argument_class_info(argument_type, function_namespace,
544 function_class_name, known_classes):
545 # type: (str, str, str, dict[str, ClassInfo]) -> ClassInfo | None
546 """Tries to find corresponding class info for the provided argument type
547
548 Args:
549 argument_type (str): Function argument type
550 function_namespace (str): Namespace of the function declaration
551 function_class_name (str): Name of the class if function is a method of class
552 known_classes (dict[str, ClassInfo]): Mapping between string class
553 identifier and ClassInfo struct.
554
555 Returns:
556 Optional[ClassInfo]: class info struct if the provided argument type
557 refers to a known C++ class, None otherwise.
558 """
559
560 possible_classes = tuple(filter(lambda cls: cls.endswith(argument_type), known_classes))
561 # If argument type is not a known class - just skip it
562 if not possible_classes:
563 return None
564 if len(possible_classes) == 1:
565 return known_classes[possible_classes[0]]
566
567 # If there is more than 1 matched class, try to select the most probable one
568 # Look for a matched class name in different scope, starting from the
569 # narrowest one
570
571 # First try to find argument inside class scope of the function (if any)
572 if function_class_name:
573 type_to_match = function_class_name + '_' + argument_type
574 if type_to_match in possible_classes:
575 return known_classes[type_to_match]
576 else:
577 type_to_match = argument_type
578
579 # Trying to find argument type in the namespace of the function
580 type_to_match = '{}_{}'.format(
581 function_namespace.lstrip('cv.').replace('.', '_'), type_to_match
582 )
583 if type_to_match in possible_classes:
584 return known_classes[type_to_match]
585
586 # Try to find argument name as is
587 if argument_type in possible_classes:
588 return known_classes[argument_type]
589
590 # NOTE: parser is broken - some classes might not be visible, depending on
591 # the order of parsed headers.
592 # print("[WARNING] Can't select an appropriate class for argument: '",
593 # argument_type, "'. Possible matches: '", possible_classes, "'")
594 return None

1403 # doesn't work for trees(graphs) with depth > 2
