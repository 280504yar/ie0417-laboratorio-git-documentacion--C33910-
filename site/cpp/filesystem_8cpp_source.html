61cv::String join(const cv::String& base, const cv::String& path)
62{
63 if (base.empty())
64 return path;
65 if (path.empty())
66 return base;
67
68 bool baseSep = isPathSeparator(base[base.size() - 1]);
69 bool pathSep = isPathSeparator(path[0]);
70 String result;
71 if (baseSep && pathSep)
72 {
73 result = base + path.substr(1);
74 }
75 else if (!baseSep && !pathSep)
76 {
77 result = base + native_separator + path;
78 }
79 else
80 {
81 result = base + path;
82 }
83 return result;
84}

146CV_EXPORTS void remove_all(const cv::String& path)
147{
148 if (!exists(path))
149 return;
150 if (isDirectory(path))
151 {
152 std::vector<String> entries;
153 utils::fs::glob(path, cv::String(), entries, false, true);
154 for (size_t i = 0; i < entries.size(); i++)
155 {
156 const String& e = entries[i];
157 remove_all(e);
158 }
159#ifdef _MSC_VER
160 bool result = _rmdir(path.c_str()) == 0;
161#else
162 bool result = rmdir(path.c_str()) == 0;
163#endif
164 if (!result)
165 {
166 CV_LOG_ERROR(NULL, "Can't remove directory: " << path);
167 }
168 }
169 else
170 {
171#ifdef _MSC_VER
172 bool result = _unlink(path.c_str()) == 0;
173#else
174 bool result = unlink(path.c_str()) == 0;
175#endif
176 if (!result)
177 {
178 CV_LOG_ERROR(NULL, "Can't remove file: " << path);

219bool createDirectory(const cv::String& path)
220{
221 CV_INSTRUMENT_REGION();
222#if defined WIN32 || defined _WIN32 || defined WINCE
223#ifdef WINRT
224 wchar_t wpath[MAX_PATH];
225 size_t copied = mbstowcs(wpath, path.c_str(), MAX_PATH);
226 CV_Assert((copied != MAX_PATH) && (copied != (size_t)-1));
227 int result = CreateDirectoryA(wpath, NULL) ? 0 : -1;
228#else
229 int result = _mkdir(path.c_str());
230#endif
231#elif defined __linux__ || defined __APPLE__ || defined __HAIKU__ || defined __FreeBSD__ || defined __EMSCRIPTEN__ || defined __QNX__ || defined __CYGWIN__
232 int result = mkdir(path.c_str(), 0777);
233#else
234 int result = -1;
235#endif
236
237 if (result == -1)
238 {
239 return isDirectory(path);
240 }
241 return true;

244bool createDirectories(const cv::String& path_)
245{
246 cv::String path = path_;
247 for (;;)
248 {
249 char last_char = path.empty() ? 0 : path[path.length() - 1];
250 if (isPathSeparator(last_char))
251 {
252 path = path.substr(0, path.length() - 1);
253 continue;
254 }
255 break;
256 }
257
258 if (path.empty() || path == "./" || path == ".\\" || path == ".")
259 return true;
260 if (isDirectory(path))
261 return true;
262
263 size_t pos = path.rfind('/');
264 if (pos == cv::String::npos)
265 pos = path.rfind('\\');
266 if (pos != cv::String::npos)
267 {
268 cv::String parent_directory = path.substr(0, pos);
269 if (!parent_directory.empty())
270 {
271 if (!createDirectories(parent_directory))
272 return false;
273 }
274 }
275
276 return createDirectory(path);
277}

282{
283 Impl(const char* fname)
284 {
285 // http://support.microsoft.com/kb/316609
286 int numRetries = 5;
287 do
288 {
289 handle = ::CreateFileA(fname, GENERIC_READ, FILE_SHARE_READ | FILE_SHARE_WRITE, NULL,
290 OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);
291 if (INVALID_HANDLE_VALUE == handle)
292 {
293 if (ERROR_SHARING_VIOLATION == GetLastError())
294 {
295 numRetries--;
296 Sleep(250);
297 continue;
298 }
299 else
300 {
301 CV_Error_(Error::StsAssert, ("Can't open lock file: %s", fname));
302 }
303 }
304 break;
305 } while (numRetries > 0);
306 }
307 ~Impl()
308 {
309 if (INVALID_HANDLE_VALUE != handle)
310 {
311 ::CloseHandle(handle);
312 }
313 }
314
315 bool lock()
316 {
317 OVERLAPPED overlapped;
318 std::memset(&overlapped, 0, sizeof(overlapped));
319 return !!::LockFileEx(handle, LOCKFILE_EXCLUSIVE_LOCK, 0, MAXDWORD, MAXDWORD, &overlapped);
320 }
321 bool unlock()
322 {
323 OVERLAPPED overlapped;
324 std::memset(&overlapped, 0, sizeof(overlapped));
325 return !!::UnlockFileEx(handle, 0, MAXDWORD, MAXDWORD, &overlapped);
326 }
327
328 bool lock_shared()
329 {
330 OVERLAPPED overlapped;
331 std::memset(&overlapped, 0, sizeof(overlapped));
332 return !!::LockFileEx(handle, 0, 0, MAXDWORD, MAXDWORD, &overlapped);
333 }
334 bool unlock_shared()
335 {
336 return unlock();
337 }

339 HANDLE handle;

341private:

430cv::String getCacheDirectory(const char* sub_directory_name, const char* configuration_name)
431{
432 String cache_path;
433 if (configuration_name)
434 {
435 cache_path = utils::getConfigurationParameterString(configuration_name, "");
436 }
437 if (cache_path.empty())
438 {
439 cv::String default_cache_path;
440#ifdef _WIN32
441 wchar_t tmp_path_buf_w[MAX_PATH+1] = {0};
442 DWORD res = GetTempPathW(MAX_PATH, tmp_path_buf_w);
443 if (res > 0 && res <= MAX_PATH)
444 {
445 // Convert from UTF-16 to UTF-8 to support Unicode paths
446 int len = WideCharToMultiByte(CP_UTF8, 0, tmp_path_buf_w, -1, NULL, 0, NULL, NULL);
447 if (len > 0)
448 {
449 std::vector<char> utf8_buf(len);
450 WideCharToMultiByte(CP_UTF8, 0, tmp_path_buf_w, -1, utf8_buf.data(), len, NULL, NULL);
451 default_cache_path = std::string(utf8_buf.data());
452 }
453 }
454#elif defined __ANDROID__
455 // no defaults
456#elif defined __APPLE__
457 const std::string tmpdir_env = utils::getConfigurationParameterString("TMPDIR");
458 if (!tmpdir_env.empty() && utils::fs::isDirectory(tmpdir_env))
459 {
460 default_cache_path = tmpdir_env;
461 }
462 else
463 {
464 default_cache_path = "/tmp/";
465 CV_LOG_WARNING(NULL, "Using world accessible cache directory. This may be not secure: " << default_cache_path);
466 }
467#elif defined __linux__ || defined __HAIKU__ || defined __FreeBSD__ || defined __GNU__
468 // https://specifications.freedesktop.org/basedir-spec/basedir-spec-latest.html
469 if (default_cache_path.empty())
470 {
471 const std::string xdg_cache_env = utils::getConfigurationParameterString("XDG_CACHE_HOME");
472 if (!xdg_cache_env.empty() && utils::fs::isDirectory(xdg_cache_env))
473 {
474 default_cache_path = xdg_cache_env;
475 }
476 }
477 if (default_cache_path.empty())
478 {
479 const std::string home_env = utils::getConfigurationParameterString("HOME");
480 if (!home_env.empty() && utils::fs::isDirectory(home_env))
481 {
482 cv::String home_path = home_env;
483 cv::String home_cache_path = utils::fs::join(home_path, ".cache/");
484 if (utils::fs::isDirectory(home_cache_path))
485 {
486 default_cache_path = home_cache_path;
487 }
488 }
489 }
490 if (default_cache_path.empty())
491 {
492 const char* temp_path = "/var/tmp/";
493 if (utils::fs::isDirectory(temp_path))
494 {
495 default_cache_path = temp_path;
496 CV_LOG_WARNING(NULL, "Using world accessible cache directory. This may be not secure: " << default_cache_path);
497 }
498 }
499 if (default_cache_path.empty())
500 {
501 default_cache_path = "/tmp/";
502 CV_LOG_WARNING(NULL, "Using world accessible cache directory. This may be not secure: " << default_cache_path);
503 }
504#else
505 // no defaults
506#endif
507 CV_LOG_VERBOSE(NULL, 0, "default_cache_path = " << default_cache_path);
508 if (!default_cache_path.empty())
509 {
510 if (utils::fs::isDirectory(default_cache_path))
511 {
512 cv::String default_cache_path_base = utils::fs::join(default_cache_path, "opencv");
513 default_cache_path = utils::fs::join(default_cache_path_base, CVAUX_STR(CV_VERSION_MAJOR) ".x" CV_VERSION_STATUS);
514 if (utils::getConfigurationParameterBool("OPENCV_CACHE_SHOW_CLEANUP_MESSAGE", true)
515 && !utils::fs::isDirectory(default_cache_path))
516 {
517 std::vector<cv::String> existedCacheDirs;
518 try
519 {
520 utils::fs::glob_relative(default_cache_path_base, "*", existedCacheDirs, false, true);
521 }
522 catch (...)
523 {
524 // ignore
525 }
526 if (!existedCacheDirs.empty())
527 {
528 CV_LOG_WARNING(NULL, "Creating new OpenCV cache directory: " << default_cache_path);
529 CV_LOG_WARNING(NULL, "There are several neighbour directories, probably created by old OpenCV versions.");
530 CV_LOG_WARNING(NULL, "Feel free to cleanup these unused directories:");
531 for (size_t i = 0; i < existedCacheDirs.size(); i++)
532 {
533 CV_LOG_WARNING(NULL, " - " << existedCacheDirs[i]);
534 }
535 CV_LOG_WARNING(NULL, "Note: This message is showed only once.");
536 }
537 }
538 if (sub_directory_name && sub_directory_name[0] != '\0')
539 default_cache_path = utils::fs::join(default_cache_path, cv::String(sub_directory_name) + native_separator);
540 if (!utils::fs::createDirectories(default_cache_path))
541 {
542 CV_LOG_DEBUG(NULL, "Can't create OpenCV cache sub-directory: " << default_cache_path);
543 }
544 else
545 {
546 cache_path = default_cache_path;
547 }
548 }
549 else
550 {
551 CV_LOG_INFO(NULL, "Can't find default cache directory (does it exist?): " << default_cache_path);
552 }
553 }
554 else
555 {
556 CV_LOG_DEBUG(NULL, "OpenCV has no support to discover default cache directory on the current platform");
557 }
558 }
559 else
560 {
561 if (cache_path == "disabled")
562 return cache_path;
563 if (!isDirectory(cache_path))
564 {
565 CV_LOG_WARNING(NULL, "Specified non-existed directory, creating OpenCV sub-directory for caching purposes: " << cache_path);
566 if (!createDirectories(cache_path))
567 {
568 CV_LOG_ERROR(NULL, "Can't create OpenCV cache sub-directory: " << cache_path);
569 cache_path.clear();
570 }
571 }
572 }
573 CV_Assert(cache_path.empty() || utils::fs::isDirectory(cache_path));
574 if (!cache_path.empty())
575 {
576 if (!isPathSeparator(cache_path[cache_path.size() - 1]))
577 {
578 cache_path += native_separator;
579 }
580 }
581 return cache_path;
582}
