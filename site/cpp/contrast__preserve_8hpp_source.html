77double Decolor::energyCalcu(const vector <double> &Cg, const vector < vector <double> > &polyGrad, const vector <double> &wei) const
78{
79 const size_t size = polyGrad[0].size();
80 vector <double> energy(size);
81 vector <double> temp(size);
82 vector <double> temp1(size);
83
84 for(size_t i=0;i< polyGrad[0].size();i++)
85 {
86 double val = 0.0;
87 for(size_t j =0;j<polyGrad.size();j++)
88 val = val + (polyGrad[j][i] * wei[j]);
89 temp[i] = val - Cg[i];
90 temp1[i] = val + Cg[i];
91 }
92
93 for(size_t i=0;i<polyGrad[0].size();i++)
94 energy[i] = -1.0*log(exp(-1.0*std::pow(temp[i],2)/sigma) + exp(-1.0*std::pow(temp1[i],2)/sigma));
95
96 double sum = 0.0;
97 for(size_t i=0;i<polyGrad[0].size();i++)
98 sum +=energy[i];
99
100 return (sum/polyGrad[0].size());
101
102}

116vector<double> Decolor::product(const vector<Vec3i> &comb, const double initRGB[3])
117{
118 vector <double> res(comb.size());
119 for (size_t i=0;i<comb.size();i++)
120 {
121 double dp = 0.0;
122 for(int j=0;j<3;j++)
123 {
124 dp = dp + (comb[i][j] * initRGB[j]);
125 }
126 res[i] = dp;
127 }
128 return res;
129}

147void Decolor::gradvector(const Mat &img, vector <double> &grad) const
148{
149 Mat dest;
150 Mat dest1;
151 singleChannelGradx(img,dest);
152 singleChannelGrady(img,dest1);
153
154 // the function uses transposed dest and dest1 here and bellow
155 const int height = dest.size().width;
156 const int width = dest.size().height;
157
158 grad.resize(width * height * 2);
159
160 for(int i=0;i<height;i++)
161 for(int j=0;j<width;j++)
162 grad[i*width + j] = dest.at<float>(j, i);
163
164 const int offset = width * height;
165 for(int i=0;i<height;i++)
166 for(int j=0;j<width;j++)
167 grad[offset + i * width + j] = dest1.at<float>(j, i);
168}

170void Decolor::colorGrad(const Mat &img, vector <double> &Cg) const
171{
172 Mat lab;
173
174 cvtColor(img,lab,COLOR_BGR2Lab);
175
176 vector <Mat> lab_channel;
177 split(lab,lab_channel);
178
179 vector <double> ImL;
180 vector <double> Ima;
181 vector <double> Imb;
182
183 gradvector(lab_channel[0],ImL);
184 gradvector(lab_channel[1],Ima);
185 gradvector(lab_channel[2],Imb);
186
187 Cg.resize(ImL.size());
188 for(size_t i=0;i<ImL.size();i++)
189 {
190 const double res = sqrt(std::pow(ImL[i],2) + std::pow(Ima[i],2) + std::pow(Imb[i],2))/100;
191 Cg[i] = res;
192 }
193}

207void Decolor::weak_order(const Mat &im, vector <double> &alf) const
208{
209 Mat img;
210 const int h = im.size().height;
211 const int w = im.size().width;
212 if((h + w) > 800)
213 {
214 const double sizefactor = double(800)/(h+w);
215 resize(im, img, Size(cvRound(w*sizefactor), cvRound(h*sizefactor)));
216 }
217 else
218 {
219 img = im;
220 }
221
222 Mat curIm = Mat(img.size(),CV_32FC1);
223 vector <Mat> rgb_channel;
224 split(img,rgb_channel);
225
226 vector <double> Rg, Gg, Bg;
227 gradvector(rgb_channel[2],Rg);
228 gradvector(rgb_channel[1],Gg);
229 gradvector(rgb_channel[0],Bg);
230
231 vector <double> t1(Rg.size()), t2(Rg.size()), t3(Rg.size());
232 vector <double> tmp1(Rg.size()), tmp2(Rg.size()), tmp3(Rg.size());
233
234 const double level = .05;
235
236 for(size_t i=0;i<Rg.size();i++)
237 {
238 t1[i] = (Rg[i] > level) ? 1.0 : 0.0;
239 t2[i] = (Gg[i] > level) ? 1.0 : 0.0;
240 t3[i] = (Bg[i] > level) ? 1.0 : 0.0;
241 tmp1[i] = (Rg[i] < -1.0*level) ? 1.0 : 0.0;
242 tmp2[i] = (Gg[i] < -1.0*level) ? 1.0 : 0.0;
243 tmp3[i] = (Bg[i] < -1.0*level) ? 1.0 : 0.0;
244 }
245
246 alf.resize(Rg.size());
247 for(size_t i =0 ;i < Rg.size();i++)
248 alf[i] = (t1[i] * t2[i] * t3[i]);
249
250 for(size_t i =0 ;i < Rg.size();i++)
251 alf[i] -= tmp1[i] * tmp2[i] * tmp3[i];
252}

254void Decolor::grad_system(const Mat &im, vector < vector < double > > &polyGrad,
255 vector < double > &Cg, vector <Vec3i>& comb) const
256{
257 Mat img;
258 int h = im.size().height;
259 int w = im.size().width;
260 if((h + w) > 800)
261 {
262 const double sizefactor = double(800)/(h+w);
263 resize(im, img, Size(cvRound(w*sizefactor), cvRound(h*sizefactor)));
264 }
265 else
266 {
267 img = im;
268 }
269
270 h = img.size().height;
271 w = img.size().width;
272 colorGrad(img,Cg);
273
274 Mat curIm = Mat(img.size(),CV_32FC1);
275 vector <Mat> rgb_channel;
276 split(img,rgb_channel);
277
278 int idx = 0, idx1 = 0;
279 for(int r=0 ;r <=order; r++)
280 {
281 for(int g=0; g<=order;g++)
282 {
283 for(int b =0; b <=order;b++)
284 {
285 if((r+g+b)<=order && (r+g+b) > 0)
286 {
287 add_vector(comb,idx,r,g,b);
288 for(int i = 0;i<h;i++)
289 for(int j=0;j<w;j++)
290 curIm.at<float>(i,j)=static_cast<float>(
291 std::pow(rgb_channel[2].at<float>(i,j),r)*std::pow(rgb_channel[1].at<float>(i,j),g)*
292 std::pow(rgb_channel[0].at<float>(i,j),b));
293 vector <double> curGrad;
294 gradvector(curIm,curGrad);
295 add_to_vector_poly(polyGrad,curGrad,idx1);
296 }
297 }
298 }
299 }

302void Decolor::wei_update_matrix(const vector < vector <double> > &poly, const vector <double> &Cg, Mat &X)
303{
304 const int size = static_cast<int>(poly.size());
305 const int size0 = static_cast<int>(poly[0].size());
306 Mat P = Mat(size, size0, CV_32FC1);
307
308 for (int i = 0; i < size; i++)
309 for (int j = 0; j < size0;j++)
310 P.at<float>(i,j) = static_cast<float>(poly[i][j]);
311
312 Mat B = Mat(size, size0, CV_32FC1);
313 for(int i =0;i < size;i++)
314 {
315 for(int j = 0, end = int(Cg.size()); j < end;j++)
316 B.at<float>(i,j) = static_cast<float>(poly[i][j] * Cg[j]);
317 }
318
319 Mat A;
320 mulTransposed(P, A, false);
321 solve(A, B, X, DECOMP_NORMAL);
322
323}

325void Decolor::wei_inti(const vector <Vec3i> &comb, vector <double> &wei)
326{
327 double initRGB[3] = { .33, .33, .33 };
328
329 wei = product(comb,initRGB);
330
331 vector <int> sum(comb.size());
332
333 for(size_t i=0;i<comb.size();i++)
334 sum[i] = (comb[i][0] + comb[i][1] + comb[i][2]);
335
336 for(size_t i=0;i<sum.size();i++)
337 {
338 if(sum[i] == 1)
339 wei[i] = wei[i] * double(1);
340 else
341 wei[i] = wei[i] * double(0);
342 }
343
344 sum.clear();
345
346}

348void Decolor::grayImContruct(vector <double> &wei, const Mat &img, Mat &Gray) const
349{
350 const int h = img.size().height;
351 const int w = img.size().width;
352
353 vector <Mat> rgb_channel;
354 split(img,rgb_channel);
355
356 int kk =0;
357
358 for(int r =0;r<=order;r++)
359 {
360 for(int g=0;g<=order;g++)
361 {
362 for(int b=0;b<=order;b++)
363 {
364 if((r + g + b) <=order && (r+g+b) > 0)
365 {
366 for(int i = 0;i<h;i++)
367 for(int j=0;j<w;j++)
368 Gray.at<float>(i,j)=static_cast<float>(Gray.at<float>(i,j) +
369 static_cast<float>(wei[kk])*std::pow(rgb_channel[2].at<float>(i,j),r)*std::pow(rgb_channel[1].at<float>(i,j),g)*
370 std::pow(rgb_channel[0].at<float>(i,j),b));
371
372 kk=kk+1;
373 }
374 }
375 }
376 }
377
378 double minval, maxval;
379 minMaxLoc(Gray, &minval, &maxval);
380
381 Gray -= minval;
382 Gray /= maxval - minval;
383}
