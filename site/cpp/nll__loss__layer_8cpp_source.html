18class NegativeLogLikelihoodLossImpl CV_FINAL : public NegativeLogLikelihoodLossLayer
19{
20public:
21 NegativeLogLikelihoodLossImpl(const LayerParams& params)
22 {
23 setParamsFrom(params);
24 String red = toLowerCase(params.get<String>("reduction", "mean"));
25 if (red == "none") reduction = LOSS_REDUCTION_NONE;
26 else if (red == "mean") reduction = LOSS_REDUCTION_MEAN;
27 else if (red == "sum") reduction = LOSS_REDUCTION_SUM;
28 else CV_Error(Error::StsBadArg, "Unsupported reduction: " + red);
29 ignoreIndex = params.get<int>("ignore_index", -1);
30 }
31
32 virtual bool supportBackend(int backendId) CV_OVERRIDE
33 {
34 return backendId == DNN_BACKEND_OPENCV;
35 }
36
37 bool getMemoryShapes(const std::vector<MatShape>& in, const int reqOut,
38 std::vector<MatShape>& out, std::vector<MatShape>& internals) const CV_OVERRIDE
39 {
40 CV_Assert(in.size() >= 2);
41 const MatShape& x = in[0];
42 CV_Assert(x.size() >= 2);
43 if (reduction == LOSS_REDUCTION_NONE)
44 {
45 MatShape shp = x;
46 MatShape y; y.reserve(shp.size()-1);
47 y.push_back(shp[0]);
48 for (size_t i = 2; i < shp.size(); ++i) {
49 y.push_back(shp[i]);
50 }
51 out.assign(1, y);
52 }
53 else
54 {
55 // Reduced (mean/sum) loss is a rank-0 scalar in ONNX, not a [1] tensor.
56 out.assign(1, MatShape::scalar());
57 }
58 return false;
59 }
60
61 void getTypes(const std::vector<MatType>&, const int reqOut, const int reqInt,
62 std::vector<MatType>& out, std::vector<MatType>& internals) const CV_OVERRIDE
63 {
64 out.assign(1, MatType(CV_32F));
65 internals.assign(reqInt, MatType(CV_32F));
66 }

68 void forward(InputArrayOfArrays in_arr, OutputArrayOfArrays out_arr, OutputArrayOfArrays) CV_OVERRIDE
69 {
70 std::vector<Mat> inp;
71 in_arr.getMatVector(inp);
72
73 const Mat& logp = inp[0];
74 const Mat& label = inp[1];
75 const bool hasWeight = inp.size() >= 3;
76
77 CV_Assert(logp.dims >= 2);
78 const int N = logp.size[0], C = logp.size[1];
79 int S = 1; for (int i = 2; i < logp.dims; ++i) S *= logp.size[i];
80
81 MatShape lossShape;
82 bool isReduced = (reduction != LOSS_REDUCTION_NONE);
83 if (!isReduced) {
84 lossShape.push_back(N);
85 for (int i = 2; i < logp.dims; ++i) {
86 lossShape.push_back(logp.size[i]);
87 }
88 }
89
90 auto kind = out_arr.kind();
91 Mat out_loss;
92 if (kind == _InputArray::STD_VECTOR_MAT) {
93 std::vector<Mat>& outs = out_arr.getMatVecRef();
94 CV_Assert(outs.size() == 1);
95 if (!isReduced) {
96 outs[0].fit(lossShape, CV_32F);
97 }
98 out_loss = outs[0];
99 } else if (kind == _InputArray::STD_VECTOR_UMAT) {
100 std::vector<UMat>& uouts = out_arr.getUMatVecRef();
101 CV_Assert(uouts.size() == 1);
102 if (!isReduced) {
103 uouts[0].fit(lossShape, CV_32F);
104 }
105 out_loss = uouts[0].getMat(ACCESS_WRITE);
106 } else {
107 CV_Error(cv::Error::StsBadArg, cv::format("Unsupported output array kind: %d", kind));
108 }
109 CV_Assert(out_loss.type() == CV_32F);
110 if (isReduced) {
111 CV_Assert(out_loss.total() == 1);
112 }
113
114 Mat w_f;
115 if (hasWeight) {
116 Mat wsrc = inp[2];
117 CV_Assert(wsrc.total() == (size_t)C);
118 Mat wflat = wsrc.reshape(1, (int)wsrc.total()).clone();
119 wflat.convertTo(w_f, CV_32F);
120 }
121
122 const float* wfDataPtr = hasWeight ? w_f.ptr<float>() : nullptr;
123
124 const int nstripes = 16;
125 Mat logp32; logp.convertTo(logp32, CV_32F);
126 const size_t sN = logp32.step1(0);
127 const size_t sC = logp32.step1(1);
128
129 CV_Assert(label.depth() == CV_32S || label.depth() == CV_64S);
130 CV_Assert((int)label.total() == N*S);
131 Mat idx1D = label.reshape(1, N*S);
132
133 Mat per(N*S, 1, CV_32F, Scalar(0));
134 Mat validMask(N*S, 1, CV_8U, Scalar(1));
135 Mat effW(N*S, 1, CV_32F, Scalar(1));
136
137 if (label.depth() == CV_32S)
138 {
139 reduceNLLPerSample<int32_t>(logp32, sN, sC, S, idx1D, C, ignoreIndex, hasWeight, wfDataPtr, per, effW, validMask, nstripes);
140 }
141 else
142 {
143 reduceNLLPerSample<int64_t>(logp32, sN, sC, S, idx1D, C, ignoreIndex, hasWeight, wfDataPtr, per, effW, validMask, nstripes);
144 }
145
146 if (!isReduced)
147 {
148 per.reshape(1, out_loss.size).copyTo(out_loss);
149 }
150 else
151 {
152 double num = 0.0, den = 0.0;
153 const float* perData = per.ptr<float>();
154 const float* effData = effW.ptr<float>();
155 const uchar* validData = validMask.ptr<uchar>();
156 const int R = per.rows;
157
158 for (int r = 0; r < R; ++r) {
159 const float m = validData[r] ? 1.f : 0.f;
160 num += static_cast<double>(perData[r] * m);
161
162 if (reduction == LOSS_REDUCTION_MEAN)
163 {
164 den += static_cast<double>(std::max(1e-12f, effData[r]) * m);
165 }
166 }
167 const float out = (reduction == LOSS_REDUCTION_SUM) ? static_cast<float>(num) : static_cast<float>((den > 0.0) ? (num / den) : 0.0);
168 out_loss.ptr<float>()[0] = out;
169 }
170 }
171
172private:
173 template<typename T>
174 static inline void reduceNLLPerSample(const Mat& logp32,
175 const size_t sN_,
176 const size_t sC_,
177 const int S_,
178 const Mat& idx1D,
179 const int C_,
180 const int ignoreIndex_,
181 const bool hasWeight_,
182 const float* wfData_,
183 Mat& per,
184 Mat& effW,
185 Mat& validMask,
186 const int nstripes)
187 {
188 CV_Assert(idx1D.cols == 1);
189 parallel_for_(Range(0, idx1D.rows), [&](const Range& rr){
190 const float* base = logp32.ptr<float>();
191 const T* idx1DData = idx1D.ptr<T>();
192 uchar* validMaskData = validMask.ptr<uchar>();
193 float* effWData = effW.ptr<float>();
194 float* perData = per.ptr<float>();
195
196 const size_t sIdx = idx1D.step1(0);
197 const size_t sVM = validMask.step1(0);
198 const size_t sEff = effW.step1(0);
199 const size_t sPer = per.step1(0);
200
201 const size_t sN = sN_;
202 const size_t sC = sC_;
203 const int S = S_;
204 const int C = C_;
205 const int ignoreIndex = ignoreIndex_;
206 const bool hasWeight = hasWeight_;
207 const float* wfData = wfData_;
208
209 for (int r = rr.start; r < rr.end; ++r)
210 {
211 const T y_raw = idx1DData[r * sIdx];
212 if ((int64_t)y_raw == (int64_t)ignoreIndex)
213 {
214 validMaskData[r * sVM] = 0;
215 effWData[r * sEff] = 0.f;
216 continue;
217 }
218
219 const int64_t yi64 = (int64_t)y_raw;
220 CV_Assert(yi64 >= 0 && yi64 < (int64_t)C);
221 const int y = static_cast<int>(yi64);
222
223 const int n = r / S;
224 const int s = r % S;
225 const float lp = base[n * sN + y * sC + s];
226 const float cw = hasWeight ? wfData[y] : 1.f;
227
228 perData[r * sPer] = -lp * cw;
229 effWData[r * sEff] = cw;
230 }
231 }, nstripes);
232 }
233};
