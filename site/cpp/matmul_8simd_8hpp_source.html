53// The SIMD GEMM kernels below reorder floating-point accumulation (and use FMA
54// where the ISA provides it), producing a result that is correct (<1e-9 vs a
55// long-double reference) but differs from the scalar reference in the last bits.
56// The multi-view camera-calibration solver is ill-conditioned - fisheye
57// ComputeHomography forms L^T*L, takes an SVD, then iterates (J^T*J)^-1 - so it
58// amplifies that tiny deviation enough to flip its outcome on some build targets.
59// CI (PR #29472, RegisterCamerasTest) shows this only on the 32-bit x86 (Windows)
60// and macOS-x64 (Intel) jobs; Windows x64, macOS-ARM64 and all Linux targets pass.
61// FP background: on 32-bit x86 the x87 FPU holds intermediates in 80-bit extended
62// precision whereas SIMD accumulates in 64-bit; on modern x86-64 and ARM both
63// scalar and SIMD use strict IEEE-754 width (64-bit double / 32-bit float, no
64// hidden extended precision). The failure is therefore a toolchain/ABI-specific
65// bifurcation rather than a single-feature effect, so we simply keep the
66// reference-matching scalar GEMM path on the affected targets.
67#ifndef CV_GEMM_SIMD_ENABLED
68#if defined(__i386__) || defined(_M_IX86) || (defined(__APPLE__) && defined(__x86_64__))
69#define CV_GEMM_SIMD_ENABLED 0
70#else
71#define CV_GEMM_SIMD_ENABLED 1
72#endif
73#endif

128GEMM_CopyBlock( const uchar* src, size_t src_step,
129 uchar* dst, size_t dst_step,
130 Size size, size_t pix_size )
131{
132 int j;
133 size.width *= (int)(pix_size / sizeof(int));
134
135 for( ; size.height--; src += src_step, dst += dst_step )
136 {
137 j=0;
138 #if CV_ENABLE_UNROLLED
139 for( ; j <= size.width - 4; j += 4 )
140 {
141 int t0 = ((const int*)src)[j];
142 int t1 = ((const int*)src)[j+1];
143 ((int*)dst)[j] = t0;
144 ((int*)dst)[j+1] = t1;
145 t0 = ((const int*)src)[j+2];
146 t1 = ((const int*)src)[j+3];
147 ((int*)dst)[j+2] = t0;
148 ((int*)dst)[j+3] = t1;
149 }
150 #endif
151 for( ; j < size.width; j++ )
152 ((int*)dst)[j] = ((const int*)src)[j];
153 }
154}

157static void
158GEMM_TransposeBlock( const uchar* src, size_t src_step,
159 uchar* dst, size_t dst_step,
160 Size size, size_t pix_size )
161{
162 int i, j;
163 for( i = 0; i < size.width; i++, dst += dst_step, src += pix_size )
164 {
165 const uchar* _src = src;
166 switch( pix_size )
167 {
168 case sizeof(int):
169 for( j = 0; j < size.height; j++, _src += src_step )
170 ((int*)dst)[j] = ((int*)_src)[0];
171 break;
172 case sizeof(int)*2:
173 for( j = 0; j < size.height*2; j += 2, _src += src_step )
174 {
175 int t0 = ((int*)_src)[0];
176 int t1 = ((int*)_src)[1];
177 ((int*)dst)[j] = t0;
178 ((int*)dst)[j+1] = t1;
179 }
180 break;
181 case sizeof(int)*4:
182 for( j = 0; j < size.height*4; j += 4, _src += src_step )
183 {
184 int t0 = ((int*)_src)[0];
185 int t1 = ((int*)_src)[1];
186 ((int*)dst)[j] = t0;
187 ((int*)dst)[j+1] = t1;
188 t0 = ((int*)_src)[2];
189 t1 = ((int*)_src)[3];
190 ((int*)dst)[j+2] = t0;
191 ((int*)dst)[j+3] = t1;
192 }
193 break;
194 default:
195 CV_Assert(0);
196 return;
197 }
198 }

222template<typename ST>
223static inline ST simdDotProduct(const ST* a, const ST* b, int n, int& k)
224{
225 typedef typename VecTraits<ST>::VT VecT;
226 const int vlanes = VTraits<VecT>::vlanes();
227 const int stride4 = vlanes * 4;
228 VecT vs0 = VecTraits<ST>::zero();
229 VecT vs1 = VecTraits<ST>::zero();
230 VecT vs2 = VecTraits<ST>::zero();
231 VecT vs3 = VecTraits<ST>::zero();
232 for( ; k <= n - stride4; k += stride4 )
233 {
234 vs0 = v_muladd(VecTraits<ST>::load(a + k), VecTraits<ST>::load(b + k), vs0);
235 vs1 = v_muladd(VecTraits<ST>::load(a + k + vlanes), VecTraits<ST>::load(b + k + vlanes), vs1);
236 vs2 = v_muladd(VecTraits<ST>::load(a + k + vlanes * 2), VecTraits<ST>::load(b + k + vlanes * 2), vs2);
237 vs3 = v_muladd(VecTraits<ST>::load(a + k + vlanes * 3), VecTraits<ST>::load(b + k + vlanes * 3), vs3);
238 }
239 vs0 = v_add(v_add(vs0, vs1), v_add(vs2, vs3));
240 for( ; k <= n - vlanes; k += vlanes )
241 vs0 = v_muladd(VecTraits<ST>::load(a + k), VecTraits<ST>::load(b + k), vs0);
242 return v_reduce_sum(vs0);
243}

247// Float input, double accumulation dot product (for 32FC1 where WT=double)
248static inline double simdDotProduct_f32f64(const float* a, const float* b, int n, int& k)
249{
250 const int vlanes_f32 = VTraits<v_float32>::vlanes();
251 const int stride2 = vlanes_f32 * 2;
252 v_float64 vs0 = vx_setzero_f64();
253 v_float64 vs1 = vx_setzero_f64();
254 v_float64 vs2 = vx_setzero_f64();
255 v_float64 vs3 = vx_setzero_f64();
256 for( ; k <= n - stride2; k += stride2 )
257 {
258 v_float32 va0 = vx_load(a + k);
259 v_float32 vb0 = vx_load(b + k);
260 vs0 = v_muladd(v_cvt_f64(va0), v_cvt_f64(vb0), vs0);
261 vs1 = v_muladd(v_cvt_f64_high(va0), v_cvt_f64_high(vb0), vs1);
262
263 v_float32 va1 = vx_load(a + k + vlanes_f32);
264 v_float32 vb1 = vx_load(b + k + vlanes_f32);
265 vs2 = v_muladd(v_cvt_f64(va1), v_cvt_f64(vb1), vs2);
266 vs3 = v_muladd(v_cvt_f64_high(va1), v_cvt_f64_high(vb1), vs3);
267 }
268 vs0 = v_add(v_add(vs0, vs1), v_add(vs2, vs3));
269 for( ; k <= n - vlanes_f32; k += vlanes_f32 )
270 {
271 v_float32 va = vx_load(a + k);
272 v_float32 vb = vx_load(b + k);
273 vs0 = v_muladd(v_cvt_f64(va), v_cvt_f64(vb), vs0);
274 vs0 = v_muladd(v_cvt_f64_high(va), v_cvt_f64_high(vb), vs0);
275 }
276 return v_reduce_sum(vs0);
277}

279// Float input, double accumulation GEMM k-outer/j-inner (for 32FC1 where WT=double)
280static inline void simdGEMM_kj_f32f64(
281 const float* a_data, const float* _b_data, size_t b_step,
282 const float* _c_data, size_t c_step1,
283 float* d_data,
284 int n, int m,
285 double alpha, double beta,
286 int& j)
287{
288 const int vlanes_f32 = VTraits<v_float32>::vlanes();
289 const int vlanes_f64 = VTraits<v_float64>::vlanes();
290
291 if( m >= 2 * vlanes_f32 && n >= 64 )
292 {
293 // k-outer / j-inner with double accumulation
294 const int max_m = 1600 / (int)sizeof(float);
295 cv::AutoBuffer<double> _s_buf(max_m);
296 double* s_buf = _s_buf.data();
297
298 {
299 v_float64 vz = vx_setzero_f64();
300 for( j = 0; j <= m - vlanes_f64; j += vlanes_f64 )
301 v_store(s_buf + j, vz);
302 for( ; j < m; j++ )
303 s_buf[j] = 0.0;
304 }
305
306 {
307 const float* b_ptr = _b_data;
308 for( int k = 0; k < n; k++, b_ptr += b_step )
309 {
310 v_float64 va = vx_setall_f64((double)a_data[k]);
311 j = 0;
312 for( ; j <= m - vlanes_f32; j += vlanes_f32 )
313 {
314 v_float32 vb = vx_load(b_ptr + j);
315 v_float64 vb_lo = v_cvt_f64(vb);
316 v_float64 vb_hi = v_cvt_f64_high(vb);
317 v_float64 vd_lo = vx_load(s_buf + j);
318 v_float64 vd_hi = vx_load(s_buf + j + vlanes_f64);
319 vd_lo = v_muladd(va, vb_lo, vd_lo);
320 vd_hi = v_muladd(va, vb_hi, vd_hi);
321 v_store(s_buf + j, vd_lo);
322 v_store(s_buf + j + vlanes_f64, vd_hi);
323 }
324 double a_val = (double)a_data[k];
325 for( ; j < m; j++ )
326 s_buf[j] += a_val * (double)b_ptr[j];
327 }
328 }
329
330 {
331 v_float64 valpha = vx_setall_f64(alpha);
332 const float* c_data = _c_data;
333 j = 0;
334 if( !_c_data )
335 {
336 for( ; j <= m - vlanes_f32; j += vlanes_f32 )
337 {
338 v_float64 vs_lo = v_mul(vx_load(s_buf + j), valpha);
339 v_float64 vs_hi = v_mul(vx_load(s_buf + j + vlanes_f64), valpha);
340 v_store(d_data + j, v_cvt_f32(vs_lo, vs_hi));
341 }
342 for( ; j < m; j++ )
343 d_data[j] = (float)(s_buf[j] * alpha);
344 }
345 else if( c_step1 == 1 )
346 {
347 v_float64 vbeta = vx_setall_f64(beta);
348 for( ; j <= m - vlanes_f32; j += vlanes_f32 )
349 {
350 v_float64 vs_lo = v_mul(vx_load(s_buf + j), valpha);
351 v_float64 vs_hi = v_mul(vx_load(s_buf + j + vlanes_f64), valpha);
352 v_float32 vc = vx_load(c_data + j);
353 vs_lo = v_muladd(v_cvt_f64(vc), vbeta, vs_lo);
354 vs_hi = v_muladd(v_cvt_f64_high(vc), vbeta, vs_hi);
355 v_store(d_data + j, v_cvt_f32(vs_lo, vs_hi));
356 }
357 for( ; j < m; j++ )
358 d_data[j] = (float)(s_buf[j] * alpha + (double)c_data[j] * beta);
359 }
360 else
361 {
362 for( j = 0; j < m; j++, c_data += c_step1 )
363 d_data[j] = (float)(s_buf[j] * alpha + (double)c_data[0] * beta);
364 }
365 }
366 j = m;
367 }
368 else if( vlanes_f32 * (int)sizeof(float) >= 32 )
369 {
370 // j-outer with SIMD: double accumulation per vlanes_f64 group
371 const float* c_data = _c_data;
372 for( ; j <= m - vlanes_f32; j += vlanes_f32, c_data += vlanes_f32 * c_step1 )
373 {
374 const float* b = _b_data + j;
375 v_float64 vs_lo = vx_setzero_f64();
376 v_float64 vs_hi = vx_setzero_f64();
377 for( int k = 0; k < n; k++, b += b_step )
378 {
379 v_float64 va = vx_setall_f64((double)a_data[k]);
380 v_float32 vb = vx_load(b);
381 vs_lo = v_muladd(va, v_cvt_f64(vb), vs_lo);
382 vs_hi = v_muladd(va, v_cvt_f64_high(vb), vs_hi);
383 }
384 v_float64 valpha = vx_setall_f64(alpha);
385 vs_lo = v_mul(vs_lo, valpha);
386 vs_hi = v_mul(vs_hi, valpha);
387 if( !_c_data )
388 {
389 v_store(d_data + j, v_cvt_f32(vs_lo, vs_hi));
390 }
391 else if( c_step1 == 1 )
392 {
393 v_float64 vbeta = vx_setall_f64(beta);
394 v_float32 vc = vx_load(c_data);
395 vs_lo = v_muladd(v_cvt_f64(vc), vbeta, vs_lo);
396 vs_hi = v_muladd(v_cvt_f64_high(vc), vbeta, vs_hi);
397 v_store(d_data + j, v_cvt_f32(vs_lo, vs_hi));
398 }
399 else
400 {
401 CV_DECL_ALIGNED(CV_SIMD_WIDTH) float buf[VTraits<v_float32>::max_nlanes];
402 v_store_aligned(buf, v_cvt_f32(vs_lo, vs_hi));
403 for( int jj = 0; jj < vlanes_f32; jj++ )
404 d_data[j + jj] = buf[jj] + c_data[jj * c_step1] * (float)beta;
405 }
406 }
407 }
408}

411#if CV_SIMD
412template<typename ST>
413static inline void simdGEMM_kj(
414 const ST* a_data, const ST* _b_data, size_t b_step,
415 const ST* _c_data, size_t c_step1,
416 ST* d_data,
417 int n, int m,
418 double alpha, double beta,
419 int& j)
420{
421 typedef typename VecTraits<ST>::VT VecT;
422 const int vlanes = VTraits<VecT>::vlanes();
423
424 if( m >= 2 * vlanes && n >= 64 )
425 {
426 // k-outer / j-inner SIMD: sequential B-row access, good for large n
427 const int max_m = 1600 / (int)sizeof(ST);
428 CV_DECL_ALIGNED(CV_SIMD_WIDTH) ST s_buf[max_m];
429
430 {
431 VecT vz = VecTraits<ST>::zero();
432 for( j = 0; j <= m - vlanes; j += vlanes )
433 v_store_aligned(s_buf + j, vz);
434 for( ; j < m; j++ )
435 s_buf[j] = (ST)0;
436 }
437
438 {
439 const ST* b_ptr = _b_data;
440 for( int k = 0; k < n; k++, b_ptr += b_step )
441 {
442 VecT va = VecTraits<ST>::setall(a_data[k]);
443 j = 0;
444 for( ; j <= m - vlanes; j += vlanes )
445 {
446 VecT vd = vx_load_aligned(s_buf + j);
447 VecT vb = VecTraits<ST>::load(b_ptr + j);
448 vd = v_muladd(va, vb, vd);
449 v_store_aligned(s_buf + j, vd);
450 }
451 ST a_val = a_data[k];
452 for( ; j < m; j++ )
453 s_buf[j] += a_val * b_ptr[j];
454 }
455 }
456
457 {
458 VecT valpha = VecTraits<ST>::setall((ST)alpha);
459 const ST* c_data = _c_data;
460 j = 0;
461 if( !_c_data )
462 {
463 for( ; j <= m - vlanes; j += vlanes )
464 {
465 VecT vs = v_mul(vx_load_aligned(s_buf + j), valpha);
466 v_store(d_data + j, vs);
467 }
468 for( ; j < m; j++ )
469 d_data[j] = s_buf[j] * (ST)alpha;
470 }
471 else if( c_step1 == 1 )
472 {
473 VecT vbeta = VecTraits<ST>::setall((ST)beta);
474 for( ; j <= m - vlanes; j += vlanes )
475 {
476 VecT vs = v_mul(vx_load_aligned(s_buf + j), valpha);
477 VecT vc = VecTraits<ST>::load(c_data + j);
478 vs = v_muladd(vc, vbeta, vs);
479 v_store(d_data + j, vs);
480 }
481 for( ; j < m; j++ )
482 d_data[j] = s_buf[j] * (ST)alpha + c_data[j] * (ST)beta;
483 }
484 else
485 {
486 for( j = 0; j < m; j++, c_data += c_step1 )
487 d_data[j] = s_buf[j] * (ST)alpha + c_data[0] * (ST)beta;
488 }
489 }
490 j = m; // all columns handled
491 }
492 else if( vlanes * (int)sizeof(ST) >= 32 )
493 {
494 // j-outer with SIMD: process vlanes columns per group
495 // (skip on narrow SIMD where 4-column scalar is faster)
496 VecT valpha = VecTraits<ST>::setall((ST)alpha);
497 const ST* c_data = _c_data;
498 for( ; j <= m - vlanes; j += vlanes, c_data += vlanes * c_step1 )
499 {
500 const ST* b = _b_data + j;
501 VecT vs = VecTraits<ST>::zero();
502 for( int k = 0; k < n; k++, b += b_step )
503 {
504 VecT va = VecTraits<ST>::setall(a_data[k]);
505 VecT vb = VecTraits<ST>::load(b);
506 vs = v_muladd(va, vb, vs);
507 }
508 vs = v_mul(vs, valpha);
509 if( !_c_data )
510 {
511 v_store(d_data + j, vs);
512 }
513 else if( c_step1 == 1 )
514 {
515 VecT vc = VecTraits<ST>::load(c_data);
516 vs = v_muladd(vc, VecTraits<ST>::setall((ST)beta), vs);
517 v_store(d_data + j, vs);
518 }
519 else
520 {
521 CV_DECL_ALIGNED(CV_SIMD_WIDTH) ST buf[VTraits<VecT>::max_nlanes];
522 v_store_aligned(buf, vs);
523 for( int jj = 0; jj < vlanes; jj++ )
524 d_data[j + jj] = buf[jj] + c_data[jj * c_step1] * (ST)beta;
525 }
526 }
527 // Remaining columns handled by 4-column scalar in caller
528 }
529}

533template<typename T, typename WT> static void
534GEMMSingleMul( const T* a_data, size_t a_step,

540{
541 int i, j, k, n = a_size.width, m = d_size.width, drows = d_size.height;
542 const T *_a_data = a_data, *_b_data = b_data, *_c_data = c_data;
543 cv::AutoBuffer<T> _a_buf;
544 T* a_buf = 0;
545 size_t a_step0, a_step1, c_step0, c_step1, t_step;
546
547 a_step /= sizeof(a_data[0]);
548 b_step /= sizeof(b_data[0]);
549 c_step /= sizeof(c_data[0]);
550 d_step /= sizeof(d_data[0]);
551 a_step0 = a_step;
552 a_step1 = 1;
553
554 if( !c_data )
555 c_step0 = c_step1 = 0;
556 else if( !(flags & GEMM_3_T) )
557 c_step0 = c_step, c_step1 = 1;
558 else
559 c_step0 = 1, c_step1 = c_step;
560
561 if( flags & GEMM_1_T )
562 {
563 CV_SWAP( a_step0, a_step1, t_step );
564 n = a_size.height;
565 if( a_step > 1 && n > 1 )
566 {
567 _a_buf.allocate(n);
568 a_buf = _a_buf.data();
569 }
570 }
571
572 if( n == 1 ) /* external product */
573 {
574 cv::AutoBuffer<T> _b_buf;
575 T* b_buf = 0;
576
577 if( a_step > 1 && a_size.height > 1 )
578 {
579 _a_buf.allocate(drows);
580 a_buf = _a_buf.data();
581 for( k = 0; k < drows; k++ )
582 a_buf[k] = a_data[a_step*k];
583 a_data = a_buf;
584 }
585
586 if( b_step > 1 )
587 {
588 _b_buf.allocate(d_size.width);
589 b_buf = _b_buf.data();
590 for( j = 0; j < d_size.width; j++ )
591 b_buf[j] = b_data[j*b_step];
592 b_data = b_buf;
593 }
594
595 for( i = 0; i < drows; i++, _c_data += c_step0, d_data += d_step )
596 {
597 WT al = WT(a_data[i])*alpha;
598 c_data = _c_data;
599 for( j = 0; j <= d_size.width - 2; j += 2, c_data += 2*c_step1 )
600 {
601 WT s0 = al*WT(b_data[j]);
602 WT s1 = al*WT(b_data[j+1]);
603 if( !c_data )
604 {
605 d_data[j] = T(s0);
606 d_data[j+1] = T(s1);
607 }
608 else
609 {
610 d_data[j] = T(s0 + WT(c_data[0])*beta);
611 d_data[j+1] = T(s1 + WT(c_data[c_step1])*beta);
612 }
613 }
614
615 for( ; j < d_size.width; j++, c_data += c_step1 )
616 {
617 WT s0 = al*WT(b_data[j]);
618 if( !c_data )
619 d_data[j] = T(s0);
620 else
621 d_data[j] = T(s0 + WT(c_data[0])*beta);
622 }
623 }
624 }
625 else if( flags & GEMM_2_T ) /* A * Bt */
626 {
627 for( i = 0; i < drows; i++, _a_data += a_step0, _c_data += c_step0, d_data += d_step )
628 {
629 a_data = _a_data;
630 b_data = _b_data;
631 c_data = _c_data;
632
633 if( a_buf )
634 {
635 for( k = 0; k < n; k++ )
636 a_buf[k] = a_data[a_step1*k];
637 a_data = a_buf;
638 }
639
640 for( j = 0; j < d_size.width; j++, b_data += b_step,
641 c_data += c_step1 )
642 {
643 WT s0(0);
644 k = 0;
645#if CV_SIMD_64F && CV_GEMM_SIMD_ENABLED
646 if( sizeof(WT) == sizeof(double) )
647 {
648 if( sizeof(T) == sizeof(double) )
649 s0 = (WT)simdDotProduct((const double*)a_data, (const double*)b_data, n, k);
650 else if( sizeof(T) == sizeof(float) )
651 s0 = (WT)simdDotProduct_f32f64((const float*)a_data, (const float*)b_data, n, k);
652 }
653 else
654#endif
655 {
656 WT s1(0), s2(0), s3(0);
657 #if CV_ENABLE_UNROLLED
658 for( ; k <= n - 4; k += 4 )
659 {
660 s0 += WT(a_data[k])*WT(b_data[k]);
661 s1 += WT(a_data[k+1])*WT(b_data[k+1]);
662 s2 += WT(a_data[k+2])*WT(b_data[k+2]);
663 s3 += WT(a_data[k+3])*WT(b_data[k+3]);
664 }
665 #endif
666 s0 += s1 + s2 + s3;
667 }
668 for( ; k < n; k++ )
669 s0 += WT(a_data[k])*WT(b_data[k]);
670 s0 = s0*alpha;
671
672 if( !c_data )
673 d_data[j] = T(s0);
674 else
675 d_data[j] = T(s0 + WT(c_data[0])*beta);
676 }
677 }
678 }
679 else if( d_size.width*sizeof(d_data[0]) <= 1600 )
680 {
681 for( i = 0; i < drows; i++, _a_data += a_step0,
682 _c_data += c_step0,
683 d_data += d_step )
684 {
685 a_data = _a_data, c_data = _c_data;
686
687 if( a_buf )
688 {
689 for( k = 0; k < n; k++ )
690 a_buf[k] = a_data[a_step1*k];
691 a_data = a_buf;
692 }
693
694 j = 0;
695#if CV_SIMD_64F && CV_GEMM_SIMD_ENABLED
696 if( sizeof(WT) == sizeof(double) )
697 {
698 if( sizeof(T) == sizeof(double) )
699 simdGEMM_kj((const double*)a_data, (const double*)_b_data, b_step,
700 (const double*)_c_data, c_step1,
701 (double*)d_data, n, m, alpha, beta, j);
702 else if( sizeof(T) == sizeof(float) )
703 simdGEMM_kj_f32f64((const float*)a_data, (const float*)_b_data, b_step,
704 (const float*)_c_data, c_step1,
705 (float*)d_data, n, m, alpha, beta, j);
706 }
707#endif
708 c_data = _c_data + j * c_step1;
709 // 4-column j-outer with register accumulators for remaining columns
710 for( ; j <= m - 4; j += 4, c_data += 4*c_step1 )
711 {
712 const T* b = _b_data + j;
713 WT s0(0), s1(0), s2(0), s3(0);
714
715 for( k = 0; k < n; k++, b += b_step )
716 {
717 WT a(a_data[k]);
718 s0 += a * WT(b[0]); s1 += a * WT(b[1]);
719 s2 += a * WT(b[2]); s3 += a * WT(b[3]);
720 }
721
722 if( !c_data )
723 {
724 d_data[j] = T(s0*alpha);
725 d_data[j+1] = T(s1*alpha);
726 d_data[j+2] = T(s2*alpha);
727 d_data[j+3] = T(s3*alpha);
728 }
729 else
730 {
731 s0 = s0*alpha; s1 = s1*alpha;
732 s2 = s2*alpha; s3 = s3*alpha;
733 d_data[j] = T(s0 + WT(c_data[0])*beta);
734 d_data[j+1] = T(s1 + WT(c_data[c_step1])*beta);
735 d_data[j+2] = T(s2 + WT(c_data[c_step1*2])*beta);
736 d_data[j+3] = T(s3 + WT(c_data[c_step1*3])*beta);
737 }
738 }
739
740 for( ; j < m; j++, c_data += c_step1 )
741 {
742 const T* b = _b_data + j;
743 WT s0(0);
744
745 for( k = 0; k < n; k++, b += b_step )
746 s0 += WT(a_data[k]) * WT(b[0]);
747
748 s0 = s0*alpha;
749 if( !c_data )
750 d_data[j] = T(s0);
751 else
752 d_data[j] = T(s0 + WT(c_data[0])*beta);
753 }
754 }
755 }
756 else
757 {
758 cv::AutoBuffer<WT> _d_buf(m);
759 WT* d_buf = _d_buf.data();
760
761 for( i = 0; i < drows; i++, _a_data += a_step0, _c_data += c_step0, d_data += d_step )
762 {
763 a_data = _a_data;
764 b_data = _b_data;
765 c_data = _c_data;
766
767 if( a_buf )
768 {
769 for( k = 0; k < n; k++ )
770 a_buf[k] = _a_data[a_step1*k];
771 a_data = a_buf;
772 }
773
774 for( j = 0; j < m; j++ )
775 d_buf[j] = WT(0);
776
777 for( k = 0; k < n; k++, b_data += b_step )
778 {
779 WT al(a_data[k]);
780 j=0;
781#if CV_SIMD_64F
782 if( sizeof(WT) == sizeof(double) )
783 {
784 if( sizeof(T) == sizeof(double) )
785 {
786 const int vlanes = VTraits<v_float64>::vlanes();
787 v_float64 val = vx_setall_f64(*((const double*)&al));
788 for( ; j <= m - vlanes; j += vlanes )
789 {
790 v_float64 vd = vx_load((const double*)(d_buf + j));
791 v_float64 vb = vx_load((const double*)(b_data + j));
792 vd = v_muladd(vb, val, vd);
793 v_store((double*)(d_buf + j), vd);
794 }
795 }
796 else
797 {
798 const int vlanes_f32 = VTraits<v_float32>::vlanes();
799 const int vlanes_f64 = VTraits<v_float64>::vlanes();
800 v_float64 val = vx_setall_f64(*((const double*)&al));
801 for( ; j <= m - vlanes_f32; j += vlanes_f32 )
802 {
803 v_float32 vb = vx_load((const float*)(b_data + j));
804 v_float64 vb_lo = v_cvt_f64(vb);
805 v_float64 vb_hi = v_cvt_f64_high(vb);
806 v_float64 vd_lo = vx_load((const double*)(d_buf + j));
807 v_float64 vd_hi = vx_load((const double*)(d_buf + j + vlanes_f64));
808 vd_lo = v_muladd(vb_lo, val, vd_lo);
809 vd_hi = v_muladd(vb_hi, val, vd_hi);
810 v_store((double*)(d_buf + j), vd_lo);
811 v_store((double*)(d_buf + j + vlanes_f64), vd_hi);
812 }
813 }
814 }
815#endif
816 for( ; j < m; j++ )
817 d_buf[j] += WT(b_data[j])*al;
818 }
819
820 if( !c_data )
821 for( j = 0; j < m; j++ )
822 d_data[j] = T(d_buf[j]*alpha);
823 else
824 for( j = 0; j < m; j++, c_data += c_step1 )
825 {
826 WT t = d_buf[j]*alpha;
827 d_data[j] = T(t + WT(c_data[0])*beta);
828 }
829 }
830 }

847template<>
848inline void simdMulAdd<float>(
849 const float* b, double* d, int m, double a_val)
850{
851 const int vlanes_f32 = VTraits<v_float32>::vlanes();
852 const int vlanes_f64 = VTraits<v_float64>::vlanes();
853 v_float64 va = vx_setall_f64(a_val);
854 int j = 0;
855 for( ; j <= m - vlanes_f32; j += vlanes_f32 )
856 {
857 v_float32 vb = vx_load(b + j);
858 v_float64 vb_lo = v_cvt_f64(vb);
859 v_float64 vb_hi = v_cvt_f64_high(vb);
860 v_float64 vd_lo = vx_load(d + j);
861 v_float64 vd_hi = vx_load(d + j + vlanes_f64);
862 vd_lo = v_muladd(va, vb_lo, vd_lo);
863 vd_hi = v_muladd(va, vb_hi, vd_hi);
864 v_store(d + j, vd_lo);
865 v_store(d + j + vlanes_f64, vd_hi);
866 }
867 for( ; j < m; j++ )
868 d[j] += a_val * (double)b[j];
869}

871template<>
872inline void simdMulAdd<double>(
873 const double* b, double* d, int m, double a_val)
874{
875 const int vlanes = VTraits<v_float64>::vlanes();
876 const int stride2 = vlanes * 2;
877 v_float64 va = vx_setall_f64(a_val);
878 int j = 0;
879 for( ; j <= m - stride2; j += stride2 )
880 {
881 v_float64 vd0 = vx_load(d + j);
882 v_float64 vb0 = vx_load(b + j);
883 v_float64 vd1 = vx_load(d + j + vlanes);
884 v_float64 vb1 = vx_load(b + j + vlanes);
885 vd0 = v_muladd(va, vb0, vd0);
886 vd1 = v_muladd(va, vb1, vd1);
887 v_store(d + j, vd0);
888 v_store(d + j + vlanes, vd1);
889 }
890 for( ; j <= m - vlanes; j += vlanes )
891 {
892 v_float64 vd = vx_load(d + j);
893 v_float64 vb = vx_load(b + j);
894 vd = v_muladd(va, vb, vd);
895 v_store(d + j, vd);
896 }
897 for( ; j < m; j++ )
898 d[j] += a_val * b[j];
899}

914template<typename T, typename WT> static void
915GEMMBlockMul( const T* a_data, size_t a_step,
916 const T* b_data, size_t b_step,
917 WT* d_data, size_t d_step,
918 Size a_size, Size d_size, int flags )
919{
920 int i, j, k, n = a_size.width, m = d_size.width;
921 const T *_a_data = a_data, *_b_data = b_data;
922 cv::AutoBuffer<T> _a_buf;
923 T* a_buf = 0;
924 size_t a_step0, a_step1, t_step;
925 int do_acc = flags & 16;
926
927 a_step /= sizeof(a_data[0]);
928 b_step /= sizeof(b_data[0]);
929 d_step /= sizeof(d_data[0]);
930
931 a_step0 = a_step;
932 a_step1 = 1;
933
934 if( flags & GEMM_1_T )
935 {
936 CV_SWAP( a_step0, a_step1, t_step );
937 n = a_size.height;
938 _a_buf.allocate(n);
939 a_buf = _a_buf.data();
940 }
941
942 if( flags & GEMM_2_T )
943 {
944 /* second operand is transposed */
945 for( i = 0; i < d_size.height; i++, _a_data += a_step0, d_data += d_step )
946 {
947 a_data = _a_data; b_data = _b_data;
948
949 if( a_buf )
950 {
951 for( k = 0; k < n; k++ )
952 a_buf[k] = a_data[a_step1*k];
953 a_data = a_buf;
954 }
955
956 for( j = 0; j < d_size.width; j++, b_data += b_step )
957 {
958 WT s0 = do_acc ? d_data[j] : WT(0);
959 k = 0;
960#if CV_SIMD_64F && CV_GEMM_SIMD_ENABLED
961 if( sizeof(WT) == sizeof(double) )
962 {
963 if( sizeof(T) == sizeof(double) )
964 s0 += (WT)simdDotProduct((const double*)a_data, (const double*)b_data, n, k);
965 else if( sizeof(T) == sizeof(float) )
966 s0 += (WT)simdDotProduct_f32f64((const float*)a_data, (const float*)b_data, n, k);
967 }
968 else
969#endif
970 {
971 WT s1(0);
972 for( ; k <= n - 2; k += 2 )
973 {
974 s0 += WT(a_data[k])*WT(b_data[k]);
975 s1 += WT(a_data[k+1])*WT(b_data[k+1]);
976 }
977 s0 += s1;
978 }
979 for( ; k < n; k++ )
980 s0 += WT(a_data[k])*WT(b_data[k]);
981
982 d_data[j] = s0;
983 }
984 }
985 }
986 else
987 {
988 for( i = 0; i < d_size.height; i++, _a_data += a_step0, d_data += d_step )
989 {
990 a_data = _a_data, b_data = _b_data;
991
992 if( a_buf )
993 {
994 for( k = 0; k < n; k++ )
995 a_buf[k] = a_data[a_step1*k];
996 a_data = a_buf;
997 }
998#if CV_SIMD_64F && CV_GEMM_SIMD_ENABLED
999 if( sizeof(WT) == sizeof(double) )
1000 {
1001 if( sizeof(T) == sizeof(double) )
1002 simdBlockMul_kj(
1003 (const double*)a_data, (const double*)b_data, b_step,
1004 (double*)d_data, n, m, do_acc != 0);
1005 else
1006 simdBlockMul_kj(
1007 (const float*)a_data, (const float*)b_data, b_step,
1008 (double*)d_data, n, m, do_acc != 0);
1009 }
1010 else
1011#endif
1012 {
1013 for( j = 0; j <= m - 4; j += 4 )
1014 {
1015 WT s0, s1, s2, s3;
1016 const T* b = b_data + j;
1017
1018 if( do_acc )
1019 {
1020 s0 = d_data[j]; s1 = d_data[j+1];
1021 s2 = d_data[j+2]; s3 = d_data[j+3];
1022 }
1023 else
1024 s0 = s1 = s2 = s3 = WT(0);
1025
1026 for( k = 0; k < n; k++, b += b_step )
1027 {
1028 WT a(a_data[k]);
1029 s0 += a * WT(b[0]); s1 += a * WT(b[1]);
1030 s2 += a * WT(b[2]); s3 += a * WT(b[3]);
1031 }
1032
1033 d_data[j] = s0; d_data[j+1] = s1;
1034 d_data[j+2] = s2; d_data[j+3] = s3;
1035 }
1036
1037 for( ; j < m; j++ )
1038 {
1039 const T* b = b_data + j;
1040 WT s0 = do_acc ? d_data[j] : WT(0);
1041
1042 for( k = 0; k < n; k++, b += b_step )
1043 s0 += WT(a_data[k]) * WT(b[0]);
1044
1045 d_data[j] = s0;
1046 }
1047 }
1048 }
1049 }

1053template<typename T, typename WT> static void
1054GEMMStore( const T* c_data, size_t c_step,
1055 const WT* d_buf, size_t d_buf_step,
1056 T* d_data, size_t d_step, Size d_size,
1057 double alpha, double beta, int flags )
1058{
1059 const T* _c_data = c_data;
1060 int j;
1061 size_t c_step0, c_step1;
1062
1063 c_step /= sizeof(c_data[0]);
1064 d_buf_step /= sizeof(d_buf[0]);
1065 d_step /= sizeof(d_data[0]);
1066
1067 if( !c_data )
1068 c_step0 = c_step1 = 0;
1069 else if( !(flags & GEMM_3_T) )
1070 c_step0 = c_step, c_step1 = 1;
1071 else
1072 c_step0 = 1, c_step1 = c_step;
1073
1074 for( ; d_size.height--; _c_data += c_step0, d_buf += d_buf_step, d_data += d_step )
1075 {
1076 if( _c_data )
1077 {
1078 c_data = _c_data;
1079 j=0;
1080 #if CV_ENABLE_UNROLLED
1081 for(; j <= d_size.width - 4; j += 4, c_data += 4*c_step1 )
1082 {
1083 WT t0 = alpha*d_buf[j];
1084 WT t1 = alpha*d_buf[j+1];
1085 t0 += beta*WT(c_data[0]);
1086 t1 += beta*WT(c_data[c_step1]);
1087 d_data[j] = T(t0);
1088 d_data[j+1] = T(t1);
1089 t0 = alpha*d_buf[j+2];
1090 t1 = alpha*d_buf[j+3];
1091 t0 += beta*WT(c_data[c_step1*2]);
1092 t1 += beta*WT(c_data[c_step1*3]);
1093 d_data[j+2] = T(t0);
1094 d_data[j+3] = T(t1);
1095 }
1096 #endif
1097 for( ; j < d_size.width; j++, c_data += c_step1 )
1098 {
1099 WT t0 = alpha*d_buf[j];
1100 d_data[j] = T(t0 + WT(c_data[0])*beta);
1101 }
1102 }
1103 else
1104 {
1105 j = 0;
1106 #if CV_ENABLE_UNROLLED
1107 for( ; j <= d_size.width - 4; j += 4 )
1108 {
1109 WT t0 = alpha*d_buf[j];
1110 WT t1 = alpha*d_buf[j+1];
1111 d_data[j] = T(t0);
1112 d_data[j+1] = T(t1);
1113 t0 = alpha*d_buf[j+2];
1114 t1 = alpha*d_buf[j+3];
1115 d_data[j+2] = T(t0);
1116 d_data[j+3] = T(t1);
1117 }
1118 #endif
1119 for( ; j < d_size.width; j++ )
1120 d_data[j] = T(alpha*d_buf[j]);
1121 }
1122 }
1123}

1932 }
1933 }
1934 else if( scn == 3 && dcn == 1 )
1935 {
1936 for( x = 0; x < len; x++, src += 3 )
1937 dst[x] = saturate_cast<T>(m[0]*src[0] + m[1]*src[1] + m[2]*src[2] + m[3]);
