16class CUDALegacyExec : public Layer
17{
18public:
19 CUDALegacyExec(const Ptr<Layer>& impl_, void* ctx_) : impl(impl_), ctx(ctx_) {}
20
21 static Ptr<Layer> create(const Ptr<LayerInfo>& data, void* backendCtx)
22 {
23 Ptr<Layer> impl = data.dynamicCast<Layer>();
24 if (!impl || !backendCtx || !impl->supportBackend(DNN_BACKEND_CUDA))
25 return Ptr<Layer>(); // unsupported -> CPU fallback
26 Ptr<CUDALegacyExec> e(new CUDALegacyExec(impl, backendCtx));
27 e->name = impl->name;
28 e->type = impl->type;
29 e->inputs = impl->inputs;
30 e->outputs = impl->outputs;
31 return e;
32 }
33
34 void finalize(InputArrayOfArrays inputs, OutputArrayOfArrays outputs) CV_OVERRIDE
35 {
36 impl->finalize(inputs, outputs);
37 }
38
39 void forwardCUDA(InputArrayOfArrays inputs_,
40 OutputArrayOfArrays outputs_,
41 void* workspace) CV_OVERRIDE
42 {
43 std::vector<cuda::GpuMatND> inputs, outputs;
44 inputs_.getGpuMatNDVector(inputs);
45 outputs_.getGpuMatNDVector(outputs);
46
47 cuda4dnn::csl::Workspace& ws = *reinterpret_cast<cuda4dnn::csl::Workspace*>(workspace);
48 if (!node) {
49 impl->preferableTarget = preferableTarget; // initCUDA may pick FP16/FP32 by target
50 cuda4dnn::csl::CSLContext context = *reinterpret_cast<cuda4dnn::csl::CSLContext*>(ctx);
51 node = impl->initCUDA(&context, inputs_, outputs_);
52 CV_Assert(node);
53 cudaNode = node.dynamicCast<CUDABackendNode>();
54 CV_Assert(cudaNode);
55 ws.require(cudaNode->get_workspace_memory_in_bytes());
56 }
57 cudaNode->forward(inputs, outputs, ws);
58 }
59
60 Ptr<Layer> impl;
61 void* ctx;
62 Ptr<BackendNode> node;
63 Ptr<CUDABackendNode> cudaNode;
64};

79void Net::Impl::initCUDABackend(const std::vector<LayerPin>& blobsToKeep_)
80{
81 CV_Assert(preferableBackend == DNN_BACKEND_CUDA);
82
83 if (!cudaInfo) /* we need to check only once */
84 cuda4dnn::checkVersions();
85
86 if (cuda4dnn::getDeviceCount() <= 0)
87 CV_Error(Error::StsError, "No CUDA capable device found.");
88
89 if (cuda4dnn::getDevice() < 0)
90 CV_Error(Error::StsError, "No CUDA capable device selected.");
91
92 if (!cuda4dnn::isDeviceCompatible())
93 CV_Error(Error::GpuNotSupported, "OpenCV was not built to work with the selected device. Please check CUDA_ARCH_PTX or CUDA_ARCH_BIN in your build configuration.");
94
95 if (preferableTarget == DNN_TARGET_CUDA_FP16 && !cuda4dnn::doesDeviceSupportFP16())
96 {
97 CV_LOG_WARNING(NULL, "The selected CUDA device does not support FP16 target; switching to FP32 target.");
98 preferableTarget = DNN_TARGET_CUDA;
99 }
100
101 if (!cudaInfo)
102 {
103 cuda4dnn::csl::CSLContext context;
104 context.stream = cuda4dnn::csl::Stream(true);
105 context.cublas_handle = cuda4dnn::csl::cublas::Handle(context.stream);
106 context.cudnn_handle = cuda4dnn::csl::cudnn::Handle(context.stream);
107
108 auto d2h_stream = cuda4dnn::csl::Stream(true); // stream for background D2H data transfers
109 cudaInfo = std::unique_ptr<CudaInfo_t>(new CudaInfo_t(std::move(context), std::move(d2h_stream)));
110 }
111
112 cudaInfo->workspace = cuda4dnn::csl::Workspace(); // release workspace memory if any
113
114 for (auto& layer : layers)
115 {
116 auto& ld = layer.second;
117
118 if (ld.id == 0 && netInputLayer->supportBackend(preferableBackend))
119 {
120 for (auto& wrapper : ld.inputBlobsWrappers)
121 {
122 auto cudaWrapper = wrapper.dynamicCast<CUDABackendWrapper>();
123 cudaWrapper->setStream(cudaInfo->context.stream, cudaInfo->d2h_stream);
124 }
125 }
126
127 for (auto& wrapper : ld.outputBlobsWrappers)
128 {
129 auto cudaWrapper = wrapper.dynamicCast<CUDABackendWrapper>();
130 cudaWrapper->setStream(cudaInfo->context.stream, cudaInfo->d2h_stream);
131 }
132 }
133
134 for (auto& layer : layers)
135 {
136 auto& ld = layer.second;
137 auto& layerInstance = ld.layerInstance;
138
139 if (!layerInstance->supportBackend(DNN_BACKEND_CUDA))
140 {
141 std::ostringstream os;
142 os << "CUDA backend will fallback to the CPU implementation for the layer \"" << ld.name
143 << "\" of type " << ld.type << '\n';
144 CV_LOG_INFO(NULL, os.str().c_str());
145 continue;
146 }
147
148 /* we make a copy so that `initCUDA` doesn't modify `cudaInfo->context` */
149 auto context = cudaInfo->context;
150 auto node = layerInstance->initCUDA(&context, ld.inputBlobsWrappers, ld.outputBlobsWrappers);
151 ld.backendNodes[DNN_BACKEND_CUDA] = node;
152
153 if(!node.empty())
154 {
155 auto cudaNode = node.dynamicCast<CUDABackendNode>();
156 cudaInfo->workspace.require(cudaNode->get_workspace_memory_in_bytes());
157 }
158 }
159
160 if (blobsToKeep_.size() > 1)
161 {
162 for (const auto& pin : blobsToKeep_)
163 {
164 LayerData& ld = layers[pin.lid];
165 ld.cudaD2HBackgroundTransfers.push_back(pin.oid);
166 }
167 }
168}
