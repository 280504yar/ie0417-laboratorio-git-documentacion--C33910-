47Color Color::to(const ColorSpaceBase& other, ChromaticAdaptationType method, bool save)
48{
49 auto it = history.find(other);
50 if ( it != history.end() )
51 {
52 return *(it->second);
53 }
54 if (cs->relate(other))
55 {
56 return Color(cs->relation(other).run(colors), other);
57 }
58 Operations ops;
59 ops.add(cs->to).add(XYZ(cs->illumobserver).cam(other.illumobserver, method)).add(other.from);
60 Mat converted = ops.run(colors);
61 if (save)
62 {
63 auto ptr = std::make_shared<Color>(converted, other);
64 history[other] = ptr;
65 return *ptr;
66 }
67 else
68 {
69 return Color(converted, other);
70 }
71}

102Mat Color::diff(Color& other, const IllumObserver& illumobserver, DistanceType method)
103{
104 Lab lab = *Lab::get(illumobserver);
105 switch (method)
106 {
107 case cv::ccm::DISTANCE_CIE76:
108 case cv::ccm::DISTANCE_CIE94_GRAPHIC_ARTS:
109 case cv::ccm::DISTANCE_CIE94_TEXTILES:
110 case cv::ccm::DISTANCE_CIE2000:
111 case cv::ccm::DISTANCE_CMC_1TO1:
112 case cv::ccm::DISTANCE_CMC_2TO1:
113 return distance(to(lab).colors, other.to(lab).colors, method);
114 case cv::ccm::DISTANCE_RGB:
115 return distance(to(*cs->nl).colors, other.to(*cs->nl).colors, method);
116 case cv::ccm::DISTANCE_RGBL:
117 return distance(to(*cs->l).colors, other.to(*cs->l).colors, method);
118 default:
119 CV_Error(Error::StsBadArg, "Wrong method!" );
120 break;
121 }
122}

124void Color::getGray(double JDN)
125{
126 if (!grays.empty())
127 {
128 return;
129 }
130 Mat lab = to(COLOR_SPACE_LAB_D65_2).colors;
131 Mat gray(colors.size(), colors.type());
132 int fromto[] = { 0, 0, -1, 1, -1, 2 };
133 mixChannels(&lab, 1, &gray, 1, fromto, 3);
134 Mat d = distance(lab, gray, DISTANCE_CIE2000);
135 this->grays = d < JDN;
136 this->colored = ~grays;
137}
