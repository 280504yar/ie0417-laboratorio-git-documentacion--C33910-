124void cv::cuda::fastNlMeansDenoising(InputArray _src, OutputArray _dst, float h, int search_window, int block_window, Stream& stream)
125{
126 const GpuMat src = _src.getGpuMat();
127
128 CV_Assert(src.depth() == CV_8U && src.channels() < 4);
129
130 int border_size = search_window/2 + block_window/2;
131 Size esize = src.size() + Size(border_size, border_size) * 2;
132
133 BufferPool pool(stream);
134
135 GpuMat extended_src = pool.getBuffer(esize, src.type());
136 cv::cuda::copyMakeBorder(src, extended_src, border_size, border_size, border_size, border_size, cv::BORDER_DEFAULT, Scalar(), stream);
137 GpuMat src_hdr = extended_src(Rect(Point2i(border_size, border_size), src.size()));
138
139 int bcols, brows;
140 device::imgproc::nln_fast_get_buffer_size(src_hdr, search_window, block_window, bcols, brows);
141 GpuMat buffer = pool.getBuffer(brows, bcols, CV_32S);
142
143 using namespace cv::cuda::device::imgproc;
144 typedef void (*nlm_fast_t)(const PtrStepSzb&, PtrStepSzb, PtrStepi, int, int, float, cudaStream_t);
145 static const nlm_fast_t funcs[] = { nlm_fast_gpu<uchar>, nlm_fast_gpu<uchar2>, nlm_fast_gpu<uchar3>, 0};
146
147 _dst.create(src.size(), src.type());
148 GpuMat dst = _dst.getGpuMat();
149
150 funcs[src.channels()-1](src_hdr, dst, buffer, search_window, block_window, h, StreamAccessor::getStream(stream));
151}

153void cv::cuda::fastNlMeansDenoisingColored(InputArray _src, OutputArray _dst, float h_luminance, float h_color, int search_window, int block_window, Stream& stream)
154{
155 const GpuMat src = _src.getGpuMat();
156
157 CV_Assert(src.type() == CV_8UC3);
158
159 BufferPool pool(stream);
160
161 GpuMat lab = pool.getBuffer(src.size(), src.type());
162 cv::cuda::cvtColor(src, lab, cv::COLOR_BGR2Lab, 0, stream);
163
164 GpuMat l = pool.getBuffer(src.size(), CV_8U);
165 GpuMat ab = pool.getBuffer(src.size(), CV_8UC2);
166 device::imgproc::fnlm_split_channels(lab, l, ab, StreamAccessor::getStream(stream));
167
168 fastNlMeansDenoising(l, l, h_luminance, search_window, block_window, stream);
169 fastNlMeansDenoising(ab, ab, h_color, search_window, block_window, stream);
170
171 device::imgproc::fnlm_merge_channels(l, ab, lab, StreamAccessor::getStream(stream));
172 cv::cuda::cvtColor(lab, _dst, cv::COLOR_Lab2BGR, 0, stream);
173}
