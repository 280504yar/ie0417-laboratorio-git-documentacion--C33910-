18 inline void checkVersions()
19 {
20 // https://docs.nvidia.com/deeplearning/cudnn/developer-guide/index.html#programming-model
21 // cuDNN API Compatibility
22 // Beginning in cuDNN 7, the binary compatibility of a patch and minor releases is maintained as follows:
23 // Any patch release x.y.z is forward or backward-compatible with applications built against another cuDNN patch release x.y.w (meaning, of the same major and minor version number, but having w!=z).
24 // cuDNN minor releases beginning with cuDNN 7 are binary backward-compatible with applications built against the same or earlier patch release (meaning, an application built against cuDNN 7.x is binary compatible with cuDNN library 7.y, where y>=x).
25 // Applications compiled with a cuDNN version 7.y are not guaranteed to work with 7.x release when y > x.
26 int cudnn_bversion = cudnnGetVersion();
27 int cudnn_major_bversion = 0, cudnn_minor_bversion = 0;
28 // CuDNN changed major version multiplier in 9.0
29 if (cudnn_bversion >= 9*10000)
30 {
31 cudnn_major_bversion = cudnn_bversion / 10000;
32 cudnn_minor_bversion = cudnn_bversion % 10000 / 100;
33 }
34 else
35 {
36 cudnn_major_bversion = cudnn_bversion / 1000;
37 cudnn_minor_bversion = cudnn_bversion % 1000 / 100;
38 }
39 if (cudnn_major_bversion != CUDNN_MAJOR || cudnn_minor_bversion < CUDNN_MINOR)
40 {
41 std::ostringstream oss;
42 oss << "cuDNN reports version " << cudnn_major_bversion << "." << cudnn_minor_bversion << " which is not compatible with the version " << CUDNN_MAJOR << "." << CUDNN_MINOR << " with which OpenCV was built";
43 CV_LOG_WARNING(NULL, oss.str().c_str());
44 }
45 }

59 inline bool isDeviceCompatible(int device_id = -1)
60 {
61 if (device_id < 0)
62 device_id = getDevice();
63
64 if (device_id < 0)
65 return false;
66
67 int major = 0, minor = 0;
68 CUDA4DNN_CHECK_CUDA(cudaDeviceGetAttribute(&major, cudaDevAttrComputeCapabilityMajor, device_id));
69 CUDA4DNN_CHECK_CUDA(cudaDeviceGetAttribute(&minor, cudaDevAttrComputeCapabilityMinor, device_id));
70
71 if (cv::cuda::TargetArchs::hasEqualOrLessPtx(major, minor))
72 return true;
73
74 for (int i = minor; i >= 0; i--)
75 if (cv::cuda::TargetArchs::hasBin(major, i))
76 return true;
77
78 return false;
79 }

81 inline bool doesDeviceSupportFP16(int device_id = -1)
82 {

84 device_id = getDevice();
85
86 if (device_id < 0)
87 return false;
88
89 int major = 0, minor = 0;
90 CUDA4DNN_CHECK_CUDA(cudaDeviceGetAttribute(&major, cudaDevAttrComputeCapabilityMajor, device_id));
91 CUDA4DNN_CHECK_CUDA(cudaDeviceGetAttribute(&minor, cudaDevAttrComputeCapabilityMinor, device_id));
92
93 int version = major * 10 + minor;
94 return (version >= 53);
