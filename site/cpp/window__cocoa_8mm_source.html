189void showImageImpl( const char* name, cv::InputArray arr)
190{
191 @autoreleasepool {
192 CVWindow *window = cvGetWindow(name);
193 if(!window)
194 {
195 namedWindowImpl(name, cv::WINDOW_AUTOSIZE);
196 window = cvGetWindow(name);
197 }
198
199 if(window)
200 {
201 bool empty = [[window contentView] image] == nil;
202 NSRect vrectOld = [[window contentView] frame];
203 NSSize oldImageSize = [[[window contentView] image] size];
204 [[window contentView] setImageData:arr];
205 if([window autosize] || [window firstContent] || empty)
206 {
207 NSSize imageSize = [[[window contentView] image] size];
208 // Only adjust the image size if the new image is a different size from the previous
209 if (oldImageSize.height != imageSize.height || oldImageSize.width != imageSize.width)
210 {
211 //Set new view size considering sliders (reserve height and min width)
212 NSSize scaledImageSize = imageSize;
213 if ([[window contentView] respondsToSelector:@selector(convertSizeFromBacking:)])
214 {
215 // Only resize for retina displays if the image is bigger than the screen
216 NSSize screenSize = NSScreen.mainScreen.visibleFrame.size;
217 CGFloat titleBarHeight = window.frame.size.height - [window contentRectForFrameRect:window.frame].size.height;
218 screenSize.height -= titleBarHeight;
219 if (imageSize.width > screenSize.width || imageSize.height > screenSize.height)
220 {
221 CGFloat fx = screenSize.width/std::max(imageSize.width, (CGFloat)1.f);
222 CGFloat fy = screenSize.height/std::max(imageSize.height, (CGFloat)1.f);
223 CGFloat min_f = std::min(fx, fy);
224 scaledImageSize = [[window contentView] convertSizeFromBacking:imageSize];
225 scaledImageSize.width = std::min(scaledImageSize.width, min_f*imageSize.width);
226 scaledImageSize.height = std::min(scaledImageSize.height, min_f*imageSize.height);
227 }
228 }
229 NSSize contentSize = vrectOld.size;
230 contentSize.height = scaledImageSize.height + [window contentView].sliderHeight;
231 contentSize.width = std::max<int>(scaledImageSize.width, MIN_SLIDER_WIDTH);
232 [window setContentSize:contentSize]; //adjust sliders to fit new window size
233 if([window firstContent])
234 {
235 int x = [window x0];
236 int y = [window y0];
237 if(x >= 0 && y >= 0)
238 {
239 NSRect visibleFrame = [[window screen] visibleFrame];
240 y = NSMaxY(visibleFrame) - y;
241 [window setFrameTopLeftPoint:NSMakePoint(x, y)];
242 }
243 }
244 }
245 }
246 [window setFirstContent:NO];
247 }
248 }
249}

436int namedWindowImpl( const char* name, int flags )
437{
438 if( !wasInitialized )
439 cocoa_InitSystem(0, 0);
440
441 @autoreleasepool {
442 CVWindow *window = cvGetWindow(name);
443 if( window )
444 {
445 [window setAutosize:(flags == cv::WINDOW_AUTOSIZE)];
446 return 0;
447 }
448
449 NSScreen* mainDisplay = [NSScreen mainScreen];
450 NSString *windowName = [NSString stringWithFormat:@"%s", name];
451 NSUInteger showResize = NSResizableWindowMask;
452 NSUInteger styleMask = NSTitledWindowMask|NSMiniaturizableWindowMask|showResize;
453 CGFloat windowWidth = [NSWindow minFrameWidthWithTitle:windowName styleMask:styleMask];
454 NSRect initContentRect = NSMakeRect(0, 0, windowWidth, 0);
455 if (mainDisplay) {
456 NSRect dispFrame = [mainDisplay visibleFrame];
457 initContentRect.origin.y = dispFrame.size.height-20;
458 }
459
460 window = [[CVWindow alloc] initWithContentRect:initContentRect
461 styleMask:NSTitledWindowMask|NSMiniaturizableWindowMask|showResize
462 backing:NSBackingStoreBuffered
463 defer:YES
464 screen:mainDisplay];
465
466 [window setFrameTopLeftPoint:initContentRect.origin];
467
468 [window setFirstContent:YES];
469 [window setX0:-1];
470 [window setY0:-1];
471
472 [window setContentView:[[CVView alloc] init]];
473
474 [NSApp activateIgnoringOtherApps:YES];
475
476 [window setHasShadow:YES];
477 [window setAcceptsMouseMovedEvents:YES];
478 [window useOptimizedDrawing:YES];
479 [window setTitle:windowName];
480 [window makeKeyAndOrderFront:nil];
481
482 [window setAutosize:(flags == cv::WINDOW_AUTOSIZE)];
483
484 [windows setValue:window forKey:windowName];
485
486 return [windows count]-1;
487 }
488}

490int waitKeyImpl (int maxWait)
491{
492 int returnCode = -1;
493 @autoreleasepool {
494 double start = [[NSDate date] timeIntervalSince1970];
495
496 while(true) {
497 if(([[NSDate date] timeIntervalSince1970] - start) * 1000 >= maxWait && maxWait>0)
498 break;
499
500 NSEvent *event =
501 [application
502 nextEventMatchingMask:NSAnyEventMask
503 untilDate://[NSDate dateWithTimeIntervalSinceNow: 1./100]
504 [NSDate distantPast]
505 inMode:NSDefaultRunLoopMode
506 dequeue:YES];
507
508 if([event type] == NSKeyDown && [[event characters] length]) {
509 returnCode = [[event characters] characterAtIndex:0];
510 break;
511 }
512
513 [application sendEvent:event];
514 [application updateWindows];
515
516 [NSThread sleepForTimeInterval:1/100.];
517 }
518
519 return returnCode;
520 }
521}

572void cvSetModeWindow_COCOA( const char* name, double prop_value )
573{
574 CVWindow *window = nil;
575
576#if MAC_OS_X_VERSION_MAX_ALLOWED < MAC_OS_X_VERSION_10_7
577 NSDictionary *fullscreenOptions = nil;
578#endif
579
580 if( name == NULL )
581 {
582 CV_Error( cv::Error::StsNullPtr, "NULL name string" );
583 }
584
585 window = cvGetWindow(name);
586 if ( window == NULL )
587 {
588 CV_Error( cv::Error::StsNullPtr, "NULL window" );
589 }
590
591 if ( [window autosize] )
592 {
593 return;
594 }
595
596 @autoreleasepool {
597#if MAC_OS_X_VERSION_MAX_ALLOWED > MAC_OS_X_VERSION_10_6
598 if ( ([window styleMask] & NSFullScreenWindowMask) && prop_value==cv::WINDOW_NORMAL )
599 {
600 [window toggleFullScreen:nil];
601
602 window.status=cv::WINDOW_NORMAL;
603 }
604 else if( !([window styleMask] & NSFullScreenWindowMask) && prop_value==cv::WINDOW_FULLSCREEN )
605 {
606 [window setCollectionBehavior:NSWindowCollectionBehaviorFullScreenPrimary];
607
608 NSScreen* screen = [window screen];
609
610 NSRect frame = [screen frame];
611 [window setFrame:frame display:YES];
612
613 [window setContentSize:frame.size];
614
615 [window toggleFullScreen:nil];
616
617 [window setFrameTopLeftPoint: frame.origin];
618
619 window.status=cv::WINDOW_FULLSCREEN;
620 }
621#else
622 fullscreenOptions = [NSDictionary dictionaryWithObject:[NSNumber numberWithBool:YES] forKey:NSFullScreenModeSetting];
623 if ( [[window contentView] isInFullScreenMode] && prop_value==cv::WINDOW_NORMAL )
624 {
625 [[window contentView] exitFullScreenModeWithOptions:fullscreenOptions];
626 window.status=cv::WINDOW_NORMAL;
627 }
628 else if( ![[window contentView] isInFullScreenMode] && prop_value==cv::WINDOW_FULLSCREEN )
629 {
630 [[window contentView] enterFullScreenMode:[NSScreen mainScreen] withOptions:fullscreenOptions];
631 window.status=cv::WINDOW_FULLSCREEN;
632 }
633#endif
634 }
635}

730static NSSize constrainAspectRatio(NSSize base, NSSize constraint) {
731 CGFloat heightDiff = (base.height / constraint.height);
732 CGFloat widthDiff = (base.width / constraint.width);
733 if (heightDiff == 0) heightDiff = widthDiff;
734 if (widthDiff == heightDiff) {
735 return base;
736 }
737 else if (widthDiff > heightDiff) {
738 NSSize out = { constraint.width / constraint.height * base.height, base.height };
739 return out;
740 }
741 else {
742 NSSize out = { base.width, constraint.height / constraint.width * base.width };
743 return out;
744 }
745}

759- (void)cvSendMouseEvent:(NSEvent *)event type:(int)type flags:(int)flags {
760 (void)event;
761 NSPoint mp = [NSEvent mouseLocation];
762 mp = [self convertScreenToBase: mp];
763 CVView *contentView = [self contentView];
764 NSSize viewSize = contentView.frame.size;
765 if (contentView.imageView) {
766 viewSize = contentView.imageView.frame.size;
767 }
768 else {
769 viewSize.height -= contentView.sliderHeight;
770 }
771 mp.y = viewSize.height - mp.y;
772
773 NSSize imageSize = contentView.image.size;
774 mp.y *= (imageSize.height / std::max(viewSize.height, 1.));
775 mp.x *= (imageSize.width / std::max(viewSize.width, 1.));
776
777 if( [event type] == NSEventTypeScrollWheel ) {
778 if( event.hasPreciseScrollingDeltas ) {
779 mp.x = int(event.scrollingDeltaX);
780 mp.y = int(event.scrollingDeltaY);
781 } else {
782 mp.x = int(event.scrollingDeltaX / 0.100006);
783 mp.y = int(event.scrollingDeltaY / 0.100006);
784 }
785 if( mp.x && !mp.y && cv::EVENT_MOUSEWHEEL == type ) {
786 type = cv::EVENT_MOUSEHWHEEL;
787 }
788 mouseCallback(type, mp.x, mp.y, flags, mouseParam);
789 } else if( mp.x >= 0 && mp.y >= 0 && mp.x < imageSize.width && mp.y < imageSize.height ) {
790 mouseCallback(type, mp.x, mp.y, flags, mouseParam);
