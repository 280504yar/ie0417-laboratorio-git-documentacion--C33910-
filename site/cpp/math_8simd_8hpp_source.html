5// Element-wise MATH kernels (sqrt/exp/log/sin/cos/tanh/erf/relu) and SELECT for the new arithmetic
6// engine, SIMD-dispatched per CPU baseline - the unary/ternary sibling of arithm.simd.hpp.
7//
8// This file is compiled once per SIMD baseline (registered via ocv_add_dispatched_file). The per-op
9// entry points get*Func_(...) live in cv::ew::CV_CPU_OPTIMIZATION_NAMESPACE and return the kernel
10// optimized for that baseline; the regular get*Func dispatchers live in math.dispatch.cpp.
11//
12// Kernel shape (house style of arithm.simd.hpp):
13// - one 2D tile; per-row outer loop with stepy (bytes); dst contiguous in x; stepx in {0,1}.
14// - continuity collapse 2D->1D when every operand+dst is gap-free.
15// - halide right-edge backoff for the SIMD tail, SUPPRESSED when dst aliases an input (in-place
16// unary math would re-apply Op to already-written values).
17//
18// Math is T -> T over the four float depths: f32/f64 compute natively (v_exp & co exist for both);
19// f16/bf16 ride the f32 hub (vx_load_pair_as widens one native vector into two f32 vectors, the
20// saturating v_store_pair_as packs them back) - more accurate than a native f16 polynomial and
21// works on every baseline. Integer inputs never reach these kernels: emitUnary computes integer
22// math in the float domain and casts.
23

62// Op functors: vec(Wvec) over the work vector (f32 or f64), scl(WT) for the scalar path/tail.
63// New unary math ops slot in here.

65struct MSqrt {
66 template<typename V> static V vec(const V& x) { return v_sqrt(x); }
67 template<typename W> static W scl(W x) { return std::sqrt(x); }
68};
69struct MExp {
70 template<typename V> static V vec(const V& x) { return v_exp(x); }
71 template<typename W> static W scl(W x) { return std::exp(x); }
72};
73struct MLog {
74 template<typename V> static V vec(const V& x) { return v_log(x); }
75 template<typename W> static W scl(W x) { return std::log(x); }
76};
77struct MSin {
78 template<typename V> static V vec(const V& x) { return v_sin(x); }
79 template<typename W> static W scl(W x) { return std::sin(x); }
80};
81struct MCos {
82 template<typename V> static V vec(const V& x) { return v_cos(x); }
83 template<typename W> static W scl(W x) { return std::cos(x); }
84};
85// tanh(x) = (e^2x - 1) / (e^2x + 1), on top of v_exp (no v_tanh intrinsic). The input is clamped
86// first: tanh saturates to +/-1 well inside |x| <= 10 (f32) / 20 (f64), while an unclamped large x
87// would push e^2x to inf and the ratio to inf/inf = NaN. (A NaN input may map to a saturated value
88// on some ISAs instead of NaN - the polynomial v_exp has relaxed NaN semantics anyway.)
89struct MTanh {
90 static v_float32 vec(const v_float32& x)
91 {
92 const v_float32 one = vx_setall_f32(1.f), lim = vx_setall_f32(10.f);
93 v_float32 cx = v_min(v_max(x, v_sub(vx_setzero_f32(), lim)), lim);
94 v_float32 e = v_exp(v_add(cx, cx));
95 return v_div(v_sub(e, one), v_add(e, one));
96 }
97#if CV_SIMD_64F || CV_SIMD_SCALABLE_64F
98 static v_float64 vec(const v_float64& x)
99 {
100 const v_float64 one = vx_setall_f64(1.), lim = vx_setall_f64(20.);
101 v_float64 cx = v_min(v_max(x, v_sub(vx_setzero_f64(), lim)), lim);
102 v_float64 e = v_exp(v_add(cx, cx));
103 return v_div(v_sub(e, one), v_add(e, one));
104 }
105#endif
106 template<typename W> static W scl(W x) { return std::tanh(x); }
107};
108struct MErf {
109 static v_float32 vec(const v_float32& x) { return v_erf(x); }
110#if CV_SIMD_64F || CV_SIMD_SCALABLE_64F
111 // no f64 SIMD erf primitive: apply std::erf per lane (keeps the one kernel shape; the
112 // store/load round-trip is noise next to libm erf itself)
113 static v_float64 vec(const v_float64& x)
114 {
115 double buf[VTraits<v_float64>::max_nlanes];
116 v_store(buf, x);
117 for (int i = 0; i < VTraits<v_float64>::vlanes(); i++) buf[i] = std::erf(buf[i]);
118 return vx_load(buf);
119 }
120#endif
121 template<typename W> static W scl(W x) { return std::erf(x); }
122};
123struct MRelu {
124 template<typename V> static V vec(const V& x) { return v_max(x, v_setzero_<V>()); }
125 template<typename W> static W scl(W x) { return x > W(0) ? x : W(0); }
126};

128// ===========================================================================
129// The unary kernel: dst = Op(src), T -> T. Wvec picks the work vector: v_float32 / v_float64 for
130// the native depths, v_float32 for the f16/bf16 hub (vx_load_pair_as does the widening).
131// ===========================================================================
132template<typename T, typename Wvec, class Op>
133static int vecUnaryKernel(const void* src0_, size_t s0y, size_t s0x,
134 const void*, size_t, size_t, const void*, size_t, size_t,
135 void* dst_, size_t dsty, int width, int height,
136 const double*, int, void*)
137{
138 s0y /= sizeof(T);
139 dsty /= sizeof(T);
140 CV_Assert(s0x <= 1u);
141
142 const T* src0 = (const T*)src0_;
143 T* dst = (T*)dst_;
144 using WT = typename VTraits<Wvec>::lane_type;
145
146 if (height > 1 && dsty == (size_t)width && s0y == s0x*(size_t)width) { width *= height; height = 1; }
147
148 // vertical broadcast (a row expanded into a matrix, s0y == 0): every output row is identical -
149 // compute the first one, memcpy the rest (transcendentals cost far more than a row copy)
150 const int urows = (s0y == 0 && height > 1) ? 1 : height;
151
152 for (int y = 0; y < urows; y++, src0 += s0y, dst += dsty)
153 {
154 if (s0x == 0) // broadcast-scalar source: one value covers the row
155 {
156 T v = saturate_cast<T>(Op::scl((WT)src0[0]));
157 for (int x = 0; x < width; x++) dst[x] = v;
158 continue;
159 }
160 int x = 0;
161#if (CV_SIMD || CV_SIMD_SCALABLE)
162 const int VECSZ = VTraits<Wvec>::vlanes();
163 // in-place (dst == src) forbids the right-edge backoff: it would re-read already-written
164 // values and apply Op twice. Those rows finish in the scalar tail instead.
165 const bool use_tail_trick = width >= VECSZ*4 && src0_ != dst_;
166 for (; x < width; x += VECSZ*2)
167 {
168 if (x + VECSZ*2 > width) { if (!use_tail_trick || x == 0) break; x = width - VECSZ*2; }
169 Wvec a0, a1;
170 vx_load_pair_as(src0 + x, a0, a1);
171 a0 = Op::vec(a0); a1 = Op::vec(a1);
172 v_store_pair_as(dst + x, a0, a1);
173 }
174#endif
175 for (; x < width; x++)
176 dst[x] = saturate_cast<T>(Op::scl((WT)src0[x]));
177 }
178 dst = (T*)dst_;
179 for (int y = urows; y < height; y++)
180 memcpy(dst + (size_t)y*dsty, dst, (size_t)width*sizeof(T));
181 return 0;
182}

184// Scalar-only variant for (op, depth) pairs with no SIMD primitive (erf on f64; every op's f64
185// when the baseline has no 64-bit float SIMD).
186template<typename T, typename WT, class Op>
187static int scalarUnaryKernel(const void* src0_, size_t s0y, size_t s0x,
188 const void*, size_t, size_t, const void*, size_t, size_t,
189 void* dst_, size_t dsty, int width, int height,
190 const double*, int, void*)
191{
192 s0y /= sizeof(T);
193 dsty /= sizeof(T);
194 CV_Assert(s0x <= 1u);
195 const T* src0 = (const T*)src0_;
196 T* dst = (T*)dst_;
197 if (height > 1 && dsty == (size_t)width && s0y == s0x*(size_t)width) { width *= height; height = 1; }
198 const int urows = (s0y == 0 && height > 1) ? 1 : height; // vertical broadcast: 1 row + copies
199 for (int y = 0; y < urows; y++, src0 += s0y, dst += dsty)
200 {
201 if (s0x == 0)
202 {
203 T v = saturate_cast<T>(Op::scl((WT)src0[0]));
204 for (int x = 0; x < width; x++) dst[x] = v;
205 continue;
206 }
207 for (int x = 0; x < width; x++)
208 dst[x] = saturate_cast<T>(Op::scl((WT)src0[x]));
209 }
210 dst = (T*)dst_;
211 for (int y = urows; y < height; y++)
212 memcpy(dst + (size_t)y*dsty, dst, (size_t)width*sizeof(T));
213 return 0;
214}

216// ===========================================================================
217// OP_POW: dst = pow(x, y), T x T -> T over the float depths. Exact std::pow semantics.
218//
219// The exponent is USUALLY a broadcast scalar (pow(x, 2), texpr literals ride as 0-dim consts with
220// stepx == 0) - dispatched PER ROW to the important special cases: y==2 -> x*x, y==3 -> x*x*x,
221// y==0.5 -> v_sqrt, y==1 -> copy, y==0 -> fill 1 (std::pow(anything, 0) == 1, NaN included).
222// Everything else - and the per-element exponent - runs the general vectorized path
223// exp(y * log(x)), which is only valid for x > 0: any lane with x <= 0 falls back to scalar
224// std::pow for the whole vector pair (v_check_any per pair; negative/zero bases are rare, and the
225// scalar path preserves every std::pow subtlety - signed results for integer y, NaN for
226// fractional y, the x == 0 family). One knowing deviation: y==0.5 uses v_sqrt, so pow(-0., .5)
227// returns -0. instead of std::pow's +0.
228//
229// The halide right-edge tail backoff is used in every SIMD loop, SUPPRESSED when dst aliases an
230// input: pow is not idempotent, so an in-place backoff would re-read already-written values (the
231// overlap region is otherwise just recomputed from the untouched source). Suppressed rows finish
232// in the scalar tail.
233#if (CV_SIMD || CV_SIMD_SCALABLE)
234static inline v_float32 vxSetallW(float v, const v_float32&) { return vx_setall_f32(v); }
235#if CV_SIMD_64F || CV_SIMD_SCALABLE_64F
236static inline v_float64 vxSetallW(double v, const v_float64&) { return vx_setall_f64(v); }
237#endif
238#endif

240// Plain scalar pow for baselines without the needed SIMD float width (f64 without 64-bit SIMD).
241template<typename T, typename WT>
242static int scalarPowKernel(const void* src0_, size_t s0y, size_t s0x,
243 const void* src1_, size_t s1y, size_t s1x,
244 const void*, size_t, size_t,
245 void* dst_, size_t dsty, int width, int height,
246 const double*, int, void*)
247{
248 s0y /= sizeof(T); s1y /= sizeof(T); dsty /= sizeof(T);
249 CV_Assert(s0x <= 1u && s1x <= 1u);
250 const T* src0 = (const T*)src0_;
251 const T* src1 = (const T*)src1_;
252 T* dst = (T*)dst_;
253 if (height > 1 && dsty == (size_t)width && s0y == s0x*(size_t)width && s1y == s1x*(size_t)width)
254 { width *= height; height = 1; }
255 for (int y = 0; y < height; y++, src0 += s0y, src1 += s1y, dst += dsty)
256 for (int x = 0; x < width; x++)
257 dst[x] = saturate_cast<T>(std::pow((WT)src0[x*s0x], (WT)src1[x*s1x]));
258 return 0;
259}

261template<typename T, typename Wvec>
262static int powKernel(const void* src0_, size_t s0y, size_t s0x,
263 const void* src1_, size_t s1y, size_t s1x,
264 const void*, size_t, size_t,
265 void* dst_, size_t dsty, int width, int height,
266 const double*, int, void*)
267{
268 s0y /= sizeof(T);
269 s1y /= sizeof(T);
270 dsty /= sizeof(T);
271 CV_Assert(s0x <= 1u && s1x <= 1u);
272
273 const T* src0 = (const T*)src0_;
274 const T* src1 = (const T*)src1_;
275 T* dst = (T*)dst_;
276 using WT = typename VTraits<Wvec>::lane_type;
277
278 if (height > 1 && dsty == (size_t)width && s0y == s0x*(size_t)width && s1y == s1x*(size_t)width)
279 { width *= height; height = 1; }
280
281 [[maybe_unused]] const bool tail_trick = src0_ != dst_ && src1_ != dst_;
282
283 // both operands vertically broadcast: every output row is identical - compute one, copy
284 const int urows = (s0y == 0 && s1y == 0 && height > 1) ? 1 : height;
285
286 for (int y = 0; y < urows; y++, src0 += s0y, src1 += s1y, dst += dsty)
287 {
288 int x = 0;
289 if (s1x == 0) // scalar exponent for this row
290 {
291 const WT p = (WT)src1[0];
292#if (CV_SIMD || CV_SIMD_SCALABLE)
293 const int VECSZ = VTraits<Wvec>::vlanes();
294 if (s0x == 1)
295 {
296 if (p == WT(2) || p == WT(3))
297 {
298 for (; x < width; x += VECSZ*2)
299 {
300 if (x + VECSZ*2 > width) { if (!tail_trick || x == 0) break; x = width - VECSZ*2; }
301 Wvec a0, a1;
302 vx_load_pair_as(src0 + x, a0, a1);
303 Wvec r0 = v_mul(a0, a0), r1 = v_mul(a1, a1);
304 if (p == WT(3)) { r0 = v_mul(r0, a0); r1 = v_mul(r1, a1); }
305 v_store_pair_as(dst + x, r0, r1);
306 }
307 }
308 else if (p == WT(0.5))
309 {
310 for (; x < width; x += VECSZ*2)
311 {
312 if (x + VECSZ*2 > width) { if (!tail_trick || x == 0) break; x = width - VECSZ*2; }
313 Wvec a0, a1;
314 vx_load_pair_as(src0 + x, a0, a1);
315 a0 = v_sqrt(a0); a1 = v_sqrt(a1);
316 v_store_pair_as(dst + x, a0, a1);
317 }
318 }
319 else if (p == WT(1))
320 {
321 if ((const void*)src0 != (const void*)dst)
322 for (; x < width; x++) dst[x] = src0[x];
323 x = width;
324 }
325 else if (p == WT(0))
326 {
327 const T one = saturate_cast<T>(1);
328 for (; x < width; x++) dst[x] = one;
329 }
330 else if (p == WT(-0.5))
331 {
332 const Wvec one = vxSetallW(WT(1), Wvec());
333 for (; x < width; x += VECSZ*2)
334 {
335 if (x + VECSZ*2 > width) { if (!tail_trick || x == 0) break; x = width - VECSZ*2; }
336 Wvec a0, a1;
337 vx_load_pair_as(src0 + x, a0, a1);
338 a0 = v_div(one, v_sqrt(a0)); a1 = v_div(one, v_sqrt(a1));
339 v_store_pair_as(dst + x, a0, a1);
340 }
341 }
342 else if (p == std::rint(p) && std::abs(p) <= WT(65536))
343 {
344 // any other INTEGER exponent: LSB-first binary exponentiation - the same
345 // multiply chain (and order) as the classic iPow, fully vectorized. Also more
346 // accurate than exp(p*log x) (a few ulp vs ~2e-7 rel) and semantically exact
347 // on non-positive bases: the sign falls out of the multiplies, 0^negative
348 // divides to inf - no scalar patching needed.
349 const int ip = (int)p, ap = ip < 0 ? -ip : ip; // ap >= 1 (0..3 handled above)
350 const Wvec one = vxSetallW(WT(1), Wvec());
351 for (; x < width; x += VECSZ*2)
352 {
353 if (x + VECSZ*2 > width) { if (!tail_trick || x == 0) break; x = width - VECSZ*2; }
354 Wvec b0, b1;
355 vx_load_pair_as(src0 + x, b0, b1);
356 Wvec a0 = one, a1 = one;
357 for (int q = ap; q > 1; q >>= 1)
358 {
359 if (q & 1) { a0 = v_mul(a0, b0); a1 = v_mul(a1, b1); }
360 b0 = v_mul(b0, b0); b1 = v_mul(b1, b1);
361 }
362 a0 = v_mul(a0, b0); a1 = v_mul(a1, b1);
363 if (ip < 0) { a0 = v_div(one, a0); a1 = v_div(one, a1); }
364 v_store_pair_as(dst + x, a0, a1);
365 }
366 }
367 else // general scalar exponent: exp(p * log(x))
368 {
369 const Wvec vp = vxSetallW(p, Wvec()), z = v_setzero_<Wvec>();
370 for (; x < width; x += VECSZ*2)
371 {
372 if (x + VECSZ*2 > width) { if (!tail_trick || x == 0) break; x = width - VECSZ*2; }
373 Wvec a0, a1;
374 vx_load_pair_as(src0 + x, a0, a1);
375 if (v_check_any(v_le(a0, z)) || v_check_any(v_le(a1, z)))
376 { // exact std::pow for x <= 0 lanes
377 for (int i = 0; i < VECSZ*2; i++)
378 dst[x + i] = saturate_cast<T>(std::pow((WT)src0[x + i], p));
379 continue;
380 }
381 a0 = v_exp(v_mul(vp, v_log(a0)));
382 a1 = v_exp(v_mul(vp, v_log(a1)));
383 v_store_pair_as(dst + x, a0, a1);
384 }
385 }
386 }
387#endif
388 for (; x < width; x++)
389 dst[x] = saturate_cast<T>(std::pow((WT)src0[x*s0x], p));
390 continue;
391 }
392 // per-element exponent
393#if (CV_SIMD || CV_SIMD_SCALABLE)
394 if (s0x == 1)
395 {
396 const int VECSZ = VTraits<Wvec>::vlanes();
397 const Wvec z = v_setzero_<Wvec>();
398 for (; x < width; x += VECSZ*2)
399 {
400 if (x + VECSZ*2 > width) { if (!tail_trick || x == 0) break; x = width - VECSZ*2; }
401 Wvec a0, a1, b0, b1;
402 vx_load_pair_as(src0 + x, a0, a1);
403 vx_load_pair_as(src1 + x, b0, b1);
404 if (v_check_any(v_le(a0, z)) || v_check_any(v_le(a1, z)))
405 {
406 for (int i = 0; i < VECSZ*2; i++)
407 dst[x + i] = saturate_cast<T>(std::pow((WT)src0[x + i], (WT)src1[x + i]));
408 continue;
409 }
410 a0 = v_exp(v_mul(b0, v_log(a0)));
411 a1 = v_exp(v_mul(b1, v_log(a1)));
412 v_store_pair_as(dst + x, a0, a1);
413 }
414 }
415#endif
416 for (; x < width; x++)
417 dst[x] = saturate_cast<T>(std::pow((WT)src0[x*s0x], (WT)src1[x]));
418 }
419 dst = (T*)dst_;
420 for (int y = urows; y < height; y++)
421 memcpy(dst + (size_t)y*dsty, dst, (size_t)width*sizeof(T));
422 return 0;
423}

425TKernel getPowFunc_(int T, int R)
426{
427 if (R != T)
428 return {};
429 KernelFunc fptr = nullptr;
430 switch (T)
431 {
432 case CV_16F: fptr = powKernel<hfloat, v_float32>; break;
433 case CV_16BF: fptr = powKernel<bfloat, v_float32>; break;
434 case CV_32F: fptr = powKernel<float, v_float32>; break;
435#if CV_SIMD_64F || CV_SIMD_SCALABLE_64F
436 case CV_64F: fptr = powKernel<double, v_float64>; break;
437#else
438 case CV_64F: fptr = scalarPowKernel<double, double>; break;
439#endif
440 default: ;
441 }
442 return {fptr, nullptr, 0};
443}

465TKernel getMathFunc_(TOp op, int T)
466{
467 KernelFunc f = nullptr;
468 switch (op)
469 {
470 case OP_SQRT: f = mathByDepth<MSqrt>(T); break;
471 case OP_EXP: f = mathByDepth<MExp >(T); break;
472 case OP_LOG: f = mathByDepth<MLog >(T); break;
473 case OP_SIN: f = mathByDepth<MSin >(T); break;
474 case OP_COS: f = mathByDepth<MCos >(T); break;
475 case OP_TANH: f = mathByDepth<MTanh>(T); break;
476 case OP_RELU: f = mathByDepth<MRelu>(T); break;
477 case OP_ERF: f = mathByDepth<MErf >(T); break;
478 default: ;
479 }
480 return {f, nullptr, 0};
481}
