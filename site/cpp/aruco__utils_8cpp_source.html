12void _copyInput2Vector(InputArrayOfArrays inp, vector<vector<Point2f> > &vec)
13{
14 size_t i, nvecs = inp.size().area();
15 int inpdepth = inp.depth();
16 CV_Assert(inpdepth == CV_32F);
17 vec.resize(nvecs);
18 if(inp.isMatVector() || inp.kind() == _InputArray::STD_VECTOR_VECTOR)
19 {
20 for (i = 0; i < nvecs; i++)
21 {
22 Mat inp_i = inp.getMat((int)i);
23 int j, npoints = inp_i.checkVector(2, inpdepth, true);
24 CV_Assert(npoints >= 0);
25 const Point2f* inpptr = inp_i.ptr<Point2f>();
26 vector<Point2f>& vec_i = vec[i];
27 vec_i.resize(npoints);
28 for (j = 0; j < npoints; j++)
29 vec_i[j] = inpptr[j];
30 }
31 }
32 else {
33 CV_Error(cv::Error::StsNotImplemented,
34 "Only Mat vector, UMat vector, and vector<vector> OutputArrays are currently supported.");
35 }
36}

38void _copyVector2Output(vector<vector<Point2f> > &vec, OutputArrayOfArrays out, const float scale) {
39 size_t i, j, nvecs = vec.size();
40 if(out.isMatVector()) {
41 vector<Mat>& out_ = out.getMatVecRef();
42 out_.resize(nvecs);
43 for (i = 0; i < nvecs; i++) {
44 const vector<Point2f>& vec_i = vec[i];
45 Mat& out_i = out_[i];
46 Mat(vec_i).reshape(2, 1).convertTo(out_i, CV_32F, scale);
47 }
48 }
49 else if(out.isUMatVector()) {
50 vector<UMat>& out_ = out.getUMatVecRef();
51 out_.resize(nvecs);
52 for (i = 0; i < nvecs; i++) {
53 const vector<Point2f>& vec_i = vec[i];
54 UMat& out_i = out_[i];
55 Mat(vec_i).reshape(2, 1).convertTo(out_i, CV_32F, scale);
56 }
57 }
58 else if(out.kind() == _OutputArray::STD_VECTOR_VECTOR &&
59 out.type() == CV_32FC2){
60 vector<vector<Point2f>>& out_ = out.getVecVecRef<Point2f>();
61 out_.resize(nvecs);
62 for (i = 0; i < nvecs; i++) {
63 const vector<Point2f>& vec_i = vec[i];
64 size_t npoints_i = vec_i.size();
65 vector<Point2f>& out_i = out_[i];
66 out_i.resize(npoints_i);
67 for (j = 0; j < npoints_i; j++) {
68 out_i[j] = vec_i[j]*scale;
69 }
70 }
71 }
72 else {
73 CV_Error(cv::Error::StsNotImplemented,
74 "Only Mat vector, UMat vector, and vector<vector> OutputArrays are currently supported.");
75 }
