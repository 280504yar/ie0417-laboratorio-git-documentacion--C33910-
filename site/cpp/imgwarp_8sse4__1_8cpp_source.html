58inline void _mm_interleave_epi16(__m128i & v_r0, __m128i & v_r1, __m128i & v_g0, __m128i & v_g1)
59{
60 __m128i v_mask = _mm_set1_epi32(0x0000ffff);
61
62 __m128i layer3_chunk0 = _mm_packus_epi32(_mm_and_si128(v_r0, v_mask), _mm_and_si128(v_r1, v_mask));
63 __m128i layer3_chunk2 = _mm_packus_epi32(_mm_srli_epi32(v_r0, 16), _mm_srli_epi32(v_r1, 16));
64 __m128i layer3_chunk1 = _mm_packus_epi32(_mm_and_si128(v_g0, v_mask), _mm_and_si128(v_g1, v_mask));
65 __m128i layer3_chunk3 = _mm_packus_epi32(_mm_srli_epi32(v_g0, 16), _mm_srli_epi32(v_g1, 16));
66
67 __m128i layer2_chunk0 = _mm_packus_epi32(_mm_and_si128(layer3_chunk0, v_mask), _mm_and_si128(layer3_chunk1, v_mask));
68 __m128i layer2_chunk2 = _mm_packus_epi32(_mm_srli_epi32(layer3_chunk0, 16), _mm_srli_epi32(layer3_chunk1, 16));
69 __m128i layer2_chunk1 = _mm_packus_epi32(_mm_and_si128(layer3_chunk2, v_mask), _mm_and_si128(layer3_chunk3, v_mask));
70 __m128i layer2_chunk3 = _mm_packus_epi32(_mm_srli_epi32(layer3_chunk2, 16), _mm_srli_epi32(layer3_chunk3, 16));
71
72 __m128i layer1_chunk0 = _mm_packus_epi32(_mm_and_si128(layer2_chunk0, v_mask), _mm_and_si128(layer2_chunk1, v_mask));
73 __m128i layer1_chunk2 = _mm_packus_epi32(_mm_srli_epi32(layer2_chunk0, 16), _mm_srli_epi32(layer2_chunk1, 16));
74 __m128i layer1_chunk1 = _mm_packus_epi32(_mm_and_si128(layer2_chunk2, v_mask), _mm_and_si128(layer2_chunk3, v_mask));
75 __m128i layer1_chunk3 = _mm_packus_epi32(_mm_srli_epi32(layer2_chunk2, 16), _mm_srli_epi32(layer2_chunk3, 16));
76
77 v_r0 = _mm_packus_epi32(_mm_and_si128(layer1_chunk0, v_mask), _mm_and_si128(layer1_chunk1, v_mask));
78 v_g0 = _mm_packus_epi32(_mm_srli_epi32(layer1_chunk0, 16), _mm_srli_epi32(layer1_chunk1, 16));
79 v_r1 = _mm_packus_epi32(_mm_and_si128(layer1_chunk2, v_mask), _mm_and_si128(layer1_chunk3, v_mask));
80 v_g1 = _mm_packus_epi32(_mm_srli_epi32(layer1_chunk2, 16), _mm_srli_epi32(layer1_chunk3, 16));
81}

83void convertMaps_nninterpolate32f1c16s_SSE41(const float* src1f, const float* src2f, short* dst1, int width)
84{
85 int x = 0;
86 for (; x <= width - 16; x += 16)
87 {
88 __m128i v_dst0 = _mm_packs_epi32(_mm_cvtps_epi32(_mm_loadu_ps(src1f + x)),
89 _mm_cvtps_epi32(_mm_loadu_ps(src1f + x + 4)));
90 __m128i v_dst1 = _mm_packs_epi32(_mm_cvtps_epi32(_mm_loadu_ps(src1f + x + 8)),
91 _mm_cvtps_epi32(_mm_loadu_ps(src1f + x + 12)));
92
93 __m128i v_dst2 = _mm_packs_epi32(_mm_cvtps_epi32(_mm_loadu_ps(src2f + x)),
94 _mm_cvtps_epi32(_mm_loadu_ps(src2f + x + 4)));
95 __m128i v_dst3 = _mm_packs_epi32(_mm_cvtps_epi32(_mm_loadu_ps(src2f + x + 8)),
96 _mm_cvtps_epi32(_mm_loadu_ps(src2f + x + 12)));
97
98 _mm_interleave_epi16(v_dst0, v_dst1, v_dst2, v_dst3);
99
100 _mm_storeu_si128((__m128i *)(dst1 + x * 2), v_dst0);
101 _mm_storeu_si128((__m128i *)(dst1 + x * 2 + 8), v_dst1);
102 _mm_storeu_si128((__m128i *)(dst1 + x * 2 + 16), v_dst2);
103 _mm_storeu_si128((__m128i *)(dst1 + x * 2 + 24), v_dst3);
104 }
105
106 for (; x < width; x++)
107 {
108 dst1[x * 2] = saturate_cast<short>(src1f[x]);
109 dst1[x * 2 + 1] = saturate_cast<short>(src2f[x]);

113void convertMaps_32f1c16s_SSE41(const float* src1f, const float* src2f, short* dst1, ushort* dst2, int width)
114{
115 int x = 0;
116 __m128 v_its = _mm_set1_ps(INTER_TAB_SIZE);
117 __m128i v_its1 = _mm_set1_epi32(INTER_TAB_SIZE - 1);
118
119 for (; x <= width - 16; x += 16)
120 {
121 __m128i v_ix0 = _mm_cvtps_epi32(_mm_mul_ps(_mm_loadu_ps(src1f + x), v_its));
122 __m128i v_ix1 = _mm_cvtps_epi32(_mm_mul_ps(_mm_loadu_ps(src1f + x + 4), v_its));
123 __m128i v_iy0 = _mm_cvtps_epi32(_mm_mul_ps(_mm_loadu_ps(src2f + x), v_its));
124 __m128i v_iy1 = _mm_cvtps_epi32(_mm_mul_ps(_mm_loadu_ps(src2f + x + 4), v_its));
125
126 __m128i v_dst10 = _mm_packs_epi32(_mm_srai_epi32(v_ix0, INTER_BITS),
127 _mm_srai_epi32(v_ix1, INTER_BITS));
128 __m128i v_dst12 = _mm_packs_epi32(_mm_srai_epi32(v_iy0, INTER_BITS),
129 _mm_srai_epi32(v_iy1, INTER_BITS));
130 __m128i v_dst20 = _mm_add_epi32(_mm_slli_epi32(_mm_and_si128(v_iy0, v_its1), INTER_BITS),
131 _mm_and_si128(v_ix0, v_its1));
132 __m128i v_dst21 = _mm_add_epi32(_mm_slli_epi32(_mm_and_si128(v_iy1, v_its1), INTER_BITS),
133 _mm_and_si128(v_ix1, v_its1));
134 _mm_storeu_si128((__m128i *)(dst2 + x), _mm_packus_epi32(v_dst20, v_dst21));
135
136 v_ix0 = _mm_cvtps_epi32(_mm_mul_ps(_mm_loadu_ps(src1f + x + 8), v_its));
137 v_ix1 = _mm_cvtps_epi32(_mm_mul_ps(_mm_loadu_ps(src1f + x + 12), v_its));
138 v_iy0 = _mm_cvtps_epi32(_mm_mul_ps(_mm_loadu_ps(src2f + x + 8), v_its));
139 v_iy1 = _mm_cvtps_epi32(_mm_mul_ps(_mm_loadu_ps(src2f + x + 12), v_its));
140
141 __m128i v_dst11 = _mm_packs_epi32(_mm_srai_epi32(v_ix0, INTER_BITS),
142 _mm_srai_epi32(v_ix1, INTER_BITS));
143 __m128i v_dst13 = _mm_packs_epi32(_mm_srai_epi32(v_iy0, INTER_BITS),
144 _mm_srai_epi32(v_iy1, INTER_BITS));
145 v_dst20 = _mm_add_epi32(_mm_slli_epi32(_mm_and_si128(v_iy0, v_its1), INTER_BITS),
146 _mm_and_si128(v_ix0, v_its1));
147 v_dst21 = _mm_add_epi32(_mm_slli_epi32(_mm_and_si128(v_iy1, v_its1), INTER_BITS),
148 _mm_and_si128(v_ix1, v_its1));
149 _mm_storeu_si128((__m128i *)(dst2 + x + 8), _mm_packus_epi32(v_dst20, v_dst21));
150
151 _mm_interleave_epi16(v_dst10, v_dst11, v_dst12, v_dst13);
152
153 _mm_storeu_si128((__m128i *)(dst1 + x * 2), v_dst10);
154 _mm_storeu_si128((__m128i *)(dst1 + x * 2 + 8), v_dst11);
155 _mm_storeu_si128((__m128i *)(dst1 + x * 2 + 16), v_dst12);
156 _mm_storeu_si128((__m128i *)(dst1 + x * 2 + 24), v_dst13);
157 }
158 for (; x < width; x++)
159 {
160 int ix = saturate_cast<int>(src1f[x] * static_cast<float>(INTER_TAB_SIZE));
161 int iy = saturate_cast<int>(src2f[x] * static_cast<float>(INTER_TAB_SIZE));
162 dst1[x * 2] = saturate_cast<short>(ix >> INTER_BITS);
163 dst1[x * 2 + 1] = saturate_cast<short>(iy >> INTER_BITS);
164 dst2[x] = (ushort)((iy & (INTER_TAB_SIZE - 1))*INTER_TAB_SIZE + (ix & (INTER_TAB_SIZE - 1)));
165 }

168void convertMaps_32f2c16s_SSE41(const float* src1f, short* dst1, ushort* dst2, int width)
169{
170 int x = 0;
171 __m128 v_its = _mm_set1_ps(INTER_TAB_SIZE);
172 __m128i v_its1 = _mm_set1_epi32(INTER_TAB_SIZE - 1);
173 __m128i v_y_mask = _mm_set1_epi32((INTER_TAB_SIZE - 1) << 16);
174
175 for (; x <= width - 4; x += 4)
176 {
177 __m128i v_src0 = _mm_cvtps_epi32(_mm_mul_ps(_mm_loadu_ps(src1f + x * 2), v_its));
178 __m128i v_src1 = _mm_cvtps_epi32(_mm_mul_ps(_mm_loadu_ps(src1f + x * 2 + 4), v_its));
179
180 __m128i v_dst1 = _mm_packs_epi32(_mm_srai_epi32(v_src0, INTER_BITS),
181 _mm_srai_epi32(v_src1, INTER_BITS));
182 _mm_storeu_si128((__m128i *)(dst1 + x * 2), v_dst1);
183
184 // x0 y0 x1 y1 . . .
185 v_src0 = _mm_packs_epi32(_mm_and_si128(v_src0, v_its1),
186 _mm_and_si128(v_src1, v_its1));
187 __m128i v_dst2 = _mm_or_si128(_mm_srli_epi32(_mm_and_si128(v_src0, v_y_mask), 16 - INTER_BITS), // y0 0 y1 0 . . .
188 _mm_and_si128(v_src0, v_its1)); // 0 x0 0 x1 . . .
189 _mm_storel_epi64((__m128i *)(dst2 + x), _mm_packus_epi32(v_dst2, v_dst2));
190 }
191 for (; x < width; x++)
192 {
193 int ix = saturate_cast<int>(src1f[x * 2] * static_cast<float>(INTER_TAB_SIZE));
194 int iy = saturate_cast<int>(src1f[x * 2 + 1] * static_cast<float>(INTER_TAB_SIZE));
195 dst1[x * 2] = saturate_cast<short>(ix >> INTER_BITS);
196 dst1[x * 2 + 1] = saturate_cast<short>(iy >> INTER_BITS);
197 dst2[x] = (ushort)((iy & (INTER_TAB_SIZE - 1))*INTER_TAB_SIZE + (ix & (INTER_TAB_SIZE - 1)));
198 }
