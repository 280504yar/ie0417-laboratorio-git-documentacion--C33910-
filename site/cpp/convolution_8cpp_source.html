28// Fast convert float 32 to float16
29static inline void _cvt32f16f(const float* src, __fp16* dst, int len)
30{
31 int j = 0;
32 const int VECSZ = 4;
33 __fp16* dst_FP16 = (__fp16 *)dst;
34 if (len > VECSZ * 4)
35 {
36 const int VECSZ4 = 4 * VECSZ;
37 for( ; j + VECSZ4 < len; j += VECSZ4)
38 {
39
40 float32x4_t v0 = vld1q_f32(src + j);
41 float32x4_t v1 = vld1q_f32(src + j + 4);
42 float32x4_t v2 = vld1q_f32(src + j + 8);
43 float32x4_t v3 = vld1q_f32(src + j + 12);
44
45 vst1q_f16(dst_FP16 + j, vcombine_f16(vcvt_f16_f32(v0), vcvt_f16_f32(v1)));
46 vst1q_f16(dst_FP16 + j + 8, vcombine_f16(vcvt_f16_f32(v2), vcvt_f16_f32(v3)));
47 }
48 }
49
50 for( ; j < len; j += VECSZ )
51 {
52 if( j > len - VECSZ )
53 {
54 if( j == 0 )
55 break;
56 j = len - VECSZ;
57 }
58
59 float16x4_t hv = vcvt_f16_f32(vld1q_f32(src + j));
60 vst1_f16(dst_FP16 + j, hv);
61 }
62 for( ; j < len; j++ )
63 dst[j] = __fp16(src[j]);
64}

87Ptr<FastConv> initFastConv(
88 InputArray _weightsMat,
89 float* srcBias,
90 int ngroups,
91 int K, int C,
92 const std::vector<size_t>& kernel_size,
93 const std::vector<size_t>& strides,
94 const std::vector<size_t>& dilations,
95 const std::vector<size_t>& pads_begin,
96 const std::vector<size_t>& pads_end,
97 int conv_dim,
98 const bool _useFP16,
99 bool useWinograd)
100{
101 Ptr<FastConv> conv = makePtr<FastConv>();
102 CV_Assert(ngroups > 0 && K > 0 && C > 0 && K % ngroups == 0);
103
104 // Weight shape, [K, C, Dk, Hk, Wk] for Conv3D, [K, C, Hk, Wk] for Conv2D, [K, C, Wk] for Conv1D.
105 int Dk = conv_dim == CONV_3D ? (int)kernel_size[0] : 1;
106 int Hk = conv_dim == CONV_1D ? 1 : (int)kernel_size[kernel_size.size() - 2];
107 int Wk = (int)kernel_size.back();
108 int karea = Wk*Hk*Dk;
109
110 conv->pad_front = conv_dim == CONV_3D ? (int)pads_begin[0] : 0;
111 conv->pad_top = conv_dim == CONV_1D ? 0 : (int)pads_begin[pads_begin.size() - 2];
112 conv->pad_left = (int)pads_begin.back();
113
114 conv->pad_behind = conv_dim == CONV_3D ? (int)pads_end[0] : 0;
115 conv->pad_bottom = conv_dim == CONV_1D ? 0 : (int)pads_end[pads_end.size() - 2];
116 conv->pad_right = (int)pads_end.back();
117
118 int stride_d = conv_dim == CONV_3D ? (int)strides[0] : 1;
119 int stride_h = conv_dim == CONV_1D ? 1 : (int)strides[strides.size() - 2];
120 int stride_w = (int)strides.back();
121
122 int dilation_d = conv_dim == CONV_3D ? (int)dilations[0] : 1;
123 int dilation_h = conv_dim == CONV_1D ? 1 : (int)dilations[dilations.size() - 2];
124 int dilation_w = (int)dilations.back();
125
126 CV_Assert(Dk > 0 && Hk > 0 && Wk > 0);
127 CV_Assert(stride_d >= 0 && stride_h >= 0 && stride_w > 0);
128 CV_Assert(dilation_d > 0 && dilation_h > 0 && dilation_w > 0);
129
130 conv->K = K; conv->C = C; conv->Hk = Hk; conv->Wk = Wk, conv->Dk = Dk;
131
132 conv->stride_d = stride_d;
133 conv->stride_h = stride_h;
134 conv->stride_w = stride_w;
135
136 conv->dilation_d = dilation_d;
137 conv->dilation_h = dilation_h;
138 conv->dilation_w = dilation_w;
139 conv->conv_dim = conv_dim;
140 conv->ngroups = ngroups;
141
142 bool ifRunDepthWise = ngroups > 1 && ngroups == K && ngroups == C;
143 bool ifRunDepthWiseRemain = false; // It's for big padding or big kernel or Conv3D depth-wise convolution.
144
145 if (ifRunDepthWise)
146 {
147 if (conv_dim == CONV_1D)
148 {
149 ifRunDepthWise &= Hk == 1 && Wk == 3 && (stride_w == 1 || (stride_w == 2 && dilation_w == 1))
150 && max(stride_w, dilation_w) >= conv->pad_left && conv->pad_left <= 1;
151 }
152 else if (conv_dim == CONV_2D)
153 {
154 ifRunDepthWise &= Hk == 3 && Wk == 3 && ((stride_w == 1) || (stride_w == 2 && dilation_w == 1)) &&
155 max(stride_w, dilation_w) >= conv->pad_left && max(stride_h, dilation_h) >= conv->pad_top
156 && conv->pad_left <= 1 && conv->pad_top <= 1;
157 }
158
159 if (!ifRunDepthWise || conv_dim == CONV_3D)
160 {
161 ifRunDepthWise = false;
162 ifRunDepthWiseRemain = true;
163 }
164 }
165
166 conv->conv_type = ifRunDepthWise && conv_dim != CONV_3D ? CONV_TYPE_DEPTHWISE :
167 useWinograd && (conv_dim == CONV_2D && (conv->useSIMD128 || conv->useAVX || conv->useAVX2 || conv->useNEON || conv->useRVV) &&
168 Hk == 3 && Wk == 3 && dilation_h == 1 && dilation_w == 1 && stride_h == 1 && stride_w == 1) ?
169 CONV_TYPE_WINOGRAD3X3 :
170 (ifRunDepthWiseRemain ? CONV_TYPE_DEPTHWISE_REMAIN : CONV_TYPE_GENERIC);
171
172#if !(CV_NEON || CV_SIMD128 || CV_TRY_AVX || CV_TRY_AVX2 || CV_RVV)
173 if (conv->conv_type == CONV_TYPE_WINOGRAD3X3) // Disabel Winograd when CV_NEON, CV_SIMD128, CV_TRY_AVX, CV_TRY_AVX2 and CV_RVV are not available.
174 conv->conv_type = CONV_TYPE_GENERIC;
175#endif
176#if CV_RVV
177 // Winograd RVV requires vsetvlmax_e32m1() >= 8 (i.e. VLEN >= 256). Fall back to generic on narrower implementations.
178 if (conv->conv_type == CONV_TYPE_WINOGRAD3X3 && conv->useRVV && (int)__riscv_vsetvlmax_e32m1() < 8)
179 conv->conv_type = CONV_TYPE_GENERIC;
180#endif
181
182 Mat weightsMat = _weightsMat.getMat();
183 const size_t wstep = weightsMat.step1();
184
185 conv->useFP16 = false;
186#ifdef CONV_ARM_FP16
187 if (_useFP16 && (conv->conv_type == CONV_TYPE_GENERIC || conv->conv_type == CONV_TYPE_DEPTHWISE_REMAIN
188 || conv->conv_type == CONV_TYPE_WINOGRAD3X3))
189 conv->useFP16 = true;
190
191 // Runtime FP16 check.
192 if (conv->useFP16 && !checkHardwareSupport(CPU_NEON_FP16))
193 {
194 conv->useFP16 = false;
195 CV_LOG_ONCE_WARNING(NULL, "DNN: the CPU does not support the instruction set required by FP16, fallback to FP32.");
196 }
197#endif
198
199 float *srcWeights = (float *)weightsMat.data;
200 if (conv->conv_type == CONV_TYPE_DEPTHWISE || conv->conv_type == CONV_TYPE_DEPTHWISE_REMAIN)
201 {
202 // Handle the Conv1D, Conv2D and Conv3D depth-wise.
203 // for depth-wise convolutions on NCHW data we just preserve the weights in KCHW layout,
204 // but add some padding to make the weights array layout more SIMD-friendly
205 int ksize = karea;
206
207 // TODO: simplify the following code with std::copy.
208 // this code aims to let memory fit with vector size.
209 int padded_ksize = ((ksize + VEC_ALIGN-1) / VEC_ALIGN) * VEC_ALIGN;
210 int nweights = C * padded_ksize;
211
212#ifdef CONV_ARM_FP16
213 if (conv->useFP16)
214 {
215 conv->weightsBuf_FP16.resize(nweights + VEC_ALIGN);
216 auto weightsPtr_FP16 = (__fp16*)conv->getWeightsFP16();
217
218 parallel_for_(Range(0, C), [&](const Range& r0){
219 for(int c = r0.start; c < r0.end; c++)
220 _cvt32f16f(srcWeights + c*wstep, weightsPtr_FP16 + c*padded_ksize, ksize);
221 });
222 }
223 else
224#endif
225 {
226 conv->weightsBuf.resize(nweights + VEC_ALIGN);
227 auto weightsPtr = conv->getWeights();
228
229 parallel_for_(Range(0, C), [&](const Range& r0) {
230 for(int c = r0.start; c < r0.end; c++)
231 memcpy(weightsPtr + c*padded_ksize, srcWeights + c*wstep, ksize*sizeof(weightsPtr[0]));
232 });
233 }
234 }
235 else if(conv->conv_type == CONV_TYPE_WINOGRAD3X3) // winograd
236 {
237 static const float ktm[8][3] = {
238 {1.0f, 0.0f, 0.0f},
239 {-2.0f / 9, -2.0f / 9, -2.0f / 9},
240 {-2.0f / 9, 2.0f / 9, -2.0f / 9},
241 {1.0f / 90, 1.0f / 45, 2.0f / 45},
242 {1.0f / 90, -1.0f / 45, 2.0f / 45},
243 {32.f/45, 16.f/45, 8.f/45},
244 {32.f/45, -16.f/45, 8.f/45},
245 {0.0f, 0.0f, 1.0f}
246 };
247
248 const int CONV_WINO_KBLOCK = 4;
249
250#if CV_TRY_AVX || CV_TRY_AVX2
251 const int CONV_WINO_ATOM_F32 = (conv->useAVX || conv->useAVX2) ? 8 : 4;
252#elif CV_RVV
253 const int CONV_WINO_ATOM_F32 = (conv->useRVV && (int)__riscv_vsetvlmax_e32m1() >= 8) ? 8 : 4;
254#else
255 const int CONV_WINO_ATOM_F32 = 4;
256#endif
257 const int CONV_WINO_NATOMS_F32 = CONV_WINO_AREA / CONV_WINO_ATOM_F32; // for AVX2, it is 8, otherwise, it's 16.
258
259#ifdef CONV_ARM_FP16
260 // FP 16
261 const int CONV_WINO_ATOM_F16 = CONV_WINO_ATOM_F32 * 2;
262 const int CONV_WINO_NATOMS_F16 = CONV_WINO_AREA / CONV_WINO_ATOM_F16;
263#endif
264
265 // the weights are packed as 6-dim tensor:
266 // ngroups * ceil((K/ngroups)/KBLOCK) * (W*W/ATOM_SIZE) * (C/ngroups) * KBLOCK * ATOM_SIZE,
267 // where W is the size of Winograd-transformed kernel (8x8),
268 // ATOM_SIZE is number of lanes in SIMD register (4 for NEON and FP32),
269 // KBLOCK is some platform-dependent constant dependent on the number of SIMD registers.
270 int ksize = CONV_WINO_KSIZE * CONV_WINO_KSIZE;
271 int Cg = C/ngroups;
272 int Kg = K/ngroups;
273 int Kg_nblocks = (Kg + CONV_WINO_KBLOCK - 1)/CONV_WINO_KBLOCK;
274 size_t nweights = ngroups*Kg_nblocks*Cg*CONV_WINO_KBLOCK*CONV_WINO_AREA;
275
276 float* wptrWino = nullptr;
277#ifdef CONV_ARM_FP16
278 __fp16* wptrWino_FP16 = nullptr;
279 if (conv->useFP16)
280 {
281 conv->weightsWinoBuf_FP16.resize(nweights + VEC_ALIGN);
282 wptrWino_FP16 = (__fp16*)conv->getWeightsWinoFP16();
283 }
284 else
285#endif
286 {
287 conv->weightsWinoBuf.resize(nweights + VEC_ALIGN);
288 wptrWino = conv->getWeightsWino();
289 }
290
291 parallel_for_(Range(0, K), [&](const Range& r0){
292 float kernelTm[CONV_WINO_AREA];
293 for (int k = r0.start; k < r0.end; k++)
294 {
295 int g = k / Kg;
296 int k_ = k - g*Kg;
297 int ki = k_ / CONV_WINO_KBLOCK;
298 int dk = k_ - ki*CONV_WINO_KBLOCK;
299
300 for (int c = 0; c < Cg; c++)
301 {
302 // wstep = Hk*Wk*Cg
303 const float *kernel0 = srcWeights + k * wstep + c * ksize;
304
305 // transform kernel, transposed
306 const float *k0 = kernel0;
307 const float *k1 = kernel0 + 3;
308 const float *k2 = kernel0 + 6;
309
310 // h
311 float tmp[8][3];
312 for (int i = 0; i < 8; i++)
313 {
314 tmp[i][0] = k0[0] * ktm[i][0] + k0[1] * ktm[i][1] + k0[2] * ktm[i][2];
315 tmp[i][1] = k1[0] * ktm[i][0] + k1[1] * ktm[i][1] + k1[2] * ktm[i][2];
316 tmp[i][2] = k2[0] * ktm[i][0] + k2[1] * ktm[i][1] + k2[2] * ktm[i][2];
317 }
318
319 // v
320 for (int j = 0; j < 8; j++)
321 {
322 float *tmpp = &tmp[j][0];
323
324 for (int i = 0; i < 8; i++)
325 kernelTm[j * 8 + i] = tmpp[0] * ktm[i][0] + tmpp[1] * ktm[i][1] + tmpp[2] * ktm[i][2];
326 }
327
328 // repack the data.
329#ifdef CONV_ARM_FP16
330 if (conv->useFP16)
331 {
332 __fp16* wptr = wptrWino_FP16 + (g*Kg_nblocks + ki) * Cg *CONV_WINO_KBLOCK*CONV_WINO_AREA +
333 (c*CONV_WINO_KBLOCK + dk)*CONV_WINO_ATOM_F16;
334 for (int i = 0; i < CONV_WINO_NATOMS_F16; i++,
335 wptr += Cg * CONV_WINO_KBLOCK * CONV_WINO_ATOM_F16)
336 {
337 CV_Assert(wptrWino_FP16 <= wptr && wptr + CONV_WINO_ATOM_F16 <= wptrWino_FP16 + nweights);
338 for (int j = 0; j < CONV_WINO_ATOM_F16; j++)
339 {
340 wptr[j] = (__fp16)kernelTm[i * CONV_WINO_ATOM_F16 + j];
341 }
342 }
343 }
344 else
345#endif
346 {
347 float* wptr = wptrWino + (g*Kg_nblocks + ki) * Cg *CONV_WINO_KBLOCK*CONV_WINO_AREA +
348 (c*CONV_WINO_KBLOCK + dk)*CONV_WINO_ATOM_F32;
349 for (int i = 0; i < CONV_WINO_NATOMS_F32; i++,
350 wptr += Cg * CONV_WINO_KBLOCK * CONV_WINO_ATOM_F32)
351 {
352 CV_Assert(wptrWino <= wptr && wptr + CONV_WINO_ATOM_F32 <= wptrWino + nweights);
353 memcpy(wptr, kernelTm + i * CONV_WINO_ATOM_F32, CONV_WINO_ATOM_F32*sizeof (wptr[0]));
354 }
355 }
356 }
357 }
358 });
359 }
360 else if (conv->conv_type == CONV_TYPE_GENERIC)
361 {
362 // The weights are packed as
363 // ngroups x (ceil((K/ngroups)/CONV_MR)*CONV_MR) x (Cg*Hk*Wk*Dk) x CONV_MR tensor
364 int Kg = K/ngroups, Cg = max(C/ngroups, 1);
365 int DkHkWkCg = Dk*Hk*Wk*Cg;
366
367 int numStripsMR = (Kg + CONV_MR_FP32 - 1) / CONV_MR_FP32;
368 int Kg_aligned = numStripsMR * CONV_MR_FP32;
369 size_t nweights = ngroups*Kg_aligned*DkHkWkCg;
370 float* weightsPtr = nullptr;
371
372#ifdef CONV_ARM_FP16
373 int numStripsMR_FP16 = (Kg + CONV_MR_FP16 - 1) / CONV_MR_FP16;
374 int Kg_aligned_FP16 = numStripsMR_FP16 * CONV_MR_FP16;
375 size_t nweights_FP16 = ngroups * Kg_aligned_FP16 * DkHkWkCg;
376 __fp16* weightsPtr_FP16 = nullptr;
377
378 if (conv->useFP16)
379 {
380 conv->weightsBuf_FP16.resize(nweights_FP16 + VEC_ALIGN);
381 weightsPtr_FP16 = (__fp16*)conv->getWeightsFP16();
382 }
383 else
384#endif
385 {
386 conv->weightsBuf.resize(nweights + VEC_ALIGN);
387 weightsPtr = conv->getWeights();
388 }
389
390 // Pack the weight.
391#ifdef CONV_ARM_FP16
392 if (conv->useFP16)
393 {
394 parallel_for_(Range(0, ngroups * numStripsMR_FP16), [&](const Range& r0){
395 for (int gsi = r0.start; gsi < r0.end; gsi++)
396 {
397 int g = gsi / numStripsMR_FP16;
398 int si = gsi - g * numStripsMR_FP16;
399
400 int startK = si * CONV_MR_FP16;
401 CV_Assert(startK < Kg_aligned_FP16);
402
403 __fp16* packed_wptr = weightsPtr_FP16 + DkHkWkCg * (startK + g * Kg_aligned_FP16);
404 int dk = Kg - startK < CONV_MR_FP16 ? Kg - startK : CONV_MR_FP16; // check if we need zero padding.
405
406 int k_idx = g*Kg + startK;
407 for(int hwd = 0; hwd < Hk*Wk*Dk; hwd++)
408 {
409 for(int c = 0; c < Cg; c++, packed_wptr += CONV_MR_FP16)
410 {
411 const float* wptr = srcWeights + wstep * k_idx + c*Hk*Wk*Dk + hwd;
412 int k = 0;
413 for(; k < dk; k++, wptr += wstep)
414 packed_wptr[k] = (__fp16)(*wptr);
415 for(; k < CONV_MR_FP16; k++)
416 packed_wptr[k] = (__fp16)0.f;
417 }
418 }
419 }});
420 }
421 else
422#endif
423 {
424 parallel_for_(Range(0, ngroups * numStripsMR), [&](const Range& r0){
425 for (int gsi = r0.start; gsi < r0.end; gsi++)
426 {
427 int g = gsi / numStripsMR;
428 int si = gsi - g * numStripsMR;
429
430 int startK = si * CONV_MR_FP32;
431 CV_Assert(startK < Kg_aligned);
432
433 float* packed_wptr = weightsPtr + DkHkWkCg * (startK + g * Kg_aligned);
434 int dk = Kg - startK < CONV_MR_FP32 ? Kg - startK : CONV_MR_FP32; // check if we need zero padding.
435
436 int k_idx = g*Kg + startK;
437 for(int hwd = 0; hwd < Hk*Wk*Dk; hwd++)
438 {
439 for(int c = 0; c < Cg; c++, packed_wptr += CONV_MR_FP32)
440 {
441 const float* wptr = srcWeights + wstep * k_idx + c*Hk*Wk*Dk + hwd;
442 int k = 0;
443 for(; k < dk; k++, wptr += wstep)
444 packed_wptr[k] = *wptr;
445 for(; k < CONV_MR_FP32; k++)
446 packed_wptr[k] = 0.f;
447 }
448 }
449 }});
450 }
451 }
452 else
453 CV_Error(cv::Error::StsUnsupportedFormat, "Unknown convolution type.");
454
455 // store bias; append some zero's to make sure that
456 // we can always read MR elements starting from any valid index
457 {
458 int k = 0, nbias = K + VEC_ALIGN;
459 conv->biasBuf.resize(nbias);
460 float* biasBufPtr = conv->biasBuf.data();
461 for(; k < K; k++)
462 biasBufPtr[k] = srcBias ? srcBias[k] : 0.f;
463 for(; k < nbias; k++)
464 biasBufPtr[k] = 0.f;
465 }
466 return conv;
467}

469static inline void packData8(char*& inpbuf, float*& inptrIn, int& in_w, int& x0, int& s0, const int* ofstab,
470 const int stride_w, const int ksize, const int esz)
471{
472 char * inpbufC = inpbuf + s0 * esz;
473 float* inptrInC = (float* )inptrIn;
474
475#ifdef CONV_ARM_FP16
476 __fp16* inpbufC_FP16 = (__fp16 *)inpbufC;
477 if (esz == sizeof(__fp16))
478 {
479 if (stride_w == 1)
480 {
481 for (int k = 0; k < ksize; k++)
482 {
483 int k1 = ofstab[k];
484
485 float32x4_t v0 = vld1q_f32(inptrInC + k1);
486 float32x4_t v1 = vld1q_f32(inptrInC + k1 + 4);
487 vst1q_f16((__fp16*)inpbufC_FP16 + k * CONV_NR_FP16, vcombine_f16(vcvt_f16_f32(v0), vcvt_f16_f32(v1)));
488 }
489 }
490 else
491 {
492 for (int k = 0; k < ksize; k++)
493 {
494 int k1 = ofstab[k];
495 float32x4_t v0 = {inptrInC[k1], inptrInC[k1 + stride_w], inptrInC[k1 + 2*stride_w], inptrInC[k1 + 3*stride_w]};
496 float32x4_t v1 = {inptrInC[k1 + 4*stride_w], inptrInC[k1 + 5*stride_w], inptrInC[k1 + 6*stride_w], inptrInC[k1 + 7*stride_w]};
497
498 vst1q_f16((__fp16*)inpbufC_FP16 + k * CONV_NR_FP16, vcombine_f16(vcvt_f16_f32(v0), vcvt_f16_f32(v1)));
499 }
500 }
501 }
502 else // float 32
503#endif
504 {
505 CV_Assert(esz == sizeof(float ));
506 float* inpbufC_FP32 = (float* )inpbufC;
507 if (stride_w == 1)
508 for (int k = 0; k < ksize; k++)
509 {
510 int k1 = ofstab[k];
511#if CV_SIMD256
512 vx_store(inpbufC_FP32 + k*CONV_NR_FP32, vx_load(inptrInC + k1));
513#elif CV_SIMD128
514 v_float32x4 vv0 = v_load(inptrInC + k1);
515 v_float32x4 vv1 = v_load(inptrInC + k1 + 4);
516 v_store(inpbufC_FP32 + k*CONV_NR_FP32, vv0);

605static inline void packInputData(char* inpbuf_task, float* inp, const int* ofstab, const int* dhwTab, int zyx0, int zyx_limit,
606 int ksize, int stride_d, int stride_h, int stride_w, int pad_front, int pad_top, int pad_left,
607 int Dk, int Hk, int Wk, int dilation_d, int dilation_h, int dilation_w, int Di, int Hi, int Wi,
608 int H0, int W0, int Cg, int stripesize, int inp_plane_ofs, int inp_planesize, int conv_dim, int conv_type,
609 const int CONV_NR, const int esz, bool fast_1x1, bool useFP16)
610{
611 for (int stripe = 0; zyx0 < zyx_limit; stripe++, zyx0 += CONV_NR)
612 {
613 char *inpbuf = inpbuf_task + stripe * stripesize * esz;
614 float *inptr = inp + inp_plane_ofs;
615
616 /*
617 1. pack the data. Copy the HkxWk CONV_NR-wide slices from
618 each feature plane of the input tensor to the input buffer.
619 */
620 if (fast_1x1)
621 {
622 int slice_len = zyx_limit - zyx0;
623 bool partial = slice_len < CONV_NR;
624 const int CONV_NR_esz = CONV_NR * esz;
625 // Superfast branch for 1x1 convolutions with sy=sx=1.
626 // in this case each feature plane can be safely treated
627 // as 1D array, and we just extract next portion
628 // of CONV_NR elements from each feature plane and
629 // put it together.
630 inptr += zyx0;
631 if (!partial)
632 {
633 // Make special branch where memcpy() is called with a constant buffer size.
634 // Compilers will likely unroll this loop properly.
635#ifdef CONV_ARM_FP16
636 if (useFP16)
637 {
638 for (int c = 0; c < Cg; c++, inptr += inp_planesize, inpbuf += CONV_NR_esz)
639 _cvt32f16f(inptr, (__fp16 *)inpbuf, CONV_NR);
640 }
641 else
642#endif
643 for (int c = 0; c < Cg; c++, inptr += inp_planesize, inpbuf += CONV_NR_esz)
644 memcpy(inpbuf, inptr, CONV_NR_esz);
645 }
646 else
647 {
648#ifdef CONV_ARM_FP16
649 if (useFP16)
650 {
651 for (int c = 0; c < Cg; c++, inptr += inp_planesize, inpbuf += CONV_NR_esz)
652 {
653 _cvt32f16f(inptr, (__fp16 *)inpbuf, slice_len);
654 }
655 }
656 else
657#endif
658 for (int c = 0; c < Cg; c++, inptr += inp_planesize, inpbuf += CONV_NR_esz)
659 {
660 memcpy(inpbuf, inptr, slice_len * esz);
661 }
662 }
663 }
664 else if (conv_type == CONV_TYPE_DEPTHWISE_REMAIN)
665 {
666 CV_Assert(Cg == 1);
667 const int HW0 = H0 * W0;
668 const int HWi = Hi * Wi;
669 int slice_len = std::min(zyx_limit - zyx0, CONV_NR);
670
671 // here some non-continuous sub-row of the row will not be
672 // filled from the tensor; we need to make sure that the uncovered
673 // elements are explicitly set to 0's. the easiest way is to
674 // set all the elements to 0's before the loop.
675 memset(inpbuf, 0, stripesize * esz);
676
677 int z0 = zyx0 / HW0, yx0 = zyx0 - z0 * HW0;
678 int y0 = yx0 / W0, x0 = yx0 - y0 * W0;
679
680 if (conv_dim == CONV_1D)
681 {
682 for (int slice_i = 0; slice_i < slice_len; y0++, x0=0)
683 {
684 int delta = std::min(slice_len - slice_i, W0 - x0);
685 int x1 = x0 + delta;
686
687 int in_w = x0 * stride_w - pad_left;
688 float* inptrIn = inptr + in_w;
689
690 int s0 = slice_i;
691
692 for (; x0 < x1; x0++, s0++, inptrIn += stride_w, in_w += stride_w)
693 {
694 // Pack 8
695 if (x0 + 8 <= x1 && 0 <= in_w &&
696 in_w + stride_w*8 <= Wi - (Wk-1)*dilation_w)
697 {
698 packData8(inpbuf, inptrIn, in_w, x0, s0, ofstab, stride_w, ksize, esz);
699 }
700 else if (x0 + 2 <= x1 && 0 <= in_w &&
701 in_w + stride_w*2 <= Wi - (Wk-1)*dilation_w)
702 {
703 packData2(inpbuf, inptrIn, in_w, x0, s0, ofstab, stride_w, ksize, esz);
704 }
705 else
706 {
707 int w0 = std::max(0, (-in_w + dilation_w-1)/dilation_w);
708 int w1 = std::min(Wk, (Wi - in_w + dilation_w-1)/dilation_w);
709 const float* inptrInC = inptrIn;
710#ifdef CONV_ARM_FP16
711 if (useFP16)
712 {
713 __fp16* inpbufC = (__fp16 *)inpbuf + s0;
714 for (int w = w0; w < w1; w++)
715 {
716 int imgofs = w*dilation_w;
717 inpbufC[w*CONV_NR] = (__fp16)inptrInC[imgofs];
718 }
719 }
720 else
721#endif
722 {
723 float* inpbufC = (float *)inpbuf + s0;
724 for (int w = w0; w < w1; w++)
725 {
726 int imgofs = w*dilation_w;

729 }
730 }
731 }
732 slice_i += delta;
733 }
734 }

1048 inpbuf_ki[1] = t1;
1049 inpbuf_ki[2] = t2;
1050 inpbuf_ki[3] = t3;
1051 }
1052 }
1053 i += 4;
1054 x0 += 4;
1055 }
1056 else
1057 {
1058#ifdef CONV_ARM_FP16
1059 if (useFP16)
1060 {
1061 for (int c = 0; c < Cg; c++, inpbuf_ki_FP16 += CONV_NR, inptr_ki += inp_planesize)
1062 inpbuf_ki_FP16[0] = (__fp16)(*inptr_ki);
1063 }
1064 else
1065#endif
1066 for (int c = 0; c < Cg; c++, inpbuf_ki += CONV_NR, inptr_ki += inp_planesize)
1067 *inpbuf_ki = *inptr_ki;
1068 i++;
1069 x0++;
1070 }
1071 }
1072 else
1073 {
1074#ifdef CONV_ARM_FP16
1075 if (useFP16)
1076 {
1077 for (int c = 0; c < Cg; c++, inpbuf_ki_FP16 += CONV_NR)
1078 inpbuf_ki_FP16[0] = (__fp16)0.f;
1079 }
1080 else
1081#endif
1082 for (int c = 0; c < Cg; c++, inpbuf_ki += CONV_NR)
1083 inpbuf_ki[0] = 0.f;
1084 i++;
1085 x0++;
1086 }
1087
1088 int mask = x0 >= W0;
1089 y0 += mask;
1090 x0 &= mask - 1;
1091
1092 mask = y0 >= H0; // Only Conv 3D need jump at z0 dimension
1093 if (mask && conv_dim != CONV_3D)
1094 break;
1095
1096 z0 += mask;
1097 y0 &= mask - 1;
1098 }
1099 }
1100 }
1101 }
1102}

1104void runFastConv(InputArray _input, OutputArray _output, const Ptr<FastConv>& conv, int ntasks,
1105 const Ptr<ActivationLayer>& actLayer, const std::vector<float>& reluslope, bool fusedAdd)
1106{
1107 Mat input = _input.getMat();
1108 Mat output = _output.getMat();
1109 int conv_dim = conv->conv_dim;
1110
1111 CV_Assert_N(input.dims == output.dims,
1112 input.size[0] == output.size[0],
1113 conv->C == input.size[1],
1114 conv->K == output.size[1],
1115 input.type() == output.type(),
1116 input.isContinuous(),
1117 output.isContinuous());
1118
1119 const bool useFP16 = conv->useFP16;
1120 Mat fusedAddMat;
1121 if (fusedAdd)
1122 {
1123 CV_Assert(conv->conv_dim != CONV_3D && "Conv3D does not support Conv+Add fusion optimization!");
1124 fusedAddMat = _output.getMat();
1125 }
1126
1127 if (conv->conv_type == CONV_TYPE_DEPTHWISE)
1128 {
1129 // Depthwise-Convolution layer should not be followed by Add layer.
1130 CV_Assert((conv_dim == CONV_1D || conv_dim == CONV_2D) && !useFP16);
1131 return runDepthwise(input, output, conv, actLayer.get(), reluslope, fusedAdd);
1132 }
1133
1134 MatShape inputShape = shape(input);
1135 MatShape outputShape = shape(output);
1136
1137 CV_Assert(inputShape.size() == outputShape.size());
1138
1139 ActivationLayer* activ = nullptr;
1140 float minval = -FLT_MAX, maxval = FLT_MAX;
1141 bool ifMinMaxAct = false;
1142
1143 if (actLayer)
1144 {
1145 Ptr<ReLULayer> activ_relu = actLayer.dynamicCast<ReLULayer>();
1146 Ptr<ReLU6Layer> activ_relu6 = actLayer.dynamicCast<ReLU6Layer>();
1147
1148 if (!activ_relu.empty())
1149 {
1150 if (activ_relu->negativeSlope == 0.0f)
1151 {
1152 minval = 0.0f;
1153 ifMinMaxAct = true;
1154 activ = nullptr;
1155 }
1156 else // Leaky ReLU
1157 {
1158 activ = actLayer.get();
1159 }
1160 }
1161 else if (!activ_relu6.empty())
1162 {
1163 minval = activ_relu6->minValue;
1164 maxval = activ_relu6->maxValue;
1165
1166 ifMinMaxAct = true;
1167 activ = nullptr;
1168 }
1169 else
1170 activ = actLayer.get();
1171 }
1172 else
1173 activ = nullptr;
1174
1175 if (conv->conv_type == CONV_TYPE_WINOGRAD3X3) // winograd
1176 {
1177 CV_Assert((!conv->weightsWinoBuf.empty() || !conv->weightsWinoBuf_FP16.empty()) && input.dims == 4 && conv_dim == CONV_2D);
1178 if (runWinograd63(input, fusedAddMat, output, conv, ntasks, minval, maxval, activ, ifMinMaxAct))
1179 return;
1180 }
1181
1182 int N = inputShape[0], C = inputShape[1];
1183
1184 // input shape: [N, C, D, H, W] for Conv3D, [N, C, H, W] for Conv2D, [N, C, W] for Conv1D.
1185 int Di = conv_dim == CONV_3D ? inputShape[2] : 1;
1186 int Hi = conv_dim == CONV_1D ? 1 : inputShape[inputShape.size() - 2];
1187 int Wi = inputShape[inputShape.size() - 1];
1188
1189 int ngroups = conv->ngroups;
1190 int K = conv->K, Dk = conv->Dk, Hk = conv->Hk, Wk = conv->Wk;
1191
1192 int D0 = conv_dim == CONV_3D ? outputShape[2] : 1;
1193 int H0 = conv_dim == CONV_1D ? 1 : outputShape[outputShape.size() - 2];
1194 int W0 = outputShape[outputShape.size() - 1];
1195
1196 int Cg = C/ngroups, Kg = K/ngroups;
1197
1198 const size_t inp_planesize = (size_t)Di*Hi*Wi;
1199 const size_t out_planesize = (size_t)D0*H0*W0;
1200
1201 int pad_front = conv->pad_front;
1202 int pad_top = conv->pad_top;
1203 int pad_left = conv->pad_left;
1204
1205 int stride_d = conv->stride_d, stride_h = conv->stride_h, stride_w = conv->stride_w;
1206 int dilation_d = conv->dilation_d, dilation_h = conv->dilation_h, dilation_w = conv->dilation_w;
1207
1208 int ksize = Dk*Hk*Wk;
1209 bool fast_1x1 = ksize == 1 && stride_d == 1 && stride_w == 1 && stride_h == 1
1210 && pad_front == 0 && pad_left == 0 && pad_top == 0;
1211 int DkHkWkCg = Dk*Hk*Wk*Cg;
1212
1213 std::vector<int> ofstab_(Hk*Wk*Dk*4, 0);
1214 int* ofstab = ofstab_.data();
1215 int* dhwTab = ofstab + Hk*Wk*Dk;
1216 int padded_ksize = ((ksize + VEC_ALIGN-1) / VEC_ALIGN) * VEC_ALIGN;
1217
1218 if (conv_dim == CONV_1D)
1219 {
1220 for( int w = 0; w < Wk; w++)
1221 {
1222 int dw = w*dilation_w;
1223 dhwTab[w*3+2] = dw;
1224 ofstab[w] = dw;
1225 }
1226 }
1227 else if (conv_dim == CONV_2D)
1228 {
1229 for (int h = 0; h < Hk; h++)
1230 for( int w = 0; w < Wk; w++)
1231 {
1232 int k = h*Wk + w;
1233 int dh = h*dilation_h, dw = w*dilation_w;
1234 dhwTab[k*3+1] = dh;
1235 dhwTab[k*3+2] = dw;
1236 ofstab[k] = dh*Wi + dw;
1237 }
1238 }
1239 else
1240 {
1241 for (int d = 0; d < Dk; d++)
1242 for (int h = 0; h < Hk; h++)
1243 {
1244 for (int w = 0; w < Wk; w++)
1245 {
1246 int k = d*Hk*Wk + h*Wk + w;
1247 int dd = d*dilation_d, dh = h*dilation_h, dw = w*dilation_w;
1248 dhwTab[k*3] = dd;
1249 dhwTab[k*3+1] = dh;
1250 dhwTab[k*3+2] = dw;
1251 ofstab[k] = dd*Hi*Wi + dh*Wi + dw;
1252 }
1253 }
1254 }
1255
1256 int CONV_NR = CONV_NR_FP32;
1257 int CONV_MR = CONV_MR_FP32;
1258 int esz = sizeof(float );
1259
1260#ifdef CONV_ARM_FP16
1261 if (useFP16)
1262 {
1263 // works at FP 16.
1264 CONV_NR = CONV_NR_FP16;
1265 CONV_MR = CONV_MR_FP16;
1266 esz = sizeof(__fp16);
1267 }
1268#endif
1269
1270 int MAX_STRIPES = conv->conv_type == CONV_TYPE_DEPTHWISE_REMAIN ? 1 : (56 + CONV_NR - 1)/CONV_NR;
1271
1272 // Friendly to L1 cache
1273 const int K_BLOCK_SIZE = conv->conv_type == CONV_TYPE_DEPTHWISE_REMAIN ? 1 : 32;
1274 const int C_BLOCK_SIZE = 256;
1275
1276 int Kg_nblocks = (Kg + CONV_MR-1)/CONV_MR;
1277 int Kg_aligned = Kg_nblocks * CONV_MR;
1278
1279 int stripes_per_plane0 = ((int)out_planesize + CONV_NR - 1) / CONV_NR;
1280 int stripes_per_plane = stripes_per_plane0;
1281
1282 if (stripes_per_plane < ntasks * 4 || conv->conv_type == CONV_TYPE_DEPTHWISE_REMAIN)
1283 {
1284 MAX_STRIPES = 1;
1285 stripes_per_plane = 1;
1286 }
1287 else
1288 Kg_nblocks = 1;
1289
1290 bool separateIm2col = (fast_1x1 || stripes_per_plane == 1) && conv->conv_type != CONV_TYPE_DEPTHWISE_REMAIN;
1291
1292 int Kstripes = Kg_nblocks * stripes_per_plane;
1293 int nsubtasks = N * ngroups * Kstripes;
1294
1295 size_t stripesize = alignSize(CONV_NR * ksize * Cg, VEC_ALIGN);
1296 size_t cbufsize = alignSize(CONV_NR * K_BLOCK_SIZE * MAX_STRIPES, VEC_ALIGN);
1297
1298 size_t taskbufsize = cbufsize * sizeof(float );
1299
1300 if (!separateIm2col)
1301 taskbufsize += MAX_STRIPES * stripesize * esz;
1302
1303 size_t totalbufsize_base = taskbufsize * ntasks;
1304 size_t totalbufsize = totalbufsize_base;
1305 if (separateIm2col)
1306 totalbufsize += N * ngroups * stripes_per_plane0 * stripesize * esz;
1307
1308 AutoBuffer<char> inpbuf_all_;
1309 char* inpbuf_all = nullptr;
1310
1311 inpbuf_all_.allocate(totalbufsize + VEC_ALIGN * sizeof(float ));
1312 inpbuf_all = alignPtr(inpbuf_all_.data(), (int)(VEC_ALIGN * sizeof(float )));
1313 char* inpbuf_all_0 = inpbuf_all + totalbufsize_base;
1314
1315 float* inp = input.ptr<float>();
1316 float* out = output.ptr<float>();
1317 float* fusedAddPtr0 = fusedAddMat.empty() ? 0 : fusedAddMat.ptr<float>();
1318
1319 // In the case of 1x1 convolution we first reorder the whole input tensor.
1320 // In general, im2row results in Hk*Wk-x unrolling factor
1321 // (e.g. 3*3=9x unrolling for 3x3 convolution), thus for 1x1 convolution
1322 // the reordered tensor will take as much space as the original tensor.
1323 if (separateIm2col)
1324 {
1325 // the optional phase 1. im2row
1326 parallel_for_(Range(0, ntasks), [&](const Range& r0) {
1327 for (int task_id = r0.start; task_id < r0.end; task_id++)
1328 {
1329 if (fast_1x1)
1330 {
1331 int nc0 = task_id*N*C/ntasks, nc1 = (task_id+1)*N*C/ntasks, dc = 0;
1332 for (; nc0 < nc1; nc0 += dc)
1333 {
1334 int n = nc0/C, c0 = nc0 - n*C;
1335 int g = c0 / Cg;
1336 c0 -= g*Cg;
1337 dc = Cg - c0 <= nc1 - nc0 ? Cg - c0 : nc1 - nc0;
1338
1339 float * inptr_ = inp + (size_t)nc0*inp_planesize;
1340 char* inpbuf_ = inpbuf_all_0 + ((n*ngroups + g)*stripes_per_plane0*stripesize + c0*CONV_NR)*esz;
1341
1342 packInputData(inpbuf_, inptr_, ofstab, dhwTab, 0, out_planesize, ksize, stride_d, stride_h,
1343 stride_w, pad_front, pad_top, pad_left, Dk, Hk, Wk, dilation_d, dilation_h, dilation_w,
1344 Di, Hi, Wi, H0, W0, dc, stripesize, 0, inp_planesize, conv->conv_dim,
1345 conv->conv_type, CONV_NR, esz, fast_1x1, useFP16);
1346 }
1347 }
1348 else
1349 {
1350 const int allTasks = N * ngroups * stripes_per_plane0;
1351 int ngs0 = task_id*allTasks/ntasks, ngs1 = (task_id+1)*allTasks/ntasks, ds = 0;
1352
1353 for (; ngs0 < ngs1; ngs0 += ds)
1354 {
1355 int n = ngs0 / (ngroups * stripes_per_plane0), gs0 = ngs0 - n*ngroups*stripes_per_plane0;
1356 int g = gs0 / stripes_per_plane0, s0 = gs0 - g*stripes_per_plane0;
1357
1358 ds = stripes_per_plane0 - s0 <= ngs1 - ngs0 ? stripes_per_plane0 - s0 : ngs1 - ngs0;
1359
1360 int zyx = s0 * CONV_NR;
1361 int zyx_limit = (s0 + ds) * CONV_NR < out_planesize ? (s0 + ds) * CONV_NR : out_planesize;
1362
1363 float * inptr_ = inp + (size_t)(n * ngroups + g) * Cg * inp_planesize;
1364 char* inpbuf_ = inpbuf_all_0 + ((n * ngroups + g) * stripes_per_plane0 * stripesize + s0 * stripesize) * esz;
1365
1366 packInputData(inpbuf_, inptr_, ofstab, dhwTab, zyx, zyx_limit, ksize, stride_d, stride_h,
1367 stride_w, pad_front, pad_top, pad_left, Dk, Hk, Wk, dilation_d, dilation_h, dilation_w,
1368 Di, Hi, Wi, H0, W0, Cg, stripesize, 0, inp_planesize, conv->conv_dim,
1369 conv->conv_type, CONV_NR, esz, fast_1x1, useFP16);
1370 }
1371 }
1372 }
1373 });
1374 }
1375
1376 // Compute
1377 parallel_for_(Range(0, ntasks), [&](const Range& r0) {
1378 for (int task_id = r0.start; task_id < r0.end; task_id++)
1379 {
1380 float * cbuf_task = (float *)(inpbuf_all + taskbufsize * task_id);
1381 char * inpbuf_task = (char*)(cbuf_task + cbufsize);
1382
1383 int ngs0 = (int)((size_t)nsubtasks * task_id / ntasks);
1384 int ngs1 = (int)((size_t)nsubtasks * (task_id+1) / ntasks);
1385 for (int subtask = ngs0; subtask < ngs1; )
1386 {
1387 int ng = subtask / Kstripes;
1388 int kzyx0 = subtask - ng * Kstripes;
1389 int kzyx1 = kzyx0 + (ngs1 - subtask);
1390 int n = ng / ngroups, g = ng % ngroups; // ng - n * ngroups;
1391 size_t inp_plane_ofs = (size_t)(n * ngroups + g) * Cg * inp_planesize;
1392 kzyx1 = kzyx1 <= Kstripes ? kzyx1 : Kstripes;
1393 subtask += kzyx1 - kzyx0;
1394 int k0, k1;
1395 int zyx0, zyx_limit, zyx_block_limit = 0;
1396
1397 if (stripes_per_plane == 1 || conv->conv_type == CONV_TYPE_DEPTHWISE_REMAIN)
1398 {
1399 k0 = kzyx0 * CONV_MR;
1400 k1 = kzyx1 * CONV_MR;
1401 k1 = k1 <= Kg ? k1 : Kg;
1402 zyx0 = 0;
1403 zyx_limit = (int)out_planesize;
1404 }
1405 else
1406 {
1407 k0 = 0;
1408 k1 = Kg;
1409 zyx0 = kzyx0 * CONV_NR;
1410 zyx_limit = kzyx1 * CONV_NR;
1411 zyx_limit = zyx_limit < out_planesize ? zyx_limit : (int)out_planesize;
1412 }
1413
1414 for (; zyx0 < zyx_limit; zyx0 = zyx_block_limit)
1415 {
1416 // step 1. extract part of input tensor and represent it in zigzag form
1417 zyx_block_limit = zyx0 + CONV_NR * MAX_STRIPES;
1418 zyx_block_limit = zyx_block_limit < zyx_limit ? zyx_block_limit : zyx_limit;
1419
1420 int nstripes = (zyx_block_limit - zyx0 + CONV_NR - 1) / CONV_NR;
1421
1422 CV_Assert(nstripes <= MAX_STRIPES);
1423
1424 if (!separateIm2col)
1425 {
1426 packInputData(inpbuf_task, inp, ofstab, dhwTab, zyx0, zyx_block_limit, ksize, stride_d, stride_h,
1427 stride_w, pad_front, pad_top, pad_left, Dk, Hk, Wk, dilation_d, dilation_h, dilation_w,
1428 Di, Hi, Wi, H0, W0, Cg, stripesize, inp_plane_ofs, inp_planesize, conv->conv_dim,
1429 conv->conv_type, CONV_NR, esz, fast_1x1, useFP16);
1430 }
1431
1432 char *weights = nullptr;
1433#ifdef CONV_ARM_FP16
1434 if (useFP16)
1435 {
1436 CV_Assert(!conv->weightsBuf_FP16.empty());
1437 weights = (char *)conv->getWeightsFP16();
1438 }
1439 else
1440#endif
1441 {
1442 CV_Assert(!conv->weightsBuf.empty());
1443 weights = (char *)conv->getWeights();
1444 }
1445 // optional branch, only for depth-wise convolution which was implemented by generic convolution.
1446 // In this case, CONV_MR is 1, and CONV_NR remains the same.
1447 if (conv->conv_type == CONV_TYPE_DEPTHWISE_REMAIN)
1448 {
1449 CV_Assert(weights);
1450 size_t outofs = (n * ngroups + g) * out_planesize + zyx0;
1451 float *cptr0 = cbuf_task;
1452 weights += g * padded_ksize * esz;
1453
1454 int out_width = zyx_block_limit - zyx0;
1455 float *outptr = out + outofs;
1456 const float biasVal = *(conv->biasBuf.data() + g);
1457 const char *inptr_ = separateIm2col ? inpbuf_all_0 + (ng * stripes_per_plane0 + zyx0 / CONV_NR) * stripesize * esz :
1458 inpbuf_task;
1459
1460 for (int stripe = 0; stripe < nstripes; stripe++)
1461 {
1462 const char *inptr = inptr_ + stripe * stripesize * esz;
1463 const int outLen = std::min(out_width - stripe * CONV_NR, CONV_NR);
1464 bool ifBuffer = outLen < CONV_NR;
1465 float *cptr = outptr + stripe * CONV_NR;
1466 if (ifBuffer)
1467 {
1468 memcpy(cptr0, cptr, outLen * sizeof(float ));
1469 cptr = cptr0;
1470 }
1471#if CV_NEON && CV_NEON_AARCH64
1472 if (conv->useNEON)
1473 {
1474#ifdef CONV_ARM_FP16
1475 if (useFP16)
1476 {
1477 opt_NEON_FP16::convBlockMR1_F16(DkHkWkCg, weights, inptr, cptr, biasVal, fusedAdd, minval, maxval, ifMinMaxAct, outLen, CONV_NR);
1478 }
1479 else
1480#endif
1481 opt_NEON::convBlockMR1_F32(DkHkWkCg, (const float *)weights, (const float *)inptr, cptr, biasVal, fusedAdd, minval, maxval, ifMinMaxAct, outLen, CONV_NR);
1482 }
1483 else
1484#endif
1485 convBlockMR1_F32(DkHkWkCg, (const float *)weights, (const float *)inptr, cptr, biasVal, fusedAdd, minval, maxval, ifMinMaxAct, outLen, CONV_NR);
1486
1487 if (ifBuffer)
1488 {
1489 memcpy(outptr + stripe * CONV_NR, cptr, outLen * sizeof(float ));
1490 }
1491 }
1492 if (activ)
1493 activ->forwardSlice(outptr, outptr, out_width, out_planesize, g, g + 1);
1494 continue;
1495 }
1496
1497 CV_Assert(weights);
1498 weights += g * Kg_aligned * DkHkWkCg * esz;
1499
1500 const float *biasptr = conv->biasBuf.data() + Kg * g;
1501 int ldc = nstripes * CONV_NR;
1502
1503 // 2. do convolution, compute Kg x (zyx_block_limit - zyx0) part of the output tensor
1504 int out_width = zyx_block_limit - zyx0;
1505 for (int k0_block = k0; k0_block < k1; k0_block += K_BLOCK_SIZE)
1506 {
1507 int k1_block = k0_block + K_BLOCK_SIZE < k1 ? k0_block + K_BLOCK_SIZE : k1;
1508 for (int c0 = 0; c0 < DkHkWkCg; c0 += C_BLOCK_SIZE)
1509 {
1510 int c1 = c0 + C_BLOCK_SIZE < DkHkWkCg ? c0 + C_BLOCK_SIZE : DkHkWkCg;
1511 const char *inptr = separateIm2col ? inpbuf_all_0 + (ng * stripes_per_plane0 + zyx0 / CONV_NR) * stripesize * esz :
1512 inpbuf_task;
1513 inptr += (c0 * CONV_NR) * esz;
1514 for (int stripe = 0; stripe < nstripes; stripe++, inptr += stripesize * esz)
1515 {
1516 const int outLen = std::min(out_width - stripe * CONV_NR, CONV_NR);
1517
1518 char *wptr = weights + (k0_block * DkHkWkCg + c0 * CONV_MR) * esz;
1519 float *cptr = cbuf_task + stripe * CONV_NR;
1520#ifdef CONV_ARM_FP16
1521 hfloat* cptr_f16 = (hfloat*)cbuf_task + stripe*CONV_NR;
1522#endif // CONV_ARM_FP16
1523 for (int k = k0_block; k < k1_block; k += CONV_MR,
1524 wptr += DkHkWkCg * CONV_MR * esz,
1525#ifdef CONV_ARM_FP16
1526 cptr_f16 += CONV_MR * ldc,
1527#endif // CONV_ARM_FP16
1528 cptr += CONV_MR * ldc)
1529 {
1530#if CV_TRY_AVX2
1531 if (conv->useAVX2)
1532 opt_AVX2::convBlock_F32(c1 - c0, (const float *)wptr, (const float *)inptr, cptr, ldc, c0 == 0, outLen, CONV_MR, CONV_NR);
1533 else
1534#endif
1535#if CV_TRY_AVX
1536 if (conv->useAVX)
1537 opt_AVX::convBlock_F32(c1 - c0, (const float *)wptr, (const float *)inptr, cptr, ldc, c0 == 0, outLen, CONV_MR, CONV_NR);
1538 else
1539#endif
1540#if CV_NEON
1541 if (conv->useNEON)
1542 {
1543#ifdef CONV_ARM_FP16
1544 if (useFP16)
1545 {
1546 opt_NEON_FP16::convBlock_F16(c1 - c0, wptr, inptr, (char *)cptr_f16, ldc, c0 == 0, outLen, CONV_MR, CONV_NR);
1547 }
1548 else
1549#endif
1550 opt_NEON::convBlock_F32(c1 - c0, (const float *)wptr, (const float *)inptr, cptr, ldc, c0 == 0, outLen, CONV_MR, CONV_NR);
1551 }
1552 else
1553#endif
1554 // The possible outLen range is 24 or 8~1.
1555 convBlock_F32(c1 - c0, (const float *)wptr, (const float *)inptr, cptr, ldc, c0 == 0, outLen, CONV_MR, CONV_NR);
1556 }
1557 }
1558 }
1559
1560 size_t outofs = ((n * ngroups + g) * Kg + k0_block) * out_planesize + zyx0;
1561 const float *cptr = cbuf_task;
1562#ifdef CONV_ARM_FP16
1563 const hfloat *cptr_fp16 = (const hfloat *)cbuf_task;
1564#endif // CONV_ARM_FP16
1565 float *outptr = out + outofs;
1566 const float *pbptr = fusedAddPtr0 ? fusedAddPtr0 + outofs : 0;
1567
1568 for (int k = k0_block; k < k1_block; k++,
1569 cptr += ldc,
1570#ifdef CONV_ARM_FP16
1571 cptr_fp16 += ldc,
1572#endif // CONV_ARM_FP16

1910 c1 = __riscv_vfmin_vf_f32m1(__riscv_vfmax_vf_f32m1(c1, minval, vl), maxval, vl);
1911 c2 = __riscv_vfmin_vf_f32m1(__riscv_vfmax_vf_f32m1(c2, minval, vl), maxval, vl);
1912 }

1914 __riscv_vse32_v_f32m1(c + vl, c1, vl);
1915 __riscv_vse32_v_f32m1(c + 2 * vl, c2, vl);
1916 return;
1917 }
1918 convBlockMR1NoSIMD(np, a, b, c, bias, init_c, minval, maxval, ifMinMaxAct, outLen, convNR);
1919}
