68 int kercn = oclop == OCL_OP_PHASE_DEGREES ||

85 _dst.createSameSize(src1, type);

103/* ************************************************************************** *\
104 Fast cube root by Ken Turkowski
105 (http://www.worldserver.com/turk/computergraphics/papers.html)
106\* ************************************************************************** */
107CV_DISABLE_UBSAN float cubeRoot( float value )
108{
109 CV_INSTRUMENT_REGION();
110
111 float fr;
112 Cv32suf v, m;
113 int ix, s;
114 int ex, shx;
115
116 v.f = value;
117 ix = v.i & 0x7fffffff;
118 s = v.i & 0x80000000;
119 ex = (ix >> 23) - 127;
120 shx = ex % 3;
121 shx -= shx >= 0 ? 3 : 0;
122 ex = (ex - shx) / 3; /* exponent of cube root */
123 v.i = (ix & ((1<<23)-1)) | ((shx + 127)<<23);
124 fr = v.f;
125
126 /* 0.125 <= fr < 1.0 */
127 /* Use quartic rational polynomial with error < 2^(-24) */
128 fr = (float)(((((45.2548339756803022511987494 * fr +
129 192.2798368355061050458134625) * fr +
130 119.1654824285581628956914143) * fr +
131 13.43250139086239872172837314) * fr +
132 0.1636161226585754240958355063)/
133 ((((14.80884093219134573786480845 * fr +
134 151.9714051044435648658557668) * fr +
135 168.5254414101568283957668343) * fr +
136 33.9905941350215598754191872) * fr +
137 1.0));
138
139 /* fr *= 2^ex * sign */
140 m.f = value;
141 v.f = fr;
142 v.i = (v.i + (ex << 23) + s) & (m.i*2 != 0 ? -1 : 0);
143 return v.f;
144}

440// The master function of the unary math family (cv::exp/log/sqrt - the analogue of arithm_op):
441// same-shape same-type output over the four float depths, computed by the element-wise engine's
442// kernels (math.simd.hpp). Two tiers:
443// - SMALL and continuous (a common pattern - exp() over one image row as a lookup substitute):
444// call the kernel DIRECTLY over the flattened elements. No TExpr, no broadcastOp, no
445// parallel_for machinery - their setup dominates at these sizes.
446// - everything else (large arrays - worth parallelizing; ROIs - need real steps): the usual
447// 1-instruction program via compile()/exec().
448enum { MATH_OP_SMALL = 100000 }; // elements; tune with a benchmark if the crossover moves
449
450static void math_op(ew::TOp op, InputArray _src, OutputArray _dst)
451{
452 int type = _src.type(), depth = CV_MAT_DEPTH(type);
453 CV_Assert(depth == CV_16F || depth == CV_16BF || depth == CV_32F || depth == CV_64F);
454
455 Mat src = _src.getMat();
456 _dst.createSameSize(_src, type); // whole-shape transfer (layout & future metadata included)
457 Mat dst = _dst.getMat();
458 if (src.empty())
459 return;
460
461 const size_t total = src.total() * src.channels();
462 if (src.isContinuous() && dst.isContinuous() && total <= (size_t)MATH_OP_SMALL)
463 {
464 ew::TKernel k = ew::getMathFunc(op, depth);
465 CV_Assert(k.fptr);
466 static const double noparams[4] = {};
467 k.fptr(src.data, 0, 1, nullptr, 0, 0, nullptr, 0, 0,
468 dst.data, 0, (int)total, 1, noparams, k.flags, k.userdata);
469 return;
470 }
471
472 ew::TExpr p;
473 const int a = p.addInput(depth);
474 const int out = p.addOutput(depth);
475 p.moveToOutput(p.emitUnary(op, a, depth), out);
476 p.compile();
477 const Mat* inputs[] = { &src };
478 p.exec(inputs, &dst);
479}

514template <typename T, typename WT>
515struct iPow_SIMD
516{
517 int operator() ( const T *, T *, int, int)
518 {
519 return 0;
520 }
521};

523#if (CV_SIMD || CV_SIMD_SCALABLE)
524
525template <>
526struct iPow_SIMD<uchar, unsigned>
527{
528 int operator() ( const uchar * src, uchar * dst, int len, int power )
529 {
530 int i = 0;
531 v_uint32 v_1 = vx_setall_u32(1u);
532
533 for ( ; i <= len - VTraits<v_uint16>::vlanes(); i += VTraits<v_uint16>::vlanes())
534 {
535 v_uint32 v_a1 = v_1, v_a2 = v_1;
536 v_uint16 v = vx_load_expand(src + i);
537 v_uint32 v_b1, v_b2;
538 v_expand(v, v_b1, v_b2);
539 int p = power;
540
541 while( p > 1 )
542 {
543 if (p & 1)
544 {
545 v_a1 = v_mul(v_a1, v_b1);
546 v_a2 = v_mul(v_a2, v_b2);
547 }
548 v_b1 = v_mul(v_b1, v_b1);
549 v_b2 = v_mul(v_b2, v_b2);
550 p >>= 1;
551 }
552
553 v_a1 = v_mul(v_a1, v_b1);
554 v_a2 = v_mul(v_a2, v_b2);
555
556 v = v_pack(v_a1, v_a2);
557 v_pack_store(dst + i, v);
558 }
559 vx_cleanup();

813template<typename T, typename WT>
814static void
815iPow_i( const T* src, T* dst, int len, int power )
816{
817 if( power < 0 )
818 {
819 T tab[5] =
820 {
821 saturate_cast<T>(power == -1 ? -1 : 0), saturate_cast<T>((power & 1) ? -1 : 1),
822 std::numeric_limits<T>::max(), 1, saturate_cast<T>(power == -1 ? 1 : 0)
823 };
824 for( int i = 0; i < len; i++ )
825 {
826 T val = src[i];
827 dst[i] = cv_abs(val) <= 2 ? tab[val + 2] : (T)0;
828 }
829 }
830 else
831 {
832 iPow_SIMD<T, WT> vop;
833 int i = vop(src, dst, len, power);
834
835 for( ; i < len; i++ )
836 {
837 WT a = 1, b = src[i];
838 int p = power;
839 while( p > 1 )
840 {
841 if( p & 1 )
842 a *= b;
843 b *= b;
844 p >>= 1;
845 }
846
847 a *= b;
848 dst[i] = saturate_cast<T>(a);
849 }
850 }
851}

855iPow_f( const T* src, T* dst, int len, int power0 )

1016void pow( InputArray _src, double power, OutputArray _dst )
1017{
1018 CV_INSTRUMENT_REGION();
1019
1020 int type = _src.type(), depth = CV_MAT_DEPTH(type),
1021 cn = CV_MAT_CN(type), ipower = cvRound(power);
1022 bool is_ipower = fabs(ipower - power) < DBL_EPSILON;
1023#ifdef HAVE_OPENCL
1024 bool useOpenCL = _dst.isUMat() && _src.dims() <= 2;
1025#endif
1026
1027 if (is_ipower)
1028 {
1029 switch( ipower )
1030 {
1031 case 0:
1032 _dst.createSameSize(_src, type);
1033 _dst.setTo(Scalar::all(1));
1034 return;
1035 case 1:
1036 _src.copyTo(_dst);
1037 return;
1038 case 2:
1039 multiply(_src, _src, _dst);
1040 return;
1041 }
1042 }
1043
1044 CV_OCL_RUN(useOpenCL, ocl_pow(_src, power, _dst, is_ipower, ipower))
1045
1046 const bool floatDepth = depth == CV_16F || depth == CV_16BF || depth == CV_32F || depth == CV_64F;
1047
1048 // INTEGER array ** INTEGER power: keep the classic iPow kernels (an exact multiply chain with
1049 // the classic wrap-around semantics) - full bit-exact compatibility for whoever relies on it.
1050 // Everything else - any power on a float array, a fractional power on an integer one (computed
1051 // in the float domain and saturated back), plus the 32U/64-bit depths iPow never supported -
1052 // goes through the engine below.
1053 if( is_ipower && !floatDepth && ipowTab[depth] )
1054 {
1055 Mat src = _src.getMat();
1056 _dst.createSameSize(_src, type);
1057 Mat dst = _dst.getMat();
1058
1059 const Mat* arrays[] = {&src, &dst, 0};
1060 uchar* ptrs[2] = {};
1061 NAryMatIterator it(arrays, ptrs);
1062 int len = (int)(it.size*cn);
1063 IPowFunc func = ipowTab[depth];
1064
1065 for( size_t i = 0; i < it.nplanes; i++, ++it )
1066 func( ptrs[0], ptrs[1], len, ipower );
1067 return;
1068 }
1069
1070 // The engine path, two tiers like math_op: the pow kernel special-cases the exponents
1071 // 3/0.5 (2/1/0 never reach here) per row and vectorizes the general exp(p*log x) with an exact
1072 // std::pow patch for non-positive bases (0^negative -> inf, negative^fractional -> NaN).
1073 Mat src = _src.getMat();
1074 _dst.createSameSize(_src, type);
1075 Mat dst = _dst.getMat();
1076 if (src.empty())
1077 return;
1078
1079 const size_t total = src.total() * cn;
1080 if (floatDepth && src.isContinuous() && dst.isContinuous() && total <= (size_t)MATH_OP_SMALL)
1081 {
1082 ew::TKernel k = ew::getPowFunc(depth, depth);
1083 if (k.fptr)
1084 {
1085 double pvstore; // the broadcast exponent, stored as T
1086 void* pv = &pvstore;
1087 switch (depth)
1088 {
1089 case CV_16F: *(hfloat*)pv = saturate_cast<hfloat>(power); break;
1090 case CV_16BF: *(bfloat*)pv = saturate_cast<bfloat>(power); break;
1091 case CV_32F: *(float*)pv = (float)power; break;
1092 default: pvstore = power; break;
1093 }
1094 static const double noparams[4] = {};
1095 k.fptr(src.data, 0, 1, pv, 0, 0, nullptr, 0, 0,
1096 dst.data, 0, (int)total, 1, noparams, k.flags, k.userdata);
1097 return;
1098 }
1099 }
1100
1101 ew::TExpr prog;
1102 const int a = prog.addInput(depth);
1103 const int c = prog.addConst(ew::EW_DEPTH_NONE, Scalar(power), 1);
1104 const int out = prog.addOutput(depth);
1105 prog.moveToOutput(prog.emitBinary(ew::OP_POW, a, c, depth), out);
1106 prog.compile();
1107 const Mat* inputs[] = { &src };
1108 prog.exec(inputs, &dst);
1109}

1345int cv::solveCubic( InputArray _coeffs, OutputArray _roots )
1346{
1347 CV_INSTRUMENT_REGION();
1348
1349 const int n0 = 3;
1350 Mat coeffs = _coeffs.getMat();
1351 int ctype = coeffs.type();
1352
1353 CV_Assert( ctype == CV_32F || ctype == CV_64F );
1354 CV_Assert( (coeffs.size() == Size(n0, 1) ||
1355 coeffs.size() == Size(n0+1, 1) ||
1356 coeffs.size() == Size(1, n0) ||
1357 coeffs.size() == Size(1, n0+1)) );
1358
1359 _roots.create(n0, 1, ctype, -1, true, _OutputArray::DEPTH_MASK_FLT);
1360 Mat roots = _roots.getMat();
1361
1362 int i = -1, n = 0;
1363 double a0 = 1., a1, a2, a3;
1364 double x0 = 0., x1 = 0., x2 = 0.;
1365 int ncoeffs = coeffs.rows + coeffs.cols - 1;
1366
1367 if( ctype == CV_32FC1 )
1368 {
1369 if( ncoeffs == 4 )
1370 a0 = coeffs.at<float>(++i);
1371
1372 a1 = coeffs.at<float>(i+1);
1373 a2 = coeffs.at<float>(i+2);
1374 a3 = coeffs.at<float>(i+3);
1375 }
1376 else
1377 {
1378 if( ncoeffs == 4 )
1379 a0 = coeffs.at<double>(++i);
1380
1381 a1 = coeffs.at<double>(i+1);
1382 a2 = coeffs.at<double>(i+2);
1383 a3 = coeffs.at<double>(i+3);
1384 }
1385
1386 // Fix for Issue #27748: Normalize coefficients to avoid overflow/underflow
1387 // and correctly detect negligible cubic terms.
1388 double max_coeff = std::max({std::abs(a0), std::abs(a1), std::abs(a2), std::abs(a3)});
1389
1390 // If max_coeff is effectively zero, the equation is 0=0
1391 if (max_coeff < std::numeric_limits<double>::epsilon())
1392 return -1;
1393
1394 // Check if the cubic term is negligible relative to the largest coefficient
1395 if( std::abs(a0) < std::numeric_limits<double>::epsilon() * max_coeff )
1396 {
1397 if( std::abs(a1) < std::numeric_limits<double>::epsilon() * max_coeff )
1398 {
1399 if( std::abs(a2) < std::numeric_limits<double>::epsilon() * max_coeff )
1400 n = std::abs(a3) < std::numeric_limits<double>::epsilon() * max_coeff ? -1 : 0;
1401 else
1402 {
1403 // linear equation
1404 x0 = -a3/a2;
1405 n = 1;
1406 }
1407 }
1408 else
1409 {
1410 // quadratic equation
1411 double d = a2*a2 - 4*a1*a3;
1412 if( d >= 0 )
1413 {
1414 d = std::sqrt(d);
1415 double q1 = (-a2 + d) * 0.5;
1416 double q2 = (a2 + d) * -0.5;
1417 if( std::abs(q1) > std::abs(q2) )
1418 {
1419 x0 = q1 / a1;
1420 x1 = a3 / q1;
1421 }
1422 else
1423 {
1424 x0 = q2 / a1;
1425 x1 = a3 / q2;
1426 }
1427 n = d > 0 ? 2 : 1;
1428 }
1429 }
1430 }
1431 else
1432 {
1433 // cubic equation
1434 // Normalize coefficients in-place to prevent overflow/instability
1435 double scale = 1.0 / max_coeff;
1436 a0 *= scale;
1437 a1 *= scale;
1438 a2 *= scale;
1439 a3 *= scale;
1440
1441 a0 = 1./a0;
1442 a1 *= a0;
1443 a2 *= a0;
1444 a3 *= a0;
1445
1446 double Q = (a1 * a1 - 3 * a2) * (1./9);
1447 double R = (a1 * (2 * a1 * a1 - 9 * a2) + 27 * a3) * (1./54);
1448 double Qcubed = Q * Q * Q;
1449
1450 /*
1451 Here we expand expression `Qcubed - R * R` for `d` variable
1452 to reduce common terms `a1^6 / 729` and `-a1^4 * a2 / 81`
1453 and thus decrease rounding error (in case of quite big coefficients).
1454
1455 And then we additionally group terms to further reduce rounding error.
1456 */
1457 double d = (a1 * a1 * (a2 * a2 - 4 * a1 * a3) + 2 * a2 * (9 * a1 * a3 - 2 * a2 * a2) - 27 * a3 * a3) * (1./108);
1458
1459 if( d > 0 )
1460 {
1461 double theta = acos(R / sqrt(Qcubed));
1462 double sqrtQ = sqrt(Q);
1463 double t0 = -2 * sqrtQ;
1464 double t1 = theta * (1./3);
1465 double t2 = a1 * (1./3);
1466 x0 = t0 * cos(t1) - t2;
1467 x1 = t0 * cos(t1 + (2.*CV_PI/3)) - t2;
1468 x2 = t0 * cos(t1 + (4.*CV_PI/3)) - t2;
1469 n = 3;
1470 }
1471 else if( d == 0 )
1472 {
1473 x0 = -2*std::cbrt(R) - a1/3;
1474 x1 = std::cbrt(R) - a1/3;
1475 x2 = 0;
1476 n = x0 == x1 ? 1 : 2;
1477 x1 = x0 == x1 ? 0 : x1;
1478 }
1479 else
1480 {
1481 double e;
1482 d = sqrt(-d);
1483 e = std::cbrt(d + fabs(R));
1484 if( R > 0 )
1485 e = -e;
1486 x0 = (e + Q / e) - a1 * (1./3);
1487 n = 1;
1488 }
1489 }
1490
1491 if( roots.type() == CV_32FC1 )
1492 {
1493 roots.at<float>(0) = (float)x0;
1494 roots.at<float>(1) = (float)x1;
1495 roots.at<float>(2) = (float)x2;
1496 }
1497 else
1498 {
1499 roots.at<double>(0) = x0;
1500 roots.at<double>(1) = x1;
1501 roots.at<double>(2) = x2;
1502 }
1503
1504 return n;
1505}

1507/* finds complex roots of a polynomial using Durand-Kerner method:
1508 http://en.wikipedia.org/wiki/Durand%E2%80%93Kerner_method */
1509double cv::solvePoly( InputArray _coeffs0, OutputArray _roots0, int maxIters )
1510{
1511 CV_INSTRUMENT_REGION();
1512
1513 typedef Complex<double> C;
1514
1515 double maxDiff = 0;
1516 int iter, i, j;
1517 Mat coeffs0 = _coeffs0.getMat();
1518 int ctype = _coeffs0.type();
1519 int cdepth = CV_MAT_DEPTH(ctype);
1520
1521 CV_Assert( CV_MAT_DEPTH(ctype) >= CV_32F && CV_MAT_CN(ctype) <= 2 );
1522 CV_Assert( coeffs0.rows == 1 || coeffs0.cols == 1 );
1523
1524 int n0 = coeffs0.cols + coeffs0.rows - 2, n = n0;
1525
1526 _roots0.create(n, 1, CV_MAKETYPE(cdepth, 2), -1, true, _OutputArray::DEPTH_MASK_FLT);
1527 Mat roots0 = _roots0.getMat();
1528
1529 AutoBuffer<C> buf(n*2+2);
1530 C *coeffs = buf.data(), *roots = coeffs + n + 1;
1531 Mat coeffs1(coeffs0.size(), CV_MAKETYPE(CV_64F, coeffs0.channels()), coeffs0.channels() == 2 ? coeffs : roots);
1532 coeffs0.convertTo(coeffs1, coeffs1.type());
1533 if( coeffs0.channels() == 1 )
1534 {
1535 const double* rcoeffs = (const double*)roots;
1536 for( i = 0; i <= n; i++ )
1537 coeffs[i] = C(rcoeffs[i], 0);
1538 }
1539
1540 for( ; n > 1; n-- )
1541 {
1542 if( std::abs(coeffs[n].re) + std::abs(coeffs[n].im) > DBL_EPSILON )
1543 break;
1544 }
1545
1546 // Related issue: https://github.com/opencv/opencv/issues/23644,
1547 // This the initialization scheme of "Initial approximations in Durand-Kerner's root finding method" by Guggenheimer.
1548 // https://link.springer.com/article/10.1007/BF01935059
1549 // We put the initial points equidistantly on a circle on the complex plane. This code computes the circle radius as in the paper.
1550 Mat absCoeffs(n + 1, 1, CV_64F);
1551 for( i = 0; i <= n; i++ )
1552 absCoeffs.at<double>(i) = abs(coeffs[i]);
1553
1554 int nonZeroCoeffs = 0;
1555 Mat u(n, 1, CV_64F, Scalar(0)), v(n, 1, CV_64F, Scalar(0));
1556 for( i = 0; i <= n; i++ )
1557 {
1558 double coeff = absCoeffs.at<double>(i);
1559 if( coeff > DBL_EPSILON )
1560 {
1561 if( i != n )
1562 u.at<double>(i) = 2.0 * pow(coeff / absCoeffs.at<double>(n), 1.0 / (n - i));
1563 if( i != 0 )
1564 v.at<double>(i - 1) = 0.5 * pow(absCoeffs.at<double>(0) / coeff, 1.0 / i);
1565 nonZeroCoeffs++;
1566 }
1567 }
1568 double scale = 1;
1569 if( nonZeroCoeffs > 2 )
1570 {
1571 Point maxU, minV;
1572 minMaxLoc(u, nullptr, nullptr, nullptr, &maxU);
1573 minMaxLoc(v, nullptr, nullptr, &minV);
1574 u.at<double>(maxU) = 0;
1575 v.at<double>(minV) = 0;
1576 scale = (sum(u).val[0] + sum(v).val[0]) / (2 * nonZeroCoeffs - 2);
1577 }
1578
1579 C p(scale, 0), r(cos(CV_2PI / n), sin(CV_2PI / n));
1580
1581 for( i = 0; i < n; i++ )
1582 {
1583 roots[i] = p;
1584 p = p * r;
1585 }
1586
1587 maxIters = maxIters <= 0 ? 1000 : maxIters;
1588 for( iter = 0; iter < maxIters; iter++ )
1589 {
1590 maxDiff = 0;
1591 for( i = 0; i < n; i++ )
1592 {
1593 p = roots[i];
1594 C num = coeffs[n], denom = coeffs[n];
1595 int num_same_root = 1;
1596 for( j = 0; j < n; j++ )
1597 {
1598 num = num*p + coeffs[n-j-1];
1599 if( j != i )
1600 {
1601 if ( (p - roots[j]).re != 0 || (p - roots[j]).im != 0 )
1602 denom = denom * (p - roots[j]);
1603 else
1604 num_same_root++;
1605 }
1606 }
1607 num /= denom;
1608 if( num_same_root > 1)
1609 {
1610 double old_num_re = num.re;
1611 double old_num_im = num.im;
1612 int square_root_times = num_same_root % 2 == 0 ? num_same_root / 2 : num_same_root / 2 - 1;
1613
1614 for( j = 0; j < square_root_times; j++)
1615 {
1616 num.re = old_num_re*old_num_re + old_num_im*old_num_im;
1617 num.re = sqrt(num.re);
1618 num.re += old_num_re;
1619 num.im = num.re - old_num_re;
1620 num.re /= 2;
1621 num.re = sqrt(num.re);
1622
1623 num.im /= 2;
1624 num.im = sqrt(num.im);
1625 if( old_num_re < 0 ) num.im = -num.im;
1626 }
1627
1628 if( num_same_root % 2 != 0){
1629 Mat cube_coefs(4, 1, CV_64FC1);
1630 Mat cube_roots(3, 1, CV_64FC2);
1631 cube_coefs.at<double>(3) = -(std::pow(old_num_re, 3));
1632 cube_coefs.at<double>(2) = -(15*std::pow(old_num_re, 2) + 27*std::pow(old_num_im, 2));
1633 cube_coefs.at<double>(1) = -48*old_num_re;
1634 cube_coefs.at<double>(0) = 64;
1635 solveCubic(cube_coefs, cube_roots);
1636
1637 num.re = std::cbrt(cube_roots.at<double>(0));
1638 num.im = sqrt(std::pow(num.re, 2) / 3 - old_num_re / (3*num.re));
1639 }
1640 }
1641 roots[i] = p - num;
1642 maxDiff = std::max(maxDiff, cv::abs(num));
1643 }
1644 if( maxDiff <= 0 )
1645 break;
1646 }
1647
1648 if( coeffs0.channels() == 1 )
1649 {
1650 const double verySmallEps = 1e-100;
1651 for( i = 0; i < n; i++ )
1652 if( fabs(roots[i].im) < verySmallEps )
1653 roots[i].im = 0;
1654 }
1655
1656 for( ; n < n0; n++ )
1657 roots[n+1] = roots[n];
1658
1659 Mat(roots0.size(), CV_64FC2, roots).convertTo(roots0, roots0.type());
1660 return maxDiff;
1661}
