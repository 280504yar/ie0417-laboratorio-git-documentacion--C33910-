14static float estimateBitrate(int codecId, size_t pixelNum, float fps)
15{
16 float bitrate = 0.f;
17 const float mp = pixelNum / 1000000.f;
18 if (codecId == MFX_CODEC_MPEG2)
19 {
20 bitrate = (mp * 43) * fps + 360;
21 }
22 else if (codecId == MFX_CODEC_AVC)
23 {
24 bitrate = (mp * 140 + 19) * std::pow(fps, 0.60f);
25 }
26 else if (codecId == MFX_CODEC_HEVC)
27 {
28 bitrate = (mp * 63 + 45) * std::pow(fps, 0.60f);
29 }
30 else
31 {
32 MSG(cerr << "MFX encoder Bitrate estimation FAILED" << endl);
33 }
34 DBG(cout << "MFX encoder Bitrate estimation (" << mp << " MP x " << fps << " fps): " << bitrate << endl);
35 return bitrate;
36
37}

51inline mfxU32 codecIdByFourCC(int fourcc)
52{
53 const int CC_MPG2 = FourCC('M', 'P', 'G', '2').vali32;
54 const int CC_H264 = FourCC('H', '2', '6', '4').vali32;
55 const int CC_X264 = FourCC('X', '2', '6', '4').vali32;
56 const int CC_AVC = FourCC('A', 'V', 'C', ' ').vali32;
57 const int CC_H265 = FourCC('H', '2', '6', '5').vali32;
58 const int CC_HEVC = FourCC('H', 'E', 'V', 'C').vali32;
59
60 if (fourcc == CC_X264 || fourcc == CC_H264 || fourcc == CC_AVC)
61 return MFX_CODEC_AVC;
62 else if (fourcc == CC_H265 || fourcc == CC_HEVC)
63 return MFX_CODEC_HEVC;
64 else if (fourcc == CC_MPG2)
65 return MFX_CODEC_MPEG2;
66 else
67 return (mfxU32)-1;
68}

70VideoWriter_IntelMFX::VideoWriter_IntelMFX(const String &filename, int _fourcc, double fps, Size frameSize_, bool)
71 : session(0), plugin(0), deviceHandler(0), bs(0), encoder(0), pool(0), outSurface(NULL), frameSize(frameSize_), good(false)
72{
73 mfxStatus res = MFX_ERR_NONE;
74
75 if (frameSize.width % 2 || frameSize.height % 2)
76 {
77 MSG(cerr << "MFX: Invalid frame size passed to encoder" << endl);
78 return;
79 }
80
81 if (fps <= 0)
82 {
83 MSG(cerr << "MFX: Invalid FPS passed to encoder" << endl);
84 return;
85 }
86
87 // Init device and session
88 deviceHandler = createDeviceHandler();
89 session = new MFXVideoSession_WRAP();
90 if (!deviceHandler->init(*session))
91 {
92 MSG(cerr << "MFX: Can't initialize session" << endl);
93 return;
94 }
95
96 // Load appropriate plugin
97
98 mfxU32 codecId = codecIdByFourCC(_fourcc);
99 if (codecId == (mfxU32)-1)
100 {
101 MSG(cerr << "MFX: Unsupported FourCC: " << FourCC(_fourcc) << endl);
102 return;
103 }
104 plugin = Plugin::loadEncoderPlugin(*session, codecId);
105 if (plugin && !plugin->isGood())
106 {
107 MSG(cerr << "MFX: LoadPlugin failed for codec: " << codecId << " (" << FourCC(_fourcc) << ")" << endl);
108 return;
109 }
110
111 // Init encoder
112
113 encoder = new MFXVideoENCODE(*session);
114 mfxVideoParam params;
115 memset(&params, 0, sizeof(params));
116 params.mfx.CodecId = codecId;
117 params.mfx.TargetUsage = MFX_TARGETUSAGE_BALANCED;
118 params.mfx.TargetKbps = saturate_cast<mfxU16>(estimateBitrate(codecId, frameSize.area(), (float)fps) * 300 / getBitrateDivisor()); // TODO: set in options
119 params.mfx.RateControlMethod = MFX_RATECONTROL_VBR;
120 params.mfx.FrameInfo.FrameRateExtN = cvRound(fps * 1000);
121 params.mfx.FrameInfo.FrameRateExtD = 1000;
122 params.mfx.FrameInfo.FourCC = MFX_FOURCC_NV12;
123 params.mfx.FrameInfo.ChromaFormat = MFX_CHROMAFORMAT_YUV420;
124 params.mfx.FrameInfo.PicStruct = MFX_PICSTRUCT_PROGRESSIVE;
125 params.mfx.FrameInfo.CropX = 0;
126 params.mfx.FrameInfo.CropY = 0;
127 params.mfx.FrameInfo.CropW = (mfxU16)frameSize.width;
128 params.mfx.FrameInfo.CropH = (mfxU16)frameSize.height;
129 params.mfx.FrameInfo.Width = (mfxU16)alignSize(frameSize.width, 32);
130 params.mfx.FrameInfo.Height = (mfxU16)alignSize(frameSize.height, 32);
131 params.IOPattern = MFX_IOPATTERN_IN_SYSTEM_MEMORY;
132 res = encoder->Query(&params, &params);
133 DBG(cout << "MFX Query: " << res << endl << params.mfx << params.mfx.FrameInfo);
134 if (res < MFX_ERR_NONE)
135 {
136 MSG(cerr << "MFX: Query failed: " << res << endl);
137 return;
138 }
139
140 // Init surface pool
141 pool = SurfacePool::create(encoder, params);
142 if (!pool)
143 {
144 MSG(cerr << "MFX: Failed to create surface pool" << endl);
145 return;
146 }
147
148 // Init encoder
149 res = encoder->Init(&params);
150 DBG(cout << "MFX encoder Init: " << res << endl << params.mfx.FrameInfo);
151 if (res < MFX_ERR_NONE)
152 {
153 MSG(cerr << "MFX: Failed to init encoder: " << res << endl);
154 return;
155 }
156
157 // Open output bitstream
158 {
159 mfxVideoParam par;
160 memset(&par, 0, sizeof(par));
161 res = encoder->GetVideoParam(&par);
162 DBG(cout << "MFX GetVideoParam: " << res << endl << "requested " << par.mfx.BufferSizeInKB << " kB" << endl);
163 CV_Assert(res >= MFX_ERR_NONE);
164 bs = new WriteBitstream(filename.c_str(), par.mfx.BufferSizeInKB * 1024 * 2);
165 if (!bs->isOpened())
166 {
167 MSG(cerr << "MFX: Failed to open output file: " << filename << endl);
168 return;
169 }
170 }
171
172 good = true;
173}

214bool VideoWriter_IntelMFX::write_one(cv::InputArray bgr)
215{
216 mfxStatus res;
217 mfxFrameSurface1 *workSurface = 0;
218 mfxSyncPoint sync;
219
220 if (!bgr.empty() && (bgr.dims() != 2 || bgr.type() != CV_8UC3 || bgr.size() != frameSize))
221 {
222 MSG(cerr << "MFX: invalid frame passed to encoder: "
223 << "dims/depth/cn=" << bgr.dims() << "/" << bgr.depth() << "/" << bgr.channels()
224 << ", size=" << bgr.size() << endl);
225 return false;
226
227 }
228 if (!bgr.empty())
229 {
230 workSurface = pool->getFreeSurface();
231 if (!workSurface)
232 {
233 // not enough surfaces
234 MSG(cerr << "MFX: Failed to get free surface" << endl);
235 return false;
236 }
237 Mat src = bgr.getMat();
238 hal::cvtBGRtoTwoPlaneYUV(src.data, src.step,
239 workSurface->Data.Y, workSurface->Data.UV, workSurface->Data.Pitch,
240 workSurface->Info.CropW, workSurface->Info.CropH,
241 3, false, 1);
242 }
243
244 while (true)
245 {
246 outSurface = 0;
247 DBG(cout << "Calling with surface: " << workSurface << endl);
248 res = encoder->EncodeFrameAsync(NULL, workSurface, &bs->stream, &sync);
249 if (res == MFX_ERR_NONE)
250 {
251 res = session->SyncOperation(sync, getWriterTimeoutMS()); // TODO: provide interface to modify timeout
252 if (res == MFX_ERR_NONE)
253 {
254 // ready to write
255 if (!bs->write())
256 {
257 MSG(cerr << "MFX: Failed to write bitstream" << endl);
258 return false;
259 }
260 else
261 {
262 DBG(cout << "Write bitstream" << endl);
263 return true;
264 }
265 }
266 else
267 {
268 MSG(cerr << "MFX: Sync error: " << res << endl);
269 return false;
270 }
271 }
272 else if (res == MFX_ERR_MORE_DATA)
273 {
274 DBG(cout << "ERR_MORE_DATA" << endl);
275 return false;
276 }
277 else if (res == MFX_WRN_DEVICE_BUSY)
278 {
279 DBG(cout << "Waiting for device" << endl);
280 sleep_ms(1000);
281 continue;
282 }
283 else
284 {
285 MSG(cerr << "MFX: Bad status: " << res << endl);
286 return false;
287 }
288 }
