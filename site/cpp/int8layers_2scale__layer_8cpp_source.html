20 ScaleLayerInt8Impl(const LayerParams& params)

24 axis = params.get<int>("axis", 1);
25 hasWeights = false;

28 output_zp = params.get<int>("zeropoints");
29
30 DictValue inpSc = params.get("input_scales");
31 DictValue inpZp = params.get("input_zeropoints");
32
33 for (int i = 0; i < inpSc.size(); i++)
34 {
35 inp_sc.push_back(inpSc.get<float>(i));
36 inp_zp.push_back(inpZp.get<int>(i));
37 }
38 }

49 virtual void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays) CV_OVERRIDE
50 {
51 std::vector<Mat> inputs;
52 inputs_arr.getMatVector(inputs);
53 hasWeights = blobs.size() == 2 || (blobs.size() <= 1 && !hasBias);
54 CV_Assert((inputs.size() == 2 && blobs.empty()) || blobs.size() == (int)hasWeights + (int)hasBias);
55
56 if (!blobs.empty())
57 {
58 Mat w = hasWeights ? blobs[0] : Mat::ones(blobs[0].size(), CV_32F);
59 Mat b = hasBias ? blobs.back() : Mat::zeros(blobs.back().size(), CV_32F);
60
61 w = w.reshape(1, 1);
62 b = b.reshape(1, 1);
63
64 w.convertTo(weights, CV_32F, inp_sc[0]/output_sc);
65 addWeighted(b, 1.0/output_sc, weights, -inp_zp[0], output_zp, bias, CV_32F);
66 }
67 else
68 {
69 // initialized during forward()
70 weights = Mat(); bias = Mat();
71 }

90 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
91 {
92 CV_TRACE_FUNCTION();
93 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
94
95 std::vector<Mat> inputs, outputs;
96 inputs_arr.getMatVector(inputs);
97 outputs_arr.getMatVector(outputs);
98
99 Mat &inpBlob = inputs[0];
100 Mat &outBlob = outputs[0];
101
102 if (blobs.empty())
103 {
104 CV_Assert(inp_sc.size() == 2 && inp_zp.size() == 2);
105 Mat inp_dequantized, w, b;
106 inputs[1].reshape(1, 1).convertTo(inp_dequantized, CV_32F, inp_sc[1], -(inp_sc[1]*inp_zp[1]));
107 w = hasWeights ? inp_dequantized : Mat::ones(inp_dequantized.size(), CV_32F);
108 b = hasBias ? inp_dequantized : Mat::zeros(inp_dequantized.size(), CV_32F);
109
110 w.convertTo(weights, CV_32F, inp_sc[0]/output_sc);
111 addWeighted(b, 1.0/output_sc, weights, -inp_zp[0], output_zp, bias, CV_32F);
112 }
113
114 MatShape inpShape = shape(inpBlob);
115 const int numWeights = weights.total();
116 CV_Assert(numWeights != 0);
117 CV_CheckEQ(weights.total(), bias.total(), "Incompatible weights/bias blobs");
118
119 int endAxis;
120 for (endAxis = axis + 1; endAxis <= inpBlob.dims; ++endAxis)
121 {
122 if (total(inpShape, axis, endAxis) == numWeights)
123 break;
124 }
125 CV_Assert(total(inpShape, axis, endAxis) == numWeights);
126 CV_CheckTypeEQ(inpBlob.type(), CV_8SC1, ""); CV_CheckTypeEQ(outBlob.type(), CV_8SC1, "");
127
128 int numSlices = total(inpShape, 0, axis);
129 int8_t* inpData = (int8_t*)inpBlob.data;
130 int8_t* outData = (int8_t*)outBlob.data;
131
132 if (endAxis != inpBlob.dims)
133 {
134 float* weightsData = (float*)weights.data;
135 float* biasesData = (float*)bias.data;
136 int spatialSize = total(inpShape, endAxis); // spatialSize != 1
137 for (int i = 0; i < numSlices; ++i)
138 {
139 for (int j = 0; j < numWeights; ++j)
140 {
141 float w = weightsData[j];
142 float b = biasesData[j];
143 Mat inpSlice(1, spatialSize, CV_8S, inpData);
144 Mat outSlice(1, spatialSize, CV_8S, outData);
145 inpSlice.convertTo(outSlice, CV_8S, w, b);
146 inpData += spatialSize;
147 outData += spatialSize;
148 }
149 }
150 }
151 else
152 {
153 for (int i = 0; i < numSlices; ++i)
154 {
155 Mat inpSlice(1, numWeights, CV_8S, inpData);
156 Mat outSlice(1, numWeights, CV_8S, outData);
157
158 multiply(inpSlice, weights, outSlice, 1.0, CV_8S);
159 add(outSlice, bias, outSlice, Mat(), CV_8S);
160
161 inpData += numWeights;
162 outData += numWeights;
163 }
164 }
165 }

191#ifdef HAVE_DNN_NGRAPH
192 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> >& inputs, const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
193 {
194 std::vector<ov::Output<ov::Node>> ieInpNodes(nodes.size());
195 for (int i = 0; i < nodes.size(); ++i) {
196 ieInpNodes[i] = nodes[i].dynamicCast<InfEngineNgraphNode>()->node;
197 }
198
199 ieInpNodes[0] = ngraphDequantize(ieInpNodes[0], inp_sc[0], inp_zp[0]);
200
201 CV_Assert(!blobs.empty() || ieInpNodes.size() == 1 + (int)hasWeights + (int)hasBias);
202
203 ov::Output<ov::Node> weights, bias;
204 if (blobs.empty()) {
205 if (hasWeights)
206 weights = ieInpNodes[1];
207 if (hasBias)
208 bias = ieInpNodes[1 + (int)hasWeights];
209 } else {
210 std::vector<size_t> shape = ieInpNodes[0].get_shape();
211 int cAxis = normalize_axis(axis, shape.size());
212
213 size_t numWeights = blobs[0].total();
214 for (int i = 0; i < cAxis; ++i) {
215 shape[i] = 1;
216 }
217 for (int i = cAxis; i < shape.size(); ++i) {
218 if (numWeights == 1) {
219 shape[i] = 1;
220 }
221 numWeights = std::max(numWeights / shape[i], (size_t)1);
222 }
223
224 if (hasWeights)
225 weights = std::make_shared<ov::op::v0::Constant>(ov::element::f32, shape, blobs[0].data);
226 if (hasBias)
227 bias = std::make_shared<ov::op::v0::Constant>(ov::element::f32, shape, blobs[(int)hasWeights].data);
228 }
229
230 ov::Output<ov::Node> res = ieInpNodes[0];
231 if (hasWeights) {
232 res = std::make_shared<ov::op::v1::Multiply>(res, weights);
233 }
234 if (hasBias) {
235 res = std::make_shared<ov::op::v1::Add>(res, bias);
236 }
237
238 res = ngraphQuantize(res, output_sc, output_zp);
239
240 return new InfEngineNgraphNode(res);
241 }
242#endif // HAVE_DNN_NGRAPH
