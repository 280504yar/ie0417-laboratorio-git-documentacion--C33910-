149template <class P>
150class CV_EXPORTS_TEMPLATE RotationWarperBase : public RotationWarper
151{
152public:
153 Point2f warpPoint(const Point2f &pt, InputArray K, InputArray R) CV_OVERRIDE;
154
155 Point2f warpPointBackward(const Point2f &pt, InputArray K, InputArray R) CV_OVERRIDE;
156
157 Rect buildMaps(Size src_size, InputArray K, InputArray R, OutputArray xmap, OutputArray ymap) CV_OVERRIDE;
158
159 Point warp(InputArray src, InputArray K, InputArray R, int interp_mode, int border_mode,
160 OutputArray dst) CV_OVERRIDE;
161
162 void warpBackward(InputArray src, InputArray K, InputArray R, int interp_mode, int border_mode,
163 Size dst_size, OutputArray dst) CV_OVERRIDE;
164
165 Rect warpRoi(Size src_size, InputArray K, InputArray R) CV_OVERRIDE;
166
167 float getScale() const CV_OVERRIDE{ return projector_.scale; }
168 void setScale(float val) CV_OVERRIDE { projector_.scale = val; }
169
170protected:
171
172 // Detects ROI of the destination image. It's correct for any projection.
173 virtual void detectResultRoi(Size src_size, Point &dst_tl, Point &dst_br);
174
175 // Detects ROI of the destination image by walking over image border.
176 // Correctness for any projection isn't guaranteed.
177 void detectResultRoiByBorder(Size src_size, Point &dst_tl, Point &dst_br);
178
179 P projector_;
180};
