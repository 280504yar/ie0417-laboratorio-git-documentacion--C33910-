90 static void run(const Mat& src, const Mat& lut, Mat& dst, int nstripes)
91 {
92 Activation p;
93
94 p.src = &src;
95 p.lut = &lut;
96 p.dst = &dst;
97 p.nstripes = nstripes;
98
99 parallel_for_(Range(0, nstripes), p, nstripes);
100 }

102 void operator()(const Range &r) const CV_OVERRIDE
103 {
104 const int8_t* table = lut->ptr<int8_t>();
105 int nsamples = 1, outCn = 1;
106 size_t planeSize = 1;
107
108 if (src->dims > 1)
109 {
110 nsamples = src->size[0];
111 outCn = src->size[1];
112 }
113 else
114 outCn = src->size[0];
115
116 for (int i = 2; i < src->dims; ++i)
117 planeSize *= src->size[i];
118
119 size_t stripeSize = (planeSize + nstripes - 1)/nstripes;
120 size_t stripeStart = r.start*stripeSize;
121 size_t stripeEnd = std::min(r.end*stripeSize, planeSize);
122 int len = (int)(stripeEnd - stripeStart);
123
124 for( int i = 0; i < nsamples; i++ )
125 {
126 const int8_t* srcptr = src->ptr<int8_t>(i) + stripeStart;
127 int8_t* dstptr = dst->ptr<int8_t>(i) + stripeStart;
128 for( int cn = 0; cn < outCn; cn++, srcptr += planeSize, dstptr += planeSize )
129 {
130 int i = 0;
131#if CV_SIMD128
132 for( ; i <= len - 16; i += 16 )
133 {
134 v_int8x16 out(table[srcptr[i] + 128], table[srcptr[i+1] + 128], table[srcptr[i+2] + 128], table[srcptr[i+3] + 128],
135 table[srcptr[i+4] + 128], table[srcptr[i+5] + 128], table[srcptr[i+6] + 128], table[srcptr[i+7] + 128],
136 table[srcptr[i+8] + 128], table[srcptr[i+9] + 128], table[srcptr[i+10] + 128], table[srcptr[i+11] + 128],
137 table[srcptr[i+12] + 128], table[srcptr[i+13] + 128], table[srcptr[i+14] + 128], table[srcptr[i+15] + 128]);
138 v_store(dstptr + i, out);
139 }
140#endif
141 for( ; i < len; i++ )
142 {
143 dstptr[i] = table[srcptr[i] + 128];
144 }
145 }
146 }
147 }

150 virtual Ptr<BackendNode> initTimVX(void* timVXInfo_,
151 const std::vector<Ptr<BackendWrapper> > &inputsWrapper,
152 const std::vector<Ptr<BackendWrapper> > &outputsWrapper,
153 bool isLast) CV_OVERRIDE
154 {
155#ifdef HAVE_TIMVX
156 // tvGraph Initialization.
157 auto timVxInfo = reinterpret_cast<TimVXInfo *>(timVXInfo_);
158 CV_Assert(timVxInfo);
159 Ptr<TimVXGraph> tvGraph = timVxInfo->getGraph();
160 CV_Assert(tvGraph);
161 Ptr<tim::vx::Graph> graph = tvGraph->graph;
162
163 std::vector<int> inputsIndex, outputsIndex;
164 int input_index, output_index;
165 CV_Assert(inputsWrapper.size() == 1);
166
167 // input Tensor
168 Ptr<TimVXBackendWrapper> inputWrapper = inputsWrapper[0].dynamicCast<TimVXBackendWrapper>();
169
170 if (inputWrapper->isTensor())
171 {
172 input_index = tvGraph->getTensorIndex(inputWrapper->getTensor());
173 if(input_index == -1)
174 {
175 // Copy To New inputWrapper
176 Mat tmp = inputWrapper->getMat();
177 inputWrapper = Ptr<TimVXBackendWrapper>(new TimVXBackendWrapper(tmp));
178 }
179 }
180
181 if (!inputWrapper->isTensor())
182 {
183 Ptr<tim::vx::Quantization> tvInputQuant = Ptr<tim::vx::Quantization>(
184 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, input_sc, input_zp));
185 inputWrapper->createTensor(graph, tim::vx::TensorAttribute::INPUT, tvInputQuant);
186 input_index = tvGraph->addWrapper(inputWrapper);
187 }
188
189 inputsIndex.push_back(input_index);
190
191 // output tensor
192 CV_Assert(outputsWrapper.size() == 1);
193 Ptr<TimVXBackendWrapper> outputWrapper = outputsWrapper[0].dynamicCast<TimVXBackendWrapper>();
194 Ptr<tim::vx::Quantization> outputQuant = Ptr<tim::vx::Quantization>(
195 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, output_sc, output_zp));
196
197 Ptr<tim::vx::Tensor> outputTensor;
198
199 if (isLast)
200 {
201 auto shapeType = getShapeTypeFromMat(outputWrapper->getMat());
202
203 // For Graph Output tensor, we need to set tensor shape before createTensor().
204 outputWrapper->setTensorShape(shapeType);
205 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::OUTPUT, outputQuant);
206 }
207 else
208 {
209 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::TRANSIENT, outputQuant);
210 }
211 output_index = tvGraph->addWrapper(outputWrapper);
212 outputsIndex.push_back(output_index);
213
214 std::shared_ptr<tim::vx::Operation> tvAct;
215
216 switch(tvActType) {
217 case tvActReLU:
218 {
219 if (slope != 0.f)
220 tvAct = graph->CreateOperation<tim::vx::ops::LeakyRelu>(slope);
221 else
222 tvAct = graph->CreateOperation<tim::vx::ops::Relu>();
223 break;
224 }
225 case tvActReLU6:
226 tvAct = graph->CreateOperation<tim::vx::ops::Relu6>();
227 break;
228 case tvActTanH:
229 tvAct = graph->CreateOperation<tim::vx::ops::Tanh>();
230 break;
231 case tvActSwish:
232 tvAct = graph->CreateOperation<tim::vx::ops::Swish>();
233 break;
234 case tvActMish:
235 tvAct = graph->CreateOperation<tim::vx::ops::Mish>();
236 break;
237 case tvActSigmoid:
238 tvAct = graph->CreateOperation<tim::vx::ops::Sigmoid>();
239 break;
240 case tvActELU:
241 tvAct = graph->CreateOperation<tim::vx::ops::Elu>();
242 break;
243 default:
244 // TODO! check the default function.
245 tvAct = graph->CreateOperation<tim::vx::ops::Relu>();
246 break;
247 }
248
249 Ptr<TimVXBackendNode> tvBackendNode = new TimVXBackendNode(tvGraph, tvAct, inputsIndex, outputsIndex);
250
251 return tvBackendNode;
252#endif // HAVE_TIMVX
253 return Ptr<BackendNode>();
254 }

257 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> > &inputs,
258 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
259 {
260 auto input = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
261
262 input = ngraphDequantize(input, input_sc, input_zp);
263
264 ov::Output<ov::Node> res;
265 if (type == "ReLU6Int8") {
266 res = std::make_shared<ov::op::v0::Clamp>(input, 0.0f, 6.0f);
267 } else if (type == "ReLUInt8") {
268 if (slope) {
269 auto param = std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{1}, &slope);
270 res = std::make_shared<ov::op::v0::PRelu>(input, param);
271 } else {
272 res = std::make_shared<ov::op::v0::Relu>(input);
273 }
274 } else if (type == "ELUInt8") {
275 res = std::make_shared<ov::op::v0::Elu>(input, 1.0f);
276 } else if (type == "MishInt8") {
277 res = std::make_shared<ov::op::v4::Mish>(input);
278 } else if (type == "HardSwishInt8") {
279 res = std::make_shared<ov::op::v4::HSwish>(input);
280 } else if (type == "AbsValInt8") {
281 res = std::make_shared<ov::op::v0::Abs>(input);
282 } else if (type == "SigmoidInt8") {
283 res = std::make_shared<ov::op::v0::Sigmoid>(input);
284 } else {
285 CV_Error(Error::StsNotImplemented, type + " activation with OpenVINO");
286 }
287
288 res = ngraphQuantize(res, output_sc, output_zp);
289
290 return new InfEngineNgraphNode(res);
291 }

294 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
295 {
296 CV_TRACE_FUNCTION();
297
298 std::vector<Mat> inputs, outputs;
299 inputs_arr.getMatVector(inputs);
300 outputs_arr.getMatVector(outputs);
301
302 for (size_t i = 0; i < inputs.size(); i++)
303 {
304 const Mat &src = inputs[i];
305 if (!activationLUT.empty())
306 {
307 const int nstripes = getNumThreads();
308 Mat &dst = outputs[i];
309 CV_Assert(src.size == dst.size && src.type() == dst.type() &&
310 src.isContinuous() && dst.isContinuous() &&
311 (src.type() == CV_8S || src.type() == CV_8U));
312
313 if (src.type() == CV_8S) {
314 Activation::run(src, activationLUT, dst, nstripes);
315 } else {
316 const uint8_t* table = activationLUT.ptr<uint8_t>();
317 const uint8_t* srcptr = src.ptr<uint8_t>();
318 uint8_t* dstptr = dst.ptr<uint8_t>();
319 int total = (int)src.total();
320 parallel_for_(Range(0, nstripes), [&](const Range& r) {
321 int start = (int)((size_t)r.start * total / nstripes);
322 int end = (int)((size_t)r.end * total / nstripes);
323 for (int j = start; j < end; j++)
324 dstptr[j] = table[srcptr[j]];
325 }, nstripes);
326 }
327 }
328 else
329 {
330 src.copyTo(outputs[i]);
331 }
332 }
333 }
