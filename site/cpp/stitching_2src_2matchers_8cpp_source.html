65struct MatchPairsBody : ParallelLoopBody
66{
67 MatchPairsBody(FeaturesMatcher &_matcher, const std::vector<ImageFeatures> &_features,
68 std::vector<MatchesInfo> &_pairwise_matches, std::vector<std::pair<int,int> > &_near_pairs)
69 : matcher(_matcher), features(_features),
70 pairwise_matches(_pairwise_matches), near_pairs(_near_pairs) {}
71
72 void operator ()(const Range &r) const CV_OVERRIDE
73 {
74 cv::RNG rng = cv::theRNG(); // save entry rng state
75 const int num_images = static_cast<int>(features.size());
76 for (int i = r.start; i < r.end; ++i)
77 {
78 cv::theRNG() = cv::RNG(rng.state + i); // force "stable" RNG seed for each processed pair
79
80 int from = near_pairs[i].first;
81 int to = near_pairs[i].second;
82 int pair_idx = from*num_images + to;
83
84 matcher(features[from], features[to], pairwise_matches[pair_idx]);
85 pairwise_matches[pair_idx].src_img_idx = from;
86 pairwise_matches[pair_idx].dst_img_idx = to;
87
88 size_t dual_pair_idx = to*num_images + from;
89
90 pairwise_matches[dual_pair_idx] = pairwise_matches[pair_idx];
91 pairwise_matches[dual_pair_idx].src_img_idx = to;
92 pairwise_matches[dual_pair_idx].dst_img_idx = from;
93
94 if (!pairwise_matches[pair_idx].H.empty())
95 pairwise_matches[dual_pair_idx].H = pairwise_matches[pair_idx].H.inv();
96
97 for (size_t j = 0; j < pairwise_matches[dual_pair_idx].matches.size(); ++j)
98 std::swap(pairwise_matches[dual_pair_idx].matches[j].queryIdx,
99 pairwise_matches[dual_pair_idx].matches[j].trainIdx);
100 LOG(".");
101 }
102 }
103
104 FeaturesMatcher &matcher;
105 const std::vector<ImageFeatures> &features;
106 std::vector<MatchesInfo> &pairwise_matches;
107 std::vector<std::pair<int,int> > &near_pairs;
108
109private:
110 void operator =(const MatchPairsBody&);
111};

149void CpuMatcher::match(const ImageFeatures &features1, const ImageFeatures &features2, MatchesInfo& matches_info)
150{
151 CV_INSTRUMENT_REGION();
152
153 CV_Assert(features1.descriptors.type() == features2.descriptors.type());
154 CV_Assert(features2.descriptors.depth() == CV_8U || features2.descriptors.depth() == CV_32F);
155
156 matches_info.matches.clear();
157
158 Ptr<cv::DescriptorMatcher> matcher;
159#if 0 // TODO check this
160 if (ocl::isOpenCLActivated())
161 {
162 matcher = makePtr<BFMatcher>((int)NORM_L2);
163 }
164 else
165#endif
166 {
167 Ptr<flann::IndexParams> indexParams = makePtr<flann::KDTreeIndexParams>();
168 Ptr<flann::SearchParams> searchParams = makePtr<flann::SearchParams>();
169
170 if (features2.descriptors.depth() == CV_8U)
171 {
172 indexParams->setAlgorithm(cvflann::FLANN_INDEX_LSH);
173 searchParams->setAlgorithm(cvflann::FLANN_INDEX_LSH);
174 }
175
176 matcher = makePtr<FlannBasedMatcher>(indexParams, searchParams);
177 }
178 std::vector< std::vector<DMatch> > pair_matches;
179 MatchesSet matches;
180
181 // Find 1->2 matches
182 matcher->knnMatch(features1.descriptors, features2.descriptors, pair_matches, 2);
183 for (size_t i = 0; i < pair_matches.size(); ++i)
184 {
185 if (pair_matches[i].size() < 2)
186 continue;
187 const DMatch& m0 = pair_matches[i][0];
188 const DMatch& m1 = pair_matches[i][1];
189 if (m0.distance < (1.f - match_conf_) * m1.distance)
190 {
191 matches_info.matches.push_back(m0);
192 matches.insert(std::make_pair(m0.queryIdx, m0.trainIdx));
193 }
194 }
195 LOG("\n1->2 matches: " << matches_info.matches.size() << endl);
196
197 // Find 2->1 matches
198 pair_matches.clear();
199 matcher->knnMatch(features2.descriptors, features1.descriptors, pair_matches, 2);
200 for (size_t i = 0; i < pair_matches.size(); ++i)
201 {
202 if (pair_matches[i].size() < 2)
203 continue;
204 const DMatch& m0 = pair_matches[i][0];
205 const DMatch& m1 = pair_matches[i][1];
206 if (m0.distance < (1.f - match_conf_) * m1.distance)
207 if (matches.find(std::make_pair(m0.trainIdx, m0.queryIdx)) == matches.end())
208 matches_info.matches.push_back(DMatch(m0.trainIdx, m0.queryIdx, m0.distance));
209 }
210 LOG("1->2 & 2->1 matches: " << matches_info.matches.size() << endl);
211}

338void FeaturesMatcher::match(const std::vector<ImageFeatures> &features, std::vector<MatchesInfo> &pairwise_matches,
339 const UMat &mask)
340{
341 const int num_images = static_cast<int>(features.size());
342
343 CV_Assert(mask.empty() || (mask.type() == CV_8U && mask.cols == num_images && mask.rows));
344 Mat_<uchar> mask_(mask.getMat(ACCESS_READ));
345 if (mask_.empty())
346 mask_ = Mat::ones(num_images, num_images, CV_8U);
347
348 std::vector<std::pair<int,int> > near_pairs;
349 for (int i = 0; i < num_images - 1; ++i)
350 for (int j = i + 1; j < num_images; ++j)
351 if (features[i].keypoints.size() > 0 && features[j].keypoints.size() > 0 && mask_(i, j))
352 near_pairs.push_back(std::make_pair(i, j));
353
354 pairwise_matches.clear(); // clear history values
355 pairwise_matches.resize(num_images * num_images);
356 MatchPairsBody body(*this, features, pairwise_matches, near_pairs);
357
358 if (is_thread_safe_)
359 parallel_for_(Range(0, static_cast<int>(near_pairs.size())), body);
360 else
361 body(Range(0, static_cast<int>(near_pairs.size())));
362 LOGLN_CHAT("");
363}

397void BestOf2NearestMatcher::match(const ImageFeatures &features1, const ImageFeatures &features2,
398 MatchesInfo &matches_info)
399{
400 CV_INSTRUMENT_REGION();
401
402 (*impl_)(features1, features2, matches_info);
403
404 // Check if it makes sense to find homography
405 if (matches_info.matches.size() < static_cast<size_t>(num_matches_thresh1_))
406 return;
407
408 // Construct point-point correspondences for homography estimation
409 Mat src_points(1, static_cast<int>(matches_info.matches.size()), CV_32FC2);
410 Mat dst_points(1, static_cast<int>(matches_info.matches.size()), CV_32FC2);
411 for (size_t i = 0; i < matches_info.matches.size(); ++i)
412 {
413 const DMatch& m = matches_info.matches[i];
414
415 Point2f p = features1.keypoints[m.queryIdx].pt;
416 p.x -= features1.img_size.width * 0.5f;
417 p.y -= features1.img_size.height * 0.5f;
418 src_points.at<Point2f>(0, static_cast<int>(i)) = p;
419
420 p = features2.keypoints[m.trainIdx].pt;
421 p.x -= features2.img_size.width * 0.5f;
422 p.y -= features2.img_size.height * 0.5f;
423 dst_points.at<Point2f>(0, static_cast<int>(i)) = p;
424 }
425
426 // Find pair-wise motion
427 matches_info.H = findHomography(src_points, dst_points, matches_info.inliers_mask, RANSAC);
428 if (matches_info.H.empty() || std::abs(determinant(matches_info.H)) < std::numeric_limits<double>::epsilon())
429 return;
430
431 // Find number of inliers
432 matches_info.num_inliers = 0;
433 for (size_t i = 0; i < matches_info.inliers_mask.size(); ++i)
434 if (matches_info.inliers_mask[i])
435 matches_info.num_inliers++;
436
437 // These coeffs are from paper M. Brown and D. Lowe. "Automatic Panoramic Image Stitching
438 // using Invariant Features"
439 matches_info.confidence = matches_info.num_inliers / (8 + 0.3 * matches_info.matches.size());
440
441 // Set zero confidence to remove matches between too close images, as they don't provide
442 // additional information anyway.
443 matches_info.confidence = matches_info.confidence > matches_confidence_thresh_ ? 0. : matches_info.confidence;
444
445 // Check if we should try to refine motion
446 if (matches_info.num_inliers < num_matches_thresh2_)
447 return;
448
449 // Construct point-point correspondences for inliers only
450 src_points.create(1, matches_info.num_inliers, CV_32FC2);
451 dst_points.create(1, matches_info.num_inliers, CV_32FC2);
452 int inlier_idx = 0;
453 for (size_t i = 0; i < matches_info.matches.size(); ++i)
454 {
455 if (!matches_info.inliers_mask[i])
456 continue;
457
458 const DMatch& m = matches_info.matches[i];
459
460 Point2f p = features1.keypoints[m.queryIdx].pt;
461 p.x -= features1.img_size.width * 0.5f;
462 p.y -= features1.img_size.height * 0.5f;
463 src_points.at<Point2f>(0, inlier_idx) = p;
464
465 p = features2.keypoints[m.trainIdx].pt;
466 p.x -= features2.img_size.width * 0.5f;
467 p.y -= features2.img_size.height * 0.5f;
468 dst_points.at<Point2f>(0, inlier_idx) = p;
469
470 inlier_idx++;
471 }
472
473 // Rerun motion estimation on inliers only
474 matches_info.H = findHomography(src_points, dst_points, RANSAC);
475}

516void AffineBestOf2NearestMatcher::match(const ImageFeatures &features1, const ImageFeatures &features2,
517 MatchesInfo &matches_info)
518{
519 (*impl_)(features1, features2, matches_info);
520
521 // Check if it makes sense to find transform
522 if (matches_info.matches.size() < static_cast<size_t>(num_matches_thresh1_))
523 return;
524
525 // Construct point-point correspondences for transform estimation
526 Mat src_points(1, static_cast<int>(matches_info.matches.size()), CV_32FC2);
527 Mat dst_points(1, static_cast<int>(matches_info.matches.size()), CV_32FC2);
528 for (size_t i = 0; i < matches_info.matches.size(); ++i)
529 {
530 const cv::DMatch &m = matches_info.matches[i];
531 src_points.at<Point2f>(0, static_cast<int>(i)) = features1.keypoints[m.queryIdx].pt;
532 dst_points.at<Point2f>(0, static_cast<int>(i)) = features2.keypoints[m.trainIdx].pt;
533 }
534
535 // Find pair-wise motion
536 if (full_affine_)
537 matches_info.H = estimateAffine2D(src_points, dst_points, matches_info.inliers_mask);
538 else
539 matches_info.H = estimateAffinePartial2D(src_points, dst_points, matches_info.inliers_mask);
540
541 if (matches_info.H.empty()) {
542 // could not find transformation
543 matches_info.confidence = 0;
544 matches_info.num_inliers = 0;
545 return;
546 }
547
548 // Find number of inliers
549 matches_info.num_inliers = 0;
550 for (size_t i = 0; i < matches_info.inliers_mask.size(); ++i)
551 if (matches_info.inliers_mask[i])
552 matches_info.num_inliers++;
553
554 // These coeffs are from paper M. Brown and D. Lowe. "Automatic Panoramic
555 // Image Stitching using Invariant Features"
556 matches_info.confidence =
557 matches_info.num_inliers / (8 + 0.3 * matches_info.matches.size());
558
559 /* should we remove matches between too close images? */
560 // matches_info.confidence = matches_info.confidence > 3. ? 0. : matches_info.confidence;
561
562 // extend H to represent linear transformation in homogeneous coordinates
563 matches_info.H.push_back(Mat::zeros(1, 3, CV_64F));
564 matches_info.H.at<double>(2, 2) = 1;
565}

586void LightGlueFeaturesMatcher::match(const ImageFeatures &features1, const ImageFeatures &features2,
587 MatchesInfo &matches_info)
588{
589 matches_info.src_img_idx = features1.img_idx;
590 matches_info.dst_img_idx = features2.img_idx;
591 matches_info.confidence = 0;
592
593 // Guard empty keypoints
594 if (features1.keypoints.empty() || features2.keypoints.empty())
595 return;
596
597 const int N = static_cast<int>(features1.keypoints.size());
598 const int M = static_cast<int>(features2.keypoints.size());
599
600 // Build Nx2 keypoint coordinate matrices
601 Mat kpts1Mat(N, 2, CV_32F);
602 Mat kpts2Mat(M, 2, CV_32F);
603 for (int i = 0; i < N; i++)
604 {
605 kpts1Mat.at<float>(i, 0) = features1.keypoints[i].pt.x;
606 kpts1Mat.at<float>(i, 1) = features1.keypoints[i].pt.y;
607 }
608 for (int i = 0; i < M; i++)
609 {
610 kpts2Mat.at<float>(i, 0) = features2.keypoints[i].pt.x;
611 kpts2Mat.at<float>(i, 1) = features2.keypoints[i].pt.y;
612 }
613
614 // Set pair context for LightGlue spatial reasoning
615 lgMatcher_->setPairInfo(kpts1Mat, kpts2Mat, features1.img_size, features2.img_size);
616
617 // Run LightGlue matching
618 Mat desc1 = features1.descriptors.getMat(ACCESS_READ);
619 Mat desc2 = features2.descriptors.getMat(ACCESS_READ);
620 std::vector<DMatch> matches;
621 lgMatcher_->match(desc1, desc2, matches);
622
623 // Filter by score threshold
624 if (lg_score_thresh_ > 0.0f)
625 {
626 float maxDist = 1.0f - lg_score_thresh_;
627 std::vector<DMatch> filtered;
628 filtered.reserve(matches.size());
629 for (const auto& m : matches)
630 {
631 if (m.distance <= maxDist)
632 filtered.push_back(m);
633 }
634 matches.swap(filtered);
635 }
636
637 // Store matches before homography estimation
638 matches_info.matches = matches;
639
640 // Guard: need at least 4 matches for findHomography
641 if (matches.size() < 4)
642 return;
643
644 // Estimate homography with centered coordinates
645 Mat src_points(1, static_cast<int>(matches.size()), CV_32FC2);
646 Mat dst_points(1, static_cast<int>(matches.size()), CV_32FC2);
647 for (size_t i = 0; i < matches.size(); ++i)
648 {
649 const DMatch& m = matches[i];
650
651 Point2f p = features1.keypoints[m.queryIdx].pt;
652 p.x -= features1.img_size.width * 0.5f;
653 p.y -= features1.img_size.height * 0.5f;
654 src_points.at<Point2f>(0, static_cast<int>(i)) = p;
655
656 p = features2.keypoints[m.trainIdx].pt;
657 p.x -= features2.img_size.width * 0.5f;
658 p.y -= features2.img_size.height * 0.5f;
659 dst_points.at<Point2f>(0, static_cast<int>(i)) = p;
660 }
661
662 matches_info.H = findHomography(src_points, dst_points, matches_info.inliers_mask, RANSAC);
663 if (matches_info.H.empty() ||
664 std::abs(determinant(matches_info.H)) < std::numeric_limits<double>::epsilon())
665 return;
666
667 // Compute inliers and confidence (Brown & Lowe formula)
668 matches_info.num_inliers = 0;
669 for (size_t i = 0; i < matches_info.inliers_mask.size(); ++i)
670 if (matches_info.inliers_mask[i])
671 matches_info.num_inliers++;
672
673 matches_info.confidence = matches_info.num_inliers /
674 (8 + 0.3 * matches_info.matches.size());
675
676 // Zero confidence for too-close image pairs
677 if (matches_info.confidence > matches_confidence_thresh_)
678 matches_info.confidence = 0.;
679
680 // Refine homography on inliers if enough
681 if (matches_info.num_inliers >= num_matches_thresh2_)
682 {
683 src_points.create(1, matches_info.num_inliers, CV_32FC2);
684 dst_points.create(1, matches_info.num_inliers, CV_32FC2);
685 int inlier_idx = 0;
686 for (size_t i = 0; i < matches_info.matches.size(); ++i)
687 {
688 if (!matches_info.inliers_mask[i])
689 continue;
690
691 const DMatch& m = matches_info.matches[i];
692
693 Point2f p = features1.keypoints[m.queryIdx].pt;
694 p.x -= features1.img_size.width * 0.5f;
695 p.y -= features1.img_size.height * 0.5f;
696 src_points.at<Point2f>(0, inlier_idx) = p;
697
698 p = features2.keypoints[m.trainIdx].pt;
699 p.x -= features2.img_size.width * 0.5f;
700 p.y -= features2.img_size.height * 0.5f;
701 dst_points.at<Point2f>(0, inlier_idx) = p;
702
703 inlier_idx++;
704 }
705
706 matches_info.H = findHomography(src_points, dst_points, RANSAC);
707 }
708}
