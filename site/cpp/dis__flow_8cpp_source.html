243void DISOpticalFlowImpl::prepareBuffers(Mat &I0, Mat &I1, Mat &flow, bool use_flow)
244{
245 CV_INSTRUMENT_REGION();
246
247 I0s.resize(coarsest_scale + 1);
248 I1s.resize(coarsest_scale + 1);
249 I1s_ext.resize(coarsest_scale + 1);
250 I0xs.resize(coarsest_scale + 1);
251 I0ys.resize(coarsest_scale + 1);
252 Ux.resize(coarsest_scale + 1);
253 Uy.resize(coarsest_scale + 1);
254
255 Mat flow_uv[2];
256 if (use_flow)
257 {
258 split(flow, flow_uv);
259 initial_Ux.resize(coarsest_scale + 1);
260 initial_Uy.resize(coarsest_scale + 1);
261 }
262
263 int fraction = 1;
264 int cur_rows = 0, cur_cols = 0;
265
266 for (int i = 0; i <= coarsest_scale; i++)
267 {
268 /* Avoid initializing the pyramid levels above the finest scale, as they won't be used anyway */
269 if (i == finest_scale)
270 {
271 cur_rows = I0.rows / fraction;
272 cur_cols = I0.cols / fraction;
273 I0s[i].create(cur_rows, cur_cols);
274 resize(I0, I0s[i], I0s[i].size(), 0.0, 0.0, INTER_AREA);
275 I1s[i].create(cur_rows, cur_cols);
276 resize(I1, I1s[i], I1s[i].size(), 0.0, 0.0, INTER_AREA);
277
278 /* These buffers are reused in each scale so we initialize them once on the finest scale: */
279 Sx.create(cur_rows / patch_stride, cur_cols / patch_stride);
280 Sy.create(cur_rows / patch_stride, cur_cols / patch_stride);
281 I0xx_buf.create(cur_rows / patch_stride, cur_cols / patch_stride);
282 I0yy_buf.create(cur_rows / patch_stride, cur_cols / patch_stride);
283 I0xy_buf.create(cur_rows / patch_stride, cur_cols / patch_stride);
284 I0x_buf.create(cur_rows / patch_stride, cur_cols / patch_stride);
285 I0y_buf.create(cur_rows / patch_stride, cur_cols / patch_stride);
286
287 I0xx_buf_aux.create(cur_rows, cur_cols / patch_stride);
288 I0yy_buf_aux.create(cur_rows, cur_cols / patch_stride);
289 I0xy_buf_aux.create(cur_rows, cur_cols / patch_stride);
290 I0x_buf_aux.create(cur_rows, cur_cols / patch_stride);
291 I0y_buf_aux.create(cur_rows, cur_cols / patch_stride);
292
293 U.create(cur_rows, cur_cols);
294 }
295 else if (i > finest_scale)
296 {
297 cur_rows = I0s[i - 1].rows / 2;
298 cur_cols = I0s[i - 1].cols / 2;
299 I0s[i].create(cur_rows, cur_cols);
300 resize(I0s[i - 1], I0s[i], I0s[i].size(), 0.0, 0.0, INTER_AREA);
301 I1s[i].create(cur_rows, cur_cols);
302 resize(I1s[i - 1], I1s[i], I1s[i].size(), 0.0, 0.0, INTER_AREA);
303 }
304
305 if (i >= finest_scale)
306 {
307 I1s_ext[i].create(cur_rows + 2 * border_size, cur_cols + 2 * border_size);
308 copyMakeBorder(I1s[i], I1s_ext[i], border_size, border_size, border_size, border_size, BORDER_REPLICATE);
309 I0xs[i].create(cur_rows, cur_cols);
310 I0ys[i].create(cur_rows, cur_cols);
311 spatialGradient(I0s[i], I0xs[i], I0ys[i]);
312 Ux[i].create(cur_rows, cur_cols);
313 Uy[i].create(cur_rows, cur_cols);
314 variational_refinement_processors[i]->setAlpha(variational_refinement_alpha);
315 variational_refinement_processors[i]->setDelta(variational_refinement_delta);
316 variational_refinement_processors[i]->setGamma(variational_refinement_gamma);
317 variational_refinement_processors[i]->setEpsilon(variational_refinement_epsilon);
318 variational_refinement_processors[i]->setSorIterations(5);
319 variational_refinement_processors[i]->setFixedPointIterations(variational_refinement_iter);
320
321 if (use_flow)
322 {
323 resize(flow_uv[0], initial_Ux[i], Size(cur_cols, cur_rows));
324 initial_Ux[i] /= fraction;
325 resize(flow_uv[1], initial_Uy[i], Size(cur_cols, cur_rows));
326 initial_Uy[i] /= fraction;
327 }
328 }
329
330 fraction *= 2;
331 }
332}

334/* This function computes the structure tensor elements (local sums of I0x^2, I0x*I0y and I0y^2).
335 * A simple box filter is not used instead because we need to compute these sums on a sparse grid
336 * and store them densely in the output buffers.
337 */
338void DISOpticalFlowImpl::precomputeStructureTensor(Mat &dst_I0xx, Mat &dst_I0yy, Mat &dst_I0xy, Mat &dst_I0x,
339 Mat &dst_I0y, Mat &I0x, Mat &I0y)
340{
341 CV_INSTRUMENT_REGION();
342
343 float *I0xx_ptr = dst_I0xx.ptr<float>();
344 float *I0yy_ptr = dst_I0yy.ptr<float>();
345 float *I0xy_ptr = dst_I0xy.ptr<float>();
346 float *I0x_ptr = dst_I0x.ptr<float>();
347 float *I0y_ptr = dst_I0y.ptr<float>();
348
349 float *I0xx_aux_ptr = I0xx_buf_aux.ptr<float>();
350 float *I0yy_aux_ptr = I0yy_buf_aux.ptr<float>();
351 float *I0xy_aux_ptr = I0xy_buf_aux.ptr<float>();
352 float *I0x_aux_ptr = I0x_buf_aux.ptr<float>();
353 float *I0y_aux_ptr = I0y_buf_aux.ptr<float>();
354
355 /* Separable box filter: horizontal pass */
356 for (int i = 0; i < h; i++)
357 {
358 float sum_xx = 0.0f, sum_yy = 0.0f, sum_xy = 0.0f, sum_x = 0.0f, sum_y = 0.0f;
359 short *x_row = I0x.ptr<short>(i);
360 short *y_row = I0y.ptr<short>(i);
361 for (int j = 0; j < patch_size; j++)
362 {
363 sum_xx += x_row[j] * x_row[j];
364 sum_yy += y_row[j] * y_row[j];
365 sum_xy += x_row[j] * y_row[j];
366 sum_x += x_row[j];
367 sum_y += y_row[j];
368 }
369 I0xx_aux_ptr[i * ws] = sum_xx;
370 I0yy_aux_ptr[i * ws] = sum_yy;
371 I0xy_aux_ptr[i * ws] = sum_xy;
372 I0x_aux_ptr[i * ws] = sum_x;
373 I0y_aux_ptr[i * ws] = sum_y;
374 int js = 1;
375 for (int j = patch_size; j < w; j++)
376 {
377 sum_xx += (x_row[j] * x_row[j] - x_row[j - patch_size] * x_row[j - patch_size]);
378 sum_yy += (y_row[j] * y_row[j] - y_row[j - patch_size] * y_row[j - patch_size]);
379 sum_xy += (x_row[j] * y_row[j] - x_row[j - patch_size] * y_row[j - patch_size]);
380 sum_x += (x_row[j] - x_row[j - patch_size]);
381 sum_y += (y_row[j] - y_row[j - patch_size]);
382 if ((j - patch_size + 1) % patch_stride == 0)
383 {
384 I0xx_aux_ptr[i * ws + js] = sum_xx;
385 I0yy_aux_ptr[i * ws + js] = sum_yy;
386 I0xy_aux_ptr[i * ws + js] = sum_xy;
387 I0x_aux_ptr[i * ws + js] = sum_x;
388 I0y_aux_ptr[i * ws + js] = sum_y;
389 js++;
390 }
391 }
392 }
393
394 AutoBuffer<float> sum_xx(ws), sum_yy(ws), sum_xy(ws), sum_x(ws), sum_y(ws);
395 for (int j = 0; j < ws; j++)
396 {
397 sum_xx[j] = 0.0f;
398 sum_yy[j] = 0.0f;
399 sum_xy[j] = 0.0f;
400 sum_x[j] = 0.0f;
401 sum_y[j] = 0.0f;
402 }
403
404 /* Separable box filter: vertical pass */
405 for (int i = 0; i < patch_size; i++)
406 for (int j = 0; j < ws; j++)
407 {
408 sum_xx[j] += I0xx_aux_ptr[i * ws + j];
409 sum_yy[j] += I0yy_aux_ptr[i * ws + j];
410 sum_xy[j] += I0xy_aux_ptr[i * ws + j];
411 sum_x[j] += I0x_aux_ptr[i * ws + j];
412 sum_y[j] += I0y_aux_ptr[i * ws + j];
413 }
414 for (int j = 0; j < ws; j++)
415 {
416 I0xx_ptr[j] = sum_xx[j];
417 I0yy_ptr[j] = sum_yy[j];
418 I0xy_ptr[j] = sum_xy[j];
419 I0x_ptr[j] = sum_x[j];
420 I0y_ptr[j] = sum_y[j];
421 }
422 int is = 1;
423 for (int i = patch_size; i < h; i++)
424 {
425 for (int j = 0; j < ws; j++)
426 {
427 sum_xx[j] += (I0xx_aux_ptr[i * ws + j] - I0xx_aux_ptr[(i - patch_size) * ws + j]);
428 sum_yy[j] += (I0yy_aux_ptr[i * ws + j] - I0yy_aux_ptr[(i - patch_size) * ws + j]);
429 sum_xy[j] += (I0xy_aux_ptr[i * ws + j] - I0xy_aux_ptr[(i - patch_size) * ws + j]);
430 sum_x[j] += (I0x_aux_ptr[i * ws + j] - I0x_aux_ptr[(i - patch_size) * ws + j]);
431 sum_y[j] += (I0y_aux_ptr[i * ws + j] - I0y_aux_ptr[(i - patch_size) * ws + j]);
432 }
433 if ((i - patch_size + 1) % patch_stride == 0)
434 {
435 for (int j = 0; j < ws; j++)
436 {
437 I0xx_ptr[is * ws + j] = sum_xx[j];
438 I0yy_ptr[is * ws + j] = sum_yy[j];
439 I0xy_ptr[is * ws + j] = sum_xy[j];
440 I0x_ptr[is * ws + j] = sum_x[j];
441 I0y_ptr[is * ws + j] = sum_y[j];
442 }
443 is++;
444 }
445 }

448int DISOpticalFlowImpl::autoSelectCoarsestScale(int img_width)
449{
450 const int fratio = 5;
451 return std::max(0, (int)std::floor(log2((2.0f*(float)img_width) / ((float)fratio * (float)patch_size))));

464 case 3:
465 patch_size = 12;
466 coarsest_scale = autoSelectCoarsestScale(img_width);
467 finest_scale = std::max(coarsest_scale-4, 0);

471 patch_size = 12;
472 coarsest_scale = autoSelectCoarsestScale(img_width);
473 finest_scale = std::max(coarsest_scale-5, 0);
474 break;
475
476 // default case, fall-through.
477 case 2:
478 default:
479 patch_size = 8;
480 coarsest_scale = autoSelectCoarsestScale(img_width);
481 finest_scale = std::max(coarsest_scale-2, 0);
482 break;
483 }

523 /* Separate the left and right halves: */ \
524 v_expand(I1_row_next_8, I1_row_next_4_left, I1_row_next_4_right); \
525 v_expand(I1_row_next_shifted_8, I1_row_next_shifted_4_left, I1_row_next_shifted_4_right); \
526 \
527 /* Load current row of I0: */ \
528 I0_row_8 = v_load_expand(I0_ptr); \
529 v_expand(I0_row_8, I0_row_4_left, I0_row_4_right); \
530 \
531 /* Compute diffs between I0 and bilinearly interpolated I1: */ \
532 I_diff_left = v_sub(v_add(v_mul(w00v, v_cvt_f32(v_reinterpret_as_s32(I1_row_4_left))), \
533 v_mul(w01v, v_cvt_f32(v_reinterpret_as_s32(I1_row_shifted_4_left))), \
534 v_mul(w10v, v_cvt_f32(v_reinterpret_as_s32(I1_row_next_4_left))), \
535 v_mul(w11v, v_cvt_f32(v_reinterpret_as_s32(I1_row_next_shifted_4_left)))), \
536 v_cvt_f32(v_reinterpret_as_s32(I0_row_4_left))); \
537 I_diff_right = v_sub(v_add(v_mul(w00v, v_cvt_f32(v_reinterpret_as_s32(I1_row_4_right))), \
538 v_mul(w01v, v_cvt_f32(v_reinterpret_as_s32(I1_row_shifted_4_right))), \
539 v_mul(w10v, v_cvt_f32(v_reinterpret_as_s32(I1_row_next_4_right))), \
540 v_mul(w11v, v_cvt_f32(v_reinterpret_as_s32(I1_row_next_shifted_4_right)))), \
541 v_cvt_f32(v_reinterpret_as_s32(I0_row_4_right)));
542
543#define HAL_BILINEAR_8x8_PATCH_EXTRACTION_NEXT_ROW \
544 I0_ptr += I0_stride; \
545 I1_ptr += I1_stride; \
546 \
547 I1_row_4_left = I1_row_next_4_left; \
548 I1_row_4_right = I1_row_next_4_right; \
549 I1_row_shifted_4_left = I1_row_next_shifted_4_left; \
550 I1_row_shifted_4_right = I1_row_next_shifted_4_right;
551
552/* This function essentially performs one iteration of gradient descent when finding the most similar patch in I1 for a
553 * given one in I0. It assumes that I0_ptr and I1_ptr already point to the corresponding patches and w00, w01, w10, w11
554 * are precomputed bilinear interpolation weights. It returns the SSD (sum of squared differences) between these patches
555 * and computes the values (dst_dUx, dst_dUy) that are used in the flow vector update. HAL acceleration is implemented
556 * only for the default patch size (8x8). Everything is processed in floats as using fixed-point approximations harms
557 * the quality significantly.
558 */
559inline float processPatch(float &dst_dUx, float &dst_dUy, uchar *I0_ptr, uchar *I1_ptr, short *I0x_ptr, short *I0y_ptr,
560 int I0_stride, int I1_stride, float w00, float w01, float w10, float w11, int patch_sz)
561{
562 float SSD = 0.0f;
563#if CV_SIMD128
564 if (patch_sz == 8)
565 {
566 /* Variables to accumulate the sums */
567 v_float32x4 Ux_vec = v_setall_f32(0);
568 v_float32x4 Uy_vec = v_setall_f32(0);
569 v_float32x4 SSD_vec = v_setall_f32(0);
570
571 v_int16x8 I0x_row, I0y_row;
572 v_int32x4 I0x_row_4_left, I0x_row_4_right, I0y_row_4_left, I0y_row_4_right;
573
574 HAL_INIT_BILINEAR_8x8_PATCH_EXTRACTION;
575 for (int row = 0; row < 8; row++)
576 {
577 HAL_PROCESS_BILINEAR_8x8_PATCH_EXTRACTION;
578 I0x_row = v_load(I0x_ptr);
579 v_expand(I0x_row, I0x_row_4_left, I0x_row_4_right);
580 I0y_row = v_load(I0y_ptr);
581 v_expand(I0y_row, I0y_row_4_left, I0y_row_4_right);
582
583 /* Update the sums: */
584 Ux_vec = v_add(Ux_vec, v_add(v_mul(I_diff_left, v_cvt_f32(I0x_row_4_left)), v_mul(I_diff_right, v_cvt_f32(I0x_row_4_right))));
585 Uy_vec = v_add(Uy_vec, v_add(v_mul(I_diff_left, v_cvt_f32(I0y_row_4_left)), v_mul(I_diff_right, v_cvt_f32(I0y_row_4_right))));
586 SSD_vec = v_add(SSD_vec, v_add(v_mul(I_diff_left, I_diff_left), v_mul(I_diff_right, I_diff_right)));
587
588 I0x_ptr += I0_stride;
589 I0y_ptr += I0_stride;
590 HAL_BILINEAR_8x8_PATCH_EXTRACTION_NEXT_ROW;
591 }
592
593 /* Final reduce operations: */
594 dst_dUx = v_reduce_sum(Ux_vec);
595 dst_dUy = v_reduce_sum(Uy_vec);
596 SSD = v_reduce_sum(SSD_vec);
597 }
598 else
599#endif
600 {
601 dst_dUx = 0.0f;
602 dst_dUy = 0.0f;
603 float diff;
604 for (int i = 0; i < patch_sz; i++)
605 for (int j = 0; j < patch_sz; j++)
606 {
607 diff = w00 * I1_ptr[i * I1_stride + j] + w01 * I1_ptr[i * I1_stride + j + 1] +
608 w10 * I1_ptr[(i + 1) * I1_stride + j] + w11 * I1_ptr[(i + 1) * I1_stride + j + 1] -
609 I0_ptr[i * I0_stride + j];
610
611 SSD += diff * diff;
612 dst_dUx += diff * I0x_ptr[i * I0_stride + j];
613 dst_dUy += diff * I0y_ptr[i * I0_stride + j];
614 }
615 }
616 return SSD;
617}

619/* Same as processPatch, but with patch mean normalization, which improves robustness under changing
620 * lighting conditions
621 */
622inline float processPatchMeanNorm(float &dst_dUx, float &dst_dUy, uchar *I0_ptr, uchar *I1_ptr, short *I0x_ptr,
623 short *I0y_ptr, int I0_stride, int I1_stride, float w00, float w01, float w10,
624 float w11, int patch_sz, float x_grad_sum, float y_grad_sum)
625{
626 float sum_diff = 0.0, sum_diff_sq = 0.0;
627 float sum_I0x_mul = 0.0, sum_I0y_mul = 0.0;
628 float n = (float)patch_sz * patch_sz;
629
630#if CV_SIMD128
631 if (patch_sz == 8)
632 {
633 /* Variables to accumulate the sums */
634 v_float32x4 sum_I0x_mul_vec = v_setall_f32(0);
635 v_float32x4 sum_I0y_mul_vec = v_setall_f32(0);
636 v_float32x4 sum_diff_vec = v_setall_f32(0);
637 v_float32x4 sum_diff_sq_vec = v_setall_f32(0);
638
639 v_int16x8 I0x_row, I0y_row;
640 v_int32x4 I0x_row_4_left, I0x_row_4_right, I0y_row_4_left, I0y_row_4_right;
641
642 HAL_INIT_BILINEAR_8x8_PATCH_EXTRACTION;
643 for (int row = 0; row < 8; row++)
644 {
645 HAL_PROCESS_BILINEAR_8x8_PATCH_EXTRACTION;
646 I0x_row = v_load(I0x_ptr);
647 v_expand(I0x_row, I0x_row_4_left, I0x_row_4_right);
648 I0y_row = v_load(I0y_ptr);
649 v_expand(I0y_row, I0y_row_4_left, I0y_row_4_right);
650
651 /* Update the sums: */
652 sum_I0x_mul_vec = v_add(sum_I0x_mul_vec, v_add(v_mul(I_diff_left, v_cvt_f32(I0x_row_4_left)), v_mul(I_diff_right, v_cvt_f32(I0x_row_4_right))));
653 sum_I0y_mul_vec = v_add(sum_I0y_mul_vec, v_add(v_mul(I_diff_left, v_cvt_f32(I0y_row_4_left)), v_mul(I_diff_right, v_cvt_f32(I0y_row_4_right))));
654 sum_diff_sq_vec = v_add(sum_diff_sq_vec, v_add(v_mul(I_diff_left, I_diff_left), v_mul(I_diff_right, I_diff_right)));
655 sum_diff_vec = v_add(sum_diff_vec, v_add(I_diff_left, I_diff_right));
656
657 I0x_ptr += I0_stride;
658 I0y_ptr += I0_stride;
659 HAL_BILINEAR_8x8_PATCH_EXTRACTION_NEXT_ROW;
660 }
661
662 /* Final reduce operations: */
663 sum_I0x_mul = v_reduce_sum(sum_I0x_mul_vec);
664 sum_I0y_mul = v_reduce_sum(sum_I0y_mul_vec);
665 sum_diff = v_reduce_sum(sum_diff_vec);
666 sum_diff_sq = v_reduce_sum(sum_diff_sq_vec);
667 }
668 else
669#endif
670 {
671 float diff;
672 for (int i = 0; i < patch_sz; i++)
673 for (int j = 0; j < patch_sz; j++)
674 {
675 diff = w00 * I1_ptr[i * I1_stride + j] + w01 * I1_ptr[i * I1_stride + j + 1] +
676 w10 * I1_ptr[(i + 1) * I1_stride + j] + w11 * I1_ptr[(i + 1) * I1_stride + j + 1] -
677 I0_ptr[i * I0_stride + j];
678
679 sum_diff += diff;
680 sum_diff_sq += diff * diff;
681
682 sum_I0x_mul += diff * I0x_ptr[i * I0_stride + j];
683 sum_I0y_mul += diff * I0y_ptr[i * I0_stride + j];
684 }
685 }
686 dst_dUx = sum_I0x_mul - sum_diff * x_grad_sum / n;
687 dst_dUy = sum_I0y_mul - sum_diff * y_grad_sum / n;
688 return sum_diff_sq - sum_diff * sum_diff / n;
689}

772void DISOpticalFlowImpl::PatchInverseSearch_ParBody::operator()(const Range &range) const
773{
774 CV_INSTRUMENT_REGION();
775
776 // force separate processing of stripes if we are using spatial propagation:
777 if (dis->use_spatial_propagation && range.end > range.start + 1)
778 {
779 for (int n = range.start; n < range.end; n++)
780 (*this)(Range(n, n + 1));
781 return;
782 }
783 int psz = dis->patch_size;
784 int psz2 = psz / 2;
785 int w_ext = dis->w + 2 * dis->border_size;
786 int bsz = dis->border_size;
787
788 /* Input dense flow */
789 float *Ux_ptr = Ux->ptr<float>();
790 float *Uy_ptr = Uy->ptr<float>();
791
792 /* Output sparse flow */
793 float *Sx_ptr = Sx->ptr<float>();
794 float *Sy_ptr = Sy->ptr<float>();
795
796 uchar *I0_ptr = I0->ptr<uchar>();
797 uchar *I1_ptr = I1->ptr<uchar>();
798 short *I0x_ptr = I0x->ptr<short>();
799 short *I0y_ptr = I0y->ptr<short>();
800
801 /* Precomputed structure tensor */
802 float *xx_ptr = dis->I0xx_buf.ptr<float>();
803 float *yy_ptr = dis->I0yy_buf.ptr<float>();
804 float *xy_ptr = dis->I0xy_buf.ptr<float>();
805 /* And extra buffers for mean-normalization: */
806 float *x_ptr = dis->I0x_buf.ptr<float>();
807 float *y_ptr = dis->I0y_buf.ptr<float>();
808
809 bool use_temporal_candidates = false;
810 float *initial_Ux_ptr = NULL, *initial_Uy_ptr = NULL;
811 if (!dis->initial_Ux.empty())
812 {
813 initial_Ux_ptr = dis->initial_Ux[pyr_level].ptr<float>();
814 initial_Uy_ptr = dis->initial_Uy[pyr_level].ptr<float>();
815 use_temporal_candidates = true;
816 }
817
818 int i, j, dir;
819 int start_is, end_is, start_js, end_js;
820 int start_i, start_j;
821 float i_lower_limit = bsz - psz + 1.0f;
822 float i_upper_limit = bsz + dis->h - 1.0f;
823 float j_lower_limit = bsz - psz + 1.0f;
824 float j_upper_limit = bsz + dis->w - 1.0f;
825 float dUx, dUy, i_I1, j_I1, w00, w01, w10, w11, dx, dy;
826
827#define INIT_BILINEAR_WEIGHTS(Ux, Uy) \
828 i_I1 = min(max(i + Uy + bsz, i_lower_limit), i_upper_limit); \
829 j_I1 = min(max(j + Ux + bsz, j_lower_limit), j_upper_limit); \
830 { \
831 float di = i_I1 - floor(i_I1); \
832 float dj = j_I1 - floor(j_I1); \
833 w11 = di * dj; \
834 w10 = di * (1 - dj); \
835 w01 = (1 - di) * dj; \
836 w00 = (1 - di) * (1 - dj); \
837 }
838
839#define COMPUTE_SSD(dst, Ux, Uy) \
840 INIT_BILINEAR_WEIGHTS(Ux, Uy); \
841 if (dis->use_mean_normalization) \
842 dst = computeSSDMeanNorm(I0_ptr + i * dis->w + j, I1_ptr + (int)i_I1 * w_ext + (int)j_I1, dis->w, w_ext, w00, \
843 w01, w10, w11, psz); \
844 else \
845 dst = computeSSD(I0_ptr + i * dis->w + j, I1_ptr + (int)i_I1 * w_ext + (int)j_I1, dis->w, w_ext, w00, w01, \
846 w10, w11, psz);
847
848 int num_inner_iter = (int)floor(dis->grad_descent_iter / (float)num_iter);
849 for (int iter = 0; iter < num_iter; iter++)
850 {
851 if (iter % 2 == 0)
852 {
853 dir = 1;
854 start_is = min(range.start * stripe_sz, hs);
855 end_is = min(range.end * stripe_sz, hs);
856 start_js = 0;
857 end_js = dis->ws;
858 start_i = start_is * dis->patch_stride;
859 start_j = 0;
860 }
861 else
862 {
863 dir = -1;
864 start_is = min(range.end * stripe_sz, hs) - 1;
865 end_is = min(range.start * stripe_sz, hs) - 1;
866 start_js = dis->ws - 1;
867 end_js = -1;
868 start_i = start_is * dis->patch_stride;
869 start_j = (dis->ws - 1) * dis->patch_stride;
870 }
871
872 i = start_i;
873 for (int is = start_is; dir * is < dir * end_is; is += dir)
874 {
875 j = start_j;
876 for (int js = start_js; dir * js < dir * end_js; js += dir)
877 {
878 if (iter == 0)
879 {
880 /* Using result form the previous pyramid level as the very first approximation: */
881 Sx_ptr[is * dis->ws + js] = Ux_ptr[(i + psz2) * dis->w + j + psz2];
882 Sy_ptr[is * dis->ws + js] = Uy_ptr[(i + psz2) * dis->w + j + psz2];
883 }
884
885 float min_SSD = INF, cur_SSD;
886 if (use_temporal_candidates || dis->use_spatial_propagation)
887 {
888 COMPUTE_SSD(min_SSD, Sx_ptr[is * dis->ws + js], Sy_ptr[is * dis->ws + js]);
889 }
890
891 if (use_temporal_candidates)
892 {
893 /* Try temporal candidates (vectors from the initial flow field that was passed to the function) */
894 COMPUTE_SSD(cur_SSD, initial_Ux_ptr[(i + psz2) * dis->w + j + psz2],
895 initial_Uy_ptr[(i + psz2) * dis->w + j + psz2]);
896 if (cur_SSD < min_SSD)
897 {
898 min_SSD = cur_SSD;
899 Sx_ptr[is * dis->ws + js] = initial_Ux_ptr[(i + psz2) * dis->w + j + psz2];
900 Sy_ptr[is * dis->ws + js] = initial_Uy_ptr[(i + psz2) * dis->w + j + psz2];
901 }
902 }
903
904 if (dis->use_spatial_propagation)
905 {
906 /* Try spatial candidates: */
907 if (dir * js > dir * start_js)
908 {
909 COMPUTE_SSD(cur_SSD, Sx_ptr[is * dis->ws + js - dir], Sy_ptr[is * dis->ws + js - dir]);
910 if (cur_SSD < min_SSD)
911 {
912 min_SSD = cur_SSD;
913 Sx_ptr[is * dis->ws + js] = Sx_ptr[is * dis->ws + js - dir];
914 Sy_ptr[is * dis->ws + js] = Sy_ptr[is * dis->ws + js - dir];
915 }
916 }
917 /* Flow vectors won't actually propagate across different stripes, which is the reason for keeping
918 * the number of stripes constant. It works well enough in practice and doesn't introduce any
919 * visible seams.
920 */
921 if (dir * is > dir * start_is)
922 {
923 COMPUTE_SSD(cur_SSD, Sx_ptr[(is - dir) * dis->ws + js], Sy_ptr[(is - dir) * dis->ws + js]);
924 if (cur_SSD < min_SSD)
925 {
926 min_SSD = cur_SSD;
927 Sx_ptr[is * dis->ws + js] = Sx_ptr[(is - dir) * dis->ws + js];
928 Sy_ptr[is * dis->ws + js] = Sy_ptr[(is - dir) * dis->ws + js];
929 }
930 }
931 }
932
933 /* Use the best candidate as a starting point for the gradient descent: */
934 float cur_Ux = Sx_ptr[is * dis->ws + js];
935 float cur_Uy = Sy_ptr[is * dis->ws + js];
936
937 /* Computing the inverse of the structure tensor: */
938 float detH = xx_ptr[is * dis->ws + js] * yy_ptr[is * dis->ws + js] -
939 xy_ptr[is * dis->ws + js] * xy_ptr[is * dis->ws + js];
940 if (abs(detH) < EPS)
941 detH = EPS;
942 float invH11 = yy_ptr[is * dis->ws + js] / detH;
943 float invH12 = -xy_ptr[is * dis->ws + js] / detH;
944 float invH22 = xx_ptr[is * dis->ws + js] / detH;
945 float prev_SSD = INF, SSD;
946 float x_grad_sum = x_ptr[is * dis->ws + js];
947 float y_grad_sum = y_ptr[is * dis->ws + js];
948
949 for (int t = 0; t < num_inner_iter; t++)
950 {
951 INIT_BILINEAR_WEIGHTS(cur_Ux, cur_Uy);
952 if (dis->use_mean_normalization)
953 SSD = processPatchMeanNorm(dUx, dUy,
954 I0_ptr + i * dis->w + j, I1_ptr + (int)i_I1 * w_ext + (int)j_I1,
955 I0x_ptr + i * dis->w + j, I0y_ptr + i * dis->w + j,
956 dis->w, w_ext, w00, w01, w10, w11, psz,
957 x_grad_sum, y_grad_sum);
958 else
959 SSD = processPatch(dUx, dUy,
960 I0_ptr + i * dis->w + j, I1_ptr + (int)i_I1 * w_ext + (int)j_I1,
961 I0x_ptr + i * dis->w + j, I0y_ptr + i * dis->w + j,
962 dis->w, w_ext, w00, w01, w10, w11, psz);
963
964 dx = invH11 * dUx + invH12 * dUy;
965 dy = invH12 * dUx + invH22 * dUy;
966 cur_Ux -= dx;
967 cur_Uy -= dy;
968
969 /* Break when patch distance stops decreasing */
970 if (SSD >= prev_SSD)
971 break;
972 prev_SSD = SSD;
973 }
974
975 /* If gradient descent converged to a flow vector that is very far from the initial approximation
976 * (more than patch size) then we don't use it. Noticeably improves the robustness.
977 */
978 if (norm(Vec2f(cur_Ux - Sx_ptr[is * dis->ws + js], cur_Uy - Sy_ptr[is * dis->ws + js])) <= psz)
979 {
980 Sx_ptr[is * dis->ws + js] = cur_Ux;

1000/* This function transforms a sparse optical flow field obtained by PatchInverseSearch (which computes flow values
1001 * on a sparse grid defined by patch_stride) into a dense optical flow field by weighted averaging of values from the
1002 * overlapping patches.
1003 */
1004void DISOpticalFlowImpl::Densification_ParBody::operator()(const Range &range) const
1005{
1006 CV_INSTRUMENT_REGION();
1007
1008 int start_i = min(range.start * stripe_sz, h);
1009 int end_i = min(range.end * stripe_sz, h);
1010
1011 /* Input sparse flow */
1012 float *Sx_ptr = Sx->ptr<float>();
1013 float *Sy_ptr = Sy->ptr<float>();
1014
1015 /* Output dense flow */
1016 float *Ux_ptr = Ux->ptr<float>();
1017 float *Uy_ptr = Uy->ptr<float>();
1018
1019 uchar *I0_ptr = I0->ptr<uchar>();
1020 uchar *I1_ptr = I1->ptr<uchar>();
1021
1022 int psz = dis->patch_size;
1023 int pstr = dis->patch_stride;
1024 int i_l, i_u;
1025 int j_l, j_u;
1026 float i_m, j_m, diff;
1027
1028 /* These values define the set of sparse grid locations that contain patches overlapping with the current dense flow
1029 * location */
1030 int start_is, end_is;
1031 int start_js, end_js;
1032
1033/* Some helper macros for updating this set of sparse grid locations */
1034#define UPDATE_SPARSE_I_COORDINATES \
1035 if (i % pstr == 0 && i + psz <= h) \
1036 end_is++; \
1037 if (i - psz >= 0 && (i - psz) % pstr == 0 && start_is < end_is) \
1038 start_is++;
1039
1040#define UPDATE_SPARSE_J_COORDINATES \
1041 if (j % pstr == 0 && j + psz <= dis->w) \
1042 end_js++; \
1043 if (j - psz >= 0 && (j - psz) % pstr == 0 && start_js < end_js) \
1044 start_js++;
1045
1046 start_is = 0;
1047 end_is = -1;
1048 for (int i = 0; i < start_i; i++)
1049 {
1050 UPDATE_SPARSE_I_COORDINATES;
1051 }
1052 for (int i = start_i; i < end_i; i++)
1053 {
1054 UPDATE_SPARSE_I_COORDINATES;
1055 start_js = 0;
1056 end_js = -1;
1057 for (int j = 0; j < dis->w; j++)
1058 {
1059 UPDATE_SPARSE_J_COORDINATES;
1060 float coef, sum_coef = 0.0f;
1061 float sum_Ux = 0.0f;
1062 float sum_Uy = 0.0f;
1063
1064 /* Iterate through all the patches that overlap the current location (i,j) */
1065 for (int is = start_is; is <= end_is; is++)
1066 for (int js = start_js; js <= end_js; js++)
1067 {
1068 j_m = min(max(j + Sx_ptr[is * dis->ws + js], 0.0f), dis->w - 1.0f - EPS);
1069 i_m = min(max(i + Sy_ptr[is * dis->ws + js], 0.0f), dis->h - 1.0f - EPS);
1070 j_l = (int)j_m;
1071 j_u = j_l + 1;
1072 i_l = (int)i_m;
1073 i_u = i_l + 1;
1074 diff = (j_m - j_l) * (i_m - i_l) * I1_ptr[i_u * dis->w + j_u] +
1075 (j_u - j_m) * (i_m - i_l) * I1_ptr[i_u * dis->w + j_l] +
1076 (j_m - j_l) * (i_u - i_m) * I1_ptr[i_l * dis->w + j_u] +
1077 (j_u - j_m) * (i_u - i_m) * I1_ptr[i_l * dis->w + j_l] - I0_ptr[i * dis->w + j];
1078 coef = 1 / max(1.0f, abs(diff));
1079 sum_Ux += coef * Sx_ptr[is * dis->ws + js];
1080 sum_Uy += coef * Sy_ptr[is * dis->ws + js];
1081 sum_coef += coef;
1082 }
1083 CV_DbgAssert(sum_coef != 0);
1084 Ux_ptr[i * dis->w + j] = sum_Ux / sum_coef;
1085 Uy_ptr[i * dis->w + j] = sum_Uy / sum_coef;
1086 }
1087 }
1088#undef UPDATE_SPARSE_I_COORDINATES
1089#undef UPDATE_SPARSE_J_COORDINATES
1090}

1508 I0xs[i], I0ys[i], 1, i));
