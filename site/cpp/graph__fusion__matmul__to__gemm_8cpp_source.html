5// Rewrites a MatMul whose B (and optionally bias) are constant blobs into a
6// Gemm with `flatten_a=false`. The Gemm path pre-packs B in MLAS layout at
7// finalize() and dispatches each forward through MlasGemmPacked, which is
8// notably faster than the OpenCV-internal packed AVX2 kernel that the MatMul
9// constant-B path otherwise hits.
10//
11// Pattern (post-importer; bias has already been absorbed into MatMul.blobs[1]
12// by BiasedMatmulSubgraph when it was a const Add):
13// A(any rank) -> MatMul[const B (2D), optional const bias [N] / scalar]
14//
15// Replacement:
16// A -> Gemm[transA=false, transB=mm.trans_b, alpha=mm.alpha, beta=mm.beta,
17// constB=true, const_C=have_bias, flatten_a=false,
18// blobs={B, bias?}]
19//
20// Restrictions (skip otherwise):
21// - MatMul.trans_a must be false (ND-A flattening assumes row-major K is the
22// last axis)
23// - blobs.size() in {1, 2} and blobs[0] (B) is 2D float32
24// - if a bias is present, it must be scalar (total==1) or 1D length-N. The
25// Gemm flatten_a=false bias path tiles a per-row pattern, so 2D / per-row
26// biases like [M, N] aren't supported by this rewriter (they would change
27// value across the flattened rows).

38struct ModelFusionMatMulToGemm
39{
40 explicit ModelFusionMatMulToGemm(Net::Impl* netimpl_) : netimpl(netimpl_) {}
41
42 void fuse() { fuseGraph(netimpl->mainGraph); }
43
44 bool fuseGraph(Ptr<Graph>& graph)
45 {
46 const vector<Ptr<LayerInfo>>& prog = graph->prog();
47 size_t nops = prog.size();
48 bool modified = false;
49
50 for (size_t i = 0; i < nops; i++) {
51 if (!prog[i]) continue;
52 vector<Ptr<Graph>>* subgraphs = prog[i]->subgraphs();
53 if (subgraphs) {
54 for (Ptr<Graph>& g : *subgraphs)
55 if (fuseGraph(g)) modified = true;
56 }
57 }
58
59 vector<Ptr<LayerInfo>> newprog = prog;
60 bool changed = false;
61
62 for (size_t i = 0; i < nops; i++) {
63 const Ptr<LayerInfo>& layer = newprog[i];
64 if (!layer) continue;
65
66 MatMulLayer* mm = dynamic_cast<MatMulLayer*>(layer.get());
67 if (!mm) continue;
68 // MatMulInt8Layer carries quantization state that this rewriter
69 // doesn't know how to translate to Gemm.
70 if (dynamic_cast<MatMulInt8Layer*>(layer.get())) continue;
71
72 // Constant B lives in blobs[0] post-parse; if blobs is empty the
73 // MatMul has a runtime B and isn't a candidate.
74 if (layer->blobs.empty() || layer->blobs.size() > 2) continue;
75
76 if (mm->trans_a) continue; // ND-A flatten assumes K = last axis
77
78 const Mat& B = layer->blobs[0];
79 if (B.dims != 2 || B.type() != CV_32F) continue;
80
81 // Single runtime input expected (the const B was absorbed into blobs).
82 if (layer->inputs.size() != 1) continue;
83
84 int N = mm->trans_b ? B.size[0] : B.size[1];
85 int K = mm->trans_b ? B.size[1] : B.size[0];
86 (void)K;
87 if (N <= 0) continue;
88
89 const bool have_bias = layer->blobs.size() == 2;
90 if (have_bias) {
91 const Mat& bias = layer->blobs[1];
92 if (bias.type() != CV_32F) continue;
93 int total = (int)bias.total();
94 if (total != 1 && total != N) continue;
95 }
96
97 // Build the Gemm replacement.
98 LayerParams gp;
99 gp.name = layer->name; // keep the name for profiling continuity
100 gp.type = "Gemm";
101 gp.set("transA", false);
102 gp.set("transB", mm->trans_b);
103 gp.set("alpha", mm->alpha);
104 gp.set("beta", mm->beta);
105 gp.set("constB", true);
106 gp.set("have_bias", have_bias);
107 gp.set("const_C", have_bias);
108 // flatten_a=false: keep A's leading dims so downstream consumers
109 // see the same shape they did when the producer was a MatMul.
110 gp.set("flatten_a", false);
111 // For the new GemmLayerImpl flatten_a=false bias path the bias is
112 // restricted to scalar or [N]; both of those map to real_ndims_C
113 // <= 1, but we don't actually consult it in that path — set it
114 // anyway so the Ngraph/CANN backends still get a sensible value.
115 if (have_bias) {
116 int total = (int)layer->blobs[1].total();
117 gp.set("real_ndims_C", total == 1 ? 0 : 1);
118 }
119
120 gp.blobs.push_back(B);
121 if (have_bias) gp.blobs.push_back(layer->blobs[1]);
122
123 Ptr<LayerInfo> gemm = LayerFactory::createLayerInstance("Gemm", gp);
124 if (!gemm) continue;
125 gemm->inputs = layer->inputs;
126 gemm->outputs = layer->outputs;
127 gemm->netimpl = netimpl;
128
129 newprog[i] = gemm;
130 changed = true;
131 modified = true;
132 }
133
134 if (changed) graph->setProg(newprog);
135 return modified;
136 }
137
138 Net::Impl* netimpl;
139};
