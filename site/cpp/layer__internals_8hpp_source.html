141 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
142 {
143 CV_TRACE_FUNCTION();
144 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
145
146 // FIXIT: add wrapper without exception suppression
147 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
148 forward_ocl(inputs_arr, outputs_arr, internals_arr))
149
150 std::vector<Mat> outputs, internals;
151 outputs_arr.getMatVector(outputs);
152 internals_arr.getMatVector(internals);
153
154 for (int i = 0; i < inputsData.size(); ++i)
155 {
156 bool isFP16 = outputs[i].depth() == CV_16F;
157 if (inputsData[i].type() != CV_32F)
158 {
159 CV_CheckTypeEQ(outputs[i].type(), inputsData[i].type(), "");
160 CV_CheckTrue(means[i] == Scalar() && scaleFactors[i] == 1.0, "Input mean and scale are supported only for float32 input");
161 inputsData[i].copyTo(outputs[i]);
162 continue;
163 }
164 double scale = scaleFactors[i];
165 Scalar& mean = means[i];
166
167 CV_Assert(mean == Scalar() || inputsData[i].size[1] <= 4);
168 if (isFP16)
169 CV_CheckTypeEQ(outputs[i].type(), CV_16FC1, "");
170 else
171 CV_CheckTypeEQ(outputs[i].type(), CV_32FC1, "");
172
173 bool singleMean = true;
174 for (int j = 1; j < std::min(4, inputsData[i].size[1]) && singleMean; ++j)
175 {
176 singleMean = mean[j] == mean[j - 1];
177 }
178
179 if (singleMean)
180 {
181 if (isFP16)
182 {
183 Mat input_f32;
184 inputsData[i].convertTo(input_f32, CV_32F, scale, -mean[0] * scale);
185 input_f32.convertTo(outputs[i], CV_16F);
186 }
187 else
188 {
189 inputsData[i].convertTo(outputs[i], CV_32F, scale, -mean[0] * scale);
190 }
191 }
192 else
193 {
194 for (int n = 0; n < inputsData[i].size[0]; ++n)
195 {
196 for (int c = 0; c < inputsData[i].size[1]; ++c)
197 {
198 Mat inp = getPlane(inputsData[i], n, c);
199 Mat out = getPlane(outputs[i], n, c);
200 if (isFP16)
201 {
202 Mat input_f32;
203 inp.convertTo(input_f32, CV_32F, scale, -mean[c] * scale);
204 input_f32.convertTo(out, CV_16F);
205 }
206 else
207 {
208 inp.convertTo(out, CV_32F, scale, -mean[c] * scale);
209 }
210 }
211 }
212 }

217 bool forward_ocl(InputArrayOfArrays, OutputArrayOfArrays outputs_, OutputArrayOfArrays internals_)
218 {
219 std::vector<UMat> outputs;
220 outputs_.getUMatVector(outputs);
221
222 for (int i = 0; i < inputsData.size(); ++i)
223 {
224 bool isFP16 = outputs[i].depth() == CV_16F;
225 if (inputsData[i].type() != CV_32F)
226 {
227 CV_CheckTypeEQ(outputs[i].type(), inputsData[i].type(), "");
228 CV_CheckTrue(means[i] == Scalar() && scaleFactors[i] == 1.0, "Input mean and scale are supported only for float32 input");
229 inputsData[i].copyTo(outputs[i]);
230 continue;
231 }
232 Mat inputData = inputsData[i];
233
234 double scale = scaleFactors[i];
235 Scalar& mean = means[i];
236
237 CV_Assert(mean == Scalar() || inputData.size[1] <= 4);
238 if (isFP16)
239 CV_CheckTypeEQ(outputs[i].type(), CV_16FC1, "");
240 else
241 CV_CheckTypeEQ(outputs[i].type(), CV_32FC1, "");
242
243 bool singleMean = true;
244 if (mean != Scalar())
245 {
246 for (int j = 1; j < std::min(4, inputData.size[1]) && singleMean; ++j)
247 {
248 singleMean = mean[j] == mean[j - 1];
249 }
250 }
251
252 if (singleMean)
253 {
254 if (isFP16)
255 {
256 UMat input_i;
257 inputData.convertTo(input_i, CV_32F, scale, -mean[0] * scale);
258 input_i.convertTo(outputs[i], CV_16F);
259 }
260 else
261 {
262 inputData.convertTo(outputs[i], CV_32F, scale, -mean[0] * scale);
263 }
264 }
265 else
266 {
267 for (int n = 0; n < inputData.size[0]; ++n)
268 {
269 for (int c = 0; c < inputData.size[1]; ++c)
270 {
271 Mat inp = getPlane(inputData, n, c);
272
273 std::vector<cv::Range> plane(4, Range::all());
274 plane[0] = Range(n, n + 1);
275 plane[1] = Range(c, c + 1);
276 UMat out = outputs[i](plane).reshape(1, inp.size);
277
278 if (isFP16)
279 {
280 UMat input_i;
281 inp.convertTo(input_i, CV_32F, scale, -mean[c] * scale);
282 input_i.convertTo(out, CV_16F);
283 }
284 else
285 {
286 inp.convertTo(out, CV_32F, scale, -mean[c] * scale);
287 }
288 }
289 }
290 }
291 }
292 return true;

340 virtual void finalize(InputArrayOfArrays, OutputArrayOfArrays outputs_arr) CV_OVERRIDE
341 {
342 std::vector<Mat> outputs;
343 outputs_arr.getMatVector(outputs);
344
345 CV_Assert_N(outputs.size() == scaleFactors.size(), outputs.size() == means.size(),
346 inputsData.size() == outputs.size());
347 skip = true;
348 for (int i = 0; skip && i < inputsData.size(); ++i)

351 skip = false;
