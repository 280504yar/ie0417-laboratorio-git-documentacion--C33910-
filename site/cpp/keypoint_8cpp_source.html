49 KeypointResponseGreaterThanOrEqualToThreshold(float _value) :

69void KeyPointsFilter::retainBest(std::vector<KeyPoint>& keypoints, int n_points)
70{
71 //this is only necessary if the keypoints size is greater than the number of desired points.
72 if( n_points >= 0 && keypoints.size() > (size_t)n_points )
73 {
74 if (n_points==0)
75 {
76 keypoints.clear();
77 return;
78 }
79 //first use nth element to partition the keypoints into the best and worst.
80 std::nth_element(keypoints.begin(), keypoints.begin() + n_points - 1, keypoints.end(), KeypointResponseGreater());
81 //this is the boundary response, and in the case of FAST may be ambiguous
82 float ambiguous_response = keypoints[n_points - 1].response;
83 //use std::partition to grab all of the keypoints with the boundary response.
84 std::vector<KeyPoint>::const_iterator new_end =
85 std::partition(keypoints.begin() + n_points, keypoints.end(),
86 KeypointResponseGreaterThanOrEqualToThreshold(ambiguous_response));
87 //resize the keypoints, given this new end point. nth_element and partition reordered the points inplace
88 keypoints.resize(new_end - keypoints.begin());
89 }
90}

173template <typename T>
174void runByPixelsMask2(std::vector<KeyPoint> &keypoints, std::vector<T> &removeFrom, const Mat &mask)
175{
176 if (mask.empty())
177 return;
178
179 MaskPredicate maskPredicate(mask);
180 removeFrom.erase(std::remove_if(removeFrom.begin(), removeFrom.end(),
181 [&](const T &x)
182 {
183 auto index = &x - &removeFrom.front();
184 return maskPredicate(keypoints[index]);
185 }),
186 removeFrom.end());
187 keypoints.erase(std::remove_if(keypoints.begin(), keypoints.end(), maskPredicate), keypoints.end());
188}

194struct KeyPoint_LessThan
195{
196 KeyPoint_LessThan(const std::vector<KeyPoint>& _kp) : kp(&_kp) {}
197 bool operator()(int i, int j) const
198 {
199 const KeyPoint& kp1 = (*kp)[i];
200 const KeyPoint& kp2 = (*kp)[j];
201 if( kp1.pt.x != kp2.pt.x )
202 return kp1.pt.x < kp2.pt.x;
203 if( kp1.pt.y != kp2.pt.y )
204 return kp1.pt.y < kp2.pt.y;
205 if( kp1.size != kp2.size )
206 return kp1.size > kp2.size;
207 if( kp1.angle != kp2.angle )
208 return kp1.angle < kp2.angle;
209 if( kp1.response != kp2.response )
210 return kp1.response > kp2.response;
211 if( kp1.octave != kp2.octave )
212 return kp1.octave > kp2.octave;
213 if( kp1.class_id != kp2.class_id )
214 return kp1.class_id > kp2.class_id;
215
216 return i < j;
217 }
218 const std::vector<KeyPoint>* kp;
219};

221void KeyPointsFilter::removeDuplicated( std::vector<KeyPoint>& keypoints )
222{
223 int i, j, n = (int)keypoints.size();
224 AutoBuffer<int> kpidx(n);
225 AutoBuffer<uchar> mask(n, (uchar)1);
226
227 for( i = 0; i < n; i++ )
228 kpidx[i] = i;
229 std::sort(kpidx.begin(), kpidx.end(), KeyPoint_LessThan(keypoints));
230 for( i = 1, j = 0; i < n; i++ )
231 {
232 KeyPoint& kp1 = keypoints[kpidx[i]];
233 KeyPoint& kp2 = keypoints[kpidx[j]];
234 if( kp1.pt.x != kp2.pt.x || kp1.pt.y != kp2.pt.y ||
235 kp1.size != kp2.size || kp1.angle != kp2.angle )
236 j = i;
237 else
238 mask[kpidx[i]] = 0;
239 }
240
241 for( i = j = 0; i < n; i++ )
242 {
243 if( mask[i] )
244 {
245 if( i != j )
246 keypoints[j] = keypoints[i];
247 j++;
248 }
249 }
250 keypoints.resize(j);
251}

273void KeyPointsFilter::removeDuplicatedSorted( std::vector<KeyPoint>& keypoints )
274{
275 int i, j, n = (int)keypoints.size();
276
277 if (n < 2) return;
278
279 std::sort(keypoints.begin(), keypoints.end(), KeyPoint12_LessThan());
280
281 for( i = 0, j = 1; j < n; ++j )
282 {
283 const KeyPoint& kp1 = keypoints[i];
284 const KeyPoint& kp2 = keypoints[j];
285 if( kp1.pt.x != kp2.pt.x || kp1.pt.y != kp2.pt.y ||
286 kp1.size != kp2.size || kp1.angle != kp2.angle ) {
287 keypoints[++i] = keypoints[j];
288 }
289 }
290 keypoints.resize(i + 1);
291}
