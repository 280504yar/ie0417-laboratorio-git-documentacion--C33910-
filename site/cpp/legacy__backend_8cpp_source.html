52Ptr<BackendWrapper> wrapMat(int backendId, int targetId, cv::Mat& m)
53{
54 if (backendId == DNN_BACKEND_OPENCV)
55 {
56 if (targetId == DNN_TARGET_CPU)
57 return Ptr<BackendWrapper>();
58#ifdef HAVE_OPENCL
59 else if (IS_DNN_OPENCL_TARGET(targetId))
60 return OpenCLBackendWrapper::create(m);
61#endif
62 else
63 CV_Error(Error::StsNotImplemented, "Unknown/unsupported target identifier");
64 }
65 else if (backendId == DNN_BACKEND_INFERENCE_ENGINE_NN_BUILDER_2019)
66 {
67 CV_ERROR_DNN_BACKEND_INFERENCE_ENGINE_NN_BUILDER_2019;
68 }
69 else if (backendId == DNN_BACKEND_INFERENCE_ENGINE_NGRAPH)
70 {
71 CV_Assert(0 && "Internal error: DNN_BACKEND_INFERENCE_ENGINE_NGRAPH must be implemented through inheritance");
72 }
73 else if (backendId == DNN_BACKEND_WEBNN)
74 {
75#ifdef HAVE_WEBNN
76 return Ptr<BackendWrapper>(new WebnnBackendWrapper(targetId, m));
77#else
78 CV_Error(Error::StsNotImplemented, "This OpenCV version is built without support of WebNN");
79#endif
80 }
81 else if (backendId == DNN_BACKEND_VKCOM)
82 {
83 CV_Assert(haveVulkan());
84#ifdef HAVE_VULKAN
85 return Ptr<BackendWrapper>(new VkComBackendWrapper(m));
86#endif // HAVE_VULKAN
87 }
88 else if (backendId == DNN_BACKEND_CUDA)
89 {
90 CV_Assert(haveCUDA());
91
92#ifdef HAVE_CUDA
93 CV_CheckType(m.depth(), m.depth() == CV_32F || m.depth() == CV_8S || m.depth() == CV_8U || m.depth() == CV_32S || m.depth() == CV_64S || m.depth() == CV_Bool, "Unsupported type for CUDA");
94 CV_Assert(IS_DNN_CUDA_TARGET(targetId));
95 switch (m.depth())
96 {
97 case CV_32F:
98 if (targetId == DNN_TARGET_CUDA_FP16)
99 return CUDABackendWrapperFP16::create(m);
100 else
101 return CUDABackendWrapperFP32::create(m);
102 case CV_8S:
103 return CUDABackendWrapperINT8::create(m);
104 case CV_8U:
105 return CUDABackendWrapperUINT8::create(m);
106 case CV_32S:
107 return CUDABackendWrapperINT32::create(m);
108 case CV_64S:
109 return CUDABackendWrapperINT64::create(m);
110 case CV_Bool:
111 return CUDABackendWrapperBOOL::create(m);
112 default:
113 CV_Error(Error::BadDepth, "Unsupported mat type for CUDA");
114 }
115#endif
116 }
117 else if (backendId == DNN_BACKEND_TIMVX)
118 {
119 CV_Assert(haveTimVX());
120#ifdef HAVE_TIMVX
121 return Ptr<BackendWrapper>(new TimVXBackendWrapper(m));
122#endif // HAVE_TIMVX
123 }
124 else if (backendId == DNN_BACKEND_CANN)
125 {
126 CV_Assert(0 && "Internal error: DNN_BACKEND_CANN must be implemented through inheritance");
127 }
128 else
129 CV_Error(Error::StsNotImplemented, "Unknown backend identifier");
130 return Ptr<BackendWrapper>(); // TODO Error?
131} // wrapMat()
