64bool CCheckerDetectorImpl::
65 _no_net_process(InputArray image, const int nc,
66 std::vector<Rect> regionsOfInterest)
67{
68 m_checkers.clear();
69
70 Mat img = image.getMat();
71 for (const Rect &region : regionsOfInterest)
72 {
73 //-------------------------------------------------------------------
74 // Run the model to find good regions
75 //-------------------------------------------------------------------
76 Mat croppedImage = img(region);
77#ifdef MCC_DEBUG
78 std::string pathOut = "./";
79#endif
80 //-------------------------------------------------------------------
81 // prepare image
82 //-------------------------------------------------------------------
83
84 Mat img_bgr, img_gray;
85 float asp;
86 prepareImage(croppedImage, img_gray, img_bgr, asp);
87
88#ifdef MCC_DEBUG
89 showAndSave("prepare_image", img_gray, pathOut);
90#endif
91 //-------------------------------------------------------------------
92 // thresholding
93 //-------------------------------------------------------------------
94 std::vector<Mat> img_bw;
95 performThreshold(img_gray, img_bw);
96
97 Mat3f img_rgb_f(img_bgr);
98 cvtColor(img_rgb_f, img_rgb_f, COLOR_BGR2RGB);
99 img_rgb_f /= 255;
100
101 Mat img_rgb_org, img_ycbcr_org;
102 std::vector<Mat> rgb_planes(3), ycbcr_planes(3);
103
104 // Convert to RGB and YCbCr space
105 cvtColor(croppedImage, img_rgb_org, COLOR_BGR2RGB);
106 cvtColor(croppedImage, img_ycbcr_org, COLOR_BGR2YCrCb);
107
108 // Get chanels
109 split(img_rgb_org, rgb_planes);
110 split(img_ycbcr_org, ycbcr_planes);
111 Mutex mtx;
112 parallel_for_(
113 Range(0, (int)img_bw.size()), [&](const Range &range) {
114 const int begin = range.start;
115 const int end = range.end;
116 for (int i = begin; i < end; i++)
117 {
118
119#ifdef MCC_DEBUG
120 showAndSave("threshold_image", img_bw[i], pathOut);
121#endif
122 // find contour
123 //-------------------------------------------------------------------
124 ContoursVector contours;
125 findContours(img_bw[i], contours);
126
127 if (contours.empty())
128 continue;
129#ifdef MCC_DEBUG
130 Mat im_contour(img_bgr.size(), CV_8UC1);
131 im_contour = Scalar(0);
132 drawContours(im_contour, contours, -1, Scalar(255), 2, LINE_AA);
133 showAndSave("find_contour", im_contour, pathOut);
134#endif
135 //-------------------------------------------------------------------
136 // find candidate
137 //-------------------------------------------------------------------
138
139 std::vector<CChart> detectedCharts;
140 findCandidates(contours, detectedCharts);
141
142 if (detectedCharts.empty())
143 continue;
144
145#ifdef MCC_DEBUG
146 Mat img_chart;
147 img_bgr.copyTo(img_chart);
148
149 for (size_t ind = 0; ind < detectedCharts.size(); ind++)
150 {
151
152 CChartDraw chrtdrw((detectedCharts[ind]), img_chart);
153 chrtdrw.drawCenter();
154 chrtdrw.drawContour();
155 }
156 showAndSave("find_candidate", img_chart, pathOut);
157#endif
158 //-------------------------------------------------------------------
159 // clusters analysis
160 //-------------------------------------------------------------------
161
162 std::vector<int> G;
163 clustersAnalysis(detectedCharts, G);
164
165 if (G.empty())
166 continue;
167
168#ifdef MCC_DEBUG
169 Mat im_gru;
170 img_bgr.copyTo(im_gru);
171 RNG rng(0xFFFFFFFF);
172 int radius = 10, thickness = -1;
173
174 std::vector<int> g;
175 unique(G, g);
176 size_t Nc = g.size();
177 std::vector<Scalar> colors(Nc);
178 for (size_t ind = 0; ind < Nc; ind++)
179 colors[ind] = randomcolor(rng);
180
181 for (size_t ind = 0; ind < detectedCharts.size(); ind++)
182 circle(im_gru, detectedCharts[ind].center, radius, colors[G[ind]],
183 thickness);
184 showAndSave("clusters_analysis", im_gru, pathOut);
185#endif
186 //-------------------------------------------------------------------
187 // checker color recognize
188 //-------------------------------------------------------------------
189
190 std::vector<std::vector<Point2f>> colorCharts;
191 checkerRecognize(img_bgr, detectedCharts, G, colorCharts);
192
193 if (colorCharts.empty())
194 continue;
195
196#ifdef MCC_DEBUG
197 Mat image_box;
198 img_bgr.copyTo(image_box);
199 for (size_t ind = 0; ind < colorCharts.size(); ind++)
200 {
201 std::vector<Point2f> ibox = colorCharts[ind];
202 Scalar color_box = CV_RGB(0, 0, 255);
203 int thickness_box = 2;
204 line(image_box, ibox[0], ibox[1], color_box, thickness_box, LINE_AA);
205 line(image_box, ibox[1], ibox[2], color_box, thickness_box, LINE_AA);
206 line(image_box, ibox[2], ibox[3], color_box, thickness_box, LINE_AA);
207 line(image_box, ibox[3], ibox[0], color_box, thickness_box, LINE_AA);
208 }
209 showAndSave("checker_recognition", image_box, pathOut);
210#endif
211 //-------------------------------------------------------------------
212 // checker color analysis
213 //-------------------------------------------------------------------
214 std::vector<Ptr<CChecker>> checkers;
215 Point2f total_offset = static_cast<Point2f>(region.tl());
216 checkerAnalysis(img_rgb_f, nc, colorCharts, checkers, asp,
217 img_rgb_org, img_ycbcr_org, rgb_planes, ycbcr_planes,
218 total_offset);
219
220#ifdef MCC_DEBUG
221 Mat image_checker;
222 croppedImage.copyTo(image_checker);
223 draw(checkers, image_checker);
224 showAndSave("checker_analysis", image_checker, pathOut);
225#endif
226 for (Ptr<CChecker> checker : checkers)
227 {
228 AutoLock lock(mtx);
229 m_checkers.push_back(checker);
230 }
231 }
232#ifdef MCC_DEBUG
233 },
234 1); //Run only one thread in debug mode
235#else
236 });
237#endif
238 }
239 //remove too close detections
240 removeTooCloseDetections();
241 m_checkers.resize(min(nc, (int)m_checkers.size()));
242 return !m_checkers.empty();

464

510void CCheckerDetectorImpl::
511 prepareImage(InputArray bgr, OutputArray grayOut,
512 OutputArray bgrOut, float &aspOut) const
513{
514
515 int min_size;
516 Size size = bgr.size();
517 aspOut = 1;
518 bgr.copyTo(bgrOut);
519
520 // Resize image
521 min_size = std::min(size.width, size.height);
522 if (m_params.minImageSize > min_size)
523 {
524 aspOut = (float)m_params.minImageSize / min_size;
525 resize(bgr, bgrOut, Size(int(size.width * aspOut), int(size.height * aspOut)), INTER_LINEAR_EXACT);
526 }
527
528 // Convert to grayscale
529 cvtColor(bgrOut, grayOut, COLOR_BGR2GRAY);
530
531 // PDiamel: wiener adaptative methods to minimize the noise effets
532 // by illumination
533
534 CWienerFilter filter;
535 filter.wiener2(grayOut, grayOut, 5, 5);
536
537 //JLeandro: perform morphological open on the equalized image
538 //to minimize the noise effects by CLAHE and to even intensities
539 //inside the MCC patches (regions)
540
541 Mat strelbox = getStructuringElement(MORPH_RECT, Size(5, 5));
542 morphologyEx(grayOut, grayOut, MORPH_OPEN, strelbox);
543}

545void CCheckerDetectorImpl::
546 performThreshold(InputArray grayscaleImg,
547 OutputArrayOfArrays thresholdImgs) const
548{
549 // number of window sizes (scales) to apply adaptive thresholding
550 int nScales = (m_params.adaptiveThreshWinSizeMax - m_params.adaptiveThreshWinSizeMin) / m_params.adaptiveThreshWinSizeStep + 1;
551 thresholdImgs.create(nScales, 1, CV_8U);
552 std::vector<Mat> _thresholdImgs(nScales);
553 parallel_for_(Range(0, nScales),[&](const Range& range) {
554 const int start = range.start;
555 const int end = range.end;
556 for (int i = start; i < end; i++) {
557 int currScale = m_params.adaptiveThreshWinSizeMin + i * m_params.adaptiveThreshWinSizeStep;
558 Mat tempThresholdImg;
559 adaptiveThreshold(grayscaleImg, tempThresholdImg, 255, ADAPTIVE_THRESH_MEAN_C,
560 THRESH_BINARY_INV, currScale, m_params.adaptiveThreshConstant);
561 _thresholdImgs[i] = tempThresholdImg;
562 }
563 });
564
565 thresholdImgs.assign(_thresholdImgs);
566}

568void CCheckerDetectorImpl::
569 findContours(
570 InputArray srcImg,
571 ContoursVector &contours) const
572{
573 // contour detected
574 // [Suzuki85] Suzuki, S. and Abe, K., Topological Structural Analysis of Digitized
575 // Binary Images by Border Following. CVGIP 30 1, pp 32-46 (1985)
576 ContoursVector allContours;
577 cv::findContours(srcImg, allContours, RETR_LIST, CHAIN_APPROX_NONE);
578
579 //select contours
580 contours.clear();
581
582 for (size_t i = 0; i < allContours.size(); i++)
583 {
584
585 PointsVector contour;
586 contour = allContours[i];
587
588 int contourSize = (int)contour.size();
589 if (contourSize <= m_params.minContourPointsAllowed)
590 continue;
591
592 double area = contourArea(contour);
593
594 if (area < m_params.minContoursArea)
595 continue;
596
597 PointsVector hull;
598 convexHull(contour, hull);
599 double area_hull = contourArea(hull);
600 double S = area / area_hull;
601 if (S < m_params.minContourSolidity)
602 continue;
603
604 contours.push_back(allContours[i]);
605 }
606}

608void CCheckerDetectorImpl::
609 findCandidates(
610 const ContoursVector &contours,
611 std::vector<CChart> &detectedCharts)
612{
613 std::vector<Point> approxCurve;
614 std::vector<CChart> possibleCharts;
615
616 // For each contour, analyze if it is a parallelepiped likely to be the chart
617 for (size_t i = 0; i < contours.size(); i++)
618 {
619 // Approximate to a polygon
620 // It uses the Douglas-Peucker algorithm
621 // http://en.wikipedia.org/wiki/Ramer-Douglas-Peucker_algorithm
622 double eps = contours[i].size() * m_params.findCandidatesApproxPolyDPEpsMultiplier;
623 approxPolyDP(contours[i], approxCurve, eps, true);
624
625 // We interested only in polygons that contains only four points
626 if (approxCurve.size() != 4)
627 continue;
628
629 // And they have to be convex
630 if (!isContourConvex(approxCurve))
631 continue;
632
633 // Ensure that the distance between consecutive points is large enough
634 float minDist = INFINITY;
635
636 for (size_t j = 0; j < 4; j++)
637 {
638 Point side = approxCurve[j] - approxCurve[(j + 1) % 4];
639 float squaredSideLength = (float)side.dot(side);
640 minDist = std::min(minDist, squaredSideLength);
641 }
642
643 // Check that distance is not very small
644 if (minDist < m_params.minContourLengthAllowed)
645 continue;
646
647 // All tests are passed. Save chart candidate:
648 CChart chart;
649
650 std::vector<Point2f> corners(4);
651 for (int j = 0; j < 4; j++)
652 corners[j] = Point2f((float)approxCurve[j].x, (float)approxCurve[j].y);
653 chart.setCorners(corners);
654
655 possibleCharts.push_back(chart);
656 }
657
658 // Remove these elements which corners are too close to each other.
659 // Eliminate overlaps!!!
660 // First detect candidates for removal:
661 std::vector<std::pair<int, int>> tooNearCandidates;
662 for (int i = 0; i < (int)possibleCharts.size(); i++)
663 {
664 const CChart &m1 = possibleCharts[i];
665
666 //calculate the average distance of each corner to the nearest corner of the other chart candidate
667 for (int j = i + 1; j < (int)possibleCharts.size(); j++)
668 {
669 const CChart &m2 = possibleCharts[j];
670
671 float distSquared = 0;
672
673 for (int c = 0; c < 4; c++)
674 {
675 Point v = m1.corners[c] - m2.corners[c];
676 distSquared += v.dot(v);
677 }
678
679 distSquared /= 4;
680
681 if (distSquared < m_params.minInterContourDistance)
682 {
683 tooNearCandidates.push_back(std::pair<int, int>(i, j));
684 }
685 }
686 }
687
688 // Mark for removal the element of the pair with smaller perimeter
689 std::vector<bool> removalMask(possibleCharts.size(), false);
690
691 for (size_t i = 0; i < tooNearCandidates.size(); i++)
692 {
693 float p1 = perimeter(possibleCharts[tooNearCandidates[i].first].corners);
694 float p2 = perimeter(possibleCharts[tooNearCandidates[i].second].corners);
695
696 size_t removalIndex;
697 if (p1 > p2)
698 removalIndex = tooNearCandidates[i].second;
699 else
700 removalIndex = tooNearCandidates[i].first;
701
702 removalMask[removalIndex] = true;
703 }
704
705 detectedCharts.clear();
706 for (size_t i = 0; i < possibleCharts.size(); i++)
707 {
708 if (removalMask[i])
709 continue;
710 detectedCharts.push_back(possibleCharts[i]);
711 }
712}

714void CCheckerDetectorImpl::
715 clustersAnalysis(
716 const std::vector<CChart> &detectedCharts,
717 std::vector<int> &groups)
718{
719 size_t N = detectedCharts.size();
720 std::vector<Point> X(N);
721 std::vector<double> B0(N), W(N);
722 std::vector<int> G;
723
724 CChart chart;
725 double b0;
726 for (size_t i = 0; i < N; i++)
727 {
728 chart = detectedCharts[i];
729 b0 = chart.large_side * m_params.B0factor;
730 X[i] = chart.center;
731 W[i] = chart.area;
732 B0[i] = b0;
733 }
734

746 InputArray img,

748 const std::vector<int> &G,
749 std::vector<std::vector<Point2f>> &colorChartsOut)
750{
751 std::vector<int> gU;
752 unique(G, gU);
753 size_t Nc = gU.size(); //numero de grupos
754 size_t Ncc = detectedCharts.size(); //numero de charts
755

757
758 for (size_t g = 0; g < Nc; g++)
759 {
760
763
764 std::vector<CChart> chartSub;
765 for (size_t i = 0; i < Ncc; i++)
766 if (G[i] == (int)g)
767 chartSub.push_back(detectedCharts[i]);

770 if (static_cast<int>(Nsc) < m_params.minGroupSize)
771 continue;
772
775
776 CBoundMin bm;
777 std::vector<Point2f> points;
778
779 bm.setCharts(chartSub);
780 bm.calculate();
781 bm.getCorners(points);
782
783 // boundary condition
784 if (points.size() == 0)
785 continue;
786
787 // sort the points in anti-clockwise order
788 polyanticlockwise(points);
789
792
793 // get physical char box model
794 std::vector<Point2f> chartPhy;
795 get_subbox_chart_physical(points, chartPhy);
796
797 // Find the perspective transformation that brings current chart to rectangular form
798 Matx33f ccT = getPerspectiveTransform(points, chartPhy);
799
800 // transformer
801 std::vector<Point2f> c(Nsc), ct;
802 std::vector<Point2f> ch(4 * Nsc), cht;
803
804 for (size_t i = 0; i < Nsc; i++)
805 {
806
807 CChart cc = chartSub[i];
808 for (size_t j = 0; j < 4; j++)
809 ch[i * 4 + j] = cc.corners[j];
810 c[i] = chartSub[i].center;
811 }
812
813 transform_points_forward(ccT, c, ct);
814 transform_points_forward(ccT, ch, cht);
815
816 float wchart = 0, hchart = 0;
817 std::vector<float> cx(Nsc), cy(Nsc);
818 for (size_t i = 0, k = 0; i < Nsc; i++)
819 {
820 k = i * 4;
821 Point2f v1 = cht[k + 1] - cht[k + 0];
822 Point2f v2 = cht[k + 3] - cht[k + 0];
823 wchart += (float)norm(v1);
824 hchart += (float)norm(v2);
825 cx[i] = ct[i].x;
826 cy[i] = ct[i].y;
827 }
828
829 wchart /= Nsc;
830 hchart /= Nsc;
831
834
835 float tolx = wchart / 2, toly = hchart / 2;
836 std::vector<float> cxr, cyr;
837 reduce_array(cx, cxr, tolx);
838 reduce_array(cy, cyr, toly);
839
840 if (cxr.size() == 1 || cyr.size() == 1) //no information can be extracted if \
841 //only one row or columns in present
842 continue;
843 // color and center rectificate
844 Size2i colorSize = Size2i((int)cxr.size(), (int)cyr.size());
845 Mat colorMat(colorSize, CV_32FC3);
846 std::vector<Point2f> cte(colorSize.area());
847
848 int k = 0;
849
850 for (int i = 0; i < colorSize.height; i++)
851 {
852 for (int j = 0; j < colorSize.width; j++)
853 {
854 Point2f vc = Point2f(cxr[j], cyr[i]);
855 cte[k] = vc;
856
857 // recovery color
858 Point2f cti;
859 Matx31f p, xt;
860
861 p(0, 0) = vc.x;
862 p(1, 0) = vc.y;
863 p(2, 0) = 1;
864 xt = ccT.inv() * p;
865 cti.x = xt(0, 0) / xt(2, 0);
866 cti.y = xt(1, 0) / xt(2, 0);
867
868 // color
869 int x, y;
870 x = (int)cti.x;
871 y = (int)cti.y;
872 Vec3f &srgb = colorMat.at<Vec3f>(i, j);
873 Vec3b rgb;

875 rgb = img.getMat().at<Vec3b>(y, x);
876
877 srgb[0] = (float)rgb[0] / 255;
878 srgb[1] = (float)rgb[1] / 255;
879 srgb[2] = (float)rgb[2] / 255;
880
881 k++;
882 }
883 }
884
885 CChartModel::SUBCCMModel scm;
886 scm.centers = cte;
887 scm.color_size = colorSize;
888 colorMat = colorMat.t();
889 scm.sub_chart = colorMat.reshape(3, colorSize.area());
890
892
893 // color chart model
894 CChartModel cccm(m_chartType);
895
896 int iTheta; // rotation angle of chart
897 int offset; // offset
898 float error; // min error
899 if (!cccm.evaluate(scm, offset, iTheta, error))
900 continue;
901 if (iTheta >= 4)
902 cccm.flip();
903
904 for (int i = 0; i < iTheta % 4; i++)
905 cccm.rotate90();
906
909
910 Size2i dim = cccm.size;
911 std::vector<Point2f> center = cccm.center;
912 std::vector<Point2f> box = cccm.box;
913 int cols = dim.height - colorSize.width + 1;
914
915 int x = (offset) / cols;
916 int y = (offset) % cols;
917
918 // seleccionar sub grid centers of model
919 std::vector<Point2f> ctss(colorSize.area());
920 Point2f point_ac = Point2f(0, 0);
921 int p = 0;
922
923 for (int i = x; i < (x + colorSize.height); i++)
924 {
925 for (int j = y; j < (y + colorSize.width); j++)
926 {
927 int iter = i * dim.height + j;
928 ctss[p] = center[iter];
929 point_ac += ctss[p];
930 p++;
931 }
932 }
933 // is colineal point
934 if (point_ac.x == ctss[0].x * p || point_ac.y == ctss[0].y * p)
935 continue;
936 // Find the perspective transformation
937 Matx33f ccTe = findHomography(ctss, cte);
938
939 std::vector<Point2f> tbox, ibox;
940 transform_points_forward(ccTe, box, tbox);
941 transform_points_inverse(ccT, tbox, ibox);
942
943 // sort the points in anti-clockwise order
944 if (iTheta < 4)
945 mcc::polyanticlockwise(ibox);
946 else
947 mcc::polyclockwise(ibox);
948 colorCharts.push_back(ibox);
949 }
950
951 colorChartsOut = colorCharts;

1265void CCheckerDetectorImpl::
1266 removeTooCloseDetections()
1267{
1268 // Provide a deterministic output by sorting the checkers by their cost and
1269 // then by their center coordinates.
1270 std::sort(m_checkers.begin(), m_checkers.end(),
1271 [](const Ptr<CChecker> &a, const Ptr<CChecker> &b) {
1272 if (a->getCost() != b->getCost())
1273 return a->getCost() < b->getCost();
1274 Point2f ca = a->getCenter();
1275 Point2f cb = b->getCenter();
1276 if (ca.x != cb.x) return ca.x < cb.x;
1277 return ca.y < cb.y;
1278 });
1279
1280 // Remove these elements which corners are too close to each other.
1281 // Eliminate overlaps!!!
1282 // First detect candidates for removal:
1283 std::vector<std::pair<int, int>> tooNearCandidates;
1284 for (int i = 0; i < (int)m_checkers.size(); i++)
1285 {
1286 const Ptr<CChecker> &m1 = m_checkers[i];
1287
1288 //calculate the average distance of each corner to the nearest corner of the other chart candidate
1289 for (int j = i + 1; j < (int)m_checkers.size(); j++)
1290 {
1291 const Ptr<CChecker> &m2 = m_checkers[j];
1292
1293 float distSquared = 0;
1294
1295 for (int c = 0; c < 4; c++)
1296 {
1297 Point v = m1->getBox()[c] - m2->getBox()[c];
1298 distSquared += v.dot(v);
1299 }
1300
1301 distSquared /= 4;
1302
1303 if (distSquared < m_params.minInterCheckerDistance)
1304 {
1305 tooNearCandidates.push_back(std::pair<int, int>(i, j));
1306 }
1307 }
1308 }
1309
1310 // Mark for removal the element of the pair with smaller cost
1311 std::vector<bool> removalMask(m_checkers.size(), false);
1312
1313 for (size_t i = 0; i < tooNearCandidates.size(); i++)
1314 {
1315 float p1 = m_checkers[tooNearCandidates[i].first]->getCost();
1316 float p2 = m_checkers[tooNearCandidates[i].second]->getCost();
1317
1318 size_t removalIndex;
1319 if (p1 < p2)
1320 removalIndex = tooNearCandidates[i].second;
1321 else if (p2 < p1)

1372 // sort array

1375 // label array

1380 // diff array
1381 for (size_t i = 1; i < n; i++)
1382 label[i] += label[i - 1];

1389 nn = ulabel.size();

1392 {
1393 float mu = 0, s = 0;
1394 for (size_t j = 0; j < n; j++)
1395 {
1396 mu += (label[j] == ulabel[i]) * xx[j];
1397 s += (label[j] == ulabel[i]);
1398 }
1399 x_new[i] = mu / s;
1400 }
1401
1402 // diff array
1403 std::vector<float> dif(nn - 1);
1404 for (size_t i = 0; i < nn - 1; i++)
1405 dif[i] = (x_new[(i + 1) % nn] - x_new[i]);
1406
1407 // max and idx
1408 float fmax = 0;
1409 size_t idx = 0;
1410 for (size_t i = 0; i < nn - 1; i++)
1411 if (fmax < dif[i])
1412 {
1413 fmax = dif[i];
1414 idx = i;
1415 }
1416
1417 // add ... X[i] MAX X[i+] ...
1418 if (fmax > 4 * tol)
1419 x_new.insert(x_new.begin() + idx + 1, (x_new[idx] + x_new[idx + 1]) / 2);
