188void Net::Impl::setUpNet(const std::vector<LayerPin>& blobsToKeep_)
189{
190 CV_TRACE_FUNCTION();
191
192 if (dumpLevel && networkDumpCounter == 0)
193 {
194 dumpNetworkToFile();
195 }
196
197 validateBackendAndTarget();
198
199 if (!netWasAllocated || this->blobsToKeep != blobsToKeep_)
200 {
201 if (preferableBackend == DNN_BACKEND_OPENCV && IS_DNN_OPENCL_TARGET(preferableTarget))
202#ifndef HAVE_OPENCL
203 {
204 CV_LOG_WARNING(NULL, "DNN: OpenCL target is not available in this OpenCV build, switching to CPU.");
205 preferableTarget = DNN_TARGET_CPU;
206 }
207#else
208 {
209 if (!getParam_DNN_OPENCL_ALLOW_ALL_DEVICES())
210 {
211 // Current implementation is only valid for GPU (#11494)
212 if (ocl::Device::getDefault().type() != ocl::Device::TYPE_GPU)
213 {
214 CV_LOG_WARNING(NULL, "DNN: OpenCL target is not supported with current OpenCL device (tested with GPUs only), switching to CPU.");
215 preferableTarget = DNN_TARGET_CPU;
216 }
217 else if (preferableTarget == DNN_TARGET_OPENCL_FP16 && !ocl::Device::getDefault().isIntel())
218 {
219 CV_LOG_WARNING(NULL,
220 "DNN: OpenCL target with fp16 precision is not supported "
221 "with current OpenCL device (tested with Intel GPUs only), "
222 "switching to OpenCL with fp32 precision.");
223 preferableTarget = DNN_TARGET_OPENCL;
224 }
225 }
226 }
227#endif
228 if (preferableBackend == DNN_BACKEND_VKCOM && !haveVulkan())
229 {
230 preferableBackend = DNN_BACKEND_OPENCV;
231 preferableTarget = DNN_TARGET_CPU;
232 }
233
234 if (preferableBackend == DNN_BACKEND_CUDA && !haveCUDA())
235 {
236#ifdef HAVE_CUDA
237 CV_LOG_WARNING(NULL, "unable to use CUDA backend; switching to CPU");
238#else
239 CV_LOG_WARNING(NULL, "DNN module was not built with CUDA backend; switching to CPU");
240#endif
241 preferableBackend = DNN_BACKEND_OPENCV;
242 preferableTarget = DNN_TARGET_CPU;
243 }
244
245 if (preferableBackend == DNN_BACKEND_TIMVX && !haveTimVX())
246 {
247 preferableBackend = DNN_BACKEND_OPENCV;
248 preferableTarget = DNN_TARGET_CPU;
249 }
250
251 clear();
252
253 this->blobsToKeep = blobsToKeep_;
254
255 allocateLayers(blobsToKeep_);
256
257 MapIdToLayerData::iterator it = layers.find(0);
258 CV_Assert(it != layers.end());
259 it->second.skip = netInputLayer->skip;
260
261 initBackend(blobsToKeep_);
262
263 netWasAllocated = true;
264
265 if (dumpLevel)
266 {
267 dumpNetworkToFile();
268 }
269 }

285 }
286 CV_Error_(Error::StsObjectNotFound, ("layer #%d is not found", layerId));
287 } else {

289 return getLayerInstance(ld);

425int Net::Impl::addLayer(const String& name, const String& type, const int& dtype, LayerParams& params)
426{
427 int id = getLayerId(name);
428 if (id >= 0)
429 {
430 if (!DNN_DIAGNOSTICS_RUN || type != "NotImplemented")
431 {
432 CV_Error(Error::StsBadArg, "Layer \"" + name + "\" has been already added into net");
433 return -1;
434 }
435 else
436 {
437 LayerData& ld = layers.find(id)->second;
438 ld.type = type;
439 ld.params = params;
440 return -1;
441 }
442 }
443
444 id = ++lastLayerId;
445 layerNameToId.insert(std::make_pair(name, id));
446 layers.insert(std::make_pair(id, LayerData(id, name, type, dtype, params)));
447 if (params.get<bool>("has_dynamic_shapes", false))
448 hasDynamicShapes = true;
449
450 if (dtype == CV_8S)
451 netWasQuantized = true;
452
453 return id;
454}

513void Net::Impl::allocateLayer(int lid, const LayersShapesMap& layersShapes)
514{
515 CV_TRACE_FUNCTION();
516
517 LayerData& ld = layers[lid];
518
519 // already allocated
520 if (ld.flag)
521 return;
522
523 size_t ninputs = ld.inputBlobsId.size();
524#if 0
525 printf("layer %s:", ld.name.c_str());
526 for (size_t i = 0; i < ninputs; i++)
527 {
528 int inp_lid = ld.inputBlobsId[i].lid;
529 LayerData &inp_ld = layers[inp_lid];
530 int inp_outputs = (int)inp_ld.outputBlobs.size();
531 std::cout << " " << inp_ld.name << "(" << inp_outputs;
532
533 for( int j = 0; j < inp_outputs; j++ )
534 {
535 std::cout << (j == 0 ? ": " : ", ") << inp_ld.outputBlobs[j].size;
536 }
537 std::cout << ")";
538 }
539 printf("\n");
540#endif
541
542 // determine parent layers
543 for (size_t i = 0; i < ninputs; i++)
544 ld.inputLayersId.insert(ld.inputBlobsId[i].lid);
545
546 // allocate parents
547 for (std::set<int>::const_iterator i = ld.inputLayersId.begin(); i != ld.inputLayersId.end(); i++)
548 allocateLayer(*i, layersShapes);
549
550 // bind inputs for DataLayer
551 if (ld.id == 0 && netInputLayer->supportBackend(preferableBackend))
552 {
553 ninputs = netInputLayer->inputsData.size();
554 ld.inputBlobsWrappers.resize(ninputs);
555 for (size_t i = 0; i < ninputs; i++)
556 ld.inputBlobsWrappers[i] = wrap(netInputLayer->inputsData[i]);
557 }
558 else
559 {
560 ld.inputBlobs.resize(ninputs);
561 ld.inputBlobsWrappers.resize(ninputs);
562 for (size_t i = 0; i < ninputs; i++)
563 {
564 LayerPin from = ld.inputBlobsId[i];
565 CV_Assert(from.valid());
566 CV_DbgAssert(layers.count(from.lid) && (int)layers[from.lid].outputBlobs.size() > from.oid);
567 ld.inputBlobs[i] = &layers[from.lid].outputBlobs[from.oid];
568 ld.inputBlobsWrappers[i] = layers[from.lid].outputBlobsWrappers[from.oid];
569 }
570 }
571
572 LayersShapesMap::const_iterator layerShapesIt = layersShapes.find(lid);
573
574 CV_Assert(layerShapesIt != layersShapes.end());
575
576 if (preferableBackend == DNN_BACKEND_OPENCV && ld.dtype == CV_32F
577 && preferableTarget == DNN_TARGET_OPENCL_FP16)
578 ld.dtype = CV_16F;
579
580 std::vector<LayerPin> pinsForInternalBlobs;
581 blobManager.allocateBlobsForLayer(ld, layerShapesIt->second, pinsForInternalBlobs);
582 ld.outputBlobsWrappers.resize(ld.outputBlobs.size());
583 for (int i = 0; i < ld.outputBlobs.size(); ++i)
584 ld.outputBlobsWrappers[i] = wrap(ld.outputBlobs[i]);
585
586 /* CUDA & CANN backend has its own system for internal blobs; we don't need these */
587 ld.internalBlobsWrappers.resize((preferableBackend == DNN_BACKEND_CUDA || preferableBackend == DNN_BACKEND_TIMVX || preferableBackend == DNN_BACKEND_CANN) ? 0 : ld.internals.size());
588 for (int i = 0; i < ld.internalBlobsWrappers.size(); ++i)
589 ld.internalBlobsWrappers[i] = wrap(ld.internals[i]);
590
591 Ptr<Layer> layerPtr = getLayerInstance(ld);
592 {
593 std::vector<Mat> inps(ld.inputBlobs.size());
594 for (int i = 0; i < ld.inputBlobs.size(); ++i)
595 {
596 inps[i] = *ld.inputBlobs[i];
597 }
598 layerPtr->finalize(inps, ld.outputBlobs);
599#if 0
600 std::cout << "\toutputs:";
601 size_t noutputs = ld.outputBlobs.size();
602 for (size_t j = 0; j < noutputs; j++)
603 {
604 std::cout << (j == 0 ? " " : ", ") << ld.outputBlobs[j].size;
605 }
606 std::cout << "\n";
607#endif
608 }
609
610 // After allocation of layer, we decrease counters to it's input blobs.
611 blobManager.releaseReferences(ld.inputBlobsId);
612 blobManager.releaseReferences(pinsForInternalBlobs);
613
614 ld.flag = 1;
615}

618void Net::Impl::allocateLayers(const std::vector<LayerPin>& blobsToKeep_)
619{
620 CV_TRACE_FUNCTION();
621
622 for (MapIdToLayerData::iterator it = layers.begin(); it != layers.end(); it++)
623 it->second.flag = 0;
624
625 CV_Assert(!layers[0].outputBlobs.empty());
626 ShapesVec inputShapes;
627 TypesVec inputTypes;
628 for (int i = 0; i < layers[0].outputBlobs.size(); i++)
629 {
630 Mat& inp = layers[0].outputBlobs[i];
631 CV_Assert(inp.total());
632 int type = inp.type();
633 if (type == CV_32F)
634 {
635 type = CV_32F;
636 if (preferableBackend == DNN_BACKEND_OPENCV &&
637 preferableTarget == DNN_TARGET_OPENCL_FP16)
638 {
639 type = CV_16F;
640 if (layers[0].dtype == CV_32F)
641 layers[0].outputBlobs[i].create(inp.size, CV_16F);
642 }
643 }
644 inputShapes.push_back(shape(inp));
645 inputTypes.push_back(type);
646 }
647
648 for (auto& layer : layers)
649 {
650 auto& ld = layer.second;
651 Ptr<Layer> layerPtr = getLayerInstance(ld);
652 layerPtr->preferableTarget = preferableTarget;
653 }
654
655 LayersShapesMap layersShapes;
656 getLayersShapes(inputShapes, inputTypes, layersShapes);
657
658 blobManager.reset();
659 backendWrappers.clear();
660
661 for (auto& layer : layers)
662 {
663 auto& ld = layer.second;
664 ld.inputBlobsWrappers.clear();
665 ld.outputBlobsWrappers.clear();
666 ld.internalBlobsWrappers.clear();
667 }
668
669 // Fake references to input blobs.
670 for (int i = 0; i < layers[0].outputBlobs.size(); ++i)
671 blobManager.addReference(LayerPin(0, i));
672 for (MapIdToLayerData::const_iterator it = layers.begin(); it != layers.end(); ++it)
673 {
674 const LayerData& ld = it->second;
675 blobManager.addReferences(ld.inputBlobsId);
676 }
677
678 for (int i = 0; i < blobsToKeep_.size(); i++)
679 {
680 blobManager.addReference(blobsToKeep_[i]);
681 }
682
683 for (MapIdToLayerData::const_iterator it = layers.begin(); it != layers.end(); it++)
684 {
685 int lid = it->first;
686 allocateLayer(lid, layersShapes);
687 }
688
689 layersTimings.resize(lastLayerId + 1, 0);
690 fuseLayers(blobsToKeep_);
691}

696void Net::Impl::forwardLayer(LayerData& ld)
697{
698 CV_TRACE_FUNCTION();
699
700 Ptr<Layer> layer = ld.layerInstance;
701
702#if TRACE_INFERENCE
703 if (layer) {
704 printf("------------------------------------------------\n");
705 printf("Running layer '%s' (%s)\n",
706 layer->name.c_str(),
707 layer->type.c_str());
708 }
709#endif
710
711 if (!ld.skip)
712 {
713 TickMeter tm;
714 tm.start();
715
716#ifndef HAVE_VULKAN
717 std::map<int, Ptr<BackendNode>>::const_iterator it = ld.backendNodes.find(preferableBackend);
718#else
719 std::map<int, Ptr<BackendNode>>::iterator it = ld.backendNodes.find(preferableBackend);
720#endif
721 if (preferableBackend == DNN_BACKEND_OPENCV || it == ld.backendNodes.end() || it->second.empty())
722 {
723 if (isAsync)
724 CV_Error(Error::StsNotImplemented, "Default implementation fallbacks in asynchronous mode");
725
726 if (!layer->supportBackend(DNN_BACKEND_OPENCV))
727 CV_Error(Error::StsNotImplemented, format("Layer \"%s\" of type \"%s\" unsupported on OpenCV backend",
728 ld.name.c_str(), ld.type.c_str()));

1239void Net::Impl::getLayerShapesRecursively(int id, LayersShapesMap& inOutShapes)
1240{
1241 CV_CheckGE(id, 0, "");
1242 CV_CheckLT(id, (int)layers.size(), "");
1243 LayerData& layerData = layers[id];
1244 std::vector<LayerPin>& inputLayerIds = layerData.inputBlobsId;
1245 LayerShapes& layerShapes = inOutShapes[id];
1246
1247 if (id == 0 && layerShapes.in[0].empty())
1248 {
1249 if (!layerData.outputBlobs.empty())
1250 {
1251 ShapesVec shapes;
1252 TypesVec types;
1253 for (int i = 0; i < layerData.outputBlobs.size(); i++)
1254 {
1255 Mat& inp = layerData.outputBlobs[i];
1256 CV_Assert(!inp.empty());
1257 shapes.push_back(shape(inp));
1258 types.push_back(inp.type());
1259 }
1260 layerShapes.in = shapes;
1261 layerShapes.inTypes = types;
1262 }
1263 else
1264 {
1265 const std::vector<MatShape>& inputShapes = netInputLayer->shapes;
1266 bool none = true;
1267 for (size_t i = 0; i < inputShapes.size(); i++)
1268 {
1269 if (!inputShapes[i].empty())
1270 {
1271 none = false;
1272 break;
1273 }
1274 }
1275 if (none)
1276 {
1277 layerShapes.out.clear();
1278 layerShapes.outTypes.clear();
1279 return;
1280 }
1281 else
1282 {
1283 layerShapes.in = inputShapes;
1284 layerShapes.inTypes.assign(inputShapes.size(), layerData.dtype);
1285 }
1286 }
1287 }
1288
1289 if (layerShapes.in.empty())
1290 {
1291 for (int i = 0; i < inputLayerIds.size(); i++)
1292 {
1293 int layerId = inputLayerIds[i].lid;
1294 LayersShapesMap::const_iterator it = inOutShapes.find(layerId);
1295 if (it == inOutShapes.end() || it->second.out.empty())
1296 {
1297 getLayerShapesRecursively(layerId, inOutShapes);
1298 it = inOutShapes.find(layerId);
1299 CV_Assert(it != inOutShapes.end());
1300 }
1301 const int out_port = inputLayerIds[i].oid;
1302 CV_CheckLT(out_port, (int)it->second.out.size(), "");
1303 const MatShape& shape = it->second.out[out_port];
1304 const MatType& type = it->second.outTypes[out_port];
1305 layerShapes.in.push_back(shape);
1306 layerShapes.inTypes.push_back(type);
1307 }
1308 }
1309 const ShapesVec& is = layerShapes.in;
1310 ShapesVec& os = layerShapes.out;
1311 ShapesVec& ints = layerShapes.internal;
1312 int requiredOutputs = layerData.requiredOutputs.size();
1313 const Ptr<Layer>& l = getLayerInstance(layerData);
1314 CV_Assert(l);
1315 bool layerSupportInPlace = false;
1316 try
1317 {
1318 l->updateMemoryShapes(layerShapes.in);
1319 layerSupportInPlace = l->getMemoryShapes(is, requiredOutputs, os, ints);
1320 l->getTypes(layerShapes.inTypes, os.size(), ints.size(), layerShapes.outTypes, layerShapes.internalTypes);
1321 CV_CheckEQ(layerShapes.out.size(), layerShapes.outTypes.size(), "Number of shapes and types should be equal");
1322 CV_CheckEQ(layerShapes.internal.size(), layerShapes.internalTypes.size(), "Number of shapes and types should be equal");
1323 }
1324 catch (const cv::Exception& e)
1325 {
1326 CV_LOG_ERROR(NULL, "OPENCV/DNN: [" << l->type << "]:(" << l->name << "): getMemoryShapes() throws exception." <<
1327 " inputs=" << is.size() <<
1328 " outputs=" << os.size() << "/" << requiredOutputs <<
1329 " blobs=" << l->blobs.size());
1330 for (size_t i = 0; i < is.size(); ++i)
1331 {
1332 CV_LOG_ERROR(NULL, " input[" << i << "] = " << toString(is[i]));
1333 }
1334 for (size_t i = 0; i < os.size(); ++i)
1335 {
1336 CV_LOG_ERROR(NULL, " output[" << i << "] = " << toString(os[i]));
1337 }
1338 for (size_t i = 0; i < l->blobs.size(); ++i)
1339 {
1340 CV_LOG_ERROR(NULL, " blobs[" << i << "] = " << typeToString(l->blobs[i].type()) << " " << toString(shape(l->blobs[i])));
1341 }
1342 CV_LOG_ERROR(NULL, "Exception message: " << e.what());
1343 throw;
1344 }
1345 layerShapes.supportInPlace = layerSupportInPlace;
1346
1347 try
1348 {
1349 for (int i = 0; i < (int)ints.size(); i++)
1350 CV_CheckGT(total(ints[i]), (size_t)0, "");
1351
1352 for (int i = 0; i < (int)os.size(); i++)
1353 CV_CheckGT(total(os[i]), (size_t)0, "");
1354 }
1355 catch (const cv::Exception& e)
1356 {
1357 CV_LOG_ERROR(NULL, "OPENCV/DNN: [" << l->type << "]:(" << l->name << "): getMemoryShapes() post validation failed." <<
1358 " inputs=" << is.size() <<
1359 " outputs=" << os.size() << "/" << requiredOutputs <<
1360 " blobs=" << l->blobs.size() <<

1440void Net::Impl::updateLayersShapes()
1441{
1442 CV_LOG_DEBUG(NULL, "updateLayersShapes() with layers.size=" << layers.size());
1443 CV_Assert(netInputLayer);
1444 DataLayer& inputLayer = *netInputLayer;
1445 LayerData& inputLayerData = layers[0];
1446 CV_Assert(inputLayerData.layerInstance.get() == &inputLayer);
1447 CV_Assert(!inputLayerData.outputBlobs.empty());
1448 ShapesVec inputShapes;
1449 TypesVec inputTypes;
1450 for (int i = 0; i < inputLayerData.outputBlobs.size(); i++)
1451 {
1452 Mat& inp = inputLayerData.outputBlobs[i];
1453 CV_Assert(!inp.empty());
1454 if (preferableBackend == DNN_BACKEND_OPENCV && // FIXIT: wrong place for output allocation
1455 preferableTarget == DNN_TARGET_OPENCL_FP16 &&
1456 inputLayerData.dtype == CV_32F)
1457 {
1458 inp.create(inp.size, CV_16F);
1459 }
1460 inputShapes.push_back(shape(inp));
1461 inputTypes.push_back(inp.type());
1462 }
1463 CV_LOG_DEBUG(NULL, toString(inputShapes, "Network input shapes"));
1464 LayersShapesMap layersShapes;
1465 layersShapes[0].in = inputShapes;
1466 layersShapes[0].inTypes = inputTypes;
1467 for (MapIdToLayerData::iterator it = layers.begin(); it != layers.end(); it++)
1468 {
1469 int layerId = it->first;
1470 LayerData& layerData = it->second;
1471 const std::vector<LayerPin>& inputLayerIds = layerData.inputBlobsId;
1472 LayerShapes& layerShapes = layersShapes[layerId];
1473 CV_LOG_DEBUG(NULL, "layer " << layerId << ": [" << layerData.type << "]:(" << layerData.name << ") with inputs.size=" << inputLayerIds.size());
1474 if (layerShapes.in.empty())
1475 {
1476 for (int i = 0; i < inputLayerIds.size(); i++)
1477 {
1478 const LayerPin& inputPin = inputLayerIds[i];
1479 int inputLayerId = inputPin.lid;
1480 CV_LOG_DEBUG(NULL, " input[" << i << "] " << inputLayerId << ":" << inputPin.oid << " as [" << layers[inputLayerId].type << "]:(" << layers[inputLayerId].name << ")");
1481 LayersShapesMap::const_iterator inputIt = layersShapes.find(inputLayerId);
1482 if (inputIt == layersShapes.end() || inputIt->second.out.empty())
1483 {
1484 getLayerShapesRecursively(inputLayerId, layersShapes);
1485 }
1486 const MatShape& shape = layersShapes[inputLayerId].out[inputPin.oid];
1487 const MatType& type = layersShapes[inputLayerId].outTypes[inputPin.oid];
1488 layerShapes.in.push_back(shape);
1489 layerShapes.inTypes.push_back(type);
1490 }
1491 getLayerInstance(layerData)->updateMemoryShapes(layerShapes.in);
1492 }
1493 CV_LOG_DEBUG(NULL, "Layer " << layerId << ": " << toString(layerShapes.in, "input shapes"));
1494 CV_LOG_IF_DEBUG(NULL, !layerShapes.out.empty(), "Layer " << layerId << ": " << toString(layerShapes.out, "output shapes"));
1495 CV_LOG_IF_DEBUG(NULL, !layerShapes.internal.empty(), "Layer " << layerId << ": " << toString(layerShapes.internal, "internal shapes"));
1496 }
1497 CV_LOG_DEBUG(NULL, "updateLayersShapes() - DONE");
1498}

1673{
1674 if (mainGraph) {
1675 auto it = argnames.find(outputTensorName);
1676 if (it == argnames.end()) {
1677 size_t excl = outputTensorName.rfind('!');
1678 if (excl != std::string::npos)
1679 it = argnames.find(outputTensorName.substr(excl + 1));
1680 }
1681 if (it == argnames.end()) {
1682 // Not a tensor name; try it as a layer name.
1683 int lid = getLayerId(outputTensorName);
1684 if (lid >= 0) {
1685 setParam(lid, numParam, blob);
1686 return;
1687 }
1688 CV_Error_(Error::StsObjectNotFound,
1689 ("DNN: tensor '%s' not found in the graph", outputTensorName.c_str()));
1690 }
1691
1692 int targetIdx = (int)it->second;
1693 const std::vector<Ptr<LayerInfo>>& prog = mainGraph->prog();
1694 for (const auto& layer : prog) {
1695 bool produces = false;
1696 for (const Arg& out : layer->outputs)
1697 if (out.idx == targetIdx) { produces = true; break; }
1698 if (!produces)
1699 continue;
1700
1701 if (numParam < (int)layer->blobs.size()) {
1702 layer->blobs[numParam] = blob;
1703 markLayerWeightsChanged(layer);
1704 return;
1705 }
1706
1707 Conv2Layer* conv = dynamic_cast<Conv2Layer*>(layer.get());
1708 if (conv && numParam == 0) {
1709 conv->setWeights(blob, Mat(), defaultC0, accuracy);
1710 markLayerWeightsChanged(layer);
1711 return;
1712 }
1713
1714 ConvTranspose2Layer* deconv = dynamic_cast<ConvTranspose2Layer*>(layer.get());
1715 if (deconv && numParam == 0) {
1716 deconv->setWeights(blob, Mat(), defaultC0, accuracy);
1717 markLayerWeightsChanged(layer);
1718 return;
1719 }
1720
1721 CV_Error_(Error::StsOutOfRange,
1722 ("DNN: op producing '%s' has fewer than %d params",
1723 outputTensorName.c_str(), numParam + 1));
1724 }
1725 CV_Error_(Error::StsObjectNotFound,
1726 ("DNN: no op found in graph producing tensor '%s'", outputTensorName.c_str()));
1727 }
1728 setParam(getLayerId(outputTensorName), numParam, blob);
1729}

2507int64 Net::Impl::getFLOPSGraph(const Ptr<Graph>& graph,
2508 const std::vector<MatShape>& shapeCache,
2509 const std::vector<MatType>& typeCache) const
2510{
2511 if (!graph)
2512 return 0;
2513
2514 int64 flops = 0;
2515 const std::vector<Ptr<LayerInfo>>& prog = graph->prog();
2516
2517 for (const Ptr<LayerInfo>& layer : prog) {
2518 if (!layer)
2519 continue;
2520
2521 const std::vector<Arg>& inputs = layer->inputs;
2522 const std::vector<Arg>& outputs = layer->outputs;
2523 int ninputs = (int)inputs.size();
2524 int noutputs = (int)outputs.size();
2525
2526 std::vector<MatShape> inpShapes(ninputs), outShapes(noutputs);
2527 for (int i = 0; i < ninputs; i++) {
2528 Arg inp = inputs[i];
2529 const ArgData& adata = args.at(inp.idx);
2530 if (adata.kind == DNN_ARG_CONST || adata.kind == DNN_ARG_EMPTY)
2531 inpShapes[i] = adata.shape;
2532 else
2533 inpShapes[i] = shapeCache[inp.idx];
2534 }
2535 for (int i = 0; i < noutputs; i++) {
2536 Arg out = outputs[i];
2537 if (out.idx > 0 && out.idx < (int)shapeCache.size())
2538 outShapes[i] = shapeCache[out.idx];
2539 }
2540
2541 // Skip FLOPS calculation if any shape is empty (unknown due to dynamic shapes)
2542 bool hasEmptyShape = false;
2543 for (int i = 0; i < ninputs && !hasEmptyShape; i++)
2544 hasEmptyShape = inpShapes[i].empty();
2545 for (int i = 0; i < noutputs && !hasEmptyShape; i++)
2546 hasEmptyShape = outShapes[i].empty();
2547
2548 if (!hasEmptyShape)
2549 flops += layer->getFLOPS(inpShapes, outShapes);
2550
2551 const std::vector<Ptr<Graph>>* subgraphs = layer->subgraphs();
2552 if (subgraphs) {
2553 for (const Ptr<Graph>& sg : *subgraphs)
2554 flops += getFLOPSGraph(sg, shapeCache, typeCache);
2555 }
2556 }
2557 return flops;
2558}
