103bool CChartModel::
104 evaluate(const SUBCCMModel &subModel, int &offset, int &iTheta, float &error)
105{
106
107 float tError;
108 int tTheta, tOffset;
109 error = INFINITY;
110 bool beval = false;
111
112 // para todas las orientaciones
113 // min_{ theta,dt } | CC_e - CC |
114 for (tTheta = 0; tTheta < 8; tTheta++)
115 {
116 if (match(subModel, tTheta, tError, tOffset) && tError < error)
117 {
118 error = tError;
119 iTheta = tTheta;
120 offset = tOffset;
121 beval = true;
122 }
123 }
124
125 return beval;
126}

154void mcc::CChartModel::
155 rotate90()
156{
157
158 size = Size2i(size.height, size.width);
159
160 //the matrix is roated clockwise 90 degree, so first row will become last column, second row second last column and so on
161 //doing this inplace will make the code a bit hard to read, so creating a temporary array
162
163 std::vector<Point2f> _cellchart(cellchart.size());
164 std::vector<Point2f> _center(center.size());
165
166 int k = 0;
167 for (int i = 0; i < size.width; i++)
168 {
169 for (int j = 0; j < size.height; j++)
170 {
171 //k contains the new coordintes,
172 int old_i = size.height - j - 1;
173 int old_j = i;
174 int old_k = (old_i)*size.width + old_j;
175
176 _cellchart[4 * k + 0] = cellchart[4 * old_k + 3];
177 _cellchart[4 * k + 1] = cellchart[4 * old_k + 0];
178 _cellchart[4 * k + 2] = cellchart[4 * old_k + 1];
179 _cellchart[4 * k + 3] = cellchart[4 * old_k + 2];
180
181 // center
182 _center[k] = center[old_k];
183 k++;
184 }
185 }
186 cellchart = _cellchart;
187 center = _center;
188
189 boxsize = Size2f(boxsize.height, boxsize.width);
190}

223float CChartModel::
224 dist_color_lab(InputArray lab1, InputArray lab2)
225{
226
227 int N = lab1.rows();
228 float dist = 0, dist_i;
229
230 Mat _lab1 = lab1.getMat(), _lab2 = lab2.getMat();
231 for (int i = 0; i < N; i++)
232 {
233 Vec3f v1 = _lab1.at<Vec3f>(i, 0);
234 Vec3f v2 = _lab2.at<Vec3f>(i, 0);
235 // v1[0] = 1;
236 // v2[0] = 1; // L <- 0
237
238 // euclidean
239 Vec3f v = v1 - v2;
240 dist_i = v.dot(v);
241 dist += sqrt(dist_i);
242
243 // cosine
244 //float costh = v1.dot(v2) / (norm(v1)*norm(v2));
245 //dist += 1 - (1 + costh) / 2;
246 }
247
248 dist /= N;
249 return dist;
250}

252bool CChartModel::
253 match(const SUBCCMModel &subModel, int iTheta, float &error, int &ierror)
254{
255
256 size_t N, M, k;
257
258 N = size.width;
259 M = size.height;
260 Mat im_lab_org((int)N, (int)M, CV_32FC3);
261 int type_color = 3;
262 k = 0;
263
264 for (size_t i = 0; i < N; i++)
265 {
266 for (size_t j = 0; j < M; j++)
267 {
268 Vec3f &lab = im_lab_org.at<Vec3f>((int)i, (int)j);
269 lab[0] = chart[k][type_color + 0];
270 lab[1] = chart[k][type_color + 1];
271 lab[2] = chart[k][type_color + 2];
272 k++;
273 }
274 }
275
276 rot90(im_lab_org, iTheta);
277 N = im_lab_org.rows;
278 M = im_lab_org.cols;
279
280 size_t n, m;
281 n = subModel.color_size.height;
282 m = subModel.color_size.width;
283
284 // boundary condition
285 if (N < n || M < m)
286 return false;
287
288 // rgb to La*b*
289 Mat rgb_est = subModel.sub_chart;
290 Mat lab_est;
291
292 cvtColor(rgb_est, lab_est, COLOR_BGR2Lab);
293
294 size_t nN, mM;
295 nN = N - n + 1;
296 mM = M - m + 1;
297 std::vector<float> lEcm(nN * mM);
298 k = 0;
299 for (size_t i = 0; i < nN; i++)
300 {
301 for (size_t j = 0; j < mM; j++)
302 {
303 Mat lab_curr, lab_roi;
304 lab_roi = im_lab_org(Rect((int)j, (int)i, (int)m, (int)n));
305 lab_roi.copyTo(lab_curr);
306 lab_curr = lab_curr.t();
307 lab_curr = lab_curr.reshape(3, (int)n * (int)m);
308
309 // Mean squared error
310 // ECM = 1 / N sum_i(Y - Yp) ^ 2
311 lEcm[k] = dist_color_lab(lab_curr, lab_est) / (M * N);
312 k++;
313 }
314 }
315
316 // minimo
317 error = lEcm[0];
318 ierror = 0;
319 for (int i = 1; i < (int)lEcm.size(); i++)
320 if (error > lEcm[i])
321 {
322 error = lEcm[i];
323 ierror = i;
324 }
325
326 return true;
327}

329void CChartModel::
330 rot90(InputOutputArray mat, int itheta)
331{
332
333 //1=CW, 2=CCW, 3=180
334 switch (itheta)
335 {
336 case 1: //transpose+flip(1)=CW
337 transpose(mat, mat);
338 cv::flip(mat, mat, 1);
339 break;
340 case 2: //flip(-1)=180
341 cv::flip(mat, mat, -1);
342 break;
343 case 3: //transpose+flip(0)=CCW
344 transpose(mat, mat);
345 cv::flip(mat, mat, 0);
346 break;
347 //flipped images start here
348 case 4: //flip(1)=no rotation, just flipped
349 cv::flip(mat, mat, 1);
350 break;
351 case 5: //flip(1)+transpose + flip(1)=CW
352 cv::flip(mat, mat, 1);
353 transpose(mat, mat);
354 cv::flip(mat, mat, 1);
355 break;
356 case 6: //flip(1)+flip(-1)=180
357 cv::flip(mat, mat, 1);
358 cv::flip(mat, mat, -1);
359 break;
360 case 7: //flip(1)+transpose+flip(0)=CCW
361 cv::flip(mat, mat, 1);
362 transpose(mat, mat);
363 cv::flip(mat, mat, 0);
364 break;
365 }

409std::vector<Point2f> CCheckerImpl::getColorCharts()
410{
411 // color chart classic model
412 CChartModel cccm(getTarget());
413 Mat lab;
414 size_t N;
415 std::vector<Point2f> fbox = cccm.box;
416 std::vector<Point2f> cellchart = cccm.cellchart;
417 std::vector<Point2f> charts(cellchart.size());
418
419 // tranformation
420 Matx33f ccT = getPerspectiveTransform(fbox, getBox());
421
422 std::vector<Point2f> bch(4), bcht(4);
423 N = cellchart.size() / 4;
424 for (size_t i = 0, k; i < N; i++)
425 {
426 k = 4 * i;
427 for (size_t j = 0ull; j < 4ull; j++)
428 bch[j] = cellchart[k + j];
429
430 polyanticlockwise(bch);
431 transform_points_forward(ccT, bch, bcht);
432
433 Point2f c(0, 0);
434 for (size_t j = 0; j < 4; j++)
435 c += bcht[j];
436 c /= 4;
437 for (size_t j = 0ull; j < 4ull; j++)
438 charts[k+j] = ((bcht[j] - c) * 0.50) + c;
439 }
440 return charts;
441}

463void transform_points_forward(const Matx33f& T, const std::vector<Point2f> &X, std::vector<Point2f> &Xt)
464{
465 size_t N = X.size();
466 if (Xt.size() != N)
467 Xt.resize(N);
468 std::fill(Xt.begin(), Xt.end(), Point2f(0.f, 0.f));
469 if (N == 0)
470 return;
471
472 Matx31f p, xt;
473 Point2f pt;
474 for (size_t i = 0; i < N; i++)
475 {
476 p(0, 0) = X[i].x;
477 p(1, 0) = X[i].y;
478 p(2, 0) = 1;
479 xt = T * p;
480 pt.x = xt(0, 0) / xt(2, 0);
481 pt.y = xt(1, 0) / xt(2, 0);
482 Xt[i] = pt;
483 }
484}
