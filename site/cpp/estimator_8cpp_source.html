302

312 minv21=(float)minv[3]; minv22=(float)minv[4]; minv23=(float)minv[5];

317 const float * points = points_mat.ptr<float>();
318 const float x1=points[idx], y1=points[idx+1], x2=points[idx+2], y2=points[idx+3];
319 const float est_z2 = 1 / (m31 * x1 + m32 * y1 + m33),
320 dx2 = x2 - (m11 * x1 + m12 * y1 + m13) * est_z2,
321 dy2 = y2 - (m21 * x1 + m22 * y1 + m23) * est_z2;

323 dx1 = x1 - (minv11 * x2 + minv12 * y2 + minv13) * est_z1,
324 dy1 = y1 - (minv21 * x2 + minv22 * y2 + minv23) * est_z1;

326 }
327 const std::vector<float> &getErrors (const Mat &model) override {
328 setModelParameters(model);
329 const float * points = points_mat.ptr<float>();
330 for (int point_idx = 0; point_idx < points_mat.rows; point_idx++) {
331 const int smpl = 4*point_idx;
332 const float x1=points[smpl], y1=points[smpl+1], x2=points[smpl+2], y2=points[smpl+3];
333 const float est_z2 = 1 / (m31 * x1 + m32 * y1 + m33),
334 dx2 = x2 - (m11 * x1 + m12 * y1 + m13) * est_z2,
335 dy2 = y2 - (m21 * x1 + m22 * y1 + m23) * est_z2;
336 const float est_z1 = 1 / (minv31 * x2 + minv32 * y2 + minv33),
337 dx1 = x1 - (minv11 * x2 + minv12 * y2 + minv13) * est_z1,
338 dy1 = y1 - (minv21 * x2 + minv22 * y2 + minv23) * est_z1;
339 errors[point_idx] = (dx2 * dx2 + dy2 * dy2 + dx1 * dx1 + dy1 * dy1) * .5f;

402class SampsonErrorImpl : public SampsonError {
403private:
404 Mat points_mat;
405 float m11, m12, m13, m21, m22, m23, m31, m32, m33;
406 std::vector<float> errors;
407public:
408 explicit SampsonErrorImpl (const Mat &points_)
409 : points_mat(points_)
410 , m11(0), m12(0), m13(0), m21(0), m22(0), m23(0), m31(0), m32(0), m33(0)
411 , errors(points_.rows)
412 {
413 CV_DbgAssert(!points_mat.empty() && points_mat.isContinuous());
414 }
415
416 inline void setModelParameters(const Mat& model) override
417 {
418 CV_Assert(!model.empty());
419 CV_CheckTypeEQ(model.depth(), CV_64F, "");
420
421 const auto * const m = (double *) model.data;
422 m11=static_cast<float>(m[0]); m12=static_cast<float>(m[1]); m13=static_cast<float>(m[2]);
423 m21=static_cast<float>(m[3]); m22=static_cast<float>(m[4]); m23=static_cast<float>(m[5]);
424 m31=static_cast<float>(m[6]); m32=static_cast<float>(m[7]); m33=static_cast<float>(m[8]);
425 }
426
427 /*
428 * (pt2^t * F * pt1)^2)
429 * Sampson error = ------------------------------------------------------------------------
430 * (((F⋅pt1)(0))^2 + ((F⋅pt1)(1))^2 + ((F^t⋅pt2)(0))^2 + ((F^t⋅pt2)(1))^2)
431 *
432 * [ x2 y2 1 ] * [ F(1,1) F(1,2) F(1,3) ] [ x1 ]
433 * [ F(2,1) F(2,2) F(2,3) ] * [ y1 ]
434 * [ F(3,1) F(3,2) F(3,3) ] [ 1 ]
435 *
436 */
437 inline float getError (int idx) const override {
438 idx *= 4;
439 const float * points = points_mat.ptr<float>();
440 const float x1=points[idx], y1=points[idx+1], x2=points[idx+2], y2=points[idx+3];
441 const float F_pt1_x = m11 * x1 + m12 * y1 + m13,
442 F_pt1_y = m21 * x1 + m22 * y1 + m23;
443 const float pt2_F_x = x2 * m11 + y2 * m21 + m31,
444 pt2_F_y = x2 * m12 + y2 * m22 + m32;
445 const float pt2_F_pt1 = x2 * F_pt1_x + y2 * F_pt1_y + m31 * x1 + m32 * y1 + m33;
446 return pt2_F_pt1 * pt2_F_pt1 / (F_pt1_x * F_pt1_x + F_pt1_y * F_pt1_y +
447 pt2_F_x * pt2_F_x + pt2_F_y * pt2_F_y);
448 }
449 const std::vector<float> &getErrors (const Mat &model) override {
450 setModelParameters(model);
451 const float * points = points_mat.ptr<float>();
452 for (int point_idx = 0; point_idx < points_mat.rows; point_idx++) {
453 const int smpl = 4*point_idx;
454 const float x1=points[smpl], y1=points[smpl+1], x2=points[smpl+2], y2=points[smpl+3];
455 const float F_pt1_x = m11 * x1 + m12 * y1 + m13,
456 F_pt1_y = m21 * x1 + m22 * y1 + m23;
457 const float pt2_F_x = x2 * m11 + y2 * m21 + m31,
458 pt2_F_y = x2 * m12 + y2 * m22 + m32;
459 const float pt2_F_pt1 = x2 * F_pt1_x + y2 * F_pt1_y + m31 * x1 + m32 * y1 + m33;
460 errors[point_idx] = pt2_F_pt1 * pt2_F_pt1 / (F_pt1_x * F_pt1_x + F_pt1_y * F_pt1_y +
461 pt2_F_x * pt2_F_x + pt2_F_y * pt2_F_y);
462 }
463 return errors;
464 }

814private:

819 // Compute normalized points and transformation matrices.
820 void getNormTransformation (Mat& norm_points, const std::vector<int> &sample,
821 int sample_size, Matx33d &T1, Matx33d &T2) const override {
822 const float * points = points_mat.ptr<float>();
823 double mean_pts1_x = 0, mean_pts1_y = 0, mean_pts2_x = 0, mean_pts2_y = 0;
824
825 // find average of each coordinate of points.
826 int smpl;
827 for (int i = 0; i < sample_size; i++) {
828 smpl = 4 * sample[i];
829 mean_pts1_x += points[smpl ];
830 mean_pts1_y += points[smpl + 1];
831 mean_pts2_x += points[smpl + 2];
832 mean_pts2_y += points[smpl + 3];
833 }
834 mean_pts1_x /= sample_size; mean_pts1_y /= sample_size;
835 mean_pts2_x /= sample_size; mean_pts2_y /= sample_size;
836
837 double avg_dist1 = 0, avg_dist2 = 0, x1_m, y1_m, x2_m, y2_m;
838 for (int i = 0; i < sample_size; i++) {
839 smpl = 4 * sample[i];
840 /*
841 * Compute a similarity transform T that takes points xi
842 * to a new set of points x̃i such that the centroid of
843 * the points x̃i is the coordinate origin and their
844 * average distance from the origin is √2
845 *
846 * sqrt(x̃*x̃ + ỹ*ỹ) = sqrt(2)
847 * ax*ax + by*by = 2
848 */
849 x1_m = points[smpl ] - mean_pts1_x;
850 y1_m = points[smpl + 1] - mean_pts1_y;
851 x2_m = points[smpl + 2] - mean_pts2_x;
852 y2_m = points[smpl + 3] - mean_pts2_y;
853
854 avg_dist1 += sqrt (x1_m * x1_m + y1_m * y1_m);
855 avg_dist2 += sqrt (x2_m * x2_m + y2_m * y2_m);
856 }
857
858 // scale
859 avg_dist1 = M_SQRT2 / (avg_dist1 / sample_size);
860 avg_dist2 = M_SQRT2 / (avg_dist2 / sample_size);
861
862 const double transl_x1 = -mean_pts1_x * avg_dist1, transl_y1 = -mean_pts1_y * avg_dist1;
863 const double transl_x2 = -mean_pts2_x * avg_dist2, transl_y2 = -mean_pts2_y * avg_dist2;
864
865 // transformation matrices
866 T1 = Matx33d (avg_dist1, 0, transl_x1,0, avg_dist1, transl_y1,0, 0, 1);
867 T2 = Matx33d (avg_dist2, 0, transl_x2,0, avg_dist2, transl_y2,0, 0, 1);
868
869 norm_points = Mat_<float>(sample_size, 4); // normalized points Nx4 matrix
870 auto * norm_points_ptr = (float *) norm_points.data;
871
872 // Normalize points: Npts = T*pts 3x3 * 3xN
873 const auto avg_dist1f = (float)avg_dist1, avg_dist2f = (float)avg_dist2;
874 const auto transl_x1f = (float)transl_x1, transl_y1f = (float)transl_y1;
875 const auto transl_x2f = (float)transl_x2, transl_y2f = (float)transl_y2;
876 for (int i = 0; i < sample_size; i++) {
877 smpl = 4 * sample[i];
878 (*norm_points_ptr++) = avg_dist1f * points[smpl ] + transl_x1f;
879 (*norm_points_ptr++) = avg_dist1f * points[smpl + 1] + transl_y1f;
880 (*norm_points_ptr++) = avg_dist2f * points[smpl + 2] + transl_x2f;
881 (*norm_points_ptr++) = avg_dist2f * points[smpl + 3] + transl_y2f;
882 }
883 }
