45bool UPCEANDecoder::findGuardPatterns(const std::vector<uchar> &row, uint rowOffset, uchar whiteFirst,
46 const std::vector<int> &pattern, Counter &counter, std::pair<uint, uint> &result)
47{
48 size_t patternLength = pattern.size();
49 size_t width = row.size();
50 uchar color = whiteFirst ? WHITE : BLACK;
51 rowOffset = (int) (std::find(row.cbegin() + rowOffset, row.cend(), color) - row.cbegin());
52 uint counterPosition = 0;
53 uint patternStart = rowOffset;
54 for (uint x = rowOffset; x < width; x++)
55 {
56 if (row[x] == color)
57 {
58 counter.pattern[counterPosition]++;
59 counter.sum++;
60 }
61 else
62 {
63 if (counterPosition == patternLength - 1)
64 {
65 if (patternMatch(counter, pattern, MAX_INDIVIDUAL_VARIANCE) < MAX_AVG_VARIANCE)
66 {
67 result.first = patternStart;
68 result.second = x;
69 return true;
70 }
71 patternStart += counter.pattern[0] + counter.pattern[1];
72 counter.sum -= counter.pattern[0] + counter.pattern[1];
73
74 std::copy(counter.pattern.begin() + 2, counter.pattern.end(), counter.pattern.begin());
75
76 counter.pattern[patternLength - 2] = 0;
77 counter.pattern[patternLength - 1] = 0;
78 counterPosition--;
79 }
80 else
81 {
82 counterPosition++;
83 }
84 counter.pattern[counterPosition] = 1;
85 counter.sum++;
86 color = (std::numeric_limits<uchar>::max() - color);
87 }
88 }
89 return false;
90}

92bool UPCEANDecoder::findStartGuardPatterns(const std::vector<uchar> &row, std::pair<uint, uint> &start_range)
93{
94 bool is_find = false;
95 int next_start = 0;
96 while (!is_find)
97 {
98 Counter guard_counters(std::vector<int>{0, 0, 0});
99 if (!findGuardPatterns(row, next_start, BLACK, BEGIN_PATTERN(), guard_counters, start_range))
100 {
101 return false;
102 }
103 int start = static_cast<int>(start_range.first);
104 next_start = static_cast<int>(start_range.second);
105 int quiet_start = max(start - (next_start - start), 0);
106 is_find = (quiet_start != start) &&
107 (std::find(std::begin(row) + quiet_start, std::begin(row) + start, BLACK) == std::begin(row) + start);
108 }
109 return true;
110}

112int UPCEANDecoder::decodeDigit(const std::vector<uchar> &row, Counter &counters, uint rowOffset,
113 const std::vector<std::vector<int>> &patterns)
114{
115 fillCounter(row, rowOffset, counters);
116 int bestMatch = -1;
117 uint bestVariance = MAX_AVG_VARIANCE; // worst variance we'll accept
118 int i = 0;
119 for (const auto &pattern : patterns)
120 {
121 uint variance = patternMatch(counters, pattern, MAX_INDIVIDUAL_VARIANCE);
122 if (variance < bestVariance)
123 {
124 bestVariance = variance;
125 bestMatch = i;
126 }
127 i++;
128 }
129 return std::max(-1, bestMatch);
130 // -1 is Mismatch or means error.
131}

134std::pair<Result, float> UPCEANDecoder::decodeROI(const Mat &bar_img) const
135{
136 if ((size_t) bar_img.cols < this->bits_num)
137 {
138 return std::make_pair(Result{string(), Result::BARCODE_NONE}, 0.0F);
139 }
140
141 std::map<std::string, int> result_vote;
142 std::map<Result::BarcodeType, int> format_vote;
143 int vote_cnt = 0;
144 int total_vote = 0;
145 std::string max_result;
146 Result::BarcodeType max_type = Result::BARCODE_NONE;
147
148 const int step = bar_img.rows / (DIVIDE_PART + BIAS_PART);
149 Result result;
150 int row_num;
151 for (int i = 0; i < DIVIDE_PART; ++i)
152 {
153 row_num = (i + BIAS_PART / 2) * step;
154 if (row_num < 0 || row_num > bar_img.rows)
155 {
156 continue;
157 }
158 const auto *ptr = bar_img.ptr<uchar>(row_num);
159 vector<uchar> line(ptr, ptr + bar_img.cols);
160 result = decodeLine(line);
161 if (result.format != Result::BARCODE_NONE)
162 {
163 total_vote++;
164 result_vote[result.result] += 1;
165 if (result_vote[result.result] > vote_cnt)
166 {
167 vote_cnt = result_vote[result.result];
168 max_result = result.result;
169 max_type = result.format;
170 }
171 }
172 }
173 if (total_vote == 0 || (vote_cnt << 2) < total_vote)
174 {
175 return std::make_pair(Result(string(), Result::BARCODE_NONE), 0.0f);
176 }
177
178 float confidence = (float) vote_cnt / (float) DIVIDE_PART;
179 //Check if it is UPC-A format
180 if (max_type == Result::BARCODE_EAN_13 && max_result[0] == '0')
181 {
182 max_result = max_result.substr(1, 12); //UPC-A length 12
183 max_type = Result::BARCODE_UPC_A;
184 }
185 return std::make_pair(Result(max_result, max_type), confidence);
186}

199bool UPCEANDecoder::isValid(const string &result) const
200{
201 if (result.size() != digit_number)
202 {
203 return false;
204 }
205 int sum = 0;
206 for (int index = (int) result.size() - 2, i = 1; index >= 0; index--, i++)
207 {
208 int temp = result[index] - '0';
209 sum += (temp + ((i & 1) != 0 ? temp << 1 : 0));
210 }
211 return (result.back() - '0') == ((10 - (sum % 10)) % 10);
212}

231const std::vector<std::vector<int>> &get_AB_Patterns()
232{
233 static const std::vector<std::vector<int>> AB_Patterns = [] {
234 constexpr uint offset = 10;
235 auto AB_Patterns_inited = std::vector<std::vector<int>>(offset << 1, std::vector<int>(PATTERN_LENGTH, 0));
236 std::copy(get_A_or_C_Patterns().cbegin(), get_A_or_C_Patterns().cend(), AB_Patterns_inited.begin());
237 //AB pattern is
238 for (uint i = 0; i < offset; ++i)
239 {
240 for (uint j = 0; j < PATTERN_LENGTH; ++j)
241 {
242 AB_Patterns_inited[i + offset][j] = AB_Patterns_inited[i][PATTERN_LENGTH - j - 1];

245 return AB_Patterns_inited;
246 }();

272 '\x01', '\x00', '\x00', '\x00', '\x00', '\x00'};

276 // 0x10110 -> 22 -> 2
