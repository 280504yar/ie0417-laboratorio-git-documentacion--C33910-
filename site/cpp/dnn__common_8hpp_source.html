85{

138static inline std::string toString(const Mat& blob, const std::string& name = std::string())
139{
140 std::ostringstream ss;
141 if (!name.empty())
142 ss << name << ' ';
143 if (blob.empty())
144 {
145 ss << "<empty>";
146 }
147 else if (blob.dims == 1)
148 {
149 Mat blob_ = blob;
150 blob_.size.dims = blob_.dims = 2; // hack
151 blob_.size[1] = blob_.size[0];
152 blob_.size[0] = 1;
153 ss << blob_.t();
154 }
155 else
156 {
157 ss << blob.reshape(1, 1);
158 }
159 return ss.str();
160}

162// Scalefactor is a common parameter used for data scaling. In OpenCV, we often use Scalar to represent it.
163// Because 0 is meaningless in scalefactor.
164// If the scalefactor is (x, 0, 0, 0), we convert it to (x, x, x, x). The following func will do this hack.
165static inline Scalar_<double> broadcastRealScalar(const Scalar_<double>& _scale)
166{
167 Scalar_<double> scale = _scale;
168 if (scale[1] == 0 && scale[2] == 0 && scale[3] == 0)
169 {
170 CV_Assert(scale[0] != 0 && "Scalefactor of 0 is meaningless.");
171 scale = Scalar_<double>::all(scale[0]);
172 }
173
174 return scale;
175}
