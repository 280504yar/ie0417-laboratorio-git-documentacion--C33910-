68Range normalizeRange(const Range& input_range, int n)
69{
70 Range range = input_range;
71
72 if (range.start != n){
73 range.start = std::min(std::max(range.start, -n), n - 1);
74 if (range.start < 0)
75 {
76 range.start += n;
77 }
78 }
79
80 range.end = std::min(std::max(range.end, -n), n);
81 if (range.end < 0)
82 {
83 range.end += n;
84 }
85
86 return range;
87}

90void transformForNegSteps(const MatShape& inpShape, std::vector<std::vector<Range> >& sliceRanges, std::vector<std::vector<int> >& sliceSteps)
91{
92 // in case of negative steps,
93 // x of shape [5, 10], x[5:0:-1, 10:1:-3] <=> np.flip(x[1:5:1, 2:10:3], aixs=(0, 1))
94 // new_end_i = start_i + 1 > dim_i ? dim_i : start_i + 1
95 // new_start_i = end + 1
96 // new_start_i = new_end_i - 1 - ((new_end_i - 1 - new_start_i) / abs(step_i)) * abs(step_i)
97 int start, end, new_start, new_end, step;
98 for (int i = 0; i < sliceSteps[0].size(); ++i)
99 {
100 step = sliceSteps[0][i];
101 if (step > 0)
102 continue;
103
104 step = -step;
105 start = sliceRanges[0][i].start;
106 end = sliceRanges[0][i].end;
107 new_end = start >= inpShape[i] ? inpShape[i] : start + 1;
108 new_start = end + 1;
109 new_start = new_end - 1 - ((new_end - 1 - new_start) / step) * step;
110
111 sliceSteps[0][i] = step;
112 sliceRanges[0][i].start = new_start;
113 sliceRanges[0][i].end = new_end;
114 }

117std::vector<std::vector<cv::Range> > finalizeSliceRange(const MatShape& inpShape, int& axis,
118 const std::vector<std::vector<cv::Range> >& inputSliceRanges)
119{
120 std::vector<std::vector<cv::Range> > sliceRanges = inputSliceRanges;
121 CV_Assert(inpShape.size() > 0);
122 bool axisNeg = (axis < 0);
123 axis = (axis + static_cast<int>(inpShape.size())) % inpShape.size();
124
125 for (size_t i = 0; i < sliceRanges.size(); ++i){
126 std::vector<Range>& ranges = sliceRanges[i];
127 if (axisNeg)
128 {
129 ranges.insert(ranges.begin(), axis, Range::all());
130 }
131
132 for (size_t j = 0; j < ranges.size(); ++j)
133 {
134 int n = inpShape[j];
135 if (n <= 0)
136 {
137 continue;
138 }
139
140 ranges[j] = normalizeRange(ranges[j], n);
141 }
142 }
143
144 return sliceRanges;
145}

147class SliceLayerImpl : public SliceLayer
148{
149public:
150 SliceLayerImpl(const LayerParams& params)
151 {
152 setParamsFrom(params);
153 hasSteps = false;
154 axis = params.get<int>("axis", 1);
155 num_split = params.get<int>("num_split", 0);
156 hasDynamicShapes = params.get<bool>("has_dynamic_shapes", false);
157 shapesInitialized = !hasDynamicShapes;
158
159 if (params.has("slice_point"))
160 {
161 CV_Assert(!params.has("begin") && !params.has("size") && !params.has("end"));
162 const DictValue &indicesValue = params.get("slice_point");
163 int size = axis > 0 ? axis + 1 : 1;
164 sliceRanges.resize(indicesValue.size() + 1,
165 std::vector<Range>(size, Range::all()));
166 int prevSlice = 0;
167 for (int i = 0; i < indicesValue.size(); ++i)
168 {
169 sliceRanges[i][size - 1].start = prevSlice;
170 sliceRanges[i][size - 1].end = indicesValue.get<int>(i);
171 prevSlice = sliceRanges[i][size - 1].end;
172 }
173 sliceRanges.back()[size - 1].start = prevSlice;
174 }
175 else if (params.has("begin"))
176 {
177 CV_Assert(params.has("size") ^ params.has("end"));
178 const DictValue &begins = params.get("begin");
179 const DictValue &sizesOrEnds = params.has("size") ? params.get("size") : params.get("end");
180 CV_Assert(begins.size() == sizesOrEnds.size());
181
182 if (params.has("steps"))
183 {
184 const DictValue &steps = params.get("steps");
185 sliceSteps.resize(1);
186 sliceSteps[0].resize(steps.size());
187
188 for (int i = 0; i < steps.size(); ++i)
189 {
190 int step = steps.get<int>(i);
191 CV_Assert(step != 0);
192 if (step < 0)
193 neg_step_dims.push_back(i);
194 if (std::abs(step) > 1 || step < 0)
195 hasSteps = true;
196 sliceSteps[0][i] = step;
197 }
198 }
199
200 sliceRanges.resize(1);
201 sliceRanges[0].resize(begins.size(), Range::all());
202 for (int i = 0; i < begins.size(); ++i)
203 {
204 int start = begins.get<int>(i);
205 int sizeOrEnd = sizesOrEnds.get<int>(i); // It may be negative to reverse indexation.
206
207 sliceRanges[0][i].start = start;
208 if (params.has("size"))
209 {
210 int size = sizeOrEnd;
211 CV_Assert(size == -1 || size > 0); // -1 value means range [start, axis_size).
212 sliceRanges[0][i].end = size > 0 ? (start + size) : INT_MAX; // We'll finalize a negative value later.
213 }
214 else
215 {
216 int end = sizeOrEnd;
217 if (hasSteps && !neg_step_dims.empty() && sliceSteps[0][i] < 0)
218 CV_Assert(end < 0 || end != start); // if current step is negative, end < start is allowed.
219 else
220 CV_Assert(end < 0 || end > start); // End index is excluded.
221 sliceRanges[0][i].end = end; // We'll finalize a negative value later.
222 }
223 }
224 }

242 bool getMemoryShapes(const std::vector<MatShape> &inputs,
243 const int requiredOutputs,
244 std::vector<MatShape> &outputs,
245 std::vector<MatShape> &internals) const CV_OVERRIDE
246 {
247 CV_Assert(inputs.size() == 1);
248 MatShape inpShape = inputs[0];
249
250 std::vector<std::vector<int> > sliceSteps_ = sliceSteps;
251 std::vector<std::vector<cv::Range> > sliceRanges_ = sliceRanges;
252 if (hasSteps && !neg_step_dims.empty())
253 transformForNegSteps(inpShape, sliceRanges_, sliceSteps_);
254
255 int axis_rw = axis;
256 std::vector<std::vector<cv::Range> > sliceRanges_rw = finalizeSliceRange(inpShape, axis_rw, sliceRanges_);
257
258 if (!sliceRanges_rw.empty())
259 {
260 outputs.resize(sliceRanges_rw.size(), inpShape);
261 for (int i = 0; i < outputs.size(); ++i)
262 {
263 CV_Assert(sliceRanges_rw[i].size() <= inpShape.size());
264 for (int j = 0; j < sliceRanges_rw[i].size(); ++j)
265 {
266 if (shapesInitialized || inpShape[j] > 0)
267 outputs[i][j] = normalizeRange(sliceRanges_rw[i][j], inpShape[j]).size();
268
269 if (!sliceSteps_.empty() && (i < sliceSteps_.size()) && (j < sliceSteps_[i].size()) && (sliceSteps_[i][j] > 1))
270 outputs[i][j] = (outputs[i][j] + sliceSteps_[i][j] - 1) / sliceSteps_[i][j];
271 }
272 }
273 }
274 else // Divide input blob on equal parts by axis.
275 {
276 CV_Assert(0 <= axis_rw && axis_rw < inpShape.size());
277 int splits = num_split ? num_split : requiredOutputs;
278 CV_Assert(splits > 0 && inpShape[axis_rw] % splits == 0);
279 inpShape[axis_rw] /= splits;
280 outputs.resize(splits, inpShape);
281 }
282 return false;
283 }

310 void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr) CV_OVERRIDE
311 {
312#ifdef HAVE_OPENCL
313 ocl_exec_cache.clear();
314#endif
315
316 std::vector<Mat> inputs, outputs;
317 inputs_arr.getMatVector(inputs);
318 outputs_arr.getMatVector(outputs);
319
320 CV_Assert(inputs.size() == 1);
321 MatShape inpShape = shape(inputs[0]);
322
323 if (hasSteps && !neg_step_dims.empty())
324 transformForNegSteps(inpShape, sliceRanges, sliceSteps);
325
326 finalSliceRanges = finalizeSliceRange(shape(inputs[0]), axis, sliceRanges);
327
328 if (sliceRanges.empty())
329 {
330 // Divide input blob on equal parts by axis.
331 int outAxisSize = inpShape[axis] / outputs.size();
332 finalSliceRanges.resize(outputs.size(),
333 std::vector<Range>(axis + 1, Range::all()));
334 int prevSlice = 0;
335 for (int i = 0; i < outputs.size(); ++i)
336 {
337 finalSliceRanges[i][axis].start = prevSlice;
338 finalSliceRanges[i][axis].end = finalSliceRanges[i][axis].start + outAxisSize;
339 prevSlice = finalSliceRanges[i][axis].end;
340 }
341 }
342 else
343 CV_Assert(outputs.size() == sliceRanges.size());
344
345 for (int i = 0; i < outputs.size(); ++i)
346 {
347 CV_Assert(finalSliceRanges[i].size() <= inpShape.size());
348 // Fill the rest of ranges.
349 for (int j = finalSliceRanges[i].size(); j < inpShape.size(); ++j)
350 {
351 finalSliceRanges[i].push_back(Range::all());
352 }
353 // Clamp.
354 for (int j = 0; j < finalSliceRanges[i].size(); ++j)
355 {
356 finalSliceRanges[i][j] = normalizeRange(finalSliceRanges[i][j], inpShape[j]);
357 }
358 }
359
360 if (!sliceSteps.empty() && sliceSteps[0].size() != inputs[0].dims)
361 sliceSteps[0].resize(inputs[0].dims, 1);
362
363#if 0
364 std::cout << "DEBUG: DNN/Slice: " << outputs.size() << " inpShape=" << inpShape << std::endl;
365 for (int i = 0; i < outputs.size(); ++i)
366 {
367 for (int j = 0; j < finalSliceRanges[i].size(); ++j)
368 {
369 std::cout << finalSliceRanges[i][j];
370 }
371 std::cout << std::endl;
372 }
373#endif

392 void ocl_prepare(const std::vector<UMat>& inputs, const std::vector<UMat>& outputs)
393 {
394 CV_TRACE_FUNCTION();
395
396 CV_Assert(outputs.size() == finalSliceRanges.size());
397 ocl_exec_cache.resize(outputs.size());
398
399 const UMat& input = inputs[0];
400 const int dims = input.dims;
401
402 size_t WSZ = 128;
403
404 const int elemSize = (int)input.elemSize();
405 String opts0 = cv::format(
406 "-DDIMS=%d -DELEMSIZE=%d",
407 dims, elemSize
408 );
409 for (int d = 0; d < dims; d++)
410 {
411 opts0 += cv::format(" -DSRC_STEP_%d=%d", d, (int)input.step[dims - 1 - d]);
412 }
413 for (size_t i = 0; i < outputs.size(); i++)
414 {
415 OpenCLExecInfo& ocl = ocl_exec_cache[i];
416
417 const UMat& output = outputs[i];
418 const std::vector<Range>& range = finalSliceRanges[i];
419
420 String opts = opts0;
421
422 CV_CheckEQ(output.dims, dims, "");
423 for (int d = 0; d < dims; d++)
424 {
425 opts += cv::format(" -DDST_STEP_%d=%d -DDST_SZ_%d=%d -DSRC_START_%d=%d",
426 d, (int)output.step[dims - 1 - d],
427 d, (int)output.size[dims - 1 - d],
428 d, (int)range[dims - 1 - d].start
429 );
430 CV_CheckEQ(range[d].size(), (int)output.size[d], "");
431 }
432
433 const size_t param_LIMIT_BLOCK_SIZE_PER_WG = WSZ * 64;
434
435 int block_dims = 0;
436 size_t block_size = elemSize;
437 for (int i = dims - 1; i >= 0; --i)
438 {
439 if (input.step[i] != output.step[i])
440 break;
441 block_size *= output.size[i];
442 block_dims++;
443 if (block_size >= param_LIMIT_BLOCK_SIZE_PER_WG)
444 break;
445 }
446
447 const size_t total = output.total() * elemSize;
448 size_t num_blocks = total / block_size;
449
450 if ((num_blocks <= 8 && block_size >= WSZ * 4) || (block_size >= param_LIMIT_BLOCK_SIZE_PER_WG))
451 {
452 // use 1D copy mode
453 opts += cv::format(" -DUSE_COPY_1D=1");
454
455 opts += cv::format(" -DBLOCK_DIMS=%d", block_dims);
456 opts += cv::format(" -DBLOCK_DIMS_CONTIGUOUS=%d", block_dims);
457 opts += cv::format(" -DBLOCK_SIZE=%d", (int)block_size);
458
459 opts += cv::format(" -DBLOCK_COLS=%d", (int)block_size);
460 }
461 else
462 {
463 // use 2D copy mode
464 int block_cols = block_size;
465 int block_dims_contiguous = block_dims;
466 size_t input_base_step = input.step[dims - 1 - block_dims_contiguous];
467 size_t output_base_step = output.step[dims - 1 - block_dims_contiguous];
468
469 size_t block_rows = 1;
470 for (int i = dims - 1 - block_dims_contiguous; i >= 0; --i)
471 {
472 if (input.step[i] * output_base_step != output.step[i] * input_base_step)
473 break;
474 block_rows *= output.size[i];
475 block_dims++;
476 }
477
478 block_size *= block_rows;
479
480 num_blocks = total / block_size;
481
482 if (block_rows > 1)
483 {
484 opts += cv::format(" -DBLOCK_DIMS=%d", block_dims);
485 opts += cv::format(" -DBLOCK_DIMS_CONTIGUOUS=%d", block_dims_contiguous);
486 opts += cv::format(" -DBLOCK_SIZE=%d", (int)block_size);
487
488 opts += cv::format(" -DBLOCK_COLS=%d", (int)block_cols);
489
490 opts += cv::format(" -DBLOCK_ROWS=%d", (int)block_rows);
491 opts += cv::format(" -DBLOCK_SRC_STRIDE=%d", (int)input_base_step);
492 }
493 else
494 {
495 // use 1D copy mode
496 opts += cv::format(" -DUSE_COPY_1D=1");
497
498 opts += cv::format(" -DBLOCK_DIMS=%d", block_dims_contiguous);
499 opts += cv::format(" -DBLOCK_DIMS_CONTIGUOUS=%d", block_dims_contiguous);
500 opts += cv::format(" -DBLOCK_SIZE=%d", (int)block_size);
501
502 opts += cv::format(" -DBLOCK_COLS=%d", (int)block_size);
503 }
504 }
505
506 const size_t MIN_WORK_ITEMS = 16;
507 if (block_size <= 4 * MIN_WORK_ITEMS)
508 WSZ = 4;
509 else if (block_size <= 8 * MIN_WORK_ITEMS)
510 WSZ = 8;
511 else if (block_size <= 16 * MIN_WORK_ITEMS)
512 WSZ = 16;
513 else if (block_size <= 32 * MIN_WORK_ITEMS)
514 WSZ = 32;
515 else if (block_size <= 64 * MIN_WORK_ITEMS)
516 WSZ = 64;
517
518 opts += cv::format(" -DWSZ=%d", (int)WSZ);
519
520 std::ostringstream kernel_suffix;
521 kernel_suffix << dims << 'x' << elemSize << "_bsz" << block_size;
522 kernel_suffix << "__src_";
523 for (int d = 0; d < dims; d++)
524 {
525 kernel_suffix << input.size[dims - 1 - d] << '_';
526 }
527 kernel_suffix << '_';
528 /*for (int d = 0; d < dims; d++)
529 {
530 kernel_suffix << input.step[dims - 1 - d] << '_';
531 }
532 kernel_suffix << '_';*/
533
534 kernel_suffix << "dst_";
535 for (int d = 0; d < dims; d++)
536 {
537 kernel_suffix << output.size[dims - 1 - d] << '_';
538 }
539 /*kernel_suffix << '_';
540 for (int d = 0; d < dims; d++)
541 {
542 kernel_suffix << output.step[dims - 1 - d] << '_';
543 }*/
544 kernel_suffix << "_slice_";
545 for (int d = 0; d < dims; d++)
546 {
547 kernel_suffix << range[dims - 1 - d].start << '_';
548 }
549 for (int d = 0; d < dims; d++)
550 {
551 kernel_suffix << '_' << range[dims - 1 - d].end;
552 }
553
554 std::string kernel_suffix_str = kernel_suffix.str();
555 opts += cv::format(" -DSLICE_KERNEL_SUFFIX=%s", kernel_suffix_str.c_str());
556
557 ocl.kernel_name = cv::format("slice_%s", kernel_suffix_str.c_str());
558 ocl.build_opts = opts;
559 ocl.local_size[0] = WSZ;
560 ocl.local_size[1] = 1;
561 ocl.global_size[0] = WSZ;
562 ocl.global_size[1] = num_blocks;
563 } // for outputs.size()
564 } // ocl_prepare

566 bool forward_ocl(InputArrayOfArrays inputs_, OutputArrayOfArrays outputs_, OutputArrayOfArrays internals_)
567 {
568 CV_TRACE_FUNCTION();
569
570 if (hasSteps)
571 return false; // TODO not implemented yet: https://github.com/opencv/opencv/pull/19546
572
573 std::vector<UMat> inputs;
574 std::vector<UMat> outputs;
575
576 inputs_.getUMatVector(inputs);
577 outputs_.getUMatVector(outputs);
578
579 CV_Assert(outputs.size() == finalSliceRanges.size());
580
581 const UMat& input = inputs[0];
582 const int dims = input.dims;
583 if (dims > 5)
584 {
585 CV_LOG_INFO(NULL, "DNN/OpenCL/Slice: implementation doesn't support dims=" << dims << ". Fallback to CPU");
586 return false;
587 }
588
589 if (ocl_exec_cache.empty())
590 {
591 ocl_prepare(inputs, outputs);
592 }
593 CV_CheckEQ(ocl_exec_cache.size(), outputs.size(), "");
594
595 for (size_t i = 0; i < outputs.size(); i++)
596 {
597 const OpenCLExecInfo& ocl = ocl_exec_cache[i];
598
599 UMat& output = outputs[i];
600
601 ocl::Kernel kernel(ocl.kernel_name.c_str(), ocl::dnn::slice_oclsrc, ocl.build_opts);
602 if (kernel.empty())
603 return false;
604 bool ret = kernel.args(
605 ocl::KernelArg::PtrReadOnly(input),
606 ocl::KernelArg::PtrWriteOnly(output)
607 )
608 .run_(2, (size_t*)ocl.global_size, (size_t*)ocl.local_size, false);
609 if (!ret)
610 return false;
611 } // for outputs.size()
612
613 return true;
614 } // forward_ocl

617 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
618 {
619 CV_TRACE_FUNCTION();
620 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
621
622 std::vector<Mat> inputs, outputs;
623 inputs_arr.getMatVector(inputs);
624 outputs_arr.getMatVector(outputs);
625
626 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
627 forward_ocl(inputs_arr, outputs_arr, internals_arr))
628
629 const Mat& inpMat = inputs[0];
630 CV_Assert(outputs.size() == finalSliceRanges.size());
631
632 if (!hasSteps)
633 {
634 for (size_t i = 0; i < outputs.size(); i++)
635 {
636 if (finalSliceRanges[i][0].start != finalSliceRanges[i][0].end){
637 inpMat(finalSliceRanges[i]).copyTo(outputs[i]);
638 }
639 }
640 }
641 if (hasSteps)
642 {
643 if (inpMat.depth() == CV_32S) run_parallel<int32_t>(inpMat, outputs);
644 else if (inpMat.depth() == CV_64S) run_parallel<int64_t>(inpMat, outputs);
645 else if (inpMat.depth() == CV_16F) run_parallel<int16_t>(inpMat, outputs);
646 else if (inpMat.depth() == CV_8S) run_parallel<int8_t>(inpMat, outputs);
647 else if (inpMat.depth() == CV_8U) run_parallel<uint8_t>(inpMat, outputs);

835 template <typename T>
836 class ParallelSlice : public cv::ParallelLoopBody
837 {
838 public:
839 ParallelSlice(const Mat& inp, Mat& out,
840 const std::vector<Range>& ranges,
841 const std::vector<int>& steps)
842 : inp_(inp), out_(out), ranges_(ranges), steps_(steps)
843 {
844 dims_ = inp.dims;
845 es_ = inp.elemSize();
846
847 inp_strides_.resize(dims_);
848 out_strides_.resize(dims_);
849 for(int i=0; i<dims_; ++i) {
850 inp_strides_[i] = inp.step.p[i];
851 out_strides_[i] = out.step.p[i];
852 }
853 }
854
855 void operator()(const Range& range) const CV_OVERRIDE
856 {
857 int b = ranges_[0].start;
858 int s = steps_[0];
859
860 const uchar* src_base = inp_.ptr();
861 uchar* dst_base = out_.ptr();
862
863 for (int i = range.start; i < range.end; ++i)
864 {
865 int k = b + i * s;
866 size_t src_offset = k * inp_strides_[0];
867 size_t dst_offset = i * out_strides_[0];
868 if (dims_ == 1)
869 std::memcpy(dst_base + dst_offset, src_base + src_offset, es_);
870 else
871 recursive_copy(1, src_base + src_offset, dst_base + dst_offset);
872 }
873 }
874
875 void recursive_copy(int dim, const uchar* src_ptr, uchar* dst_ptr) const
876 {
877 if (dim >= dims_) return;
878
879 int begin = ranges_[dim].start;
880 int end = ranges_[dim].end;
881 int step = steps_[dim];
882
883 if (dim == dims_ - 1)
884 {
885 if (step == 1)
886 {
887 size_t count = end - begin;
888 std::memcpy(dst_ptr, src_ptr + begin * inp_strides_[dim], count * es_);
889 }
890 else
891 {
892 const uchar* s_ptr = src_ptr + begin * inp_strides_[dim];
893 uchar* d_ptr = dst_ptr;
894 size_t s_stride = step * inp_strides_[dim];
895 size_t d_stride = out_strides_[dim];
896
897 for (int k = begin; k < end; k += step)
898 {
899 *(T*)d_ptr = *(const T*)s_ptr;
900 s_ptr += s_stride;
901 d_ptr += d_stride;
902 }
903 }
904 return;
905 }
906
907 if (step == 1 && is_fully_contiguous(dim))
908 {
909 size_t count = end - begin;
910 size_t bytes = count * inp_strides_[dim];
911 std::memcpy(dst_ptr, src_ptr + begin * inp_strides_[dim], bytes);
912 return;
913 }
914
915 size_t src_stride = step * inp_strides_[dim];
916 size_t dst_stride = out_strides_[dim];
917
918 const uchar* s_ptr = src_ptr + begin * inp_strides_[dim];
919 uchar* d_ptr = dst_ptr;
920
921 for (int k = begin; k < end; k += step)
922 {
923 recursive_copy(dim + 1, s_ptr, d_ptr);
924 s_ptr += src_stride;
925 d_ptr += dst_stride;
926 }
927 }
928
929 bool is_fully_contiguous(int dim) const
930 {
931 size_t expected_step = es_;
932 for (int d = dims_ - 1; d >= dim; --d)
933 {
934 if (inp_.step[d] != expected_step) return false;
935 if (d > dim) {
936 if (steps_[d] != 1) return false;
937 if (ranges_[d].start != 0 || ranges_[d].end != inp_.size[d]) return false;
938 expected_step *= inp_.size[d];
939 }
940 }
941 return true;
942 }
943
944 private:
945 const Mat& inp_;
946 Mat& out_;
947 const std::vector<Range>& ranges_;
948 const std::vector<int>& steps_;
949 int dims_;
950 size_t es_;
951 std::vector<size_t> inp_strides_;
952 std::vector<size_t> out_strides_;
953 };
954
955 template <typename T>
956 void run_parallel(const Mat& inp, std::vector<Mat>& outputs)
957 {
958 int dimsNum = inp.dims;
959 for(size_t i=0; i<outputs.size(); ++i) {
960 std::vector<int> steps;
961 if (!sliceSteps.empty() && i < sliceSteps.size() && !sliceSteps[i].empty())
962 steps = sliceSteps[i];
963 else
964 steps.assign(dimsNum, 1);
965
966 ParallelSlice<T> body(inp, outputs[i], finalSliceRanges[i], steps);
967 int dim0_size = outputs[i].size[0];
968 parallel_for_(Range(0, dim0_size), body);
969 }
970 }
971 void flip(Mat& output) // break if 1d tensor?
972 {
973 for (int i = 0; i < neg_step_dims.size(); ++i)
974 cv::flipND(output, output, neg_step_dims[i]);
975 }
976protected:
977 // The actual non-negative values determined from @p sliceRanges depends on input size.
978 std::vector<std::vector<Range> > finalSliceRanges;
979 std::vector<int> neg_step_dims;
980 bool hasDynamicShapes;
981 bool shapesInitialized;
982 bool hasSteps;
983};
