144static std::string path_join(const std::string& prefix, const std::string& subpath)
145{
146 CV_Assert(subpath.empty() || subpath[0] != '/');
147 if (prefix.empty())
148 return subpath;
149 bool skipSlash = prefix.size() > 0 ? (prefix[prefix.size()-1] == '/' || prefix[prefix.size()-1] == '\\') : false;
150 std::string path = prefix + (skipSlash ? "" : "/") + subpath;
151 return path;
152}

227// reads 16-digit hexadecimal number (i.e. 64-bit integer)
228int64 readSeed( const char* str )
229{
230 int64 val = 0;
231 if( str && strlen(str) == 16 )
232 {
233 for( int i = 0; str[i]; i++ )
234 {
235 int c = tolower(str[i]);
236 if( !isxdigit(c) )
237 return 0;
238 val = val * 16 +
239 (str[i] < 'a' ? str[i] - '0' : str[i] - 'a' + 10);
240 }
241 }
242 return val;
243}

285void BaseTest::safe_run( int start_from )
286{
287 CV_TRACE_FUNCTION();
288 ts->update_context( 0, -1, true );
289 ts->update_context( this, -1, true );
290
291 if( !::testing::GTEST_FLAG(catch_exceptions) )
292 run( start_from );
293 else
294 {
295 try
296 {
297 #if !defined _WIN32
298 int _code = setjmp( tsJmpMark );
299 if( !_code )
300 run( start_from );
301 else
302 throw TS::FailureCode(_code);
303 #else
304 run( start_from );
305 #endif
306 }
307 catch (const cv::Exception& exc)
308 {
309 const char* errorStr = exc.codeMessage();
310 char buf[1 << 16];
311
312 const char* delim = exc.err.find('\n') == cv::String::npos ? "" : "\n";
313 snprintf( buf, sizeof(buf), "OpenCV Error:\n\t%s (%s%s) in %s, file %s, line %d",
314 errorStr, delim, exc.err.c_str(), exc.func.size() > 0 ?
315 exc.func.c_str() : "unknown function", exc.file.c_str(), exc.line );
316 ts->printf(TS::LOG, "%s\n", buf);
317
318 ts->set_failed_test_info( TS::FAIL_ERROR_IN_CALLED_FUNC );
319 }
320 catch (const TS::FailureCode& fc)
321 {
322 std::string errorStr = TS::str_from_code(fc);
323 ts->printf(TS::LOG, "General failure:\n\t%s (%d)\n", errorStr.c_str(), fc);
324
325 ts->set_failed_test_info( fc );
326 }
327 catch (...)
328 {
329 ts->printf(TS::LOG, "Unknown failure\n");
330
331 ts->set_failed_test_info( TS::FAIL_EXCEPTION );
332 }
333 }
334
335 ts->set_gtest_status();
336}

339void BaseTest::run( int start_from )
340{
341 int test_case_idx, count = get_test_case_count();
342 int64 t_start = cv::getTickCount();
343 double freq = cv::getTickFrequency();
344 bool ff = can_do_fast_forward();
345 int progress = 0, code;
346 int64 t1 = t_start;
347
348 for( test_case_idx = ff && start_from >= 0 ? start_from : 0;
349 count < 0 || test_case_idx < count; test_case_idx++ )
350 {
351 ts->update_context( this, test_case_idx, ff );
352 progress = update_progress( progress, test_case_idx, count, (double)(t1 - t_start)/(freq*1000) );
353
354 code = prepare_test_case( test_case_idx );
355 if( code < 0 || ts->get_err_code() < 0 )
356 return;
357
358 if( code == 0 )
359 continue;
360
361 run_func();
362
363 if( ts->get_err_code() < 0 )
364 return;
365
366 if( validate_test_results( test_case_idx ) < 0 || ts->get_err_code() < 0 )
367 {
368 std::stringstream ss;
369 dump_test_case(test_case_idx, &ss);
370 std::string s = ss.str();
371 ts->printf( TS::LOG, "%s", s.c_str());
372 return;
373 }
374 }
375}

402int BaseTest::update_progress( int progress, int test_case_idx, int count, double dt )
403{
404 int width = 60 - (int)get_name().size();
405 if( count > 0 )
406 {
407 int t = cvRound( ((double)test_case_idx * width)/count );
408 if( t > progress )
409 {
410 ts->printf( TS::CONSOLE, "." );
411 progress = t;
412 }
413 }
414 else if( cvRound(dt) > progress )
415 {
416 ts->printf( TS::CONSOLE, "." );
417 progress = cvRound(dt);
418 }
419
420 return progress;
421}

441int BadArgTest::run_test_case( int expected_code, const string& _descr )
442{
443 int errcount = 0;
444 bool thrown = false;
445 const char* descr = _descr.c_str() ? _descr.c_str() : "";
446
447 try
448 {
449 run_func();
450 }
451 catch(const cv::Exception& e)
452 {
453 thrown = true;
454 if (e.code != expected_code &&
455 e.code != cv::Error::StsError && e.code != cv::Error::StsAssert // Exact error codes support will be dropped. Checks should provide proper text messages instead.
456 )
457 {
458 ts->printf(TS::LOG, "%s (test case #%d): the error code %d is different from the expected %d\n",
459 descr, test_case_idx, e.code, expected_code);
460 errcount = 1;
461 }
462 }
463 catch(...)
464 {
465 thrown = true;
466 ts->printf(TS::LOG, "%s (test case #%d): unknown exception was thrown (the function has likely crashed)\n",
467 descr, test_case_idx);
468 errcount = 1;
469 }
470
471 if(!thrown)
472 {
473 ts->printf(TS::LOG, "%s (test case #%d): no expected exception was thrown\n",
474 descr, test_case_idx);
475 errcount = 1;
476 }
477 test_case_idx++;
478
479 return errcount;
480}

516{
517 switch( code )

533 case FAIL_HANG: return "Infinite loop(?)";
534 case FAIL_BAD_ARG_CHECK: return "Incorrect handling of bad arguments";

536 ;
537 }
538 return "Generic/Unknown";

649void TS::vprintf( int streams, const char* fmt, va_list l )
650{
651 char str[1 << 14];
652 vsnprintf( str, sizeof(str)-1, fmt, l );
653
654 for( int i = 0; i < MAX_IDX; i++ )
655 if( (streams & (1 << i)) )
656 {
657 output_buf[i] += std::string(str);
658 // in the new GTest-based framework we do not use
659 // any output files (except for the automatically generated xml report).
660 // if a test fails, all the buffers are printed, so we do not want to duplicate the information and
661 // thus only add the new information to a single buffer and return from the function.
662 break;
663 }
664}

685template<>
686void fillGradient<uint8_t>(Mat& img, int delta)
687{
688 const int ch = img.channels();
689 CV_Assert(!img.empty() && img.depth() == CV_8U && ch <= 4);
690
691 int n = 255 / delta;
692 int r, c, i;
693 for(r=0; r<img.rows; r++)
694 {
695 int kR = r % (2*n);
696 int valR = (kR<=n) ? delta*kR : delta*(2*n-kR);
697 for(c=0; c<img.cols; c++)
698 {
699 int kC = c % (2*n);
700 int valC = (kC<=n) ? delta*kC : delta*(2*n-kC);
701 uchar vals[] = {uchar(valR), uchar(valC), uchar(200*r/img.rows), uchar(255)};
702 uchar *p = img.ptr(r, c);
703 for(i=0; i<ch; i++) p[i] = vals[i];
704 }
705 }
706}

759void testTearDown()
760{
761 ::cvtest::checkIppStatus();
762 uint64_t memory_usage = 0;
763 uint64_t ocv_memory_usage = 0, ocv_peak = 0;
764 if (malloc_peak) // if memory profiler is available
765 {
766 size_t peak = malloc_peak();
767 memory_usage = peak - memory_usage_base;
768 if (peak > 0)
769 {
770 CV_LOG_INFO(NULL, "Memory_usage (malloc): " << memory_usage << " (base=" << memory_usage_base << ")");
771 }
772 }
773 {
774 // core/src/alloc.cpp: #define OPENCV_ALLOC_ENABLE_STATISTICS
775 // handle large buffers via fastAlloc()
776 // (not always accurate on heavy 3rdparty usage, like protobuf)
777 cv::utils::AllocatorStatisticsInterface& ocv_stats = cv::getAllocatorStatistics();

801 uint64_t ocl_memory_usage = 0;
802#endif
803 if (malloc_peak // external memory profiler is available
804 || ocv_peak > 0 // or enabled OpenCV builtin allocation statistics
805 )
806 {
807 CV_LOG_INFO(NULL, "Memory usage total: " << (memory_usage + ocl_memory_usage));
808 ::testing::Test::RecordProperty("memory_usage",

871static bool isDirectory(const std::string& path)
872{
873#if defined _WIN32 || defined WINCE
874 WIN32_FILE_ATTRIBUTE_DATA all_attrs;
875#ifdef WINRT
876 wchar_t wpath[MAX_PATH];
877 size_t copied = mbstowcs(wpath, path.c_str(), MAX_PATH);
878 CV_Assert((copied != MAX_PATH) && (copied != (size_t)-1));
879 BOOL status = ::GetFileAttributesExW(wpath, GetFileExInfoStandard, &all_attrs);
880#else
881 BOOL status = ::GetFileAttributesExA(path.c_str(), GetFileExInfoStandard, &all_attrs);
882#endif
883 DWORD attributes = all_attrs.dwFileAttributes;
884 return status && ((attributes & FILE_ATTRIBUTE_DIRECTORY) != 0);
885#else
886 struct stat s;
887 if (0 != stat(path.c_str(), &s))
888 return false;
889 return S_ISDIR(s.st_mode);
890#endif
891}

908static std::string findData(const std::string& relative_path, bool required, bool findDirectory)
909{
910#define CHECK_FILE_WITH_PREFIX(prefix, result) \
911{ \
912 result.clear(); \
913 std::string path = path_join(prefix, relative_path); \
914 /*printf("Trying %s\n", path.c_str());*/ \
915 if (findDirectory) \
916 { \
917 if (isDirectory(path)) \
918 result = path; \
919 } \
920 else \
921 { \
922 FILE* f = fopen(path.c_str(), "rb"); \
923 if(f) { \
924 fclose(f); \
925 result = path; \
926 } \
927 } \
928}
929
930#define TEST_TRY_FILE_WITH_PREFIX(prefix) \
931{ \
932 std::string result__; \
933 CHECK_FILE_WITH_PREFIX(prefix, result__); \
934 if (!result__.empty()) \
935 return result__; \
936}
937
938
939 const std::vector<std::string>& search_path = TS::ptr()->data_search_path;
940 for(size_t i = search_path.size(); i > 0; i--)
941 {
942 const std::string& prefix = search_path[i - 1];
943 TEST_TRY_FILE_WITH_PREFIX(prefix);
944 }
945
946 const std::vector<std::string>& search_subdir = TS::ptr()->data_search_subdir;
947
948 std::string datapath_dir = cv::utils::getConfigurationParameterString("OPENCV_TEST_DATA_PATH");
949
950 std::string datapath;
951 if (!datapath_dir.empty())
952 {
953 datapath = datapath_dir;
954 //CV_Assert(isDirectory(datapath) && "OPENCV_TEST_DATA_PATH is specified but it doesn't exist");
955 if (isDirectory(datapath))
956 {
957 for(size_t i = search_subdir.size(); i > 0; i--)
958 {
959 const std::string& subdir = search_subdir[i - 1];
960 std::string prefix = path_join(datapath, subdir);
961 std::string result_;
962 CHECK_FILE_WITH_PREFIX(prefix, result_);
963 if (!required && !result_.empty())
964 {
965 static bool checkOptionalFlag = cv::utils::getConfigurationParameterBool("OPENCV_TEST_CHECK_OPTIONAL_DATA", false);
966 if (checkOptionalFlag)
967 {
968 std::cout << "TEST ERROR: Don't use 'optional' findData() for " << relative_path << std::endl;
969 CV_Assert(required || result_.empty());
970 }
971 }
972 if (!result_.empty())
973 return result_;
974 }
975 }
976 }
977#ifdef OPENCV_TEST_DATA_INSTALL_PATH
978 datapath = OPENCV_TEST_DATA_INSTALL_PATH;
979
980 if (isDirectory(datapath))
981 {
982 for(size_t i = search_subdir.size(); i > 0; i--)
983 {
984 const std::string& subdir = search_subdir[i - 1];
985 std::string prefix = path_join(datapath, subdir);
986 TEST_TRY_FILE_WITH_PREFIX(prefix);
987 }
988 }
989#ifdef OPENCV_INSTALL_PREFIX
990 else
991 {
992 datapath = path_join(OPENCV_INSTALL_PREFIX, OPENCV_TEST_DATA_INSTALL_PATH);
993 if (isDirectory(datapath))
994 {
995 for(size_t i = search_subdir.size(); i > 0; i--)
996 {
997 const std::string& subdir = search_subdir[i - 1];
998 std::string prefix = path_join(datapath, subdir);
999 TEST_TRY_FILE_WITH_PREFIX(prefix);
1000 }
1001 }
1002 }
1003#endif
1004#endif
1005 const char* type = findDirectory ? "directory" : "data file";
1006 if (required || checkTestData)
1007 CV_Error(cv::Error::StsError, cv::format("OpenCV tests: Can't find required %s: %s", type, relative_path.c_str()));
1008 throw SkipTestException(cv::format("OpenCV tests: Can't find %s: %s", type, relative_path.c_str()));
1009}

1021inline static std::string getSnippetFromConfig(const std::string & start, const std::string & end)
1022{
1023 const std::string buildInfo = cv::getBuildInformation();
1024 size_t pos1 = buildInfo.find(start);
1025 if (pos1 != std::string::npos)
1026 {
1027 pos1 += start.length();
1028 pos1 = buildInfo.find_first_not_of(" \t\n\r", pos1);
1029 }
1030 size_t pos2 = buildInfo.find(end, pos1);
1031 if (pos2 != std::string::npos)
1032 {
1033 pos2 = buildInfo.find_last_not_of(" \t\n\r", pos2);
1034 }
1035 if (pos1 != std::string::npos && pos2 != std::string::npos && pos1 < pos2)
1036 {
1037 return buildInfo.substr(pos1, pos2 - pos1 + 1);
1038 }
1039 return std::string();
1040}
