95Mat TFLiteImporter::parseTensor(const Tensor& tensor)
96{
97 std::vector<int> shape;
98 const auto tensor_shape = tensor.shape();
99 if (tensor_shape && tensor_shape->size())
100 shape.assign(tensor_shape->begin(), tensor_shape->end());
101 else
102 shape.resize(1, 1);
103 int bufferIdx = tensor.buffer();
104 CV_Assert(bufferIdx != 0); // 0th buffer is a no-data buffer
105 const Buffer* buffer = model->buffers()->Get(bufferIdx);
106 CV_Assert(buffer);
107 const auto buffer_data = buffer->data();
108 if (!buffer_data)
109 return Mat();
110
111 const void* data = buffer_data->data();
112
113 int dtype = -1;
114 switch (tensor.type()) {
115 case TensorType_FLOAT32:
116 dtype = CV_32F;
117 break;
118 case TensorType_INT32:
119 dtype = CV_32S;
120 break;
121 case TensorType_FLOAT16:
122 dtype = CV_16F;
123 break;
124 case TensorType_INT8:
125 dtype = CV_8S;
126 break;
127 default:
128 CV_Error(Error::StsNotImplemented, format("Parse tensor with type %s", EnumNameTensorType(tensor.type())));
129 }
130 Mat res = Mat(shape, dtype, const_cast<void*>(data));
131 // workaround for scalars support
132 if (!tensor_shape || shape.size() == 1)
133 res.size.dims = res.dims = 1;
134 return res;
135}

137TFLiteImporter::TFLiteImporter(Net& dstNet, const char* modelBuffer, size_t bufSize, bool newEngine)
138 : newEngine(newEngine), dstNet(dstNet), dispatch(buildDispatchMap())
139{
140 flatbuffers::Verifier verifier((const uint8_t*)modelBuffer, bufSize);
141 if (!VerifyModelBuffer(verifier)) {
142 CV_Error(Error::StsError, "DNN/TFLite: model is incorrect");
143 }
144
145 model = GetModel(modelBuffer);
146 CV_Assert(model);
147 CV_Assert(model->subgraphs());
148 CV_Assert(model->buffers());
149 CV_CheckEQ((size_t)model->subgraphs()->size(), 1u, "");
150
151 modelTensors = model->subgraphs()->Get(0)->tensors();
152 CV_Assert(modelTensors);
153 for (int i = 0; i < modelTensors->size(); ++i) {
154 const Tensor* tensor = modelTensors->Get(i);
155 CV_Assert(tensor);
156 if (tensor->buffer() != 0) {
157 allTensors[i] = parseTensor(*tensor);
158 }
159 }
160
161 populateNet();
162}

168 switch (t_shape->size()) {
169 case 5: return DNN_LAYOUT_NDHWC;

176void TFLiteImporter::populateNet()
177{
178 CV_Assert(model);
179 const auto model_subgraphs = model->subgraphs();
180 CV_Assert(model_subgraphs);
181 const SubGraph* subgraph = model_subgraphs->Get(0);
182 CV_Assert(subgraph);
183 const auto subgraph_inputs = subgraph->inputs();
184 CV_Assert(subgraph_inputs);
185 const auto subgraph_operators = subgraph->operators();
186 CV_Assert(subgraph_operators);
187 const auto opCodes = model->operator_codes();
188 CV_Assert(opCodes);
189
190 CV_Assert(modelTensors);
191 layouts.resize(modelTensors->size(), DNN_LAYOUT_UNKNOWN);
192 size_t subgraph_inputs_size = subgraph_inputs->size();
193 std::vector<std::string> inputsNames(subgraph_inputs_size);
194 std::vector<MatShape> inputsShapes(subgraph_inputs_size);
195
196 // NEW ENGINE
197 Net::Impl* netImpl = dstNet.getImpl();
198 std::vector<Arg> modelInputs, modelOutputs;
199 Ptr<Graph> curr_graph;
200
201 for (size_t i = 0; i < subgraph_inputs_size; ++i)
202 {
203 size_t idx = subgraph_inputs->Get(i);
204 layerIds[idx] = std::make_pair(0, i);
205 const auto tensor = modelTensors->Get(idx);
206 if (!tensor)
207 CV_Error(Error::StsError, cv::format("DNN/TFLite: subgraph input %zu (%zu) is NULL", i, idx));
208 layouts[idx] = estimateLayout(*tensor);
209
210 // Keep info about origin inputs names and shapes
211 inputsNames[i] = tensor->name()->str();
212 MatShape shape(tensor->shape()->begin(), tensor->shape()->end());
213 if (layouts[idx] == DNN_LAYOUT_NHWC) {
214 CV_CheckEQ(shape.size(), (size_t)4, "");
215 std::swap(shape[2], shape[3]);
216 std::swap(shape[1], shape[2]);
217 }
218 inputsShapes[i] = shape;
219
220 if (newEngine)
221 {
222 modelInputs.push_back(netImpl->newArg(tensor->name()->str(), DNN_ARG_INPUT));
223 netImpl->args.at(modelInputs.back().idx).type = CV_32F;
224 netImpl->args.at(modelInputs.back().idx).shape = shape;
225 }
226 }
227
228 if (!newEngine)
229 {
230 dstNet.setInputsNames(inputsNames);
231 for (size_t i = 0; i < subgraph_inputs_size; ++i)
232 {
233 dstNet.setInputShape(inputsNames[i], inputsShapes[i]);
234 }
235 }
236
237 const auto& all_operators = *subgraph_operators;
238 const size_t all_operators_size = all_operators.size();
239
240 if (newEngine)
241 {
242 const auto last_op = all_operators[all_operators_size - 1];
243 const auto op_outputs = last_op->outputs();
244 std::string type = EnumNameBuiltinOperator(
245 BuiltinOperator(opCodes->Get(last_op->opcode_index())->deprecated_builtin_code()));
246 for (int idx : *op_outputs)
247 {
248 std::string tensorName = modelTensors->Get(idx)->name()->str();
249 modelOutputs.push_back(netImpl->newArg(tensorName, DNN_ARG_OUTPUT));
250
251 // TFLite_Detection_PostProcess layer returns 4 outputs
252 // (num_bboxes, bboxes, classes, conf)
253 // but our detection_output layer returns one output with all the data
254 if (type == "TFLite_Detection_PostProcess")
255 break;
256 }
257 }
258
259 for (size_t op_idx = 0; op_idx < all_operators_size; ++op_idx)
260 {
261 const auto op = all_operators[op_idx];
262 CV_Assert(op);
263 const auto op_inputs = op->inputs();
264 CV_Assert(op_inputs);
265 const auto op_outputs = op->outputs();
266 CV_Assert(op_outputs);
267 int idx = op->opcode_index();
268
269 LayerParams layerParams;
270 layerParams.name = modelTensors->Get(op_outputs->Get(0))->name()->str();
271
272 std::string type = EnumNameBuiltinOperator(BuiltinOperator(opCodes->Get(idx)->deprecated_builtin_code()));
273 if (type == "CUSTOM") {
274 type = opCodes->Get(idx)->custom_code()->str();
275 }
276
277 CV_LOG_DEBUG(NULL, "DNN/TFLite: processing operator (" << op_idx << "/" << all_operators_size << ") with " << op_inputs->size() << " inputs: "
278 << cv::format("[%s]:(%s)", type.c_str(), layerParams.name.c_str()));
279
280 try
281 {
282 if (type == "DEQUANTIZE") {
283 // Convert from FP16 to FP32
284 Mat data = allTensors[op_inputs->Get(0)];
285 if (!data.empty()) {
286 // Dequantize a buffer
287 Mat dataFP32;
288 data.convertTo(dataFP32, CV_32F);
289 // workaround for scalars support
290 dataFP32.size.dims = dataFP32.dims = data.dims;
291 allTensors[op_outputs->Get(0)] = dataFP32;
292 continue;
293 }
294 }
295
296 DispatchMap::const_iterator iter = dispatch.find(type);
297 if (iter == dispatch.end())
298 CV_Error(Error::StsNotImplemented, "Unsupported operator type " + type);
299
300 CALL_MEMBER_FN(*this, iter->second)(*op, type, layerParams);
301 }
302 catch (const cv::Exception& e)
303 {
304 CV_LOG_ERROR(NULL, "DNN/TFLite: Problem during import of operator "
305 << cv::format("[%s]:(%s)", type.c_str(), layerParams.name.c_str())
306 << " (" << op_idx << "/" << all_operators_size << "). Exception: " << e.what());
307 if (DNN_DIAGNOSTICS_RUN)
308 {
309 continue;
310 }
311 throw;
312 }
313 }
314 if (newEngine)
315 {
316 Ptr<Graph> curr_graph = netImpl->newGraph(subgraph->name()->str(), modelInputs, true);
317 curr_graph->setOutputs(modelOutputs);
318 curr_graph->setProg(curProg);
319
320 netImpl->mainGraph = curr_graph;
321 netImpl->modelFormat = DNN_MODEL_TFLITE;
322 netImpl->originalLayout = DATA_LAYOUT_NCHW; // TODO Should we set NHWC?
323 netImpl->prepareForInference();
324 }

363void TFLiteImporter::addLayer(LayerParams& layerParams, const Operator& op, bool additionalPreLayer, bool additionalPostLayer) {
364 const auto op_inputs = op.inputs();
365 const auto op_outputs = op.outputs();
366
367 // Collect input blobs
368 if (layerParams.blobs.empty()) {
369 for (int idx : *op_inputs) {
370 if (layerIds.find(idx) != layerIds.end()) {
371 continue; // Output from a different layer
372 }
373 Mat blob = allTensors[idx];
374 layerParams.blobs.push_back(blob.u ? blob : blob.clone()); // some tensors are owned by OpenCV
375 }
376 } else {
377 for (auto& blob : layerParams.blobs) {
378 CV_Assert(blob.u);
379 }
380 }
381
382 int dtype = CV_32F;
383 if (isInt8(op)) {
384 dtype = CV_8S;
385 if (layerParams.type != "Quantize")
386 layerParams.type += "Int8";
387
388 if (!layerParams.has("zeropoints")) {
389 float inpScale, outScale;
390 int inpZero, outZero;
391 getQuantParams(op, inpScale, inpZero, outScale, outZero);
392
393 layerParams.set("input_scale", inpScale);
394 layerParams.set("input_zeropoint", inpZero);
395 layerParams.set("scales", outScale);
396 layerParams.set("zeropoints", outZero);
397 }
398 }
399
400 int layerId = -1;
401 if (!newEngine)
402 layerId = dstNet.addLayer(layerParams.name, layerParams.type, dtype, layerParams);
403
404 // Connect layer to inputs
405 int i = 0;
406 std::vector<string> inputTensors;
407 std::vector<DataLayout> inpLayouts;
408 for (int idx : *op_inputs) {
409 if (layerIds.find(idx) == layerIds.end()) {
410 continue; // Const input
411 }
412 inpLayouts.push_back(layouts[idx]);
413
414 if (newEngine)
415 {
416 std::string tensorName = modelTensors->Get(idx)->name()->str();
417 if (additionalPreLayer)
418 tensorName += "_additional_pre_layer";
419 inputTensors.push_back(tensorName);
420 }
421 else
422 {
423 auto it = layerIds.find(idx);
424 CV_Assert(it != layerIds.end());
425 dstNet.connect(it->second.first, it->second.second, layerId, i++);
426 }
427 }
428
429 // Predict output layout. Some layer-specific parsers may set them explicitly.
430 // Otherwise, propagate input layout.
431 if (layouts[op_outputs->Get(0)] == DNN_LAYOUT_UNKNOWN) {
432 DataLayout predictedLayout = DNN_LAYOUT_UNKNOWN;
433 for (auto layout : inpLayouts) {
434 if (layout != DNN_LAYOUT_UNKNOWN) {
435 if (predictedLayout == DNN_LAYOUT_UNKNOWN)
436 predictedLayout = layout;
437 else if (predictedLayout != layout) {
438 predictedLayout = DNN_LAYOUT_UNKNOWN;
439 break;
440 }
441 }
442 }
443 layouts[op_outputs->Get(0)] = predictedLayout;
444 }
445
446 // Register outputs
447 i = 0;
448 std::vector<string> outputTensors;
449 for (int idx : *op_outputs) {
450 if (newEngine)
451 {
452 std::string tensorName = modelTensors->Get(idx)->name()->str();
453 if (additionalPostLayer)
454 tensorName = tensorName + "_additional_post_layer";
455 outputTensors.push_back(tensorName);
456 layerIds[idx] = std::make_pair(-1, -1);
457 }
458 else
459 layerIds[idx] = std::make_pair(layerId, i++);
460
461 // TFLite_Detection_PostProcess layer returns 4 outputs
462 // (num_bboxes, bboxes, classes, conf)
463 // but our detection_output layer returns one output with all the data
464 if (layerParams.type == "DetectionOutput")
465 break;
466 }
467
468 if (newEngine)
469 {
470 if (additionalPostLayer)
471 layerParams.name += "_pre";
472 addLayer(layerParams, inputTensors, outputTensors);
473 }
474}

488void TFLiteImporter::parseConvolution(const Operator& op, const std::string& opcode, LayerParams& layerParams) {
489 layerParams.type = "Convolution";
490
491 int inpId = op.inputs()->Get(0);
492 bool additionalPreLayer = false;
493 if (layouts[inpId] == DNN_LAYOUT_UNKNOWN && modelTensors->Get(inpId)->shape()->size() == 4)
494 {
495 int permId = addPermuteLayer({0, 3, 1, 2}, layerParams.name + "/permute_input", layerIds[inpId], isInt8(op) ? CV_8S : CV_32F, op.inputs()->Get(0)); // NHWC -> NCHW
496 layerIds[inpId] = std::make_pair(permId, 0);
497 layouts[op.outputs()->Get(0)] = DNN_LAYOUT_NHWC;
498 additionalPreLayer = true;
499 }
500
501 auto options = reinterpret_cast<const Conv2DOptions*>(op.builtin_options());
502 layerParams.set("pad_mode", EnumNamePadding(options->padding()));
503 layerParams.set("stride_w", options->stride_w());
504 layerParams.set("stride_h", options->stride_h());
505 layerParams.set("dilation_w", options->dilation_w_factor());
506 layerParams.set("dilation_h", options->dilation_h_factor());
507
508 // Get filter size
509 int filterIdx = op.inputs()->Get(1);
510 Mat filter = allTensors[filterIdx];
511 int oc = filter.size[0];
512 int kh = filter.size[1];
513 int kw = filter.size[2];
514 layerParams.set("kernel_w", kw);
515 layerParams.set("kernel_h", kh);
516 layerParams.set("num_output", oc);
517
518 bool isInt8 = filter.depth() == CV_8S;
519
520 // Fill convolutions blobs here because of two reasons:
521 // 1. Kernel transposition
522 // 2. Extra blob with kernel scales in case of INT8 mode
523 bool hasBias = op.inputs()->size() > 2;
524 layerParams.blobs.resize(1 + (int)hasBias + (int)isInt8);
525 if (hasBias) {
526 Mat bias = allTensors[op.inputs()->Get(2)];
527 layerParams.blobs[1] = bias.u ? bias : bias.clone();
528 }
529
530 // Reorder filter data from OHWI to OIHW and change shape correspondingly.
531 transposeND(filter, {0, 3, 1, 2}, layerParams.blobs[0]);
532
533 if (isInt8) {
534 float inpScale, outScale;
535 int inpZero, outZero;
536 getQuantParams(op, inpScale, inpZero, outScale, outZero);
537
538 layerParams.blobs[2] = Mat(oc, 1, CV_32F);
539 auto filterScales = modelTensors->Get(filterIdx)->quantization()->scale();
540 if (filterScales->size() == 1) {
541 layerParams.blobs[2].setTo(inpScale * filterScales->Get(0) / outScale);
542 } else {
543 CV_CheckEQ((int)filterScales->size(), oc, "TFLite: number of filter quantization scales must match the number of output channels");
544 for (size_t i = 0; i < filterScales->size(); ++i) {
545 layerParams.blobs[2].at<float>(i) = inpScale * filterScales->Get(i) / outScale;
546 }
547 }
548
549 if (hasBias) {
550 Mat bias = layerParams.blobs[1].reshape(1, oc);
551 Mat weights_2d = layerParams.blobs[0].reshape(1, oc);
552 for (int i = 0; i < oc; i++)
553 {
554 bias.at<int>(i) -= inpZero * (cv::sum(weights_2d.row(i))[0]);
555 }
556 }
557 }
558
559 std::string fusedActivationType = EnumNameActivationFunctionType(options->fused_activation_function());
560 bool haveFusedActivation = fusedActivationType != "NONE";
561 addLayer(layerParams, op, additionalPreLayer, haveFusedActivation);
562 parseFusedActivation(op, options->fused_activation_function());
563}

565void TFLiteImporter::parseDWConvolution(const Operator& op, const std::string& opcode, LayerParams& layerParams) {
566 layerParams.type = "Convolution";
567
568 auto options = reinterpret_cast<const DepthwiseConv2DOptions*>(op.builtin_options());
569 layerParams.set("pad_mode", EnumNamePadding(options->padding()));
570 layerParams.set("stride_w", options->stride_w());
571 layerParams.set("stride_h", options->stride_h());
572 layerParams.set("dilation_w", options->dilation_w_factor());
573 layerParams.set("dilation_h", options->dilation_h_factor());
574
575 int filterIdx = op.inputs()->Get(1);
576 Mat filter = allTensors[filterIdx];
577 int kh = filter.size[1];
578 int kw = filter.size[2];
579 int oc = filter.size[3];
580 layerParams.set("kernel_w", kw);
581 layerParams.set("kernel_h", kh);
582 layerParams.set("num_output", oc);
583 layerParams.set("group", oc);
584
585 bool isInt8 = filter.depth() == CV_8S;
586
587 // Fill convolutions blobs here because of two reasons:
588 // 1. Kernel transposition
589 // 2. Extra blob with kernel scales in case of INT8 mode
590 bool hasBias = op.inputs()->size() > 2;
591 layerParams.blobs.resize(1 + (int)hasBias + (int)isInt8);
592 if (hasBias) {
593 Mat bias = allTensors[op.inputs()->Get(2)];
594 layerParams.blobs[1] = bias.u ? bias : bias.clone();
595 }
596
597 transposeND(filter, {3, 0, 1, 2}, layerParams.blobs[0]);
598
599 if (isInt8) {
600 float inpScale, outScale;
601 int inpZero, outZero;
602 getQuantParams(op, inpScale, inpZero, outScale, outZero);
603
604 layerParams.blobs[2] = Mat(oc, 1, CV_32F);
605 auto filterScales = modelTensors->Get(filterIdx)->quantization()->scale();
606 if (filterScales->size() == 1) {
607 layerParams.blobs[2].setTo(inpScale * filterScales->Get(0) / outScale);
608 } else {
609 CV_CheckEQ((int)filterScales->size(), oc, "TFLite: number of filter quantization scales must match the number of output channels");
610 for (size_t i = 0; i < filterScales->size(); ++i) {
611 layerParams.blobs[2].at<float>(i) = inpScale * filterScales->Get(i) / outScale;
612 }
613 }
614
615 if (hasBias) {
616 Mat bias = layerParams.blobs[1].reshape(1, oc);
617 Mat weights_2d = layerParams.blobs[0].reshape(1, oc);
618 for (int i = 0; i < oc; i++)
619 {
620 bias.at<int>(i) -= inpZero * (cv::sum(weights_2d.row(i))[0]);
621 }
622 }
623 }
624
625 std::string fusedActivationType = EnumNameActivationFunctionType(options->fused_activation_function());
626 bool haveFusedActivation = fusedActivationType != "NONE";
627 addLayer(layerParams, op, false, haveFusedActivation);
628 parseFusedActivation(op, options->fused_activation_function());
629}

632 layerParams.type = "Padding";

635 CV_CheckTypeEQ(paddings.type(), CV_32S, "");

654void TFLiteImporter::parseEltwise(const Operator& op, const std::string& opcode, LayerParams& layerParams) {
655 bool isOpInt8 = isInt8(op);
656 ActivationFunctionType activ = ActivationFunctionType_NONE;
657 layerParams.type = isOpInt8 ? "Eltwise" : "NaryEltwise";
658 if (opcode == "ADD") {
659 auto options = reinterpret_cast<const AddOptions*>(op.builtin_options());
660 activ = options->fused_activation_function();
661 layerParams.set("operation", "sum");
662 }
663 else if (opcode == "MUL") {
664 auto options = reinterpret_cast<const MulOptions*>(op.builtin_options());
665 activ = options->fused_activation_function();
666 layerParams.set("operation", "mul");
667 }
668 else if (opcode == "DIV") {
669 auto options = reinterpret_cast<const DivOptions*>(op.builtin_options());
670 activ = options->fused_activation_function();
671 layerParams.set("operation", "div");
672 }
673 else if (opcode == "SUB" && !isOpInt8) {
674 auto options = reinterpret_cast<const SubOptions*>(op.builtin_options());
675 activ = options->fused_activation_function();
676 layerParams.set("operation", "sub");
677 }
678 else if (opcode == "NEG") {
679 layerParams.type = "Scale";
680 layerParams.blobs.resize(1, Mat(1, 1, CV_32F, Scalar(-1)));
681 }
682 else if (opcode == "SQUARED_DIFFERENCE" && !isOpInt8) {
683 layerParams.set("operation", "sub");
684 }
685 else if (opcode == "RSQRT" && !isOpInt8) {
686 layerParams.type = "Sqrt";
687 }
688 else if (opcode == "SQRT" && !isOpInt8) {
689 layerParams.type = "Sqrt";
690 }
691 else if (opcode == "MAXIMUM" && !isOpInt8) {
692 layerParams.set("operation", "max");
693 }
694 else if (opcode == "MINIMUM" && !isOpInt8) {
695 layerParams.set("operation", "min");
696 }else {
697 CV_Error(Error::StsNotImplemented, cv::format("DNN/TFLite: Unknown opcode for %s Eltwise layer '%s'", isOpInt8 ? "INT8" : "FP32", opcode.c_str()));
698 }
699
700 if (isOpInt8) {
701 const Tensor* out = modelTensors->Get(op.outputs()->Get(0));
702 float outScale = out->quantization()->scale()->Get(0);
703 int outZero = out->quantization()->zero_point()->Get(0);
704
705 const size_t numInps = op.inputs()->size();
706 std::vector<float> inputScales(numInps);
707 std::vector<int> inputZeros(numInps);
708 std::vector<float> coeffs(numInps);
709 float offset = outZero;
710 for (int i = 0; i < numInps; ++i) {
711 const Tensor* inp = modelTensors->Get(op.inputs()->Get(i));
712 float inpScale = inp->quantization()->scale()->Get(0);
713 int inpZero = inp->quantization()->zero_point()->Get(0);
714 inputScales[i] = inpScale;
715 inputZeros[i] = inpZero;
716 coeffs[i] = inpScale / outScale;
717 offset -= coeffs[i] * inpZero;
718 }
719
720 layerParams.set("input_scales", DictValue::arrayReal(inputScales.data(), numInps));
721 layerParams.set("input_zeropoints", DictValue::arrayInt(inputZeros.data(), numInps));
722 layerParams.set("coeff", DictValue::arrayReal(coeffs.data(), numInps));
723 layerParams.set("offset", offset);
724 layerParams.set("scales", outScale);
725 layerParams.set("zeropoints", outZero);
726 }
727
728 // Force all inputs to be in graph, not as blobs
729 for (int idx : *op.inputs()) {
730 if (layerIds.find(idx) != layerIds.end()) {
731 continue; // Output from a different layer
732 }
733 Mat blob = allTensors[idx];
734 if (layouts[op.inputs()->Get(0)] == DNN_LAYOUT_NHWC && blob.dims == 1) {
735 blob = blob.reshape(1, {1, (int)blob.total(), 1, 1});
736 }
737 addConstLayer(blob, idx);
738 }
739
740
741 std::string fusedActivationType = EnumNameActivationFunctionType(activ);
742 bool haveFusedActivation = fusedActivationType != "NONE";
743 addLayer(layerParams, op, false, haveFusedActivation || opcode == "SQUARED_DIFFERENCE" || opcode == "RSQRT");
744 parseFusedActivation(op, activ);
745
746 if (opcode == "SQUARED_DIFFERENCE" || opcode == "RSQRT") {
747 if (haveFusedActivation)
748 CV_LOG_WARNING(NULL, format("%s with fused activation on new engine is not tested", opcode.c_str()));
749 LayerParams lp;
750 if (opcode == "RSQRT")
751 lp.type = "Reciprocal";
752 else {
753 lp.type = "Power";
754 lp.set("power", 2);
755 }
756 if (newEngine) {
757 std::string tensorName = modelTensors->Get(op.outputs()->Get(0))->name()->str();
758 addLayer(lp, {tensorName + "_additional_post_layer"}, {tensorName});
759 layerIds[op.outputs()->Get(0)] = std::make_pair(-1, -1);
760 } else {
761 int id = dstNet.addLayerToPrev(layerParams.name + "/post", lp.type, isOpInt8 ? CV_8S : CV_32F, lp);
762 layerIds[op.outputs()->Get(0)] = std::make_pair(id, 0);
763 }
764 }
765}

824void TFLiteImporter::parseReshape(const Operator& op, const std::string& opcode, LayerParams& layerParams) {
825 DataLayout inpLayout = layouts[op.inputs()->Get(0)];
826
827 layerParams.type = "Reshape";
828 std::vector<int> shape;
829 if (op.inputs()->size() > 1) {
830 shape = allTensors[op.inputs()->Get(1)];
831 } else {
832 auto options = op.builtin_options_as_ReshapeOptions();
833 CV_Assert(options);
834 shape.assign(options->new_shape()->begin(), options->new_shape()->end());
835 }
836
837 bool additionalPreLayer = false;
838 if (inpLayout == DNN_LAYOUT_NHWC) {
839 if (shape.size() == 4) {
840 // Keep data but change a shape to OpenCV's NCHW order
841 std::swap(shape[2], shape[3]);
842 std::swap(shape[1], shape[2]);
843 } else {
844 // Permute to NCHW entire data and reshape to given a shape
845 std::vector<int> order = {0, 2, 3, 1};
846 const std::string name = layerParams.name + "/permute";
847 auto inpId = layerIds[op.inputs()->Get(0)];
848 int permId = addPermuteLayer(order, name, inpId, isInt8(op) ? CV_8S : CV_32F, op.inputs()->Get(0)); // NCHW -> NHWC
849 layerIds[op.inputs()->Get(0)] = std::make_pair(permId, 0);
850 layouts[op.outputs()->Get(0)] = DNN_LAYOUT_NCHW;
851 additionalPreLayer = true;
852 }
853 }
854 layerParams.set("dim", DictValue::arrayInt<int*>(shape.data(), shape.size()));
855 addLayer(layerParams, op, additionalPreLayer);
856}

858void TFLiteImporter::parseConcat(const Operator& op, const std::string& opcode, LayerParams& layerParams) {
859 layerParams.type = "Concat";
860 auto options = reinterpret_cast<const ConcatenationOptions*>(op.builtin_options());
861 int axis = options->axis();
862
863 bool hasNHWCInput = false;
864 for (int idx : *op.inputs()) {
865 DataLayout inpLayout = layouts[idx];
866 if (inpLayout == DNN_LAYOUT_NHWC) {
867 // OpenCV works in NCHW data layout. So change the axis correspondingly.
868 axis = normalize_axis(axis, 4);
869 static const int remap[] = {0, 2, 3, 1};
870 axis = remap[axis];
871 hasNHWCInput = true;
872 break;
873 }
874 }
875 layerParams.set("axis", axis);
876
877 // Force all inputs to be in graph, not as blobs
878 for (int idx : *op.inputs()) {
879 if (layerIds.find(idx) != layerIds.end()) {
880 continue; // Output from a different layer
881 }
882 Mat blob = allTensors[idx];
883 if (hasNHWCInput && blob.dims == 4)
884 {
885 Mat nchwBlob;
886 transposeND(blob, {0, 3, 1, 2}, nchwBlob);
887 blob = nchwBlob;
888 }
889 addConstLayer(blob, idx);
890 }
891
892 std::string fusedActivationType = EnumNameActivationFunctionType(options->fused_activation_function());
893 bool haveFusedActivation = fusedActivationType != "NONE";
894 addLayer(layerParams, op, false, haveFusedActivation);
895 parseFusedActivation(op, options->fused_activation_function());
896}

898void TFLiteImporter::parsePack(const Operator& op, const std::string& opcode, LayerParams& layerParams) {
899 auto options = reinterpret_cast<const PackOptions*>(op.builtin_options());
900 int axis = options->axis();
901
902 DataLayout inpLayout = layouts[op.inputs()->Get(0)];
903 if (inpLayout == DNN_LAYOUT_NHWC) {
904 // OpenCV works in NCHW data layout. So change the axis correspondingly.
905 axis = normalize_axis(axis, 5); // 5 because Pack adds a new axis so -1 would mean 4
906 static const int remap[] = {0, 1, 3, 4, 2};
907 axis = remap[axis];
908 }
909
910 // Replace Pack layer to Reshape + Concat
911 // Use a set because there are models which replicate single layer data by Pack.
912 std::set<int> op_inputs(op.inputs()->begin(), op.inputs()->end());
913 std::map<int, std::pair<int, int> > originLayerIds;
914 for (int inp : op_inputs) {
915 auto inpId = layerIds[inp];
916 int dims = modelTensors->Get(inp)->shape()->size();
917
918 std::vector<int> shape{1, -1};
919 if (axis == dims) {
920 std::swap(shape[0], shape[1]);
921 }
922 const auto name = modelTensors->Get(inp)->name()->str() + "/reshape";
923 int reshapeId = addReshapeLayer(shape, axis == dims ? dims - 1 : axis, 1,
924 name, inpId, isInt8(op) ? CV_8S : CV_32F, inp);
925
926 originLayerIds[inp] = layerIds[inp];
927 layerIds[inp] = std::make_pair(reshapeId, 0);
928 }
929 layerParams.type = "Concat";
930 layerParams.set("axis", axis);
931 addLayer(layerParams, op, true, false);
932
933 // Restore origin layer inputs
934 for (const auto& ids : originLayerIds) {
935 layerIds[ids.first] = ids.second;
936 }
937}

960void TFLiteImporter::parseTranspose(const Operator& op, const std::string& opcode, LayerParams& layerParams)
961{
962 layerParams.type = "Permute";
963 std::vector<int> perm = allTensors[op.inputs()->Get(1)];
964
965 DataLayout inpLayout = layouts[op.inputs()->Get(0)];
966 if (inpLayout == DNN_LAYOUT_NHWC && perm.size() == 4) {
967
968 // OpenCV operates under the assumption that NCHW format, whereas TFLite defaults to NHWC.
969 // Therfore, to align these layouts, the axes of the permutation vector should be adjusted accordingly.
970 // For implementation details, please refer to the disscusion:
971 // https://github.com/opencv/opencv/pull/25297#issuecomment-2049762298
972
973 if (perm[0] != 0) {
974 CV_Error(Error::StsParseError, "The first axis should not be permuted.");
975 }
976 if (perm[1] == 1 && perm[2] == 2 && perm[3] == 3) {
977 std::vector<int> orderLP = {0, 1, 2, 3};
978 layerParams.set("order", DictValue::arrayInt<int*>(orderLP.data(), orderLP.size()));
979 layouts[op.outputs()->Get(0)] = DNN_LAYOUT_NCHW;
980 }
981 else if (perm[1] == 1 && perm[2] == 3 && perm[3] == 2) {
982 std::vector<int> orderLP = {0, 3, 2, 1};
983 layerParams.set("order", DictValue::arrayInt<int*>(orderLP.data(), orderLP.size()));
984 }
985 else if (perm[1] == 2 && perm[2] == 1 && perm[3] == 3) {
986 std::vector<int> orderLP = {0, 1, 3, 2};
987 layerParams.set("order", DictValue::arrayInt<int*>(orderLP.data(), orderLP.size()));
988 layouts[op.outputs()->Get(0)] = DNN_LAYOUT_NCHW;
989 }
990 else if (perm[1] == 2 && perm[2] == 3 && perm[3] == 1) {
991 std::vector<int> orderLP = {0, 2, 3, 1};
992 layerParams.set("order", DictValue::arrayInt<int*>(orderLP.data(), orderLP.size()));
993 }
994
995 }
996 else {
997 layerParams.set("order", DictValue::arrayInt<int*>(perm.data(), perm.size()));
998 }
999
1000 addLayer(layerParams, op);
1001}

1139 layerParams.set("kernel_h", kh);

1154 layerParams.set("adj_h", (outH - kh) % params->stride_height);

1168 for (int i_h = 0; i_h < kh; i_h++) {

1209void TFLiteImporter::parseStridedSlice(const Operator& op, const std::string& opcode, LayerParams& layerParams) {
1210 layerParams.type = "Slice";
1211 auto options = op.builtin_options_as_StridedSliceOptions();
1212 CV_Assert(options);
1213 int endMask = options->end_mask();
1214 if (options->new_axis_mask())
1215 CV_Error(Error::StsNotImplemented, "New axis during StridedSlice");
1216 int shrinkMask = options->shrink_axis_mask();
1217
1218 Mat begins = allTensors[op.inputs()->Get(1)];
1219 Mat ends = allTensors[op.inputs()->Get(2)];
1220 Mat strides = allTensors[op.inputs()->Get(3)];
1221
1222 CV_CheckTypeEQ(begins.type(), CV_32SC1, "");
1223 CV_CheckTypeEQ(ends.type(), CV_32SC1, "");
1224 CV_CheckTypeEQ(strides.type(), CV_32SC1, "");
1225 const int num = begins.total();
1226 CV_Assert_N(num == ends.total(), num == strides.total());
1227 for (int i = 0; i < num; ++i)
1228 {
1229 if (endMask & (1 << i))
1230 ends.at<int>(i) = INT_MAX;
1231 }
1232 if (begins.total() == 4 && layouts[op.inputs()->Get(0)] == DNN_LAYOUT_NHWC)
1233 {
1234 // Swap NHWC parameters' order to NCHW.
1235 std::swap(begins.at<int>(2), begins.at<int>(3));
1236 std::swap(begins.at<int>(1), begins.at<int>(2));
1237 std::swap(ends.at<int>(2), ends.at<int>(3));
1238 std::swap(ends.at<int>(1), ends.at<int>(2));
1239 std::swap(strides.at<int>(2), strides.at<int>(3));
1240 std::swap(strides.at<int>(1), strides.at<int>(2));
1241 }
1242 layerParams.set("begin", DictValue::arrayInt((int*)begins.data, begins.total()));
1243 layerParams.set("end", DictValue::arrayInt((int*)ends.data, ends.total()));
1244 layerParams.set("steps", DictValue::arrayInt((int*)strides.data, strides.total()));
1245
1246 int lastShrinkAxis = -1;
1247 for (int axis = 0; axis < num; ++axis)
1248 {
1249 if (shrinkMask & (1 << axis))
1250 lastShrinkAxis = axis;
1251 }
1252 std::string layerName = layerParams.name;
1253 if (!newEngine && lastShrinkAxis != -1)
1254 {
1255 layerParams.name += "/slice";
1256 }
1257
1258 addLayer(layerParams, op, false, lastShrinkAxis != -1);
1259
1260 for (int axis = 0; axis < num; ++axis)
1261 {
1262 if (!(shrinkMask & (1 << axis)))
1263 continue;
1264 if (newEngine)
1265 {
1266 if (axis != lastShrinkAxis)
1267 CV_LOG_WARNING(NULL, "StridedSlice with multiple axes shrink in new engine is not tested");
1268 addFlattenLayer(axis, axis + 1, layerName,
1269 layerIds[op.outputs()->Get(0)], isInt8(op) ? CV_8S : CV_32F, op.outputs()->Get(0));
1270 }
1271 else
1272 {
1273 std::string name = (axis == lastShrinkAxis) ? layerName : format("%s/shrink_axis_%d", layerName.c_str(), axis);
1274 int layerId = addFlattenLayer(axis, axis + 1, name,
1275 layerIds[op.outputs()->Get(0)], isInt8(op) ? CV_8S : CV_32F, op.inputs()->Get(0));
1276 layerIds[op.inputs()->Get(0)] = std::make_pair(layerId, 0);
1277 }
1278 }

1305void TFLiteImporter::parseDetectionPostProcess(const Operator& op, const std::string& opcode, LayerParams& layerParams) {
1306 // Parse parameters;
1307 std::vector<std::string> keys(1, "");
1308 const uint8_t* data = op.custom_options()->Data();
1309 int offset = 0;
1310
1311 // Read zero delimited keys
1312 while (data[offset] != 10 && offset < op.custom_options()->size()) {
1313 if (data[offset]) {
1314 keys.back() += data[offset];
1315 } else {
1316 keys.emplace_back("");
1317 }
1318 offset += 1;
1319 }
1320 keys.pop_back();
1321 std::sort(keys.begin(), keys.end());
1322
1323 // TODO: Replace empirical offset to something more reliable.
1324 offset += 25;
1325 std::map<std::string, uint32_t> parameters;
1326 for (int i = 0; i < keys.size(); ++i) {
1327 parameters[keys[i]] = *reinterpret_cast<const uint32_t*>(data + offset + i * 4);
1328 }
1329
1330 parameters["num_classes"] = modelTensors->Get(op.inputs()->Get(1))->shape()->Get(2);
1331
1332 layerParams.type = "DetectionOutput";
1333 layerParams.set("num_classes", parameters["num_classes"]);
1334 layerParams.set("share_location", true);
1335 layerParams.set("background_label_id", parameters["num_classes"] + 1);
1336 layerParams.set("nms_threshold", *(float*)&parameters["nms_iou_threshold"]);
1337 layerParams.set("confidence_threshold", *(float*)&parameters["nms_score_threshold"]);
1338 layerParams.set("top_k", parameters["max_detections"]);
1339 layerParams.set("keep_top_k", parameters["max_detections"]);
1340 layerParams.set("code_type", "CENTER_SIZE");
1341 layerParams.set("loc_pred_transposed", true);
1342
1343 // Replace third input from tensor to Const layer with the priors
1344 Mat priors = allTensors[op.inputs()->Get(2)].clone();
1345
1346 // Change priors data from (ycenter, xcenter, h, w) to (xmin, ymin, xmax, ymax)
1347 priors = priors.reshape(1, priors.total() / 4);
1348 Mat tmp = priors.col(0).clone();
1349 priors.col(0) = priors.col(1) - 0.5 * priors.col(3);
1350 priors.col(1) = tmp - 0.5 * priors.col(2);
1351
1352 tmp = priors.col(2).clone();
1353 priors.col(2) = priors.col(0) + priors.col(3);
1354 priors.col(3) = priors.col(1) + tmp;
1355
1356 float x_scale = *(float*)&parameters["x_scale"];
1357 float y_scale = *(float*)&parameters["y_scale"];
1358 float w_scale = *(float*)&parameters["w_scale"];
1359 float h_scale = *(float*)&parameters["h_scale"];
1360 if (x_scale != 1.0f || y_scale != 1.0f || w_scale != 1.0f || h_scale != 1.0f) {
1361 int numPriors = priors.rows;
1362 priors.resize(numPriors * 2);
1363 Mat_<float> scales({1, 4}, {1.f / x_scale, 1.f / y_scale,
1364 1.f / w_scale, 1.f / h_scale});
1365 repeat(scales, numPriors, 1, priors.rowRange(numPriors, priors.rows));
1366 priors = priors.reshape(1, {1, 2, (int)priors.total() / 2});
1367 layerParams.set("variance_encoded_in_target", false);
1368 } else {
1369 priors = priors.reshape(1, {1, 1, (int)priors.total()});
1370 layerParams.set("variance_encoded_in_target", true);
1371 }
1372
1373 addConstLayer(priors, op.inputs()->Get(2));
1374 addLayer(layerParams, op);
1375}

1389void TFLiteImporter::parseActivation(const Operator& op, const std::string& opcode, LayerParams& activParams, bool isFused) {
1390 float slope = 0.;
1391 if (opcode == "NONE")
1392 return;
1393 else if (opcode == "RELU6")
1394 activParams.type = "ReLU6";
1395 else if (opcode == "PRELU")
1396 activParams.type = "PReLU";
1397 else if (opcode == "RELU")
1398 activParams.type = "ReLU";
1399 else if (opcode == "HARD_SWISH")
1400 activParams.type = "HardSwish";
1401 else if (opcode == "LOGISTIC")
1402 activParams.type = "Sigmoid";
1403 else if (opcode == "LEAKY_RELU")
1404 {
1405 activParams.type = "ReLU";
1406 auto options = reinterpret_cast<const LeakyReluOptions*>(op.builtin_options());
1407 slope = options->alpha();
1408 activParams.set("negative_slope", slope);
1409 }
1410 else
1411 CV_Error(Error::StsNotImplemented, "Unsupported activation " + opcode);
1412
1413 if (isInt8(op)) {
1414 float inpScale, outScale;
1415 int inpZero, outZero;
1416 getQuantParams(op, inpScale, inpZero, outScale, outZero);
1417
1418 if (isFused) {
1419 activParams.type += "Int8";
1420 activParams.set("input_scale", outScale);
1421 activParams.set("input_zeropoint", outZero);
1422 activParams.set("scales", outScale);
1423 activParams.set("zeropoints", outZero);
1424 }
1425
1426 Mat lookUpTable(1, 256, CV_8S);
1427 int8_t* table = lookUpTable.ptr<int8_t>();
1428 for (int i = -128; i < 128; i++) {
1429 float x, y = i;
1430 if (isFused)
1431 x = outScale * (i - outZero);
1432 else
1433 x = inpScale * (i - inpZero);
1434
1435 if (opcode == "RELU6")
1436 y = std::min(std::max(x, 0.f), 6.f);
1437 else if (opcode == "LOGISTIC")
1438 y = 1.0f / (1.0f + std::exp(-x));
1439 else if (opcode == "HARD_SWISH")
1440 y = x * max(0.f, min(1.f, x / 6.f + 0.5f));
1441 else if (opcode == "LEAKY_RELU")
1442 y = x >= 0.f ? x : slope*x;
1443 else
1444 CV_Error(Error::StsNotImplemented, "Lookup table for " + opcode);
1445
1446 int quantized = outZero + cvRound(y / outScale);
1447 table[i + 128] = saturate_cast<int8_t>(quantized);
1448 }
1449 activParams.blobs.resize(1, lookUpTable);
1450 }
1451
1452 if (isFused) {
1453 if (newEngine)
1454 {
1455 std::string tensorName = modelTensors->Get(op.outputs()->Get(0))->name()->str();
1456 addLayer(activParams, {tensorName + "_additional_post_layer"}, {tensorName});
1457 layerIds[op.outputs()->Get(0)] = std::make_pair(-1, -1);
1458 }
1459 else
1460 {
1461 int dtype = isInt8(op) ? CV_8S : CV_32F;
1462 int layerId = dstNet.addLayerToPrev(activParams.name, activParams.type, dtype, activParams);
1463
1464 // Override layer ids mapping
1465 int i = 0;
1466 for (int idx : *op.outputs()) {
1467 layerIds[idx] = std::make_pair(layerId, i++);
1468 }
1469 }
1470 } else {
1471 addLayer(activParams, op);
1472 }
1473}
