59struct CalcRotation
60{
61 CalcRotation(int _num_images, const std::vector<MatchesInfo> &_pairwise_matches, std::vector<CameraParams> &_cameras)
62 : num_images(_num_images), pairwise_matches(&_pairwise_matches[0]), cameras(&_cameras[0]) {}
63
64 void operator ()(const GraphEdge &edge)
65 {
66 int pair_idx = edge.from * num_images + edge.to;
67
68 Mat_<double> K_from = Mat::eye(3, 3, CV_64F);
69 K_from(0,0) = cameras[edge.from].focal;
70 K_from(1,1) = cameras[edge.from].focal * cameras[edge.from].aspect;
71 K_from(0,2) = cameras[edge.from].ppx;
72 K_from(1,2) = cameras[edge.from].ppy;
73
74 Mat_<double> K_to = Mat::eye(3, 3, CV_64F);
75 K_to(0,0) = cameras[edge.to].focal;
76 K_to(1,1) = cameras[edge.to].focal * cameras[edge.to].aspect;
77 K_to(0,2) = cameras[edge.to].ppx;
78 K_to(1,2) = cameras[edge.to].ppy;
79
80 Mat R = K_from.inv() * pairwise_matches[pair_idx].H.inv() * K_to;
81 cameras[edge.to].R = cameras[edge.from].R * R;
82 }
83
84 int num_images;
85 const MatchesInfo* pairwise_matches;
86 CameraParams* cameras;
87};

199 std::vector<CameraParams> &cameras)
200{

203
204 // find maximum spaning tree on pairwise matches
205 cv::detail::Graph span_tree;
206 std::vector<int> span_tree_centers;
207 // uses number of inliers as weights
208 findMaxSpanningTree(num_images, pairwise_matches, span_tree,

222bool BundleAdjusterBase::estimate(const std::vector<ImageFeatures> &features,
223 const std::vector<MatchesInfo> &pairwise_matches,
224 std::vector<CameraParams> &cameras)
225{
226 LOG_CHAT("Bundle adjustment");
227#if ENABLE_LOG
228 int64 t = getTickCount();
229#endif
230
231 num_images_ = static_cast<int>(features.size());
232 features_ = &features[0];
233 pairwise_matches_ = &pairwise_matches[0];
234
235 setUpInitialCameraParams(cameras);
236
237 // Leave only consistent image pairs
238 edges_.clear();
239 for (int i = 0; i < num_images_ - 1; ++i)
240 {
241 for (int j = i + 1; j < num_images_; ++j)
242 {
243 const MatchesInfo& matches_info = pairwise_matches_[i * num_images_ + j];
244 if (matches_info.confidence > conf_thresh_)
245 edges_.push_back(std::make_pair(i, j));
246 }
247 }
248
249 // Compute number of correspondences
250 total_num_matches_ = 0;
251 for (size_t i = 0; i < edges_.size(); ++i)
252 total_num_matches_ += static_cast<int>(pairwise_matches[edges_[i].first * num_images_ +
253 edges_[i].second].num_inliers);
254
255 int nerrs = total_num_matches_ * num_errs_per_measurement_;
256
257 auto callb = [&](InputOutputArray param, OutputArray err, OutputArray jac) -> bool
258 {
259 // workaround against losing value
260 Mat backup = cam_params_.clone();
261 param.copyTo(cam_params_);
262 if (jac.needed())
263 {
264 Mat m = jac.getMat();
265 calcJacobian(m);
266 }
267 if (err.needed())
268 {
269 Mat m = err.getMat();
270 calcError(m);
271 }
272 backup.copyTo(cam_params_);
273 return true;
274 };
275
276 LevMarq solver(cam_params_, callb,
277 LevMarq::Settings()
278 .setMaxIterations((unsigned int)term_criteria_.maxCount)
279 .setStepNormTolerance(term_criteria_.epsilon)
280 .setSmallEnergyTolerance(term_criteria_.epsilon * term_criteria_.epsilon)
281 .setGeodesic(true),
282 noArray(), MatrixType::AUTO, VariableType::LINEAR, nerrs);
283 solver.optimize();
284
285 LOGLN_CHAT("");
286 LOGLN_CHAT("Bundle adjustment, final RMS error: " << std::sqrt(err.dot(err) / total_num_matches_));
287 LOGLN_CHAT("Bundle adjustment, iterations done: " << iter);
288
289 // Check if all camera parameters are valid
290 bool ok = true;
291 for (int i = 0; i < cam_params_.rows; ++i)
292 {
293 if (cvIsNaN(cam_params_.at<double>(i,0)))
294 {
295 ok = false;
296 break;
297 }
298 }
299 if (!ok)
300 return false;
301
302 obtainRefinedCameraParams(cameras);
303
304 // Normalize motion to center image
305 Graph span_tree;
306 std::vector<int> span_tree_centers;
307 findMaxSpanningTree(num_images_, pairwise_matches, span_tree, span_tree_centers);
308 Mat R_inv = cameras[span_tree_centers[0]].R.inv();
309 for (int i = 0; i < num_images_; ++i)
310 cameras[i].R = R_inv * cameras[i].R;
311
312 LOGLN_CHAT("Bundle adjustment, time: " << ((getTickCount() - t) / getTickFrequency()) << " sec");
313 return true;
314}

367void BundleAdjusterReproj::calcError(Mat &err)
368{
369 err.create(total_num_matches_ * 2, 1, CV_64F);
370
371 int match_idx = 0;
372 for (size_t edge_idx = 0; edge_idx < edges_.size(); ++edge_idx)
373 {
374 int i = edges_[edge_idx].first;
375 int j = edges_[edge_idx].second;
376 double f1 = cam_params_.at<double>(i * 7, 0);
377 double f2 = cam_params_.at<double>(j * 7, 0);
378 double ppx1 = cam_params_.at<double>(i * 7 + 1, 0);
379 double ppx2 = cam_params_.at<double>(j * 7 + 1, 0);
380 double ppy1 = cam_params_.at<double>(i * 7 + 2, 0);
381 double ppy2 = cam_params_.at<double>(j * 7 + 2, 0);
382 double a1 = cam_params_.at<double>(i * 7 + 3, 0);
383 double a2 = cam_params_.at<double>(j * 7 + 3, 0);
384
385 double R1[9];
386 Mat R1_(3, 3, CV_64F, R1);
387 Mat rvec(3, 1, CV_64F);
388 rvec.at<double>(0, 0) = cam_params_.at<double>(i * 7 + 4, 0);
389 rvec.at<double>(1, 0) = cam_params_.at<double>(i * 7 + 5, 0);
390 rvec.at<double>(2, 0) = cam_params_.at<double>(i * 7 + 6, 0);
391 Rodrigues(rvec, R1_);
392
393 double R2[9];
394 Mat R2_(3, 3, CV_64F, R2);
395 rvec.at<double>(0, 0) = cam_params_.at<double>(j * 7 + 4, 0);
396 rvec.at<double>(1, 0) = cam_params_.at<double>(j * 7 + 5, 0);
397 rvec.at<double>(2, 0) = cam_params_.at<double>(j * 7 + 6, 0);
398 Rodrigues(rvec, R2_);
399
400 const ImageFeatures& features1 = features_[i];
401 const ImageFeatures& features2 = features_[j];
402 const MatchesInfo& matches_info = pairwise_matches_[i * num_images_ + j];
403
404 Mat_<double> K1 = Mat::eye(3, 3, CV_64F);
405 K1(0,0) = f1; K1(0,2) = ppx1;
406 K1(1,1) = f1*a1; K1(1,2) = ppy1;
407
408 Mat_<double> K2 = Mat::eye(3, 3, CV_64F);
409 K2(0,0) = f2; K2(0,2) = ppx2;
410 K2(1,1) = f2*a2; K2(1,2) = ppy2;
411
412 Mat_<double> H = K2 * R2_.inv() * R1_ * K1.inv();
413
414 for (size_t k = 0; k < matches_info.matches.size(); ++k)
415 {
416 if (!matches_info.inliers_mask[k])
417 continue;
418
419 const DMatch& m = matches_info.matches[k];
420 Point2f p1 = features1.keypoints[m.queryIdx].pt;
421 Point2f p2 = features2.keypoints[m.trainIdx].pt;
422 double x = H(0,0)*p1.x + H(0,1)*p1.y + H(0,2);
423 double y = H(1,0)*p1.x + H(1,1)*p1.y + H(1,2);
424 double z = H(2,0)*p1.x + H(2,1)*p1.y + H(2,2);
425
426 err.at<double>(2 * match_idx, 0) = p2.x - x/z;
427 err.at<double>(2 * match_idx + 1, 0) = p2.y - y/z;
428 match_idx++;
429 }
430 }
431}

434void BundleAdjusterReproj::calcJacobian(Mat &jac)
435{
436 jac.create(total_num_matches_ * 2, num_images_ * 7, CV_64F);
437 jac.setTo(0);
438
439 double val;
440 const double step = 1e-4;
441
442 for (int i = 0; i < num_images_; ++i)
443 {
444 if (refinement_mask_.at<uchar>(0, 0))
445 {
446 val = cam_params_.at<double>(i * 7, 0);
447 cam_params_.at<double>(i * 7, 0) = val - step;
448 calcError(err1_);
449 cam_params_.at<double>(i * 7, 0) = val + step;
450 calcError(err2_);
451 calcDeriv(err1_, err2_, 2 * step, jac.col(i * 7));
452 cam_params_.at<double>(i * 7, 0) = val;
453 }
454 if (refinement_mask_.at<uchar>(0, 2))
455 {
456 val = cam_params_.at<double>(i * 7 + 1, 0);
457 cam_params_.at<double>(i * 7 + 1, 0) = val - step;
458 calcError(err1_);
459 cam_params_.at<double>(i * 7 + 1, 0) = val + step;
460 calcError(err2_);
461 calcDeriv(err1_, err2_, 2 * step, jac.col(i * 7 + 1));
462 cam_params_.at<double>(i * 7 + 1, 0) = val;
463 }
464 if (refinement_mask_.at<uchar>(1, 2))
465 {
466 val = cam_params_.at<double>(i * 7 + 2, 0);
467 cam_params_.at<double>(i * 7 + 2, 0) = val - step;
468 calcError(err1_);
469 cam_params_.at<double>(i * 7 + 2, 0) = val + step;
470 calcError(err2_);
471 calcDeriv(err1_, err2_, 2 * step, jac.col(i * 7 + 2));
472 cam_params_.at<double>(i * 7 + 2, 0) = val;
473 }
474 if (refinement_mask_.at<uchar>(1, 1))
475 {
476 val = cam_params_.at<double>(i * 7 + 3, 0);
477 cam_params_.at<double>(i * 7 + 3, 0) = val - step;
478 calcError(err1_);
479 cam_params_.at<double>(i * 7 + 3, 0) = val + step;
480 calcError(err2_);
481 calcDeriv(err1_, err2_, 2 * step, jac.col(i * 7 + 3));
482 cam_params_.at<double>(i * 7 + 3, 0) = val;
483 }
484 for (int j = 4; j < 7; ++j)
485 {
486 val = cam_params_.at<double>(i * 7 + j, 0);
487 cam_params_.at<double>(i * 7 + j, 0) = val - step;
488 calcError(err1_);
489 cam_params_.at<double>(i * 7 + j, 0) = val + step;
490 calcError(err2_);
491 calcDeriv(err1_, err2_, 2 * step, jac.col(i * 7 + j));
492 cam_params_.at<double>(i * 7 + j, 0) = val;
493 }
494 }
495}

542void BundleAdjusterRay::calcError(Mat &err)
543{
544 err.create(total_num_matches_ * 3, 1, CV_64F);
545
546 int match_idx = 0;
547 for (size_t edge_idx = 0; edge_idx < edges_.size(); ++edge_idx)
548 {
549 int i = edges_[edge_idx].first;
550 int j = edges_[edge_idx].second;
551 double f1 = cam_params_.at<double>(i * 4, 0);
552 double f2 = cam_params_.at<double>(j * 4, 0);
553
554 double R1[9];
555 Mat R1_(3, 3, CV_64F, R1);
556 Mat rvec(3, 1, CV_64F);
557 rvec.at<double>(0, 0) = cam_params_.at<double>(i * 4 + 1, 0);
558 rvec.at<double>(1, 0) = cam_params_.at<double>(i * 4 + 2, 0);
559 rvec.at<double>(2, 0) = cam_params_.at<double>(i * 4 + 3, 0);
560 Rodrigues(rvec, R1_);
561
562 double R2[9];
563 Mat R2_(3, 3, CV_64F, R2);
564 rvec.at<double>(0, 0) = cam_params_.at<double>(j * 4 + 1, 0);
565 rvec.at<double>(1, 0) = cam_params_.at<double>(j * 4 + 2, 0);
566 rvec.at<double>(2, 0) = cam_params_.at<double>(j * 4 + 3, 0);
567 Rodrigues(rvec, R2_);
568
569 const ImageFeatures& features1 = features_[i];
570 const ImageFeatures& features2 = features_[j];
571 const MatchesInfo& matches_info = pairwise_matches_[i * num_images_ + j];
572
573 Mat_<double> K1 = Mat::eye(3, 3, CV_64F);
574 K1(0,0) = f1; K1(0,2) = features1.img_size.width * 0.5;
575 K1(1,1) = f1; K1(1,2) = features1.img_size.height * 0.5;
576
577 Mat_<double> K2 = Mat::eye(3, 3, CV_64F);
578 K2(0,0) = f2; K2(0,2) = features2.img_size.width * 0.5;
579 K2(1,1) = f2; K2(1,2) = features2.img_size.height * 0.5;
580
581 Mat_<double> H1 = R1_ * K1.inv();
582 Mat_<double> H2 = R2_ * K2.inv();
583
584 for (size_t k = 0; k < matches_info.matches.size(); ++k)
585 {
586 if (!matches_info.inliers_mask[k])
587 continue;
588
589 const DMatch& m = matches_info.matches[k];
590
591 Point2f p1 = features1.keypoints[m.queryIdx].pt;
592 double x1 = H1(0,0)*p1.x + H1(0,1)*p1.y + H1(0,2);
593 double y1 = H1(1,0)*p1.x + H1(1,1)*p1.y + H1(1,2);
594 double z1 = H1(2,0)*p1.x + H1(2,1)*p1.y + H1(2,2);
595 double len = std::sqrt(x1*x1 + y1*y1 + z1*z1);
596 x1 /= len; y1 /= len; z1 /= len;
597
598 Point2f p2 = features2.keypoints[m.trainIdx].pt;
599 double x2 = H2(0,0)*p2.x + H2(0,1)*p2.y + H2(0,2);

878WaveCorrectKind autoDetectWaveCorrectKind(const std::vector<Mat> &rmats)
879{
880 std::vector<float> xs, ys;
881 xs.reserve(rmats.size());
882 ys.reserve(rmats.size());
883
884 // Project a [0, 0, 1, 1] point to the camera image frame
885 // Ignore intrinsic parameters and camera translation as they
886 // have little influence
887 // This also means we can simply use "rmat.col(2)" as the
888 // projected point homogeneous coordinate
889 for (const Mat& rmat: rmats)
890 {
891 CV_Assert(rmat.type() == CV_32F);
892 xs.push_back(rmat.at<float>(0, 2) / rmat.at<float>(2, 2));
893 ys.push_back(rmat.at<float>(1, 2) / rmat.at<float>(2, 2));
894 }
895
896 // Calculate the delta between the max and min values for
897 // both the X and Y axis
898 auto min_max_x = std::minmax_element(xs.begin(), xs.end());
899 auto min_max_y = std::minmax_element(ys.begin(), ys.end());
900 double delta_x = *min_max_x.second - *min_max_x.first;
901 double delta_y = *min_max_y.second - *min_max_y.first;
902
903 // If the Y delta is the biggest, it means the images
904 // mostly span along the vertical axis: correct this axis
905 if (delta_y > delta_x)
906 {
907 LOGLN(" using vertical wave correction");
908 return WAVE_CORRECT_VERT;
909 }
910 else
911 {
912 LOGLN(" using horizontal wave correction");
913 return WAVE_CORRECT_HORIZ;
914 }

917void waveCorrect(std::vector<Mat> &rmats, WaveCorrectKind kind)
918{
919 LOGLN("Wave correcting...");
920#if ENABLE_LOG
921 int64 t = getTickCount();
922#endif
923 if (rmats.size() <= 1)
924 {
925 LOGLN("Wave correcting, time: " << ((getTickCount() - t) / getTickFrequency()) << " sec");
926 return;
927 }
928
929 if (kind == WAVE_CORRECT_AUTO)
930 {
931 kind = autoDetectWaveCorrectKind(rmats);
932 }
933
934 Mat moment = Mat::zeros(3, 3, CV_32F);
935 for (size_t i = 0; i < rmats.size(); ++i)
936 {
937 Mat col = rmats[i].col(0);
938 moment += col * col.t();
939 }
940
941 Mat eigen_vals, eigen_vecs;
942 eigen(moment, eigen_vals, eigen_vecs);
943
944 Mat rg1;
945 if (kind == WAVE_CORRECT_HORIZ)
946 rg1 = eigen_vecs.row(2).t();
947 else if (kind == WAVE_CORRECT_VERT)
948 rg1 = eigen_vecs.row(0).t();
949 else
950 CV_Error(cv::Error::StsBadArg, "unsupported kind of wave correction");
951
952 Mat img_k = Mat::zeros(3, 1, CV_32F);
953 for (size_t i = 0; i < rmats.size(); ++i)
954 img_k += rmats[i].col(2);
955 Mat rg0 = rg1.cross(img_k);
956 double rg0_norm = norm(rg0);
957
958 if( rg0_norm <= DBL_MIN )
959 {
960 return;
961 }
962
963 rg0 /= rg0_norm;
964
965 Mat rg2 = rg0.cross(rg1);
966
967 double conf = 0;
968 if (kind == WAVE_CORRECT_HORIZ)
969 {
970 for (size_t i = 0; i < rmats.size(); ++i)
971 conf += rg0.dot(rmats[i].col(0));
972 if (conf < 0)
973 {
974 rg0 *= -1;
975 rg1 *= -1;
976 }
977 }
978 else if (kind == WAVE_CORRECT_VERT)
979 {
980 for (size_t i = 0; i < rmats.size(); ++i)
981 conf -= rg1.dot(rmats[i].col(0));
982 if (conf < 0)
983 {
984 rg0 *= -1;
985 rg1 *= -1;
986 }
987 }
988
989 Mat R = Mat::zeros(3, 3, CV_32F);
990 Mat tmp = R.row(0);
991 Mat(rg0.t()).copyTo(tmp);
992 tmp = R.row(1);
993 Mat(rg1.t()).copyTo(tmp);
994 tmp = R.row(2);
995 Mat(rg2.t()).copyTo(tmp);
996
997 for (size_t i = 0; i < rmats.size(); ++i)
998 rmats[i] = R * rmats[i];
999
1000 LOGLN("Wave correcting, time: " << ((getTickCount() - t) / getTickFrequency()) << " sec");
1001}

1131void findMaxSpanningTree(int num_images, const std::vector<MatchesInfo> &pairwise_matches,
1132 Graph &span_tree, std::vector<int> &centers)
1133{
1134 Graph graph(num_images);
1135 std::vector<GraphEdge> edges;
1136
1137 // Construct images graph and remember its edges
1138 for (int i = 0; i < num_images; ++i)
1139 {
1140 for (int j = 0; j < num_images; ++j)
1141 {
1142 if (pairwise_matches[i * num_images + j].H.empty())
1143 continue;
1144 float conf = static_cast<float>(pairwise_matches[i * num_images + j].num_inliers);
1145 graph.addEdge(i, j, conf);
1146 edges.push_back(GraphEdge(i, j, conf));
1147 }
1148 }
1149
1150 DisjointSets comps(num_images);
1151 span_tree.create(num_images);
1152 std::vector<int> span_tree_powers(num_images, 0);
1153
1154 // Find maximum spanning tree
1155 sort(edges.begin(), edges.end(), std::greater<GraphEdge>());
1156 for (size_t i = 0; i < edges.size(); ++i)
1157 {
1158 int comp1 = comps.findSetByElem(edges[i].from);
1159 int comp2 = comps.findSetByElem(edges[i].to);
1160 if (comp1 != comp2)
1161 {
1162 comps.mergeSets(comp1, comp2);
1163 span_tree.addEdge(edges[i].from, edges[i].to, edges[i].weight);
1164 span_tree.addEdge(edges[i].to, edges[i].from, edges[i].weight);
1165 span_tree_powers[edges[i].from]++;
1166 span_tree_powers[edges[i].to]++;
1167 }
1168 }
1169
1170 // Find spanning tree leafs
1171 std::vector<int> span_tree_leafs;
1172 for (int i = 0; i < num_images; ++i)
1173 if (span_tree_powers[i] == 1)
1174 span_tree_leafs.push_back(i);
1175
1176 // Find maximum distance from each spanning tree vertex
1177 std::vector<int> max_dists(num_images, 0);
1178 std::vector<int> cur_dists;
1179 for (size_t i = 0; i < span_tree_leafs.size(); ++i)
1180 {
1181 cur_dists.assign(num_images, 0);
1182 span_tree.walkBreadthFirst(span_tree_leafs[i], IncDistance(cur_dists));
1183 for (int j = 0; j < num_images; ++j)
1184 max_dists[j] = std::max(max_dists[j], cur_dists[j]);
1185 }
1186
1187 // Find min-max distance
1188 int min_max_dist = max_dists[0];
1189 for (int i = 1; i < num_images; ++i)
1190 if (min_max_dist > max_dists[i])
1191 min_max_dist = max_dists[i];
1192
1193 // Find spanning tree centers
1194 centers.clear();
1195 for (int i = 0; i < num_images; ++i)
1196 if (max_dists[i] == min_max_dist)
1197 centers.push_back(i);
1198 CV_Assert(centers.size() > 0 && centers.size() <= 2);
1199}
