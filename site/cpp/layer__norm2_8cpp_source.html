35 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
36 const int requiredOutputs,
37 std::vector<MatShape> &outputs,
38 std::vector<MatShape> &internals) const CV_OVERRIDE
39 {
40 int noutputs = std::max(requiredOutputs > 0 ? requiredOutputs : (int)this->outputs.size(), 1);
41 CV_Assert(noutputs >= 1 && noutputs <= 3);
42
43 int num_inputs = inputs.size() + blobs.size();
44 CV_Check(num_inputs, num_inputs >= 2 && num_inputs <= 3, "LayerNorm2: require two (x, weight) or three (x, weight, bias) inputs");
45
46 auto x_shape = inputs[0];
47 int x_ndims = static_cast<int>(x_shape.size());
48 int axis_ = normalize_axis(axis0, x_shape.dims);
49
50 auto w_shape = blobs.empty() ? inputs[1] : shape(blobs.front());
51 int w_ndims = static_cast<int>(w_shape.size());
52 w_ndims = (axis_ == x_ndims - 1 && w_ndims == 2) ? (w_ndims - 1) : w_ndims;
53 CV_CheckEQ(x_ndims - axis_, w_ndims, "LayerNorm2: weight rank mismatch");
54 for (int i = 0; i < w_ndims; ++i)
55 CV_CheckEQ(x_shape[axis_ + i], w_shape[i], "LayerNorm2: weight dims mismatch");
56 if (num_inputs >= 3)
57 {
58 auto b_shape = blobs.empty() ? inputs[2] : shape(blobs.back());
59 CV_CheckEQ(w_shape.size(), b_shape.size(), "LayerNorm2: bias rank mismatch");
60 for (size_t i = 0; i < w_shape.size(); ++i)
61 CV_CheckEQ(w_shape[i], b_shape[i], "LayerNorm2: bias dims mismatch");
62 }
63
64 outputs.resize(noutputs, inputs[0]);
65 for (int i = 1; i < noutputs; i++) {
66 for (int j = axis_; j < x_ndims; j++)
67 outputs[i][j] = 1;
68 }
69 internals.clear();
70 return false;
71 }

73 void forward(InputArrayOfArrays inputs_arr,
74 OutputArrayOfArrays outputs_arr,
75 OutputArrayOfArrays internals_arr) CV_OVERRIDE
76 {
77 CV_TRACE_FUNCTION();
78 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
79
80 std::vector<Mat> inputs, outputs;
81 inputs_arr.getMatVector(inputs);
82 outputs_arr.getMatVector(outputs);
83
84 const Mat& input = inputs[0];
85 const Mat& scale = blobs.empty() ? inputs[1] : blobs.front();
86 Mat& output = outputs[0];
87
88 int axis_ = normalize_axis(axis0, input.dims);
89
90 if (outputs.size() >= 3)
91 {
92 Mat& mean = outputs[1];
93 Mat& invStdDev = outputs[2];
94 fastNormMeanInvStdDev(input, mean, invStdDev, epsilon, (size_t)axis_);
95 }
96
97 if ((int)inputs.size() + (int)blobs.size() >= 3)
98 {
99 const Mat& bias = blobs.empty() ? inputs[2] : blobs.back();
100 fastNorm(input, scale, bias, output, epsilon, (size_t)axis_);
101 }
102 else
103 {
104 fastNorm(input, scale, output, epsilon, (size_t)axis_);
105 }
106 }
107};
