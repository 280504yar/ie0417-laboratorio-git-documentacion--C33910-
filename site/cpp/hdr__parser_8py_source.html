36def evaluate_conditional_inclusion_directive(directive, preprocessor_definitions):
37 """Evaluates C++ conditional inclusion directive.
38 Reference: https://en.cppreference.com/w/cpp/preprocessor/conditional
39
40 Args:
41 directive(str): input C++ conditional directive.
42 preprocessor_definitions(dict[str, int]): defined preprocessor identifiers.
43
44 Returns:
45 bool: True, if directive is evaluated to 1, False otherwise.
46
47 >>> evaluate_conditional_inclusion_directive("#ifdef A", {"A": 0})
48 True
49
50 >>> evaluate_conditional_inclusion_directive("#ifdef A", {"B": 0})
51 False
52
53 >>> evaluate_conditional_inclusion_directive("#ifndef A", {})
54 True
55
56 >>> evaluate_conditional_inclusion_directive("#ifndef A", {"A": 1})
57 False
58
59 >>> evaluate_conditional_inclusion_directive("#if 0", {})
60 False
61
62 >>> evaluate_conditional_inclusion_directive("#if 1", {})
63 True
64
65 >>> evaluate_conditional_inclusion_directive("#if VAR", {"VAR": 0})
66 False
67
68 >>> evaluate_conditional_inclusion_directive("#if VAR ", {"VAR": 1})
69 True
70
71 >>> evaluate_conditional_inclusion_directive("#if defined(VAR)", {"VAR": 0})
72 True
73
74 >>> evaluate_conditional_inclusion_directive("#if !defined(VAR)", {"VAR": 0})
75 False
76
77 >>> evaluate_conditional_inclusion_directive("#if defined(VAR_1)", {"VAR_2": 0})
78 False
79
80 >>> evaluate_conditional_inclusion_directive(
81 ... "#if defined(VAR) && VAR", {"VAR": 0}
82 ... )
83 False
84
85 >>> evaluate_conditional_inclusion_directive(
86 ... "#if VAR_1 || VAR_2", {"VAR_1": 1, "VAR_2": 0}
87 ... )
88 True
89
90 >>> evaluate_conditional_inclusion_directive(
91 ... "#if defined VAR && defined (VAR)", {"VAR": 1}
92 ... )
93 True
94
95 >>> evaluate_conditional_inclusion_directive(
96 ... "#if strangedefinedvar", {}
97 ... )
98 Traceback (most recent call last):
99 ...
100 ValueError: Failed to evaluate '#if strangedefinedvar' directive, stripped down to 'strangedefinedvar'
101 """
102 OPERATORS1 = {"&&": "and", "||": "or"}
103 OPERATORS2 = { "!": "not ", "&": "and", "|": "or" }
104
105 input_directive = directive
106
107 # Ignore all directives if they contain __cplusplus check
108 if "__cplusplus" in directive:
109 return True
110
111 directive = directive.strip()
112 if directive.startswith("#ifdef "):
113 var = directive[len("#ifdef "):].strip()
114 return var in preprocessor_definitions
115 if directive.startswith("#ifndef "):
116 var = directive[len("#ifndef "):].strip()
117 return var not in preprocessor_definitions
118
119 if directive.startswith("#if "):
120 directive = directive[len("#if "):].strip()
121 elif directive.startswith("#elif "):
122 directive = directive[len("#elif "):].strip()
123 else:
124 raise ValueError("{} is not known conditional directive".format(directive))
125
126 if directive.isdigit():
127 return int(directive) != 0
128
129 if directive in preprocessor_definitions:
130 return bool(preprocessor_definitions[directive])
131
132 # Converting all `defined` directives to their boolean representations
133 # they have 2 forms: `defined identifier` and `defined(identifier)`
134 directive = re.sub(
135 r"\bdefined\s*(\w+|\‍(\w+\‍))",
136 lambda m: "True" if m.group(1).strip("() ") in preprocessor_definitions else "False",
137 directive
138 )
139
140 for src_op, dst_op in OPERATORS1.items():
141 directive = directive.replace(src_op, dst_op)
142
143 for src_op, dst_op in OPERATORS2.items():
144 directive = directive.replace(src_op, dst_op)
145
146 try:
147 if sys.version_info >= (3, 13):
148 eval_directive = eval(directive,
149 globals={"__builtins__": {}},
150 locals=preprocessor_definitions)
151 else:
152 eval_directive = eval(directive,
153 {"__builtins__": {}},
154 preprocessor_definitions)
155 except Exception as e:
156 raise ValueError(
157 "Failed to evaluate '{}' directive, stripped down to '{}'".format(
158 input_directive, directive
159 )
160 ) from e
161
162 if not isinstance(eval_directive, (bool, int)):
163 raise TypeError(
164 "'{}' directive is evaluated to unexpected type: {}".format(
165 input_directive, type(eval_directive).__name__
166 )
167 )
168 if isinstance(eval_directive, bool):
169 return eval_directive
170
171 return eval_directive != 0
