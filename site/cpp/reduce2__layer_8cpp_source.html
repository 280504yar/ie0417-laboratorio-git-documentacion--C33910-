99 bool getMemoryShapes(const std::vector<MatShape> &inps,
100 const int /*requiredOutputs*/,
101 std::vector<MatShape> &outs,
102 std::vector<MatShape> &/*internals*/) const CV_OVERRIDE
103 {
104 CV_Assert(!inps.empty());
105 outs.resize(1);
106 const MatShape& inp0 = inps[0];
107 if (inp0.empty()) {
108 outs[0] = MatShape();
109 return false;
110 }
111
112 std::vector<int> axes;
113 if (!this->axes.empty()) {
114 axes = this->axes;
115 } else if (inps.size() >= 2) {
116 Net::Impl* netimpl_ = getNetImpl(this);
117 if (netimpl_ && netimpl_->isConstArg(inputs[1])) {
118 Mat axesTensor = netimpl_->argTensor(inputs[1]);
119 tensorToIntVec(axesTensor, axes);
120 }
121 }
122
123 if (axes.empty()) {
124 if (noop_with_empty_axes) {
125 outs[0] = inp0;
126 } else {
127 if (keepdims) {
128 MatShape shape_out = inp0;
129 std::fill(shape_out.begin(), shape_out.end(), 1);
130 outs[0] = shape_out;
131 } else {
132 outs[0] = MatShape::scalar();
133 }
134 }
135 return false;
136 }
137
138 std::vector<int> norm_axes = axes;
139 for (size_t i = 0; i < norm_axes.size(); ++i)
140 norm_axes[i] = normalize_axis(norm_axes[i], inp0);
141
142 auto shape_output_ = inp0;
143 for (int axis : norm_axes) shape_output_[axis] = -1;
144 MatShape shape_output;
145 for (size_t i = 0; i < shape_output_.size(); ++i) {
146 if (shape_output_[i] == -1) {
147 if (keepdims) shape_output.push_back(1);
148 } else {
149 shape_output.push_back(shape_output_[i]);
150 }
151 }
152 if (shape_output.empty()) shape_output = MatShape::scalar();
153 outs[0] = shape_output;
154 return false;

171 template <typename T, typename WT, typename AccT>
172 class ReduceBase {
173 public:
174 using dtype_input = T;
175 using work_type = WT;
176 using acc_type = AccT;
177 typedef T dtype;
178 typedef WT WorkT;
179 ReduceBase(size_t n, const T& init) : n_(n), accumulator_(static_cast<AccT>(static_cast<WT>(init))) {}
180 AccT finalize() const { return accumulator_; }
181 protected:
182 size_t n_;
183 AccT accumulator_;
184 };

191 inline void update(const WT& a) { this->accumulator_ = a < static_cast<WT>(this->accumulator_) ? static_cast<AccT>(a) : this->accumulator_; }
192 static WT identity() { return std::numeric_limits<WT>::has_infinity ? std::numeric_limits<WT>::infinity() : std::numeric_limits<WT>::max(); }

283 template <typename Op>
284 class ReduceAllInvoker : public ParallelLoopBody {
285 public:
286 using dtype = typename Op::dtype_input;
287 using WT = typename Op::work_type;
288 const Mat& src;
289 Mat& dst;
290 int n_reduce;
291 int loop_size;
292 int total;
293 int cost_per_thread;
294
295 ReduceAllInvoker(const Mat& src_, Mat& dst_) : src(src_), dst(dst_) {
296 auto shape_src = shape(src);
297 n_reduce = std::accumulate(shape_src.begin(), shape_src.end(), 1, std::multiplies<int>());
298 loop_size = n_reduce;
299 total = 1;
300 cost_per_thread = 1;
301 }
302
303 void operator()(const Range& r) const CV_OVERRIDE {
304 int start = r.start;
305 int end = r.end;
306 const dtype* p_src = src.ptr<const dtype>();
307 dtype* p_dst = dst.ptr<dtype>();
308 for (int i = start; i < end; ++i) {
309 Op accumulator(n_reduce, static_cast<WT>(*p_src));
310 for (int l = 0; l < loop_size; ++l) {
311 accumulator.update(static_cast<WT>(p_src[l]));
312 }
313 auto val = accumulator.finalize();
314 p_dst[i] = saturate_cast<dtype>(static_cast<double>(val));
315 }
316 }

319 template <typename Op>
320 class ReduceInvoker : public ParallelLoopBody {
321 public:
322 using dtype = typename Op::dtype_input;
323 using WT = typename Op::work_type;
324 const Mat& src;
325 Mat& dst;
326 std::vector<int> reduced_axes;
327 int n_reduce;
328 int loop_size;
329 int last_reduced_dim;
330 int last_reduced_step;
331 std::vector<int> projected_steps;
332 int last_unreduced_dim;
333 int last_unreduced_step;
334 std::vector<int> unprojected_steps;
335 int total;
336 int cost_per_thread;
337
338 ReduceInvoker(const Mat& src_, Mat& dst_, std::vector<int> axes_) : src(src_), dst(dst_), reduced_axes(axes_) {
339 auto shape_src = shape(src);
340
341 auto steps_src = shape_src;
342 steps_src[steps_src.size() - 1] = 1;
343 for (int i = (int)steps_src.size() - 2; i >= 0; --i)
344 steps_src[i] = steps_src[i + 1] * shape_src[i + 1];
345
346 size_t projection_size = 1;
347 for (auto axis : reduced_axes) projection_size *= shape_src[axis];
348 n_reduce = (int)projection_size;
349
350 last_reduced_dim = shape_src[reduced_axes.back()];
351 last_reduced_step = steps_src[reduced_axes.back()];
352 loop_size = last_reduced_dim * last_reduced_step;
353 projection_size /= last_reduced_dim;
354
355 int last_reduced_axis = (int)reduced_axes.size() - 1;
356 if (last_reduced_axis == 0) {
357 projected_steps.resize(1, 0);
358 } else {
359 projected_steps.resize(projection_size);
360 std::vector<int> projected_indices(last_reduced_axis, 0);
361 for (size_t i = 0, current_step = 0; i < projection_size; ++i) {
362 projected_steps[i] = current_step;
363 ++projected_indices[last_reduced_axis - 1];
364 current_step += steps_src[reduced_axes[last_reduced_axis - 1]];
365 for (int j = last_reduced_axis - 1; j > 0; --j) {
366 if (projected_indices[j] < shape_src[reduced_axes[j]])
367 break;
368 projected_indices[j] = 0;
369 ++projected_indices[j - 1];
370 current_step = steps_src[reduced_axes[j - 1]];
371 }
372 }
373 }
374
375 std::vector<int> unreduced_axes;
376 for (int i = 0; i < (int)shape_src.size(); ++i) {
377 if (std::find(reduced_axes.begin(), reduced_axes.end(), i) == reduced_axes.end())
378 unreduced_axes.push_back(i);
379 }
380 size_t unprojection_size = 1;
381 for (auto axis : unreduced_axes) unprojection_size *= shape_src[axis];
382 last_unreduced_dim = shape_src[unreduced_axes.back()];
383 last_unreduced_step = steps_src[unreduced_axes.back()];
384 unprojection_size /= last_unreduced_dim;
385
386 std::vector<int> unprojected_indices(unreduced_axes.size(), 0);
387 unprojected_steps.reserve(unprojection_size);
388 if (unprojected_indices.size() <= 1) {
389 unprojected_steps.push_back(0);
390 } else {
391 for (size_t i = 0, current_step = 0; i < unprojection_size; ++i) {
392 unprojected_steps.push_back(current_step);
393 ++unprojected_indices[unprojected_indices.size() - 2];
394 current_step += steps_src[unreduced_axes[unreduced_axes.size() - 2]];
395 for (int j = (int)unreduced_axes.size() - 2; j > 0; --j) {
396 if (unprojected_indices[j] < shape_src[unreduced_axes[j]])
397 break;
398 unprojected_indices[j] -= shape_src[unreduced_axes[j]];
399 current_step -= shape_src[unreduced_axes[j]] * steps_src[unreduced_axes[j]];
400 ++unprojected_indices[j - 1];
401 current_step += steps_src[unreduced_axes[j - 1]];
402 }
403 }
404 }
405
406 auto shape_dst = shape(dst);
407 total = std::accumulate(shape_dst.begin(), shape_dst.end(), 1, std::multiplies<int>());
408 cost_per_thread = (int)(projected_steps.size() * last_reduced_step);
409 }

411 static void run(const Mat& src, Mat& dst, std::vector<int> axes, bool noop_with_empty_axes) {
412 CV_Assert(src.isContinuous());
413 CV_Assert(dst.isContinuous());
414 if (src.total() == 0) {
415 dst.setTo(Scalar(static_cast<double>(Op::identity())));
416 return;
417 }
418
419 if (shape(src).empty() || (shape(src).size() == 1)){
420 ReduceAllInvoker<Op> p(src, dst);
421 p(Range(0, p.total));
422 return;
423 }
424
425 if (axes.empty()) {
426 if (noop_with_empty_axes) {
427 const auto p_src = src.ptr<const dtype>();
428 auto p_dst = dst.ptr<dtype>();
429 std::memcpy(p_dst, p_src, sizeof(dtype) * dst.total());
430 return;
431 }
432 ReduceAllInvoker<Op> p(src, dst);
433 double nstripes = (size_t)p.total * (size_t)p.cost_per_thread * (1 / 1024.0);
434 parallel_for_(Range(0, p.total), p, nstripes);
435 return;
436 }
437
438 ReduceInvoker<Op> p(src, dst, axes);
439 double nstripes = (size_t)p.total * (size_t)p.cost_per_thread * (1 / 1024.0);
440 parallel_for_(Range(0, p.total), p, nstripes);
441 }

443 void operator()(const Range& r) const CV_OVERRIDE {
444 int start = r.start;
445 int end = r.end;
446 const dtype* p_src = src.ptr<const dtype>();
447 dtype* p_dst = dst.ptr<dtype>();
448 size_t main_index = start / last_unreduced_dim;
449 size_t loop = start % last_unreduced_dim;
450 size_t origin = unprojected_steps[main_index] + loop * last_unreduced_step;
451 for (int i = start; i < end; ++i) {
452 Op accumulator(n_reduce, static_cast<WT>(p_src[origin + projected_steps[0]]));
453 for (auto projected_step : projected_steps) {
454 const dtype* loop_p_src = p_src + origin + projected_step;
455 for (auto l = 0; l < loop_size; l += last_reduced_step) {
456 accumulator.update(static_cast<WT>(loop_p_src[l]));
457 }
458 }
459 auto val = accumulator.finalize();
460 p_dst[i] = saturate_cast<dtype>(static_cast<double>(val));
461
462 ++loop;
463 if (loop >= last_unreduced_dim) {
464 loop = 0;
465 ++main_index;
466 if (main_index < unprojected_steps.size())
467 origin = unprojected_steps[main_index];
468 } else {
469 origin += last_unreduced_step;
470 }
471 }
472 }

475 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
476 {
477 CV_TRACE_FUNCTION();
478 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
479
480 std::vector<Mat> inputs, outputs;
481 inputs_arr.getMatVector(inputs);
482
483 CV_Assert(!inputs.empty());
484 Mat& src = inputs[0];
485 std::vector<int> axes;
486 if (!this->axes.empty()) {
487 axes = this->axes;
488 } else if (inputs.size() >= 2) {
489 tensorToIntVec(inputs[1], axes);
490 }
491
492 MatShape inpShape = shape(src);
493 MatShape outShape;
494 if (axes.empty()) {
495 if (noop_with_empty_axes) {
496 outShape = inpShape;
497 } else {
498 if (keepdims) {
499 outShape.assign(inpShape.size(), 1);
500 } else {
501 outShape = MatShape::scalar();
502 }
503 }
504 } else {
505 std::vector<int> norm_axes = axes;
506 for (size_t i = 0; i < norm_axes.size(); ++i)
507 norm_axes[i] = normalize_axis(norm_axes[i], inpShape);
508 MatShape tmp = inpShape;
509 for (int a : norm_axes) tmp[a] = -1;
510 for (size_t i = 0; i < tmp.size(); ++i) {
511 if (tmp[i] == -1) {
512 if (keepdims) outShape.push_back(1);
513 } else {
514 outShape.push_back(tmp[i]);
515 }
516 }
517 if (outShape.size() == 0) outShape = MatShape{1};
518 axes = norm_axes;
519 }
520
521 auto kind = outputs_arr.kind();
522 if (kind == _InputArray::STD_VECTOR_MAT) {
523 outputs_arr.getMatVecRef()[0].fit(outShape, src.type());
524 } else {
525 CV_Assert(kind == _InputArray::STD_VECTOR_UMAT);
526 outputs_arr.getUMatVecRef()[0].fit(outShape, src.type());
527 }
528 outputs_arr.getMatVector(outputs);
529 Mat& dst = outputs[0];
530
531 if (src.depth() == CV_32F && src.isContinuous() && dst.isContinuous() &&
532 reduce_type != ReduceType::LOG_SUM_EXP) {
533 if (dst.total() == 1) {
534 CV_CPU_DISPATCH(reduceAllFloatParallel_, (src, dst, (int)reduce_type),
535 NEON, AVX2, AVX, BASELINE);
536 return;
537 }
538 size_t innerLen = 1;
539 if (reduceTrailingAxesLen(src, axes, innerLen) && innerLen > 1) {
540 CV_CPU_DISPATCH(reduceLastAxesFloatParallel_, (src, dst, innerLen, (int)reduce_type),
541 NEON, AVX2, AVX, BASELINE);
542 return;
543 }
544 }
545
546 typeDispatch(dst.type(), src, dst, axes, noop_with_empty_axes);
547 }

549 static bool reduceTrailingAxesLen(const Mat& src, const std::vector<int>& axes,
550 size_t& innerLen)
551 {
552 MatShape s = shape(src);
553 int nd = s.dims;
554 if (axes.empty() || (int)axes.size() > nd) return false;
555 std::vector<int> sorted_axes(axes.begin(), axes.end());
556 std::sort(sorted_axes.begin(), sorted_axes.end());
557 // Must be the trailing [nd - k .. nd - 1] block.
558 int k = (int)sorted_axes.size();
559 for (int i = 0; i < k; i++)
560 if (sorted_axes[i] != nd - k + i) return false;
561 innerLen = 1;
562 for (int i = nd - k; i < nd; i++) innerLen *= (size_t)s[i];
563 return true;
564 }

592 case ReduceType::MIN: ReduceInvoker<ReduceMin<T, WT, WT>>::run(std::forward<Args>(args)...); break;
593 case ReduceType::MEAN: ReduceInvoker<ReduceMean<T, WT, AccT>>::run(std::forward<Args>(args)...); break;
594 case ReduceType::SUM: ReduceInvoker<ReduceSum<T, WT, AccT>>::run(std::forward<Args>(args)...); break;
595 case ReduceType::L1: ReduceInvoker<ReduceL1<T, WT, AccT>>::run(std::forward<Args>(args)...); break;
596 case ReduceType::L2: ReduceInvoker<ReduceL2<T, WT, AccT>>::run(std::forward<Args>(args)...); break;
597 case ReduceType::PROD: ReduceInvoker<ReduceProd<T, WT, WT>>::run(std::forward<Args>(args)...); break;
598 case ReduceType::SUM_SQUARE: ReduceInvoker<ReduceSumSquare<T, WT, AccT>>::run(std::forward<Args>(args)...); break;

602 }
