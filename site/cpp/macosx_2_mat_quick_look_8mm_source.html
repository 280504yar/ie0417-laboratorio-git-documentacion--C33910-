44+ (NSString*)makeLabel:(BOOL)isIntType val:(NSNumber*)num {
45 if (isIntType) {
46 return [NSString stringWithFormat:@"%d", num.intValue];
47 } else {
48 int exponent = 1 + (int)log10(abs(num.doubleValue));
49 if (num.doubleValue == (double)num.intValue && num.doubleValue < 10000 && num.doubleValue > -10000) {
50 return [NSString stringWithFormat:@"%d", num.intValue];;
51 } else if (exponent <= 5 && exponent >= -1) {
52 return [NSString stringWithFormat:[NSString stringWithFormat:@"%%%d.%df", 6, MIN(5 - exponent, 4)], num.doubleValue];
53 } else {
54 return [[[NSString stringWithFormat:@"%.2e", num.doubleValue] stringByReplacingOccurrencesOfString:@"e+0" withString:@"e"] stringByReplacingOccurrencesOfString:@"e-0" withString:@"e-"];
55 }
56 }

59+ (id)matDebugQuickLookObject:(Mat*)mat {
60 // for smallish Mat objects display as a matrix
61 if ([mat dims] == 2 && [mat rows] <= 10 && [mat cols] <= 10 && [mat channels] == 1) {
62 FontGetter fontGetters[] = { getCMU, getBodoni72, getAnySerif, getSystemFont };
63 NSFont* font = nil;
64 for (int fontGetterIndex = 0; font==nil && fontGetterIndex < (sizeof(fontGetters)) / (sizeof(fontGetters[0])); fontGetterIndex++) {
65 font = fontGetters[fontGetterIndex]();
66 }
67 int elements = [mat rows] * [mat cols];
68 NSDictionary<NSAttributedStringKey,id>* textFontAttributes = @{ NSFontAttributeName: font, NSForegroundColorAttributeName: NSColor.blackColor };
69 NSMutableArray<NSNumber*>* rawData = [NSMutableArray new];
70 for (int dataIndex = 0; dataIndex < elements; dataIndex++) {
71 [rawData addObject:[NSNumber numberWithDouble:0]];
72 }
73 [mat get:0 col: 0 data: rawData];
74 BOOL isIntType = [mat depth] <= CV_32S;
75 NSMutableArray<NSString*>* labels = [NSMutableArray new];
76 NSMutableDictionary<NSString*, NSValue*>* boundingRects = [NSMutableDictionary dictionaryWithCapacity:elements];
77 int maxWidth = 0, maxHeight = 0;
78 for (NSNumber* number in rawData) {
79 NSString* label = [MatQuickLook makeLabel:isIntType val:number];
80 [labels addObject:label];
81 NSRect boundingRect = [label boundingRectWithSize:NSMakeSize(CGFLOAT_MAX, CGFLOAT_MAX) options:NSStringDrawingUsesLineFragmentOrigin attributes:textFontAttributes];
82 if (boundingRect.size.width > maxWidth) {
83 maxWidth = boundingRect.size.width;
84 }
85 if (boundingRect.size.height > maxHeight) {
86 maxHeight = boundingRect.size.height;
87 }
88 boundingRects[label] = [NSValue valueWithRect:boundingRect];
89 }

91 int rowGap = 8;
92 int colGap = 8;
93 int borderGap = 9;
94 int lineThickness = 4;
95 int lipWidth = 8;
96 int imageWidth = 2 * (borderGap + lipWidth) + maxWidth * [mat cols] + colGap * ([mat cols] - 1);
97 int imageHeight = 2 * (borderGap + lipWidth) + maxHeight * [mat rows] + rowGap * ([mat rows] - 1);
98 NSImage* image = [[NSImage alloc] initWithSize:NSMakeSize(imageWidth, imageHeight)];
99 NSBezierPath* leftBracket = [NSBezierPath new];
100 [leftBracket moveToPoint:NSMakePoint(borderGap, borderGap)];
101 [leftBracket relativeLineToPoint:NSMakePoint(0, imageHeight - 2 * borderGap)];
102 [leftBracket relativeLineToPoint:NSMakePoint(lineThickness + lipWidth, 0)];
103 [leftBracket relativeLineToPoint:NSMakePoint(0, -lineThickness)];
104 [leftBracket relativeLineToPoint:NSMakePoint(-lipWidth, 0)];
105 [leftBracket relativeLineToPoint:NSMakePoint(0, -(imageHeight - 2 * (borderGap + lineThickness)))];
106 [leftBracket relativeLineToPoint:NSMakePoint(lipWidth, 0)];
107 [leftBracket relativeLineToPoint:NSMakePoint(0, -lineThickness)];
108 [leftBracket relativeLineToPoint:NSMakePoint(-(lineThickness + lipWidth), 0)];
109 NSAffineTransform* reflect = [NSAffineTransform new];
110 [reflect scaleXBy:-1 yBy:1];
111 [reflect translateXBy:-imageWidth yBy:0];
112 NSBezierPath* rightBracket = [leftBracket copy];
113 [rightBracket transformUsingAffineTransform:reflect];
114
115 [image lockFocus];
116 [NSColor.whiteColor drawSwatchInRect:NSMakeRect(0, 0, imageWidth, imageHeight)];
117 [NSColor.blackColor set];
118 [leftBracket fill];
119 [rightBracket fill];

121 [labels enumerateObjectsUsingBlock:^(id label, NSUInteger index, BOOL *stop)
122 {
123 NSRect boundingRect = boundingRects[label].rectValue;
124 int row = [mat rows] - 1 - ((int)index / [mat cols]);
125 int col = (int)index % [mat cols];
126 int x = borderGap + lipWidth + col * (maxWidth + colGap) + (maxWidth - boundingRect.size.width) / 2;
127 int y = borderGap + lipWidth + row * (maxHeight + rowGap) + (maxHeight - boundingRect.size.height) / 2;
128 NSRect textRect = NSMakeRect(x, y, boundingRect.size.width, boundingRect.size.height);
129 [label drawInRect:textRect withAttributes:textFontAttributes];
130 }];
131 [image unlockFocus];
132 return image;
133 } else if (([mat dims] == 2) && ([mat type] == CV_8U || [mat type] == CV_8UC3 || [mat type] == CV_8UC4)) {
134 // convert to NSImage if the Mats has 2 dimensions and a type and number of channels consistent with it being a image
135 return [mat toNSImage];
136 } else if ([mat dims] == 2 && [mat channels] == 1) {
137 // for other Mats with 2 dimensions and one channel - generate heat map
138 Mat* normalized = [Mat new];
139 [Core normalize:mat dst:normalized alpha:0 beta:255 norm_type:NORM_MINMAX dtype:CV_8U];
140 Mat* normalizedKey = [[Mat alloc] initWithRows:[mat rows] + 10 cols:[mat cols] type:CV_8U];
141 std::vector<char> key;
142 for (int index = 0; index < [mat cols]; index++) {
143 key.push_back((char)(index * 256 / [mat cols]));
144 }
145 for (int index = 0; index < 10; index++) {
146 [normalizedKey put:@[[NSNumber numberWithInt:index], [NSNumber numberWithInt:0]] count:[mat cols] byteBuffer:key.data()];
147 }
148 [normalized copyTo:[normalizedKey submatRoi:[[Rect2i alloc] initWithX:0 y:10 width:[mat cols] height:[mat rows]]]];
149 Mat* colorMap = [Mat new];
150 [Imgproc applyColorMap:normalizedKey dst:colorMap colormap:COLORMAP_JET];
151 [Imgproc cvtColor:colorMap dst:colorMap code:COLOR_BGR2RGB];
152 return [colorMap toNSImage];
153 }
154 //everything just return the Mat description
155 return [mat description];
156}
