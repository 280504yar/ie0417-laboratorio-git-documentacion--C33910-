33class MlesacLoss {
34 public:
35 MlesacLoss(double threshold) : squared_thr(threshold * threshold), norm_thr(squared_thr*3), one_over_thr(1/norm_thr), inv_sq_thr(1/squared_thr) {}
36 double loss(double r2) const {
37 return r2 < norm_thr ? r2 * one_over_thr - 1 : 0;
38 }
39 double weight(double r2) const {
40 // use Cauchly weight
41 return 1.0 / (1.0 + r2 * inv_sq_thr);
42 }
43 const double squared_thr;
44 private:
45 const double norm_thr, one_over_thr, inv_sq_thr;
46};

57 RelativePoseJacobianAccumulator(
58 const Mat& correspondences_,
59 const std::vector<int> &sample_,

69 Matx33d essential_from_motion(const CameraPose &pose) const {
70 return Matx33d(0.0, -pose.t(2), pose.t(1),
71 pose.t(2), 0.0, -pose.t(0),
72 -pose.t(1), pose.t(0), 0.0) * pose.R;
73 }

75 double residual(const CameraPose &pose) const {
76 const Matx33d E = essential_from_motion(pose);
77 const float m11=static_cast<float>(E(0,0)), m12=static_cast<float>(E(0,1)), m13=static_cast<float>(E(0,2));
78 const float m21=static_cast<float>(E(1,0)), m22=static_cast<float>(E(1,1)), m23=static_cast<float>(E(1,2));
79 const float m31=static_cast<float>(E(2,0)), m32=static_cast<float>(E(2,1)), m33=static_cast<float>(E(2,2));
80 const auto * const pts = (float *) correspondences->data;
81 double cost = 0.0;
82 for (int k = 0; k < sample_size; ++k) {
83 const int idx = 4*sample[k];
84 const float x1=pts[idx], y1=pts[idx+1], x2=pts[idx+2], y2=pts[idx+3];
85 const float F_pt1_x = m11 * x1 + m12 * y1 + m13,
86 F_pt1_y = m21 * x1 + m22 * y1 + m23;
87 const float pt2_F_x = x2 * m11 + y2 * m21 + m31,
88 pt2_F_y = x2 * m12 + y2 * m22 + m32;
89 const float pt2_F_pt1 = x2 * F_pt1_x + y2 * F_pt1_y + m31 * x1 + m32 * y1 + m33;
90 const float r2 = pt2_F_pt1 * pt2_F_pt1 / (F_pt1_x * F_pt1_x + F_pt1_y * F_pt1_y +
91 pt2_F_x * pt2_F_x + pt2_F_y * pt2_F_y);
92 if (weights == nullptr)
93 cost += loss_fn.loss(r2);
94 else cost += weights[k] * loss_fn.loss(r2);
95 }
96 return cost;
97 }

99 void accumulate(const CameraPose &pose, Matx<double, 5, 5> &JtJ, Matx<double, 5, 1> &Jtr, Matx<double, 3, 2> &tangent_basis) const {
100 const auto * const pts = (float *) correspondences->data;
101 // We start by setting up a basis for the updates in the translation (orthogonal to t)
102 // We find the minimum element of t and cross product with the corresponding basis vector.
103 // (this ensures that the first cross product is not close to the zero vector)
104 Vec3d tangent_basis_col0;
105 if (std::abs(pose.t(0)) < std::abs(pose.t(1))) {
106 // x < y
107 if (std::abs(pose.t(0)) < std::abs(pose.t(2))) {
108 tangent_basis_col0 = pose.t.cross(Vec3d(1,0,0));
109 } else {
110 tangent_basis_col0 = pose.t.cross(Vec3d(0,0,1));
111 }
112 } else {
113 // x > y
114 if (std::abs(pose.t(1)) < std::abs(pose.t(2))) {
115 tangent_basis_col0 = pose.t.cross(Vec3d(0,1,0));
116 } else {
117 tangent_basis_col0 = pose.t.cross(Vec3d(0,0,1));
118 }
119 }
120 tangent_basis_col0 /= norm(tangent_basis_col0);
121 Vec3d tangent_basis_col1 = pose.t.cross(tangent_basis_col0);
122 tangent_basis_col1 /= norm(tangent_basis_col1);
123 for (int i = 0; i < 3; i++) {
124 tangent_basis(i,0) = tangent_basis_col0(i);
125 tangent_basis(i,1) = tangent_basis_col1(i);
126 }
127
128 const Matx33d E = essential_from_motion(pose);
129
130 // Matrices contain the jacobians of E w.r.t. the rotation and translation parameters
131 // Each column is vec(E*skew(e_k)) where e_k is k:th basis vector
132 const Matx<double, 9, 3> dR = {0., -E(0,2), E(0,1),
133 0., -E(1,2), E(1,1),
134 0., -E(2,2), E(2,1),
135 E(0,2), 0., -E(0,0),
136 E(1,2), 0., -E(1,0),
137 E(2,2), 0., -E(2,0),
138 -E(0,1), E(0,0), 0.,
139 -E(1,1), E(1,0), 0.,
140 -E(2,1), E(2,0), 0.};
141
142 Matx<double, 9, 2> dt;
143 // Each column is vec(skew(tangent_basis[k])*R)
144 for (int i = 0; i <= 2; i+=1) {
145 const Vec3d r_i(pose.R(0,i), pose.R(1,i), pose.R(2,i));
146 for (int j = 0; j <= 1; j+= 1) {
147 const Vec3d v = (j == 0 ? tangent_basis_col0 : tangent_basis_col1).cross(r_i);
148 for (int k = 0; k < 3; k++) {
149 dt(3*i+k,j) = v[k];
150 }
151 }
152 }
153
154 for (int k = 0; k < sample_size; ++k) {
155 const auto point_idx = 4*sample[k];
156 const Vec3d pt1 (pts[point_idx], pts[point_idx+1], 1), pt2 (pts[point_idx+2], pts[point_idx+3], 1);
157 const double C = pt2.dot(E * pt1);
158
159 // J_C is the Jacobian of the epipolar constraint w.r.t. the image points
160 const Vec4d J_C ((E.col(0).t() * pt2)[0], (E.col(1).t() * pt2)[0], (E.row(0) * pt1)[0], (E.row(1) * pt1)[0]);
161 const double nJ_C = norm(J_C);
162 const double inv_nJ_C = 1.0 / nJ_C;
163 const double r = C * inv_nJ_C;
164
165 if (r*r > loss_fn.squared_thr) continue;
166
167 // Compute weight from robust loss function (used in the IRLS)
168 double weight = loss_fn.weight(r * r) / sample_size;
169 if (weights != nullptr)
170 weight = weights[k] * weight;
171
172 if(weight < DBL_EPSILON)
173 continue;
174
175 // Compute Jacobian of Sampson error w.r.t the fundamental/essential matrix (3x3)
176 Matx<double, 1, 9> dF (pt1(0) * pt2(0), pt1(0) * pt2(1), pt1(0), pt1(1) * pt2(0), pt1(1) * pt2(1), pt1(1), pt2(0), pt2(1), 1.0);
177 const double s = C * inv_nJ_C * inv_nJ_C;
178 dF(0) -= s * (J_C(2) * pt1(0) + J_C(0) * pt2(0));
179 dF(1) -= s * (J_C(3) * pt1(0) + J_C(0) * pt2(1));
180 dF(2) -= s * (J_C(0));
181 dF(3) -= s * (J_C(2) * pt1(1) + J_C(1) * pt2(0));
182 dF(4) -= s * (J_C(3) * pt1(1) + J_C(1) * pt2(1));
183 dF(5) -= s * (J_C(1));
184 dF(6) -= s * (J_C(2));
185 dF(7) -= s * (J_C(3));
186 dF *= inv_nJ_C;
187
188 // and then w.r.t. the pose parameters (rotation + tangent basis for translation)
189 const Matx13d dFdR = dF * dR;
190 const Matx12d dFdt = dF * dt;
191 const Matx<double, 1, 5> J (dFdR(0), dFdR(1), dFdR(2), dFdt(0), dFdt(1));
192
193 // Accumulate into JtJ and Jtr
194 Jtr += weight * C * inv_nJ_C * J.t();
195 JtJ(0, 0) += weight * (J(0) * J(0));
196 JtJ(1, 0) += weight * (J(1) * J(0));
197 JtJ(1, 1) += weight * (J(1) * J(1));
198 JtJ(2, 0) += weight * (J(2) * J(0));
199 JtJ(2, 1) += weight * (J(2) * J(1));
200 JtJ(2, 2) += weight * (J(2) * J(2));
201 JtJ(3, 0) += weight * (J(3) * J(0));
202 JtJ(3, 1) += weight * (J(3) * J(1));
203 JtJ(3, 2) += weight * (J(3) * J(2));
204 JtJ(3, 3) += weight * (J(3) * J(3));
205 JtJ(4, 0) += weight * (J(4) * J(0));
206 JtJ(4, 1) += weight * (J(4) * J(1));
207 JtJ(4, 2) += weight * (J(4) * J(2));
208 JtJ(4, 3) += weight * (J(4) * J(3));
209 JtJ(4, 4) += weight * (J(4) * J(4));
210 }
211 }

214bool satisfyCheirality (const Matx33d& R, const Vec3d &t, const Vec3d &x1, const Vec3d &x2) {
215 // This code assumes that x1 and x2 are unit vectors
216 const auto Rx1 = R * x1;
217 // lambda_2 * x2 = R * ( lambda_1 * x1 ) + t
218 // [1 a; a 1] * [lambda1; lambda2] = [b1; b2]
219 // [lambda1; lambda2] = [1 -a; -a 1] * [b1; b2] / (1 - a*a)
220 const double a = -Rx1.dot(x2), b1 = -Rx1.dot(t), b2 = x2.dot(t);
221 // Note that we drop the factor 1.0/(1-a*a) since it is always positive.
222 return (b1 - a * b2 > 0) && (-a * b1 + b2 > 0);
223}

225int refine_relpose(const Mat &correspondences_,
226 const std::vector<int> &sample_,
227 const int sample_size_,
228 CameraPose *pose,
229 const BundleOptions &opt,
230 const double* weights) {
231 MlesacLoss loss_fn(opt.loss_scale);
232 RelativePoseJacobianAccumulator accum(correspondences_, sample_, sample_size_, loss_fn, weights);
233 // return lm_5dof_impl(accum, pose, opt);
234
235 Matx<double, 5, 5> JtJ;
236 Matx<double, 5, 1> Jtr;
237 Matx<double, 3, 2> tangent_basis;
238 Matx33d sw = Matx33d::zeros();
239 double lambda = opt.initial_lambda;
240
241 // Compute initial cost
242 double cost = accum.residual(*pose);
243 bool recompute_jac = true;
244 int iter;
245 for (iter = 0; iter < opt.max_iterations; ++iter) {
246 // We only recompute jacobian and residual vector if last step was successful
247 if (recompute_jac) {
248 std::fill(JtJ.val, JtJ.val+25, 0);
249 std::fill(Jtr.val, Jtr.val +5, 0);
250 accum.accumulate(*pose, JtJ, Jtr, tangent_basis);
251 if (norm(Jtr) < opt.gradient_tol)
252 break;
253 }
254
255 // Add dampening
256 JtJ(0, 0) += lambda;
257 JtJ(1, 1) += lambda;
258 JtJ(2, 2) += lambda;
259 JtJ(3, 3) += lambda;
260 JtJ(4, 4) += lambda;
261
262 Matx<double, 5, 1> sol;
263 Matx<double, 5, 5> JtJ_symm = JtJ;
264 for (int i = 0; i < 5; i++)
265 for (int j = i+1; j < 5; j++)
266 JtJ_symm(i,j) = JtJ(j,i);
267
268 const bool success = solve(-JtJ_symm, Jtr, sol);
269 if (!success || norm(sol) < opt.step_tol)
270 break;
271
272 Vec3d w (sol(0,0), sol(1,0), sol(2,0));
273 const double theta = norm(w);
274 w /= theta;
275 const double a = std::sin(theta);
276 const double b = std::cos(theta);
277 sw(0, 1) = -w(2);
278 sw(0, 2) = w(1);
279 sw(1, 2) = -w(0);
280 sw(1, 0) = w(2);
281 sw(2, 0) = -w(1);
282 sw(2, 1) = w(0);
283
284 CameraPose pose_new;
285 pose_new.R = pose->R + pose->R * (a * sw + (1 - b) * sw * sw);
286 // In contrast to the 6dof case, we don't apply R here
287 // (since this can already be added into tangent_basis)
288 pose_new.t = pose->t + Vec3d(Mat(tangent_basis * Matx21d(sol(3,0), sol(4,0))));
289 double cost_new = accum.residual(pose_new);
290
291 if (cost_new < cost) {
292 *pose = pose_new;
293 lambda /= 10;
294 cost = cost_new;
295 recompute_jac = true;
296 } else {
297 JtJ(0, 0) -= lambda;
298 JtJ(1, 1) -= lambda;
299 JtJ(2, 2) -= lambda;
300 JtJ(3, 3) -= lambda;
301 JtJ(4, 4) -= lambda;
302 lambda *= 10;
303 recompute_jac = false;
304 }
305 }
306 return iter;
307}
