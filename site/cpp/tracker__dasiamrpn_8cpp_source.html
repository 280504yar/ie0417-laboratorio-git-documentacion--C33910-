39template <typename T> static
40T sizeCal(const T& w, const T& h)
41{
42 T pad = (w + h) * T(0.5);
43 T sz2 = (w + pad) * (h + pad);
44 return sqrt(sz2);
45}
46
47template <>
48Mat sizeCal(const Mat& w, const Mat& h)
49{
50 Mat pad = (w + h) * 0.5;
51 Mat sz2 = (w + pad).mul((h + pad));
52
53 cv::sqrt(sz2, sz2);
54 return sz2;
55}

148void TrackerDaSiamRPNImpl::trackerInit(Mat img)
149{
150 Rect2f targetBox = trackState.targetBox;
151 Mat anchors = generateAnchors();
152 trackState.anchors = anchors;
153
154 Mat windows = generateHanningWindow();
155
156 trackState.windows = windows;
157 trackState.imgSize = img.size();
158
159 trackState.avgChans = mean(img);
160 float wc = targetBox.width + trackState.contextAmount * (targetBox.width + targetBox.height);
161 float hc = targetBox.height + trackState.contextAmount * (targetBox.width + targetBox.height);
162 float sz = (float)cvRound(sqrt(wc * hc));
163
164 Mat zCrop = getSubwindow(img, targetBox, sz, trackState.avgChans);
165 Mat blob;
166
167 dnn::blobFromImage(zCrop, blob, 1.0, Size(trackState.exemplarSize, trackState.exemplarSize), Scalar(), trackState.swapRB, false, CV_32F);
168 siamRPN.setInput(blob);
169 Mat out1;
170 siamRPN.forward(out1, "onnx_node_output_0!63");
171
172 siamKernelCL1.setInput(out1);
173 siamKernelR1.setInput(out1);
174
175 Mat cls1 = siamKernelCL1.forward();
176 Mat r1 = siamKernelR1.forward();
177 std::vector<int> r1_shape = { 20, 256, 4, 4 }, cls1_shape = { 10, 256, 4, 4 };
178
179 siamRPN.setParam("onnx_node_output_0!65", 0, r1.reshape(0, r1_shape));
180 siamRPN.setParam("onnx_node_output_0!68", 0, cls1.reshape(0, cls1_shape));
181}

196void TrackerDaSiamRPNImpl::trackerEval(Mat img)
197{
198 Rect2f targetBox = trackState.targetBox;
199
200 float wc = targetBox.height + trackState.contextAmount * (targetBox.width + targetBox.height);
201 float hc = targetBox.width + trackState.contextAmount * (targetBox.width + targetBox.height);
202
203 float sz = sqrt(wc * hc);
204 float scaleZ = trackState.exemplarSize / sz;
205
206 float searchSize = float((trackState.instanceSize - trackState.exemplarSize) / 2);
207 float pad = searchSize / scaleZ;
208 float sx = sz + 2 * pad;
209
210 Mat xCrop = getSubwindow(img, targetBox, (float)cvRound(sx), trackState.avgChans);
211
212 Mat blob;
213 std::vector<Mat> outs;
214 std::vector<String> outNames;
215 Mat delta, score;
216 Mat sc, rc, penalty, pscore;
217
218 dnn::blobFromImage(xCrop, blob, 1.0, Size(trackState.instanceSize, trackState.instanceSize), Scalar(), trackState.swapRB, false, CV_32F);
219
220 siamRPN.setInput(blob);
221
222 outNames = siamRPN.getUnconnectedOutLayersNames();
223 siamRPN.forward(outs, outNames);
224
225 delta = outs[0];
226 score = outs[1];
227
228 score = score.reshape(0, { 2, trackState.anchorNum, trackState.scoreSize, trackState.scoreSize });
229 delta = delta.reshape(0, { 4, trackState.anchorNum, trackState.scoreSize, trackState.scoreSize });
230
231 softmax(score, score);
232
233 targetBox.width *= scaleZ;
234 targetBox.height *= scaleZ;
235
236 score = score.row(1);
237 score = score.reshape(0, { 5, 19, 19 });
238
239 // Post processing
240 delta.row(0) = delta.row(0).mul(trackState.anchors.row(2)) + trackState.anchors.row(0);
241 delta.row(1) = delta.row(1).mul(trackState.anchors.row(3)) + trackState.anchors.row(1);
242 exp(delta.row(2), delta.row(2));
243 delta.row(2) = delta.row(2).mul(trackState.anchors.row(2));
244 exp(delta.row(3), delta.row(3));
245 delta.row(3) = delta.row(3).mul(trackState.anchors.row(3));
246
247 sc = sizeCal(delta.row(2), delta.row(3)) / sizeCal(targetBox.width, targetBox.height);
248 elementMax(sc);
249
250 rc = delta.row(2).mul(1 / delta.row(3));
251 rc = (targetBox.width / targetBox.height) / rc;
252 elementMax(rc);
253
254 // Calculating the penalty
255 exp(((rc.mul(sc) - 1.) * trackState.penaltyK * (-1.0)), penalty);
256 penalty = penalty.reshape(0, { trackState.anchorNum, trackState.scoreSize, trackState.scoreSize });
257
258 pscore = penalty.mul(score);
259 pscore = pscore * (1.0 - trackState.windowInfluence) + trackState.windows * trackState.windowInfluence;
260
261 int bestID[2] = { 0, 0 };
262 // Find the index of best score.
263 minMaxIdx(pscore.reshape(0, { trackState.anchorNum * trackState.scoreSize * trackState.scoreSize, 1 }), 0, 0, 0, bestID);
264 delta = delta.reshape(0, { 4, trackState.anchorNum * trackState.scoreSize * trackState.scoreSize });
265 penalty = penalty.reshape(0, { trackState.anchorNum * trackState.scoreSize * trackState.scoreSize, 1 });
266 score = score.reshape(0, { trackState.anchorNum * trackState.scoreSize * trackState.scoreSize, 1 });
267
268 int index[2] = { 0, bestID[0] };
269 Rect2f resBox = { 0, 0, 0, 0 };
270
271 resBox.x = delta.at<float>(index) / scaleZ;
272 index[0] = 1;
273 resBox.y = delta.at<float>(index) / scaleZ;
274 index[0] = 2;
275 resBox.width = delta.at<float>(index) / scaleZ;
276 index[0] = 3;
277 resBox.height = delta.at<float>(index) / scaleZ;
278
279 float lr = penalty.at<float>(bestID) * score.at<float>(bestID) * trackState.lr;
280
281 resBox.x = resBox.x + targetBox.x;
282 resBox.y = resBox.y + targetBox.y;
283 targetBox.width /= scaleZ;
284 targetBox.height /= scaleZ;
285
286 resBox.width = targetBox.width * (1 - lr) + resBox.width * lr;
287 resBox.height = targetBox.height * (1 - lr) + resBox.height * lr;
288
289 resBox.x = float(fmax(0., fmin(float(trackState.imgSize.width), resBox.x)));
290 resBox.y = float(fmax(0., fmin(float(trackState.imgSize.height), resBox.y)));
291 resBox.width = float(fmax(10., fmin(float(trackState.imgSize.width), resBox.width)));
292 resBox.height = float(fmax(10., fmin(float(trackState.imgSize.height), resBox.height)));
293
294 trackState.targetBox = resBox;
295 trackState.tracking_score = score.at<float>(bestID);
296}

303void TrackerDaSiamRPNImpl::softmax(const Mat& src, Mat& dst)
304{
305 Mat maxVal;
306 cv::max(src.row(1), src.row(0), maxVal);
307
308 src.row(1) -= maxVal;
309 src.row(0) -= maxVal;
310
311 exp(src, dst);
312
313 Mat sumVal = dst.row(0) + dst.row(1);
314 dst.row(0) = dst.row(0) / sumVal;
315 dst.row(1) = dst.row(1) / sumVal;
316}

318void TrackerDaSiamRPNImpl::elementMax(Mat& src)
319{
320 int* p = src.size.p;
321 int index[4] = { 0, 0, 0, 0 };
322 for (int n = 0; n < *p; n++)
323 {
324 for (int k = 0; k < *(p + 1); k++)
325 {
326 for (int i = 0; i < *(p + 2); i++)
327 {
328 for (int j = 0; j < *(p + 3); j++)
329 {
330 index[0] = n, index[1] = k, index[2] = i, index[3] = j;
331 float& v = src.at<float>(index);
332 v = fmax(v, 1.0f / v);
333 }
334 }
335 }
336 }

354Mat TrackerDaSiamRPNImpl::generateAnchors()
355{
356 int totalStride = trackState.totalStride, scales = trackState.scale, scoreSize = trackState.scoreSize;
357 std::vector<float> ratios = trackState.ratios;
358 std::vector<Rect2f> baseAnchors;
359 int anchorNum = int(ratios.size());
360 int size = totalStride * totalStride;
361
362 float ori = -(float(scoreSize / 2)) * float(totalStride);
363
364 for (auto i = 0; i < anchorNum; i++)
365 {
366 int ws = int(sqrt(size / ratios[i]));
367 int hs = int(ws * ratios[i]);
368
369 float wws = float(ws) * scales;
370 float hhs = float(hs) * scales;
371 Rect2f anchor = { 0, 0, wws, hhs };
372 baseAnchors.push_back(anchor);
373 }
374
375 int anchorIndex[4] = { 0, 0, 0, 0 };
376 const int sizes[4] = { 4, (int)ratios.size(), scoreSize, scoreSize };
377 Mat anchors(4, sizes, CV_32F);
378
379 for (auto i = 0; i < scoreSize; i++)
380 {
381 for (auto j = 0; j < scoreSize; j++)
382 {
383 for (auto k = 0; k < anchorNum; k++)
384 {
385 anchorIndex[0] = 1, anchorIndex[1] = k, anchorIndex[2] = i, anchorIndex[3] = j;
386 anchors.at<float>(anchorIndex) = ori + totalStride * i;
387
388 anchorIndex[0] = 0;
389 anchors.at<float>(anchorIndex) = ori + totalStride * j;
390
391 anchorIndex[0] = 2;
392 anchors.at<float>(anchorIndex) = baseAnchors[k].width;
393
394 anchorIndex[0] = 3;
395 anchors.at<float>(anchorIndex) = baseAnchors[k].height;
396 }
397 }
398 }
399
400 return anchors;
401}

403Mat TrackerDaSiamRPNImpl::getSubwindow(Mat& img, const Rect2f& targetBox, float originalSize, Scalar avgChans)
404{
405 Mat zCrop, dst;
406 Size imgSize = img.size();
407 float c = (originalSize + 1) / 2;
408 float xMin = (float)cvRound(targetBox.x - c);
409 float xMax = xMin + originalSize - 1;
410 float yMin = (float)cvRound(targetBox.y - c);
411 float yMax = yMin + originalSize - 1;
412
413 int leftPad = (int)(fmax(0., -xMin));
414 int topPad = (int)(fmax(0., -yMin));
415 int rightPad = (int)(fmax(0., xMax - imgSize.width + 1));
416 int bottomPad = (int)(fmax(0., yMax - imgSize.height + 1));
417
418 xMin = xMin + leftPad;
419 xMax = xMax + leftPad;
420 yMax = yMax + topPad;
421 yMin = yMin + topPad;
422
423 if (topPad == 0 && bottomPad == 0 && leftPad == 0 && rightPad == 0)
424 {
425 img(Rect(int(xMin), int(yMin), int(xMax - xMin + 1), int(yMax - yMin + 1))).copyTo(zCrop);
426 }
427 else
428 {
429 copyMakeBorder(img, dst, topPad, bottomPad, leftPad, rightPad, BORDER_CONSTANT, avgChans);
430 dst(Rect(int(xMin), int(yMin), int(xMax - xMin + 1), int(yMax - yMin + 1))).copyTo(zCrop);
431 }
432
433 return zCrop;
434}
