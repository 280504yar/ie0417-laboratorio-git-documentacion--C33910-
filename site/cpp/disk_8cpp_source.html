51 void detectAndCompute(InputArray _image, InputArray _mask,
52 std::vector<KeyPoint>& keypoints,
53 OutputArray _descriptors,
54 bool useProvidedKeypoints) CV_OVERRIDE
55 {
56 CV_Assert(!useProvidedKeypoints && "DISK does not support providing keypoints externally");
57
58 keypoints.clear();
59
60 Mat image = _image.getMat();
61 if (image.empty())
62 {
63 if (_descriptors.needed())
64 _descriptors.release();
65 return;
66 }
67
68 Mat mask = _mask.getMat();
69 if (!mask.empty())
70 {
71 CV_Assert(mask.type() == CV_8UC1);
72 CV_Assert(mask.size() == image.size());
73 }
74
75 const Size netSize = (imageSize_.width > 0 && imageSize_.height > 0) ? imageSize_ : kDefaultDiskInputSize;
76 const float scaleX = static_cast<float>(image.cols) / netSize.width;
77 const float scaleY = static_cast<float>(image.rows) / netSize.height;
78
79 Mat blob;
80 // 1/255 normalization, swap BGR->RGB, no mean subtraction.
81 blobFromImage(image, blob, 1.0 / 255.0, netSize, Scalar(), /*swapRB=*/true, /*crop=*/false);
82 net_.setInput(blob, "image");
83
84 const std::vector<String> outNames = {"keypoints", "scores", "descriptors"};
85 std::vector<Mat> outs;
86 net_.forward(outs, outNames);
87 CV_Assert(outs.size() == 3);
88
89 // DISK's ONNX export emits keypoints as int64 pixel coordinates, scores and
90 // descriptors as float32. Reshape each output to (N, *) for row-wise indexing.
91 Mat kptsBlob = outs[0].reshape(1, outs[0].size[1]); // N x 2, int64
92 Mat scoresBlob = outs[1].reshape(1, outs[1].size[1]); // N x 1, float32
93 Mat descBlob = outs[2].reshape(1, outs[2].size[1]); // N x D, float32
94
95 CV_Assert(kptsBlob.depth() == CV_64S);
96 CV_Assert(scoresBlob.depth() == CV_32F);
97 CV_Assert(descBlob.depth() == CV_32F);
98
99 const int numFeatures = kptsBlob.rows;
100 CV_Assert(scoresBlob.rows == numFeatures);
101 CV_Assert(descBlob.rows == numFeatures);
102
103 const int64_t* kptsData = kptsBlob.ptr<int64_t>();
104 const float* scoresData = scoresBlob.ptr<float>();
105
106 std::vector<int> validIndices;
107 validIndices.reserve(numFeatures);
108 keypoints.reserve(numFeatures);
109
110 for (int i = 0; i < numFeatures; ++i)
111 {
112 const float score = scoresData[i];
113 if (score <= scoreThreshold_)
114 continue;
115
116 const float x = static_cast<float>(kptsData[i * 2]) * scaleX;
117 const float y = static_cast<float>(kptsData[i * 2 + 1]) * scaleY;
118
119 if (!mask.empty())
120 {
121 const int ix = cvFloor(x);
122 const int iy = cvFloor(y);
123 if (ix < 0 || iy < 0 || ix >= mask.cols || iy >= mask.rows)
124 continue;
125 if (mask.at<uchar>(iy, ix) == 0)
126 continue;
127 }
128
129 keypoints.emplace_back(x, y, 1.0f, -1.0f, score);
130 validIndices.push_back(i);
131 }
132
133 if (maxKeypoints_ > 0 && static_cast<int>(keypoints.size()) > maxKeypoints_)
134 {
135 std::vector<int> order(keypoints.size());
136 std::iota(order.begin(), order.end(), 0);
137 std::partial_sort(order.begin(), order.begin() + maxKeypoints_, order.end(),
138 [&](int a, int b) { return keypoints[a].response > keypoints[b].response; });
139 order.resize(maxKeypoints_);
140
141 std::vector<KeyPoint> kept;
142 std::vector<int> keptIdx;
143 kept.reserve(maxKeypoints_);
144 keptIdx.reserve(maxKeypoints_);
145 for (int idx : order)
146 {
147 kept.push_back(keypoints[idx]);
148 keptIdx.push_back(validIndices[idx]);
149 }
150 keypoints.swap(kept);
151 validIndices.swap(keptIdx);
152 }
153
154 if (_descriptors.needed())
155 {
156 if (validIndices.empty())
157 {
158 _descriptors.release();
159 return;
160 }
161 const int dim = descBlob.cols;
162 _descriptors.create(static_cast<int>(validIndices.size()), dim, CV_32F);
163 Mat descriptors = _descriptors.getMat();
164 for (size_t i = 0; i < validIndices.size(); ++i)
165 descBlob.row(validIndices[i]).copyTo(descriptors.row(static_cast<int>(i)));
166 }
167 }

199 {
200 if (size.width == 0 && size.height == 0)

203 CV_Assert(size.width % kDiskStride == 0);
