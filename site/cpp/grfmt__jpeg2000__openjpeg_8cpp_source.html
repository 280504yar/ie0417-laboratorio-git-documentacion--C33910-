64template <class Traits>
65class ChannelsIterator
66{
67public:
68 using difference_type = typename Traits::difference_type;
69 using value_type = typename Traits::value_type;
70 using pointer = typename Traits::pointer;
71 using reference = typename Traits::reference;
72 using iterator_category = std::random_access_iterator_tag;
73
74 ChannelsIterator(pointer ptr, std::size_t channel, std::size_t channels_count)
75 : ptr_ { ptr + channel }, step_ { channels_count }
76 {
77 }
78
79 /* Element Access */
80 reference operator*() const
81 {
82 return *ptr_;
83 }
84
85 pointer operator->() const
86 {
87 return &(operator*());
88 }
89
90 reference operator[](difference_type n) const
91 {
92 return *(*this + n);
93 }
94
95 /* Iterator movement */
96 ChannelsIterator<Traits>& operator++()
97 {
98 ptr_ += step_;
99 return *this;
100 }
101
102 ChannelsIterator<Traits> operator++(int)
103 {
104 ChannelsIterator ret(*this);
105 ++(*this);
106 return ret;
107 }
108
109 ChannelsIterator<Traits>& operator--()
110 {
111 ptr_ -= step_;
112 return *this;
113 }
114
115 ChannelsIterator<Traits> operator--(int)
116 {
117 ChannelsIterator ret(*this);
118 --(*this);
119 return ret;
120 }
121
122 ChannelsIterator<Traits>& operator-=(difference_type n)
123 {
124 ptr_ -= n * step_;
125 return *this;
126 }
127
128 ChannelsIterator<Traits>& operator+=(difference_type n)
129 {
130 ptr_ += n * step_;
131 return *this;
132 }
133
134 ChannelsIterator<Traits> operator-(difference_type n) const
135 {
136 return ChannelsIterator<Traits>(*this) -= n;
137 }
138
139 ChannelsIterator<Traits> operator+(difference_type n) const
140 {
141 return ChannelsIterator<Traits>(*this) += n;
142 }
143
144 difference_type operator-(const ChannelsIterator<Traits>& other) const
145 {
146 return (ptr_ - other.ptr_) / step_;
147 }
148
149 /* Comparison */
150 bool operator==(const ChannelsIterator<Traits>& other) const CV_NOEXCEPT
151 {
152 return ptr_ == other.ptr_;
153 }
154
155 bool operator!=(const ChannelsIterator<Traits>& other) const CV_NOEXCEPT
156 {
157 return !(*this == other);
158 }
159
160 bool operator<(const ChannelsIterator<Traits>& other) const CV_NOEXCEPT
161 {
162 return ptr_ < other.ptr_;
163 }
164
165 bool operator>(const ChannelsIterator<Traits>& other) const CV_NOEXCEPT
166 {
167 return other < *this;
168 }
169
170 bool operator>=(const ChannelsIterator<Traits>& other) const CV_NOEXCEPT
171 {
172 return !(*this < other);
173 }
174
175 bool operator<=(const ChannelsIterator<Traits>& other) const CV_NOEXCEPT
176 {
177 return !(other < *this);
178 }
179
180private:
181 pointer ptr_{nullptr};
182 std::size_t step_{1};
183};

191template<typename OutT, typename InT>
192void copyToMatImpl(std::vector<InT*>&& in, Mat& out, uint8_t shift)
193{
194 using ChannelsIt = ChannelsIterator<NonConstItTraits<OutT>>;
195
196 Size size = out.size();
197 if (out.isContinuous())
198 {
199 size.width *= size.height;
200 size.height = 1;
201 }
202
203 const bool isShiftRequired = shift != 0;
204
205 const std::size_t channelsCount = in.size();
206
207 if (isShiftRequired)
208 {
209 for (int i = 0; i < size.height; ++i)
210 {
211 auto rowPtr = out.ptr<OutT>(i);
212 for (std::size_t c = 0; c < channelsCount; ++c)
213 {
214 const auto first = in[c];
215 const auto last = first + size.width;
216 auto dOut = ChannelsIt(rowPtr, c, channelsCount);
217 std::transform(first, last, dOut, [shift](InT val) -> OutT { return static_cast<OutT>(val >> shift); });
218 in[c] += size.width;
219 }
220 }
221 }
222 else
223 {
224 for (int i = 0; i < size.height; ++i)
225 {
226 auto rowPtr = out.ptr<OutT>(i);
227 for (std::size_t c = 0; c < channelsCount; ++c)
228 {
229 const auto first = in[c];
230 const auto last = first + size.width;
231 auto dOut = ChannelsIt(rowPtr, c, channelsCount);
232 std::transform(first, last, dOut, [](InT val) -> OutT { return static_cast<OutT>(val); });
233 in[c] += size.width;
234 }
235 }
236 }
237}

239template<typename InT>
240void copyToMat(std::vector<const InT*>&& in, Mat& out, uint8_t shift)
241{
242 switch (out.depth())
243 {
244 case CV_8U:
245 copyToMatImpl<uint8_t>(std::move(in), out, shift);
246 break;
247 case CV_16U:
248 copyToMatImpl<uint16_t>(std::move(in), out, shift);
249 break;
250 default:
251 CV_Error(Error::StsNotImplemented, "only depth CV_8U and CV16_U are supported");
252 }

255template<typename InT, typename OutT>
256void copyFromMatImpl(const Mat& in, std::vector<OutT*>&& out)
257{
258 using ChannelsIt = ChannelsIterator<ConstItTraits<InT>>;
259
260 Size size = in.size();
261 if (in.isContinuous())
262 {
263 size.width *= size.height;
264 size.height = 1;
265 }
266
267 const std::size_t outChannelsCount = out.size();
268
269 for (int i = 0; i < size.height; ++i)
270 {
271 const InT* row = in.ptr<InT>(i);
272 for (std::size_t c = 0; c < outChannelsCount; ++c)
273 {
274 auto first = ChannelsIt(row, c, outChannelsCount);
275 auto last = first + size.width;
276 out[c] = std::copy(first, last, out[c]);
277 }
278 }
279}

326opj_cparameters setupEncoderParameters(const std::vector<int>& params)
327{
328 opj_cparameters parameters;
329 opj_set_default_encoder_parameters(&parameters);
330 bool rate_is_specified = false;
331 for (size_t i = 0; i < params.size(); i += 2)
332 {
333 const int value = params[i + 1];
334 switch (params[i])
335 {
336 case cv::IMWRITE_JPEG2000_COMPRESSION_X1000:
337 {
338 const int compression = std::min(std::max(value, 1), 1000);
339 parameters.tcp_rates[0] = 1000.f / static_cast<float>(compression);
340 if(value != compression) {
341 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_JPEG2000_COMPRESSION_X1000 must be between 1 to 1000. It is fallbacked to 1", value));
342 }
343 rate_is_specified = true;
344 }
345 break;
346 default:
347 break;
348 }
349 }
350 parameters.tcp_numlayers = 1;
351 parameters.cp_disto_alloc = 1;
352 if (!rate_is_specified)
353 {
354 parameters.tcp_rates[0] = 4;
355 }
356 return parameters;
357}

359bool decodeSRGBData(const opj_image_t& inImg, cv::Mat& outImg, uint8_t shift, bool use_rgb)
360{
361 using ImageComponents = std::vector<const OPJ_INT32*>;
362
363 const int inChannels = inImg.numcomps;
364 const int outChannels = outImg.channels();
365
366 if (outChannels == 1)
367 {
368 // Assume gray (+ alpha) for 1 channels -> gray
369 if (inChannels <= 2)
370 {
371 copyToMat(ImageComponents { inImg.comps[0].data }, outImg, shift);
372 }
373 // Assume RGB for >= 3 channels -> gray
374 else
375 {
376 Mat tmp(outImg.size(), CV_MAKETYPE(outImg.depth(), 3));
377 copyToMat(ImageComponents { inImg.comps[2].data, inImg.comps[1].data, inImg.comps[0].data },
378 tmp, shift);
379 cvtColor(tmp, outImg, COLOR_BGR2GRAY);
380 }
381 return true;
382 }
383
384 if (inChannels >= 3)
385 {
386 int swap_rb = use_rgb ? 0 : 2;
387 // Assume RGB (+ alpha) for 3 channels -> BGR
388 ImageComponents incomps { inImg.comps[swap_rb].data, inImg.comps[1].data, inImg.comps[swap_rb^2].data };
389 // Assume RGBA for 4 channels -> BGRA
390 if (outChannels > 3)
391 {
392 incomps.push_back(inImg.comps[3].data);
393 }
394 copyToMat(std::move(incomps), outImg, shift);
395 return true;
396 }
397 CV_LOG_ERROR(NULL,
398 cv::format("OpenJPEG2000: unsupported conversion from %d components to %d for SRGB image decoding",
399 inChannels, outChannels));
400 return false;
401}

403bool decodeGrayscaleData(const opj_image_t& inImg, cv::Mat& outImg, uint8_t shift, bool)
404{
405 using ImageComponents = std::vector<const OPJ_INT32*>;
406
407 const int inChannels = inImg.numcomps;
408 const int outChannels = outImg.channels();
409
410 if (outChannels == 1 || outChannels == 3)
411 {
412 copyToMat(ImageComponents(outChannels, inImg.comps[0].data), outImg, shift);
413 return true;
414 }
415 CV_LOG_ERROR(NULL,
416 cv::format("OpenJPEG2000: unsupported conversion from %d components to %d for Grayscale image decoding",
417 inChannels, outChannels));
418 return false;
419}

421bool decodeSYCCData(const opj_image_t& inImg, cv::Mat& outImg, uint8_t shift, bool use_rgb)
422{
423 using ImageComponents = std::vector<const OPJ_INT32*>;
424
425 const int inChannels = inImg.numcomps;
426 const int outChannels = outImg.channels();
427
428 if (outChannels == 1) {
429 copyToMat(ImageComponents { inImg.comps[0].data }, outImg, shift);
430 return true;
431 }
432
433 if (outChannels == 3 && inChannels >= 3) {
434 copyToMat(ImageComponents { inImg.comps[0].data, inImg.comps[1].data, inImg.comps[2].data },
435 outImg, shift);
436 if (use_rgb)
437 cvtColor(outImg, outImg, COLOR_YUV2RGB);
438 else
439 cvtColor(outImg, outImg, COLOR_YUV2BGR);
440 return true;
441 }
442
443 CV_LOG_ERROR(NULL,
444 cv::format("OpenJPEG2000: unsupported conversion from %d components to %d for YUV image decoding",
445 inChannels, outChannels));
446 return false;
447}

449OPJ_SIZE_T opjReadFromBuffer(void* dist, OPJ_SIZE_T count, detail::OpjMemoryBuffer* buffer)
450{
451 const OPJ_SIZE_T bytesToRead = std::min(buffer->availableBytes(), count);
452 if (bytesToRead > 0)
453 {
454 memcpy(dist, buffer->pos, bytesToRead);
455 buffer->pos += bytesToRead;
456 return bytesToRead;
457 }
458 else
459 {
460 return static_cast<OPJ_SIZE_T>(-1);
461 }
462}

466 if (bytesToSkip > 0)
467 {
468 buffer->pos += bytesToSkip;
469 return bytesToSkip;
470 }
471 else
472 {
473 return static_cast<OPJ_SIZE_T>(-1);
474 }

477OPJ_BOOL opjSeekFromBuffer(OPJ_OFF_T count, detail::OpjMemoryBuffer* buffer)
478{
479 // Count should stay positive to prevent unsigned overflow
480 CV_DbgAssert(count > 0);
481 // To provide proper comparison between OPJ_OFF_T and OPJ_SIZE_T, both should be
482 // casted to uint64_t (On 32-bit systems sizeof(size_t) might be 4)
483 CV_DbgAssert(static_cast<uint64_t>(count) < static_cast<uint64_t>(std::numeric_limits<OPJ_SIZE_T>::max()));
484 const OPJ_SIZE_T pos = std::min(buffer->length, static_cast<OPJ_SIZE_T>(count));
485 buffer->pos = buffer->begin + pos;
486 return OPJ_TRUE;
487}

517bool Jpeg2KOpjDecoderBase::readHeader()
518{
519 if (!m_buf.empty()) {
520 opjBuf_ = detail::OpjMemoryBuffer(m_buf);
521 stream_ = opjCreateBufferInputStream(&opjBuf_);
522 }
523 else
524 {
525 stream_.reset(opj_stream_create_default_file_stream(m_filename.c_str(), OPJ_STREAM_READ));
526 }
527 if (!stream_)
528 return false;
529
530 codec_.reset(opj_create_decompress(format_));
531 if (!codec_)
532 return false;
533
534 // Callbacks are cleared, when opj_destroy_codec is called,
535 // They can provide some additional information for the user, about what goes wrong
536 setupLogCallbacks(codec_.get());
537
538 opj_dparameters parameters = setupDecoderParameters();
539 if (!opj_setup_decoder(codec_.get(), &parameters))
540 return false;
541
542 {
543 opj_image_t* rawImage;
544 if (!opj_read_header(stream_.get(), codec_.get(), &rawImage))
545 return false;
546
547 image_.reset(rawImage);
548 }
549
550 m_width = image_->x1 - image_->x0;
551 m_height = image_->y1 - image_->y0;
552
553 /* Different components may have different precision,
554 * so check all.
555 */
556 bool hasAlpha = false;
557 const int numcomps = image_->numcomps;
558 CV_Check(numcomps, numcomps >= 1 && numcomps <= 4, "Unsupported number of components");
559 for (int i = 0; i < numcomps; i++)
560 {
561 const opj_image_comp_t& comp = image_->comps[i];
562
563 if (comp.sgnd)
564 {
565 CV_Error(Error::StsNotImplemented, cv::format("OpenJPEG2000: Component %d/%d is signed", i, numcomps));
566 }
567
568 if (hasAlpha && comp.alpha)
569 {
570 CV_Error(Error::StsNotImplemented, cv::format("OpenJPEG2000: Component %d/%d is duplicate alpha channel", i, numcomps));
571 }
572
573 hasAlpha |= comp.alpha != 0;
574
575 if (comp.prec > 64)
576 {
577 CV_Error(Error::StsNotImplemented, "OpenJPEG2000: precision > 64 is not supported");
578 }
579 m_maxPrec = std::max(m_maxPrec, comp.prec);
580 }
581
582 if (m_maxPrec < 8) {
583 CV_Error(Error::StsNotImplemented, "OpenJPEG2000: Precision < 8 not supported");
584 } else if (m_maxPrec == 8) {
585 m_type = CV_MAKETYPE(CV_8U, numcomps);
586 } else if (m_maxPrec <= 16) {
587 m_type = CV_MAKETYPE(CV_16U, numcomps);
588 } else if (m_maxPrec <= 23) {
589 m_type = CV_MAKETYPE(CV_32F, numcomps);
590 } else {
591 m_type = CV_MAKETYPE(CV_64F, numcomps);
592 }
593 return true;
594}

596bool Jpeg2KOpjDecoderBase::readData( Mat& img )
597{
598 using DecodeFunc = bool(*)(const opj_image_t&, cv::Mat&, uint8_t shift, bool use_rgb);
599
600 if (!opj_decode(codec_.get(), stream_.get(), image_.get()))
601 {
602 CV_Error(Error::StsError, "OpenJPEG2000: Decoding is failed");
603 }
604
605 if (img.channels() == 2)
606 {
607 CV_Error(Error::StsNotImplemented,
608 cv::format("OpenJPEG2000: Unsupported number of output channels. IN: %d OUT: 2", image_->numcomps));
609 }
610
611 DecodeFunc decode = nullptr;
612 switch (image_->color_space)
613 {
614 case OPJ_CLRSPC_UNKNOWN:
615 /* FALLTHRU */
616 case OPJ_CLRSPC_UNSPECIFIED:
617 CV_LOG_WARNING(NULL, "OpenJPEG2000: Image has unknown or unspecified color space, SRGB is assumed");
618 /* FALLTHRU */
619 case OPJ_CLRSPC_SRGB:
620 decode = decodeSRGBData;
621 break;
622 case OPJ_CLRSPC_GRAY:
623 decode = decodeGrayscaleData;
624 break;
625 case OPJ_CLRSPC_SYCC:
626 decode = decodeSYCCData;
627 break;
628 default:
629 CV_Error(Error::StsNotImplemented,
630 cv::format("OpenJPEG2000: Unsupported color space conversion: %s -> %s",
631 colorspaceName(image_->color_space).c_str(),
632 (img.channels() == 1) ? "gray" : "BGR"));
633 }
634
635 const int depth = img.depth();
636 const OPJ_UINT32 outPrec = [depth]() {
637 if (depth == CV_8U) return 8;
638 if (depth == CV_16U) return 16;
639 CV_Error(Error::StsNotImplemented,
640 cv::format("OpenJPEG2000: output precision > 16 not supported: target depth %d", depth));
641 }();
642 const uint8_t shift = outPrec > m_maxPrec ? 0 : (uint8_t)(m_maxPrec - outPrec); // prec <= 64
643
644 const int inChannels = image_->numcomps;
645
646 CV_Assert(inChannels > 0);
647 CV_Assert(image_->comps);
648 for (int c = 0; c < inChannels; c++)
649 {
650 const opj_image_comp_t& comp = image_->comps[c];
651 CV_CheckEQ((int)comp.dx, 1, "OpenJPEG2000: tiles are not supported");
652 CV_CheckEQ((int)comp.dy, 1, "OpenJPEG2000: tiles are not supported");
653 CV_CheckEQ((int)comp.x0, 0, "OpenJPEG2000: tiles are not supported");
654 CV_CheckEQ((int)comp.y0, 0, "OpenJPEG2000: tiles are not supported");
655 CV_CheckEQ((int)comp.w, img.cols, "OpenJPEG2000: tiles are not supported");
656 CV_CheckEQ((int)comp.h, img.rows, "OpenJPEG2000: tiles are not supported");
657 CV_Assert(comp.data && "OpenJPEG2000: missing component data (unsupported / broken input)");
658 }
659
660 return decode(*image_, img, shift, m_use_rgb);
661}

707bool Jpeg2KOpjEncoder::write(const Mat& img, const std::vector<int>& params)
708{
709 CV_Assert(params.size() % 2 == 0);
710
711 const int channels = img.channels();
712 CV_DbgAssert(channels > 0); // passed matrix is not empty
713 if (channels > 4)
714 {
715 CV_Error(Error::StsNotImplemented, "OpenJPEG2000: only BGR(a) and gray (+ alpha) images supported");
716 }
717
718 const int depth = img.depth();
719
720 const OPJ_UINT32 outPrec = [depth]() {
721 if (depth == CV_8U) return 8;
722 if (depth == CV_16U) return 16;
723 CV_Error(Error::StsNotImplemented,
724 cv::format("OpenJPEG2000: image precision > 16 not supported. Got: %d", depth));
725 }();
726
727 opj_cparameters parameters = setupEncoderParameters(params);
728
729 std::vector<opj_image_cmptparm_t> compparams(channels);
730 for (int i = 0; i < channels; i++) {
731 compparams[i].prec = outPrec;
732 compparams[i].bpp = outPrec;
733 compparams[i].sgnd = 0; // unsigned for now
734 compparams[i].dx = parameters.subsampling_dx;
735 compparams[i].dy = parameters.subsampling_dy;
736 compparams[i].w = img.size().width;
737 compparams[i].h = img.size().height;
738 }
