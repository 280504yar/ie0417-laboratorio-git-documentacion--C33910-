25 FStructData startWriteStruct(const FStructData& parent, const char* key,
26 int struct_flags, const char* type_name=0)
27 {
28 char buf[CV_FS_MAX_LEN + 1024];
29 const char* data = 0;
30
31 if ( type_name && *type_name == '\0' )
32 type_name = 0;
33
34 struct_flags = (struct_flags & (FileNode::TYPE_MASK|FileNode::FLOW)) | FileNode::EMPTY;
35 if( !FileNode::isCollection(struct_flags))
36 CV_Error( cv::Error::StsBadArg,
37 "Some collection type - FileNode::SEQ or FileNode::MAP, must be specified" );
38
39 if (type_name && memcmp(type_name, "binary", 6) == 0)
40 {
41 /* reset struct flag. in order not to print ']' */
42 struct_flags = FileNode::SEQ;
43 snprintf(buf, sizeof(buf), "!!binary |");
44 data = buf;
45 }
46 else if( FileNode::isFlow(struct_flags))
47 {
48 char c = FileNode::isMap(struct_flags) ? '{' : '[';
49 struct_flags |= FileNode::FLOW;
50
51 if( type_name )
52 snprintf( buf, sizeof(buf), "!!%s %c", type_name, c );
53 else
54 {
55 buf[0] = c;
56 buf[1] = '\0';
57 }
58 data = buf;
59 }
60 else if( type_name )
61 {
62 snprintf( buf, sizeof(buf), "!!%s", type_name );
63 data = buf;
64 }
65
66 writeScalar( key, data );
67
68 FStructData fsd;
69 fsd.indent = parent.indent;
70 fsd.flags = struct_flags;
71
72 if( !FileNode::isFlow(parent.flags) )
73 fsd.indent += CV_YML_INDENT + FileNode::isFlow(struct_flags);
74
75 return fsd;
76 }

122 void write(const char* key, const char* str, bool quote)
123 {
124 char buf[CV_FS_MAX_LEN*4+16];
125 char* data = (char*)str;
126 int i, len;
127
128 if( !str )
129 CV_Error( cv::Error::StsNullPtr, "Null string pointer" );
130
131 len = (int)strlen(str);
132 if( len > CV_FS_MAX_LEN )
133 CV_Error( cv::Error::StsBadArg, "The written string is too long" );
134
135 if( quote || len == 0 || str[0] != str[len-1] || (str[0] != '\"' && str[0] != '\'') )
136 {
137 int need_quote = quote || len == 0 || str[0] == ' ';
138 data = buf;
139 *data++ = '\"';
140 for( i = 0; i < len; i++ )
141 {
142 char c = str[i];
143
144 if( !need_quote && !cv_isalnum(c) && c != '_' && c != ' ' && c != '-' &&
145 c != '(' && c != ')' && c != '/' && c != '+' && c != ';' )
146 need_quote = 1;
147
148 if( !cv_isalnum(c) && (!cv_isprint(c) || c == '\\' || c == '\'' || c == '\"') )
149 {
150 *data++ = '\\';
151 if( cv_isprint(c) )
152 *data++ = c;
153 else if( c == '\n' )
154 *data++ = 'n';
155 else if( c == '\r' )
156 *data++ = 'r';
157 else if( c == '\t' )
158 *data++ = 't';
159 else
160 {
161 snprintf( data, sizeof(buf) - (data - buf), "x%02x", c );
162 data += 3;
163 }
164 }
165 else
166 *data++ = c;
167 }
168 if( !need_quote && (cv_isdigit(str[0]) ||
169 str[0] == '+' || str[0] == '-' || str[0] == '.' ))
170 need_quote = 1;
171
172 if( need_quote )
173 *data++ = '\"';
174 *data++ = '\0';
175 data = buf + !need_quote;
176 }
177
178 writeScalar( key, data);
179 }

181 void writeScalar(const char* key, const char* data)
182 {
183 fs->check_if_write_struct_is_delayed(false);
184 if ( fs->get_state_of_writing_base64() == FileStorage_API::Uncertain )
185 {
186 fs->switch_to_Base64_state( FileStorage_API::NotUse );
187 }
188 else if ( fs->get_state_of_writing_base64() == FileStorage_API::InUse )
189 {
190 CV_Error( cv::Error::StsError, "At present, output Base64 data only." );
191 }
192
193 int i, keylen = 0;
194 int datalen = 0;
195 char* ptr;
196
197 FStructData& current_struct = fs->getCurrentStruct();
198
199 int struct_flags = current_struct.flags;
200
201 if( key && key[0] == '\0' )
202 key = 0;
203
204 if( FileNode::isCollection(struct_flags) )
205 {
206 if( (FileNode::isMap(struct_flags) ^ (key != 0)) )
207 CV_Error( cv::Error::StsBadArg, "An attempt to add element without a key to a map, "
208 "or add element with key to sequence" );
209 }
210 else
211 {
212 fs->setNonEmpty();
213 struct_flags = FileNode::EMPTY | (key ? FileNode::MAP : FileNode::SEQ);
214 }
215
216 if( key )
217 {
218 keylen = (int)strlen(key);
219 if( keylen == 0 )
220 CV_Error( cv::Error::StsBadArg, "The key is an empty" );
221
222 if( keylen > CV_FS_MAX_LEN )
223 CV_Error( cv::Error::StsBadArg, "The key is too long" );
224 }
225
226 if( data )
227 datalen = (int)strlen(data);
228
229 if( FileNode::isFlow(struct_flags) )
230 {
231 ptr = fs->bufferPtr();
232 if( !FileNode::isEmptyCollection(struct_flags) )
233 *ptr++ = ',';
234 int new_offset = (int)(ptr - fs->bufferStart()) + keylen + datalen;
235 if( new_offset > fs->wrapMargin() && new_offset - current_struct.indent > 10 )
236 {
237 fs->setBufferPtr(ptr);
238 ptr = fs->flush();
239 }
240 else
241 *ptr++ = ' ';
242 }
243 else
244 {
245 ptr = fs->flush();
246 if( !FileNode::isMap(struct_flags) )
247 {
248 *ptr++ = '-';
249 if( data )
250 *ptr++ = ' ';
251 }
252 }
253
254 if( key )
255 {
256 if( !cv_isalpha(key[0]) && key[0] != '_' )
257 CV_Error( cv::Error::StsBadArg, "Key must start with a letter or _" );
258
259 ptr = fs->resizeWriteBuffer( ptr, keylen );
260
261 for( i = 0; i < keylen; i++ )
262 {
263 char c = key[i];
264
265 ptr[i] = c;
266 if( !cv_isalnum(c) && c != '-' && c != '_' && c != ' ' )
267 CV_Error( cv::Error::StsBadArg, "Key names may only contain alphanumeric characters [a-zA-Z0-9], '-', '_' and ' '" );
268 }
269
270 ptr += keylen;
271 *ptr++ = ':';
272 if( !FileNode::isFlow(struct_flags) && data )
273 *ptr++ = ' ';
274 }
275
276 if( data )
277 {
278 ptr = fs->resizeWriteBuffer( ptr, datalen );
279 memcpy( ptr, data, datalen );
280 ptr += datalen;
281 }
282
283 fs->setBufferPtr(ptr);
284 current_struct.flags &= ~FileNode::EMPTY;
285 }

287 void writeComment(const char* comment, bool eol_comment)
288 {
289 if( !comment )
290 CV_Error( cv::Error::StsNullPtr, "Null comment" );
291
292 int len = (int)strlen(comment);
293 const char* eol = strchr(comment, '\n');
294 bool multiline = eol != 0;
295 char* ptr = fs->bufferPtr();
296
297 if( !eol_comment || multiline ||
298 fs->bufferEnd() - ptr < len || ptr == fs->bufferStart() )
299 ptr = fs->flush();
300 else
301 *ptr++ = ' ';
302
303 while( comment )
304 {
305 *ptr++ = '#';
306 *ptr++ = ' ';
307 if( eol )
308 {
309 ptr = fs->resizeWriteBuffer( ptr, (int)(eol - comment) + 1 );
310 memcpy( ptr, comment, eol - comment + 1 );
311 fs->setBufferPtr(ptr + (eol - comment));
312 comment = eol + 1;
313 eol = strchr( comment, '\n' );
314 }
315 else
316 {
317 len = (int)strlen(comment);
318 ptr = fs->resizeWriteBuffer( ptr, len );
319 memcpy( ptr, comment, len );
320 fs->setBufferPtr(ptr + len);
321 comment = 0;
322 }
323 ptr = fs->flush();
324 }

347 char* skipSpaces( char* ptr, int min_indent, int max_comment_indent )
348 {
349 if (!ptr)
350 CV_PARSE_ERROR_CPP("Invalid input");
351
352 for(;;)
353 {
354 while( *ptr == ' ' )
355 ptr++;
356 if( *ptr == '#' )
357 {
358 if( ptr - fs->bufferStart() > max_comment_indent )
359 return ptr;
360 *ptr = '\0';
361 }
362 else if( cv_isprint(*ptr) )
363 {
364 if( ptr - fs->bufferStart() < min_indent )
365 CV_PARSE_ERROR_CPP( "Incorrect indentation" );
366 break;
367 }
368 else if( *ptr == '\0' || *ptr == '\n' || *ptr == '\r' )
369 {
370 ptr = fs->gets();
371 if( !ptr )
372 {
373 // emulate end of stream
374 ptr = fs->bufferStart();
375 ptr[0] = ptr[1] = ptr[2] = '.';
376 ptr[3] = '\0';
377 fs->setEof();
378 break;
379 }
380 else
381 {
382 int l = (int)strlen(ptr);
383 if( ptr[l-1] != '\n' && ptr[l-1] != '\r' && !fs->eof() )
384 CV_PARSE_ERROR_CPP( "Too long string or a last string w/o newline" );
385 }
386 }
387 else
388 CV_PARSE_ERROR_CPP( *ptr == '\t' ? "Tabs are prohibited in YAML!" : "Invalid character" );
389 }
390
391 return ptr;
392 }

413 return true;

417 char* parseKey( char* ptr, FileNode& map_node, FileNode& value_placeholder )
418 {
419 if (!ptr)
420 CV_PARSE_ERROR_CPP("Invalid input");
421
422 char c;
423 char *endptr = ptr - 1, *saveptr;
424
425 if( *ptr == '-' )
426 CV_PARSE_ERROR_CPP( "Key may not start with \'-\'" );
427
428 do c = *++endptr;
429 while( cv_isprint(c) && c != ':' );
430
431 if( c != ':' )
432 CV_PARSE_ERROR_CPP( "Missing \':\'" );
433
434 saveptr = endptr + 1;
435 if( endptr == ptr )
436 CV_PARSE_ERROR_CPP( "An empty key" );
437
438 do c = *--endptr;
439 while( c == ' ' );
440
441 ++endptr;
442
443 value_placeholder = fs->addNode(map_node, std::string(ptr, endptr - ptr), FileNode::NONE);
444 ptr = saveptr;
445
446 return ptr;

449 char* parseValue( char* ptr, FileNode& node, int min_indent, bool is_parent_flow )
450 {
451 if (!ptr)
452 CV_PARSE_ERROR_CPP("Invalid input");
453
454 char* endptr = 0;
455 char c = ptr[0], d = ptr[1];
456 int value_type = FileNode::NONE;
457 int len;
458 bool is_binary_string = false;
459 bool is_user = false;
460
461 if( c == '!' ) // handle explicit type specification
462 {
463 if( d == '!' || d == '^' )
464 {
465 ptr++;
466 is_user = true;
467 //value_type |= FileNode::USER;
468 }
469 if ( d == '<') //support of full type heading from YAML 1.2
470 {
471 const char* yamlTypeHeading = "<tag:yaml.org,2002:";
472 const size_t headingLength = strlen(yamlTypeHeading);
473
474 char* typeEndPtr = ++ptr;
475
476 do d = *++typeEndPtr;
477 while( cv_isprint(d) && d != ' ' && d != '>' );
478
479 if ( d == '>' && (size_t)(typeEndPtr - ptr) > headingLength )
480 {
481 if ( memcmp(ptr, yamlTypeHeading, headingLength) == 0 )
482 {
483 *typeEndPtr = ' ';
484 ptr += headingLength - 1;
485 is_user = true;
486 //value_type |= FileNode::USER;
487 }
488 }
489 }
490
491 endptr = ptr++;
492 do d = *++endptr;
493 while( cv_isprint(d) && d != ' ' );
494 len = (int)(endptr - ptr);
495 if( len == 0 )
496 CV_PARSE_ERROR_CPP( "Empty type name" );
497 d = *endptr;
498 *endptr = '\0';
499
500 if( len == 3 && !is_user )
501 {
502 if( memcmp( ptr, "str", 3 ) == 0 )
503 value_type = FileNode::STRING;
504 else if( memcmp( ptr, "int", 3 ) == 0 )
505 value_type = FileNode::INT;
506 else if( memcmp( ptr, "seq", 3 ) == 0 )
507 value_type = FileNode::SEQ;
508 else if( memcmp( ptr, "map", 3 ) == 0 )
509 value_type = FileNode::MAP;
510 }
511 else if( len == 5 && !is_user )
512 {
513 if( memcmp( ptr, "float", 5 ) == 0 )
514 value_type = FileNode::REAL;
515 }
516 else if (len == 6 && is_user)
517 {
518 if( memcmp( ptr, "binary", 6 ) == 0 ) {
519 value_type = FileNode::SEQ;
520 is_binary_string = true;
521
522 /* for ignore '|' */
523
524 /**** operation with endptr ****/
525 *endptr = d;
526
527 do {
528 d = *++endptr;
529 if (d == '|')
530 break;
531 } while (d == ' ');
532
533 d = *++endptr;
534 *endptr = '\0';
535 }
536 }
537
538 *endptr = d;
539 ptr = skipSpaces( endptr, min_indent, INT_MAX );
540 if (!ptr)
541 CV_PARSE_ERROR_CPP("Invalid input");
542
543 c = *ptr;
544
545 if( !is_user )
546 {
547 if (value_type == FileNode::STRING && c != '\'' && c != '\"')
548 goto force_string;
549 if( value_type == FileNode::INT )
550 goto force_int;
551 if( value_type == FileNode::REAL )
552 goto force_real;
553 }
554 }
555
556 if (is_binary_string)
557 {
558 int indent = static_cast<int>(ptr - fs->bufferStart());
559 ptr = fs->parseBase64(ptr, indent, node);
560 }
561 else if( cv_isdigit(c) ||
562 ((c == '-' || c == '+') && (cv_isdigit(d) || d == '.')) ||
563 (c == '.' && cv_isalnum(d))) // a number
564 {
565 endptr = ptr + (c == '-' || c == '+');
566 while( cv_isdigit(*endptr) )
567 endptr++;
568 if( *endptr == '.' || *endptr == 'e' )
569 {
570 force_real:
571 double fval = fs->strtod( ptr, &endptr );
572 node.setValue(FileNode::REAL, &fval);
573 }
574 else
575 {
576 force_int:
577 int64_t ival = strtoll( ptr, &endptr, 0 );
578 node.setValue(FileNode::INT, &ival);
579 }
580
581 if( !endptr || endptr == ptr )
582 CV_PARSE_ERROR_CPP( "Invalid numeric value (inconsistent explicit type specification?)" );
583
584 ptr = endptr;
585 CV_PERSISTENCE_CHECK_END_OF_BUFFER_BUG_CPP();
586 }
587 else if( c == '\'' || c == '\"' ) // an explicit string
588 {
589 if( c == '\'' )
590 for( len = 0; len < CV_FS_MAX_LEN; )
591 {
592 c = *++ptr;
593 if( cv_isalnum(c) || (c != '\'' && cv_isprint(c)))
594 buf[len++] = c;
595 else if( c == '\'' )
596 {
597 c = *++ptr;
598 if( c != '\'' )
599 break;
600 buf[len++] = c;
601 }
602 else
603 CV_PARSE_ERROR_CPP( "Invalid character" );
604 }
605 else
606 for( len = 0; len < CV_FS_MAX_LEN; )
607 {
608 c = *++ptr;
609 if( cv_isalnum(c) || (c != '\\' && c != '\"' && cv_isprint(c)))
610 buf[len++] = c;
611 else if( c == '\"' )
612 {
613 ++ptr;
614 break;
615 }
616 else if( c == '\\' )
617 {
618 d = *++ptr;
619 if( d == '\'' )
620 buf[len++] = d;
621 else if( d == '\"' || d == '\\' || d == '\'' )
622 buf[len++] = d;
623 else if( d == 'n' )
624 buf[len++] = '\n';
625 else if( d == 'r' )
626 buf[len++] = '\r';
627 else if( d == 't' )
628 buf[len++] = '\t';
629 else if( d == 'x' || (cv_isdigit(d) && d < '8') )
630 {
631 int val, is_hex = d == 'x';
632 c = ptr[3];
633 ptr[3] = '\0';
634 val = (int)strtol( ptr + is_hex, &endptr, is_hex ? 8 : 16 );
635 ptr[3] = c;
636 if( endptr == ptr + is_hex )
637 buf[len++] = 'x';
638 else
639 {
640 buf[len++] = (char)val;
641 ptr = endptr;
642 }
643 }
644 }
645 else
646 CV_PARSE_ERROR_CPP( "Invalid character" );
647 }
648
649 if( len >= CV_FS_MAX_LEN )
650 CV_PARSE_ERROR_CPP( "Too long string literal" );
651
652 node.setValue(FileNode::STRING, buf, len);
653 }
654 else if( c == '[' || c == '{' ) // collection as a flow
655 {
656 int new_min_indent = min_indent + !is_parent_flow;
657 int struct_type = c == '{' ? FileNode::MAP : FileNode::SEQ;
658 int nelems = 0;
659
660 fs->convertToCollection(struct_type, node);
661 d = c == '[' ? ']' : '}';
662
663 for( ++ptr ;; nelems++ )
664 {
665 FileNode elem;
666
667 ptr = skipSpaces( ptr, new_min_indent, INT_MAX );
668 if (!ptr)
669 CV_PARSE_ERROR_CPP("Invalid input");
670 if( *ptr == '}' || *ptr == ']' )
671 {
672 if( *ptr != d )
673 CV_PARSE_ERROR_CPP( "The wrong closing bracket" );
674 ptr++;
675 break;
676 }
677
678 if( nelems != 0 )
679 {
680 if( *ptr != ',' )
681 CV_PARSE_ERROR_CPP( "Missing , between the elements" );
682 ptr = skipSpaces( ptr + 1, new_min_indent, INT_MAX );
683 if (!ptr)
684 CV_PARSE_ERROR_CPP("Invalid input");
685 }
686
687 if( struct_type == FileNode::MAP )
688 {
689 ptr = parseKey( ptr, node, elem );
690 ptr = skipSpaces( ptr, new_min_indent, INT_MAX );
691 }
692 else
693 {
694 if( *ptr == ']' )
695 break;
696 elem = fs->addNode(node, std::string(), FileNode::NONE);
697 }
698 ptr = parseValue( ptr, elem, new_min_indent, true );
699 }
700 fs->finalizeCollection(node);
701 }
702 else
703 {
704 int indent, struct_type;
705
706 if( is_parent_flow || c != '-' )
707 {
708 // implicit (one-line) string or nested block-style collection
709 if( !is_parent_flow )
710 {
711 if( c == '?' )
712 CV_PARSE_ERROR_CPP( "Complex keys are not supported" );
713 if( c == '|' || c == '>' )
714 CV_PARSE_ERROR_CPP( "Multi-line text literals are not supported" );
715 }
716
717 force_string:
718 endptr = ptr - 1;
719
720 do c = *++endptr;
721 while( cv_isprint(c) &&
722 (!is_parent_flow || (c != ',' && c != '}' && c != ']')) &&
723 (is_parent_flow || c != ':' || value_type == FileNode::STRING));
724
725 if( endptr == ptr )
726 CV_PARSE_ERROR_CPP( "Invalid character" );
727
728 if( is_parent_flow || c != ':' )
729 {
730 char* str_end = endptr;
731 do c = *--str_end;
732 while( str_end > ptr && c == ' ' );
733 str_end++;
734 if ((fs->getFormat() & FileStorage::FORMAT_MASK) != FileStorage::FORMAT_YAML_1_0)
735 {
736 size_t llen = str_end - ptr;
737 if (llen == 4 && memcmp(ptr, "true", 4) == 0) {
738 int64_t ival = 1;
739 node.setValue(FileNode::INT, &ival);
740 ptr = endptr;
741 return ptr;
742 }
743 else if (llen == 5 && memcmp(ptr, "false", 5) == 0) {
744 int64_t ival = 0;
745 node.setValue(FileNode::INT, &ival);
746 ptr = endptr;
747 return ptr;
748 }
749 else if (llen == 4 && memcmp(ptr, "null", 4) == 0) {
750 ptr = endptr;
751 return ptr;
752 }
753 }
754
755 node.setValue(FileNode::STRING, ptr, (int)(str_end - ptr));
756 ptr = endptr;
757 return ptr;
758 }
759 struct_type = FileNode::MAP;
760 }
761 else
762 struct_type = FileNode::SEQ;
763
764 fs->convertToCollection( struct_type, node );
765 indent = (int)(ptr - fs->bufferStart());
766
767 for(;;)
768 {
769 FileNode elem;
770
771 if( struct_type == FileNode::MAP )
772 {
773 ptr = parseKey( ptr, node, elem );
774 }
775 else
776 {
777 c = *ptr++;
778 if( c != '-' )
779 CV_PARSE_ERROR_CPP( "Block sequence elements must be preceded with \'-\'" );
780
781 elem = fs->addNode(node, std::string(), FileNode::NONE);
782 }
783 ptr = skipSpaces( ptr, indent + 1, INT_MAX );

802 bool parse( char* ptr )
803 {
804 if (!ptr)
805 CV_PARSE_ERROR_CPP("Invalid input");
806
807 bool first = true;
808 bool ok = true;
809 FileNode root_collection(fs->getFS(), 0, 0);
810
811 for(;;)
812 {
813 // 0. skip leading comments and directives and ...
814 // 1. reach the first item
815 for(;;)
816 {
817 ptr = skipSpaces( ptr, 0, INT_MAX );
818 if( !ptr || *ptr == '\0' )
819 {
820 ok = !first;
821 break;
822 }
823
824 if( *ptr == '%' )
825 {
826 if( memcmp( ptr, "%YAML", 5 ) == 0 &&
827 memcmp( ptr, "%YAML:1.", 8 ) != 0 &&
828 memcmp( ptr, "%YAML 1.", 8 ) != 0)
829 {
830 if ((fs->getFormat() & FileStorage::FORMAT_MASK) == FileStorage::FORMAT_YAML_1_0)
831 CV_PARSE_ERROR_CPP( "Unsupported YAML version (it must be 1.x)" );
832 }
833 *ptr = '\0';
834 }
835 else if( *ptr == '-' )
836 {
837 if( memcmp(ptr, "---", 3) == 0 )
838 {
839 ptr += 3;
840 break;
841 }
842 else if( first )
843 break;
844 }
845 else if( cv_isalnum(*ptr) || *ptr=='_')
846 {
847 if( !first )
848 CV_PARSE_ERROR_CPP( "The YAML streams must start with '---', except the first one" );
849 break;
850 }
851 else if( fs->eof() )
852 break;
853 else
854 CV_PARSE_ERROR_CPP( "Invalid or unsupported syntax" );
855 }
856
857 if( ptr )
858 ptr = skipSpaces( ptr, 0, INT_MAX );
859 if( !ptr || !ptr[0] )
860 break;
861 if( memcmp( ptr, "...", 3 ) != 0 )
862 {
863 // 2. parse the collection
864 FileNode root_node = fs->addNode(root_collection, std::string(), FileNode::NONE);
865
866 ptr = parseValue( ptr, root_node, 0, false );
867 if( !root_node.isMap() && !root_node.isSeq() )
868 CV_PARSE_ERROR_CPP( "Only collections as YAML streams are supported by this parser" );
869
870 // 3. parse until the end of file or next collection
871 ptr = skipSpaces( ptr, 0, INT_MAX );
872 if( !ptr )
873 break;
874 }
875
876 if( fs->eof() )
877 break;
878 ptr += 3;
879 first = false;
880 }
881
882 return ok;
883 }
