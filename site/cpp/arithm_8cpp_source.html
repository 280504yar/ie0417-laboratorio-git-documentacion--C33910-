81static bool ocl_binary_op(InputArray _src1, InputArray _src2, OutputArray _dst,
82 InputArray _mask, bool bitwise, int oclop, bool haveScalar )
83{
84 bool haveMask = !_mask.empty();
85 int srctype = _src1.type();
86 int srcdepth = CV_MAT_DEPTH(srctype);
87 int cn = CV_MAT_CN(srctype);
88
89 const ocl::Device d = ocl::Device::getDefault();
90 bool doubleSupport = d.doubleFPConfig() > 0;
91 if( oclop < 0 || ((haveMask || haveScalar) && cn > 4) ||
92 (!doubleSupport && srcdepth == CV_64F && !bitwise))
93 return false;
94
95 char opts[1024];
96 int kercn = haveMask || haveScalar ? cn : ocl::predictOptimalVectorWidth(_src1, _src2, _dst);
97 int scalarcn = kercn == 3 ? 4 : kercn;
98 int rowsPerWI = d.isIntel() ? 4 : 1;
99
100 const int dstDepth = srcdepth;
101 const int dstType = CV_MAKETYPE(dstDepth, kercn);
102 const int dstType1 = CV_MAKETYPE(dstDepth, 1);
103 const int scalarType = CV_MAKETYPE(srcdepth, scalarcn);
104
105 snprintf(opts, sizeof(opts), "-D %s%s -D %s%s -D dstT=%s -D DEPTH_dst=%d -D dstT_C1=%s -D workST=%s -D cn=%d -D rowsPerWI=%d",
106 haveMask ? "MASK_" : "", haveScalar ? "UNARY_OP" : "BINARY_OP", oclop2str[oclop],
107 doubleSupport ? " -D DOUBLE_SUPPORT" : "",
108 bitwise ? ocl::memopTypeToStr(dstType) : ocl::typeToStr(dstType),
109 dstDepth,
110 bitwise ? ocl::memopTypeToStr(dstType1) : ocl::typeToStr(dstType1),
111 bitwise ? ocl::memopTypeToStr(scalarType) : ocl::typeToStr(scalarType),
112 kercn, rowsPerWI);
113
114 ocl::Kernel k("KF", ocl::core::arithm_oclsrc, opts);
115 if (k.empty())
116 return false;
117
118 UMat src1 = _src1.getUMat(), src2;
119 UMat dst = _dst.getUMat(), mask = _mask.getUMat();
120
121 ocl::KernelArg src1arg = ocl::KernelArg::ReadOnlyNoSize(src1, cn, kercn);
122 ocl::KernelArg dstarg = haveMask ? ocl::KernelArg::ReadWrite(dst, cn, kercn) :
123 ocl::KernelArg::WriteOnly(dst, cn, kercn);
124 ocl::KernelArg maskarg = ocl::KernelArg::ReadOnlyNoSize(mask, 1);
125
126 if( haveScalar )
127 {
128 size_t esz = CV_ELEM_SIZE1(srctype)*scalarcn;
129 double buf[4] = {0,0,0,0};
130
131 if( oclop != OCL_OP_NOT )
132 {
133 Mat src2sc = _src2.getMat();
134 convertAndUnrollScalar(src2sc, srctype, (uchar*)buf, 1);
135 }
136
137 ocl::KernelArg scalararg = ocl::KernelArg(ocl::KernelArg::CONSTANT, 0, 0, 0, buf, esz);
138
139 if( !haveMask )
140 k.args(src1arg, dstarg, scalararg);
141 else
142 k.args(src1arg, maskarg, dstarg, scalararg);
143 }
144 else
145 {
146 src2 = _src2.getUMat();
147 ocl::KernelArg src2arg = ocl::KernelArg::ReadOnlyNoSize(src2, cn, kercn);
148
149 if( !haveMask )
150 k.args(src1arg, src2arg, dstarg);
151 else
152 k.args(src1arg, src2arg, maskarg, dstarg);
153 }
154
155 size_t globalsize[] = { (size_t)src1.cols * cn / kercn, ((size_t)src1.rows + rowsPerWI - 1) / rowsPerWI };
156 return k.run(2, globalsize, 0, false);
157}

161// OpenCL path for the bitwise ops (and/or/xor/not), lifted out of binary_op: choose the (possibly
162// scalar) operand order, size the destination, and dispatch the shared bitwise OpenCL kernel. Returns
163// true if OpenCL handled the op, false to fall through to the CPU engine. NOT arrives unary
164// (src2 == src1, forced-scalar). The caller gates this on a UMat operand + dims <= 2 (via CV_OCL_RUN).
165#ifdef HAVE_OPENCL
166static bool bitwise_op_ocl(InputArray _src1, InputArray _src2, OutputArray _dst,
167 InputArray _mask, int oclop)
168{
169 const _InputArray *psrc1 = &_src1, *psrc2 = &_src2;
170 _InputArray::KindFlag kind1 = psrc1->kind(), kind2 = psrc2->kind();
171 int type1 = psrc1->type(), type2 = psrc2->type();
172 bool haveScalar = false;
173
174 if( oclop == OCL_OP_NOT )
175 haveScalar = true;
176 else if( (kind1 == _InputArray::MATX) + (kind2 == _InputArray::MATX) == 1 ||
177 !psrc1->sameSize(*psrc2) || type1 != type2 )
178 {
179 if( checkScalar(*psrc1, type2, kind1, kind2) )
180 { std::swap(psrc1, psrc2); std::swap(type1, type2); } // src1 is the scalar; swap it out
181 else if( !checkScalar(*psrc2, type1, kind2, kind1) )
182 return false;
183 haveScalar = true;
184 }
185 _dst.createSameSize(*psrc1, type1);
186 return ocl_binary_op(*psrc1, *psrc2, _dst, _mask, true, oclop, haveScalar);
187}
188#endif // HAVE_OPENCL
189
190}

286// used by the OpenCL branch only (arithm_op_ocl working-type selection)
287static int actualScalarDepth(const double* data, int len)
288{
289 int i = 0, minval = INT_MAX, maxval = INT_MIN;
290 for(; i < len; ++i)
291 {
292 int ival = cvRound(data[i]);
293 if( ival != data[i] )
294 break;
295 minval = MIN(minval, ival);
296 maxval = MAX(maxval, ival);
297 }
298 return i < len ? CV_64F :
299 minval >= 0 && maxval <= (int)UCHAR_MAX ? CV_8U :
300 minval >= (int)SCHAR_MIN && maxval <= (int)SCHAR_MAX ? CV_8S :
301 minval >= 0 && maxval <= (int)USHRT_MAX ? CV_16U :
302 minval >= (int)SHRT_MIN && maxval <= (int)SHRT_MAX ? CV_16S :
303 CV_32S;
304}

306static int coerceTypes(int depth1, int depth2, bool muldiv)
307{
308 return depth1 == depth2 ? depth1 :
309 ((depth1 <= CV_32S) & (depth2 <= CV_32S)) != 0 ?
310 (((int)!muldiv & (depth1 <= CV_8S) & (depth2 <= CV_8S)) != 0 ? CV_16S : CV_32S) :
311 ((CV_ELEM_SIZE1(depth1) > 4) | (CV_ELEM_SIZE1(depth2) > 4)) != 0 ? CV_64F : CV_32F;
312}

314static bool ocl_arithm_op(InputArray _src1, InputArray _src2, OutputArray _dst,
315 InputArray _mask, int wtype,
316 void* usrdata, int oclop,
317 bool haveScalar )
318{
319 const ocl::Device d = ocl::Device::getDefault();
320 bool doubleSupport = d.doubleFPConfig() > 0;
321 int type1 = _src1.type(), depth1 = CV_MAT_DEPTH(type1), cn = CV_MAT_CN(type1);
322 bool haveMask = !_mask.empty();
323
324 if ( (haveMask || haveScalar) && cn > 4 )
325 return false;
326
327#ifdef __APPLE__
328 if (depth1 == CV_16U && (oclop == OCL_OP_MUL || oclop == OCL_OP_MUL_SCALE))
329 return false;
330#endif
331
332 int dtype = _dst.type(), ddepth = CV_MAT_DEPTH(dtype), wdepth = std::max(CV_32S, CV_MAT_DEPTH(wtype));
333 if (!doubleSupport)
334 wdepth = std::min(wdepth, CV_32F);
335
336 wtype = CV_MAKETYPE(wdepth, cn);
337 int type2 = haveScalar ? wtype : _src2.type(), depth2 = CV_MAT_DEPTH(type2);
338 if (!doubleSupport && (depth2 == CV_64F || depth1 == CV_64F))
339 return false;
340
341 int kercn = haveMask || haveScalar ? cn : ocl::predictOptimalVectorWidth(_src1, _src2, _dst);
342 int scalarcn = kercn == 3 ? 4 : kercn, rowsPerWI = d.isIntel() ? 4 : 1;
343
344 char cvtstr[4][50], opts[1024];
345 snprintf(opts, sizeof(opts), "-D %s%s -D %s -D srcT1=%s -D srcT1_C1=%s -D srcT2=%s -D srcT2_C1=%s "
346 "-D dstT=%s -D DEPTH_dst=%d -D dstT_C1=%s -D workT=%s -D workST=%s -D scaleT=%s -D wdepth=%d -D convertToWT1=%s "
347 "-D convertToWT2=%s -D convertToDT=%s%s -D cn=%d -D rowsPerWI=%d -D convertFromU=%s",
348 (haveMask ? "MASK_" : ""), (haveScalar ? "UNARY_OP" : "BINARY_OP"),
349 oclop2str[oclop], ocl::typeToStr(CV_MAKETYPE(depth1, kercn)),
350 ocl::typeToStr(depth1), ocl::typeToStr(CV_MAKETYPE(depth2, kercn)),
351 ocl::typeToStr(depth2), ocl::typeToStr(CV_MAKETYPE(ddepth, kercn)), ddepth,
352 ocl::typeToStr(ddepth), ocl::typeToStr(CV_MAKETYPE(wdepth, kercn)),
353 ocl::typeToStr(CV_MAKETYPE(wdepth, scalarcn)),
354 ocl::typeToStr(wdepth), wdepth,
355 ocl::convertTypeStr(depth1, wdepth, kercn, cvtstr[0], sizeof(cvtstr[0])),
356 ocl::convertTypeStr(depth2, wdepth, kercn, cvtstr[1], sizeof(cvtstr[1])),
357 ocl::convertTypeStr(wdepth, ddepth, kercn, cvtstr[2], sizeof(cvtstr[2])),
358 doubleSupport ? " -D DOUBLE_SUPPORT" : "", kercn, rowsPerWI,
359 oclop == OCL_OP_ABSDIFF && wdepth == CV_32S && ddepth == wdepth ?
360 ocl::convertTypeStr(CV_8U, ddepth, kercn, cvtstr[3], sizeof(cvtstr[3])) : "noconvert");
361
362 size_t usrdata_esz = CV_ELEM_SIZE(wdepth);
363 const uchar* usrdata_p = (const uchar*)usrdata;
364 const double* usrdata_d = (const double*)usrdata;
365 float usrdata_f[3];
366 int i, n = oclop == OCL_OP_MUL_SCALE || oclop == OCL_OP_DIV_SCALE ||
367 oclop == OCL_OP_RDIV_SCALE || oclop == OCL_OP_RECIP_SCALE ? 1 : oclop == OCL_OP_ADDW ? 3 : 0;
368 if( usrdata && n > 0 && wdepth == CV_32F )
369 {
370 for( i = 0; i < n; i++ )
371 usrdata_f[i] = (float)usrdata_d[i];
372 usrdata_p = (const uchar*)usrdata_f;
373 }
374
375 ocl::Kernel k("KF", ocl::core::arithm_oclsrc, opts);
376 if (k.empty())
377 return false;
378
379 UMat src1 = _src1.getUMat(), src2;
380 UMat dst = _dst.getUMat(), mask = _mask.getUMat();
381
382 ocl::KernelArg src1arg = ocl::KernelArg::ReadOnlyNoSize(src1, cn, kercn);
383 ocl::KernelArg dstarg = haveMask ? ocl::KernelArg::ReadWrite(dst, cn, kercn) :
384 ocl::KernelArg::WriteOnly(dst, cn, kercn);
385 ocl::KernelArg maskarg = ocl::KernelArg::ReadOnlyNoSize(mask, 1);
386
387 if( haveScalar )
388 {
389 size_t esz = CV_ELEM_SIZE1(wtype)*scalarcn;
390 double buf[4]={0,0,0,0};
391 Mat src2sc = _src2.getMat();
392
393 if( !src2sc.empty() )
394 convertAndUnrollScalar(src2sc, wtype, (uchar*)buf, 1);
395 ocl::KernelArg scalararg = ocl::KernelArg(ocl::KernelArg::CONSTANT, 0, 0, 0, buf, esz);
396
397 if( !haveMask )
398 {
399 if(n == 0)
400 k.args(src1arg, dstarg, scalararg);
401 else if(n == 1)
402 k.args(src1arg, dstarg, scalararg,
403 ocl::KernelArg(ocl::KernelArg::CONSTANT, 0, 0, 0, usrdata_p, usrdata_esz));
404 else
405 CV_Error(Error::StsNotImplemented, "unsupported number of extra parameters");
406 }
407 else
408 k.args(src1arg, maskarg, dstarg, scalararg);
409 }
410 else
411 {
412 src2 = _src2.getUMat();
413 ocl::KernelArg src2arg = ocl::KernelArg::ReadOnlyNoSize(src2, cn, kercn);
414
415 if( !haveMask )
416 {
417 if (n == 0)
418 k.args(src1arg, src2arg, dstarg);
419 else if (n == 1)
420 k.args(src1arg, src2arg, dstarg,
421 ocl::KernelArg(ocl::KernelArg::CONSTANT, 0, 0, 0, usrdata_p, usrdata_esz));
422 else if (n == 3)
423 k.args(src1arg, src2arg, dstarg,
424 ocl::KernelArg(ocl::KernelArg::CONSTANT, 0, 0, 0, usrdata_p, usrdata_esz),
425 ocl::KernelArg(ocl::KernelArg::CONSTANT, 0, 0, 0, usrdata_p + usrdata_esz, usrdata_esz),
426 ocl::KernelArg(ocl::KernelArg::CONSTANT, 0, 0, 0, usrdata_p + usrdata_esz*2, usrdata_esz));
427 else
428 CV_Error(Error::StsNotImplemented, "unsupported number of extra parameters");
429 }
430 else
431 k.args(src1arg, src2arg, maskarg, dstarg);
432 }
433
434 size_t globalsize[] = { (size_t)src1.cols * cn / kercn, ((size_t)src1.rows + rowsPerWI - 1) / rowsPerWI };
435 return k.run(2, globalsize, NULL, false);
436}

440// OpenCL branch of the arithmetic ops, extracted from the former arithm_op (whose CPU compute is now
441// the element-wise engine). Returns true if an OpenCL kernel handled the op, false to let the caller
442// fall through to the CPU engine. min/max use the bitwise-style kernel; every other op the arithm
443// kernel with a coerced working type. `usrdata` carries the mul/div/addWeighted scale block, `oclop`
444// the kernel id. Only reached for a UMat dst; sizes/types that don't fit the scalar/same-size pattern
445// (e.g. a broadcast) decline (return false) so the engine handles them on the CPU.
446static bool arithm_op_ocl(InputArray _src1, InputArray _src2, OutputArray _dst,
447 InputArray _mask, int dtype, int oclop, bool muldiv, void* usrdata)
448{
449#ifndef HAVE_OPENCL
450 CV_UNUSED(_src1); CV_UNUSED(_src2); CV_UNUSED(_dst); CV_UNUSED(_mask);
451 CV_UNUSED(dtype); CV_UNUSED(oclop); CV_UNUSED(muldiv); CV_UNUSED(usrdata);
452 return false;
453#else
454 const _InputArray *psrc1 = &_src1, *psrc2 = &_src2;
455 _InputArray::KindFlag kind1 = psrc1->kind(), kind2 = psrc2->kind();
456 bool haveMask = !_mask.empty();
457 bool reallocate = false;
458 int type1 = psrc1->type(), depth1 = CV_MAT_DEPTH(type1), cn = CV_MAT_CN(type1);
459 int type2 = psrc2->type(), depth2 = CV_MAT_DEPTH(type2), cn2 = CV_MAT_CN(type2);
460 int wtype, dims1 = psrc1->dims(), dims2 = psrc2->dims();
461 Size sz1 = dims1 <= 2 ? psrc1->size() : Size();
462 Size sz2 = dims2 <= 2 ? psrc2->size() : Size();
463
464 if (!(OCL_PERFORMANCE_CHECK(_dst.isUMat()) && dims1 <= 2 && dims2 <= 2))
465 return false;
466
467 // min/max: the bitwise-style OpenCL kernel (same-type element-wise, no working-type coercion).
468 if (oclop == OCL_OP_MIN || oclop == OCL_OP_MAX)
469 {
470 bool haveScalar = false;
471 if ((kind1 == _InputArray::MATX) + (kind2 == _InputArray::MATX) == 1 ||
472 !psrc1->sameSize(*psrc2) || type1 != type2)
473 {
474 if (checkScalar(*psrc1, type2, kind1, kind2))
475 { std::swap(psrc1, psrc2); std::swap(type1, type2); }
476 else if (!checkScalar(*psrc2, type1, kind2, kind1))
477 return false;
478 haveScalar = true;
479 }
480 _dst.createSameSize(*psrc1, type1);
481 return ocl_binary_op(*psrc1, *psrc2, _dst, _mask, false, oclop, haveScalar);
482 }
483
484 // reciprocal (scale/src): the OpenCL kernel is unary on src - use src2 for both operands (the CPU
485 // engine received a 0-dim `1` numerator as src1, which the recip kernel would ignore). [TODO.VP: tidy]
486 if (oclop == OCL_OP_RECIP_SCALE)
487 {
488 psrc1 = psrc2; kind1 = kind2; type1 = type2; depth1 = depth2; cn = cn2; dims1 = dims2; sz1 = sz2;
489 }
490
491 // add/sub/mul/div/absdiff/addWeighted/recip: the arithm OpenCL kernel with a coerced work type.
492 bool src1Scalar = checkScalar(*psrc1, type2, kind1, kind2);
493 bool src2Scalar = checkScalar(*psrc2, type1, kind2, kind1);
494 bool haveScalar = false;
495
496 if (dims1 != dims2 || sz1 != sz2 || cn != cn2 ||
497 (kind1 == _InputArray::MATX && (sz1 == Size(1,4) || sz1 == Size(1,1))) ||
498 (kind2 == _InputArray::MATX && (sz2 == Size(1,4) || sz2 == Size(1,1))))
499 {
500 if ((type1 == CV_64F && (sz1.height == 1 || sz1.height == 4)) && src1Scalar)
501 {
502 // src1 is a scalar; swap it with src2
503 std::swap(psrc1, psrc2); std::swap(sz1, sz2); std::swap(type1, type2);
504 std::swap(depth1, depth2); std::swap(cn, cn2); std::swap(dims1, dims2);
505 if (oclop == OCL_OP_SUB) oclop = OCL_OP_RSUB;
506 if (oclop == OCL_OP_DIV_SCALE) oclop = OCL_OP_RDIV_SCALE;
507 }
508 else if (!src2Scalar)
509 return false; // array op array with mismatched size/cn: engine broadcasts on the CPU
510 haveScalar = true;
511 CV_Assert((type2 == CV_64F || type2 == CV_32F) && (sz2.height == 1 || sz2.height == 4));
512
513 if (!muldiv)
514 {
515 Mat sc = psrc2->getMat();
516 depth2 = actualScalarDepth(sc.ptr<double>(), sz2 == Size(1, 1) ? cn2 : cn);
517 if (depth2 == CV_64F && CV_ELEM_SIZE1(depth1) < 8)
518 depth2 = CV_32F;
519 }
520 else
521 depth2 = CV_64F;
522 }
523
524 if (dtype < 0)
525 {
526 if (_dst.fixedType())
527 dtype = _dst.type();
528 else
529 {
530 if (!haveScalar && type1 != type2)
531 CV_Error(cv::Error::StsBadArg,
532 "When the input arrays in add/subtract/multiply/divide functions have different types, "
533 "the output array type must be explicitly specified");
534 dtype = type1;
535 }
536 }
537 dtype = CV_MAT_DEPTH(dtype);
538
539 if (depth1 == depth2 && dtype == depth1)
540 wtype = dtype;
541 else if (!muldiv)
542 {
543 wtype = coerceTypes(depth1, depth2, false);
544 wtype = coerceTypes(wtype, dtype, false);
545 if (dtype < CV_32F && (depth1 < CV_32F || depth2 < CV_32F))
546 wtype = CV_32S;
547 }
548 else
549 {
550 wtype = coerceTypes(depth1, depth2, true);
551 wtype = coerceTypes(wtype, dtype, true);
552 }
553
554 // The scaled OpenCL kernels (mul/div by a scale, reciprocal, addWeighted) compute in float; the old
555 // same-type fast path forced max(depth, CV_32F). coerceTypes keeps same-type integer, so bump here.
556 if (oclop == OCL_OP_MUL_SCALE || oclop == OCL_OP_DIV_SCALE || oclop == OCL_OP_RDIV_SCALE ||
557 oclop == OCL_OP_RECIP_SCALE || oclop == OCL_OP_ADDW)
558 wtype = std::max(wtype, (int)CV_32F);
559
560 dtype = CV_MAKETYPE(dtype, cn);
561 wtype = CV_MAKETYPE(wtype, cn);
562
563 if (haveMask)
564 {
565 int mtype = _mask.type();
566 CV_Assert((mtype == CV_8UC1 || mtype == CV_8SC1 || mtype == CV_Bool) && _mask.sameSize(*psrc1));
567 reallocate = !_dst.sameSize(*psrc1) || _dst.type() != dtype;
568 }
569
570 _dst.createSameSize(*psrc1, dtype);
571 if (reallocate)
572 _dst.setTo(0.);
573
574 return ocl_arithm_op(*psrc1, *psrc2, _dst, _mask, wtype, usrdata, oclop, haveScalar);
575#endif
576}

578// Element-wise binary op (ADD/SUB/...) on two CPU operands - each an array Mat or a MATX scalar -
579// through the new broadcasting engine. Handles mixed input types (emitBinary inserts the casts) and
580// broadcasting (a scalar rides as a 0-dim per-channel CONST; arrays broadcast via outputShape/exec).
581// Returns false (declining) for UMat, a write-mask, or two scalar operands, so the caller falls
582// through to arithm_op. Incompatible shapes make the engine throw (matching cv::add's error).
583// Does the output array already have exactly this shape+type (=> create() reuses it, no realloc)?
584// Works for Mat and UMat alike (shape/type queries, no data-pointer peeking).
585static bool outArrayMatches(const _OutputArray& a, const MatShape& shp, int type)
586{
587 if (a.empty() || a.type() != type)
588 return false;
589 int sz[MatShape::MAX_DIMS];
590 int nd = a.sizend(sz);
591 if (nd != (int)shp.size())
592 return false;
593 for (int i = 0; i < nd; i++)
594 if (sz[i] != shp[i])
595 return false;
596 return true;
597}
598
599// Are two operands' iteration shapes (spatial dims + channels innermost, exactly how the engine
600// iterates) numpy-broadcast-compatible? Powers the scalar-like-Mat compat fallback in arithm_op.
601// Cheap by design: sizend copies the dim arrays inline - no Mat headers, no allocations.
602static bool shapesBroadcastCompat(InputArray a, int acn, InputArray b, int bcn)
603{
604 int asz[MatShape::MAX_DIMS + 1], bsz[MatShape::MAX_DIMS + 1];
605 int ad = a.sizend(asz), bd = b.sizend(bsz);
606 asz[ad++] = acn;
607 bsz[bd++] = bcn;
608 for (int i1 = ad - 1, i2 = bd - 1; i1 >= 0 && i2 >= 0; i1--, i2--)
609 if (asz[i1] != bsz[i2] && asz[i1] != 1 && bsz[i2] != 1)
610 return false;
611 return true;
612}

614// The one entry point for every element-wise binary op (add/sub/mul/div/min/max/absdiff/addWeighted/
615// reciprocal). Handles empty inputs, the UMat/OpenCL branch (arithm_op_ocl) and the CPU element-wise
616// engine. `oclop` selects the OpenCL kernel; `muldiv` drives the OpenCL working-type coercion + scale;
617// `params` are the op scalars (params[0]=mul/div scale; {alpha,beta,gamma} for addWeighted).
618// Below this many ELEMENTS (total * channels), the per-call constant overhead of the engine path
619// (program build + compile + executor setup) is comparable to the actual work - take the direct
620// kernel call instead. Same tier boundary as math_op's MATH_OP_SMALL for cv::exp.
621enum { ARITHM_SMALL_DIRECT = 100000 };
622
623static void arithm_op(ew::TOp op, InputArray src1, InputArray src2, OutputArray dst,
624 InputArray mask, int dtype, int oclop, bool muldiv, const Scalar& params)
625{
626 CV_Assert(src1.empty() == src2.empty());
627 if (src1.empty() && src2.empty()) // empty inputs -> empty result
628 {
629 dst.release();
630 if (dtype >= 0)
631 dst.create(0, 0, dtype);
632 return;
633 }
634
635 // UMat -> classic OpenCL kernel. It declines (false) when OpenCL can't apply (no device, dims>2, a
636 // broadcast, cn>4 masked, ...) - then the CPU engine below handles it (mapping the UMat via getMat).
637 // oclop < 0 is a sentinel: the caller (e.g. cv::compare) already ran its own OpenCL path, so this
638 // helper only runs the CPU engine (a UMat operand is mapped via getMat).
639 if (oclop >= 0 && (src1.isUMat() || src2.isUMat() || dst.isUMat()))
640 {
641 double abg[3] = { params[0], params[1], params[2] }, scale = params[0];
642 void* usrdata = (op == ew::OP_ADDW) ? (void*)abg : (muldiv ? (void*)&scale : nullptr);
643 if (arithm_op_ocl(src1, src2, dst, mask, dtype, oclop, muldiv, usrdata))
644 return;
645 }
646
647 const bool haveMask = !mask.empty();
648 const int cn1 = src1.channels(), cn2 = src2.channels();
649 bool s1 = isScalarArg(src1, cn2), s2 = isScalarArg(src2, cn1);
650 if (s1 && s2)
651 s1 = s2 = false; // two scalars: treat both as tiny array operands (element-wise + broadcast),
652 // matching arithm_op (Scalar+Scalar -> 4x1, number+number -> 1x1, ...)
653 // A 4x1 CV_64F Mat/UMat pseudo-scalar (see isScalarArg) is hijacked only against a REAL array:
654 // when the partner is itself a tiny scalar-shaped array (e.g. compare(Mat 4x1, Mat 1x1) -
655 // issue #8999), both are honest data and ride the broadcast. A genuine MATX Scalar never demotes.
656 else if (s1 && src1.kind() != _InputArray::MATX && isScalarLikeMat(src2, cn1))
657 s1 = false;
658 else if (s2 && src2.kind() != _InputArray::MATX && isScalarLikeMat(src1, cn2))
659 s2 = false;
660
661 // Compat fallback: Java/Python/user code passes scalars as real little Mats (the classic 4x1
662 // CV_64F column, a 1xcn/cnx1 vector - see isScalarLikeMat). Those normally ride broadcasting
663 // now, so hijack one as a per-channel scalar ONLY when the shapes do NOT broadcast - just the
664 // calls that would otherwise throw get the 4.x scalar semantics, every valid broadcast keeps
665 // its numpy meaning. Cheap: two geometry probes, and the O(ndims) shape walk runs only when a
666 // probe hits (never for ordinary same-size or MATX-scalar calls).
667 if (!s1 && !s2)
668 {
669 bool like1 = isScalarLikeMat(src1, cn2), like2 = isScalarLikeMat(src2, cn1);
670 if ((like1 || like2) && !shapesBroadcastCompat(src1, cn1, src2, cn2))
671 {
672 s1 = like1; // src1 preferred when both qualify, like the old checkScalar order
673 s2 = like2 && !like1;
674 }
675 }
676
677 // Direct Mat pointers - NO header copy / refcount atomics (the InputArrays outlive this call, so
678 // their data stays alive). A MAT operand is used in place via getObj(); a non-MAT one (UMat, or a
679 // two-scalar operand handled as an array) is mapped once into a local. EXCEPTION - in-place with a
680 // realloc: if dst IS one of the sources, dst.create() may free that source's data mid-op, so take a
681 // header copy (increfs, keeping the old data alive) of the sources in that case only.
682 const void* dstObj = (dst.kind() == _InputArray::MAT) ? dst.getObj() : nullptr;
683 const bool aliased = dstObj && (dstObj == src1.getObj() || dstObj == src2.getObj());
684 Mat m1loc, m2loc, mloc;
685 auto asMat = [&](InputArray a, Mat& loc) -> const Mat* {
686 if (a.kind() == _InputArray::MAT && !aliased) return (const Mat*)a.getObj();
687 loc = a.getMat(); return &loc;
688 };
689 const Mat* pm1 = s1 ? nullptr : asMat(src1, m1loc);
690 const Mat* pm2 = s2 ? nullptr : asMat(src2, m2loc);
691 const int adepth = s1 ? pm2->depth() : pm1->depth(); // the (first) array operand's depth
692 // auto result depth (no explicit dtype, no fixed-type dst): add/sub/mul/div keep the first array
693 // operand's depth (cv::'s dtype==-1 convention), but min/max/absdiff have NO dtype argument and
694 // must promote mixed inputs to their common type (min(u32,f32) in u32 would drop the float). For
695 // same-type inputs promoteArith(da,db)==da, so classic behaviour is preserved (e.g. absdiff of two
696 // s16 stays s16, saturating - the wide unsigned |a-b| is computed then cast back down by emitBinary).
697 int autoDepth = adepth;
698 if (op == ew::OP_MIN || op == ew::OP_MAX || op == ew::OP_ABSDIFF)
699 {
700 const int d1 = s1 ? ew::EW_DEPTH_NONE : pm1->depth();
701 const int d2 = s2 ? ew::EW_DEPTH_NONE : pm2->depth();
702 autoDepth = ew::promoteArith(d1, d2);
703 }
704 // result depth: explicit dtype wins; else a fixed-type dst dictates it; else the auto depth above.
705 const int rdepth = dtype >= 0 ? CV_MAT_DEPTH(dtype)
706 : (dst.fixedType() ? dst.depth() : autoDepth);
707
708 // FAST PATH for the classic hot call: two same-type same-shape continuous arrays, no mask, no
709 // scalar, no broadcast, result depth == input depth, and the array is SMALL. Building and
710 // compiling the 1-instruction program plus the BroadcastOp setup costs ~40-250ns per call -
711 // negligible on big images, dominant at 127x61-class sizes. Call the T x T -> T kernel directly
712 // over the flattened elements instead (the same tier design as math_op for cv::exp). Only ops
713 // whose emitBinary lowering for this exact type combination IS the plain direct kernel are
714 // listed - compare (boundary rewrite/flags) and divide (int guards) lower to more than one
715 // kernel call and keep the ordinary path. addWeighted qualifies exactly when its direct fused
716 // T -> T kernel exists (u8..f32; the 32/64-bit ints lower to wide-compute + cast and get {}
717 // from getElemwiseFunc below, falling through naturally).
718 if (!haveMask && !s1 && !s2 &&
719 (op == ew::OP_ADD || op == ew::OP_SUB || op == ew::OP_MIN || op == ew::OP_MAX ||
720 op == ew::OP_ABSDIFF || op == ew::OP_MUL || op == ew::OP_ADDW ||
721 op == ew::OP_AND || op == ew::OP_OR || op == ew::OP_XOR) &&
722 rdepth == pm1->depth() && pm1->type() == pm2->type() && pm1->size == pm2->size &&
723 pm1->isContinuous() && pm2->isContinuous() &&
724 pm1->total()*cn1 <= (size_t)ARITHM_SMALL_DIRECT)
725 {
726 ew::TKernel k = ew::getElemwiseFunc(op, rdepth, rdepth, ew::EW_DEPTH_NONE, rdepth);
727 if (k.fptr)
728 {
729 dst.createSameSize(src1, pm1->type()); // whole-shape transfer, not piecemeal dims+sizes
730 Mat dloc;
731 Mat* pd = (dst.kind() == _InputArray::MAT) ? (Mat*)dst.getObj() : &(dloc = dst.getMat());
732 if (pd->isContinuous()) // a reused non-continuous dst view falls through to the engine
733 {
734 k.fptr(pm1->data, 0, 1, pm2->data, 0, 1, nullptr, 0, 0,
735 pd->data, 0, (int)(pm1->total()*cn1), 1, params.val, k.flags, k.userdata);
736 return;
737 }
738 }
739 }
740
741 // operand order = (src1, src2): an array becomes an INPUT, a scalar a flexible per-channel CONST
742 // read straight from the caller's inline storage (getObj()), no Mat / convertTo.
743 ew::TExpr p;
744 auto addOperand = [&](InputArray src, bool isScalar, const Mat* m, int otherCn) -> int {
745 if (!isScalar)
746 return p.addInput(m->depth());
747 const uchar* sp; int sd;
748 uchar scbuf[EW_SCALAR_BUF_SIZE]; // stack room for a UMAT scalar; addConst copies the values out
749 const int scn0 = scalarArgElems(src, sp, sd, scbuf), cn = otherCn;
750 const int scn = (cn == scn0 || (cn < 4 && scn0 == 4)) ? cn
751 : (scn0 == 1) ? 1 : 0;
752 CV_Assert(scn > 0);
753 return p.addConst(ew::EW_DEPTH_NONE, sd, sp, scn);
754 };
755 const int a0 = addOperand(src1, s1, pm1, cn2);
756 const int a1 = addOperand(src2, s2, pm2, cn1);
757
758 // Write-mask: a single-channel 1-byte array (u8/s8/bool) the size of the output spatial shape,
759 // added as another broadcast input. The arithmetic result lands in a temp; a final
760 // select(mask, r, dst) -> dst overwrites only mask!=0 positions of the (pre-existing) output.
761 int sMask = 0;
762 const Mat* pmask = nullptr;
763 if (haveMask)
764 {
765 CV_Assert(mask.type() == CV_8U || mask.type() == CV_8S || mask.type() == CV_Bool);
766 pmask = asMat(mask, mloc);
767 sMask = p.addInput(pmask->depth());
768 }
769 const int sOut = p.addOutput(rdepth); // output BEFORE emit -> moveToOutput drops the
770 const int r = p.emitBinary(op, a0, a1, rdepth, params); // result temp (mask-free => none); scale
771 // rides params[0] (mul/div), {a,b,g} for addW
772 if (haveMask)
773 p.addInsn(ew::OP_SELECT, sMask, r, sOut, sOut);
774 else
775 p.moveToOutput(r, sOut);
776 p.compile();
777
778 const Mat* inputs[3]; int ni = 0; // inputs in program (input-index) order
779 if (!s1) inputs[ni++] = pm1;
780 if (!s2) inputs[ni++] = pm2;
781 if (haveMask) inputs[ni++] = pmask;
782
783 // Pre-create the output at the broadcast shape, ALWAYS via dst.create() - never hand exec a raw
784 // dst Mat. Routing through the _OutputArray is what enforces its contract: a FIXED_SIZE/FIXED_TYPE
785 // dst (in-place 'a += b' passes '(const Mat&)a') throws here if a broadcast would change its shape
786 // or type (Mat(1x1) += Mat(1x4)), instead of silently reallocating - exec sees only a Mat and has
787 // no view of the array's fixed flags. Cost over a direct exec is negligible: exec keeps its own
788 // fast path for a matching-shape op, and dst.create() with an unchanged shape/type is a no-op.
789 // With a mask, a freshly (re)allocated dst is zeroed so mask==0 reads 0 (matches arithm_op); a
790 // reused dst keeps its prior content there - detect reuse by shape+type BEFORE create (correct for
791 // a reused UMat dst too).
792 MatShape oshape; int ocn;
793 p.outputShape(inputs, oshape, ocn);
794 const int otype = CV_MAKETYPE(rdepth, ocn);
795 const bool reused = haveMask && outArrayMatches(dst, oshape, otype);
796 dst.create(oshape, otype);
797 Mat dstloc;
798 Mat* pdst = (dst.kind() == _InputArray::MAT) ? (Mat*)dst.getObj() : &(dstloc = dst.getMat());
799 if (haveMask && !reused)
800 pdst->setZero(); // setZero (not setTo, which caps at 4 channels)
801 p.exec(inputs, pdst);
802}
803
804// Unary element-wise op through the engine (bitwise NOT now; unary math later). One array input, an
805// optional write-mask; result depth = dtype (>=0), else a fixed-type dst's depth, else the input
806// depth. Mirrors arithm_op's build/mask/exec tail with a single input and no scalar operand. OpenCL is
807// run by the caller (bitwise_op_ocl), so this is the CPU engine only.
808static void unary_op(ew::TOp op, InputArray src, OutputArray dst, InputArray mask,
809 int dtype, const Scalar& params)
810{
811 if (src.empty())
812 {
813 dst.release();
814 if (dtype >= 0)
815 dst.create(0, 0, dtype);
816 return;
817 }
818
819 const bool haveMask = !mask.empty();
820 // in-place (dst aliases src): keep a header copy so a shape/type-changing dst.create() can't free
821 // src's data mid-op (matches arithm_op). A fixed-size/type dst never reallocs, so this is only for
822 // an explicit dtype change.
823 const void* dstObj = (dst.kind() == _InputArray::MAT) ? dst.getObj() : nullptr;
824 const bool aliased = dstObj && src.kind() == _InputArray::MAT && dstObj == src.getObj();
825 Mat sloc, mloc;
826 const Mat* psrc = (src.kind() == _InputArray::MAT && !aliased) ? (const Mat*)src.getObj()
827 : &(sloc = src.getMat());
828 const int rdepth = dtype >= 0 ? CV_MAT_DEPTH(dtype)
829 : (dst.fixedType() ? dst.depth() : psrc->depth());
830
831 ew::TExpr p;
832 const int a0 = p.addInput(psrc->depth());
833 int sMask = 0;
834 const Mat* pmask = nullptr;
835 if (haveMask)
836 {
837 CV_Assert(mask.type() == CV_8U || mask.type() == CV_8S || mask.type() == CV_Bool);
838 pmask = (mask.kind() == _InputArray::MAT) ? (const Mat*)mask.getObj() : &(mloc = mask.getMat());
839 sMask = p.addInput(pmask->depth());
840 }
841 const int sOut = p.addOutput(rdepth);
842 const int r = p.emitUnary(op, a0, rdepth, params);
843 if (haveMask)
844 p.addInsn(ew::OP_SELECT, sMask, r, sOut, sOut);
845 else
846 p.moveToOutput(r, sOut);
847 p.compile();
848
849 const Mat* inputs[2]; int ni = 0;
850 inputs[ni++] = psrc;
851 if (haveMask) inputs[ni++] = pmask;
852
853 MatShape oshape; int ocn;
854 p.outputShape(inputs, oshape, ocn);
855 const int otype = CV_MAKETYPE(rdepth, ocn);
856 const bool reused = haveMask && outArrayMatches(dst, oshape, otype);
857 dst.create(oshape, otype);
858 Mat dstloc;
859 Mat* pdst = (dst.kind() == _InputArray::MAT) ? (Mat*)dst.getObj() : &(dstloc = dst.getMat());
860 if (haveMask && !reused)
861 pdst->setZero();
862 p.exec(inputs, pdst);
863}

921void divide(double scale, InputArray src2,
922 OutputArray dst, int dtype)
923{
924 CV_INSTRUMENT_REGION();
925
926 if (src2.empty())
927 {
928 dst.release();
929 return;
930 }
931
932 // scale / src2 == scale * 1 / src2: feed a 0-dim `1` as the numerator so the CPU engine reuses the
933 // normal divide (scale*a/b). Give the numerator src2's OWN depth: then OP_DIV(T, T) is same-type and
934 // takes the fast per-type kernel (f32 work for <=16-bit / f16 / bf16, not the s32->f64 path a CV_32S
935 // numerator would force). An integer src2 keeps its div-by-zero->0 guard (both operands integer); a
936 // float src2 divides as float (1/0->inf), both matching cv::divide. dtype<0 follows src2's depth (a
937 // fixed-type dst still wins). The UMat path uses the dedicated reciprocal kernel (OCL_OP_RECIP_SCALE).
938 double one = 1;
939 Mat numerator(MatShape::scalar(), src2.depth(), &one);
940 scalarToRawData(Scalar(1), &one, src2.depth(), 1); // write `1` in src2's depth (no allocation)
941 const int rtype = (dtype < 0 && !dst.fixedType()) ? src2.depth() : dtype;
942 arithm_op(ew::OP_DIV, numerator, src2, dst, noArray(), rtype, OCL_OP_RECIP_SCALE, /*muldiv*/ true, Scalar(scale));
943}

1002 type2 = _src2.type(), depth2 = CV_MAT_DEPTH(type2);

1005 return false;

1007 if (!haveScalar && (!_src1.sameSize(_src2) || type1 != type2))
1008 return false;

1011 // per-channel (scalar[c] for channel c), so a multichannel array against a scalar with DISTINCT
1012 // channel values would disagree - bail to the CPU engine in that case (rare; not worth a kernel fix).
1013 if (haveScalar && cn > 1)

1021 for (int i = 1, n = std::min(cn, scn); i < n; i++)
1022 if (v[i] != v[0])
1023 return false;
1024 }
1025 }

1027 int kercn = haveScalar ? cn : ocl::predictOptimalVectorWidth(_src1, _src2, _dst), rowsPerWI = dev.isIntel() ? 4 : 1;

1029 if (depth1 >= CV_16U)
1030 kercn = 1;

1032 int scalarcn = kercn == 3 ? 4 : kercn;
1033 const char * const operationMap[] = { "==", ">", ">=", "<", "<=", "!=" };
1034 char cvt[50];

1036 String opts = format("-D %s -D srcT1=%s -D dstT=%s -D DEPTH_dst=%d -D workT=srcT1 -D cn=%d"
1037 " -D convertToDT=%s -D OP_CMP -D CMP_OPERATOR=%s -D srcT1_C1=%s"
1038 " -D srcT2_C1=%s -D dstT_C1=%s -D workST=%s -D rowsPerWI=%d%s",
1039 haveScalar ? "UNARY_OP" : "BINARY_OP",
1040 ocl::typeToStr(CV_MAKE_TYPE(depth1, kercn)),
1041 ocl::typeToStr(CV_8UC(kercn)), CV_8U, kercn,
1042 ocl::convertTypeStr(depth1, CV_8U, kercn, cvt, sizeof(cvt)),
1043 operationMap[op], ocl::typeToStr(depth1),
1044 ocl::typeToStr(depth1), ocl::typeToStr(CV_8U),
1045 ocl::typeToStr(CV_MAKE_TYPE(depth1, scalarcn)), rowsPerWI,
1046 doubleSupport ? " -D DOUBLE_SUPPORT" : "");

1049 if (k.empty())
1050 return false;

1053 Size size = src1.size();
1054 _dst.create(size, CV_8UC(cn));
1055 UMat dst = _dst.getUMat();

1057 if (haveScalar)
1058 {
1059 size_t esz = CV_ELEM_SIZE1(type1) * scalarcn;
1060 double buf[4] = { 0, 0, 0, 0 };

1064 convertAndUnrollScalar( src2, depth1, (uchar *)buf, kercn );

1071
1072 if( fval > getMaxVal(depth1) )
1073 return dst.setTo(Scalar::all(op == CMP_LT || op == CMP_LE || op == CMP_NE ? 255 : 0)), true;

1076 if( fval != ival )
1077 {
1078 if( op == CMP_LT || op == CMP_GE )
1079 ival = cvCeil(fval);
1080 else if( op == CMP_LE || op == CMP_GT )
1081 ival = cvFloor(fval);
1082 else
1083 return dst.setTo(Scalar::all(op == CMP_NE ? 255 : 0)), true;
1084 }
1085 convertAndUnrollScalar(Mat(1, 1, CV_32S, &ival), depth1, (uchar *)buf, kercn);
1086 }
1087
1088 ocl::KernelArg scalararg = ocl::KernelArg(ocl::KernelArg::CONSTANT, 0, 0, 0, buf, esz);
1089

1092 }
1093 else

1099 ocl::KernelArg::WriteOnly(dst, cn, kercn));

1110void cv::compare(InputArray _src1, InputArray _src2, OutputArray _dst, int op)
1111{
1112 CV_INSTRUMENT_REGION();
1113
1114 CV_Assert( op == CMP_LT || op == CMP_LE || op == CMP_EQ ||
1115 op == CMP_NE || op == CMP_GE || op == CMP_GT );
1116
1117 CV_Assert(_src1.empty() == _src2.empty());
1118 if (_src1.empty() && _src2.empty())
1119 {
1120 _dst.release();
1121 return;
1122 }
1123
1124 bool haveScalar = false;
1125
1126 if ((_src1.isMatx() + _src2.isMatx()) == 1
1127 || !_src1.sameSize(_src2)
1128 || _src1.type() != _src2.type())
1129 {
1130 bool is_src1_scalar = checkScalar(_src1, _src2.type(), _src1.kind(), _src2.kind());
1131 bool is_src2_scalar = checkScalar(_src2, _src1.type(), _src2.kind(), _src1.kind());
1132
1133 // exactly one scalar operand: keep it as src2 (swapping flips the ordering ops) so the OpenCL
1134 // path sees the array-op-scalar form. Otherwise - two arrays of different size (broadcast),
1135 // mixed types, or two scalars - the CPU engine broadcasts / promotes them, so there is NO error
1136 // here: fall through to arithm_op (which handles scalar detection, broadcast and mixed types).
1137 if (is_src1_scalar && !is_src2_scalar)
1138 {
1139 op = op == CMP_LT ? CMP_GT : op == CMP_LE ? CMP_GE :
1140 op == CMP_GE ? CMP_LE : op == CMP_GT ? CMP_LT : op;
1141 // src1 is a scalar; swap it with src2
1142 compare(_src2, _src1, _dst, op);
1143 return;
1144 }
1145 haveScalar = is_src2_scalar && !is_src1_scalar;
1146 }
1147 CV_UNUSED(haveScalar); // consumed by CV_OCL_RUN only - a no-op without OpenCL
1148
1149 // OpenCL handles only the same-size and array-op-scalar forms; a broadcast between two arrays of
1150 // different size falls through to the CPU engine.
1151 CV_OCL_RUN(_src1.dims() <= 2 && _src2.dims() <= 2 && OCL_PERFORMANCE_CHECK(_dst.isUMat())
1152 && (haveScalar || _src1.sameSize(_src2)),
1153 ocl_compare(_src1, _src2, _dst, op, haveScalar))
1154
1155 // CPU: the element-wise engine. A CMP_* code maps to the corresponding OP_CMP_* TOp; the engine
1156 // forces a u8 mask (dtype = CV_8U) and compares in the common operand type (a fractional scalar
1157 // threshold promotes both operands to float - matching the classic ceil/floor threshold logic).
1158 // The scalar / broadcast / mixed-shape cases are handled inside arithm_op. oclop = -1 tells it the
1159 // OpenCL path was already tried above (via ocl_compare), so it only runs the CPU engine.
1160 ew::TOp cmpOp = op == CMP_EQ ? ew::OP_CMP_EQ : op == CMP_NE ? ew::OP_CMP_NE :
1161 op == CMP_LT ? ew::OP_CMP_LT : op == CMP_LE ? ew::OP_CMP_LE :
1162 op == CMP_GT ? ew::OP_CMP_GT : ew::OP_CMP_GE;
1163 arithm_op(cmpOp, _src1, _src2, _dst, noArray(), CV_8U, /*oclop=*/-1, /*muldiv=*/false);
1164}
