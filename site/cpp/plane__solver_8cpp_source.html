37 models.clear();

40 const int p1_idx = sample[0], p2_idx = sample[1], p3_idx = sample[2];
41 float x1 = pts_ptr_x[p1_idx], y1 = pts_ptr_y[p1_idx], z1 = pts_ptr_z[p1_idx];
42 float x2 = pts_ptr_x[p2_idx], y2 = pts_ptr_y[p2_idx], z2 = pts_ptr_z[p2_idx];
43 float x3 = pts_ptr_x[p3_idx], y3 = pts_ptr_y[p3_idx], z3 = pts_ptr_z[p3_idx];
44
45 // v1 = p1p2 v2 = p1p3
46 float a1 = x2 - x1;
47 float b1 = y2 - y1;
48 float c1 = z2 - z1;
49 float a2 = x3 - x1;
50 float b2 = y3 - y1;
51 float c2 = z3 - z1;
52
53 // Get the plane normal vector v = v1 x v2
54 float a = b1 * c2 - b2 * c1;
55 float b = a2 * c1 - a1 * c2;
56 float c = a1 * b2 - b1 * a2;
57 float d = (-a * x1 - b * y1 - c * z1);
58
59 double plane_coeff[4] = {a, b, c, d};
60 models.emplace_back(cv::Mat(1, 4, CV_64F, plane_coeff).clone());
