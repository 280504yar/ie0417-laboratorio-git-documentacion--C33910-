77void ALIKEDImpl::runNetwork(InputArray _image, std::vector<KeyPoint>& keypoints,
78 Mat& descriptors, Mat& scores)
79{
80 Mat image = _image.getMat();
81 Size inputSz = params.inputSize;
82 Size origSize = image.size();
83
84 // BGR->RGB conversion via swapRB=true
85 Mat blob = dnn::blobFromImage(image, 1.0/255.0, inputSz, Scalar(), /*swapRB=*/true, /*crop=*/false);
86
87 net.setInput(blob, "image");
88
89 std::vector<String> outNames = {"keypoints", "descriptors", "scores"};
90 std::vector<Mat> outputs;
91 net.forward(outputs, outNames);
92
93 CV_Assert(outputs.size() == 3);
94
95 // ORT engine drops the batch dimension, so outputs are:
96 // keypoints: [N, 2] (not [1, N, 2])
97 // descriptors: [N, 128] (not [1, N, 128])
98 // scores: [N] (not [1, N, 1])
99 int N = outputs[0].rows;
100
101 Mat normKpts = outputs[0].reshape(0, N); // Nx2
102 Mat desc = outputs[1].reshape(0, N); // Nx128
103 Mat scr = outputs[2].reshape(0, N); // Nx1
104
105 // Store normalized keypoints for LightGlue context
106 lastContext.normalizedKeypoints = normKpts.clone();
107 lastContext.imageSize = origSize;
108
109 // Convert normalized [-1,1] coordinates to pixel coordinates
110 keypoints.resize(N);
111 for (int i = 0; i < N; i++)
112 {
113 float nx = normKpts.at<float>(i, 0);
114 float ny = normKpts.at<float>(i, 1);
115 float px = (nx + 1.0f) * 0.5f * (float)origSize.width;
116 float py = (ny + 1.0f) * 0.5f * (float)origSize.height;
117 float score = scr.at<float>(i, 0);
118 keypoints[i] = KeyPoint(px, py, 1.0f, -1.0f, score, 0, -1);
119 }
120
121 // Optionally L2-normalize descriptors
122 if (params.normalizeDescriptors)
123 {
124 for (int i = 0; i < N; i++)
125 {
126 Mat row = desc.row(i);
127 normalize(row, row);
128 }
129 }
130
131 descriptors = desc;
132 scores = scr;
133}

135void ALIKEDImpl::detectAndCompute(InputArray image, InputArray mask,
136 std::vector<KeyPoint>& keypoints,
137 OutputArray descriptors,
138 bool useProvidedKeypoints)
139{
140 CV_INSTRUMENT_REGION();
141 CV_UNUSED(mask);
142 CV_UNUSED(useProvidedKeypoints);
143
144 if (image.empty())
145 {
146 keypoints.clear();
147 descriptors.release();
148 return;
149 }
150
151 Mat desc;
152 Mat sc;
153 runNetwork(image, keypoints, desc, sc);
154
155 if (descriptors.needed())
156 desc.copyTo(descriptors);
157}
