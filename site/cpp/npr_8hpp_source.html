134void Domain_Filter::find_magnitude(Mat &img, Mat &mag)
135{
136 int h = img.rows;
137 int w = img.cols;
138
139 vector <Mat> planes;
140 split(img, planes);
141
142 Mat magXR = Mat(h, w, CV_32FC1);
143 Mat magYR = Mat(h, w, CV_32FC1);
144
145 Mat magXG = Mat(h, w, CV_32FC1);
146 Mat magYG = Mat(h, w, CV_32FC1);
147
148 Mat magXB = Mat(h, w, CV_32FC1);
149 Mat magYB = Mat(h, w, CV_32FC1);
150
151 Sobel(planes[0], magXR, CV_32FC1, 1, 0, 3);
152 Sobel(planes[0], magYR, CV_32FC1, 0, 1, 3);
153
154 Sobel(planes[1], magXG, CV_32FC1, 1, 0, 3);
155 Sobel(planes[1], magYG, CV_32FC1, 0, 1, 3);
156
157 Sobel(planes[2], magXB, CV_32FC1, 1, 0, 3);
158 Sobel(planes[2], magYB, CV_32FC1, 0, 1, 3);
159
160 Mat mag1 = Mat(h,w,CV_32FC1);
161 Mat mag2 = Mat(h,w,CV_32FC1);
162 Mat mag3 = Mat(h,w,CV_32FC1);
163
164 magnitude(magXR,magYR,mag1);
165 magnitude(magXG,magYG,mag2);
166 magnitude(magXB,magYB,mag3);
167
168 mag = mag1 + mag2 + mag3;
169 mag = 1.0f - mag;
170}

172void Domain_Filter::compute_Rfilter(Mat &output, Mat &hz, float sigma_h)
173{
174 int h = output.rows;
175 int w = output.cols;
176 int channel = output.channels();
177
178 float a = (float) exp((-1.0 * sqrt(2.0)) / sigma_h);
179
180 Mat temp = Mat(h,w,CV_32FC3);
181
182 output.copyTo(temp);
183 Mat V = Mat(h,w,CV_32FC1);
184
185 for(int i=0;i<h;i++)
186 for(int j=0;j<w;j++)
187 V.at<float>(i,j) = std::pow(a,hz.at<float>(i,j));
188
189 for(int i=0; i<h; i++)
190 {
191 for(int j =1; j < w; j++)
192 {
193 for(int c = 0; c<channel; c++)
194 {
195 temp.at<float>(i,j*channel+c) = temp.at<float>(i,j*channel+c) +
196 (temp.at<float>(i,(j-1)*channel+c) - temp.at<float>(i,j*channel+c)) * V.at<float>(i,j);
197 }
198 }
199 }
200
201 for(int i=0; i<h; i++)
202 {
203 for(int j =w-2; j >= 0; j--)
204 {
205 for(int c = 0; c<channel; c++)
206 {
207 temp.at<float>(i,j*channel+c) = temp.at<float>(i,j*channel+c) +
208 (temp.at<float>(i,(j+1)*channel+c) - temp.at<float>(i,j*channel+c))*V.at<float>(i,j+1);
209 }
210 }
211 }
212
213 temp.copyTo(output);
214}

216void Domain_Filter::compute_boxfilter(Mat &output, Mat &hz, Mat &psketch, float radius)
217{
218 int h = output.rows;
219 int w = output.cols;
220 Mat lower_pos = Mat(h,w,CV_32FC1);
221 Mat upper_pos = Mat(h,w,CV_32FC1);
222
223 lower_pos = hz - radius;
224 upper_pos = hz + radius;
225
226 lower_idx = Mat::zeros(h,w,CV_32FC1);
227 upper_idx = Mat::zeros(h,w,CV_32FC1);
228
229 Mat domain_row = Mat::zeros(1,w+1,CV_32FC1);
230
231 for(int i=0;i<h;i++)
232 {
233 for(int j=0;j<w;j++)
234 domain_row.at<float>(0,j) = hz.at<float>(i,j);
235 domain_row.at<float>(0,w) = (float) myinf;
236
237 Mat lower_pos_row = Mat::zeros(1,w,CV_32FC1);
238 Mat upper_pos_row = Mat::zeros(1,w,CV_32FC1);
239
240 for(int j=0;j<w;j++)
241 {
242 lower_pos_row.at<float>(0,j) = lower_pos.at<float>(i,j);
243 upper_pos_row.at<float>(0,j) = upper_pos.at<float>(i,j);
244 }
245
246 Mat temp_lower_idx = Mat::zeros(1,w,CV_32FC1);
247 Mat temp_upper_idx = Mat::zeros(1,w,CV_32FC1);
248
249 for(int j=0;j<w;j++)
250 {
251 if(domain_row.at<float>(0,j) > lower_pos_row.at<float>(0,0))
252 {
253 temp_lower_idx.at<float>(0,0) = (float) j;
254 break;
255 }
256 }
257 for(int j=0;j<w;j++)
258 {
259 if(domain_row.at<float>(0,j) > upper_pos_row.at<float>(0,0))
260 {
261 temp_upper_idx.at<float>(0,0) = (float) j;
262 break;
263 }
264 }
265
266 int temp = 0;
267 for(int j=1;j<w;j++)
268 {
269 int count=0;
270 for(int k=(int) temp_lower_idx.at<float>(0,j-1);k<w+1;k++)
271 {
272 if(domain_row.at<float>(0,k) > lower_pos_row.at<float>(0,j))
273 {
274 temp = count;
275 break;
276 }
277 count++;
278 }
279
280 temp_lower_idx.at<float>(0,j) = temp_lower_idx.at<float>(0,j-1) + temp;
281
282 count = 0;
283 for(int k=(int) temp_upper_idx.at<float>(0,j-1);k<w+1;k++)
284 {
285
286
287 if(domain_row.at<float>(0,k) > upper_pos_row.at<float>(0,j))
288 {
289 temp = count;
290 break;
291 }
292 count++;
293 }
294
295 temp_upper_idx.at<float>(0,j) = temp_upper_idx.at<float>(0,j-1) + temp;
296 }
297
298 for(int j=0;j<w;j++)
299 {
300 lower_idx.at<float>(i,j) = temp_lower_idx.at<float>(0,j) + 1;
301 upper_idx.at<float>(i,j) = temp_upper_idx.at<float>(0,j) + 1;
302 }
303
304 }
305 psketch = upper_idx - lower_idx;
306}
307void Domain_Filter::compute_NCfilter(Mat &output, Mat &hz, Mat &psketch, float radius)
308{
309 int h = output.rows;
310 int w = output.cols;
311 int channel = output.channels();
312
313 if (w < 2)
314 return;
315
316 compute_boxfilter(output,hz,psketch,radius);
317
318 Mat box_filter = Mat::zeros(h,w+1,CV_32FC3);
319
320 for(int i = 0; i < h; i++)
321 {
322 box_filter.at<float>(i,1*channel+0) = output.at<float>(i,0*channel+0);
323 box_filter.at<float>(i,1*channel+1) = output.at<float>(i,0*channel+1);
324 box_filter.at<float>(i,1*channel+2) = output.at<float>(i,0*channel+2);
325 for(int j = 2; j < w+1; j++)
326 {
327 for(int c=0;c<channel;c++)
328 box_filter.at<float>(i,j*channel+c) = output.at<float>(i,(j-1)*channel+c) + box_filter.at<float>(i,(j-1)*channel+c);
329 }
330 }
331
332 Mat indices = Mat::zeros(h,w,CV_32FC1);
333 Mat final = Mat::zeros(h,w,CV_32FC3);
334
335 for(int i=0;i<h;i++)
336 for(int j=0;j<w;j++)
337 indices.at<float>(i,j) = (float) i+1;
338
339 Mat a = Mat::zeros(h,w,CV_32FC1);
340 Mat b = Mat::zeros(h,w,CV_32FC1);
341
342 // Compute the box filter using a summed area table.
343 for(int c=0;c<channel;c++)
344 {
345 Mat flag = Mat::ones(h,w,CV_32FC1);
346 multiply(flag,c+1,flag);
347
348 Mat temp1, temp2;
349 multiply(flag - 1,h*(w+1),temp1);
350 multiply(lower_idx - 1,h,temp2);
351 a = temp1 + temp2 + indices;
352
353 multiply(flag - 1,h*(w+1),temp1);
354 multiply(upper_idx - 1,h,temp2);
355 b = temp1 + temp2 + indices;
356
357 int p,q,r,rem;
358 int p1,q1,r1,rem1;
359
360 // Calculating indices
361 for(int i=0;i<h;i++)
362 {
363 for(int j=0;j<w;j++)
364 {
365
366 r = (int) b.at<float>(i,j)/(h*(w+1));
367 rem = (int) b.at<float>(i,j) - r*h*(w+1);
368 q = rem/h;
369 p = rem - q*h;
370 if(q==0)
371 {
372 p=h;
373 q=w;
374 r=r-1;
375 }
376 if(p==0)
377 {
378 p=h;
379 q=q-1;
380 }
381
382 r1 = (int) a.at<float>(i,j)/(h*(w+1));
383 rem1 = (int) a.at<float>(i,j) - r1*h*(w+1);
384 q1 = rem1/h;
385 p1 = rem1 - q1*h;
386 if(p1==0)
387 {
388 p1=h;
389 q1=q1-1;
390 }
391
392 final.at<float>(i,j*channel+2-c) = (box_filter.at<float>(p-1,q*channel+(2-r)) - box_filter.at<float>(p1-1,q1*channel+(2-r1)))
393 /(upper_idx.at<float>(i,j) - lower_idx.at<float>(i,j));
394 }
395 }
396 }
397
398 final.copyTo(output);

400void Domain_Filter::init(const Mat &img, int flags, float sigma_s, float sigma_r)
401{
402 int h = img.size().height;
403 int w = img.size().width;
404 int channel = img.channels();
405
407
408 Mat derivx = Mat::zeros(h,w-1,CV_32FC3);
409 Mat derivy = Mat::zeros(h-1,w,CV_32FC3);
410
411 diffx(img,derivx);
412 diffy(img,derivy);
413
414 Mat distx = Mat::zeros(h,w,CV_32FC1);
415 Mat disty = Mat::zeros(h,w,CV_32FC1);
416
418
419 for(int i = 0; i < h; i++)
420 for(int j = 0,k=1; j < w-1; j++,k++)
421 for(int c = 0; c < channel; c++)
422 {
423 distx.at<float>(i,k) =
424 distx.at<float>(i,k) + abs(derivx.at<float>(i,j*channel+c));
425 }
426
427 for(int i = 0,k=1; i < h-1; i++,k++)
428 for(int j = 0; j < w; j++)
429 for(int c = 0; c < channel; c++)
430 {
431 disty.at<float>(k,j) =
432 disty.at<float>(k,j) + abs(derivy.at<float>(i,j*channel+c));
433 }
434
436
437 horiz = Mat(h,w,CV_32FC1);
438 vert = Mat(h,w,CV_32FC1);
439
440 Mat final = Mat(h,w,CV_32FC3);
441
442 Mat tempx,tempy;
443 multiply(distx,sigma_s/sigma_r,tempx);
444 multiply(disty,sigma_s/sigma_r,tempy);
445
446 horiz = 1.0f + tempx;
447 vert = 1.0f + tempy;
448
449 O = Mat(h,w,CV_32FC3);
450 img.copyTo(O);
451
452 O_t = Mat(w,h,CV_32FC3);
453
454 if(flags == 2)
455 {
456
457 ct_H = Mat(h,w,CV_32FC1);
458 ct_V = Mat(h,w,CV_32FC1);
459
460 for(int i = 0; i < h; i++)
461 {
462 ct_H.at<float>(i,0) = horiz.at<float>(i,0);
463 for(int j = 1; j < w; j++)
464 {
465 ct_H.at<float>(i,j) = horiz.at<float>(i,j) + ct_H.at<float>(i,j-1);
466 }
467 }
468
469 for(int j = 0; j < w; j++)
470 {
471 ct_V.at<float>(0,j) = vert.at<float>(0,j);
472 for(int i = 1; i < h; i++)
473 {
474 ct_V.at<float>(i,j) = vert.at<float>(i,j) + ct_V.at<float>(i-1,j);
475 }
476 }
477 }

481void Domain_Filter::filter(const Mat &img, Mat &res, float sigma_s = 60, float sigma_r = 0.4, int flags = 1)
482{
483 int no_of_iter = 3;
484 int h = img.size().height;
485 int w = img.size().width;
486 float sigma_h = sigma_s;
487
488 init(img,flags,sigma_s,sigma_r);
489
490 if(flags == 1)
491 {
492 Mat vert_t = vert.t();
493
494 for(int i=0;i<no_of_iter;i++)
495 {
496 sigma_h = (float) (sigma_s * sqrt(3.0) * std::pow(2,(no_of_iter - (i+1))) / sqrt(std::pow(4,no_of_iter) -1));
497
498 compute_Rfilter(O, horiz, sigma_h);
499
500 O_t = O.t();
501
502 compute_Rfilter(O_t, vert_t, sigma_h);
503
504 O = O_t.t();
505
506 }
507 }
508 else if(flags == 2)
509 {
510
511 Mat vert_t = ct_V.t();
512 Mat temp = Mat(h,w,CV_32FC1);
513 Mat temp1 = Mat(w,h,CV_32FC1);
514
515 float radius;
516
517 for(int i=0;i<no_of_iter;i++)
518 {
519 sigma_h = (float) (sigma_s * sqrt(3.0) * std::pow(2,(no_of_iter - (i+1))) / sqrt(std::pow(4,no_of_iter) -1));
520
521 radius = (float) sqrt(3.0) * sigma_h;
522
523 compute_NCfilter(O, ct_H, temp,radius);
524
525 O_t = O.t();
526
527 compute_NCfilter(O_t, vert_t, temp1, radius);
528
529 O = O_t.t();
530 }
531 }
532
533 res = O.clone();
534}

536void Domain_Filter::pencil_sketch(const Mat &img, Mat &sketch, Mat &color_res, float sigma_s, float sigma_r, float shade_factor)
537{
538
539 int no_of_iter = 3;
540 init(img,2,sigma_s,sigma_r);
541 int h = img.size().height;
542 int w = img.size().width;
543
545
546 Mat color_sketch = Mat(h,w,CV_32FC3);
547
548 cvtColor(img,color_sketch,COLOR_BGR2YCrCb);
549
550 vector <Mat> YUV_channel;
551 Mat vert_t = ct_V.t();
552
553 float sigma_h = sigma_s;
554
555 Mat penx = Mat(h,w,CV_32FC1);
556
557 Mat pen_res = Mat::zeros(h,w,CV_32FC1);
558 Mat peny = Mat(w,h,CV_32FC1);
559
560 Mat peny_t;
561
562 float radius;
563
564 for(int i=0;i<no_of_iter;i++)
565 {
566 sigma_h = (float) (sigma_s * sqrt(3.0) * std::pow(2,(no_of_iter - (i+1))) / sqrt(std::pow(4,no_of_iter) -1));
567
568 radius = (float) sqrt(3.0) * sigma_h;
569
570 compute_boxfilter(O, ct_H, penx, radius);
571
572 O_t = O.t();
573
574 compute_boxfilter(O_t, vert_t, peny, radius);
575
576 O = O_t.t();
577
578 peny_t = peny.t();
579
580 for(int k=0;k<h;k++)
581 for(int j=0;j<w;j++)
582 pen_res.at<float>(k,j) = (shade_factor * (penx.at<float>(k,j) + peny_t.at<float>(k,j)));
583
584 if(i==0)
585 {
586 sketch = pen_res.clone();
587 split(color_sketch,YUV_channel);
588 pen_res.copyTo(YUV_channel[0]);
589 merge(YUV_channel,color_sketch);
590 cvtColor(color_sketch,color_res,COLOR_YCrCb2BGR);
591 }
592
593 }
