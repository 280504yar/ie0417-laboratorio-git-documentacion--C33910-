16/*
17 Implementation of LinearAttention, as defined in ONNX specification:
18 https://onnx.ai/onnx/operators/onnx__LinearAttention.html
19
20 Opset 27 is covered.
21*/
22// Recurrent (O(T)) attention over a fixed-size state S of shape [Dk, Dv], maintained
23// per (batch, kv-head). For each time step t the state is updated and read out:
24// linear : S += k_t (x) v_t
25// delta : S += k_t (x) (beta_t * (v_t - S^T k_t))
26// gated : S = S * exp(decay_t); S += k_t (x) v_t
27// gated_delta : S = S * exp(decay_t); S += k_t (x) (beta_t * (v_t - S^T k_t)) (default)
28// out_t = scale * (q_t . S)
29// GQA/MQA: q has q_num_heads, the state has kv_num_heads; the state is shared across
30// the q_num_heads / kv_num_heads queries in each group.
31// chunk_size is a prefill tuning hint only; ignored.
32
33class LinearAttentionLayerImpl CV_FINAL : public LinearAttentionLayer
34{
35public:
36 LinearAttentionLayerImpl(const LayerParams& params)
37 {
38 setParamsFrom(params);
39 q_num_heads = params.get<int>("q_num_heads", 0);
40 kv_num_heads = params.get<int>("kv_num_heads", 0);
41 scale = params.get<float>("scale", 0.f);
42 // Spec: scale 0.0 means "derive 1/sqrt(d_k)", so test the value, not presence.
43 has_scale = (scale != 0.f);
44 update_rule = params.get<std::string>("update_rule", "gated_delta");
45 CV_Check(update_rule, update_rule == "linear" || update_rule == "gated" ||
46 update_rule == "delta" || update_rule == "gated_delta",
47 "LinearAttention: unknown update_rule");
48 CV_CheckGT(q_num_heads, 0, "LinearAttention: q_num_heads is required");
49 CV_CheckGT(kv_num_heads, 0, "LinearAttention: kv_num_heads is required");
50 CV_CheckEQ(q_num_heads % kv_num_heads, 0, "LinearAttention: q_num_heads must be divisible by kv_num_heads");
51 use_decay = (update_rule == "gated" || update_rule == "gated_delta");
52 use_delta = (update_rule == "delta" || update_rule == "gated_delta");
53 }

60 // Optional inputs arrive as empty Mats in fixed slots: 3=past_state, 4=decay, 5=beta.
61 static bool present(const std::vector<MatShape>& in, size_t i) { return in.size() > i && in[i].dims > 0; }
62 static bool present(const std::vector<Mat>& in, size_t i) { return in.size() > i && !in[i].empty(); }

64 void getTypes(const std::vector<MatType>& inputs,
65 const int requiredOutputs,
66 const int requiredInternals,
67 std::vector<MatType>& outputs,
68 std::vector<MatType>& internals) const CV_OVERRIDE
69 {
70 CV_CheckGE(inputs.size(), (size_t)3, "LinearAttention needs query, key, value");
71 CV_CheckType(inputs[0], inputs[0] == CV_32F || inputs[0] == CV_16F, "LinearAttention: only FP32/FP16 are supported");
72 outputs.assign(requiredOutputs, inputs[0]);
73 internals.assign(requiredInternals, CV_32F); // recurrence-state scratch is always fp32
74 }

76 // requiredOutputs can be 1 (trimmed graph); state then lives in internals.
77 bool getMemoryShapes(const std::vector<MatShape>& inputs,
78 const int requiredOutputs,
79 std::vector<MatShape>& outputs,
80 std::vector<MatShape>& internals) const CV_OVERRIDE
81 {
82 CV_CheckGE(inputs.size(), (size_t)3, "LinearAttention needs query, key, value");
83 CV_CheckEQ(inputs[0].dims, 3, "LinearAttention: query must be 3D [batch, seq, q_num_heads*head_size]");
84 CV_Check(requiredOutputs, requiredOutputs == 1 || requiredOutputs == 2,
85 "LinearAttention: expects 1 (output) or 2 (output, present_state) outputs");
86
87 const int batch = inputs[0][0];
88 const int seq = inputs[0][1];
89 const int dk = inputs[0][2] / q_num_heads;
90 const int dv = inputs[2][2] / kv_num_heads;
91 CV_CheckEQ(inputs[1][2] / kv_num_heads, dk, "LinearAttention: key head_size must equal query head_size");
92
93 const MatShape stateShape{batch, kv_num_heads, dk, dv};
94 outputs.assign(1, MatShape{batch, seq, q_num_heads * dv}); // output
95 internals.clear();
96 if (requiredOutputs >= 2)
97 outputs.push_back(stateShape); // present_state
98 else
99 internals.assign(1, stateShape); // recurrence scratch, not exposed
100 return false;
101 }

103 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
104 {
105 std::vector<Mat> rawInputs, rawOutputs, internals;
106 inputs_arr.getMatVector(rawInputs);
107 outputs_arr.getMatVector(rawOutputs);
108 internals_arr.getMatVector(internals);
109 const bool has_state_output = rawOutputs.size() >= 2; // else present_state lives in internals[0]
110
111 // past_state/present_state use an independent ONNX dtype from Q/K/V; convert per-tensor.
112 std::vector<Mat> in32(rawInputs.size()), out32(rawOutputs.size());
113 for (size_t i = 0; i < rawInputs.size(); ++i)
114 {
115 if (rawInputs[i].empty()) continue;
116 if (rawInputs[i].depth() == CV_16F) rawInputs[i].convertTo(in32[i], CV_32F);
117 else in32[i] = rawInputs[i];
118 }
119 for (size_t i = 0; i < rawOutputs.size(); ++i)
120 {
121 if (rawOutputs[i].depth() == CV_16F) out32[i].create(rawOutputs[i].dims, rawOutputs[i].size.p, CV_32F);
122 else out32[i] = rawOutputs[i];
123 }
124 std::vector<Mat>& inputs = in32;
125 std::vector<Mat>& outputs = out32;
126
127 const Mat& query = inputs[0];
128 const Mat& key = inputs[1];
129 const Mat& value = inputs[2];
130 const bool has_past = present(inputs, 3);
131 const bool has_decay = present(inputs, 4) && use_decay;
132 const bool has_beta = present(inputs, 5) && use_delta;
133
134 const int batch = query.size[0];
135 const int seq = query.size[1];
136 const int Hq = q_num_heads;
137 const int Hkv = kv_num_heads;
138 const int group = Hq / Hkv;
139 const int Dk = query.size[2] / Hq;
140 const int Dv = value.size[2] / Hkv;
141
142 const float scl = has_scale ? scale : 1.0f / std::sqrt(static_cast<float>(Dk));
143
144 // decay: per-kv-head vector of length Dk, or a single value broadcast over Dk (per_head_decay).
145 const int decayDim = has_decay ? inputs[4].size[2] / Hkv : 0;
146 // beta: one scalar per kv-head, or one scalar broadcast over all heads.
147 const bool betaPerHead = has_beta && inputs[5].size[2] >= Hkv;
148
149 const float* Qp = query.ptr<float>();
150 const float* Kp = key.ptr<float>();
151 const float* Vp = value.ptr<float>();
152 const float* Dp = has_decay ? inputs[4].ptr<float>() : nullptr;
153 const float* Bp = has_beta ? inputs[5].ptr<float>() : nullptr;
154 const float* Pp = has_past ? inputs[3].ptr<float>() : nullptr;
155
156 float* Op = outputs[0].ptr<float>();
157 float* Sp = has_state_output ? outputs[1].ptr<float>() : internals[0].ptr<float>();
158
159 const size_t qStride = (size_t)seq * Hq * Dk; // per batch
160 const size_t kStride = (size_t)seq * Hkv * Dk;
161 const size_t vStride = (size_t)seq * Hkv * Dv;
162 const size_t oStride = (size_t)seq * Hq * Dv;
163 const size_t stateSz = (size_t)Dk * Dv;
164
165 parallel_for_(Range(0, batch * Hkv), [&](const Range& r)
166 {
167 std::vector<float> retrieved(Dv);
168 for (int bh = r.start; bh < r.end; ++bh)
169 {
170 const int b = bh / Hkv;
171 const int h = bh % Hkv;
172
173 // state[i*Dv + j], initialised from past_state or zeros.
174 float* S = Sp + (size_t)bh * stateSz;
175 if (has_past)
176 std::memcpy(S, Pp + (size_t)bh * stateSz, stateSz * sizeof(float));
177 else
178 std::memset(S, 0, stateSz * sizeof(float));
179
180 for (int t = 0; t < seq; ++t)
181 {
182 const float* k_t = Kp + b * kStride + (size_t)t * Hkv * Dk + (size_t)h * Dk;
183 const float* v_t = Vp + b * vStride + (size_t)t * Hkv * Dv + (size_t)h * Dv;
184
185 // 1) forget gate: S[i,:] *= exp(decay_i). Resolve the per-head-scalar vs
186 // per-Dk-vector layout outside the i-loop so it stays branchless.
187 if (has_decay)
188 {
189 const float* d_t = Dp + b * (size_t)seq * Hkv * decayDim + (size_t)t * Hkv * decayDim + (size_t)h * decayDim;
190 if (decayDim == 1)
191 {
192 const float gate = std::exp(d_t[0]);
193 for (int i = 0; i < Dk; ++i)
194 {
195 float* Si = S + (size_t)i * Dv;
196 for (int j = 0; j < Dv; ++j) Si[j] *= gate;
197 }
198 }
199 else
200 {
201 for (int i = 0; i < Dk; ++i)
202 {
203 const float gate = std::exp(d_t[i]);
204 float* Si = S + (size_t)i * Dv;
205 for (int j = 0; j < Dv; ++j) Si[j] *= gate;
206 }
207 }
208 }
209
210 // 2) write: outer-product update, optionally delta-corrected
211 if (has_beta)
212 {
213 // retrieved = S^T k_t ([Dv])
214 for (int j = 0; j < Dv; ++j) retrieved[j] = 0.f;
215 for (int i = 0; i < Dk; ++i)
216 {
217 const float ki = k_t[i];
218 const float* Si = S + (size_t)i * Dv;
219 for (int j = 0; j < Dv; ++j) retrieved[j] += Si[j] * ki;
220 }
221 const float beta = Bp[b * (size_t)seq * (betaPerHead ? Hkv : 1) + (size_t)t * (betaPerHead ? Hkv : 1) + (betaPerHead ? h : 0)];
222 for (int i = 0; i < Dk; ++i)
223 {
224 const float ki = k_t[i];
225 float* Si = S + (size_t)i * Dv;
226 for (int j = 0; j < Dv; ++j)
227 Si[j] += ki * (beta * (v_t[j] - retrieved[j]));
228 }
229 }
230 else
231 {
232 for (int i = 0; i < Dk; ++i)
233 {
234 const float ki = k_t[i];
235 float* Si = S + (size_t)i * Dv;
236 for (int j = 0; j < Dv; ++j) Si[j] += ki * v_t[j];
237 }
238 }
239
240 // 3) read-out for every query head in this kv group: out = scale * (q_t . S)
241 for (int g = 0; g < group; ++g)
242 {
243 const int n = h * group + g;
244 const float* q_t = Qp + b * qStride + (size_t)t * Hq * Dk + (size_t)n * Dk;
245 float* o_t = Op + b * oStride + (size_t)t * Hq * Dv + (size_t)n * Dv;
246 for (int j = 0; j < Dv; ++j) o_t[j] = 0.f;
247 for (int i = 0; i < Dk; ++i)
248 {
249 const float qi = q_t[i];
250 const float* Si = S + (size_t)i * Dv;
251 for (int j = 0; j < Dv; ++j) o_t[j] += qi * Si[j];
252 }
253 for (int j = 0; j < Dv; ++j) o_t[j] *= scl;
254 }
255 }
256 }
257 });
258
259 for (size_t i = 0; i < rawOutputs.size(); ++i)
260 if (rawOutputs[i].depth() == CV_16F)
261 out32[i].convertTo(rawOutputs[i], CV_16F);
262 }

264 int64 getFLOPS(const std::vector<MatShape>& inputs, const std::vector<MatShape>& /*outputs*/) const CV_OVERRIDE
265 {
266 const int64 batch = inputs[0][0], seq = inputs[0][1];
267 const int64 Dk = inputs[0][2] / q_num_heads, Dv = inputs[2][2] / kv_num_heads;
268 // per step, per kv-head: ~2 outer products (write) + one read per q head
269 int64 perStep = kv_num_heads * CV_BIG_INT(4) * Dk * Dv + q_num_heads * CV_BIG_INT(2) * Dk * Dv;
270 return batch * seq * perStep;
271 }
272
273private:
274 int q_num_heads = 0;
275 int kv_num_heads = 0;
276 float scale = 0.f;
277 bool has_scale = false;
278 std::string update_rule;
279 bool use_decay = false;
280 bool use_delta = false;
281};
