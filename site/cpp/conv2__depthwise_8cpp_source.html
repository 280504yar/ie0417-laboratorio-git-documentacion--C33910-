19template <typename InpT, typename OutT>
20static void repackDepthwiseWeightsBlock(const InpT* inpw, OutT* outw,
21 int ksize, int C0, int currC0)
22{
23 for (int xy = 0; xy < ksize; xy++, inpw++, outw += C0) {
24 for (int c0 = 0; c0 < currC0; c0++) {
25 outw[c0] = OutT(inpw[ksize*c0]);
26 }
27 }
28}

30// C x 1 x ksize ... => C1 x ksize x C0
31void repackDepthwiseConvWeights(const Mat& weights, Mat& wpack, int outtype_, int C0_)
32{
33 int inptype_ = weights.type();
34 MatShape wshape = weights.shape();
35 CV_Assert(wshape.dims >= 3);
36 CV_Assert(inptype_ == CV_32F || inptype_ == CV_16F || inptype_ == CV_16BF);
37 CV_Assert(outtype_ == CV_32F || outtype_ == CV_16F || outtype_ == CV_16BF);
38 int ksize_ = 1;
39 for (int i = 2; i < wshape.dims; i++)
40 ksize_ *= wshape[i];
41
42 int C1_ = (wshape[0] + C0_ - 1)/C0_;
43
44 if (!wpack.isContinuous())
45 wpack.release();
46
47 wpack.create(MatShape({C1_, ksize_, C0_}, DATA_LAYOUT_UNKNOWN), outtype_);
48
49 parallel_for_(Range(0, C1_), [&](const Range& r) {
50 int inptype = inptype_, outtype = outtype_;
51 size_t inpEsz = CV_ELEM_SIZE(inptype);
52 size_t outEsz = CV_ELEM_SIZE(outtype);
53 int C = wshape[0], C0 = C0_;
54 int ksize = ksize_;
55
56 for (int c1 = r.start; c1 < r.end; c1++) {
57 const uint8_t* inpw_ = (const uint8_t*)weights.data + c1*ksize*C0*inpEsz;
58 uint8_t* outw_ = (uint8_t*)wpack.data + c1*ksize*C0*outEsz;
59 int currC0 = std::min(C - c1*C0, C0);
60 if (currC0 < C0)
61 memset(outw_, 0, ksize*C0*outEsz);
62
63 if (inptype == CV_32F) {
64 const float* inpw = (const float*)inpw_;
65 if (outtype == CV_32F) {
66 float* outw = (float*)outw_;
67 repackDepthwiseWeightsBlock(inpw, outw, ksize, C0, currC0);
68 } else if (outtype == CV_16F) {
69 hfloat* outw = (hfloat*)outw_;
70 repackDepthwiseWeightsBlock(inpw, outw, ksize, C0, currC0);
71 } else if (outtype == CV_16BF) {
72 bfloat* outw = (bfloat*)outw_;
73 repackDepthwiseWeightsBlock(inpw, outw, ksize, C0, currC0);
74 }
75 } else if (inptype == CV_16F) {
76 const hfloat* inpw = (const hfloat*)inpw_;
77 if (outtype == CV_32F) {
78 float* outw = (float*)outw_;

80 } else if (outtype == CV_16F) {
