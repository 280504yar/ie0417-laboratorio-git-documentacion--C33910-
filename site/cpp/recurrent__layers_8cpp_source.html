60template<typename Dtype>
61static void tanh(const Mat &src, Mat &dst)
62{
63 MatConstIterator_<Dtype> itSrc = src.begin<Dtype>();
64 MatIterator_<Dtype> itDst = dst.begin<Dtype>();
65
66 for (; itSrc != src.end<Dtype>(); itSrc++, itDst++)
67 *itDst = std::tanh(*itSrc);
68}

70//TODO: make utils method
71static void tanh(const Mat &src, Mat &dst)
72{
73 dst.create(src.size, src.type());
74
75 if (src.type() == CV_32F)
76 tanh<float>(src, dst);
77 else if (src.type() == CV_64F)
78 tanh<double>(src, dst);
79 else
80 CV_Error(Error::StsUnsupportedFormat, "Function supports only floating point types");
81}

83static void sigmoid(const Mat &src, Mat &dst)
84{
85 cv::exp(-src, dst);
86 cv::pow(1 + dst, -1, dst);
87}

89typedef void (*ActivationFunction)(const Mat &src, Mat &dst);
90static ActivationFunction get_activation_function(const String& activation) {
91 // most used activations for PyTorch and TF : Tanh, Sigmoid
92 // if you need to support more optional activations use std::map instead
93 if (activation == "Tanh")
94 {
95 return tanh;
96 }
97 else if (activation == "Sigmoid")
98 {
99 return sigmoid;
100 }
101 else
102 {
103 CV_Error(Error::StsNotImplemented,
104 cv::format("Activation function [%s] for layer LSTM is not supported", activation.c_str()));
105 }
106}

155 : numTimeStamps(0), numSamples(0)

157 , useAVX(checkHardwareSupport(CPU_AVX))

169 setParamsFrom(params);
170
171 if (params.get<bool>("is_onnx", false))
172 {
173 // collect copies of onnx blobs
174 originalBlobs.insert(originalBlobs.begin(), blobs.begin(), blobs.begin() + 3);
175 blobs.erase(blobs.begin(), blobs.begin() + 3);
176 }
177
178 bidirectional = params.get<bool>("bidirectional", false);
179 if (!blobs.empty())
180 {
181 CV_Assert(blobs.size() >= 3);
182
183 blobs[2] = blobs[2].reshape(1, 1);
184
185 const Mat& Wh = blobs[0];
186 const Mat& Wx = blobs[1];
187 const Mat& bias = blobs[2];
188 const Mat& hInternal = blobs[3];
189 const Mat& cInternal = blobs[4];
190 CV_CheckEQ(Wh.dims, 2, "");
191 CV_CheckEQ(Wx.dims, 2, "");
192 CV_CheckEQ(Wh.rows, Wx.rows, "");
193 CV_CheckEQ(Wh.rows, (1 + static_cast<int>(bidirectional))*4*Wh.cols, "");
194 CV_CheckEQ(Wh.rows, (int)bias.total(), "");
195 // Only perform these checks if hInternal and cInternal are not empty matrices
196 // e.g. inputs are not given by a user
197 if(!hInternal.empty()){
198 CV_CheckEQ(hInternal.cols, Wh.cols, "");
199 }
200 if(!cInternal.empty()){
201 CV_CheckEQ(cInternal.cols, Wh.cols, "");
202 }
203 if (!hInternal.empty() && !cInternal.empty()){ //otherwise check in forward
204 CV_CheckEQ(hInternal.rows, cInternal.rows, "");
205 }
206 CV_Assert(Wh.type() == Wx.type() && Wx.type() == bias.type());
207
208 // Peephole weights.
209 if (blobs.size() > 5)
210 {
211 CV_Assert(blobs.size() == 8);
212 const int N = Wh.cols;
213 for (int i = 5; i < 8; ++i)
214 {
215 CV_Assert(blobs[i].rows == N && blobs[i].cols == N);
216 CV_Assert(blobs[i].type() == bias.type());
217 }
218 }
219 }
220 layout = (layout_t) params.get<int>("layout", SEQ_BATCH_HID);
221 useTimestampDim = params.get<bool>("use_timestamp_dim", true);
222 produceCellOutput = params.get<bool>("produce_cell_output", false);
223 forgetBias = params.get<float>("forget_bias", 0.0f);
224 cellClip = params.get<float>("cell_clip", 0.0f);
225 useCellClip = params.get<bool>("use_cell_clip", false);
226 usePeephole = params.get<bool>("use_peephole", false);
227 reverse = params.get<bool>("reverse", false);
228 numHidden = params.get<int>("hidden_size", 1);
229 CV_Assert(!reverse || !bidirectional);
230
231 // read activations
232 DictValue activations = params.get<DictValue>("activations", DictValue(String()));
233 if (activations.size() == 1) // if activations wasn't specified use default
234 {
235 f_activation = sigmoid;
236 g_activation = tanh;
237 h_activation = tanh;
238 isDefaultActivations = true;
239 } else {
240 CV_Assert(activations.size() == 3);
241 f_activation = get_activation_function(activations.getStringValue(0));
242 g_activation = get_activation_function(activations.getStringValue(1));
243 h_activation = get_activation_function(activations.getStringValue(2));
244 isDefaultActivations = activations.getStringValue(0) == "Sigmoid"
245 && activations.getStringValue(1) == "Tanh"
246 && activations.getStringValue(2) == "Tanh";
247 }
248
249 allocated = false;
250 outTailShape.clear();
251 }

291 bool getMemoryShapes(const std::vector<MatShape> &inputs,
292 const int requiredOutputs,
293 std::vector<MatShape> &outputs,
294 std::vector<MatShape> &internals) const CV_OVERRIDE
295 {
296 CV_Assert((!usePeephole && blobs.size() == 5) || (usePeephole && blobs.size() == 8));
297 CV_Assert((inputs.size() == 1 || inputs.size() == 3));
298 const MatShape& inp0 = inputs[0];
299
300 const Mat &Wh = blobs[0], &Wx = blobs[1];
301 int _numOut = Wh.size[1];
302 int _numInp = Wx.size[1];
303 MatShape outTailShape_(outTailShape), outResShape;
304
305 if (!outTailShape_.empty())
306 CV_Assert(total(outTailShape_) == _numOut);
307 else
308 outTailShape_.assign(1, _numOut);
309
310 int _numSamples;
311 if (useTimestampDim)
312 {
313 CV_Assert(inp0.size() >= 2 && total(inp0, 2) == _numInp);
314 if (layout == SEQ_BATCH_HID) {
315 _numSamples = inp0[1];
316 outResShape.push_back(inp0[0]);
317 outResShape.push_back(_numSamples);
318 } else {
319 // batch-first layout: the output keeps the (batch, seq, ...) order - this must match
320 // what forward() actually writes, the graph engine preallocates outputs by this shape
321 _numSamples = inp0[0];
322 outResShape.push_back(_numSamples);
323 outResShape.push_back(inp0[1]);
324 }
325 }
326 else
327 {
328 CV_Assert(inp0.size() >= 2 && total(inp0, 1) == _numInp);
329 _numSamples = inp0[0];
330 outResShape.push_back(_numSamples);
331 }
332
333 outResShape.insert(outResShape.end(), outTailShape_.begin(), outTailShape_.end());
334 outResShape.back() *= (1 + static_cast<int>(bidirectional));
335
336 outputs.assign(1, outResShape);
337 if (produceCellOutput)
338 {
339 // the producer is ONNX, so CellState is different
340 if (!originalBlobs.empty())
341 {
342 int shp[] = {(1 + static_cast<int>(bidirectional)), _numSamples, numHidden};
343 MatShape newShape(shp, shp + sizeof(shp)/sizeof(shp[0]));
344 outputs.push_back(newShape);
345 }
346 else
347 {
348 outputs.push_back(outResShape);
349 }
350 }
351
352 internals.assign(1, shape(_numSamples, _numOut)); // hInternal
353 internals.push_back(shape(_numSamples, _numOut)); // cInternal
354 internals.push_back(shape(_numSamples, 1)); // dummyOnes
355 internals.push_back(shape(_numSamples, 4*_numOut)); // gates
356
357 return false;
358 }

360 void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays) CV_OVERRIDE
361 {
362 std::vector<Mat> input;
363 inputs_arr.getMatVector(input);
364
365 CV_Assert((!usePeephole && blobs.size() == 5) || (usePeephole && blobs.size() == 8));
366 CV_Assert((input.size() == 1 || input.size() == 3));
367 const Mat& inp0 = input[0];
368
369 Mat &Wh = blobs[0], &Wx = blobs[1];
370 int numOut = Wh.size[1];
371 int numInp = Wx.size[1];
372
373 if (!outTailShape.empty())
374 CV_Assert(total(outTailShape) == numOut);
375 else
376 outTailShape.assign(1, numOut);
377
378 if (useTimestampDim)
379 {
380 CV_Assert(inp0.dims >= 2 && (int)inp0.total(2) == numInp);
381 if (layout == SEQ_BATCH_HID){
382 numTimeStamps = inp0.size[0];
383 numSamples = inp0.size[1];
384 }else{
385 numTimeStamps = inp0.size[1];
386 numSamples = inp0.size[0];
387 }
388 }
389 else
390 {
391 CV_Assert(inp0.dims >= 2 && (int)inp0.total(1) == numInp);
392 numTimeStamps = 1;
393 numSamples = inp0.size[0];
394 }
395
396 outTsShape.clear();
397 outTsShape.push_back(numSamples);
398 outTsShape.insert(outTsShape.end(), outTailShape.begin(), outTailShape.end());
399 outTsShape.back() *= (1 + static_cast<int>(bidirectional));
400
401 allocated = true;
402 }

404 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
405 {
406 CV_TRACE_FUNCTION();
407 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
408
409 if (inputs_arr.depth() == CV_16F)
410 {
411 forward_fallback(inputs_arr, outputs_arr, internals_arr);
412 return;
413 }
414
415 std::vector<Mat> input, output, internals;
416 inputs_arr.getMatVector(input);
417 outputs_arr.getMatVector(output);
418 internals_arr.getMatVector(internals);
419
420 if (layout == BATCH_SEQ_HID){
421 //swap axis 0 and 1 input x
422 cv::Mat tmp;
423 // Since python input is 4 dimensional and C++ input 3 dimensional
424 // we need to process each differently
425 if (input[0].dims == 4){
426 // here !!!
427 CV_Assert(input[0].size[3] == 1);
428 cv::transposeND(input[0], {1, 0, 2, 3}, tmp); //back to seq_len, batch_size, hidden_size format
429 }else{
430 cv::transposeND(input[0], {1, 0, 2}, tmp); //back to seq_len, batch_size, hidden_size format
431 }
432 input[0] = tmp;
433 }
434
435 // For the batch-first layout the (preallocated) output[0] is (batch, seq, ...), but the
436 // recurrence below assembles rows in seq-major order - run it on a seq-first temp, then
437 // transpose INTO output[0] at the end. output[0] itself must never be reallocated/replaced:
438 // it is the tensor the graph engine preallocated, a new header would silently detach from it.
439 Mat hOutSeqFirst = output[0];
440 if (layout == BATCH_SEQ_HID)
441 {
442 MatShape seqFirstShape = output[0].shape();
443 std::swap(seqFirstShape[0], seqFirstShape[1]);
444 hOutSeqFirst = Mat(seqFirstShape, output[0].type());
445 }
446
447 Mat cOut = produceCellOutput ? hOutSeqFirst.clone() : Mat();
448 const bool needYcTransform = !originalBlobs.empty(); // if the producer is onnx
449 const int numDirs = 1 + static_cast<int>(bidirectional);
450 for (int i = 0; i < numDirs; ++i)
451 {
452 Mat Wh = blobs[0];
453 Mat Wx = blobs[1];
454 Mat bias = blobs[2];
455
456 Mat h_0, c_0;
457 // Handle h_0 and c_0 based on input size
458 h_0 = (input.size() >= 2) ? input[1].reshape(1, input[1].size[0] * input[1].size[1]) : blobs[3];
459 c_0 = (input.size() == 3) ? input[2].reshape(1, input[2].size[0] * input[2].size[1]) : blobs[4];
460
461 // Perform checks if input size is 2 or 3
462 if (input.size() >= 2) {
463 CV_CheckEQ(h_0.cols, Wh.cols, "");
464 CV_CheckEQ(h_0.cols, c_0.cols, "");
465 CV_CheckEQ(h_0.rows, c_0.rows, "");
466 }
467
468
469 Mat pI, pF, pO;
470
471 Wh = Wh.rowRange(i * Wh.rows / numDirs, (i + 1) * Wh.rows / numDirs);
472 Wx = Wx.rowRange(i * Wx.rows / numDirs, (i + 1) * Wx.rows / numDirs);
473 bias = bias.colRange(i * bias.cols / numDirs, (i + 1) * bias.cols / numDirs);
474 h_0 = h_0.rowRange(i * h_0.rows / numDirs, (i + 1) * h_0.rows / numDirs);
475 c_0 = c_0.rowRange(i * c_0.rows / numDirs, (i + 1) * c_0.rows / numDirs);
476
477 if (usePeephole)
478 {
479 pI = blobs[5];
480 pF = blobs[6];
481 pO = blobs[7];
482
483 pI = pI.rowRange(i * pI.rows / numDirs, (i + 1) * pI.rows / numDirs);
484 pI = pI.colRange(i * pI.cols / numDirs, (i + 1) * pI.cols / numDirs);
485
486 pF = pF.rowRange(i * pF.rows / numDirs, (i + 1) * pF.rows / numDirs);
487 pF = pF.colRange(i * pF.cols / numDirs, (i + 1) * pF.cols / numDirs);
488
489 pO = pO.rowRange(i * pO.rows / numDirs, (i + 1) * pO.rows / numDirs);
490 pO = pO.colRange(i * pO.cols / numDirs, (i + 1) * pO.cols / numDirs);
491 }
492
493 int numOut = Wh.size[1];
494 Mat hInternal = internals[0], cInternal = internals[1],
495 dummyOnes = internals[2], gates = internals[3];
496 h_0.copyTo(hInternal);
497 c_0.copyTo(cInternal);
498 dummyOnes.setTo(1.);
499
500 int numSamplesTotal = numTimeStamps*numSamples;
501 Mat xTs = input[0].reshape(1, numSamplesTotal);
502
503 Mat hOutTs = hOutSeqFirst.reshape(1, numSamplesTotal);
504 hOutTs = hOutTs.colRange(i * hOutTs.cols / numDirs, (i + 1) * hOutTs.cols / numDirs);
505 Mat cOutTs;
506 if (produceCellOutput)
507 {
508 cOutTs = cOut.reshape(1, numSamplesTotal);
509 cOutTs = cOutTs.colRange(i * cOutTs.cols / numDirs, (i + 1) * cOutTs.cols / numDirs);
510 }
511
512#if CV_TRY_AVX2 || CV_TRY_AVX
513 bool canUseAvx = gates.isContinuous() && bias.isContinuous()
514 && Wx.depth() == CV_32F && gates.depth() == CV_32F
515 && bias.depth() == CV_32F && Wx.cols >= 8;
516 bool canUseAvx_hInternal = hInternal.isContinuous() && gates.isContinuous() && bias.isContinuous()
517 && Wh.depth() == CV_32F && hInternal.depth() == CV_32F && gates.depth() == CV_32F
518 && Wh.cols >= 8;
519#endif
520#if CV_TRY_SVE
521 bool canUseSVE = gates.isContinuous() && bias.isContinuous()
522 && Wx.depth() == CV_32F && gates.depth() == CV_32F
523 && bias.depth() == CV_32F;
524 bool canUseSVE_hInternal = hInternal.isContinuous() && gates.isContinuous() && bias.isContinuous()
525 && Wh.depth() == CV_32F && hInternal.depth() == CV_32F && gates.depth() == CV_32F;
526#endif
527#if CV_TRY_NEON
528 bool canUseNeon = gates.isContinuous() && bias.isContinuous()
529 && Wx.depth() == CV_32F && gates.depth() == CV_32F
530 && bias.depth() == CV_32F && Wx.cols >= 4;
531 bool canUseNeon_hInternal = hInternal.isContinuous() && gates.isContinuous() && bias.isContinuous()
532 && Wh.depth() == CV_32F && hInternal.depth() == CV_32F && gates.depth() == CV_32F
533 && Wh.cols >= 4;
534#endif
535
536 int tsStart, tsEnd, tsInc;
537 if (reverse || i == 1) {
538 tsStart = numTimeStamps - 1;
539 tsEnd = -1;
540 tsInc = -1;
541 }
542 else {
543 tsStart = 0;
544 tsEnd = numTimeStamps;
545 tsInc = 1;
546 }
547 for (int ts = tsStart; ts != tsEnd; ts += tsInc)
548 {
549 Range curRowRange(ts*numSamples, (ts + 1)*numSamples);
550 Mat xCurr = xTs.rowRange(curRowRange);
551
552#if CV_TRY_AVX2
553 if (useAVX2 && canUseAvx && xCurr.isContinuous())
554 {
555 for (int n = 0; n < xCurr.rows; n++) {
556 opt_AVX2::fastGEMM1T(
557 xCurr.ptr<float>(n),
558 Wx.ptr<float>(),
559 Wx.step1(),
560 bias.ptr<float>(),
561 gates.ptr<float>(n),
562 Wx.rows,
563 Wx.cols
564 );
565 }
566 }
567 else
568#endif
569#if CV_TRY_AVX
570 if (useAVX && canUseAvx && xCurr.isContinuous())
571 {
572 for (int n = 0; n < xCurr.rows; n++) {
573 opt_AVX::fastGEMM1T(
574 xCurr.ptr<float>(n),
575 Wx.ptr<float>(),
576 Wx.step1(),
577 bias.ptr<float>(),
578 gates.ptr<float>(n),
579 Wx.rows,
580 Wx.cols
581 );
582 }
583 }
584 else
585#endif
586#if CV_TRY_SVE
587 if (useSVE && canUseSVE && xCurr.isContinuous())
588 {
589 for (int n = 0; n < xCurr.rows; n++) {
590 opt_SVE::fastGEMM1T(
591 xCurr.ptr<float>(n),
592 Wx.ptr<float>(),
593 Wx.step1(),
594 bias.ptr<float>(),
595 gates.ptr<float>(n),
596 Wx.rows,
597 Wx.cols
598 );
599 }
600 }
601 else
602#endif
603#if CV_TRY_NEON
604 if (useNEON && canUseNeon && xCurr.isContinuous())
605 {
606 for (int n = 0; n < xCurr.rows; n++) {
607 opt_NEON::fastGEMM1T(
608 xCurr.ptr<float>(n),
609 Wx.ptr<float>(),
610 Wx.step1(),
611 bias.ptr<float>(),
612 gates.ptr<float>(n),
613 Wx.rows,
614 Wx.cols
615 );
616 }
617 }
618 else
619#endif
620 {
621 gemm(xCurr, Wx, 1, gates, 0, gates, GEMM_2_T); // Wx * x_t
622 gemm(dummyOnes, bias, 1, gates, 1, gates); //+b
623 }
624
625#if CV_TRY_AVX2
626 if (useAVX2 && canUseAvx_hInternal)
627 {
628 for (int n = 0; n < hInternal.rows; n++) {
629 opt_AVX2::fastGEMM1T(
630 hInternal.ptr<float>(n),
631 Wh.ptr<float>(),
632 Wh.step1(),
633 gates.ptr<float>(n),
634 gates.ptr<float>(n),
635 Wh.rows,
636 Wh.cols
637 );
638 }
639 }
640 else
641#endif

648 Wh.ptr<float>(),
649 Wh.step1(),

651 gates.ptr<float>(n),
652 Wh.rows,
653 Wh.cols
654 );
655 }
656 }
657 else
658#endif
659#if CV_TRY_SVE

662 for (int n = 0; n < hInternal.rows; n++) {
663 opt_SVE::fastGEMM1T(
664 hInternal.ptr<float>(n),
665 Wh.ptr<float>(),
666 Wh.step1(),
667 gates.ptr<float>(n),
668 gates.ptr<float>(n),
669 Wh.rows,
670 Wh.cols
671 );
672 }

674 else
675#endif

677 if (useNEON && canUseNeon_hInternal)
678 {
679 for (int n = 0; n < hInternal.rows; n++) {
680 opt_NEON::fastGEMM1T(
681 hInternal.ptr<float>(n),
682 Wh.ptr<float>(),
683 Wh.step1(),
684 gates.ptr<float>(n),
685 gates.ptr<float>(n),
686 Wh.rows,

688 );

691 else
692#endif
693 {
694 gemm(hInternal, Wh, 1, gates, 1, gates, GEMM_2_T); //+Wh * h_{t-1}
695 }
696
697 Mat gateI = gates.colRange(0*numOut, 1*numOut);
698 Mat gateF = gates.colRange(1*numOut, 2*numOut);
699 Mat gateO = gates.colRange(2*numOut, 3*numOut);
700 Mat gateG = gates.colRange(3*numOut, 4*numOut);
701
702 if (forgetBias)
703 add(gateF, forgetBias, gateF);
704
705 if (usePeephole)
706 {
707 Mat gatesIF = gates.colRange(0, 2*numOut);
708 gemm(cInternal, pI, 1, gateI, 1, gateI);
709 gemm(cInternal, pF, 1, gateF, 1, gateF);
710 f_activation(gatesIF, gatesIF);
711 }
712 else
713 {
714 Mat gatesIFO = gates.colRange(0, 3*numOut);
715 f_activation(gatesIFO, gatesIFO);
716 }
717
718 g_activation(gateG, gateG);
719
720 //compute c_t
721 multiply(gateF, cInternal, gateF); // f_t (*) c_{t-1}
722 multiply(gateI, gateG, gateI); // i_t (*) g_t
723 add(gateF, gateI, cInternal); // c_t = f_t (*) c_{t-1} + i_t (*) g_t
724
725 if (useCellClip)
726 {
727 min(cInternal, cellClip, cInternal);
728 max(cInternal, -cellClip, cInternal);
729 }
730 if (usePeephole)
731 {
732 gemm(cInternal, pO, 1, gateO, 1, gateO);
733 f_activation(gateO, gateO);
734 }
735
736 //compute h_t
737 h_activation(cInternal, hInternal);
738 multiply(gateO, hInternal, hInternal);
739
740 //save results in output blobs
741 hInternal.copyTo(hOutTs.rowRange(curRowRange));
742 if (produceCellOutput)
743 cInternal.copyTo(cOutTs.rowRange(curRowRange));
744 }
745 }
746 // transpose to match batch first output - into the preallocated tensor (exact-shape create()
747 // inside transposeND reuses it, so the graph's output blob is written in place)
748 if (layout == BATCH_SEQ_HID){
749 cv::transposeND(hOutSeqFirst, {1, 0, 2}, output[0]);
750 }
751 if (needYcTransform && produceCellOutput)
752 {
753 fixCellState(cOut, numDirs);
754 }
755 if (produceCellOutput)
756 {
757 cOut.copyTo(output[1]);
758 }

761 void fixCellState(Mat& cOut, int numDirs)
762 {
763 // seq, batch, dirs, hidden
764 int shp[] = {0, numSamples, numDirs, numHidden};
765 cOut = cOut.reshape(1, sizeof(shp)/sizeof(shp[0]), shp);
766
767 // permute to {0, 2, 1, 3};
768 cv::Mat newCellState;
769 // transpose to match batch first output
770 if (layout == BATCH_SEQ_HID){
771 cv::transposeND(cOut, {2, 0, 1, 3}, newCellState);
772 }
773 else{
774 cv::transposeND(cOut, {0, 2, 1, 3}, newCellState);
775 }
776 cOut = newCellState;
777
778 if (numDirs == 1)
779 {
780 // Slice: Yh = Y[-1, :, :, :]
781 Range ranges[] = {cv::Range(cOut.size[0] - 1, cOut.size[0]), cv::Range::all(), cv::Range::all(), cv::Range::all()};
782 cOut = cOut(ranges);
783 // Reshape: 1x1xBxH -> 1xBxH
784 int shp[] = {1, numSamples, numHidden};
785 cOut = cOut.reshape(1, sizeof(shp)/sizeof(shp[0]), shp);
786 }
787 else
788 {
789 // Slice: SxDxBxH -> last sequence, first direction
790 Range ranges1[] = {cv::Range(cOut.size[0] - 1, cOut.size[0]), cv::Range(0, 1), cv::Range::all(), cv::Range::all()};
791 Mat part1 = cOut(ranges1);
792
793 // Slice: SxDxBxH -> first sequence, last direction
794 Range ranges2[] = {cv::Range(0, 1), cv::Range(cOut.size[1] - 1, cOut.size[1]), cv::Range::all(), cv::Range::all()};
795 Mat part2 = cOut(ranges2);
796
797 int shp[] = {1, part1.size[2] * part1.size[3]};
798 part1 = part1.reshape(1, sizeof(shp)/sizeof(shp[0]), shp);
799 part2 = part2.reshape(1, sizeof(shp)/sizeof(shp[0]), shp);
800
801 vconcat(part1, part2, cOut);
802
803 // Reshape: 1x2xBxH -> 2xBxH
804 int finalShape[] = {2, numSamples, numHidden};
805 cOut = cOut.reshape(1, sizeof(finalShape)/sizeof(finalShape[0]), finalShape);
806 }
807 }

810 Ptr<BackendNode> initCUDA(void *context_, const std::vector<Ptr<BackendWrapper>> &inputs,
811 const std::vector<Ptr<BackendWrapper>> &outputs) override
812 {
813 const int numDirs = 1 + static_cast<int>(bidirectional);
814 auto toIFCO = [numDirs] (Mat& in) {
815 int first = in.size[0];
816 int rest = in.total() / first / 4;
817 // every weight blob contains weights for Input, Output, Forget and Cell gates
818 Mat m = in.reshape(1, {first, 4, rest});
819 Mat outputGate = m.col(1);
820 Mat forgetGate = m.col(2);
821 Mat cellGate = m.col(3);
822 // IOFC -> IFOC
823 std::swap_ranges(outputGate.begin<float>(), outputGate.end<float>(), forgetGate.begin<float>());
824 std::swap(outputGate, forgetGate);
825 // IFOC -> IFCO
826 std::swap_ranges(outputGate.begin<float>(), outputGate.end<float>(), cellGate.begin<float>());
827 in = in.reshape(1, numDirs);
828 };
829
830 Mat& b = originalBlobs[2];
831 // B is a concatenation of biases for Wh and Wx
832 b = b.reshape(1, originalBlobs[2].size[0]*2);
833
834 for (auto& m : originalBlobs)
835 {
836 toIFCO(m);
837 }
838
839 b = b.reshape(1, static_cast<int>(b.total()));
840
841 Mat ordered_weights;
842 // Wx_f, Wh_f, [Wx_b, Wh_b,] b
843 for (int i = 0; i < numDirs; ++i)
844 {
845 for (size_t j = 0; j < 2; ++j) // Wx, Wh
846 {
847 Mat oneDirection = originalBlobs[j].row(i);
848 ordered_weights.push_back(oneDirection.reshape(1, static_cast<int>(oneDirection.total())));
849 }
850 }
851 ordered_weights.push_back(b);
852
853 // Pass hidden states as is
854 Mat h0 = blobs[3];
855 Mat c0 = blobs[4];
856
857 CV_Assert(!inputs.empty());
858 auto input_wrapper = inputs[0].dynamicCast<CUDABackendWrapper>();
859 auto input_shape = input_wrapper->getShape();
860
861 RNNConfiguration config
862 {
863 input_shape[0], // seqLength;
864 1, // numLayers;
865 numHidden, // hiddenSize;
866 input_shape[2], // inputSize;
867 input_shape[1], // miniBatch;
868 bidirectional
869 };
870
871
872 auto *context = reinterpret_cast<cuda4dnn::csl::CSLContext *>(context_);
873 return make_cuda_node<cuda4dnn::LSTMOp>(preferableTarget, std::move(context->stream),
874 std::move(context->cudnn_handle),
875 ordered_weights, h0, c0,
876 config);
877 }

1015 {
1016 CV_TRACE_FUNCTION();
1017 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
1018
1019 if (inputs_arr.depth() == CV_16F)
1020 {
1021 forward_fallback(inputs_arr, outputs_arr, internals_arr);
1022 return;
1023 }
1024
1025 std::vector<Mat> input, output, internals;
1026 inputs_arr.getMatVector(input);
1027 outputs_arr.getMatVector(output);
1028 internals_arr.getMatVector(internals);
1029
1030 Mat xTs = input[0].reshape(1, numSamplesTotal);
1031 Mat oTs = output[0].reshape(1, numSamplesTotal);
1032 Mat hTs = produceH ? output[1].reshape(1, numSamplesTotal) : Mat();
1033 Mat hCurr = internals[0];
1034 Mat hPrev = internals[1];
1035 Mat dummyBiasOnes = internals[2];
1036
1037 hPrev.setTo(0.);
1038 dummyBiasOnes.setTo(1.);
1039
1040 for (int ts = 0; ts < numTimestamps; ts++)
1041 {
1042 Range curRowRange = Range(ts * numSamples, (ts + 1) * numSamples);
1043 Mat xCurr = xTs.rowRange(curRowRange);
1044
1045 gemm(hPrev, Whh, 1, hCurr, 0, hCurr, GEMM_2_T); // W_{hh} * h_{prev}
1046 gemm(xCurr, Wxh, 1, hCurr, 1, hCurr, GEMM_2_T); //+W_{xh} * x_{curr}
1047 gemm(dummyBiasOnes, bh, 1, hCurr, 1, hCurr); //+bh
1048 tanh(hCurr, hPrev);
1049
1050 Mat oCurr = oTs.rowRange(curRowRange);
1051 gemm(hPrev, Who, 1, oCurr, 0, oCurr, GEMM_2_T); // W_{ho} * h_{prev}
1052 gemm(dummyBiasOnes, bo, 1, oCurr, 1, oCurr); //+b_o
1053 tanh(oCurr, oCurr);
1054
1055 if (produceH)
1056 hPrev.copyTo(hTs.rowRange(curRowRange));
1057 }
1058 }
