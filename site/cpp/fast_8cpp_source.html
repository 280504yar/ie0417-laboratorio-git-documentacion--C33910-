55template<int patternSize>
56void FAST_t(InputArray _img, std::vector<KeyPoint>& keypoints, int threshold, bool nonmax_suppression)
57{
58 Mat img = _img.getMat();
59 const int K = patternSize/2, N = patternSize + K + 1;
60 int i, j, k, pixel[25];
61 makeOffsets(pixel, (int)img.step, patternSize);
62
63#if CV_SIMD128
64 const int quarterPatternSize = patternSize/4;
65 v_uint8x16 delta = v_setall_u8(0x80), t = v_setall_u8((char)threshold), K16 = v_setall_u8((char)K);
66#if CV_TRY_AVX2
67 Ptr<opt_AVX2::FAST_t_patternSize16_AVX2> fast_t_impl_avx2;
68 if(CV_CPU_HAS_SUPPORT_AVX2)
69 fast_t_impl_avx2 = opt_AVX2::FAST_t_patternSize16_AVX2::getImpl(img.cols, threshold, nonmax_suppression, pixel);
70#endif
71
72#endif
73
74 keypoints.clear();
75
76 threshold = std::min(std::max(threshold, 0), 255);
77
78 uchar threshold_tab[512];
79 for( i = -255; i <= 255; i++ )
80 threshold_tab[i+255] = (uchar)(i < -threshold ? 1 : i > threshold ? 2 : 0);
81
82 uchar* buf[3] = { 0 };
83 int* cpbuf[3] = { 0 };
84 utils::BufferArea area;
85 for (unsigned idx = 0; idx < 3; ++idx)
86 {
87 area.allocate(buf[idx], img.cols);
88 area.allocate(cpbuf[idx], img.cols + 1);
89 }
90 area.commit();
91
92 for (unsigned idx = 0; idx < 3; ++idx)
93 {
94 memset(buf[idx], 0, img.cols);
95 }
96
97 for(i = 3; i < img.rows-2; i++)
98 {
99 const uchar* ptr = img.ptr<uchar>(i) + 3;
100 uchar* curr = buf[(i - 3)%3];
101 int* cornerpos = cpbuf[(i - 3)%3] + 1; // cornerpos[-1] is used to store a value
102 memset(curr, 0, img.cols);
103 int ncorners = 0;
104
105 if( i < img.rows - 3 )
106 {
107 j = 3;
108#if CV_SIMD128
109 {
110 if( patternSize == 16 )
111 {
112#if CV_TRY_AVX2
113 if (fast_t_impl_avx2)
114 fast_t_impl_avx2->process(j, ptr, curr, cornerpos, ncorners);
115#endif
116 //vz if (j <= (img.cols - 27)) //it doesn't make sense using vectors for less than 8 elements
117 {
118 for (; j < img.cols - 16 - 3; j += 16, ptr += 16)
119 {
120 v_uint8x16 v = v_load(ptr);
121 v_int8x16 v0 = v_reinterpret_as_s8(v_xor(v_add(v, t), delta));
122 v_int8x16 v1 = v_reinterpret_as_s8(v_xor(v_sub(v, t), delta));
123
124 v_int8x16 x0 = v_reinterpret_as_s8(v_sub_wrap(v_load(ptr + pixel[0]), delta));
125 v_int8x16 x1 = v_reinterpret_as_s8(v_sub_wrap(v_load(ptr + pixel[quarterPatternSize]), delta));
126 v_int8x16 x2 = v_reinterpret_as_s8(v_sub_wrap(v_load(ptr + pixel[2*quarterPatternSize]), delta));
127 v_int8x16 x3 = v_reinterpret_as_s8(v_sub_wrap(v_load(ptr + pixel[3*quarterPatternSize]), delta));
128
129 v_int8x16 m0, m1;
130 m0 = v_and(v_lt(v0, x0), v_lt(v0, x1));
131 m1 = v_and(v_lt(x0, v1), v_lt(x1, v1));
132 m0 = v_or(m0, v_and(v_lt(v0, x1), v_lt(v0, x2)));
133 m1 = v_or(m1, v_and(v_lt(x1, v1), v_lt(x2, v1)));
134 m0 = v_or(m0, v_and(v_lt(v0, x2), v_lt(v0, x3)));
135 m1 = v_or(m1, v_and(v_lt(x2, v1), v_lt(x3, v1)));
136 m0 = v_or(m0, v_and(v_lt(v0, x3), v_lt(v0, x0)));
137 m1 = v_or(m1, v_and(v_lt(x3, v1), v_lt(x0, v1)));
138 m0 = v_or(m0, m1);
139
140 if( !v_check_any(m0) )
141 continue;
142 if( !v_check_any(v_combine_low(m0, m0)) )
143 {
144 j -= 8;
145 ptr -= 8;
146 continue;
147 }
148
149 v_int8x16 c0 = v_setzero_s8();
150 v_int8x16 c1 = v_setzero_s8();
151 v_uint8x16 max0 = v_setzero_u8();
152 v_uint8x16 max1 = v_setzero_u8();
153 for( k = 0; k < N; k++ )
154 {
155 v_int8x16 x = v_reinterpret_as_s8(v_xor(v_load((ptr + pixel[k])), delta));
156 m0 = v_lt(v0, x);
157 m1 = v_lt(x, v1);
158
159 c0 = v_and(v_sub_wrap(c0, m0), m0);
160 c1 = v_and(v_sub_wrap(c1, m1), m1);
161
162 max0 = v_max(max0, v_reinterpret_as_u8(c0));
163 max1 = v_max(max1, v_reinterpret_as_u8(c1));
164 }
165
166 max0 = v_lt(K16, v_max(max0, max1));
167 unsigned int m = v_signmask(v_reinterpret_as_s8(max0));
168
169 for( k = 0; m > 0 && k < 16; k++, m >>= 1 )
170 {
171 if( m & 1 )
172 {
173 cornerpos[ncorners++] = j+k;
174 if(nonmax_suppression)
175 {
176 short d[25];
177 int _k = 0;
178 #if CV_ENABLE_UNROLLED
179 for (; _k + 4 < 25; _k += 5)
180 {
181 d[_k] = (short)(ptr[k] - ptr[k + pixel[_k]]);
182 d[_k + 1] = (short)(ptr[k] - ptr[k + pixel[_k + 1]]);
183 d[_k + 2] = (short)(ptr[k] - ptr[k + pixel[_k + 2]]);
184 d[_k + 3] = (short)(ptr[k] - ptr[k + pixel[_k + 3]]);
185 d[_k + 4] = (short)(ptr[k] - ptr[k + pixel[_k + 4]]);
186 }
187 #else
188 for ( ; _k < 25; _k++)
189 d[_k] = (short)(ptr[k] - ptr[k + pixel[_k]]);
190 #endif
191
192 v_int16x8 a0, b0, a1, b1;
193 a0 = b0 = a1 = b1 = v_load(d + 8);
194 for(int shift = 0; shift < 8; ++shift)
195 {
196 v_int16x8 v_nms = v_load(d + shift);
197 a0 = v_min(a0, v_nms);
198 b0 = v_max(b0, v_nms);
199 v_nms = v_load(d + 9 + shift);
200 a1 = v_min(a1, v_nms);
201 b1 = v_max(b1, v_nms);
202 }
203 curr[j + k] = (uchar)(v_reduce_max(v_max(v_max(a0, a1), v_sub(v_setzero_s16(), v_min(b0, b1)))) - 1);
204 }
205 }
206 }
207 }
208 }
209 }
210 }
211#endif
212 for( ; j < img.cols - 3; j++, ptr++ )
213 {
214 int v = ptr[0];
215 const uchar* tab = &threshold_tab[0] - v + 255;
216 int d = tab[ptr[pixel[0]]] | tab[ptr[pixel[8]]];
217
218 if( d == 0 )
219 continue;
220
221 d &= tab[ptr[pixel[2]]] | tab[ptr[pixel[10]]];
222 d &= tab[ptr[pixel[4]]] | tab[ptr[pixel[12]]];
223 d &= tab[ptr[pixel[6]]] | tab[ptr[pixel[14]]];
224
225 if( d == 0 )
226 continue;
227
228 d &= tab[ptr[pixel[1]]] | tab[ptr[pixel[9]]];
229 d &= tab[ptr[pixel[3]]] | tab[ptr[pixel[11]]];
230 d &= tab[ptr[pixel[5]]] | tab[ptr[pixel[13]]];
231 d &= tab[ptr[pixel[7]]] | tab[ptr[pixel[15]]];
232
233 if( d & 1 )
234 {
235 int vt = v - threshold, count = 0;
236
237 for( k = 0; k < N; k++ )
238 {
239 int x = ptr[pixel[k]];
240 if(x < vt)
241 {
242 if( ++count > K )
243 {
244 cornerpos[ncorners++] = j;
245 if(nonmax_suppression)
246 curr[j] = (uchar)cornerScore<patternSize>(ptr, pixel, threshold);
247 break;
248 }
249 }
250 else
251 count = 0;
252 }
253 }
254
255 if( d & 2 )
256 {
257 int vt = v + threshold, count = 0;
258
259 for( k = 0; k < N; k++ )
260 {
261 int x = ptr[pixel[k]];
262 if(x > vt)
263 {
264 if( ++count > K )
265 {
266 cornerpos[ncorners++] = j;
267 if(nonmax_suppression)
268 curr[j] = (uchar)cornerScore<patternSize>(ptr, pixel, threshold);
269 break;
270 }
271 }
272 else
273 count = 0;
274 }
275 }
276 }
277 }
278
279 cornerpos[-1] = ncorners;
280
281 if( i == 3 )
282 continue;
283
284 const uchar* prev = buf[(i - 4 + 3)%3];
285 const uchar* pprev = buf[(i - 5 + 3)%3];
286 cornerpos = cpbuf[(i - 4 + 3)%3] + 1; // cornerpos[-1] is used to store a value
287 ncorners = cornerpos[-1];
288
289 for( k = 0; k < ncorners; k++ )
290 {
291 j = cornerpos[k];
292 int score = prev[j];
293 if( !nonmax_suppression ||
294 (score > prev[j+1] && score > prev[j-1] &&
295 score > pprev[j-1] && score > pprev[j] && score > pprev[j+1] &&
296 score > curr[j-1] && score > curr[j] && score > curr[j+1]) )
297 {
298 keypoints.push_back(KeyPoint((float)j, (float)(i-1), 7.f, -1, (float)score));
299 }
300 }
301 }
302}

311static bool ocl_FAST( InputArray _img, std::vector<KeyPoint>& keypoints,
312 int threshold, bool nonmax_suppression, int maxKeypoints )
313{
314 UMat img = _img.getUMat();
315 if( img.cols < 7 || img.rows < 7 )
316 return false;
317 size_t globalsize[] = { (size_t)img.cols-6, (size_t)img.rows-6 };
318
319 ocl::Kernel fastKptKernel("FAST_findKeypoints", ocl::features::fast_oclsrc);
320 if (fastKptKernel.empty())
321 return false;
322
323 UMat kp1(1, maxKeypoints*2+1, CV_32S);
324
325 UMat ucounter1(kp1, Rect(0,0,1,1));
326 ucounter1.setTo(Scalar::all(0));
327
328 if( !fastKptKernel.args(ocl::KernelArg::ReadOnly(img),
329 ocl::KernelArg::PtrReadWrite(kp1),
330 maxKeypoints, threshold).run(2, globalsize, 0, true))
331 return false;
332
333 Mat mcounter;
334 ucounter1.copyTo(mcounter);
335 int i, counter = mcounter.at<int>(0);
336 counter = std::min(counter, maxKeypoints);
337
338 keypoints.clear();
339
340 if( counter == 0 )
341 return true;
342
343 if( !nonmax_suppression )
344 {
345 Mat m;
346 kp1(Rect(0, 0, counter*2+1, 1)).copyTo(m);
347 const Point* pt = (const Point*)(m.ptr<int>() + 1);
348 for( i = 0; i < counter; i++ )
349 keypoints.push_back(KeyPoint((float)pt[i].x, (float)pt[i].y, 7.f, -1, 1.f));
350 }
351 else
352 {
353 UMat kp2(1, maxKeypoints*3+1, CV_32S);
354 UMat ucounter2 = kp2(Rect(0,0,1,1));
355 ucounter2.setTo(Scalar::all(0));
356
357 ocl::Kernel fastNMSKernel("FAST_nonmaxSupression", ocl::features::fast_oclsrc);
358 if (fastNMSKernel.empty())
359 return false;
360
361 size_t globalsize_nms[] = { (size_t)counter };
362 if( !fastNMSKernel.args(ocl::KernelArg::PtrReadOnly(kp1),
363 ocl::KernelArg::PtrReadWrite(kp2),
364 ocl::KernelArg::ReadOnly(img),
365 counter, counter).run(1, globalsize_nms, 0, true))
366 return false;
367
368 Mat m2;
369 kp2(Rect(0, 0, counter*3+1, 1)).copyTo(m2);
370 Point3i* pt2 = (Point3i*)(m2.ptr<int>() + 1);
371 int newcounter = std::min(m2.at<int>(0), counter);
372
373 std::sort(pt2, pt2 + newcounter, cmp_pt<Point3i>());
374
375 for( i = 0; i < newcounter; i++ )
376 keypoints.push_back(KeyPoint((float)pt2[i].x, (float)pt2[i].y, 7.f, -1, (float)pt2[i].z));
377 }
378
379 return true;
380}

383static inline int hal_FAST(cv::Mat& src, std::vector<KeyPoint>& keypoints, int threshold, bool nonmax_suppression, FastFeatureDetector::DetectorType type)
384{
385 if (threshold > 20)
386 return CV_HAL_ERROR_NOT_IMPLEMENTED;
387
388 cv::Mat scores(src.size(), src.type());
389
390 int error = cv_hal_FAST_dense(src.data, src.step, scores.data, scores.step, src.cols, src.rows, type);
391
392 if (error != CV_HAL_ERROR_OK)
393 return error;
394
395 cv::Mat suppressedScores(src.size(), src.type());
396
397 if (nonmax_suppression)
398 {
399 error = cv_hal_FAST_NMS(scores.data, scores.step, suppressedScores.data, suppressedScores.step, scores.cols, scores.rows);
400
401 if (error != CV_HAL_ERROR_OK)
402 return error;
403 }
404 else
405 {
406 suppressedScores = scores;
407 }
408
409 if (!threshold && nonmax_suppression) threshold = 1;
410
411 cv::KeyPoint kpt(0, 0, 7.f, -1, 0);
412
413 unsigned uthreshold = (unsigned) threshold;
414
415 int ofs = 3;
416
417 int stride = (int)suppressedScores.step;
418 const unsigned char* pscore = suppressedScores.data;
419
420 keypoints.clear();
421
422 for (int y = ofs; y + ofs < suppressedScores.rows; ++y)
423 {
424 kpt.pt.y = (float)(y);
425 for (int x = ofs; x + ofs < suppressedScores.cols; ++x)
426 {
427 unsigned score = pscore[y * stride + x];
428 if (score > uthreshold)
429 {
430 kpt.pt.x = (float)(x);
431 kpt.response = (nonmax_suppression != 0) ? (float)((int)score - 1) : 0.f;
432 keypoints.push_back(kpt);
433 }
434 }
435 }
436
437 return CV_HAL_ERROR_OK;
438}

440void FAST(InputArray _img, std::vector<KeyPoint>& keypoints, int threshold, bool nonmax_suppression, FastFeatureDetector::DetectorType type)
441{
442 CV_INSTRUMENT_REGION();
443
444 if (type != FastFeatureDetector::TYPE_5_8 &&
445 type != FastFeatureDetector::TYPE_7_12 &&
446 type != FastFeatureDetector::TYPE_9_16)
447 {
448 CV_Error_(Error::StsBadArg,
449 ("Unknown FastFeatureDetector detector type: %d", static_cast<int>(type)));
450 }
451
452 const size_t max_fast_features = std::max(_img.total()/100, size_t(1000)); // Simple heuristic that depends on resolution.
453
454 CV_OCL_RUN(_img.isUMat() && type == FastFeatureDetector::TYPE_9_16,
455 ocl_FAST(_img, keypoints, threshold, nonmax_suppression, (int)max_fast_features));
456
457 cv::Mat img = _img.getMat();
458 CALL_HAL(fast_dense, hal_FAST, img, keypoints, threshold, nonmax_suppression, type);
459
460 size_t keypoints_count = 1;
461 keypoints.clear();
462 KeyPoint* kps = (KeyPoint*)malloc(sizeof(KeyPoint) * keypoints_count);
463 int hal_ret = cv_hal_FASTv2(img.data, img.step, img.cols, img.rows, (void**)&kps,
464 &keypoints_count, threshold, nonmax_suppression, type, realloc);
465 if (hal_ret == CV_HAL_ERROR_OK) {
466 keypoints.assign(kps, kps + keypoints_count);
467 free(kps);
468 return;
469 } else {
470 free(kps);
471 keypoints_count = max_fast_features;
472 keypoints.clear();
473 keypoints.resize(keypoints_count);
474 CALL_HAL(fast, cv_hal_FAST, img.data, img.step, img.cols, img.rows,
475 (uchar*)(keypoints.data()), &keypoints_count, threshold, nonmax_suppression, type);
476 }
477
478 switch(type) {
479 case FastFeatureDetector::TYPE_5_8:
480 FAST_t<8>(_img, keypoints, threshold, nonmax_suppression);
481 break;
482 case FastFeatureDetector::TYPE_7_12:
483 FAST_t<12>(_img, keypoints, threshold, nonmax_suppression);
484 break;
485 case FastFeatureDetector::TYPE_9_16:
486 FAST_t<16>(_img, keypoints, threshold, nonmax_suppression);
487 break;
488 }
489}
