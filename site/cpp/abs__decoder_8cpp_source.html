12void cropROI(const Mat &src, Mat &dst, const std::vector<Point2f> &rects)
13{
14 std::vector<Point2f> vertices = rects;
15 int height = cvRound(norm(vertices[0] - vertices[1]));
16 int width = cvRound(norm(vertices[1] - vertices[2]));
17 if (height > width)
18 {
19 std::swap(height, width);
20 Point2f v0 = vertices[0];
21 vertices.erase(vertices.begin());
22 vertices.push_back(v0);
23 }
24 std::vector<Point2f> dst_vertices{
25 Point2f(0, (float) (height - 1)), Point2f(0, 0), Point2f((float) (width - 1), 0),
26 Point2f((float) (width - 1), (float) (height - 1))};
27 dst.create(Size(width, height), CV_8UC1);
28 Mat M = getPerspectiveTransform(vertices, dst_vertices);
29 warpPerspective(src, dst, M, dst.size(), cv::INTER_LINEAR, BORDER_CONSTANT, Scalar(255));
30}

32void fillCounter(const std::vector<uchar> &row, uint start, Counter &counter)
33{
34 size_t counter_length = counter.pattern.size();
35 std::fill(counter.pattern.begin(), counter.pattern.end(), 0);
36 counter.sum = 0;
37 size_t end = row.size();
38 uchar color = row[start];
39 uint counterPosition = 0;
40 while (start < end)
41 {
42 if (row[start] == color)
43 { // that is, exactly one is true
44 counter.pattern[counterPosition]++;
45 counter.sum++;
46 }
47 else
48 {
49 counterPosition++;
50 if (counterPosition == counter_length)
51 {
52 break;
53 }
54 else
55 {
56 counter.pattern[counterPosition] = 1;
57 counter.sum++;
58 color = 255 - color;
59 }
60 }
61 ++start;
62 }
63}

65static inline uint
66patternMatchVariance(const Counter &counter, const std::vector<int> &pattern, uint maxIndividualVariance)
67{
68 size_t numCounters = counter.pattern.size();
69 int total = static_cast<int>(counter.sum);
70 int patternLength = std::accumulate(pattern.cbegin(), pattern.cend(), 0);
71 if (total < patternLength)
72 {
73 // If we don't even have one pixel per unit of bar width, assume this is too small
74 // to reliably match, so fail:
75 // and use constexpr functions
76 return WHITE;// max
77 }
78 // We're going to fake floating-point math in integers. We just need to use more bits.
79 // Scale up patternLength so that intermediate values below like scaledCounter will have
80 // more "significant digits"
81
82 int unitBarWidth = (total << INTEGER_MATH_SHIFT) / patternLength;
83 maxIndividualVariance = (maxIndividualVariance * unitBarWidth) >> INTEGER_MATH_SHIFT;
84 uint totalVariance = 0;
85 for (uint x = 0; x < numCounters; x++)
86 {
87 int cnt = counter.pattern[x] << INTEGER_MATH_SHIFT;
88 int scaledPattern = pattern[x] * unitBarWidth;
89 uint variance = std::abs(cnt - scaledPattern);
90 if (variance > maxIndividualVariance)
91 {
92 return WHITE;
93 }
94 totalVariance += variance;
95 }
96 return totalVariance / total;
97}
