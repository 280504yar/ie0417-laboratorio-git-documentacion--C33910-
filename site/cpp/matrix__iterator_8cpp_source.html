27void NAryMatIterator::init(const Mat** _arrays, Mat* _planes, uchar** _ptrs, int _narrays)
28{
29 CV_Assert( _arrays && (_ptrs || _planes) );
30 int i, j, d1=0, i0 = -1, d = -1;
31
32 arrays = _arrays;
33 ptrs = _ptrs;
34 planes = _planes;
35 narrays = _narrays;
36 nplanes = 0;
37 size = 0;
38
39 if( narrays < 0 )
40 {
41 for( i = 0; _arrays[i] != 0; i++ )
42 ;
43 narrays = i;
44 CV_Assert(narrays <= 1000);
45 }
46
47 iterdepth = 0;
48
49 for( i = 0; i < narrays; i++ )
50 {
51 CV_Assert(arrays[i] != 0);
52 const Mat& A = *arrays[i];
53 if( ptrs )
54 ptrs[i] = A.data;
55
56 if( !A.data )
57 continue;
58
59 if( i0 < 0 )
60 {
61 i0 = i;
62 d = A.dims;
63
64 // find the first dimensionality which is different from 1;
65 // in any of the arrays the first "d1" step do not affect the continuity
66 for( d1 = 0; d1 < d; d1++ )
67 if( A.size[d1] > 1 )
68 break;
69 }
70 else
71 CV_Assert( A.size == arrays[i0]->size );
72
73 if( !A.isContinuous() )
74 {
75 CV_Assert( A.step[d-1] == A.elemSize() );
76 for( j = d-1; j > d1; j-- )
77 if( A.step[j]*A.size[j] < A.step[j-1] )
78 break;
79 iterdepth = std::max(iterdepth, j);
80 }
81 }
82
83 if( i0 >= 0 )
84 {
85 size = arrays[i0]->size[std::max(d-1, 0)];
86 for( j = d-1; j > iterdepth; j-- )
87 {
88 int64 total1 = (int64)size*arrays[i0]->size[j-1];
89 if( total1 != (int)total1 )
90 break;
91 size = (int)total1;
92 }
93
94 iterdepth = j;
95 if( iterdepth == d1 )
96 iterdepth = 0;
97
98 nplanes = 1;
99 for( j = iterdepth-1; j >= 0; j-- )
100 nplanes *= arrays[i0]->size[j];
101 }
102 else
103 iterdepth = 0;
104
105 idx = 0;
106
107 if( !planes )
108 return;
109
110 for( i = 0; i < narrays; i++ )
111 {
112 CV_Assert(arrays[i] != 0);
113 const Mat& A = *arrays[i];
114
115 if( !A.data )
116 {
117 planes[i] = Mat();
118 continue;
119 }
120
121 planes[i] = Mat(1, (int)size, A.type(), A.data);
122 }
123}

126NAryMatIterator& NAryMatIterator::operator ++()
127{
128 if( idx >= nplanes-1 )
129 return *this;
130 ++idx;
131
132 if( iterdepth == 1 )
133 {
134 if( ptrs )
135 {
136 for( int i = 0; i < narrays; i++ )
137 {
138 if( !ptrs[i] )
139 continue;
140 ptrs[i] = arrays[i]->data + arrays[i]->step[0]*idx;
141 }
142 }
143 if( planes )
144 {
145 for( int i = 0; i < narrays; i++ )
146 {
147 if( !planes[i].data )
148 continue;
149 planes[i].data = arrays[i]->data + arrays[i]->step[0]*idx;
150 }
151 }
152 }
153 else
154 {
155 for( int i = 0; i < narrays; i++ )
156 {
157 const Mat& A = *arrays[i];
158 if( !A.data )
159 continue;
160 int _idx = (int)idx;
161 uchar* data = A.data;
162 for( int j = iterdepth-1; j >= 0 && _idx > 0; j-- )
163 {
164 int szi = A.size[j], t = _idx/szi;
165 data += (_idx - t * szi)*A.step[j];
166 _idx = t;
167 }
168 if( ptrs )
169 ptrs[i] = data;
170 if( planes )
171 planes[i].data = data;
172 }
173 }
174
175 return *this;
176}

210ptrdiff_t MatConstIterator::lpos() const
211{
212 if(!m)
213 return 0;
214 if( m->isContinuous() )
215 return (ptr - sliceStart)/elemSize;
216 ptrdiff_t ofs = ptr - m->ptr();
217 int i, d = m->dims;
218 if( d == 2 )
219 {
220 ptrdiff_t y = ofs/m->step[0];
221 return y*m->cols + (ofs - y*m->step[0])/elemSize;
222 }
223 ptrdiff_t result = 0;
224 for( i = 0; i < d; i++ )
225 {
226 size_t s = m->step[i], v = ofs/s;
227 ofs -= v*s;
228 result = result*m->size[i] + v;
229 }
230 return result;
231}

233void MatConstIterator::seek(ptrdiff_t ofs, bool relative)
234{
235 if( m->isContinuous() )
236 {
237 ptr = (relative ? ptr : sliceStart) + ofs*elemSize;
238 if( ptr < sliceStart )
239 ptr = sliceStart;
240 else if( ptr > sliceEnd )
241 ptr = sliceEnd;
242 return;
243 }
244
245 int d = m->dims;
246 if( d == 2 )
247 {
248 ptrdiff_t ofs0, y;
249 if( relative )
250 {
251 ofs0 = ptr - m->ptr();
252 y = ofs0/m->step[0];
253 ofs += y*m->cols + (ofs0 - y*m->step[0])/elemSize;
254 }
255 y = ofs/m->cols;
256 int y1 = std::min(std::max((int)y, 0), m->rows-1);
257 sliceStart = m->ptr(y1);
258 sliceEnd = sliceStart + m->cols*elemSize;
259 ptr = y < 0 ? sliceStart : y >= m->rows ? sliceEnd :
260 sliceStart + (ofs - y*m->cols)*elemSize;
261 return;
262 }
263
264 if( relative )
265 ofs += lpos();
266
267 if( ofs < 0 )
268 ofs = 0;
269
270 int szi = m->size[d-1];
271 ptrdiff_t t = ofs/szi;
272 int v = (int)(ofs - t*szi);
273 ofs = t;
274 ptr = m->ptr() + v*elemSize;
275 sliceStart = m->ptr();
276
277 for( int i = d-2; i >= 0; i-- )
278 {
279 szi = m->size[i];
280 t = ofs/szi;
281 v = (int)(ofs - t*szi);
282 ofs = t;
283 sliceStart += v*m->step[i];
284 }
285
286 sliceEnd = sliceStart + m->size[d-1]*elemSize;
287 if( ofs > 0 )
288 ptr = sliceEnd;
289 else
290 ptr = sliceStart + (ptr - m->ptr());
291}

293void MatConstIterator::seek(const int* _idx, bool relative)
294{
295 int d = m->dims;
296 ptrdiff_t ofs = 0;
297 if( !_idx )
298 ;
299 else if( d == 2 )
300 ofs = _idx[0]*m->size[1] + _idx[1];
301 else
302 {
303 for( int i = 0; i < d; i++ )
304 ofs = ofs*m->size[i] + _idx[i];
305 }
306 seek(ofs, relative);
307}

311SparseMatConstIterator::SparseMatConstIterator(const SparseMat* _m) : m((SparseMat*)_m), hashidx(0), ptr(0)
312{
313 if(!_m || !_m->hdr)
314 return;
315 SparseMat::Hdr& hdr = *m->hdr;
316 const std::vector<size_t>& htab = hdr.hashtab;
317 size_t i, hsize = htab.size();
318 for( i = 0; i < hsize; i++ )
319 {
320 size_t nidx = htab[i];
321 if( nidx )
322 {
323 hashidx = i;
324 ptr = &hdr.pool[nidx] + hdr.valueOffset;
325 return;
326 }
327 }
328}

330SparseMatConstIterator& SparseMatConstIterator::operator ++()
331{
332 if( !ptr || !m || !m->hdr )
333 return *this;
334 SparseMat::Hdr& hdr = *m->hdr;
335 size_t next = ((const SparseMat::Node*)(ptr - hdr.valueOffset))->next;
336 if( next )
337 {
338 ptr = &hdr.pool[next] + hdr.valueOffset;
339 return *this;
340 }
341 size_t i = hashidx + 1, sz = hdr.hashtab.size();
342 for( ; i < sz; i++ )
343 {
344 size_t nidx = hdr.hashtab[i];
345 if( nidx )
346 {
347 hashidx = i;
348 ptr = &hdr.pool[nidx] + hdr.valueOffset;
349 return *this;
350 }
351 }
352 hashidx = sz;
353 ptr = 0;
354 return *this;
355}
