13template<typename T>
14class ComparatorGreater {
15public:
16 ComparatorGreater(const T* data, size_t step)
17 : data_(data), step_(step) {}
18
19 void addOffset(size_t offset) {
20 data_ += offset;
21 }
22
23 void minusOffset(size_t offset) {
24 data_ -= offset;
25 }
26
27 bool operator()(const size_t lhs_idx, const size_t rhs_idx) {
28 T lhs = *(data_ + lhs_idx * step_),
29 rhs = *(data_ + rhs_idx * step_);
30 return (lhs > rhs || (lhs == rhs && lhs_idx < rhs_idx));
31 }
32
33private:
34 const T* data_;
35 size_t step_;
36};

85 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
86 const int requiredOutputs,
87 std::vector<MatShape> &outputs,
88 std::vector<MatShape> &internals) const CV_OVERRIDE
89 {
90 const auto &input_shape = inputs.front();
91 int input_dims = input_shape.size();
92
93 // Check if axis is valid
94 CV_CheckGE(axis, -input_dims, "TopK: axis is out of range");
95 CV_CheckLT(axis, input_dims, "TopK: axis is out of range");
96 // Normalize axis
97 int axis_normalized = normalize_axis(axis, input_shape.size());
98
99 // Check if K is in range (0, input_shape[axis]]
100 CV_CheckGT(K, 0, "TopK: K needs to be a positive integer");
101 CV_CheckLE(K, input_shape[axis_normalized], "TopK: K is out of range");
102
103 // Assign output shape
104 auto output_shape = input_shape;
105 output_shape[axis_normalized] = K;
106 outputs.assign(2, output_shape);
107
108 return false;
109 }

133 template<class Comparator, typename T>
134 void FindTopK(const Mat &input, Mat &output_value, Mat &output_index) {
135 const auto input_shape = shape(input);
136 size_t loops = std::accumulate(input_shape.begin(), input_shape.begin() + axis, 1, std::multiplies<int>());
137 size_t step = std::accumulate(input_shape.begin() + axis + 1, input_shape.end(), 1, std::multiplies<int>());
138 int dim_axis = input_shape[axis];
139 if (loops == 1) {
140 auto worker = [&](const Range &r) {
141 const auto *input_ptr = input.ptr<const T>();
142 auto *output_value_ptr = output_value.ptr<T>();
143 auto *output_index_ptr = output_index.ptr<int64_t>();
144
145 Comparator cmp(input_ptr, step);
146
147 AutoBuffer<int> buffer_index(dim_axis);
148 auto *buffer_index_ptr = buffer_index.data();
149 for (int offset = r.start; offset < r.end; offset++) {
150 const auto *input_offset_ptr = input_ptr + offset;
151 cmp.addOffset(offset);
152
153 std::iota(buffer_index_ptr, buffer_index_ptr + dim_axis, 0);
154 std::stable_sort(buffer_index_ptr, buffer_index_ptr + dim_axis, cmp);
155
156 auto *output_value_offset_ptr = output_value_ptr + offset;
157 auto *output_index_offset_ptr = output_index_ptr + offset;
158 for (int i = 0; i < K; i++) {
159 int source_index = buffer_index_ptr[i];
160 output_value_offset_ptr[i * step] = *(input_offset_ptr + source_index * step);
161 output_index_offset_ptr[i * step] = source_index;
162 }
163 cmp.minusOffset(offset);
164 }
165 };
166 parallel_for_(Range(0, step), worker);
167 } else {
168 auto worker = [&](const Range &r) {
169 const auto *input_ptr = input.ptr<const T>();
170 auto *output_value_ptr = output_value.ptr<T>();
171 auto *output_index_ptr = output_index.ptr<int64_t>();
172
173 Comparator cmp(input_ptr, step);
174
175 AutoBuffer<int> buffer_index(dim_axis);
176 auto *buffer_index_ptr = buffer_index.data();
177 for (int batch_index = r.start; batch_index < r.end; batch_index++) {
178 for (size_t offset = 0; offset < step; offset++) {
179 const auto *input_offset_ptr = input_ptr + batch_index * dim_axis * step + offset;
180 cmp.addOffset(batch_index * dim_axis * step + offset);
181
182 std::iota(buffer_index_ptr, buffer_index_ptr + dim_axis, 0);
183 std::stable_sort(buffer_index_ptr, buffer_index_ptr + dim_axis, cmp);
184
185 auto *output_value_offset_ptr = output_value_ptr + batch_index * K * step + offset;
186 auto *output_index_offset_ptr = output_index_ptr + batch_index * K * step + offset;
187 for (int i = 0; i < K; i++) {
188 int source_index = buffer_index_ptr[i];
189 output_value_offset_ptr[i * step] = *(input_offset_ptr + source_index * step);
190 output_index_offset_ptr[i * step] = source_index;
191 }
192 cmp.minusOffset(batch_index * dim_axis * step + offset);
193 }
194 }
195 };
196 parallel_for_(Range(0, loops), worker);
197 }
198 }

200 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
201 {
202 CV_TRACE_FUNCTION();
203 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
204
205 if (inputs_arr.depth() == CV_16F)
206 {
207 forward_fallback(inputs_arr, outputs_arr, internals_arr);
208 return;
209 }
210
211 std::vector<Mat> inputs, outputs;
212 inputs_arr.getMatVector(inputs);
213 outputs_arr.getMatVector(outputs);
214
215 const auto &input = inputs.front();
216 auto &output_value = outputs.front();
217 auto &output_index = outputs.back();
218
219 if (largest) {
220 switch (input.depth()) {
221 case CV_8U: FindTopK<ComparatorGreater<uint8_t>, uint8_t>(input, output_value, output_index); break;
222 case CV_8S: FindTopK<ComparatorGreater<int8_t>, int8_t>(input, output_value, output_index); break;
223 case CV_16U: FindTopK<ComparatorGreater<uint16_t>, uint16_t>(input, output_value, output_index); break;
224 case CV_16S: FindTopK<ComparatorGreater<int16_t>, int16_t>(input, output_value, output_index); break;
225 case CV_16F: FindTopK<ComparatorGreater<hfloat>, hfloat>(input, output_value, output_index); break;
226 case CV_32U: FindTopK<ComparatorGreater<unsigned>, unsigned>(input, output_value, output_index); break;
227 case CV_32S: FindTopK<ComparatorGreater<int>, int>(input, output_value, output_index); break;
228 case CV_32F: FindTopK<ComparatorGreater<float>, float>(input, output_value, output_index); break;
229 case CV_64U: FindTopK<ComparatorGreater<uint64_t>, uint64_t>(input, output_value, output_index); break;
230 case CV_64S: FindTopK<ComparatorGreater<int64_t>, int64_t>(input, output_value, output_index); break;
231 case CV_64F: FindTopK<ComparatorGreater<double>, double>(input, output_value, output_index); break;
232 default: CV_Error(Error::BadDepth, "Unsupported input data type");
233 }
234 } else {
235 switch (input.depth()) {
236 case CV_8U: FindTopK<ComparatorLess<uint8_t>, uint8_t>(input, output_value, output_index); break;
237 case CV_8S: FindTopK<ComparatorLess<int8_t>, int8_t>(input, output_value, output_index); break;
238 case CV_16U: FindTopK<ComparatorLess<uint16_t>, uint16_t>(input, output_value, output_index); break;
239 case CV_16S: FindTopK<ComparatorLess<int16_t>, int16_t>(input, output_value, output_index); break;
240 case CV_16F: FindTopK<ComparatorLess<hfloat>, hfloat>(input, output_value, output_index); break;
241 case CV_32U: FindTopK<ComparatorLess<unsigned>, unsigned>(input, output_value, output_index); break;
242 case CV_32S: FindTopK<ComparatorLess<int>, int>(input, output_value, output_index); break;
243 case CV_32F: FindTopK<ComparatorLess<float>, float>(input, output_value, output_index); break;
244 case CV_64U: FindTopK<ComparatorLess<uint64_t>, uint64_t>(input, output_value, output_index); break;
245 case CV_64S: FindTopK<ComparatorLess<int64_t>, int64_t>(input, output_value, output_index); break;
246 case CV_64F: FindTopK<ComparatorLess<double>, double>(input, output_value, output_index); break;
247 default: CV_Error(Error::BadDepth, "Unsupported input data type");
248 }
249 }
250 }
