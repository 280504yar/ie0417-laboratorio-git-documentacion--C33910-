25bool ocl_sum( InputArray _src, Scalar & res, int sum_op, InputArray _mask,
26 InputArray _src2, bool calc2, const Scalar & res2 )
27{
28 CV_Assert(sum_op == OCL_OP_SUM || sum_op == OCL_OP_SUM_ABS || sum_op == OCL_OP_SUM_SQR);
29
30 const ocl::Device & dev = ocl::Device::getDefault();
31 bool doubleSupport = dev.doubleFPConfig() > 0,
32 haveMask = _mask.kind() != _InputArray::NONE,
33 haveSrc2 = _src2.kind() != _InputArray::NONE;
34 int type = _src.type(), depth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type),
35 kercn = cn == 1 && !haveMask ? ocl::predictOptimalVectorWidth(_src, _src2) : 1,
36 mcn = std::max(cn, kercn);
37 CV_Assert(!haveSrc2 || _src2.type() == type);
38 int convert_cn = haveSrc2 ? mcn : cn;
39
40 if ( (!doubleSupport && depth == CV_64F) || cn > 4 )
41 return false;
42
43 if (depth >= CV_16F)
44 return false;
45
46 int ngroups = dev.maxComputeUnits(), dbsize = ngroups * (calc2 ? 2 : 1);
47 size_t wgs = dev.maxWorkGroupSize();
48
49 int ddepth = std::max(sum_op == OCL_OP_SUM_SQR ? CV_32F : CV_32S, depth),
50 dtype = CV_MAKE_TYPE(ddepth, cn);
51 CV_Assert(!haveMask || _mask.type() == CV_8U || _mask.type() == CV_8S || _mask.type() == CV_Bool);
52
53 int wgs2_aligned = 1;
54 while (wgs2_aligned < (int)wgs)
55 wgs2_aligned <<= 1;
56 wgs2_aligned >>= 1;
57
58 static const char * const opMap[3] = { "OP_SUM", "OP_SUM_ABS", "OP_SUM_SQR" };
59 char cvt[2][50];
60 String opts = format("-D srcT=%s -D srcT1=%s -D dstT=%s -D dstTK=%s -D dstT1=%s -D ddepth=%d -D cn=%d"
61 " -D convertToDT=%s -D %s -D WGS=%d -D WGS2_ALIGNED=%d%s%s%s%s -D kercn=%d%s%s%s -D convertFromU=%s",
62 ocl::typeToStr(CV_MAKE_TYPE(depth, mcn)), ocl::typeToStr(depth),
63 ocl::typeToStr(dtype), ocl::typeToStr(CV_MAKE_TYPE(ddepth, mcn)),
64 ocl::typeToStr(ddepth), ddepth, cn,
65 ocl::convertTypeStr(depth, ddepth, mcn, cvt[0], sizeof(cvt[0])),
66 opMap[sum_op], (int)wgs, wgs2_aligned,
67 doubleSupport ? " -D DOUBLE_SUPPORT" : "",
68 haveMask ? " -D HAVE_MASK" : "",
69 _src.isContinuous() ? " -D HAVE_SRC_CONT" : "",
70 haveMask && _mask.isContinuous() ? " -D HAVE_MASK_CONT" : "", kercn,
71 haveSrc2 ? " -D HAVE_SRC2" : "", calc2 ? " -D OP_CALC2" : "",
72 haveSrc2 && _src2.isContinuous() ? " -D HAVE_SRC2_CONT" : "",
73 depth <= CV_32S && ddepth == CV_32S ? ocl::convertTypeStr(CV_8U, ddepth, convert_cn, cvt[1], sizeof(cvt[1])) : "noconvert");
74
75 ocl::Kernel k("reduce", ocl::core::reduce_oclsrc, opts);
76 if (k.empty())
77 return false;
78
79 UMat src = _src.getUMat(), src2 = _src2.getUMat(),
80 db(1, dbsize, dtype), mask = _mask.getUMat();
81
82 ocl::KernelArg srcarg = ocl::KernelArg::ReadOnlyNoSize(src),
83 dbarg = ocl::KernelArg::PtrWriteOnly(db),
84 maskarg = ocl::KernelArg::ReadOnlyNoSize(mask),
85 src2arg = ocl::KernelArg::ReadOnlyNoSize(src2);
86
87 if (haveMask)
88 {
89 if (haveSrc2)
90 k.args(srcarg, src.cols, (int)src.total(), ngroups, dbarg, maskarg, src2arg);
91 else
92 k.args(srcarg, src.cols, (int)src.total(), ngroups, dbarg, maskarg);
93 }
94 else
95 {
96 if (haveSrc2)
97 k.args(srcarg, src.cols, (int)src.total(), ngroups, dbarg, src2arg);
98 else
99 k.args(srcarg, src.cols, (int)src.total(), ngroups, dbarg);
100 }
101
102 size_t globalsize = ngroups * wgs;
103 if (k.run(1, &globalsize, &wgs, true))
104 {
105 typedef Scalar (*part_sum)(Mat m);
106 part_sum funcs[3] = { ocl_part_sum<int>, ocl_part_sum<float>, ocl_part_sum<double> },
107 func = funcs[ddepth - CV_32S];
108
109 Mat mres = db.getMat(ACCESS_READ);
110 if (calc2)
111 const_cast<Scalar &>(res2) = func(mres.colRange(ngroups, dbsize));
112
113 res = func(mres.colRange(0, ngroups));
114 return true;
115 }
116 return false;
117}

121Scalar sum(InputArray _src)
122{
123 CV_INSTRUMENT_REGION();
124
125 Scalar _res;
126#ifdef HAVE_OPENCL
127 CV_OCL_RUN_(OCL_PERFORMANCE_CHECK(_src.isUMat()) && _src.dims() <= 2,
128 ocl_sum(_src, _res, OCL_OP_SUM),
129 _res);
130#endif
131
132 Mat src = _src.getMat();
133 int cn = src.channels();
134 CV_CheckLE( cn, 4, "cv::sum does not support more than 4 channels" );
135
136 if (_src.dims() <= 2)
137 {
138 CALL_HAL_RET2(sum, cv_hal_sum, _res, src.data, src.step, src.type(), src.cols, src.rows, &_res[0]);
139 }
140 else if (_src.isContinuous())
141 {
142 CALL_HAL_RET2(sum, cv_hal_sum, _res, src.data, 0, src.type(), (int)src.total(), 1, &_res[0]);
143 }
144
145 int k, depth = src.depth();
146 SumFunc func = getSumFunc(depth);
147 if (func == nullptr) {
148 if (depth == CV_Bool && cn == 1)
149 return Scalar((double)countNonZero(src));
150 CV_Error(Error::StsNotImplemented, "");
151 }
152
153 const Mat* arrays[] = {&src, 0};
154 uchar* ptrs[1] = {};
155 NAryMatIterator it(arrays, ptrs);
156 int total = (int)it.size, blockSize = total, partialBlockSize = 0;
157 int j, count = 0;
158 int _buf[CV_CN_MAX];
159 int* buf = (int*)&_res[0];
160 size_t esz = 0;
161 bool partialSumIsInt = depth < CV_32S;
162 bool blockSum = partialSumIsInt || depth == CV_16F || depth == CV_16BF;
163
164 if( blockSum )
165 {
166 partialBlockSize = depth <= CV_8S ? (1 << 23) : (1 << 15);
167 blockSize = std::min(blockSize, partialBlockSize);
168 buf = _buf;
169 for( k = 0; k < cn; k++ )
170 buf[k] = 0;
171 esz = src.elemSize();
172 }
173
174 for( size_t i = 0; i < it.nplanes; i++, ++it )
175 {
176 for( j = 0; j < total; j += blockSize )
177 {
178 int bsz = std::min(total - j, blockSize);
179 func( ptrs[0], 0, (uchar*)buf, bsz, cn );
180 count += bsz;
181 if( blockSum && (count + blockSize >= partialBlockSize || (i+1 >= it.nplanes && j+bsz >= total)) )
182 {
183 if (partialSumIsInt) {
184 for( k = 0; k < cn; k++ )
185 {
186 _res[k] += buf[k];
187 buf[k] = 0;
188 }
189 } else {
190 for( k = 0; k < cn; k++ )
191 {
192 _res[k] += ((float*)buf)[k];
193 buf[k] = 0;
194 }
195 }
196 count = 0;
197 }
198 ptrs[0] += bsz*esz;
199 }
200 }
201 return _res;
202}
