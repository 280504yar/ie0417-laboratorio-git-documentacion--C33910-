22static void concat(const std::vector<Mat>& inps, Mat& out, int axis)
23{
24 CV_Assert(out.isContinuous());
25
26 MatShape outShape = out.shape();
27 int ndims = outShape.dims, nslices = 1;
28 size_t esz = out.elemSize();
29 size_t sliceSize = esz;
30 size_t totalSize = 0;
31 size_t outStep = 0;
32 int ninputs = (int)inps.size();
33 for (int i = ndims-1; i > axis; i--)
34 sliceSize *= outShape[i];
35 outStep = sliceSize*outShape[axis];
36 for (int i = 0; i < axis; i++)
37 nslices *= outShape[i];
38 for (int i = 0; i < ninputs; i++) {
39 CV_Assert(inps[i].isContinuous());
40 totalSize += inps[i].total()*esz;
41 }
42
43 // Precompute per-input destination offset and per-slice size.
44 std::vector<size_t> dstOffset(ninputs);
45 std::vector<size_t> sliceSize_k_vec(ninputs);
46 {
47 size_t acc = 0;
48 for (int k = 0; k < ninputs; k++) {
49 int sz_a = inps[k].size[axis];
50 dstOffset[k] = acc;
51 sliceSize_k_vec[k] = sliceSize * sz_a;
52 acc += sliceSize_k_vec[k];
53 }
54 }
55 const size_t CHUNK = 64 * 1024;
56
57 // Precompute per-input chunk counts and a prefix sum for fast index decode.
58 std::vector<int> chunkOff(ninputs + 1, 0);
59 for (int k = 0; k < ninputs; k++)
60 chunkOff[k + 1] = chunkOff[k] + (int)((sliceSize_k_vec[k] + CHUNK - 1) / CHUNK);
61 int chunksPerSlice = chunkOff[ninputs];
62 int totalChunks = chunksPerSlice * nslices;
63
64 if (totalSize > CHUNK && totalChunks > 0) {
65 parallel_for_(Range(0, totalChunks), [&](const Range& r) {
66 for (int c = r.start; c < r.end; c++) {
67 int s = c / chunksPerSlice;
68 int local = c % chunksPerSlice;
69 int k = 0;
70 while (local >= chunkOff[k + 1]) k++;
71 int chunkInK = local - chunkOff[k];
72 size_t byteStart = (size_t)chunkInK * CHUNK;
73 size_t byteEnd = std::min(byteStart + CHUNK, sliceSize_k_vec[k]);
74
75 const uchar* inptr_k = inps[k].data;
76 uchar* outptr = out.data + dstOffset[k];
77 memcpy(outptr + (size_t)s * outStep + byteStart,
78 inptr_k + (size_t)s * sliceSize_k_vec[k] + byteStart,
79 byteEnd - byteStart);
80 }
81 });
82 } else {
83 for (int k = 0; k < ninputs; k++) {
84 const uchar* inptr_k = inps[k].data;
85 uchar* outptr = out.data + dstOffset[k];
86 size_t sliceSize_k = sliceSize_k_vec[k];
87 for (int s = 0; s < nslices; s++)
88 memcpy(outptr + (size_t)s * outStep, inptr_k + (size_t)s * sliceSize_k, sliceSize_k);
89 }
90 }
91}

107 MatShape getOutShape(const std::vector<MatShape>& inpShapes) const
108 {
109 size_t ninputs = inpShapes.size();
110 CV_Assert(ninputs == inputs.size());
111
112 const MatShape& inpShape0 = inpShapes[0];
113 int inpDims = inpShape0.dims;
114 int axis_ = normalize_axis(axis, inpDims);
115 CV_Assert(0 <= axis_ && axis_ < inpDims);
116 MatShape outShape = inpShape0;
117 outShape[axis_] = 0;
118
119 for (size_t i = 0; i < ninputs; i++) {
120 const MatShape& inpShape_i = inpShapes[i];
121 CV_Assert(inpShape_i.dims == inpDims);
122 for (int j = 0; j < inpDims; j++) {
123 if (j == axis_) {
124 outShape[j] += inpShape_i[j];
125 continue;
126 }
127 CV_Assert(inpShape0[j] == inpShape_i[j]);
128 }
129 }
130
131 // BLOCK axis=1: per-input c-block sum overcounts on misaligned inputs
132 // (YOLOX 80+4+1: 12 vs correct ceil(85/8)=11).
133 if (outShape.layout == DATA_LAYOUT_BLOCK && axis_ == 1) {
134 int total_C = 0;
135 for (size_t i = 0; i < ninputs; i++) total_C += inpShapes[i].C;
136 int C0 = outShape[outShape.dims - 1];
137 outShape[1] = (total_C + C0 - 1) / C0;
138 outShape.C = total_C;
139 }
140
141 return outShape;
142 }

170 int getLayouts(const std::vector<DataLayout>& actualInputs,
171 std::vector<DataLayout>& desiredInputs,
172 const int requiredOutputs,
173 std::vector<DataLayout>& outputs) const CV_OVERRIDE
174 {
175 auto* netimpl_ = getNetImpl(this);
176 DataLayout defaultLayout = netimpl_->originalLayout;
177 const size_t ninputs = actualInputs.size();
178 desiredInputs = actualInputs;
179 outputs.assign(requiredOutputs, DATA_LAYOUT_UNKNOWN);
180
181 bool allBlock = ninputs > 0;
182 for (size_t i = 0; i < ninputs; ++i)
183 if (actualInputs[i] != DATA_LAYOUT_BLOCK) { allBlock = false; break; }
184
185 // BLOCK layout on the channel axis would expose inner-block padding as real channels; let TransformLayout repack instead.
186 const bool canKeepBlock = allBlock && axis >= 0;
187
188 if (canKeepBlock) {
189 outputs.assign(requiredOutputs, DATA_LAYOUT_BLOCK);
190 } else {
191 for (size_t i = 0; i < ninputs; ++i)
192 if (actualInputs[i] == DATA_LAYOUT_BLOCK) desiredInputs[i] = defaultLayout;
193 }
194 return outputs[0] == DATA_LAYOUT_BLOCK ? netimpl_->defaultC0 : 0;
195 }

201 void forward(InputArrayOfArrays inputs_arr,
202 OutputArrayOfArrays outputs_arr,
203 OutputArrayOfArrays) CV_OVERRIDE
204 {
205 CV_TRACE_FUNCTION();
206 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
207
208 Size size = inputs_arr.size();
209 int ninputs = size.area();
210
211 CV_Assert(ninputs > 0);
212
213 std::vector<MatShape> inpShapes(ninputs);
214 int inpType = inputs_arr.type(0);
215
216 for (int i = 0; i < ninputs; i++) {
217 inpShapes[i] = inputs_arr.shape(i);
218 CV_Assert(inputs_arr.type(i) == inpType);
219 }
220
221 MatShape outShape = getOutShape(inpShapes);
222 int outKind = outputs_arr.kind();
223 int axis_ = normalize_axis(axis, inpShapes[0].dims);
224
225 CV_Assert(outKind == _InputArray::STD_VECTOR_MAT ||
226 outKind == _InputArray::STD_VECTOR_UMAT);
227
228 if (outKind == _InputArray::STD_VECTOR_MAT) {
229 std::vector<Mat> inps;
230 inputs_arr.getMatVector(inps);
231 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
232 outs.resize(1);
233 outs[0].fit(outShape, inpType);
234 runOp(inps, outs[0], axis_);
235 } else {
236 // [TODO] more efficient OpenCL implementation
237 std::vector<Mat> inps;
238 inputs_arr.getMatVector(inps);
239 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
240 outs.resize(1);
241 outs[0].fit(outShape, inpType);
242 Mat temp(outShape, inpType);
243 runOp(inps, temp, axis_);
244 temp.copyTo(outs[0]);
245 }
246 }

248 void runOp(const std::vector<Mat>& inps, Mat& out, int axis_)
249 {
250 // Byte-level BLOCK concat needs every input's C to be a multiple of
251 // C0; misaligned inputs (YOLOX 80+4+1) route through NCHW.
252 if (axis_ == 1 && out.size.layout == DATA_LAYOUT_BLOCK) {
253 const int C0 = out.size[out.dims - 1];
254 bool allAligned = true;
255 for (const auto& inp : inps) {
256 if (inp.size.layout != DATA_LAYOUT_BLOCK || (inp.size.channels() % C0) != 0) {
257 allAligned = false;
258 break;
259 }
260 }
261 if (!allAligned) {
262 runOpBlockAxis1Misaligned(inps, out);
263 return;
264 }
265 }
266 concat(inps, out, axis_);
267 }
268
269 // Fallback for axis=1 BLOCK concat when inputs aren't C0-aligned.
270 void runOpBlockAxis1Misaligned(const std::vector<Mat>& inps, Mat& out)
271 {
272 auto* netimpl_ = getNetImpl(this);
273 DataLayout origLayout = netimpl_->originalLayout;
274 const int C0 = out.size[out.dims - 1];
275
276 std::vector<Mat> nchwInps(inps.size());
277 for (size_t i = 0; i < inps.size(); i++) {
278 transformLayout(inps[i], nchwInps[i], origLayout, origLayout, C0);
279 }
280
281 // NCHW output shape: drop the trailing C0 dim, use logical C from out.size.C.
282 const int dims = out.dims;
283 std::vector<int> nchwDims(dims - 1);
284 nchwDims[0] = out.size[0];
285 nchwDims[1] = out.size.C;
286 for (int d = 2; d < dims - 1; d++) nchwDims[d] = out.size[d];
287 MatShape nchwShape(nchwDims, origLayout, out.size.C);
288 Mat nchwOut;
289 nchwOut.fit(nchwShape, out.type());
290
291 concat(nchwInps, nchwOut, /*axis=*/1);
292 transformLayout(nchwOut, out, DATA_LAYOUT_BLOCK, origLayout, C0);
293 }
