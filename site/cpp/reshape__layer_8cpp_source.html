64static void computeShapeByReshapeMask(const MatShape &srcShape,
65 const MatShape &maskShape,
66 Range srcRange /*= Range::all()*/,
67 MatShape& dstShape)
68{
69 int srcShapeSize = (int)srcShape.size();
70 int maskShapeSize = (int)maskShape.size();
71
72 srcRange = normalize_axis_range(srcRange, srcShapeSize);
73
74 bool explicitMask = !maskShape.empty(); // All mask values are positive.
75 for (int i = 0, n = maskShape.size(); i < n && explicitMask; ++i)
76 {
77 explicitMask = maskShape[i] > 0;
78 }
79 // Working range of source shape is a range where area(src) == area(mask).
80 if (explicitMask)
81 {
82 int maskTotal = total(maskShape);
83 // Go from the end of mask until we collect required total.
84 bool matched = false;
85 for (int i = srcRange.end - 1; i >= srcRange.start; --i)
86 {
87 if (matched)
88 {
89 if (total(srcShape, i, srcRange.end) != maskTotal)
90 {
91 srcRange.start = i + 1;
92 break;
93 }
94 else if (i == 0)
95 {
96 srcRange.start = 0;
97 break;
98 }
99 }
100 else
101 {
102 matched = total(srcShape, i, srcRange.end) == maskTotal;
103 }
104 }
105 while (total(srcShape, srcRange.start, srcRange.end) != maskTotal && srcRange.start > 0)
106 {
107 srcRange.start -= 1;
108 }
109 CV_Assert(total(srcShape, srcRange.start, srcRange.end) == maskTotal);
110 }
111
112 CV_Assert(0 <= srcRange.start && srcRange.start <= srcRange.end && srcRange.end <= srcShapeSize);
113 int dstShapeSize = srcShapeSize - srcRange.size() + maskShapeSize;
114 dstShape.resize(dstShapeSize);
115
116 std::copy(srcShape.begin(), srcShape.begin() + srcRange.start, dstShape.begin());
117 std::copy(srcShape.begin() + srcRange.end, srcShape.begin() + srcShapeSize, dstShape.begin() + srcRange.start + maskShapeSize);
118
119 int inferDim = -1;
120 for (int i = 0; i < maskShapeSize; i++)
121 {
122 if (maskShape[i] > 0)
123 {
124 dstShape[srcRange.start + i] = maskShape[i];
125 }
126 else if (maskShape[i] == 0)
127 {
128 if (srcRange.start + i >= srcShapeSize)
129 CV_Error(Error::StsBadArg, format("Copy dim[%d] (which has zero size) is out of the source shape bounds", srcRange.start + i));
130 dstShape[srcRange.start + i] = srcShape[srcRange.start + i];
131 }
132 else if (maskShape[i] == -1)
133 {
134 if (inferDim != -1)
135 CV_Error(Error::StsAssert, "Duplicate of inferred dim (which is denoted by -1)");
136 inferDim = srcRange.start + i;
137 dstShape[inferDim] = 1;
138 }
139 else
140 CV_Error(Error::StsBadArg, "maskShape[i] >= -1");
141 }
142
143 size_t srcTotal = total(srcShape);
144 size_t dstTotal = total(dstShape);
145 CV_Assert(dstTotal != 0);
146
147 if (inferDim != -1)
148 {
149 if (srcTotal % dstTotal != 0)
150 CV_Error(Error::StsBackTrace, "Can't infer a dim denoted by -1");
151
152 dstShape[inferDim] = (int)(srcTotal / dstTotal);
153 }
154 else
155 {
156 CV_Assert(srcTotal == dstTotal);
157 }
158}

164 ReshapeLayerImpl(const LayerParams& params)
165 {
166 setParamsFrom(params);
167 axis = params.get<int>("axis", 0);
168 numAxes = params.get<int>("num_axes", -1);
169 hasDynamicShapes = params.get<bool>("has_dynamic_shapes", false);
170 shapesInitialized = !hasDynamicShapes;
171
172 zeropoint = params.get<int>("zeropoints", 0);
173 scale = params.get<float>("scales", 1.0f);
174
175 CV_Assert(numAxes >= -1);
176 newShapeRange = (numAxes == -1) ? Range(axis, INT_MAX) : Range(axis, axis + numAxes);
177
178 newShapeDesc.clear();
179 if (params.has("dim"))
180 {
181 const DictValue &paramShape = params.get("dim");
182 int i, dims = paramShape.size();
183 newShapeDesc.resize(dims);
184 for (i = 0; i < dims; i++)
185 newShapeDesc[i] = paramShape.get<int>(i);
186 }
187 if (params.has("unsqueeze_axes"))
188 {
189 const DictValue& param_unsqueeze_axes = params.get("unsqueeze_axes");
190 int len_axes = param_unsqueeze_axes.size();
191 unsqueeze_axes.resize(len_axes);
192 for (int i = 0; i < len_axes; ++i)
193 {
194 unsqueeze_axes[i] = (int64_t)param_unsqueeze_axes.get<int>(i);
195 }
196 }
197 if (hasDynamicShapes)
198 {
199 dynamicShapes.clear();
200 inputIndices.clear();
201 if (params.has("dynamic_axes")) {
202 CV_Assert(params.has("input_indices"));
203 const DictValue &dynamicAxes = params.get("dynamic_axes");
204 const DictValue &dynamicInputShapes = params.get("input_indices");
205 int i, dims = dynamicAxes.size();
206 CV_Assert(dims == dynamicInputShapes.size());
207 CV_Assert(dims > 0);
208 dynamicShapes.resize(dims);
209 inputIndices.resize(dims);
210 for (i = 0; i < dims; i++) {
211 dynamicShapes[i] = dynamicAxes.get<int>(i);
212 inputIndices[i] = dynamicInputShapes.get<int>(i);
213 }
214 }
215 }

244 bool getMemoryShapes(const std::vector<MatShape> &inputs,
245 const int requiredOutputs,
246 std::vector<MatShape> &outputs,
247 std::vector<MatShape> &internals) const CV_OVERRIDE
248 {
249 if (inputs.size() == 1 || inputs.size() == requiredOutputs)
250 {
251 outputs.clear();
252 for (size_t i = 0; i < inputs.size(); i++)
253 {
254 if (hasDynamicShapes && !shapesInitialized)
255 {
256 outputs.push_back(newShapeDesc);
257 }
258 else
259 {
260 outputs.push_back(MatShape());
261 computeShapeByReshapeMask(inputs[i], newShapeDesc, newShapeRange, outputs.back());
262 }
263 }
264 }
265 else
266 {
267 CV_Assert_N(inputs.size() == 2, total(inputs[0]) == total(inputs[1]));
268 outputs.assign(1, inputs[1]);
269 }
270 return true;
271 }

316 bool forward_ocl(InputArrayOfArrays inps, OutputArrayOfArrays outs, OutputArrayOfArrays internals)
317 {
318 std::vector<UMat> inputs;
319 std::vector<UMat> outputs;
320
321 inps.getUMatVector(inputs);
322 outs.getUMatVector(outputs);
323
324 for (size_t i = 0; i < outputs.size(); i++)
325 {
326 UMat srcBlob = inputs[i];
327 void *src_handle = inputs[i].handle(ACCESS_READ);
328 void *dst_handle = outputs[i].handle(ACCESS_WRITE);
329 if (src_handle != dst_handle)
330 {
331 UMat umat = srcBlob.reshape(1, (int)outShapes[i].size(), &outShapes[i][0]);
332 umat.copyTo(outputs[i]);
333 }
334 }
335 outs.assign(outputs);
336
337 return true;
338 }

340 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
341 {
342 CV_TRACE_FUNCTION();
343 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
344
345 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
346 forward_ocl(inputs_arr, outputs_arr, internals_arr))
347
348 std::vector<Mat> inputs, outputs;
349 inputs_arr.getMatVector(inputs);
350 outputs_arr.getMatVector(outputs);
351 for (size_t i = 0; i < outputs.size(); i++)
352 {
353 Mat srcBlob = inputs[i];
354 if (outputs[i].data != srcBlob.data)
355 srcBlob.reshape(1, shape(outputs[i])).copyTo(outputs[i]);
356 }
357 }

360 virtual Ptr<BackendNode> initCann(const std::vector<Ptr<BackendWrapper> > &inputs,

363 {

372

377 auto input_desc = input_wrapper->getTensorDesc();
378 op->update_input_desc_x(*input_desc);
379
380 // set outputs
381 auto desc_y = std::make_shared<ge::TensorDesc>(ge::Shape(), ge::FORMAT_NCHW, ge::DT_FLOAT);
382 op->update_output_desc_y(*desc_y);
383
384 return Ptr<BackendNode>(new CannBackendNode(op));
385 }
386 else
387 {
388 // create operator
389 auto op = std::make_shared<ge::op::Reshape>(name);
390
391 // set attributes
392 op->set_attr_axis(axis);
393 op->set_attr_num_axes(numAxes);
394
395 // set inputs
396 // set inputs : x
397 auto input_node = nodes[0].dynamicCast<CannBackendNode>()->getOp();
398 op->set_input_x_by_name(*input_node, input_wrapper->name.c_str());
399 auto input_desc = input_wrapper->getTensorDesc();
400 op->update_input_desc_x(*input_desc);
401 // set inputs : shape
402 std::vector<int> shape_of_shape{(int)newShapeDesc.size()};
403 Mat shape_mat(shape_of_shape, CV_32S, newShapeDesc.data());
404 auto op_const_shape = std::make_shared<CannConstOp>(shape_mat.data, shape_mat.type(), shape_of_shape, cv::format("%s_shape", name.c_str()));
405 op->set_input_shape(*(op_const_shape->getOp()));
406 op->update_input_desc_shape(*(op_const_shape->getTensorDesc()));
407
408 // set outputs
409 auto desc_y = std::make_shared<ge::TensorDesc>(ge::Shape(), ge::FORMAT_NCHW, ge::DT_FLOAT);
410 op->update_output_desc_y(*desc_y);
411
412 return Ptr<BackendNode>(new CannBackendNode(op));

459 virtual Ptr<BackendNode> initTimVX(void* timVXInfo_,
460 const std::vector<Ptr<BackendWrapper> > &inputsWrapper,
461 const std::vector<Ptr<BackendWrapper> > &outputsWrapper,
462 bool isLast) CV_OVERRIDE
463 {
464#ifdef HAVE_TIMVX
465 // tvGraph Initialization.
466 auto timVxInfo = reinterpret_cast<TimVXInfo *>(timVXInfo_);
467 CV_Assert(timVxInfo);
468 Ptr<TimVXGraph> tvGraph = timVxInfo->getGraph();
469 CV_Assert(tvGraph);
470 Ptr<tim::vx::Graph> graph = tvGraph->graph;
471
472 std::vector<int> inputsIndex, outputsIndex;
473 int input_index = -1, output_index = -1;
474
475 int reshapeNum = 0;
476 Ptr<TimVXBackendWrapper> tmpWrapper, inputWrapper, outputWrapper;
477 for (size_t i = 0; i < outputsWrapper.size(); i++)
478 {
479 tmpWrapper = inputsWrapper[i].dynamicCast<TimVXBackendWrapper>();
480 Mat srcBlob = tmpWrapper->getMat();
481
482 tmpWrapper = outputsWrapper[i].dynamicCast<TimVXBackendWrapper>();
483 Mat dstBlob = tmpWrapper->getMat();
484 if (dstBlob.data != srcBlob.data)
485 {
486 reshapeNum++;
487 inputWrapper = inputsWrapper[i].dynamicCast<TimVXBackendWrapper>();
488 outputWrapper = outputsWrapper[i].dynamicCast<TimVXBackendWrapper>();
489 }
490 }
491
492 // Only work for single reshape Mat
493 if (reshapeNum != 1)
494 {
495 return Ptr<BackendNode>();
496 }
497
498 // Input
499 if (inputWrapper->isTensor())
500 {
501 input_index = tvGraph->getTensorIndex(inputWrapper->getTensor());
502 if (input_index == -1)
503 {
504 // Copy To New inputWrapper
505 Mat tmp = inputWrapper->getMat();
506 inputWrapper = Ptr<TimVXBackendWrapper>(new TimVXBackendWrapper(tmp));
507 }
508 }
509
510 if (!inputWrapper->isTensor() || input_index == -1)
511 {
512 Ptr<tim::vx::Quantization> tvInputQuant = Ptr<tim::vx::Quantization>(
513 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, scale, zeropoint));
514 inputWrapper->createTensor(graph,tim::vx::TensorAttribute::INPUT,tvInputQuant);
515 input_index = tvGraph->addWrapper(inputWrapper);
516 }
517 inputsIndex.push_back(input_index);
518
519 //Output
520 // Output Tensor has the same quantized attrib as Input Tesor.
521 Ptr<tim::vx::Quantization> outputQuant = inputWrapper->getTensorQuantization();
522 if (isLast)
523 {
524 auto shapeType = getShapeTypeFromMat(outputWrapper->getMat());
525
526 // For Graph Output tensor, we need to set tensor shape before createTensor().
527 outputWrapper->setTensorShape(shapeType);
528 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::OUTPUT, outputQuant);
529 }
530 else
531 {
532 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::TRANSIENT, outputQuant);
533 }
534 output_index = tvGraph->addWrapper(outputWrapper);
535 outputsIndex.push_back(output_index);
536
537 // generate output shape.
538 MatShape outputShape = shape(outputWrapper->getMat());
539 // reverse shape, from NCHW to WHCN
540 std::reverse(outputShape.begin(), outputShape.end());
541 std::vector<uint32_t> tvShape(outputShape.begin(), outputShape.end());
542
543 std::shared_ptr<tim::vx::Operation> tvReshape = graph->CreateOperation<tim::vx::ops::Reshape>(tvShape);
544
545 Ptr<TimVXBackendNode> tvBackendNode = new TimVXBackendNode(tvGraph, tvReshape, inputsIndex, outputsIndex);
546
547 return tvBackendNode;
548#endif // HAVE_TIMVX
549 return Ptr<BackendNode>();
550 }

558 bool hasDynamicShapes;
