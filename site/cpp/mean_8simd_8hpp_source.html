30template <>
31struct SumSqr_SIMD<uchar, int, int>
32{
33 int operator () (const uchar * src0, const uchar * mask, int * sum, int * sqsum, int len, int cn) const
34 {
35 if (mask)
36 return 0;
37 // cn==3: deinterleave so each lane belongs to one channel, then reduce per
38 // channel. sum/sqsum stay exact in s32 within the dispatcher's 1<<15 block.
39 if (cn == 3)
40 {
41 const int vl8 = VTraits<v_uint8>::vlanes();
42 v_int32 vs0 = vx_setzero_s32(), vs1 = vx_setzero_s32(), vs2 = vx_setzero_s32();
43 v_int32 vq0 = vx_setzero_s32(), vq1 = vx_setzero_s32(), vq2 = vx_setzero_s32();
44 auto acc = [](const v_uint8& ch, v_int32& vs, v_int32& vq)
45 {
46 v_uint16 lo, hi; v_expand(ch, lo, hi);
47 v_uint32 s0, s1, s2, s3; v_expand(lo, s0, s1); v_expand(hi, s2, s3);
48 vs = v_add(vs, v_reinterpret_as_s32(v_add(v_add(s0, s1), v_add(s2, s3))));
49 v_int16 d0 = v_reinterpret_as_s16(lo), d1 = v_reinterpret_as_s16(hi);
50 vq = v_add(vq, v_add(v_dotprod(d0, d0), v_dotprod(d1, d1)));
51 };
52 int e = 0;
53 for (; e <= len - vl8; e += vl8)
54 {
55 v_uint8 a, b, c; v_load_deinterleave(src0 + e * 3, a, b, c);
56 acc(a, vs0, vq0); acc(b, vs1, vq1); acc(c, vs2, vq2);
57 }
58 sum[0] += v_reduce_sum(vs0); sum[1] += v_reduce_sum(vs1); sum[2] += v_reduce_sum(vs2);
59 sqsum[0] += v_reduce_sum(vq0); sqsum[1] += v_reduce_sum(vq1); sqsum[2] += v_reduce_sum(vq2);
60 v_cleanup();
61 return e;
62 }
63 if (cn != 1 && cn != 2 && cn != 4)
64 return 0;
65 len *= cn;
66
67 int x = 0;
68 v_int32 v_sum = vx_setzero_s32();
69 v_int32 v_sqsum = vx_setzero_s32();
70
71 const int len0 = len & -VTraits<v_uint8>::vlanes();
72 while(x < len0)
73 {
74 const int len_tmp = min(x + 256*VTraits<v_uint16>::vlanes(), len0);
75 v_uint16 v_sum16 = vx_setzero_u16();
76 for ( ; x < len_tmp; x += VTraits<v_uint8>::vlanes())
77 {
78 v_uint16 v_src0 = vx_load_expand(src0 + x);
79 v_uint16 v_src1 = vx_load_expand(src0 + x + VTraits<v_uint16>::vlanes());
80 v_sum16 = v_add(v_sum16, v_add(v_src0, v_src1));
81 v_int16 v_tmp0, v_tmp1;
82 v_zip(v_reinterpret_as_s16(v_src0), v_reinterpret_as_s16(v_src1), v_tmp0, v_tmp1);
83 v_sqsum = v_add(v_sqsum, v_add(v_dotprod(v_tmp0, v_tmp0), v_dotprod(v_tmp1, v_tmp1)));
84 }
85 v_uint32 v_half0, v_half1;
86 v_expand(v_sum16, v_half0, v_half1);
87 v_sum = v_add(v_sum, v_reinterpret_as_s32(v_add(v_half0, v_half1)));
88 }
89 if (x <= len - VTraits<v_uint16>::vlanes())
90 {
91 v_uint16 v_src = vx_load_expand(src0 + x);
92 v_uint16 v_half = v_combine_high(v_src, v_src);
93
94 v_uint32 v_tmp0, v_tmp1;
95 v_expand(v_add(v_src, v_half), v_tmp0, v_tmp1);
96 v_sum = v_add(v_sum, v_reinterpret_as_s32(v_tmp0));
97
98 v_int16 v_tmp2, v_tmp3;
99 v_zip(v_reinterpret_as_s16(v_src), v_reinterpret_as_s16(v_half), v_tmp2, v_tmp3);
100 v_sqsum = v_add(v_sqsum, v_dotprod(v_tmp2, v_tmp2));
101 x += VTraits<v_uint16>::vlanes();
102 }
103
104 if (cn == 1)
105 {
106 *sum += v_reduce_sum(v_sum);
107 *sqsum += v_reduce_sum(v_sqsum);
108 }
109 else
110 {
111 int CV_DECL_ALIGNED(CV_SIMD_WIDTH) ar[2 * VTraits<v_int32>::max_nlanes];
112 v_store(ar, v_sum);
113 v_store(ar + VTraits<v_int32>::vlanes(), v_sqsum);
114 for (int i = 0; i < VTraits<v_int32>::vlanes(); ++i)
115 {
116 sum[i % cn] += ar[i];
117 sqsum[i % cn] += ar[VTraits<v_int32>::vlanes() + i];
118 }
119 }
120 v_cleanup();
121 return x / cn;
122 }
123};

219// Expand a 16-bit vector into two s32 halves. The unsigned overload reinterprets
220// after a widening expand (values 0..65535 stay non-negative in s32, so the
221// subsequent v_cvt_f64 is exact); the signed overload expands directly. This is
222// the only real difference between the ushort and short reductions below.
223static inline void sumSqrExpandS32(const v_uint16& v, v_int32& lo, v_int32& hi)
224{
225 v_uint32 a, b; v_expand(v, a, b);
226 lo = v_reinterpret_as_s32(a); hi = v_reinterpret_as_s32(b);
227}
228static inline void sumSqrExpandS32(const v_int16& v, v_int32& lo, v_int32& hi)
229{
230 v_expand(v, lo, hi);
231}

233// Shared 16-bit -> {s32 sum, f64 sqsum} reduction for ushort/short. The register
234// type is deduced from the pointer, so the body is identical for both depths.
235template <typename T>
236static inline int sumSqr16ToF64(const T* src0, int* sum, double* sqsum, int len, int cn)
237{
238 typedef decltype(vx_load(src0)) VT; // v_uint16 or v_int16
239
240 if (cn == 3)
241 {
242 const int vl = VTraits<VT>::vlanes();
243 v_int32 vs0 = vx_setzero_s32(), vs1 = vx_setzero_s32(), vs2 = vx_setzero_s32();
244 v_float64 vq0 = vx_setzero_f64(), vq1 = vx_setzero_f64(), vq2 = vx_setzero_f64();
245 auto acc = [](const VT& ch, v_int32& vs, v_float64& vq)
246 {
247 v_int32 lo, hi; sumSqrExpandS32(ch, lo, hi);
248 vs = v_add(vs, v_add(lo, hi));
249 v_float64 f0 = v_cvt_f64(lo), f1 = v_cvt_f64_high(lo);
250 v_float64 f2 = v_cvt_f64(hi), f3 = v_cvt_f64_high(hi);
251 vq = v_fma(f0, f0, v_fma(f1, f1, v_fma(f2, f2, v_fma(f3, f3, vq))));
252 };
253 int e = 0;
254 for (; e <= len - vl; e += vl)
255 {
256 VT a, b, c; v_load_deinterleave(src0 + e * 3, a, b, c);
257 acc(a, vs0, vq0); acc(b, vs1, vq1); acc(c, vs2, vq2);
258 }
259 sum[0] += v_reduce_sum(vs0); sum[1] += v_reduce_sum(vs1); sum[2] += v_reduce_sum(vs2);
260 sqsum[0] += v_reduce_sum(vq0); sqsum[1] += v_reduce_sum(vq1); sqsum[2] += v_reduce_sum(vq2);
261 v_cleanup();
262 return e;
263 }
264 if (cn != 1 && cn != 2 && cn != 4)
265 return 0;
266 len *= cn;
267
268 const int vl16 = VTraits<VT>::vlanes();
269 const int vl32 = VTraits<v_int32>::vlanes();
270 const int vl64 = VTraits<v_float64>::vlanes();
271
272 // The lane->channel scatter below (i % cn) and the returned pixel count
273 // (x / cn) require the lane counts to be a multiple of cn. This always holds
274 // for fixed-width SIMD (vlanes is a power of two >= cn for cn in {1,2,4}),
275 // but scalable backends (e.g. RVV) may report a non-multiple; fall back to
276 // scalar there. vl16 == 2*vl32, so checking vl32 covers both scatter loops.
277 if (vl32 % cn != 0)
278 return 0;
279
280 v_int32 v_sum = vx_setzero_s32();
281 v_float64 v_sq0 = vx_setzero_f64(), v_sq1 = vx_setzero_f64();
282 v_float64 v_sq2 = vx_setzero_f64(), v_sq3 = vx_setzero_f64();
283
284 int x = 0;
285 for (; x <= len - vl16; x += vl16)
286 {
287 VT v_src = vx_load(src0 + x);
288 v_int32 lo, hi; sumSqrExpandS32(v_src, lo, hi);
289 v_sum = v_add(v_sum, v_add(lo, hi));
290
291 v_float64 f0 = v_cvt_f64(lo);
292 v_float64 f1 = v_cvt_f64_high(lo);
293 v_float64 f2 = v_cvt_f64(hi);
294 v_float64 f3 = v_cvt_f64_high(hi);
295 v_sq0 = v_fma(f0, f0, v_sq0);
296 v_sq1 = v_fma(f1, f1, v_sq1);
297 v_sq2 = v_fma(f2, f2, v_sq2);
298 v_sq3 = v_fma(f3, f3, v_sq3);
299 }
300
301 int CV_DECL_ALIGNED(CV_SIMD_WIDTH) sbuf[VTraits<v_int32>::max_nlanes];
302 double CV_DECL_ALIGNED(CV_SIMD_WIDTH) qbuf[VTraits<v_float64>::max_nlanes * 4];
303 v_store_aligned(sbuf, v_sum);
304 v_store_aligned(qbuf + vl64 * 0, v_sq0);
305 v_store_aligned(qbuf + vl64 * 1, v_sq1);
306 v_store_aligned(qbuf + vl64 * 2, v_sq2);
307 v_store_aligned(qbuf + vl64 * 3, v_sq3);
308
309 // sbuf lane j (j<vl32) holds samples j and j+vl32 -> channel j%cn (vl32%cn==0).
310 // qbuf index i (i<vl16) holds sample lane i -> channel i%cn (vl16%cn==0).
311 for (int i = 0; i < vl32; ++i)
312 sum[i % cn] += sbuf[i];
313 for (int i = 0; i < vl16; ++i)
314 sqsum[i % cn] += qbuf[i];
315
316 v_cleanup();
317 return x / cn;
318}

320// Shared 32-bit -> f64 reduction for int/float. Both sum and sqsum accumulate in
321// f64 to match the scalar reference's double accumulation; values are widened to
322// f64 before squaring so the square is computed in double (same as (double)v*v).
323// The register type is deduced from the pointer, so int and float share one body.
324template <typename T>
325static inline int sumSqr32ToF64(const T* src0, double* sum, double* sqsum, int len, int cn)
326{
327 typedef decltype(vx_load(src0)) VT; // v_int32 or v_float32
328
329 if (cn == 3)
330 {
331 const int vl = VTraits<VT>::vlanes();
332 v_float64 vs0 = vx_setzero_f64(), vs1 = vx_setzero_f64(), vs2 = vx_setzero_f64();
333 v_float64 vq0 = vx_setzero_f64(), vq1 = vx_setzero_f64(), vq2 = vx_setzero_f64();
334 auto acc = [](const VT& ch, v_float64& vs, v_float64& vq)
335 {
336 v_float64 f0 = v_cvt_f64(ch), f1 = v_cvt_f64_high(ch);
337 vs = v_add(vs, v_add(f0, f1));
338 vq = v_fma(f0, f0, v_fma(f1, f1, vq));
339 };
340 int e = 0;
341 for (; e <= len - vl; e += vl)
342 {
343 VT a, b, c; v_load_deinterleave(src0 + e * 3, a, b, c);
344 acc(a, vs0, vq0); acc(b, vs1, vq1); acc(c, vs2, vq2);
345 }
346 sum[0] += v_reduce_sum(vs0); sum[1] += v_reduce_sum(vs1); sum[2] += v_reduce_sum(vs2);
347 sqsum[0] += v_reduce_sum(vq0); sqsum[1] += v_reduce_sum(vq1); sqsum[2] += v_reduce_sum(vq2);
348 v_cleanup();
349 return e;
350 }
351 if (cn != 1 && cn != 2 && cn != 4)
352 return 0;
353 len *= cn;
354
355 const int vl32 = VTraits<VT>::vlanes();
356 const int vl64 = VTraits<v_float64>::vlanes();
357
358 // See note in sumSqr16ToF64: guard scalable backends whose lane count may
359 // not divide cn (the i % cn scatter / x / cn rely on it).
360 if (vl32 % cn != 0)
361 return 0;
362
363 v_float64 vs0 = vx_setzero_f64(), vs1 = vx_setzero_f64();
364 v_float64 vq0 = vx_setzero_f64(), vq1 = vx_setzero_f64();
365
366 int x = 0;
367 for (; x <= len - vl32; x += vl32)
368 {
369 VT v_src = vx_load(src0 + x);
370 v_float64 f0 = v_cvt_f64(v_src);
371 v_float64 f1 = v_cvt_f64_high(v_src);
372 vs0 = v_add(vs0, f0);
373 vs1 = v_add(vs1, f1);
374 vq0 = v_fma(f0, f0, vq0);
375 vq1 = v_fma(f1, f1, vq1);
376 }
377
378 double CV_DECL_ALIGNED(CV_SIMD_WIDTH) sbuf[VTraits<v_float64>::max_nlanes * 2];
379 double CV_DECL_ALIGNED(CV_SIMD_WIDTH) qbuf[VTraits<v_float64>::max_nlanes * 2];
380 v_store_aligned(sbuf, vs0); v_store_aligned(sbuf + vl64, vs1);
381 v_store_aligned(qbuf, vq0); v_store_aligned(qbuf + vl64, vq1);
382
383 for (int i = 0; i < vl32; ++i)
384 {
385 sum[i % cn] += sbuf[i];
386 sqsum[i % cn] += qbuf[i];
387 }
388
389 v_cleanup();
390 return x / cn;
391}

425template<typename T, typename ST, typename SQT>
426static int sumsqr_(const T* src0, const uchar* mask, ST* sum, SQT* sqsum, int len, int cn )
427{
428 const T* src = src0;
429
430 if( !mask )
431 {
432 SumSqr_SIMD<T, ST, SQT> vop;
433 int x = vop(src0, mask, sum, sqsum, len, cn), k = cn % 4;
434 src = src0 + x * cn;
435
436 if( k == 1 )
437 {
438 ST s0 = sum[0];
439 SQT sq0 = sqsum[0];
440 for(int i = x; i < len; i++, src += cn )
441 {
442 ST v = (ST)src[0];
443 s0 += v; sq0 += (SQT)v*v;
444 }
445 sum[0] = s0;
446 sqsum[0] = sq0;
447 }
448 else if( k == 2 )
449 {
450 ST s0 = sum[0], s1 = sum[1];
451 SQT sq0 = sqsum[0], sq1 = sqsum[1];
452 for(int i = x; i < len; i++, src += cn )
453 {
454 ST v0 = (ST)src[0], v1 = (ST)src[1];
455 s0 += v0; sq0 += (SQT)v0*v0;
456 s1 += v1; sq1 += (SQT)v1*v1;
457 }
458 sum[0] = s0; sum[1] = s1;
459 sqsum[0] = sq0; sqsum[1] = sq1;
460 }
461 else if( k == 3 )
462 {
463 ST s0 = sum[0], s1 = sum[1], s2 = sum[2];
464 SQT sq0 = sqsum[0], sq1 = sqsum[1], sq2 = sqsum[2];
465 for(int i = x; i < len; i++, src += cn )
466 {
467 ST v0 = (ST)src[0], v1 = (ST)src[1], v2 = (ST)src[2];
468 s0 += v0; sq0 += (SQT)v0*v0;
469 s1 += v1; sq1 += (SQT)v1*v1;
470 s2 += v2; sq2 += (SQT)v2*v2;
471 }
472 sum[0] = s0; sum[1] = s1; sum[2] = s2;
473 sqsum[0] = sq0; sqsum[1] = sq1; sqsum[2] = sq2;
474 }
475
476 for( ; k < cn; k += 4 )
477 {
478 src = src0 + x * cn + k;
479 ST s0 = sum[k], s1 = sum[k+1], s2 = sum[k+2], s3 = sum[k+3];
480 SQT sq0 = sqsum[k], sq1 = sqsum[k+1], sq2 = sqsum[k+2], sq3 = sqsum[k+3];
481 for(int i = x; i < len; i++, src += cn )
482 {
483 ST v0, v1;
484 v0 = (ST)src[0], v1 = (ST)src[1];
485 s0 += v0; sq0 += (SQT)v0*v0;
486 s1 += v1; sq1 += (SQT)v1*v1;
487 ST v2, v3;
