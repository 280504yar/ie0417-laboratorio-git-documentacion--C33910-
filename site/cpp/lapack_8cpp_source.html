96template<typename _Tp> static inline _Tp hypot(_Tp a, _Tp b)
97{
98 a = std::abs(a);
99 b = std::abs(b);
100 if( a > b )
101 {
102 b /= a;
103 return a*std::sqrt(1 + b*b);
104 }
105 if( b > 0 )
106 {
107 a /= b;
108 return b*std::sqrt(1 + a*a);
109 }
110 return 0;
111}

114template<typename _Tp> bool
115JacobiImpl_( _Tp* A, size_t astep, _Tp* W, _Tp* V, size_t vstep, int n, uchar* buf )
116{
117 const _Tp eps = std::numeric_limits<_Tp>::epsilon();
118 int i, j, k, m;
119
120 astep /= sizeof(A[0]);
121 if( V )
122 {
123 vstep /= sizeof(V[0]);
124 for( i = 0; i < n; i++ )
125 {
126 for( j = 0; j < n; j++ )
127 V[i*vstep + j] = (_Tp)0;
128 V[i*vstep + i] = (_Tp)1;
129 }
130 }
131
132 int iters, maxIters = n*n*30;
133
134 int* indR = (int*)alignPtr(buf, sizeof(int));
135 int* indC = indR + n;
136 _Tp mv = (_Tp)0;
137
138 for( k = 0; k < n; k++ )
139 {
140 W[k] = A[(astep + 1)*k];
141 if( k < n - 1 )
142 {
143 for( m = k+1, mv = std::abs(A[astep*k + m]), i = k+2; i < n; i++ )
144 {
145 _Tp val = std::abs(A[astep*k+i]);
146 if( mv < val )
147 mv = val, m = i;
148 }
149 indR[k] = m;
150 }
151 if( k > 0 )
152 {
153 for( m = 0, mv = std::abs(A[k]), i = 1; i < k; i++ )
154 {
155 _Tp val = std::abs(A[astep*i+k]);
156 if( mv < val )
157 mv = val, m = i;
158 }
159 indC[k] = m;
160 }
161 }
162
163 if( n > 1 ) for( iters = 0; iters < maxIters; iters++ )
164 {
165 // find index (k,l) of pivot p
166 for( k = 0, mv = std::abs(A[indR[0]]), i = 1; i < n-1; i++ )
167 {
168 _Tp val = std::abs(A[astep*i + indR[i]]);
169 if( mv < val )
170 mv = val, k = i;
171 }
172 int l = indR[k];
173 for( i = 1; i < n; i++ )
174 {
175 _Tp val = std::abs(A[astep*indC[i] + i]);
176 if( mv < val )
177 mv = val, k = indC[i], l = i;
178 }
179
180 _Tp p = A[astep*k + l];
181 if( std::abs(p) <= eps )
182 break;
183 _Tp y = (_Tp)((W[l] - W[k])*0.5);
184 _Tp t = std::abs(y) + hypot(p, y);
185 _Tp s = hypot(p, t);
186 _Tp c = t/s;
187 s = p/s; t = (p/t)*p;
188 if( y < 0 )
189 s = -s, t = -t;
190 A[astep*k + l] = 0;
191
192 W[k] -= t;
193 W[l] += t;
194
195 _Tp a0, b0;
196
197#undef rotate
198#define rotate(v0, v1) a0 = v0, b0 = v1, v0 = a0*c - b0*s, v1 = a0*s + b0*c
199
200 // rotate rows and columns k and l
201 for( i = 0; i < k; i++ )
202 rotate(A[astep*i+k], A[astep*i+l]);
203 for( i = k+1; i < l; i++ )
204 rotate(A[astep*k+i], A[astep*i+l]);
205 for( i = l+1; i < n; i++ )
206 rotate(A[astep*k+i], A[astep*l+i]);
207
208 // rotate eigenvectors
209 if( V )
210 for( i = 0; i < n; i++ )
211 rotate(V[vstep*k+i], V[vstep*l+i]);
212
213#undef rotate
214
215 for( j = 0; j < 2; j++ )
216 {
217 int idx = j == 0 ? k : l;
218 if( idx < n - 1 )
219 {
220 for( m = idx+1, mv = std::abs(A[astep*idx + m]), i = idx+2; i < n; i++ )
221 {
222 _Tp val = std::abs(A[astep*idx+i]);
223 if( mv < val )
224 mv = val, m = i;
225 }
226 indR[idx] = m;
227 }
228 if( idx > 0 )
229 {
230 for( m = 0, mv = std::abs(A[idx]), i = 1; i < idx; i++ )
231 {
232 _Tp val = std::abs(A[astep*i+idx]);
233 if( mv < val )
234 mv = val, m = i;
235 }
236 indC[idx] = m;
237 }
238 }
239 }
240
241 // sort eigenvalues & eigenvectors
242 for( k = 0; k < n-1; k++ )
243 {
244 m = k;
245 for( i = k+1; i < n; i++ )
246 {
247 if( W[m] < W[i] )
248 m = i;
249 }
250 if( k != m )
251 {
252 std::swap(W[m], W[k]);
253 if( V )
254 for( i = 0; i < n; i++ )
255 std::swap(V[vstep*m + i], V[vstep*k + i]);
256 }
257 }
258
259 return true;
260}

300{
301 if( n < VTraits<v_float32>::vlanes())
302 return 0;
303 int k = 0;
304 v_float32 c4 = vx_setall_f32(c), s4 = vx_setall_f32(s);

309 v_float32 b0 = vx_load(b + k);

312 v_store(a + k, t0);
313 v_store(b + k, t1);
314 }
315 vx_cleanup();
316 return k;

362template<typename _Tp> void
363JacobiSVDImpl_(_Tp* At, size_t astep, _Tp* _W, _Tp* Vt, size_t vstep,
364 int m, int n, int n1, double minval, _Tp eps)
365{
366 VBLAS<_Tp> vblas;
367 AutoBuffer<double> Wbuf(n);
368 double* W = Wbuf.data();
369 int i, j, k, iter, max_iter = std::max(m, 30);
370 _Tp c, s;
371 double sd;
372 astep /= sizeof(At[0]);
373 vstep /= sizeof(Vt[0]);
374
375 for( i = 0; i < n; i++ )
376 {
377 for( k = 0, sd = 0; k < m; k++ )
378 {
379 _Tp t = At[i*astep + k];
380 sd += (double)t*t;
381 }
382 W[i] = sd;
383
384 if( Vt )
385 {
386 for( k = 0; k < n; k++ )
387 Vt[i*vstep + k] = 0;
388 Vt[i*vstep + i] = 1;
389 }
390 }
391
392 for( iter = 0; iter < max_iter; iter++ )
393 {
394 bool changed = false;
395
396 for( i = 0; i < n-1; i++ )
397 for( j = i+1; j < n; j++ )
398 {
399 _Tp *Ai = At + i*astep, *Aj = At + j*astep;
400 double a = W[i], p = 0, b = W[j];
401
402 for( k = 0; k < m; k++ )
403 p += (double)Ai[k]*Aj[k];
404
405 if( std::abs(p) <= eps*std::sqrt((double)a*b) )
406 continue;
407
408 p *= 2;
409 double beta = a - b, gamma = hypot((double)p, beta);
410 if( beta < 0 )
411 {
412 double delta = (gamma - beta)*0.5;
413 s = (_Tp)std::sqrt(delta/gamma);
414 c = (_Tp)(p/(gamma*s*2));
415 }
416 else
417 {
418 c = (_Tp)std::sqrt((gamma + beta)/(gamma*2));
419 s = (_Tp)(p/(gamma*c*2));
420 }
421
422 a = b = 0;
423 for( k = 0; k < m; k++ )
424 {
425 _Tp t0 = c*Ai[k] + s*Aj[k];
426 _Tp t1 = -s*Ai[k] + c*Aj[k];
427 Ai[k] = t0; Aj[k] = t1;
428
429 a += (double)t0*t0; b += (double)t1*t1;
430 }
431 W[i] = a; W[j] = b;
432
433 changed = true;
434
435 if( Vt )
436 {
437 _Tp *Vi = Vt + i*vstep, *Vj = Vt + j*vstep;
438 k = vblas.givens(Vi, Vj, n, c, s);
439
440 for( ; k < n; k++ )
441 {
442 _Tp t0 = c*Vi[k] + s*Vj[k];
443 _Tp t1 = -s*Vi[k] + c*Vj[k];
444 Vi[k] = t0; Vj[k] = t1;
445 }
446 }
447 }
448 if( !changed )
449 break;
450 }
451
452 for( i = 0; i < n; i++ )
453 {
454 for( k = 0, sd = 0; k < m; k++ )
455 {
456 _Tp t = At[i*astep + k];
457 sd += (double)t*t;
458 }
459 W[i] = std::sqrt(sd);
460 }
461
462 for( i = 0; i < n-1; i++ )
463 {
464 j = i;
465 for( k = i+1; k < n; k++ )
466 {
467 if( W[j] < W[k] )
468 j = k;
469 }
470 if( i != j )
471 {
472 std::swap(W[i], W[j]);
473 if( Vt )
474 {
475 for( k = 0; k < m; k++ )
476 std::swap(At[i*astep + k], At[j*astep + k]);
477
478 for( k = 0; k < n; k++ )
479 std::swap(Vt[i*vstep + k], Vt[j*vstep + k]);
480 }
481 }
482 }
483
484 for( i = 0; i < n; i++ )
485 _W[i] = (_Tp)W[i];
486
487 if( !Vt )
488 return;
489
490 RNG rng(0x12345678);
491 for( i = 0; i < n1; i++ )
492 {
493 sd = i < n ? W[i] : 0;
494
495 for( int ii = 0; ii < 100 && sd <= minval; ii++ )
496 {
497 // if we got a zero singular value, then in order to get the corresponding left singular vector
498 // we generate a random vector, project it to the previously computed left singular vectors,
499 // subtract the projection and normalize the difference.
500 const _Tp val0 = (_Tp)(1./m);
501 for( k = 0; k < m; k++ )
502 {
503 _Tp val = (rng.next() & 256) != 0 ? val0 : -val0;
504 At[i*astep + k] = val;
505 }
506 for( iter = 0; iter < 2; iter++ )
507 {
508 for( j = 0; j < i; j++ )
509 {
510 sd = 0;
511 for( k = 0; k < m; k++ )
512 sd += At[i*astep + k]*At[j*astep + k];
513 _Tp asum = 0;
514 for( k = 0; k < m; k++ )
515 {
516 _Tp t = (_Tp)(At[i*astep + k] - sd*At[j*astep + k]);
517 At[i*astep + k] = t;
518 asum += std::abs(t);
519 }
520 asum = asum > eps*100 ? 1/asum : 0;
521 for( k = 0; k < m; k++ )
522 At[i*astep + k] *= asum;
523 }
524 }
525 sd = 0;
526 for( k = 0; k < m; k++ )
527 {
528 _Tp t = At[i*astep + k];
529 sd += (double)t*t;
530 }
531 sd = std::sqrt(sd);
532 }
533
534 s = (_Tp)(sd > minval ? 1/sd : 0.);
535 for( k = 0; k < m; k++ )
536 At[i*astep + k] *= s;
537 }
538}

612template<typename T> static void
613SVBkSbImpl_( int m, int n, const T* w, int incw,
614 const T* u, int ldu, bool uT,
615 const T* v, int ldv, bool vT,
616 const T* b, int ldb, int nb,
617 T* x, int ldx, double* buffer, T eps )
618{
619 double threshold = 0;
620 int udelta0 = uT ? ldu : 1, udelta1 = uT ? 1 : ldu;
621 int vdelta0 = vT ? ldv : 1, vdelta1 = vT ? 1 : ldv;
622 int i, j, nm = std::min(m, n);
623
624 if( !b )
625 nb = m;
626
627 for( i = 0; i < n; i++ )
628 for( j = 0; j < nb; j++ )
629 x[i*ldx + j] = 0;
630
631 for( i = 0; i < nm; i++ )
632 threshold += w[i*incw];
633 threshold *= eps;
634
635 // v * inv(w) * uT * b
636 for( i = 0; i < nm; i++, u += udelta0, v += vdelta0 )
637 {
638 double wi = w[i*incw];
639 if( (double)std::abs(wi) <= threshold )
640 continue;
641 wi = 1/wi;
642
643 if( nb == 1 )
644 {
645 double s = 0;
646 if( b )
647 for( j = 0; j < m; j++ )
648 s += u[j*udelta1]*b[j*ldb];
649 else
650 s = u[0];
651 s *= wi;
652
653 for( j = 0; j < n; j++ )
654 x[j*ldx] = (T)(x[j*ldx] + s*v[j*vdelta1]);
655 }
656 else
657 {

706/****************************************************************************************\
707* Determinant of the matrix *
708\****************************************************************************************/
709
710#define det2(m) ((double)m(0,0)*m(1,1) - (double)m(0,1)*m(1,0))
711#define det3(m) (m(0,0)*((double)m(1,1)*m(2,2) - (double)m(1,2)*m(2,1)) - \
712 m(0,1)*((double)m(1,0)*m(2,2) - (double)m(1,2)*m(2,0)) + \
713 m(0,2)*((double)m(1,0)*m(2,1) - (double)m(1,1)*m(2,0)))
714
715double determinant( InputArray _mat )
716{
717 CV_INSTRUMENT_REGION();
718
719 Mat mat = _mat.getMat();
720 double result = 0;
721 int type = mat.type(), rows = mat.rows;
722 size_t step = mat.step;
723 const uchar* m = mat.ptr();
724
725 CV_Assert( !mat.empty() );
726 CV_Assert( mat.rows == mat.cols && (type == CV_32F || type == CV_64F));
727
728 #define Mf(y, x) ((float*)(m + y*step))[x]
729 #define Md(y, x) ((double*)(m + y*step))[x]
730
731 if( type == CV_32F )
732 {
733 if( rows == 2 )
734 result = det2(Mf);
735 else if( rows == 3 )
736 result = det3(Mf);
737 else if( rows == 1 )
738 result = Mf(0,0);
739 else
740 {
741 size_t bufSize = rows*rows*sizeof(float);
742 AutoBuffer<uchar> buffer(bufSize);
743 Mat a(rows, rows, CV_32F, buffer.data());
744 mat.copyTo(a);
745
746 result = hal::LU32f(a.ptr<float>(), a.step, rows, 0, 0, 0);
747 if( result )
748 {
749 for( int i = 0; i < rows; i++ )
750 result *= a.at<float>(i,i);
751 }
752 }
753 }
754 else
755 {
756 if( rows == 2 )
757 result = det2(Md);
758 else if( rows == 3 )
759 result = det3(Md);
760 else if( rows == 1 )
761 result = Md(0,0);
762 else
763 {
764 size_t bufSize = rows*rows*sizeof(double);
765 AutoBuffer<uchar> buffer(bufSize);
766 Mat a(rows, rows, CV_64F, buffer.data());
767 mat.copyTo(a);
768
769 result = hal::LU64f(a.ptr<double>(), a.step, rows, 0, 0, 0);
770 if( result )
771 {
772 for( int i = 0; i < rows; i++ )
773 result *= a.at<double>(i,i);
774 }
775 }
776 }
777
778 #undef Mf
779 #undef Md
780
781 return result;
782}

784/****************************************************************************************\
785* Inverse (or pseudo-inverse) of a matrix *
786\****************************************************************************************/
787
788#define Sf( y, x ) ((float*)(srcdata + y*srcstep))[x]
789#define Sd( y, x ) ((double*)(srcdata + y*srcstep))[x]
790#define Df( y, x ) ((float*)(dstdata + y*dststep))[x]
791#define Dd( y, x ) ((double*)(dstdata + y*dststep))[x]
792
793double invert( InputArray _src, OutputArray _dst, int method )
794{
795 CV_INSTRUMENT_REGION();
796
797 bool result = false;
798 Mat src = _src.getMat();
799 int type = src.type();
800
801 CV_Assert(type == CV_32F || type == CV_64F);
802
803 size_t esz = CV_ELEM_SIZE(type);
804 int m = src.rows, n = src.cols;
805
806 if( method == DECOMP_SVD )
807 {
808 int nm = std::min(m, n);
809
810 AutoBuffer<uchar> _buf((m*nm + nm + nm*n)*esz + sizeof(double));
811 uchar* buf = alignPtr((uchar*)_buf.data(), (int)esz);
812 Mat u(m, nm, type, buf);
813 Mat w(nm, 1, type, u.ptr() + m*nm*esz);
814 Mat vt(nm, n, type, w.ptr() + nm*esz);
815
816 SVD::compute(src, w, u, vt);
817 SVD::backSubst(w, u, vt, Mat(), _dst);
818 return type == CV_32F ?
819 (w.ptr<float>()[0] >= FLT_EPSILON ?
820 w.ptr<float>()[n-1]/w.ptr<float>()[0] : 0) :
821 (w.ptr<double>()[0] >= DBL_EPSILON ?
822 w.ptr<double>()[n-1]/w.ptr<double>()[0] : 0);
823 }
824
825 CV_Assert( m == n );
826
827 if( method == DECOMP_EIG )
828 {
829 AutoBuffer<uchar> _buf((n*n*2 + n)*esz + sizeof(double));
830 uchar* buf = alignPtr((uchar*)_buf.data(), (int)esz);
831 Mat u(n, n, type, buf);
832 Mat w(n, 1, type, u.ptr() + n*n*esz);
833 Mat vt(n, n, type, w.ptr() + n*esz);
834
835 eigen(src, w, vt);
836 transpose(vt, u);
837 SVD::backSubst(w, u, vt, Mat(), _dst);
838 return type == CV_32F ?
839 (w.ptr<float>()[0] >= FLT_EPSILON ?
840 w.ptr<float>()[n-1]/w.ptr<float>()[0] : 0) :
841 (w.ptr<double>()[0] >= DBL_EPSILON ?
842 w.ptr<double>()[n-1]/w.ptr<double>()[0] : 0);
843 }
844
845 CV_Assert( method == DECOMP_LU || method == DECOMP_CHOLESKY );
846
847 _dst.create( n, n, type );
848 Mat dst = _dst.getMat();
849
850 if( n <= 3 )
851 {
852 const uchar* srcdata = src.ptr();
853 uchar* dstdata = dst.ptr();
854 size_t srcstep = src.step;
855 size_t dststep = dst.step;
856
857 if( n == 2 )
858 {
859 if( type == CV_32FC1 )
860 {
861 double d = det2(Sf);
862 if( d != 0. )
863 {
864 result = true;
865 d = 1./d;
866 #if CV_SIMD128
867 const float d_32f = (float)d;
868 const v_float32x4 d_vec(d_32f, -d_32f, -d_32f, d_32f);
869 v_float32x4 s0 = v_mul(v_load_halves((const float *)srcdata, (const float *)(srcdata + srcstep)), d_vec);//0123//3120
870 s0 = v_extract<3>(s0, v_combine_low(v_rotate_right<1>(s0), s0));
871 v_store_low((float*)dstdata, s0);
872 v_store_high((float*)(dstdata + dststep), s0);
873 #else
874 double t0, t1;
875 t0 = Sf(0,0)*d;
876 t1 = Sf(1,1)*d;
877 Df(1,1) = (float)t0;
878 Df(0,0) = (float)t1;
879 t0 = -Sf(0,1)*d;
880 t1 = -Sf(1,0)*d;
881 Df(0,1) = (float)t0;
882 Df(1,0) = (float)t1;
883 #endif
884 }
885 }
886 else
887 {
888 double d = det2(Sd);
889 if( d != 0. )
890 {
891 result = true;
892 d = 1./d;
893 #if CV_SIMD128_64F
894 v_float64x2 det = v_setall_f64(d);
895 v_float64x2 s0 = v_mul(v_load((const double *)srcdata), det);
896 v_float64x2 s1 = v_mul(v_load((const double *)(srcdata + srcstep)), det);
897 v_float64x2 sm = v_extract<1>(s1, s0);//30
898 v_float64x2 ss = v_sub(v_setall<double>(0), v_extract<1>(s0, s1));//12
899 v_store((double*)dstdata, v_combine_low(sm, ss));//31
900 v_store((double*)(dstdata + dststep), v_combine_high(ss, sm));//20
901 #else
902 double t0, t1;
903 t0 = Sd(0,0)*d;
904 t1 = Sd(1,1)*d;
905 Dd(1,1) = t0;
906 Dd(0,0) = t1;
907 t0 = -Sd(0,1)*d;
908 t1 = -Sd(1,0)*d;
909 Dd(0,1) = t0;
910 Dd(1,0) = t1;
911 #endif
912 }
913 }
914 }
915 else if( n == 3 )
916 {
917 if( type == CV_32FC1 )
918 {
919 double d = det3(Sf);
920
921 if( d != 0. )
922 {
923 double t[12];
924
925 result = true;
926 d = 1./d;
927 t[0] = (((double)Sf(1,1) * Sf(2,2) - (double)Sf(1,2) * Sf(2,1)) * d);
928 t[1] = (((double)Sf(0,2) * Sf(2,1) - (double)Sf(0,1) * Sf(2,2)) * d);
929 t[2] = (((double)Sf(0,1) * Sf(1,2) - (double)Sf(0,2) * Sf(1,1)) * d);
930
931 t[3] = (((double)Sf(1,2) * Sf(2,0) - (double)Sf(1,0) * Sf(2,2)) * d);
932 t[4] = (((double)Sf(0,0) * Sf(2,2) - (double)Sf(0,2) * Sf(2,0)) * d);
933 t[5] = (((double)Sf(0,2) * Sf(1,0) - (double)Sf(0,0) * Sf(1,2)) * d);
934
935 t[6] = (((double)Sf(1,0) * Sf(2,1) - (double)Sf(1,1) * Sf(2,0)) * d);
936 t[7] = (((double)Sf(0,1) * Sf(2,0) - (double)Sf(0,0) * Sf(2,1)) * d);
937 t[8] = (((double)Sf(0,0) * Sf(1,1) - (double)Sf(0,1) * Sf(1,0)) * d);
938
939 Df(0,0) = (float)t[0]; Df(0,1) = (float)t[1]; Df(0,2) = (float)t[2];
940 Df(1,0) = (float)t[3]; Df(1,1) = (float)t[4]; Df(1,2) = (float)t[5];
941 Df(2,0) = (float)t[6]; Df(2,1) = (float)t[7]; Df(2,2) = (float)t[8];
942 }
943 }
944 else
945 {
946 double d = det3(Sd);
947 if( d != 0. )
948 {
949 result = true;
950 d = 1./d;
951 double t[9];
952
953 t[0] = (Sd(1,1) * Sd(2,2) - Sd(1,2) * Sd(2,1)) * d;
954 t[1] = (Sd(0,2) * Sd(2,1) - Sd(0,1) * Sd(2,2)) * d;
955 t[2] = (Sd(0,1) * Sd(1,2) - Sd(0,2) * Sd(1,1)) * d;
956
957 t[3] = (Sd(1,2) * Sd(2,0) - Sd(1,0) * Sd(2,2)) * d;
958 t[4] = (Sd(0,0) * Sd(2,2) - Sd(0,2) * Sd(2,0)) * d;
959 t[5] = (Sd(0,2) * Sd(1,0) - Sd(0,0) * Sd(1,2)) * d;
960
961 t[6] = (Sd(1,0) * Sd(2,1) - Sd(1,1) * Sd(2,0)) * d;
962 t[7] = (Sd(0,1) * Sd(2,0) - Sd(0,0) * Sd(2,1)) * d;
963 t[8] = (Sd(0,0) * Sd(1,1) - Sd(0,1) * Sd(1,0)) * d;
964
965 Dd(0,0) = t[0]; Dd(0,1) = t[1]; Dd(0,2) = t[2];
966 Dd(1,0) = t[3]; Dd(1,1) = t[4]; Dd(1,2) = t[5];
967 Dd(2,0) = t[6]; Dd(2,1) = t[7]; Dd(2,2) = t[8];
968 }
969 }
970 }
971 else
972 {
973 CV_Assert( n == 1 );
974
975 if( type == CV_32FC1 )
976 {
977 double d = Sf(0,0);
978 if( d != 0. )
979 {
980 result = true;
981 Df(0,0) = (float)(1./d);
982 }
983 }
984 else
985 {
986 double d = Sd(0,0);
987 if( d != 0. )
988 {
989 result = true;
990 Dd(0,0) = 1./d;
991 }
992 }
993 }
994 if( !result )
995 dst = Scalar(0);
996 return result;
997 }
998
999 int elem_size = CV_ELEM_SIZE(type);
1000 AutoBuffer<uchar> buf(n*n*elem_size);
1001 Mat src1(n, n, type, buf.data());
1002 src.copyTo(src1);
1003 setIdentity(dst);
1004
1005 if( method == DECOMP_LU && type == CV_32F )
1006 result = hal::LU32f(src1.ptr<float>(), src1.step, n, dst.ptr<float>(), dst.step, n) != 0;
1007 else if( method == DECOMP_LU && type == CV_64F )
1008 result = hal::LU64f(src1.ptr<double>(), src1.step, n, dst.ptr<double>(), dst.step, n) != 0;
1009 else if( method == DECOMP_CHOLESKY && type == CV_32F )
1010 result = hal::Cholesky32f(src1.ptr<float>(), src1.step, n, dst.ptr<float>(), dst.step, n);
1011 else
1012 result = hal::Cholesky64f(src1.ptr<double>(), src1.step, n, dst.ptr<double>(), dst.step, n);
1013
1014 if( !result )
1015 dst = Scalar(0);
1016
1017 return result;
1018}
