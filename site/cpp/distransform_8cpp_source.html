51static void
52initTopBottom( Mat& temp, int border, unsigned int value )
53{
54 Size size = temp.size();
55 unsigned int* ttop = (unsigned int*)temp.ptr<int>(0);
56 unsigned int* tbottom = (unsigned int*)temp.ptr<int>(size.height - 1);
57 for( int i = 0; i < border; i++ )
58 {
59 for( int j = 0; j < size.width; j++ )
60 {
61 ttop[j] = value;
62 tbottom[j] = value;
63 }
64 ttop += size.width;
65 tbottom -= size.width;
66 }
67}

70// Scale one row of fixed-point distances to the float output (3x3/5x5 backward pass).
71static inline void
72writeDistRowF32( const unsigned int* cur, float* drow, int width, float scale )
73{
74#if defined(CV_SIMD)
75 int j = 0;
76 const uint32_t* tptr = (const uint32_t*)cur;
77 float* dptr = drow;
78 const v_float32 scale_v = vx_setall<v_float32>(scale);
79
80 for (; j <= width - 8; j += 8)
81 {
82 v_uint32 v0 = vx_load(tptr + j);
83 v_uint32 v1 = vx_load(tptr + j + 4);
84 v_float32 f0 = v_mul(v_cvt_f32(v0), scale_v);
85 v_float32 f1 = v_mul(v_cvt_f32(v1), scale_v);
86 v_store(dptr + j, f0);
87 v_store(dptr + j + 4, f1);
88 }
89
90 for (; j < width; j++)
91 dptr[j] = (float)(tptr[j] * scale);
92#else
93 for (int j = 0; j < width; j++)
94 drow[j] = (float)(cur[j] * scale);
95#endif
96}

99// Scalar forward-pass step for the 3x3 chamfer transform (SIMD remainder tail + non-SIMD build).
100static inline void
101dtForwardScalar3x3( unsigned int* cur, const uchar* s, const unsigned int* top,
102 int j0, int width, unsigned int& left,
103 unsigned int HV_DIST, unsigned int DIAG_DIST, unsigned int DIST_MAX )
104{
105 for (int j = j0; j < width; j++)
106 {
107 if (!s[j])
108 {
109 cur[j] = 0;
110 left = 0;
111 }
112 else
113 {
114 unsigned int t0 = top[j - 1] + DIAG_DIST;
115 unsigned int t1 = top[j] + HV_DIST;
116 unsigned int t2 = top[j + 1] + DIAG_DIST;
117 unsigned int t3 = left + HV_DIST;
118
119 unsigned int m = t0 < t1 ? t0 : t1;
120 m = m < t2 ? m : t2;
121 m = m < t3 ? m : t3;
122
123 cur[j] = m > DIST_MAX ? DIST_MAX : m;
124 left = cur[j];
125 }
126 }
127}

130// Scalar forward-pass step for the 5x5 chamfer transform (SIMD remainder tail + non-SIMD build).
131static inline void
132dtForwardScalar5x5( unsigned int* cur, const uchar* s,
133 const unsigned int* top1, const unsigned int* top2,
134 int j0, int width, unsigned int& left,
135 unsigned int HV_DIST, unsigned int DIAG_DIST,
136 unsigned int LONG_DIST, unsigned int DIST_MAX )
137{
138 for (int j = j0; j < width; j++)
139 {
140 if (!s[j])
141 {
142 cur[j] = 0;
143 left = 0;
144 }
145 else
146 {
147 unsigned int t0 = top2[j - 1] + LONG_DIST;
148 unsigned int t = top2[j + 1] + LONG_DIST;
149 if (t < t0) t0 = t;
150 t = top1[j - 2] + LONG_DIST;
151 if (t < t0) t0 = t;
152 t = top1[j - 1] + DIAG_DIST;
153 if (t < t0) t0 = t;
154 t = top1[j] + HV_DIST;
155 if (t < t0) t0 = t;
156 t = top1[j + 1] + DIAG_DIST;
157 if (t < t0) t0 = t;
158 t = top1[j + 2] + LONG_DIST;
159 if (t < t0) t0 = t;
160 t = left + HV_DIST;
161 if (t < t0) t0 = t;
162 cur[j] = t0 > DIST_MAX ? DIST_MAX : t0;
163 left = cur[j];
164 }
165 }

169static void
170distanceTransform_3x3( const Mat& _src, Mat& _temp, Mat& _dist, const float* metrics )
171{
172 const int BORDER = 1;
173
174 const unsigned int HV_DIST = CV_FLT_TO_FIX( metrics[0], DIST_SHIFT );
175 const unsigned int DIAG_DIST = CV_FLT_TO_FIX( metrics[1], DIST_SHIFT );
176 const unsigned int DIST_MAX = UINT_MAX - DIAG_DIST;
177 const float scale = 1.f/(1 << DIST_SHIFT);
178
179 const uchar* src = _src.ptr();
180 int* temp = _temp.ptr<int>();
181 float* dist = _dist.ptr<float>(_dist.rows - 1);
182
183 const int srcstep = (int)_src.step;
184 const int step = (int)(_temp.step / sizeof(int));
185 const int dststep = (int)(_dist.step / sizeof(float));
186
187 const int width = _src.cols;
188 const int height = _src.rows;
189
190 initTopBottom( _temp, BORDER, DIST_MAX );
191
192 // forward pass
193 unsigned int* row = (unsigned int*)(temp + BORDER * step) + BORDER;
194 const uchar* srow = src;
195
196 for (int i = 0; i < height; i++)
197 {
198 unsigned int* cur = row;
199 const uchar* s = srow;
200 const unsigned int* top = (const unsigned int*)(cur - step);
201
202 // set horizontal border sentinels
203 cur[-1] = DIST_MAX;
204 cur[width] = DIST_MAX;
205
206 unsigned int left = cur[-1];
207
208#if defined(CV_SIMD)
209 const int BLOCK = 8;
210 int j = 0;
211 const v_uint32 v_diag = vx_setall<v_uint32>(DIAG_DIST);
212 const v_uint32 v_hv = vx_setall<v_uint32>(HV_DIST);
213
214 for (; j <= width - BLOCK; j += BLOCK)
215 {
216 v_uint32 tl0 = vx_load(top + j - 1);
217 v_uint32 t0 = vx_load(top + j);
218 v_uint32 tr0 = vx_load(top + j + 1);
219 v_uint32 tl1 = vx_load(top + j + 3);
220 v_uint32 t1 = vx_load(top + j + 4);
221 v_uint32 tr1 = vx_load(top + j + 5);
222 tl0 = v_add(tl0, v_diag);
223 t0 = v_add(t0, v_hv);
224 tr0 = v_add(tr0, v_diag);
225 tl1 = v_add(tl1, v_diag);
226 t1 = v_add(t1, v_hv);
227 tr1 = v_add(tr1, v_diag);
228 v_uint32 m0 = v_min(tl0, t0);
229 m0 = v_min(m0, tr0);
230 v_uint32 m1 = v_min(tl1, t1);
231 m1 = v_min(m1, tr1);
232 v_store(cur + j, m0);
233 v_store(cur + j + 4, m1);
234
235 for (int k = 0; k < BLOCK; k++)
236 {
237 int idx = j + k;
238 if (!s[idx])
239 {
240 cur[idx] = 0;
241 left = 0;
242 }
243 else
244 {
245 unsigned int m = cur[idx];
246 unsigned int lm = left + HV_DIST;
247 if (lm < m) m = lm;
248 if (m > DIST_MAX) m = DIST_MAX;
249 cur[idx] = m;
250 left = m;
251 }
252 }
253 }
254
255 dtForwardScalar3x3(cur, s, top, j, width, left, HV_DIST, DIAG_DIST, DIST_MAX);
256#else
257 dtForwardScalar3x3(cur, s, top, 0, width, left, HV_DIST, DIAG_DIST, DIST_MAX);
258#endif
259
260 row += step;
261 srow += srcstep;
262 }
263
264 // backward pass
265 float* drow = dist;
266 unsigned int* cur = row - step;
267 for (int i = height - 1; i >= 0; i--)
268 {
269 unsigned int* bottom = cur + step;
270
271 unsigned int right = cur[width];
272 for (int j = width - 1; j >= 0; j--)
273 {
274 unsigned int t0 = cur[j];
275
276 if (t0 > HV_DIST)
277 {
278 unsigned int b1 = bottom[j + 1] + DIAG_DIST;
279 unsigned int b2 = bottom[j] + HV_DIST;
280 unsigned int b3 = bottom[j - 1] + DIAG_DIST;
281 unsigned int r = right + HV_DIST;
282
283 unsigned int t = b1 < b2 ? b1 : b2;
284 t = t < b3 ? t : b3;
285 t = t < r ? t : r;
286
287 if (t < t0)
288 {
289 t0 = t;
290 cur[j] = t0;
291 }
292 }
293
294 right = cur[j];
295 }
296
297 writeDistRowF32(cur, drow, width, scale);
298
299 cur -= step;
300 drow -= dststep;
301 }
302}

305static void
306distanceTransform_5x5( const Mat& _src, Mat& _temp, Mat& _dist, const float* metrics )
307{
308 const int BORDER = 2;
309
310 const unsigned int HV_DIST = CV_FLT_TO_FIX( metrics[0], DIST_SHIFT );
311 const unsigned int DIAG_DIST = CV_FLT_TO_FIX( metrics[1], DIST_SHIFT );
312 const unsigned int LONG_DIST = CV_FLT_TO_FIX( metrics[2], DIST_SHIFT );
313 const unsigned int DIST_MAX = UINT_MAX - LONG_DIST;
314 const float scale = 1.f/(1 << DIST_SHIFT);
315
316 const uchar* src = _src.ptr<uchar>();
317 int* temp = _temp.ptr<int>();
318 float* dist = _dist.ptr<float>(_dist.rows - 1);
319
320 const int srcstep = (int)_src.step;
321 const int step = (int)(_temp.step / sizeof(int));
322 const int dststep = (int)(_dist.step / sizeof(float));
323
324 const int width = _src.cols;
325 const int height = _src.rows;
326
327 initTopBottom(_temp, BORDER, DIST_MAX);
328
329 // forward pass
330 unsigned int* row = (unsigned int*)(temp + BORDER * step) + BORDER;
331 const uchar* srow = src;
332
333 for (int i = 0; i < height; i++)
334 {
335 unsigned int* cur = row;
336 const uchar* s = srow;
337 const unsigned int* top1 = cur - step;
338 const unsigned int* top2 = cur - step * 2;
339 cur[-1] = cur[-2] = DIST_MAX;
340 cur[width] = cur[width + 1] = DIST_MAX;
341#if defined(CV_SIMD)
342 unsigned int left1 = DIST_MAX;
343 const int BLOCK = 8;
344 int j = 0;
345 const v_uint32 v_hv = vx_setall<v_uint32>(HV_DIST);
346 const v_uint32 v_diag = vx_setall<v_uint32>(DIAG_DIST);
347 const v_uint32 v_long = vx_setall<v_uint32>(LONG_DIST);
348 const v_uint32 v_max = vx_setall<v_uint32>(DIST_MAX);
349
350 for (; j <= width - BLOCK; j += BLOCK)
351 {
352 v_uint32 t2m1_0 = vx_load(top2 + j - 1);
353 v_uint32 t2p1_0 = vx_load(top2 + j + 1);
354 v_uint32 t1m2_0 = vx_load(top1 + j - 2);
355 v_uint32 t1m1_0 = vx_load(top1 + j - 1);
356 v_uint32 t1_0 = vx_load(top1 + j);
357 v_uint32 t1p1_0 = vx_load(top1 + j + 1);
358 v_uint32 t1p2_0 = vx_load(top1 + j + 2);
359
360 t2m1_0 = v_add(t2m1_0, v_long);
361 t2p1_0 = v_add(t2p1_0, v_long);
362 t1m2_0 = v_add(t1m2_0, v_long);
363 t1m1_0 = v_add(t1m1_0, v_diag);
364 t1_0 = v_add(t1_0, v_hv);
365 t1p1_0 = v_add(t1p1_0, v_diag);
366 t1p2_0 = v_add(t1p2_0, v_long);
367
368 v_uint32 m0 = v_min(t2m1_0, t2p1_0);
369 v_uint32 m0_1 = v_min(t1m2_0, t1m1_0);
370 v_uint32 m0_2 = v_min(t1p1_0, t1p2_0);
371 m0 = v_min(m0, m0_1);
372 m0_2 = v_min(m0_2, t1_0);
373 m0 = v_min(m0, m0_2);
374 m0 = v_min(m0, v_max);
375
376 v_store(cur + j, m0);
377
378 v_uint32 t2m1_1 = vx_load(top2 + j + 3);
379 v_uint32 t2p1_1 = vx_load(top2 + j + 5);
380 v_uint32 t1m2_1 = vx_load(top1 + j + 2);
381 v_uint32 t1m1_1 = vx_load(top1 + j + 3);
382 v_uint32 t1_1 = vx_load(top1 + j + 4);
383 v_uint32 t1p1_1 = vx_load(top1 + j + 5);
384 v_uint32 t1p2_1 = vx_load(top1 + j + 6);
385
386 t2m1_1 = v_add(t2m1_1, v_long);
387 t2p1_1 = v_add(t2p1_1, v_long);
388 t1m2_1 = v_add(t1m2_1, v_long);
389 t1m1_1 = v_add(t1m1_1, v_diag);
390 t1_1 = v_add(t1_1, v_hv);
391 t1p1_1 = v_add(t1p1_1, v_diag);
392 t1p2_1 = v_add(t1p2_1, v_long);
393
394 v_uint32 m1 = v_min(t2m1_1, t2p1_1);
395 v_uint32 m1_1 = v_min(t1m2_1, t1m1_1);
396 v_uint32 m1_2 = v_min(t1p1_1, t1p2_1);
397 m1 = v_min(m1, m1_1);
398 m1_2 = v_min(m1_2, t1_1);
399 m1 = v_min(m1, m1_2);
400 m1 = v_min(m1, v_max);
401
402 v_store(cur + j + 4, m1);
403
404 for (int k = 0; k < BLOCK; k++)
405 {
406 int idx = j + k;
407 if (!s[idx])
408 {
409 cur[idx] = 0;
410 left1 = 0;
411 }
412 else
413 {
414 unsigned int m = cur[idx];
415 unsigned int lm = left1 + HV_DIST;
416 if (lm < m) m = lm;
417 if (m > DIST_MAX) m = DIST_MAX;
418 cur[idx] = m;
419 left1 = m;
420 }
421 }
422 }
423
424 dtForwardScalar5x5(cur, s, top1, top2, j, width, left1, HV_DIST, DIAG_DIST, LONG_DIST, DIST_MAX);
425#else
426 unsigned int left = DIST_MAX;
427 dtForwardScalar5x5(cur, s, top1, top2, 0, width, left, HV_DIST, DIAG_DIST, LONG_DIST, DIST_MAX);
428#endif
429 row += step;
430 srow += srcstep;
431 }
432
433 // backward pass
434 float* drow = dist;
435 unsigned int* cur = row - step;
436
437 for (int i = height - 1; i >= 0; i--)
438 {
439 unsigned int* bot1 = cur + step;
440 unsigned int* bot2 = cur + step * 2;
441 unsigned int right = DIST_MAX;
442
443 for (int j = width - 1; j >= 0; j--)
444 {
445 unsigned int t0 = cur[j];
446 if( t0 > HV_DIST )
447 {
448 unsigned int t = bot2[j + 1] + LONG_DIST;
449 if (t < t0) t0 = t;
450 t = bot2[j - 1] + LONG_DIST;
451 if (t < t0) t0 = t;
452 t = bot1[j + 2] + LONG_DIST;
453 if (t < t0) t0 = t;
454 t = bot1[j + 1] + DIAG_DIST;
455 if (t < t0) t0 = t;
456 t = bot1[j] + HV_DIST;
457 if (t < t0) t0 = t;
458 t = bot1[j - 1] + DIAG_DIST;
459 if (t < t0) t0 = t;
460 t = bot1[j - 2] + LONG_DIST;
461 if (t < t0) t0 = t;
462 t = right + HV_DIST;
463 if (t < t0) t0 = t;
464
465 cur[j] = t0;
466 }
467 right = cur[j];
468 }
469
470 writeDistRowF32(cur, drow, width, scale);
471
472 cur -= step;
473 drow -= dststep;
474 }
475}

478static void
479distanceTransformEx_5x5( const Mat& _src, Mat& _temp, Mat& _dist, Mat& _labels, const float* metrics )
480{
481 const int BORDER = 2;
482
483 int i, j;
484 const unsigned int HV_DIST = CV_FLT_TO_FIX( metrics[0], DIST_SHIFT );
485 const unsigned int DIAG_DIST = CV_FLT_TO_FIX( metrics[1], DIST_SHIFT );
486 const unsigned int LONG_DIST = CV_FLT_TO_FIX( metrics[2], DIST_SHIFT );
487 const unsigned int DIST_MAX = UINT_MAX - LONG_DIST;
488 const float scale = 1.f/(1 << DIST_SHIFT);
489
490 const uchar* src = _src.ptr();
491 int* temp = _temp.ptr<int>();
492 float* dist = _dist.ptr<float>(_dist.rows - 1);
493 int* labels = _labels.ptr<int>();
494 int srcstep = (int)(_src.step/sizeof(src[0]));
495 int step = (int)(_temp.step/sizeof(temp[0]));
496 int dststep = (int)(_dist.step/sizeof(dist[0]));
497 int lstep = (int)(_labels.step/sizeof(labels[0]));
498 Size size = _src.size();
499
500 initTopBottom( _temp, BORDER, DIST_MAX );
501
502 // forward pass
503 const uchar* s = src;
504 unsigned int* tmp = (unsigned int*)(temp + BORDER*step) + BORDER;
505 int* lls = (int*)labels;
506 for( i = 0; i < size.height; i++ )
507 {
508 for( j = 0; j < BORDER; j++ )
509 tmp[-j-1] = tmp[size.width + j] = DIST_MAX;
510
511 for( j = 0; j < size.width; j++ )
512 {
513 if( !s[j] )
514 {
515 tmp[j] = 0;
516 //CV_Assert( lls[j] != 0 );
517 }
518 else
519 {
520 unsigned int t0 = DIST_MAX, t;
521 int l0 = 0;
522
523 t = tmp[j-step*2-1] + LONG_DIST;
524 if( t0 > t )
525 {
526 t0 = t;
527 l0 = lls[j-lstep*2-1];
528 }
529 t = tmp[j-step*2+1] + LONG_DIST;
530 if( t0 > t )
531 {
532 t0 = t;
533 l0 = lls[j-lstep*2+1];
534 }
535 t = tmp[j-step-2] + LONG_DIST;
536 if( t0 > t )
537 {
538 t0 = t;
539 l0 = lls[j-lstep-2];
540 }
541 t = tmp[j-step-1] + DIAG_DIST;
542 if( t0 > t )
543 {
544 t0 = t;
545 l0 = lls[j-lstep-1];
546 }
547 t = tmp[j-step] + HV_DIST;
548 if( t0 > t )
549 {
550 t0 = t;
551 l0 = lls[j-lstep];
552 }
553 t = tmp[j-step+1] + DIAG_DIST;
554 if( t0 > t )
555 {
556 t0 = t;
557 l0 = lls[j-lstep+1];
558 }
559 t = tmp[j-step+2] + LONG_DIST;
560 if( t0 > t )
561 {
562 t0 = t;
563 l0 = lls[j-lstep+2];
564 }
565 t = tmp[j-1] + HV_DIST;
566 if( t0 > t )
567 {
568 t0 = t;
569 l0 = lls[j-1];
570 }
571
572 tmp[j] = t0;
573 lls[j] = l0;
574 }
575 }
576 s += srcstep;
577 tmp += step;
578 lls += lstep;
579 }
580
581 // backward pass
582 float* d = (float*)dist;
583 for( i = size.height - 1; i >= 0; i-- )
584 {
585 tmp -= step;
586 lls -= lstep;
587
588 for( j = size.width - 1; j >= 0; j-- )
589 {
590 unsigned int t0 = tmp[j];
591 int l0 = lls[j];
592 if( t0 > HV_DIST )
593 {
594 unsigned int t = tmp[j+step*2+1] + LONG_DIST;
595 if( t0 > t )
596 {
597 t0 = t;
598 l0 = lls[j+lstep*2+1];
599 }
600 t = tmp[j+step*2-1] + LONG_DIST;
601 if( t0 > t )
602 {
603 t0 = t;
604 l0 = lls[j+lstep*2-1];
605 }
606 t = tmp[j+step+2] + LONG_DIST;
607 if( t0 > t )
608 {
609 t0 = t;
610 l0 = lls[j+lstep+2];
611 }
612 t = tmp[j+step+1] + DIAG_DIST;
613 if( t0 > t )
614 {
615 t0 = t;
616 l0 = lls[j+lstep+1];
617 }
618 t = tmp[j+step] + HV_DIST;
619 if( t0 > t )
620 {
621 t0 = t;
622 l0 = lls[j+lstep];
623 }
624 t = tmp[j+step-1] + DIAG_DIST;
625 if( t0 > t )
626 {
627 t0 = t;
628 l0 = lls[j+lstep-1];
629 }
630 t = tmp[j+step-2] + LONG_DIST;
631 if( t0 > t )
632 {
633 t0 = t;
634 l0 = lls[j+lstep-2];
635 }
636 t = tmp[j+1] + HV_DIST;
637 if( t0 > t )
638 {
639 t0 = t;
640 l0 = lls[j+1];
641 }
642 tmp[j] = t0;
643 lls[j] = l0;
644 }
645 d[j] = (float)(t0 * scale);
646 }
647 d -= dststep;
648 }

652static void getDistanceTransformMask( int maskType, float *metrics )
653{
654 CV_Assert( metrics != 0 );
655
656 switch (maskType)
657 {
658 case 30:
659 metrics[0] = 1.0f;
660 metrics[1] = 1.0f;
661 break;
662
663 case 31:
664 metrics[0] = 1.0f;
665 metrics[1] = 2.0f;
666 break;
667
668 case 32:
669 metrics[0] = 0.955f;
670 metrics[1] = 1.3693f;
671 break;
672
673 case 50:
674 metrics[0] = 1.0f;
675 metrics[1] = 1.0f;
676 metrics[2] = 2.0f;
677 break;

741static const int PRECISE_DIST_MAX = 1 << 16;
742
743struct DTRowInvoker : ParallelLoopBody
744{
745 DTRowInvoker( Mat* _dst, const unsigned int* _sqr_tab, const float* _inv_tab )
746 {
747 dst = _dst;
748 sqr_tab = _sqr_tab;
749 inv_tab = _inv_tab;
750 }
751
752 void operator()(const Range& range) const CV_OVERRIDE
753 {
754 const float inf = 1e15f;
755 int i, i1 = range.start, i2 = range.end;
756 int n = dst->cols;
757 AutoBuffer<uchar> _buf((n+2)*2*sizeof(float) + (n+2)*sizeof(int));
758 float* f = (float*)_buf.data();
759 float* z = f + n;
760 int* v = alignPtr((int*)(z + n + 1), sizeof(int));
761
762 for( i = i1; i < i2; i++ )
763 {
764 float* d = dst->ptr<float>(i);
765 int p, q, k;
766
767 v[0] = 0;
768 z[0] = -inf;
769 z[1] = inf;
770 f[0] = d[0];
771
772 for( q = 1, k = 0; q < std::min(PRECISE_DIST_MAX, n); q++ )
773 {
774 float fq = d[q];
775 f[q] = fq;
776
777 for(;;k--)
778 {
779 p = v[k];
780 float s = (fq - d[p] + (sqr_tab[q]-sqr_tab[p]))*inv_tab[q - p];
781 if( s > z[k] )
782 {
783 k++;
784 v[k] = q;
785 z[k] = s;
786 z[k+1] = inf;
787 break;
788 }
789 }
790 }
791 for(; q < n; q++ )
792 {
793 float fq = d[q];
794 f[q] = fq;
795
796 for(;;k--)
797 {
798 p = v[k];
799 float s = (fq - d[p] + static_cast<float>(q + p) * (q - p))*inv_tab[q - p];
800 if( s > z[k] )
801 {
802 k++;
803 v[k] = q;
804 z[k] = s;
805 z[k+1] = inf;
806 break;
807 }
808 }
809 }
810
811 for( q = 0, k = 0; q < n; q++ )
812 {
813 while( z[k+1] < q )
814 k++;
815 p = v[k];
816 d[q] = std::sqrt(sqr_tab[std::abs(q - p)] + f[p]);
817 }
818 }
819 }
820
821 Mat* dst;
822 const unsigned int* sqr_tab;
823 const float* inv_tab;
824};

826static void
827trueDistTrans( const Mat& src, Mat& dst )
828{
829 const unsigned int inf = UINT_MAX;
830
831 CV_Assert( src.size() == dst.size() );
832
833 CV_Assert( (src.type() == CV_8UC1 || src.type() == CV_BoolC1) && dst.type() == CV_32FC1 );
834 int i, m = src.rows, n = src.cols;
835
836 cv::AutoBuffer<uchar> _buf(std::max(m*2*sizeof(int) + (m*3+1)*sizeof(int), n*2*sizeof(float)));
837 // stage 1: compute 1d distance transform of each column
838 unsigned int* sqr_tab = (unsigned int*)_buf.data();
839 int* sat_tab = cv::alignPtr((int*)(sqr_tab + m*2), sizeof(int));
840 int shift = m*2;
841
842 for( i = 0; i < m; i++ )
843 sqr_tab[i] = i >= PRECISE_DIST_MAX ? inf : static_cast<unsigned int>(i) * i;
844 for( i = m; i < m*2; i++ )
845 sqr_tab[i] = inf;
846 for( i = 0; i < shift; i++ )
847 sat_tab[i] = 0;
848 for( ; i <= m*3; i++ )
849 sat_tab[i] = i - shift;
850
851 cv::parallel_for_(cv::Range(0, n), cv::DTColumnInvoker(&src, &dst, sat_tab, sqr_tab), src.total()/(double)(1<<16));
852
853 // stage 2: compute modified distance transform for each row
854 float* inv_tab = (float*)sqr_tab + n;
855
856 inv_tab[0] = 0.f;
857 sqr_tab[0] = 0;
858 for( i = 1; i < n; i++ )
859 {
860 inv_tab[i] = (float)(0.5/i);
861 sqr_tab[i] = i >= PRECISE_DIST_MAX ? inf : static_cast<unsigned int>(i) * i;
862 }
863
864 cv::parallel_for_(cv::Range(0, m), cv::DTRowInvoker(&dst, sqr_tab, inv_tab));
865}

873//BEGIN ATS ADDITION
874// 8-bit grayscale distance transform function
875static void
876distanceATS_L1_8u( const Mat& src, Mat& dst )
877{
878 int width = src.cols, height = src.rows;
879
880 int a;
881 uchar lut[256];
882 int x, y;
883
884 const uchar *sbase = src.ptr();
885 uchar *dbase = dst.ptr();
886 int srcstep = (int)src.step;
887 int dststep = (int)dst.step;
888
889 CV_Assert( (src.type() == CV_8UC1 || src.type() == CV_BoolC1) && dst.type() == CV_8UC1 );
890 CV_Assert( src.size() == dst.size() );
891
893 for( x = 0; x < 256; x++ )
894 lut[x] = cv::saturate_cast<uchar>(x+1);
895
896 //init first pixel to max (we're going to be skipping it)
897 dbase[0] = (uchar)(sbase[0] == 0 ? 0 : 255);
898
899 //first row (scan west only, skip first pixel)
900 for( x = 1; x < width; x++ )
901 dbase[x] = (uchar)(sbase[x] == 0 ? 0 : lut[dbase[x-1]]);
902
903 for( y = 1; y < height; y++ )
904 {
905 sbase += srcstep;
906 dbase += dststep;
907
908 //for left edge, scan north only
909 a = sbase[0] == 0 ? 0 : lut[dbase[-dststep]];
910 dbase[0] = (uchar)a;
911
912 for( x = 1; x < width; x++ )
913 {
914 a = sbase[x] == 0 ? 0 : lut[MIN(a, dbase[x - dststep])];
915 dbase[x] = (uchar)a;
916 }
917 }
918
920
921 a = dbase[width-1];
922
923 // do last row east pixel scan here (skip bottom right pixel)
924 for( x = width - 2; x >= 0; x-- )
925 {
926 a = lut[a];
927 dbase[x] = (uchar)(CV_CALC_MIN_8U(a, dbase[x]));
928 }
929
930 // right edge is the only error case
931 for( y = height - 2; y >= 0; y-- )
932 {
933 dbase -= dststep;
934
935 // do right edge
936 a = lut[dbase[width-1+dststep]];
937 a = dbase[width-1] = (uchar)(MIN(a, dbase[width-1]));
938
939 for( x = width - 2; x >= 0; x-- )
940 {
941 int b = dbase[x+dststep];
942 a = lut[MIN(a, b)];
943 a = MIN(a, dbase[x]);
944 dbase[x] = (uchar)(a);
945 }
946 }
947}
948//END ATS ADDITION
