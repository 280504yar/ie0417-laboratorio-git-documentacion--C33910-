27// A lookup table of LogTags using full name, first part of name, and any part of name.

159 struct NameTable
160 {
161 // All data structures in this class are append-only. The item count
162 // is being used as an incrementing integer key.
163
164 public:
165 // Full name information struct.
166 std::vector<FullNameInfo> m_fullNameInfos;
167
168 // Name part information struct.
169 std::vector<NamePartInfo> m_namePartInfos;
170
171 // key: full name (string)
172 // value: full name ID
173 // .... (index into the vector of m_fullNameInfos)
174 // .... (key into m_fullNameToNamePartIds)
175 // .... (value.second in m_namePartToFullNameIds)
176 std::unordered_map<std::string, size_t> m_fullNameIds;
177
178 // key: name part (string)
179 // value: name part ID
180 // .... (index into the vector of m_namePartInfos)
181 // .... (key into m_namePartToFullNameIds)
182 // .... (value.second in m_fullNameToNamePartIds)
183 std::unordered_map<std::string, size_t> m_namePartIds;
184
185 // key: full name ID
186 // value.first: name part ID
187 // value.second: occurrence position of name part in the full name
188 std::unordered_multimap<size_t, std::pair<size_t, size_t>> m_fullNameToNamePartIds;
189
190 // key: name part ID
191 // value.first: full name ID
192 // value.second: occurrence position of name part in the full name
193 std::unordered_multimap<size_t, std::pair<size_t, size_t>> m_namePartToFullNameIds;
194
195 public:
196 void addOrLookupFullName(FullNameLookupResult& result);
197 void addOrLookupNamePart(NamePartLookupResult& result);
198 FullNameInfo* getFullNameInfo(const std::string& fullName);
199
200 private:
201 // Add or get full name. Does not compute name parts or access them in the table.
202 // Returns the full name ID, and a bool indicating if the full name is new.
203 std::pair<size_t, bool> internal_addOrLookupFullName(const std::string& fullName);
204
205 // Add or get multiple name parts. Saves name part IDs into a vector.
206 void internal_addOrLookupNameParts(const std::vector<std::string>& nameParts, std::vector<size_t>& namePartIds);
207
208 // Add or get name part. Returns namePartId.
209 size_t internal_addOrLookupNamePart(const std::string& namePart);
210
211 // For each name part ID, insert the tuples (full name, name part ID, name part position)
212 // into the cross reference table
213 void internal_addCrossReference(size_t fullNameId, const std::vector<size_t>& namePartIds);
214
215 // Gather pointer for full name info struct.
216 // Note: The pointer is interior to the table vector. The pointers are invalidated
217 // if the table is modified.
218 FullNameInfo* internal_getFullNameInfo(size_t fullNameId);
219
220 // Gather pointers for name part info struct.
221 // Note: The pointers are interior to the table vector. The pointers are invalidated
222 // if the table is modified.
223 NamePartInfo* internal_getNamePartInfo(size_t namePartId);
224
225 void internal_findMatchingNamePartsForFullName(FullNameLookupResult& fullNameResult);
226 void internal_findMatchingFullNamesForNamePart(NamePartLookupResult& result);
227 };
