60#define CV_WARP_SCALAR_FETCH_PIXEL_C1(dy, dx, pxy) \
61 if ((((unsigned)(ix + dx) < (unsigned)srccols) & ((unsigned)(iy + dy) < (unsigned)srcrows)) != 0) { \
62 size_t ofs = dy*srcstep + dx; \
63 pxy##g = srcptr[ofs]; \
64 } else if (border_type == BORDER_CONSTANT) { \
65 pxy##g = bval[0]; \
66 } else if (border_type == BORDER_TRANSPARENT) { \
67 pxy##g = dstptr[x]; \
68 } else { \
69 int ix_ = borderInterpolate_fast(ix + dx, srccols, border_type_x); \
70 int iy_ = borderInterpolate_fast(iy + dy, srcrows, border_type_y); \
71 size_t glob_ofs = iy_*srcstep + ix_; \
72 pxy##g = src[glob_ofs]; \
73 }
