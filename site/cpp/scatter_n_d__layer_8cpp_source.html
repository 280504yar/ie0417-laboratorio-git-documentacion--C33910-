51 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
52 const int requiredOutputs,
53 std::vector<MatShape> &outputs,
54 std::vector<MatShape> &internals) const CV_OVERRIDE
55 {
56 CV_CheckEQ(inputs.size(), 3ull, "ScatterND: require three inputs.");
57
58 size_t r = inputs[0].size(), q = inputs[1].size(), p = inputs[2].size(), k = inputs[1].back();
59 CV_CheckEQ(r + q - inputs[1].back() - 1, p, "ScatterND: updates should have rank of data.dims + indices.dims - indices.size[-1] - 1");
60 CV_CheckLE(k, r, "ScatterND: indices.shape[-1] must be less than (or equal to) the rank of input data.");
61
62 for (int i = 0; i < q - 1; i++) // np.ndindex(indices.shape[-1])
63 {
64 CV_CheckEQ(inputs[2][i], inputs[1][i], "ScatterND: updates.shape[0 : rank(indices)-1] must equal to indices.shape[0 : rank(indices)-1].");
65 }
66 for (int i = q - 1, j = k, m = 0; i + m < p; m++)
67 {
68 CV_CheckEQ(inputs[2][i + m], inputs[0][j + m], "ScatterND: updates.shape[rank(indices)-1 : ] must equal to data[indices.shape[-1] : rank(data)-1].");
69 }
70
71 outputs.assign(1, inputs[0]);
72 return false;
73 }

89 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
90 {
91 CV_TRACE_FUNCTION();
92 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
93
94 if (inputs_arr.depth() == CV_16F) {
95 forward_fallback(inputs_arr, outputs_arr, internals_arr);
96 return;
97 }
98
99 std::vector<Mat> inputs, outputs;
100 inputs_arr.getMatVector(inputs);
101 outputs_arr.getMatVector(outputs);
102
103 const Mat& data = inputs[0];
104 const Mat& indices = inputs[1];
105 const Mat& updates = inputs[2];
106 Mat& out = outputs[0];
107
108 indexTypeDispatch(outputs[0].type(), indices.type(), data, indices, updates, out);
109 }

111 // NOTE: This impl does not check whether indices have duplicate entries.
112 // The last duplicate entry will overwrite the previous.
113 template<typename T, typename T_INDEX, typename Functor>
114 void forward_impl(const Functor &reduce_operation, const Mat &input_mat, const Mat &indices_mat, const Mat &updates_mat, Mat& output_mat) {
115 input_mat.copyTo(output_mat);
116
117 const auto &input_mat_shape = shape(input_mat);
118 std::vector<size_t> input_mat_step(input_mat_shape.size());
119 for (int i = 0; i < input_mat.dims; i++) {
120 input_mat_step[i] = static_cast<size_t>(input_mat.step.p[i] / sizeof(T));
121 }
122
123 const int indices_mat_ndims = indices_mat.dims;
124 const auto &indices_mat_shape = shape(indices_mat);
125
126 const int updates_mat_ndims = updates_mat.dims;
127 const auto &updates_mat_shape = shape(updates_mat);
128
129 int indices_last_dim = indices_mat_shape[indices_mat_ndims - 1]; // last dim of indices
130
131 size_t updates_size = 1;
132 for (int i = indices_mat_ndims - 1; i < updates_mat_ndims; i++)
133 updates_size *= updates_mat_shape[i];
134
135 auto fn = [&](const Range &r) {
136 size_t input_offset = 0,
137 indices_offset = r.start * indices_last_dim,
138 updates_offset = r.start * updates_size;
139 for (int i = r.start; i < r.end; i++) {
140 const T_INDEX* indices = indices_mat.ptr<const T_INDEX>();
141 const T* updates = updates_mat.ptr<const T>();
142 T* output = output_mat.ptr<T>();
143
144 input_offset = 0;
145 indices += indices_offset;
146 for (int j = 0; j < indices_last_dim; j++) {
147 int index = *(indices + j);
148 index = (index + input_mat_shape[j]) % input_mat_shape[j];
149 CV_Assert(index < input_mat_shape[j] && index >= 0);
150 input_offset += index * input_mat_step[j];
151 }
152
153 updates += updates_offset;
154 output += input_offset;
155 for (int j = 0; j < updates_size; j++) {
156 output[j] = reduce_operation(output[j], updates[j]);
157 }
158
159 indices_offset += indices_last_dim;
160 updates_offset += updates_size;
161 }
162 };
163
164 size_t total = (size_t)(indices_mat.total() / indices_last_dim);
165 double nstripes = (size_t)total * (indices_last_dim + updates_size) * (1 / 1024.0);
166 parallel_for_(Range(0, total), fn, nstripes);
167 }

229 template<typename T, typename T_INDEX, typename... Args>
230 inline void reductionDispatch(Args&&... args)
231 {
232 switch (reduction)
233 {
234 case REDUCTION::NONE:
235 {
236 auto rd = [](const T& a, const T& b) { return b; }; // a from input data, b from updates
237 forward_impl<T, T_INDEX>(rd, std::forward<Args>(args)...);
238 break;
239 }
240 case REDUCTION::ADD:
241 {
242 auto rd = [](const T& a, const T& b) { return a + b; };
243 forward_impl<T, T_INDEX>(rd, std::forward<Args>(args)...);
244 break;
245 }
246 case REDUCTION::MUL:
247 {
248 auto rd = [](const T& a, const T& b) { return a * b; };
249 forward_impl<T, T_INDEX>(rd, std::forward<Args>(args)...);
250 break;
251 }
252 case REDUCTION::MAX:
253 {
254 auto rd = [](const T& a, const T& b) { return std::max(a, b); };
255 forward_impl<T, T_INDEX>(rd, std::forward<Args>(args)...);
256 break;
257 }
258 case REDUCTION::MIN:
259 {
260 auto rd = [](const T& a, const T& b) { return std::min(a, b); };
261 forward_impl<T, T_INDEX>(rd, std::forward<Args>(args)...);
262 break;
263 }
264 default:
265 CV_Error(Error::StsBadArg, "Unsupported reduction.");
266 };
