36 def parse(cls, full_symbol_name: str,
37 known_namespaces: Sequence[str],
38 symbol_parts_delimiter: str = '.') -> "SymbolName":
39 """Performs contextual symbol name parsing into namespaces, classes
40 and "bare" symbol name.
41
42 Args:
43 full_symbol_name (str): Input string to parse symbol name from.
44 known_namespaces (Sequence[str]): Collection of namespace that was
45 met during C++ headers parsing.
46 symbol_parts_delimiter (str, optional): Delimiter string used to
47 split `full_symbol_name` string into chunks. Defaults to '.'.
48
49 Returns:
50 SymbolName: Parsed symbol name structure.
51
52 >>> SymbolName.parse('cv.ns.Feature', ('cv', 'cv.ns'))
53 (namespace="cv::ns", classes="", name="Feature")
54
55 >>> SymbolName.parse('cv.ns.Feature', ())
56 (namespace="", classes="cv::ns", name="Feature")
57
58 >>> SymbolName.parse('cv.ns.Feature.Params', ('cv', 'cv.ns'))
59 (namespace="cv::ns", classes="Feature", name="Params")
60
61 >>> SymbolName.parse('cv::ns::Feature::Params::serialize',
62 ... known_namespaces=('cv', 'cv.ns'),
63 ... symbol_parts_delimiter='::')
64 (namespace="cv::ns", classes="Feature::Params", name="serialize")
65 """
66
67 chunks = full_symbol_name.split(symbol_parts_delimiter)
68 namespaces, name = chunks[:-1], chunks[-1]
69 classes: List[str] = []
70 while len(namespaces) > 0 and '.'.join(namespaces) not in known_namespaces:
71 classes.insert(0, namespaces.pop())
72 return SymbolName(tuple(namespaces), tuple(classes), name)

75def find_scope(root: NamespaceNode, symbol_name: SymbolName,
76 create_missing_namespaces: bool = True) -> Union[NamespaceNode, ClassNode]:
77 """Traverses down nodes hierarchy to the direct parent of the node referred
78 by `symbol_name`.
79
80 Args:
81 root (NamespaceNode): Root node of the hierarchy.
82 symbol_name (SymbolName): Full symbol name to find scope for.
83 create_missing_namespaces (bool, optional): Set to True to create missing
84 namespaces while traversing the hierarchy. Defaults to True.
85
86 Raises:
87 ScopeNotFoundError: If direct parent for the node referred by `symbol_name`
88 can't be found e.g. one of classes doesn't exist.
89
90 Returns:
91 Union[NamespaceNode, ClassNode]: Direct parent for the node referred by
92 `symbol_name`.
93
94 >>> root = NamespaceNode('cv')
95 >>> algorithm_node = root.add_class('Algorithm')
96 >>> find_scope(root, SymbolName(('cv', ), ('Algorithm',), 'Params')) == algorithm_node
97 True
98
99 >>> root = NamespaceNode('cv')
100 >>> scope = find_scope(root, SymbolName(('cv', 'gapi', 'detail'), (), 'function'))
101 >>> scope.full_export_name
102 'cv.gapi.detail'
103
104 >>> root = NamespaceNode('cv')
105 >>> scope = find_scope(root, SymbolName(('cv', 'gapi'), ('GOpaque',), 'function'))
106 Traceback (most recent call last):
107 ...
108 ast_utils.ScopeNotFoundError: Can't find a scope for 'function', with \
109'(namespace="cv::gapi", classes="GOpaque", name="function")', \
110because 'GOpaque' class is not registered yet
111 """
112 assert isinstance(root, NamespaceNode), \
113 'Wrong hierarchy root type: {}'.format(type(root))
114
115 assert symbol_name.namespaces[0] == root.name, \
116 "Trying to find scope for '{}' with root namespace different from: '{}'".format(
117 symbol_name, root.name
118 )
119
120 scope: Union[NamespaceNode, ClassNode] = root
121 for namespace in symbol_name.namespaces[1:]:
122 if namespace not in scope.namespaces: # type: ignore
123 if not create_missing_namespaces:
124 raise ScopeNotFoundError(
125 "Can't find a scope for '{}', with '{}', because namespace"
126 " '{}' is not created yet and `create_missing_namespaces`"
127 " flag is set to False".format(
128 symbol_name.name, symbol_name, namespace
129 )
130 )
131 scope = scope.add_namespace(namespace) # type: ignore
132 else:
133 scope = scope.namespaces[namespace] # type: ignore
134 for class_name in symbol_name.classes:
135 if class_name not in scope.classes:
136 raise ScopeNotFoundError(
137 "Can't find a scope for '{}', with '{}', because '{}' "
138 "class is not registered yet".format(
139 symbol_name.name, symbol_name, class_name
140 )
141 )
142 scope = scope.classes[class_name]
143 return scope

166def create_function_node_in_scope(scope: Union[NamespaceNode, ClassNode],
167 func_info) -> FunctionNode:
168 def prepare_overload_arguments_and_return_type(variant):
169 arguments = [] # type: list[FunctionNode.Arg]
170 # Enumerate is required, because `argno` in `variant.py_arglist`
171 # refers to position of argument in C++ function interface,
172 # but `variant.py_noptargs` refers to position in `py_arglist`
173 for i, (_, argno) in enumerate(variant.py_arglist):
174 arg_info = variant.args[argno]
175 type_node = create_type_node(arg_info.tp)
176 # Special handling for string representation of the file system path
177 if arg_info.pathlike and type_node.typename == "str":
178 type_node = PathLikeTypeNode.string_or_pathlike_()
179
180 default_value = None
181 if len(arg_info.defval):
182 default_value = arg_info.defval
183 # If argument is optional and can be None - make its type optional
184 if variant.is_arg_optional(i):
185 # NOTE: should UMat be always mandatory for better type hints?
186 # otherwise overload won't be selected e.g. VideoCapture.read()
187 if arg_info.py_outputarg:
188 type_node = OptionalTypeNode(type_node)
189 default_value = "None"
190 elif arg_info.isbig() and "None" not in type_node.typename:
191 # but avoid duplication of the optioness
192 type_node = OptionalTypeNode(type_node)
193 arguments.append(
194 FunctionNode.Arg(arg_info.export_name, type_node=type_node,
195 default_value=default_value)
196 )
197 if func_info.isconstructor:
198 return arguments, None
199
200 # Function has more than 1 output argument, so its return type is a tuple
201 if len(variant.py_outlist) > 1:
202 ret_types = []
203 # Actual returned value of the function goes first
204 if variant.py_outlist[0][1] == -1:
205 ret_types.append(create_type_node(variant.rettype))
206 outlist = variant.py_outlist[1:]
207 else:
208 outlist = variant.py_outlist
209 for _, argno in outlist:
210 assert argno >= 0, \
211 f"Logic Error! Outlist contains function return type: {outlist}"
212
213 ret_types.append(create_type_node(variant.args[argno].tp))
214
215 return arguments, FunctionNode.RetType(
216 TupleTypeNode("return_type", ret_types)
217 )
218 # Function with 1 output argument in Python
219 if len(variant.py_outlist) == 1:
220 # Can be represented as a function with a non-void return type in C++
221 if variant.rettype:
222 return arguments, FunctionNode.RetType(
223 create_type_node(variant.rettype)
224 )
225 # or a function with void return type and output argument type
226 # such non-const reference
227 ret_type = variant.args[variant.py_outlist[0][1]].tp
228 return arguments, FunctionNode.RetType(
229 create_type_node(ret_type)
230 )
231 # Function without output types returns None in Python
232 return arguments, None
233
234 function_node = FunctionNode(func_info.name)
235 function_node.parent = scope
236 if func_info.isconstructor:
237 function_node.export_name = "__init__"
238 for variant in func_info.variants:
239 arguments, ret_type = prepare_overload_arguments_and_return_type(variant)
240 if isinstance(scope, ClassNode):
241 if func_info.is_static:
242 if ret_type is not None and ret_type.typename.endswith(scope.name):
243 function_node.is_classmethod = True
244 arguments.insert(0, FunctionNode.Arg("cls"))
245 else:
246 function_node.is_static = True
247 else:
248 arguments.insert(0, FunctionNode.Arg("self"))
249 function_node.add_overload(arguments, ret_type)
250 return function_node

263def create_class_node_in_scope(scope: Union[NamespaceNode, ClassNode],
264 symbol_name: SymbolName,
265 class_info) -> ClassNode:
266 properties = []
267 for property in class_info.props:
268 export_property_name = property.name
269 if keyword.iskeyword(export_property_name):
270 export_property_name += "_"
271 properties.append(
272 ClassProperty(
273 name=export_property_name,
274 type_node=create_type_node(property.tp),
275 is_readonly=property.readonly
276 )
277 )
278 class_node = scope.add_class(symbol_name.name,
279 properties=properties)
280 class_node.export_name = class_info.export_name
281 if class_info.constructor is not None:
282 create_function_node_in_scope(class_node, class_info.constructor)
283 for method in class_info.methods.values():
284 create_function_node_in_scope(class_node, method)
285 return class_node

295def resolve_enum_scopes(root: NamespaceNode,
296 enums: Dict[SymbolName, EnumerationNode]):
297 """Attaches all enumeration nodes to the appropriate classes and modules
298
299 If classes containing enumeration can't be found in the AST - they will
300 be created and marked as not exportable. This behavior is required to cover
301 cases, when enumeration is defined in base class, but only its derivatives
302 are used. Example:
303 ```cpp
304 class CV_EXPORTS TermCriteria {
305 public:
306 enum Type { /* ... */ };
307 // ...
308 };
309 ```
310
311 Args:
312 root (NamespaceNode): root of the reconstructed AST
313 enums (Dict[SymbolName, EnumerationNode]): Mapping between enumerations
314 symbol names and corresponding nodes without parents.
315 """
316
317 for symbol_name, enum_node in enums.items():
318 if symbol_name.classes:
319 try:
320 scope = find_scope(root, symbol_name)
321 except ScopeNotFoundError:
322 # Scope can't be found if enumeration is a part of class
323 # that is not exported.
324 # Create class node, but mark it as not exported
325 for i, class_name in enumerate(symbol_name.classes):
326 scope = find_scope(root,
327 SymbolName(symbol_name.namespaces,
328 classes=symbol_name.classes[:i],
329 name=class_name))
330 if class_name in scope.classes:
331 continue
332 class_node = scope.add_class(class_name)
333 class_node.is_exported = False
334 scope = find_scope(root, symbol_name)
335 else:
336 scope = find_scope(root, symbol_name)
337 enum_node.parent = scope

340def get_enclosing_namespace(
341 node: ASTNode,
342 class_node_callback: Optional[Callable[[ClassNode], None]] = None
343) -> NamespaceNode:
344 """Traverses up nodes hierarchy to find closest enclosing namespace of the
345 passed node
346
347 Args:
348 node (ASTNode): Node to find a namespace for.
349 class_node_callback (Optional[Callable[[ClassNode], None]]): Optional
350 callable object invoked for each traversed class node in bottom-up
351 order. Defaults: None.
352
353 Returns:
354 NamespaceNode: Closest enclosing namespace of the provided node.
355
356 Raises:
357 AssertionError: if nodes hierarchy missing a namespace node.
358
359 >>> root = NamespaceNode('cv')
360 >>> feature_class = root.add_class("Feature")
361 >>> get_enclosing_namespace(feature_class) == root
362 True
363
364 >>> root = NamespaceNode('cv')
365 >>> feature_class = root.add_class("Feature")
366 >>> feature_params_class = feature_class.add_class("Params")
367 >>> serialize_params_func = feature_params_class.add_function("serialize")
368 >>> get_enclosing_namespace(serialize_params_func) == root
369 True
370
371 >>> root = NamespaceNode('cv')
372 >>> detail_ns = root.add_namespace('detail')
373 >>> flags_enum = detail_ns.add_enumeration('Flags')
374 >>> get_enclosing_namespace(flags_enum) == detail_ns
375 True
376 """
377 parent_node = node.parent
378 while not isinstance(parent_node, NamespaceNode):
379 assert parent_node is not None, \
380 "Can't find enclosing namespace for '{}' known as: '{}'".format(
381 node.full_export_name, node.native_name
382 )
383 if class_node_callback:
384 class_node_callback(cast(ClassNode, parent_node))
385 parent_node = parent_node.parent
386 return parent_node
