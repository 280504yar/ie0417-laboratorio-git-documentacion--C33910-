51void checkDeviceMaxMemoryAllocSize(const Size& size, int type, int factor)
52{
53 CV_Assert(factor > 0);
54
55 if (!cv::ocl::useOpenCL())
56 return;
57
58 size_t memSize = size.area() * CV_ELEM_SIZE(type);
59 const cv::ocl::Device& dev = cv::ocl::Device::getDefault();
60
61 if (memSize * factor >= dev.maxMemAllocSize())
62 throw ::perf::TestBase::PerfSkipTestException();
63}

65void randu(InputOutputArray dst)
66{
67 if (dst.depth() == CV_8U)
68 cv::randu(dst, 0, 256);
69 else if (dst.depth() == CV_8S)
70 cv::randu(dst, -128, 128);
71 else if (dst.depth() == CV_16U)
72 cv::randu(dst, 0, 1024);
73 else if (dst.depth() == CV_32F || dst.depth() == CV_64F || dst.depth() == CV_16F)
74 cv::randu(dst, -1.0, 1.0);
75 else if (dst.depth() == CV_16S || dst.depth() == CV_32S)
76 cv::randu(dst, -4096, 4096);
77 else
78 CV_Error(Error::StsUnsupportedFormat, "Unsupported format");
79}
