16 Rect select(const String &windowName, Mat img, bool showCrossair = true, bool fromCenter = true, bool printNotice = true)
17 {
18 if(printNotice)
19 {
20 // show notice to user
21 printf("Select a ROI and then press SPACE or ENTER button!\n");
22 printf("Cancel the selection process by pressing c button!\n");
23 }
24
25 key = 0;
26 imageSize = img.size();
27
28 // set the drawing mode
29 selectorParams.drawFromCenter = fromCenter;
30
31 // show the image and give feedback to user
32 imshow(windowName, img);
33
34 // copy the data, rectangle should be drawn in the fresh image
35 selectorParams.image = img.clone();
36
37 // select the object
38 setMouseCallback(windowName, mouseHandler, (void*)this);
39
40 // end selection process on SPACE (32) ESC (27) or ENTER (13)
41 while (!(key == 32 || key == 27 || key == 13))
42 {
43 // draw the selected object
44 rectangle(selectorParams.image, selectorParams.box, Scalar(255, 0, 0), 2, 1);
45
46 // draw cross air in the middle of bounding box
47 if (showCrossair)
48 {
49 // horizontal line
50 line(selectorParams.image,
51 Point((int)selectorParams.box.x,
52 (int)(selectorParams.box.y + selectorParams.box.height / 2)),
53 Point((int)(selectorParams.box.x + selectorParams.box.width),
54 (int)(selectorParams.box.y + selectorParams.box.height / 2)),
55 Scalar(255, 0, 0), 2, 1);
56
57 // vertical line
58 line(selectorParams.image,
59 Point((int)(selectorParams.box.x + selectorParams.box.width / 2),
60 (int)selectorParams.box.y),
61 Point((int)(selectorParams.box.x + selectorParams.box.width / 2),
62 (int)(selectorParams.box.y + selectorParams.box.height)),
63 Scalar(255, 0, 0), 2, 1);
64 }
65
66 // show the image bounding box
67 imshow(windowName, selectorParams.image);
68
69 // reset the image
70 selectorParams.image = img.clone();
71
72 // get keyboard event
73 key = waitKey(30);
74
75 if (key == 'c' || key == 'C')//cancel selection
76 {
77 selectorParams.box = Rect();
78 break;
79 }
80 }
81
82 //cleanup callback
83 setMouseCallback(windowName, emptyMouseHandler, NULL);
84
85 return selectorParams.box;

135 void opencv_mouse_callback(int event, int x, int y, int)
136 {
137 switch (event)
138 {
139 // update the selected bounding box
140 case EVENT_MOUSEMOVE:
141 if (selectorParams.isDrawing)
142 {
143 if (selectorParams.drawFromCenter)
144 {
145 // limit half extends to imageSize
146 float halfWidth = std::min(std::min(
147 std::abs(x - selectorParams.startPos.x),
148 selectorParams.startPos.x),
149 imageSize.width - selectorParams.startPos.x);
150 float halfHeight = std::min(std::min(
151 std::abs(y - selectorParams.startPos.y),
152 selectorParams.startPos.y),
153 imageSize.height - selectorParams.startPos.y);
154
155 selectorParams.box.width = halfWidth * 2;
156 selectorParams.box.height = halfHeight * 2;
157 selectorParams.box.x = selectorParams.startPos.x - halfWidth;
158 selectorParams.box.y = selectorParams.startPos.y - halfHeight;
159
160 }
161 else
162 {
163 // limit x and y to imageSize
164 int lx = std::min(std::max(x, 0), imageSize.width);
165 int by = std::min(std::max(y, 0), imageSize.height);
166 selectorParams.box.width = std::abs(lx - selectorParams.startPos.x);
167 selectorParams.box.height = std::abs(by - selectorParams.startPos.y);
168 selectorParams.box.x = std::min((float)lx, selectorParams.startPos.x);
169 selectorParams.box.y = std::min((float)by, selectorParams.startPos.y);
170 }
171 }
172 break;
173
174 // start to select the bounding box
175 case EVENT_LBUTTONDOWN:
176 selectorParams.isDrawing = true;
177 selectorParams.box = Rect2d(x, y, 0, 0);
178 selectorParams.startPos = Point2f((float)x, (float)y);
179 break;
180
181 // cleaning up the selected bounding box
182 case EVENT_LBUTTONUP:
183 selectorParams.isDrawing = false;
184 if (selectorParams.box.width < 0)
185 {
186 selectorParams.box.x += selectorParams.box.width;
187 selectorParams.box.width *= -1;
188 }
189 if (selectorParams.box.height < 0)
190 {
191 selectorParams.box.y += selectorParams.box.height;
192 selectorParams.box.height *= -1;
193 }
194 break;
195 }
196 }
197
198 // save the keypressed character
199 int key;
200 Size imageSize;
201};
202}
