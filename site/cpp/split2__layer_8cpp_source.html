30// axis must be normalized
31static void split(const Mat& inp, std::vector<Mat>& outs, int axis)
32{
33 CV_Assert(inp.isContinuous());
34
35 MatShape inpShape = inp.shape();
36 int ndims = inpShape.dims;
37
38 CV_Assert_N(0 <= axis, axis <= inp.dims);
39
40 int nslices = 1;
41 int inpType = inp.type();
42 size_t esz = inp.elemSize();
43 size_t sliceSize = esz;
44 size_t inpStep = 0;
45 int outSize_a = 0;
46 for (int i = ndims-1; i > axis; i--)
47 sliceSize *= inpShape[i];
48 inpStep = sliceSize*inpShape[axis];
49 for (int i = 0; i < axis; i++)
50 nslices *= inpShape[i];
51
52 size_t noutputs = outs.size();
53 for (size_t k = 0; k < noutputs; k++) {
54 Mat& out = outs[k];
55 MatShape outShape = out.shape();
56 CV_Assert(out.isContinuous());
57 CV_Assert(out.type() == inpType);
58 CV_Assert(out.dims == ndims);
59 for (int i = 0; i < ndims; i++) {
60 if (i == axis)
61 outSize_a += outShape[i];
62 else {
63 CV_Assert(inpShape[i] == outShape[i]);
64 }
65 }
66 }
67
68 CV_Assert(outSize_a == inpShape[axis]);
69
70 // Precompute per-output source offset and per-slice size.
71 std::vector<size_t> srcOffset(noutputs);
72 std::vector<size_t> sliceSize_k_vec(noutputs);
73 {
74 size_t acc = 0;
75 for (size_t k = 0; k < noutputs; k++) {
76 int sz_a = outs[k].size[axis];
77 srcOffset[k] = acc;
78 sliceSize_k_vec[k] = sliceSize * sz_a;
79 acc += sliceSize_k_vec[k];
80 }
81 }
82
83 int64_t nTasks = (int64_t)nslices * (int64_t)noutputs;
84 parallel_for_(Range(0, (int)nTasks), [&](const Range& r) {
85 for (int64_t idx = r.start; idx < r.end; idx++) {
86 int k = (int)(idx % (int64_t)noutputs);
87 int s = (int)(idx / (int64_t)noutputs);
88 uchar* outptr_k = outs[k].data;
89 const uchar* inptr = inp.data + srcOffset[k];
90 size_t sliceSize_k = sliceSize_k_vec[k];
91 memcpy(outptr_k + (size_t)s*sliceSize_k, inptr + (size_t)s*inpStep, sliceSize_k);
92 }
93 });
94}

111 void getOutShapes(const MatShape& inpShape, int axis_,
112 const std::vector<int>& split,
113 std::vector<MatShape>& outShapes) const
114 {
115 size_t noutputs = split.size();
116 CV_Assert(noutputs == outputs.size());
117
118 int inpDims = inpShape.dims;
119 CV_Assert(0 <= axis_ && axis_ < inpDims);
120 int totalSize_a = 0;
121
122 outShapes.resize(noutputs);
123 for (size_t i = 0; i < noutputs; i++) {
124 MatShape outShape = inpShape;
125 int s = split[i];
126 CV_Assert(s >= 0);
127 CV_Assert(s <= inpShape[axis_] - totalSize_a);
128 outShape[axis_] = s;
129 outShapes[i] = outShape;
130 totalSize_a += s;
131 }
132 }

134 void makeDefaultSplit(int totalSize, size_t noutputs, std::vector<int>& split_) const
135 {
136 split_.resize(noutputs);
137 int chunkSize = (int)((totalSize + noutputs - 1) / noutputs);
138 for (size_t i = 0; i < noutputs; i++) {
139 int sz_i = std::min(totalSize, chunkSize);
140 split_[i] = sz_i;
141 totalSize -= sz_i;
142 }
143 }

147 bool getMemoryShapes(const std::vector<MatShape> &inputs,
148 const int noutputs,
149 std::vector<MatShape> &outputs,
150 std::vector<MatShape> &internals) const CV_OVERRIDE
151 {
152 CV_Assert(noutputs == (int)this->outputs.size());
153
154 size_t ninputs = inputs.size();
155 CV_Assert(ninputs == 1 || ninputs == 2);
156
157 MatShape inpShape = inputs[0];
158 std::vector<int> tempSplit;
159 const std::vector<int>* split_ = &split;
160 int axis_ = normalize_axis(axis, inpShape.dims);
161
162 if (ninputs == 2) {
163 Net::Impl* netimpl_ = getNetImpl(this);
164 Mat splitTensor = netimpl_->argTensor(this->inputs[1]);
165 tensorToIntVec(splitTensor, tempSplit);
166 split_ = &tempSplit;
167 }
168 else if (split.empty()) {
169 makeDefaultSplit(inpShape[axis_], noutputs, tempSplit);
170 split_ = &tempSplit;
171 }
172
173 getOutShapes(inputs[0], axis_, *split_, outputs);
174 internals.clear();
175 return true;
176 }

195 void forward(InputArrayOfArrays inputs_arr,
196 OutputArrayOfArrays outputs_arr,
197 OutputArrayOfArrays) CV_OVERRIDE
198 {
199 CV_TRACE_FUNCTION();
200 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
201
202 Size size = inputs_arr.size();
203 int ninputs = size.area();
204 int noutputs = (int)outputs.size();
205
206 CV_Assert(ninputs == 1 || ninputs == 2);
207
208 int inpType = inputs_arr.type(0);
209 MatShape inpShape = inputs_arr.shape(0);
210 std::vector<int> tempSplit;
211 const std::vector<int>* split_ = &split;
212 std::vector<MatShape> outShapes;
213
214 int axis_ = normalize_axis(axis, inpShape.dims);
215
216 if (ninputs == 2) {
217 Mat splitTensor = inputs_arr.getMat(1);
218 tensorToIntVec(splitTensor, tempSplit);
219 split_ = &tempSplit;
220 }
221 else if (split.empty()) {
222 makeDefaultSplit(inpShape[axis_], noutputs, tempSplit);
223 split_ = &tempSplit;
224 }
225 getOutShapes(inpShape, axis_, *split_, outShapes);
226 CV_Assert(outShapes.size() == (size_t)noutputs);
227
228 int outKind = outputs_arr.kind();
229
230 CV_Assert(outKind == _InputArray::STD_VECTOR_MAT ||
231 outKind == _InputArray::STD_VECTOR_UMAT);
232
233 if (outKind == _InputArray::STD_VECTOR_MAT) {
234 Mat inp = inputs_arr.getMat(0);
235 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
236 outs.resize(noutputs);
237 for (int i = 0; i < noutputs; i++) {
238 MatShape outShape = outShapes[i];
239 outs[i].fit(outShape, inpType);
240 }
241 runOp(inp, outs, axis_);
242 } else {
243 // [TODO] more efficient OpenCL implementation
244 Mat inp = inputs_arr.getMat(0);
245 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
246 outs.resize(noutputs);
247
248 std::vector<Mat> temps(noutputs);
249 for (int i = 0; i < noutputs; i++) {
250 MatShape outShape = outShapes[i];
251 temps[i].fit(outShape, inpType);
252 }
253 runOp(inp, temps, axis_);
254 for (int i = 0; i < noutputs; i++) {
255 MatShape outShape = outShapes[i];
256 outs[i].fit(outShape, inpType);
257 temps[i].copyTo(outs[i]);
258 temps[i].release();
259 }
260 }
261 }

263 void runOp(const Mat& inp, std::vector<Mat>& outs, int axis_)
264 {
265 cv::dnn::split(inp, outs, axis_);
266 }
