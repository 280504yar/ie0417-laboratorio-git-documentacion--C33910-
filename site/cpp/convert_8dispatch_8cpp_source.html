56static bool ocl_convertTo(InputArray src_, OutputArray dst_, int ddepth, bool noScale, double alpha, double beta)
57{
58 CV_INSTRUMENT_REGION();
59
60 CV_Assert(ddepth >= 0);
61
62 int stype = src_.type();
63 int sdepth = CV_MAT_DEPTH(stype);
64 int cn = CV_MAT_CN(stype);
65
66 // FP8 types are not supported by the OpenCL kernels yet; fall back to CPU.
67 if ((sdepth >= CV_8F_E4M3FN && sdepth <= CV_8F_E4M3FNUZ) ||
68 (ddepth >= CV_8F_E4M3FN && ddepth <= CV_8F_E4M3FNUZ))
69 return false;
70
71 int dtype = CV_MAKETYPE(ddepth, cn);
72
73 int wdepth = (sdepth == CV_64F) ? CV_64F : CV_32F;
74
75 bool needDouble = sdepth == CV_64F || ddepth == CV_64F;
76 bool doubleCheck = true;
77 if (needDouble)
78 {
79 doubleCheck = ocl::Device::getDefault().hasFP64();
80 }
81 bool halfCheck = true;
82 bool needHalf = sdepth == CV_16F || ddepth == CV_16F;
83 if (needHalf)
84 {
85 halfCheck = ocl::Device::getDefault().hasFP16();
86 }
87
88 if (!doubleCheck)
89 return false;
90 if (!halfCheck)
91 return false;
92
93 const int rowsPerWI = 4;
94
95 char cvt[2][50];
96 ocl::Kernel k("convertTo", ocl::core::convert_oclsrc,
97 format("-D srcT=%s -D WT=%s -D dstT=%s -D convertToWT=%s -D convertToDT=%s -D rowsPerWI=%d%s%s%s",
98 ocl::typeToStr(sdepth), ocl::typeToStr(wdepth), ocl::typeToStr(ddepth),
99 ocl::convertTypeStr(sdepth, wdepth, 1, cvt[0], sizeof(cvt[0])),
100 ocl::convertTypeStr(wdepth, ddepth, 1, cvt[1], sizeof(cvt[1])),
101 rowsPerWI,
102 needDouble ? " -D DOUBLE_SUPPORT" : "",
103 needHalf ? " -D HALF_SUPPORT" : "",
104 noScale ? " -D NO_SCALE" : ""
105 )
106 );
107
108 if (k.empty())
109 return false;
110
111 UMat src = src_.getUMat();
112 dst_.createSameSize(src_, dtype);
113 UMat dst = dst_.getUMat();
114
115 float alphaf = (float)alpha, betaf = (float)beta;
116
117 if (noScale)
118 k.args(ocl::KernelArg::ReadOnlyNoSize(src), ocl::KernelArg::WriteOnly(dst, cn));
119 else if (wdepth == CV_32F)
120 k.args(ocl::KernelArg::ReadOnlyNoSize(src), ocl::KernelArg::WriteOnly(dst, cn), alphaf, betaf);
121 else
122 k.args(ocl::KernelArg::ReadOnlyNoSize(src), ocl::KernelArg::WriteOnly(dst, cn), alpha, beta);
123
124 size_t globalsize[2] = {
125 (size_t)dst.cols * cn,
126 divUp((size_t)dst.rows, rowsPerWI)
127 };
128 if (!k.run(2, globalsize, NULL, false))
129 return false;
130
131 CV_IMPL_ADD(CV_IMPL_OCL);
132 return true;
133}

136void Mat::convertTo(OutputArray dst, int type_, double alpha, double beta) const
137{
138 CV_INSTRUMENT_REGION();
139
140 if (empty())
141 {
142 int ddepth = CV_MAT_DEPTH(type_);
143 if (ddepth < 0)
144 ddepth = dst.fixedType() ? dst.depth() : depth();
145 const int dtype = CV_MAKETYPE(ddepth, channels());
146
147 dst.release();
148 bool allowTransposed = dims == 1 ||
149 dst.kind() == _InputArray::STD_VECTOR ||
150 (dst.fixedSize() && dst.dims() == 1);
151 dst.create(size, dtype, -1, allowTransposed);
152 return;
153 }
154
155 int stype = type();
156 int sdepth = CV_MAT_DEPTH(stype);
157
158 int ddepth = sdepth;
159 if (type_ >= 0)
160 ddepth = CV_MAT_DEPTH(type_);
161 else
162 ddepth = dst.fixedType() ? dst.depth() : sdepth;
163
164 bool noScale = std::fabs(alpha - 1) < DBL_EPSILON && std::fabs(beta) < DBL_EPSILON;
165 if (sdepth == ddepth && noScale)
166 {
167 copyTo(dst);
168 return;
169 }
170
171 CV_OCL_RUN(dims <= 2 && dst.isUMat(),
172 ocl_convertTo(*this, dst, ddepth, noScale, alpha, beta))
173
174 int cn = channels();
175 int dtype = CV_MAKETYPE(ddepth, cn);
176
177 Mat src = *this;
178 bool allowTransposed = dims == 1 ||
179 dst.kind() == _InputArray::STD_VECTOR ||
180 (dst.fixedSize() && dst.dims() == 1);
181 dst.create( size, dtype, -1, allowTransposed );
182 Mat dstMat = dst.getMat();
183
184 if( dims <= 2 )
185 {
186 CALL_HAL(convertScale, cv_hal_convertScale, src.data, src.step, dstMat.data, dstMat.step, src.cols * cn, src.rows, sdepth, ddepth, alpha, beta);
187 }
188 else if( src.isContinuous() && dstMat.isContinuous() )
189 {
190 CALL_HAL(convertScale, cv_hal_convertScale, src.data, 0, dstMat.data, 0, (int)src.total() * cn, 1, sdepth, ddepth, alpha, beta);
191 }
192
193 BinaryFunc func = noScale ? getConvertFunc(sdepth, ddepth) : getConvertScaleFunc(sdepth, ddepth);
194 double scale[] = {alpha, beta};
195 CV_Assert( func != 0 );
196
197 if( dims <= 2 )
198 {
199 Size sz = getContinuousSize2D(src, dstMat, cn);
200 func(src.data, src.step, 0, 0, dstMat.data, dstMat.step, sz, scale);
201 }
202 else
203 {
204 const Mat* arrays[] = {&src, &dstMat, 0};
205 uchar* ptrs[2] = {};
206 NAryMatIterator it(arrays, ptrs);
207 Size sz((int)(it.size*cn), 1);
208
209 for( size_t i = 0; i < it.nplanes; i++, ++it )
210 func(ptrs[0], 1, 0, 0, ptrs[1], 1, sz, scale);
211 }
212}

214void UMat::convertTo(OutputArray dst, int type_, double alpha, double beta) const
215{
216 CV_INSTRUMENT_REGION();
217
218 if (empty())
219 {
220 dst.release();
221 dst.create(size(), type_ >= 0 ? type_ : type());
222 return;
223 }
224
225#ifdef HAVE_OPENCL
226 int stype = type();
227 int sdepth = CV_MAT_DEPTH(stype);
228
229 int ddepth = sdepth;
230 if (type_ >= 0)
231 ddepth = CV_MAT_DEPTH(type_);
232 else
233 ddepth = dst.fixedType() ? dst.depth() : sdepth;
234
235 bool noScale = std::fabs(alpha - 1) < DBL_EPSILON && std::fabs(beta) < DBL_EPSILON;
236 if (sdepth == ddepth && noScale)
237 {
238 copyTo(dst);
239 return;
240 }
241
242 CV_OCL_RUN(dims <= 2,
243 ocl_convertTo(*this, dst, ddepth, noScale, alpha, beta))
244#endif // HAVE_OPENCL
245
246 UMat src = *this; // Fake reference to itself.
247 // Resolves issue 8693 in case of src == dst.
248 Mat m = getMat(ACCESS_READ);
249 m.convertTo(dst, type_, alpha, beta);
250 (void)src;
251}
