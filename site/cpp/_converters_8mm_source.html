41+ (Mat*)vector_Point2d_to_Mat:(NSArray<Point2d*>*)pts {
42 Mat* res = [[Mat alloc] initWithRows:(int)pts.count cols:1 type:CV_64FC2];
43 NSMutableArray<NSNumber*>* buff = [NSMutableArray arrayWithCapacity:pts.count*2];

45 [buff addObject:[NSNumber numberWithDouble:pt.x]];

47 }
48 [res put:0 col:0 data:buff];
