44bool SunRasterDecoder::readHeader()
45{
46 bool result = false;
47
48 if (!m_buf.empty())
49 m_strm.open(m_buf);
50 else
51 m_strm.open(m_filename);
52
53 if( !m_strm.isOpened()) return false;
54
55 try
56 {
57 m_strm.skip( 4 );
58 m_width = m_strm.getDWord();
59 m_height = m_strm.getDWord();
60 m_bpp = m_strm.getDWord();
61 int palSize = (m_bpp > 0 && m_bpp <= 8) ? (3*(1 << m_bpp)) : 0;
62
63 m_strm.skip( 4 );
64 // Read as plain integers first; validate before casting to enum types.
65 // Casting an out-of-range integer directly to an enum with no fixed
66 // underlying type is undefined behavior (C++11 §7.2/8). Reject invalid
67 // values early so no downstream code ever touches an ill-formed enum.
68 const int raw_encoding = (int)m_strm.getDWord();
69 const int raw_maptype = (int)m_strm.getDWord();
70 m_maplength = m_strm.getDWord();
71
72 if (raw_encoding < RAS_OLD || raw_encoding > RAS_FORMAT_RGB)
73 return false;
74 if (raw_maptype < RMT_NONE || raw_maptype > RMT_EQUAL_RGB)
75 return false;
76
77 m_encoding = (SunRasType)raw_encoding;
78 m_maptype = (SunRasMapType)raw_maptype;
79
80 if( m_width > 0 && m_height > 0 &&
81 (m_bpp == 1 || m_bpp == 8 || m_bpp == 24 || m_bpp == 32) &&
82 (m_encoding == RAS_OLD || m_encoding == RAS_STANDARD ||
83 (m_type == RAS_BYTE_ENCODED && m_bpp == 8) || m_type == RAS_FORMAT_RGB) &&
84 ((m_maptype == RMT_NONE && m_maplength == 0) ||
85 (m_maptype == RMT_EQUAL_RGB && m_maplength <= palSize && m_maplength > 0 && m_bpp <= 8)))
86 {
87 memset( m_palette, 0, sizeof(m_palette));
88
89 if( m_maplength != 0 )
90 {
91 uchar buffer[256*3];
92
93 if( m_strm.getBytes( buffer, m_maplength ) == m_maplength )
94 {
95 int i;
96 palSize = m_maplength/3;
97
98 for( i = 0; i < palSize; i++ )
99 {
100 m_palette[i].b = buffer[i + 2*palSize];
101 m_palette[i].g = buffer[i + palSize];
102 m_palette[i].r = buffer[i];
103 m_palette[i].a = 0;
104 }
105
106 m_type = IsColorPalette( m_palette, m_bpp ) ? CV_8UC3 : CV_8UC1;
107 m_offset = m_strm.getPos();
108
109 CV_Assert(m_offset == static_cast<int64_t>(32 + m_maplength));
110 result = true;
111 }
112 }
113 else
114 {
115 m_type = m_bpp > 8 ? CV_8UC3 : CV_8UC1;
116
117 if( CV_MAT_CN(m_type) == 1 )
118 FillGrayPalette( m_palette, m_bpp );
119
120 m_offset = m_strm.getPos();
121
122 CV_Assert(m_offset == static_cast<int64_t>(32 + m_maplength));
123 result = true;
124 }
125 }
126 }
127 catch(...)
128 {
129 }
130
131 if( !result )
132 {
133 m_offset = -1;
134 m_width = m_height = -1;
135 m_strm.close();
136 }
137 return result;
138}

141bool SunRasterDecoder::readData( Mat& img )
142{
143 bool color = img.channels() > 1;
144 uchar* data = img.ptr();
145 size_t step = img.step;
146 uchar gray_palette[256] = {0};
147 bool result = false;
148 int nch = color ? 3 : 1;
149
150 const RowPitchParams pitch_params = calculateRowPitch(m_width, m_bpp, 2, "SunRaster");
151 const int src_pitch = pitch_params.src_pitch;
152 const size_t bytes_per_row = pitch_params.bytes_per_row;
153 const int width3 = calculateRowSize(m_width, nch, "SunRaster");
154 int y;
155
156 if( m_offset < 0 || !m_strm.isOpened())
157 return false;
158
159 AutoBuffer<uchar> _src(src_pitch + 32);
160 uchar* src = _src.data();
161
162 if( !color && m_maptype == RMT_EQUAL_RGB )
163 CvtPaletteToGray( m_palette, gray_palette, 1 << m_bpp );
164
165 try
166 {
167 m_strm.setPos( m_offset );
168
169 switch( m_bpp )
170 {
171 /************************* 1 BPP ************************/
172 case 1:
173 if( m_type != RAS_BYTE_ENCODED )
174 {
175 for( y = 0; y < m_height; y++, data += step )
176 {
177 m_strm.getBytes( src, src_pitch );
178 if( color )
179 FillColorRow1( data, src, m_width, m_palette );
180 else
181 FillGrayRow1( data, src, m_width, gray_palette );
182 }
183 result = true;
184 }
185 else
186 {
187 uchar* line_end = src + bytes_per_row;
188 uchar* tsrc = src;
189 y = 0;
190
191 for(;;)
192 {
193 int max_count = (int)(line_end - tsrc);
194 int code = 0, len = 0, len1 = 0;
195
196 do
197 {
198 code = m_strm.getByte();
199 if( code == 0x80 )
200 {
201 len = m_strm.getByte();
202 if( len != 0 ) break;
203 }
204 tsrc[len1] = (uchar)code;
205 }
206 while( ++len1 < max_count );
207
208 tsrc += len1;
209
210 if( len > 0 ) // encoded mode
211 {
212 ++len;
213 code = m_strm.getByte();
214 if( len > line_end - tsrc )
215 {
216 CV_Error(Error::StsInternal, "");
217 goto bad_decoding_1bpp;
218 }
219
220 memset( tsrc, code, len );
221 tsrc += len;
222 }
223
224 if( tsrc >= line_end )
225 {
226 tsrc = src;
227 if( color )
228 FillColorRow1( data, src, m_width, m_palette );
229 else
230 FillGrayRow1( data, src, m_width, gray_palette );
231 data += step;
232 if( ++y >= m_height ) break;
233 }
234 }
235 result = true;
236bad_decoding_1bpp:
237 ;
238 }
239 break;
240 /************************* 8 BPP ************************/
241 case 8:
242 if( m_type != RAS_BYTE_ENCODED )
243 {
244 for( y = 0; y < m_height; y++, data += step )
245 {
246 m_strm.getBytes( src, src_pitch );
247 if( color )
248 FillColorRow8( data, src, m_width, m_palette );
249 else
250 FillGrayRow8( data, src, m_width, gray_palette );
251 }
252 result = true;
253 }
254 else // RLE-encoded
255 {
256 uchar* line_end = data + width3;
257 y = 0;
258
259 for(;;)
260 {
261 int max_count = (int)(line_end - data);
262 int code = 0, len = 0, len1;
263 uchar* tsrc = src;
264
265 do
266 {
267 code = m_strm.getByte();
268 if( code == 0x80 )
269 {
270 len = m_strm.getByte();
271 if( len != 0 ) break;
272 }
273 *tsrc++ = (uchar)code;
274 }
275 while( (max_count -= nch) > 0 );
276
277 len1 = (int)(tsrc - src);
278
279 if( len1 > 0 )
280 {
281 if( color )
282 FillColorRow8( data, src, len1, m_palette );
283 else
284 FillGrayRow8( data, src, len1, gray_palette );
285 data += len1*nch;
286 }
287
288 if( len > 0 ) // encoded mode
289 {
290 len = (len + 1)*nch;
291 code = m_strm.getByte();
292
293 if( color )
294 data = FillUniColor( data, line_end, validateToInt(step), width3,
295 y, m_height, len,
296 m_palette[code] );
297 else
298 data = FillUniGray( data, line_end, validateToInt(step), width3,
299 y, m_height, len,
300 gray_palette[code] );
301 if( y >= m_height )
302 break;
303 }
304
305 if( data == line_end )
306 {
307 if( m_strm.getByte() != 0 )
308 goto bad_decoding_end;
309 line_end += step;
310 data = line_end - width3;
311 if( ++y >= m_height ) break;
312 }
313 }
314
315 result = true;
316bad_decoding_end:
317 ;
318 }
319 break;
320 /************************* 24 BPP ************************/
321 case 24:
322 for( y = 0; y < m_height; y++, data += step )
323 {
324 m_strm.getBytes(src, src_pitch );
325
326 if( color )
327 {
328 if( m_type == RAS_FORMAT_RGB || m_use_rgb)
329 icvCvt_RGB2BGR_8u_C3R(src, 0, data, 0, Size(m_width,1) );
330 else
331 memcpy(data, src, std::min(step, (size_t)src_pitch));
332 }
333 else
334 {
335 icvCvt_BGR2Gray_8u_C3C1R(src, 0, data, 0, Size(m_width,1),
336 m_type == RAS_FORMAT_RGB ? 2 : 0 );
337 }
338 }
339 result = true;
340 break;
341 /************************* 32 BPP ************************/
342 case 32:
343 for( y = 0; y < m_height; y++, data += step )
344 {
345 /* hack: a0 b0 g0 r0 a1 b1 g1 r1 ... are written to src + 3,
346 so when we look at src + 4, we see b0 g0 r0 x b1 g1 g1 x ... */
347 m_strm.getBytes( src + 3, src_pitch );
348
349 if( color )
350 icvCvt_BGRA2BGR_8u_C4C3R( src + 4, 0, data, 0, Size(m_width,1),
351 (m_type == RAS_FORMAT_RGB || m_use_rgb) ? 2 : 0 );
352 else
353 icvCvt_BGRA2Gray_8u_C4C1R( src + 4, 0, data, 0, Size(m_width,1),
354 m_type == RAS_FORMAT_RGB ? 2 : 0 );
355 }
356 result = true;
357 break;
358 default:
359 CV_Error(Error::StsInternal, "");
360 }
361 }
362 catch( ... )
363 {
364 }
365
366 return result;
367}

388bool SunRasterEncoder::write( const Mat& img, const std::vector<int>& )
389{
390 bool result = false;
391 int y, width = img.cols, height = img.rows, channels = img.channels();
392 int fileStep = (width*channels + 1) & -2;
393 WMByteStream strm;
394
395 if (m_buf) {
396 if (!strm.open(*m_buf)) {
397 return false;
398 }
399 else {
400 m_buf->reserve(height * fileStep + 32);
401 }
402 }
403 else
404 strm.open(m_filename);
405
406 if( strm.isOpened() )
407 {
408 CHECK_WRITE(strm.putBytes( fmtSignSunRas, (int)strlen(fmtSignSunRas) ));
409 CHECK_WRITE(strm.putDWord( width ));
410 CHECK_WRITE(strm.putDWord( height ));
411 CHECK_WRITE(strm.putDWord( channels*8 ));
412 CHECK_WRITE(strm.putDWord( fileStep*height ));
413 CHECK_WRITE(strm.putDWord( RAS_STANDARD ));
414 CHECK_WRITE(strm.putDWord( RMT_NONE ));
415 CHECK_WRITE(strm.putDWord( 0 ));
416
417 for( y = 0; y < height; y++ )
418 CHECK_WRITE(strm.putBytes( img.ptr(y), fileStep ));
419
420 strm.close();
421 result = true;
422 }
423 return result;
424}
