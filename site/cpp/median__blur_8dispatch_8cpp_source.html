60static bool ocl_medianFilter(InputArray _src, OutputArray _dst, int m)
61{
62 size_t localsize[2] = { 16, 16 };
63 size_t globalsize[2];
64 int type = _src.type(), depth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type);
65
66 if ( !((depth == CV_8U || depth == CV_16U || depth == CV_16S || depth == CV_32F) && cn <= 4 && (m == 3 || m == 5)) )
67 return false;
68
69 Size imgSize = _src.size();
70 bool useOptimized = (1 == cn) &&
71 (size_t)imgSize.width >= localsize[0] * 8 &&
72 (size_t)imgSize.height >= localsize[1] * 8 &&
73 imgSize.width % 4 == 0 &&
74 imgSize.height % 4 == 0 &&
75 (ocl::Device::getDefault().isIntel());
76
77 cv::String kname = format( useOptimized ? "medianFilter%d_u" : "medianFilter%d", m) ;
78 cv::String kdefs = useOptimized ?
79 format("-D T=%s -D T1=%s -D T4=%s%d -D cn=%d -D USE_4OPT", ocl::typeToStr(type),
80 ocl::typeToStr(depth), ocl::typeToStr(depth), cn*4, cn)
81 :
82 format("-D T=%s -D T1=%s -D cn=%d", ocl::typeToStr(type), ocl::typeToStr(depth), cn) ;
83
84 ocl::Kernel k(kname.c_str(), ocl::imgproc::medianFilter_oclsrc, kdefs.c_str() );
85
86 if (k.empty())
87 return false;
88
89 UMat src = _src.getUMat();
90 _dst.create(src.size(), type);
91 UMat dst = _dst.getUMat();
92
93 k.args(ocl::KernelArg::ReadOnlyNoSize(src), ocl::KernelArg::WriteOnly(dst));
94
95 if( useOptimized )
96 {
97 globalsize[0] = DIVUP(src.cols / 4, localsize[0]) * localsize[0];
98 globalsize[1] = DIVUP(src.rows / 4, localsize[1]) * localsize[1];
99 }
100 else
101 {
102 globalsize[0] = (src.cols + localsize[0] + 2) / localsize[0] * localsize[0];
103 globalsize[1] = (src.rows + localsize[1] - 1) / localsize[1] * localsize[1];
104 }
105
106 return k.run(2, globalsize, localsize, false);
107}

109#undef DIVUP

186void medianBlur( InputArray _src0, OutputArray _dst, int ksize )
187{
188 CV_INSTRUMENT_REGION();
189
190 CV_Assert(!_src0.empty());
191
192 CV_Assert( (ksize % 2 == 1) && (_src0.dims() <= 2 ));
193
194 if( ksize <= 1 || _src0.empty() )
195 {
196 _src0.copyTo(_dst);
197 return;
198 }
199
200 CV_OCL_RUN(_dst.isUMat(),
201 ocl_medianFilter(_src0,_dst, ksize))
202
203 Mat src0 = _src0.getMat();
204 _dst.create( src0.size(), src0.type() );
205 Mat dst = _dst.getMat();
206
207 CALL_HAL(medianBlur, cv_hal_medianBlur, src0.data, src0.step, dst.data, dst.step, src0.cols, src0.rows, src0.depth(),
208 src0.channels(), ksize);
209
210 //CV_IPP_RUN_FAST(ipp_medianFilter(src0, dst, ksize));
211
212 CV_CPU_DISPATCH(medianBlur, (src0, dst, ksize),
213 CV_CPU_DISPATCH_MODES_ALL);
214}
