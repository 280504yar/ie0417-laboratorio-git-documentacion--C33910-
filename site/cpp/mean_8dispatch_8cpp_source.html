16Scalar mean(InputArray _src, InputArray _mask)
17{
18 CV_INSTRUMENT_REGION();
19
20 Mat src = _src.getMat(), mask = _mask.getMat();
21 CV_Assert( mask.empty() || mask.type() == CV_8U || mask.type() == CV_8S || mask.type() == CV_Bool);
22
23 int k, cn = src.channels(), depth = src.depth();
24 Scalar s = Scalar::all(0.0);
25
26 CV_Assert( cn <= 4 );
27
28 if (src.isContinuous() && mask.isContinuous())
29 {
30 CALL_HAL_RET2(meanStdDev, cv_hal_meanStdDev, s, src.data, 0, (int)src.total(), 1, src.type(),
31 &s[0], nullptr /*stddev*/, mask.data, 0);
32 }
33 else
34 {
35 if (src.dims <= 2)
36 {
37 CALL_HAL_RET2(meanStdDev, cv_hal_meanStdDev, s, src.data, src.step, src.cols, src.rows, src.type(),
38 &s[0], nullptr, mask.data, mask.step);
39 }
40 }
41
42 SumFunc func = getSumFunc(depth);
43
44 CV_Assert( func != 0 );
45
46 const Mat* arrays[] = {&src, &mask, 0};
47 uchar* ptrs[2] = {};
48 NAryMatIterator it(arrays, ptrs);
49 int total = (int)it.size, blockSize = total, partialBlockSize = 0;
50 int j, count = 0;
51 int _buf[CV_CN_MAX];
52 int* buf = (int*)&s[0];
53 bool partialSumIsInt = depth < CV_32S;
54 bool blockSum = partialSumIsInt || depth == CV_16F || depth == CV_16BF;
55 size_t esz = 0, nz0 = 0;
56
57 if( blockSum )
58 {
59 partialBlockSize = depth <= CV_8S ? (1 << 23) : (1 << 15);
60 blockSize = std::min(blockSize, partialBlockSize);
61 buf = _buf;
62 for( k = 0; k < cn; k++ )
63 buf[k] = 0;
64 esz = src.elemSize();
65 }
66
67 for( size_t i = 0; i < it.nplanes; i++, ++it )
68 {
69 for( j = 0; j < total; j += blockSize )
70 {
71 int bsz = std::min(total - j, blockSize);
72 int nz = func( ptrs[0], ptrs[1], (uchar*)buf, bsz, cn );
73 count += nz;
74 nz0 += nz;
75 if( blockSum && (count + blockSize >= partialBlockSize || (i+1 >= it.nplanes && j+bsz >= total)) )
76 {
77 if (partialSumIsInt) {
78 for( k = 0; k < cn; k++ )
79 {
80 s[k] += buf[k];
81 buf[k] = 0;
82 }
83 } else {
84 for( k = 0; k < cn; k++ )
85 {
86 s[k] += ((float*)buf)[k];
87 buf[k] = 0;
88 }
89 }
90 count = 0;
91 }
92 ptrs[0] += bsz*esz;
93 if( ptrs[1] )
94 ptrs[1] += bsz;
95 }
96 }
97 return s*(nz0 ? 1./nz0 : 0);
98}

108static bool ocl_meanStdDev( InputArray _src, OutputArray _mean, OutputArray _sdv, InputArray _mask )
109{
110 CV_INSTRUMENT_REGION_OPENCL();
111
112 bool haveMask = _mask.kind() != _InputArray::NONE;
113 int nz = haveMask ? -1 : (int)_src.total();
114 Scalar mean(0), stddev(0);
115 const int cn = _src.channels();
116 if (cn > 4)
117 return false;
118
119 {
120 int type = _src.type(), depth = CV_MAT_DEPTH(type);
121 bool doubleSupport = ocl::Device::getDefault().doubleFPConfig() > 0,
122 isContinuous = _src.isContinuous(),
123 isMaskContinuous = _mask.isContinuous();
124 const ocl::Device &defDev = ocl::Device::getDefault();
125 int groups = defDev.maxComputeUnits();
126 if (defDev.isIntel())
127 {
128 static const int subSliceEUCount = 10;
129 groups = (groups / subSliceEUCount) * 2;
130 }
131 size_t wgs = defDev.maxWorkGroupSize();
132
133 int ddepth = std::max(CV_32S, depth), sqddepth = std::max(CV_32F, depth),
134 dtype = CV_MAKE_TYPE(ddepth, cn),
135 sqdtype = CV_MAKETYPE(sqddepth, cn);
136 CV_Assert(!haveMask || _mask.type() == CV_8U || _mask.type() == CV_8S || _mask.type() == CV_Bool);
137
138 int wgs2_aligned = 1;
139 while (wgs2_aligned < (int)wgs)
140 wgs2_aligned <<= 1;
141 wgs2_aligned >>= 1;
142
143 if ( (!doubleSupport && depth == CV_64F) )
144 return false;
145
146 char cvt[2][50];
147 String opts = format("-D srcT=%s -D srcT1=%s -D dstT=%s -D dstT1=%s -D sqddepth=%d"
148 " -D sqdstT=%s -D sqdstT1=%s -D convertToSDT=%s -D cn=%d%s%s"
149 " -D convertToDT=%s -D WGS=%d -D WGS2_ALIGNED=%d%s%s",
150 ocl::typeToStr(type), ocl::typeToStr(depth),
151 ocl::typeToStr(dtype), ocl::typeToStr(ddepth), sqddepth,
152 ocl::typeToStr(sqdtype), ocl::typeToStr(sqddepth),
153 ocl::convertTypeStr(depth, sqddepth, cn, cvt[0], sizeof(cvt[0])),
154 cn, isContinuous ? " -D HAVE_SRC_CONT" : "",
155 isMaskContinuous ? " -D HAVE_MASK_CONT" : "",
156 ocl::convertTypeStr(depth, ddepth, cn, cvt[1], sizeof(cvt[1])),
157 (int)wgs, wgs2_aligned, haveMask ? " -D HAVE_MASK" : "",
158 doubleSupport ? " -D DOUBLE_SUPPORT" : "");
159
160 ocl::Kernel k("meanStdDev", ocl::core::meanstddev_oclsrc, opts);
161 if (k.empty())
162 return false;
163
164 int dbsize = groups * ((haveMask ? CV_ELEM_SIZE1(CV_32S) : 0) +
165 CV_ELEM_SIZE(sqdtype) + CV_ELEM_SIZE(dtype));
166 UMat src = _src.getUMat(), db(1, dbsize, CV_8UC1), mask = _mask.getUMat();
167
168 ocl::KernelArg srcarg = ocl::KernelArg::ReadOnlyNoSize(src),
169 dbarg = ocl::KernelArg::PtrWriteOnly(db),
170 maskarg = ocl::KernelArg::ReadOnlyNoSize(mask);
171
172 if (haveMask)
173 k.args(srcarg, src.cols, (int)src.total(), groups, dbarg, maskarg);
174 else
175 k.args(srcarg, src.cols, (int)src.total(), groups, dbarg);
176
177 size_t globalsize = groups * wgs;
178
179 if(!k.run(1, &globalsize, &wgs, false))
180 return false;
181
182 typedef Scalar (* part_sum)(Mat m);
183 part_sum funcs[3] = { ocl_part_sum<int>, ocl_part_sum<float>, ocl_part_sum<double> };
184 Mat dbm = db.getMat(ACCESS_READ);
185
186 mean = funcs[ddepth - CV_32S](Mat(1, groups, dtype, dbm.ptr()));
187 stddev = funcs[sqddepth - CV_32S](Mat(1, groups, sqdtype, dbm.ptr() + groups * CV_ELEM_SIZE(dtype)));
188
189 if (haveMask)
190 nz = saturate_cast<int>(funcs[0](Mat(1, groups, CV_32SC1, dbm.ptr() +
191 groups * (CV_ELEM_SIZE(dtype) +
192 CV_ELEM_SIZE(sqdtype))))[0]);
193 }
194
195 double total = nz != 0 ? 1.0 / nz : 0;
196 int k, j;
197 for (int i = 0; i < cn; ++i)
198 {
199 mean[i] *= total;
200 stddev[i] = std::sqrt(std::max(stddev[i] * total - mean[i] * mean[i] , 0.));
201 }
202
203 for( j = 0; j < 2; j++ )
204 {
205 const double * const sptr = j == 0 ? &mean[0] : &stddev[0];
206 _OutputArray _dst = j == 0 ? _mean : _sdv;
207 if( !_dst.needed() )
208 continue;
209
210 if( !_dst.fixedSize() )
211 _dst.create(cn, 1, CV_64F, -1, true);
212 Mat dst = _dst.getMat();
213 int dcn = (int)dst.total();
214 CV_Assert( dst.type() == CV_64F && dst.isContinuous() &&
215 (dst.cols == 1 || dst.rows == 1) && dcn >= cn );
216 double* dptr = dst.ptr<double>();
217 for( k = 0; k < cn; k++ )
218 dptr[k] = sptr[k];
219 for( ; k < dcn; k++ )
220 dptr[k] = 0;
221 }
222
223 return true;
224}

227void meanStdDev(InputArray _src, OutputArray _mean, OutputArray _sdv, InputArray _mask)
228{
229 CV_INSTRUMENT_REGION();
230
231 CV_Assert(!_src.empty());
232 CV_Assert( _mask.empty() || _mask.type() == CV_8U || _mask.type() == CV_8S || _mask.type() == CV_Bool );
233
234 CV_OCL_RUN(OCL_PERFORMANCE_CHECK(_src.isUMat()) && _src.dims() <= 2,
235 ocl_meanStdDev(_src, _mean, _sdv, _mask))
236
237 Mat src = _src.getMat(), mask = _mask.getMat();
238
239 CV_Assert(mask.empty() || ((mask.type() == CV_8U || mask.type() == CV_8S || mask.type() == CV_Bool) && src.size == mask.size));
240
241 int k, cn = src.channels(), depth = src.depth();
242 Mat mean_mat, stddev_mat;
243
244 if(_mean.needed())
245 {
246 if( !_mean.fixedSize() )
247 _mean.create(cn, 1, CV_64F, -1, true);
248
249 mean_mat = _mean.getMat();
250 int dcn = (int)mean_mat.total();
251 CV_Assert( mean_mat.type() == CV_64F && mean_mat.isContinuous() &&
252 (mean_mat.cols == 1 || mean_mat.rows == 1) && dcn >= cn );
253
254 double* dptr = mean_mat.ptr<double>();
255 for(k = cn ; k < dcn; k++ )
256 dptr[k] = 0;
257 }
258
259 if (_sdv.needed())
260 {
261 if( !_sdv.fixedSize() )
262 _sdv.create(cn, 1, CV_64F, -1, true);
263
264 stddev_mat = _sdv.getMat();
265 int dcn = (int)stddev_mat.total();
266 CV_Assert( stddev_mat.type() == CV_64F && stddev_mat.isContinuous() &&
267 (stddev_mat.cols == 1 || stddev_mat.rows == 1) && dcn >= cn );
268
269 double* dptr = stddev_mat.ptr<double>();
270 for(k = cn ; k < dcn; k++ )
271 dptr[k] = 0;
272
273 }
274
275 if (src.isContinuous() && mask.isContinuous())
276 {
277 CALL_HAL(meanStdDev, cv_hal_meanStdDev, src.data, 0, (int)src.total(), 1, src.type(),
278 _mean.needed() ? mean_mat.ptr<double>() : nullptr,
279 _sdv.needed() ? stddev_mat.ptr<double>() : nullptr,
280 mask.data, 0);
281 }
282 else
283 {
284 if (src.dims <= 2)
285 {
286 CALL_HAL(meanStdDev, cv_hal_meanStdDev, src.data, src.step, src.cols, src.rows, src.type(),
287 _mean.needed() ? mean_mat.ptr<double>() : nullptr,
288 _sdv.needed() ? stddev_mat.ptr<double>() : nullptr,
289 mask.data, mask.step);
290 }
291 }
292
293 SumSqrFunc func = getSumSqrFunc(depth);
294
295 CV_Assert( func != 0 );
296
297 const Mat* arrays[] = {&src, &mask, 0};
298 uchar* ptrs[2] = {};
299 NAryMatIterator it(arrays, ptrs);
300 int total = (int)it.size, blockSize = total, partialBlockSize = 0;
301 int j;
302 int64_t count = 0, nz0 = 0;
303 double _buf[CV_CN_MAX*4];
304 double *s = _buf, *sq = s + cn;
305 int *sbuf = (int*)s, *sqbuf = (int*)sq;
306 bool partialSumIsInt = depth < CV_32S;
307 bool blockSum = partialSumIsInt || depth == CV_16F || depth == CV_16BF;
308 bool blockSqSum = depth <= CV_8S;
309 size_t esz = 0;
310
311 for( k = 0; k < cn; k++ )
312 s[k] = sq[k] = 0;
313
314 if( blockSum )
315 {
316 partialBlockSize = 1 << 15;
317 blockSize = std::min(blockSize, partialBlockSize);
318 sbuf = (int*)(sq + cn);
319 if( blockSqSum )
320 sqbuf = sbuf + cn;
321 for( k = 0; k < cn; k++ )
322 sbuf[k] = sqbuf[k] = 0;
323 esz = src.elemSize();
324 }
325
326 for( size_t i = 0; i < it.nplanes; i++, ++it )
327 {
328 for( j = 0; j < total; j += blockSize )
329 {
330 int bsz = std::min(total - j, blockSize);
331 int nz = func( ptrs[0], ptrs[1], (uchar*)sbuf, (uchar*)sqbuf, bsz, cn );
332 count += nz;
333 nz0 += nz;
334 if( blockSum && (count + blockSize >= partialBlockSize || (i+1 >= it.nplanes && j+bsz >= total)) )
335 {
336 if (partialSumIsInt) {
337 for( k = 0; k < cn; k++ )
338 {
339 s[k] += sbuf[k];
340 sbuf[k] = 0;
341 }
342 } else {
343 for( k = 0; k < cn; k++ )
344 {
345 s[k] += ((float*)sbuf)[k];
346 sbuf[k] = 0;
347 }
348 }
349 if( blockSqSum )
350 {
351 for( k = 0; k < cn; k++ )
352 {
353 sq[k] += sqbuf[k];
354 sqbuf[k] = 0;
355 }
356 }
357 count = 0;
358 }
359 ptrs[0] += bsz*esz;
360 if( ptrs[1] )
361 ptrs[1] += bsz;
362 }
363 }
364
365 double scale = nz0 ? 1./nz0 : 0.;
366 for( k = 0; k < cn; k++ )
367 {
368 s[k] *= scale;
369 sq[k] = std::sqrt(std::max(sq[k]*scale - s[k]*s[k], 0.));
370 }
371
372 if (_mean.needed())
373 {
374 const double* sptr = s;
375 double* dptr = mean_mat.ptr<double>();
376 for( k = 0; k < cn; k++ )
377 dptr[k] = sptr[k];
378 }
379
380 if (_sdv.needed())
381 {
382 const double* sptr = sq;
383 double* dptr = stddev_mat.ptr<double>();
384 for( k = 0; k < cn; k++ )
385 dptr[k] = sptr[k];
386 }
387}
