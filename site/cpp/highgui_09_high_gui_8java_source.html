107 public static int waitKey(int delay) {
108 // Reset control values
109 latch = new CountDownLatch(1);
110 n_closed_windows = 0;
111 pressedKey = -1;
112
113 // If there are no windows to be shown return
114 if (windows.isEmpty()) {
115 throw new IllegalStateException("No windows created. Call imshow() first");
116 }
117
118 // Remove the unused windows
119 Iterator<Map.Entry<String,
120 ImageWindow>> iter = windows.entrySet().iterator();
121 while (iter.hasNext()) {
122 Map.Entry<String,
123 ImageWindow> entry = iter.next();
124 ImageWindow win = entry.getValue();
125 if (win.alreadyUsed) {
126 iter.remove();
127 win.frame.dispose();
128 }
129 }
130
131 // (if) Create (else) Update frame
132 for (ImageWindow win : windows.values()) {
133
134 if (win.img != null) {
135
136 ImageIcon icon = new ImageIcon(toBufferedImage(win.img));
137
138 if (win.lbl == null) {
139 JFrame frame = createJFrame(win.name, win.flag);
140 JLabel lbl = new JLabel(icon);
141 win.setFrameLabelVisible(frame, lbl);
142 } else {
143 win.lbl.setIcon(icon);
144 }
145 } else {
146 throw new IllegalStateException("No image set for window: \"" + win.name + "\". Call imshow() first");
147 }
148 }
149
150 try {
151 if (delay == 0) {
152 latch.await();
153 } else {
154 latch.await(delay, TimeUnit.MILLISECONDS);
155 }
156 } catch (InterruptedException e) {
157 e.printStackTrace();
158 Thread.currentThread().interrupt();
159 }
160
161 // Set all windows as already used
162 for (ImageWindow win : windows.values())
163 win.alreadyUsed = true;
164
165 return pressedKey;
166 }
