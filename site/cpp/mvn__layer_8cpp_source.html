71 MVNLayerImpl(const LayerParams& params)
72 {
73 setParamsFrom(params);
74
75 // Caffe params
76 normVariance = params.get<bool>("normalize_variance", true);
77 acrossChannels = params.get<bool>("across_channels", false);
78 eps = params.get<double>("eps", 1e-9);
79
80 fuse_batch_norm = false;
81 fuse_relu = false;
82 relu_slope = 0.f;
83 zeroDev = false;

92 Ptr<ReLULayer> activ_relu;

94 bool fuse_relu;
95 bool zeroDev; // TODO: Doesn't considered in Intel's Inference Engine backend.
96 bool setActivation(const Ptr<ActivationLayer>& layer) CV_OVERRIDE
97 {
98 if (!layer.empty() && !fuse_relu && !fuse_batch_norm)
99 {
100 layer->getScaleShift(scale, shift);
101 fuse_batch_norm = !scale.empty() || !shift.empty();
102 return fuse_batch_norm;
103 }
104
105 if (!layer.empty() && preferableTarget == DNN_TARGET_OPENCL)
106 {
107 activ_relu = layer.dynamicCast<ReLULayer>();
108 if( !activ_relu.empty() )
109 relu_slope = activ_relu->negativeSlope;
110 }
111 fuse_relu = !activ_relu.empty();
112 return fuse_relu;
113 }

142 bool fast_forward_ocl(std::vector<UMat> &inputs, std::vector<UMat> &outputs)
143 {
144 if (umat_scale.empty() && !scale.empty())
145 scale.copyTo(umat_scale);
146 if (umat_shift.empty() && !shift.empty())
147 shift.copyTo(umat_shift);
148 UMat& bnorm_weight = umat_scale;
149 UMat& bnorm_bias = umat_shift;
150
151 const unsigned LOCAL_SIZE = 128;
152 bool use_half = (inputs[0].depth() == CV_16F);
153 String opts = format(" -DT=%s -DT4=%s -Dconvert_T=%s -DLOCAL_SIZE=%u", use_half ? "half" : "float",
154 use_half ? "half4" : "float4", use_half ? "convert_half4" : "convert_float4",
155 LOCAL_SIZE
156 );
157
158 int splitDim = (acrossChannels) ? 1 : 2;
159 for (size_t inpIdx = 0; inpIdx < inputs.size(); inpIdx++)
160 {
161 UMat &inpMat = inputs[inpIdx];
162 UMat &outMat = outputs[inpIdx];
163 int newRows = total(shape(inpMat), 0, splitDim);
164 CV_Assert(newRows != 0);
165
166 MatShape s = shape(newRows, inpMat.total() / newRows);
167 UMat meanMat = UMat(s[0], 1, (use_half) ? CV_16F : CV_32F);
168 UMat tmpMat = UMat(s[0], s[1], CV_32F);
169 float alpha = 1.0f / s[1];
170
171 String buildopt = "-DNUM=4" + opts;
172 ocl::Kernel k("mean_fuse4", ocl::dnn::mvn_oclsrc, buildopt + " -DKERNEL_MEAN_FUSE");
173 size_t localsize[] = { LOCAL_SIZE };
174 size_t groups = std::max<size_t>(1, s[0] / 4);
175 size_t globalsize[] = { groups * localsize[0] };
176
177
178 int argId = 0;
179 k.set(argId++, ocl::KernelArg::PtrReadOnly(inpMat));
180 k.set(argId++, (int)s[1]);
181 k.set(argId++, alpha);
182 k.set(argId++, ocl::KernelArg::PtrWriteOnly(meanMat));
183 k.set(argId++, ocl::KernelArg::PtrWriteOnly(tmpMat));
184 bool ret = k.run(1, globalsize, localsize, false);
185 if (!ret)
186 return false;
187
188 buildopt += format(" %s %s", (fuse_batch_norm) ? "-DFUSE_BATCH_NORM" : "",
189 (fuse_relu) ? "-DFUSE_RELU" : "");
190
191 ocl::Kernel k1("mvn_fuse4", ocl::dnn::mvn_oclsrc, buildopt + " -DKERNEL_MVN_FUSE");
192 argId = 0;
193 k1.set(argId++, ocl::KernelArg::PtrReadOnly(tmpMat));
194 k1.set(argId++, ocl::KernelArg::PtrReadOnly(inpMat));
195 k1.set(argId++, ocl::KernelArg::PtrReadOnly(meanMat));
196 k1.set(argId++, (int)s[1]);
197 k1.set(argId++, (float)alpha);
198 k1.set(argId++, (float)eps);
199 k1.set(argId++, (float)relu_slope);
200 k1.set(argId++, ocl::KernelArg::PtrReadOnly(bnorm_weight));
201 k1.set(argId++, ocl::KernelArg::PtrReadOnly(bnorm_bias));
202 k1.set(argId++, ocl::KernelArg::PtrWriteOnly(outMat));
203 ret = k1.run_(1, globalsize, localsize, false);
204 if (!ret)
205 return false;
206 }
207 return true;
208 }

210 bool forward_ocl(InputArrayOfArrays inputs_, OutputArrayOfArrays outputs_, OutputArrayOfArrays internals_)
211 {
212 if (umat_scale.empty() && !scale.empty())
213 scale.copyTo(umat_scale);
214 if (umat_shift.empty() && !shift.empty())
215 shift.copyTo(umat_shift);
216 UMat& bnorm_weight = umat_scale;
217 UMat& bnorm_bias = umat_shift;
218
219 std::vector<UMat> inputs;
220 std::vector<UMat> outputs;
221
222 inputs_.getUMatVector(inputs);
223 outputs_.getUMatVector(outputs);
224
225 int splitDim = (acrossChannels) ? 1 : 2;
226 int row_size = total(shape(inputs[0]), 0, splitDim);
227 int plane_size = total(shape(inputs[0]), splitDim);
228 if (normVariance && (row_size % 4 == 0) && (plane_size % 4 == 0))
229 return fast_forward_ocl(inputs, outputs);
230
231 if (inputs[0].depth() == CV_16F)
232 return false;
233
234 String opts = format(" -DT=float -DT4=float4 -Dconvert_T=convert_float4");
235
236 for (size_t inpIdx = 0; inpIdx < inputs.size(); inpIdx++)
237 {
238 UMat &inpMat = inputs[inpIdx];
239 UMat &outMat = outputs[inpIdx];
240 int newRows = total(shape(inpMat), 0, splitDim);
241 CV_Assert(newRows != 0);
242
243 MatShape s = shape(newRows, inpMat.total() / newRows);
244 UMat oneMat = UMat::ones(s[1], 1, CV_32F);
245 UMat meanMat = UMat(s[0], 1, CV_32F);
246 UMat devMat = UMat(s[0], 1, CV_32F);
247 UMat tmpMat = UMat(s[0], s[1], CV_32F);
248 float alpha = 1.0f / s[1];
249
250 bool ret = ocl4dnn::ocl4dnnGEMV<float>(ocl4dnn::CblasNoTrans, s[0], s[1], alpha,
251 inpMat, 0, oneMat, 0, 0.0f, meanMat, 0);
252 if (!ret)
253 return false;
254
255 int number = (s[1] % 8 == 0) ? 8 : ((s[1] % 4 == 0) ? 4 : 1);
256 size_t global[] = { (size_t)s[0], (size_t)(s[1] / number) };
257 String buildopt = format("-DNUM=%d", number) + opts;
258 if (normVariance)
259 {
260 String kname = format("calc_mean%d", number);
261 ocl::Kernel kernel(kname.c_str(), ocl::dnn::mvn_oclsrc, buildopt + " -DKERNEL_MEAN");
262 if (kernel.empty())
263 return false;
264
265 kernel.set(0, ocl::KernelArg::PtrReadOnly(inpMat));
266 kernel.set(1, (int)s[0]);
267 kernel.set(2, (int)s[1]);
268 kernel.set(3, ocl::KernelArg::PtrReadOnly(meanMat));
269 kernel.set(4, ocl::KernelArg::PtrWriteOnly(tmpMat));
270 ret = kernel.run(2, global, NULL, false);
271 if (!ret)
272 return false;
273
274 ret = ocl4dnn::ocl4dnnGEMV<float>(ocl4dnn::CblasNoTrans, s[0], s[1], alpha,
275 tmpMat, 0, oneMat, 0, 0.0f, devMat, 0);
276 if (!ret)
277 return false;
278 }
279
280 String kname = format("mvn%d", number);
281 buildopt += format("%s%s%s -DKERNEL_MVN", (normVariance) ? " -DNORM_VARIANCE" : "",
282 (fuse_batch_norm) ? " -DFUSE_BATCH_NORM" : "",
283 (fuse_relu) ? " -DFUSE_RELU" : "");
284 ocl::Kernel kernel1(kname.c_str(), ocl::dnn::mvn_oclsrc, buildopt);
285 if (kernel1.empty())
286 return false;
287 kernel1.set(0, ocl::KernelArg::PtrReadOnly(inpMat));
288 kernel1.set(1, (int)s[0]);
289 kernel1.set(2, (int)s[1]);
290 kernel1.set(3, (float)eps);
291 kernel1.set(4, ocl::KernelArg::PtrReadOnly(meanMat));
292 kernel1.set(5, ocl::KernelArg::PtrReadOnly(devMat));
293 kernel1.set(6, ocl::KernelArg::PtrReadOnly(bnorm_weight));
294 kernel1.set(7, ocl::KernelArg::PtrReadOnly(bnorm_bias));
295 kernel1.set(8, (int)inpMat.size[1]);
296 kernel1.set(9, (float)relu_slope);
297 kernel1.set(10, ocl::KernelArg::PtrWriteOnly(outMat));
298 ret = kernel1.run(2, global, NULL, false);
299 if (!ret)
300 return false;
301 }
302 return true;
303 }

306 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
307 {
308 CV_TRACE_FUNCTION();
309 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
310
311 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
312 forward_ocl(inputs_arr, outputs_arr, internals_arr))
313
314 if (inputs_arr.depth() == CV_16F)
315 {
316 forward_fallback(inputs_arr, outputs_arr, internals_arr);
317 return;
318 }
319
320 std::vector<Mat> inputs, outputs;
321 inputs_arr.getMatVector(inputs); // assume only one input
322 outputs_arr.getMatVector(outputs);
323
324 const auto &input = inputs[0];
325
326 if (fuse_batch_norm) { // channel-wise scale/bias of shape (C)
327 CV_CheckTrue(normVariance, "DNN/MVN: not supported");
328 fastNormChannel(input, scale, shift, outputs[0], eps);
329 } else {
330 size_t axis = acrossChannels ? 1 : 2;
331 fastNorm(input, outputs[0], eps, axis, normVariance);
332 }
333 }

337 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> >& inputs,
338 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
339 {
340 auto& ieInpNode = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
341 int64_t start_axis = acrossChannels ? 1 : 2;
342 std::vector<int64_t> axes_v(ieInpNode.get_shape().size() - start_axis);
343 std::iota(axes_v.begin(), axes_v.end(), start_axis);
344 auto axes = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{axes_v.size()}, axes_v.data());
345 auto mvn = std::make_shared<ov::op::v6::MVN>(ieInpNode, axes, normVariance, eps, ov::op::MVNEpsMode::INSIDE_SQRT);
346 return Ptr<BackendNode>(new InfEngineNgraphNode(mvn));
347 }

350#ifdef HAVE_CUDA
351 Ptr<BackendNode> initCUDA(
352 void *context_,
353 const std::vector<Ptr<BackendWrapper>>& inputs,
354 const std::vector<Ptr<BackendWrapper>>& outputs
355 ) override
356 {
357 auto context = reinterpret_cast<csl::CSLContext*>(context_);
358
359 cuda4dnn::MVNConfiguration config;
360 config.split_axis = acrossChannels ? 1 : 2;
361 config.normalize_variance = normVariance;
362 config.epsilon = eps;
363 config.input_shapes.resize(inputs.size());
364 for (int i = 0; i < inputs.size(); i++)
365 {
366 auto wrapper = inputs[i].dynamicCast<CUDABackendWrapper>();
367 auto shape = wrapper->getShape();
368 config.input_shapes[i].assign(std::begin(shape), std::end(shape));
369 }
370
371 return make_cuda_node<cuda4dnn::MVNOp>(preferableTarget, std::move(context->stream), config);
372 }

375 virtual int64 getFLOPS(const std::vector<MatShape> &inputs,
376 const std::vector<MatShape> &outputs) const CV_OVERRIDE
377 {
378 CV_UNUSED(outputs); // suppress unused variable warning
379 int64 flops = 0;
380 for(int i = 0; i < inputs.size(); i++)
381 {
382 flops += 6*total(inputs[i]) + 3*total(inputs[i], 0, normVariance ? 2 : 1);
383 }
384 return flops;
385 }
