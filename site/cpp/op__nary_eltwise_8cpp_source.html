19OpNary::OpNary(const OpNary::OPERATION _naryOpType, int _ninputs, int _max_ndims,
20 const std::vector<std::vector<int>> shapes, const std::vector<std::vector<size_t>> steps)
21 : naryOpType(_naryOpType), ninputs(_ninputs), max_ndims(_max_ndims)
22{
23 CV_Assert(ninputs > 1);
24
25 shapesBuf.resize((ninputs + 1) * max_ndims);
26 stepsBuf.resize((ninputs + 1) * max_ndims);
27 for (int i = 0; i <= ninputs; i++)
28 {
29 std::copy(shapes[i].begin(), shapes[i].end(), shapesBuf.data() + i * max_ndims);
30 std::copy(steps[i].begin(), steps[i].end(), stepsBuf.data() + i * max_ndims);
31 }
32
33 // TODO(VK): support more types of operation
34 switch(naryOpType) {
35 // case OPERATION::EQUAL:
36 // case OPERATION::GREATER:
37 // case OPERATION::GREATER_EQUAL:
38 // case OPERATION::LESS:
39 // case OPERATION::LESS_EQUAL:
40 // case OPERATION::POW:
41 // case OPERATION::BITSHIFT:
42 // case OPERATION::MOD:
43 case OPERATION::PROD:
44 case OPERATION::SUB:
45 case OPERATION::ADD:
46 case OPERATION::DIV:
47 // case OPERATION::AND:
48 // case OPERATION::OR:
49 // case OPERATION::XOR:
50 {
51 CV_Assert(ninputs == 2);
52 CV_Assert(max_ndims >= 2);
53 shaderType = kNaryShaderTypeBinary;
54 shader_name = "nary_eltwise_binary_forward_spv";
55
56 // TODO(VK): confirm if this makes any sense
57 nplanes = std::accumulate(shapesBuf.data(), shapesBuf.data() + max_ndims - 2, 1, [](int32_t a, int32_t b) { return a * b; } );
58 N2 = shapesBuf.data()[max_ndims - 2];
59 N1 = shapesBuf.data()[max_ndims - 1];
60 CV_LOG_DEBUG(NULL, "max_ndims="<<max_ndims<<", nplanes="<<nplanes<<", N2="<<N2<<", N1="<<N1);
61 break;
62 }
63 case OPERATION::WHERE:
64 {
65 CV_Assert(ninputs == 3);
66 CV_Assert(max_ndims >= 2);
67 shaderType = kNaryShaderTypeTrinary;
68 shader_name = "nary_eltwise_trinary_forward_spv";
69 break;
70 }
71 // case OPERATION::MAX:
72 // case OPERATION::MEAN:
73 // case OPERATION::MIN:
74 case OPERATION::SUM:
75 {
76 CV_Assert(max_ndims >= 2);
77 shaderType = kNaryShaderTypeNary;
78 shader_name = "nary_eltwise_nary_forward_spv";
79 break;
80 }
81 //TODO(VK) add other cases
82 default:
83 CV_Error(Error::StsNotImplemented, "Unsupported nary operation type");
84 }

102bool OpNary::binaryForward(std::vector<Tensor>& ins, std::vector<Tensor>& outs)
103{
104 std::vector<int32_t> param = {(int32_t)naryOpType, max_ndims};
105 std::vector<int32_t> paramSize = {(int32_t)param.size()};
106 std::vector<int32_t> dimSizes = {(ninputs + 1) * max_ndims};
107 std::vector<int32_t> actualSteps;
108
109 // TODO(VK): compute step for different dtype. Currently this is for kFormatFp32.
110 actualSteps.resize(stepsBuf.size());
111 std::transform(stepsBuf.data(), stepsBuf.data() + dimSizes[0], actualSteps.begin(), [](int32_t sz){ return sz / 4; });
112
113 Tensor paramTensor = Tensor(reinterpret_cast<const char *>(param.data()), paramSize, kFormatInt32, VK_BUFFER_USAGE_UNIFORM_BUFFER_BIT);
114 Tensor shapeTensor = Tensor(reinterpret_cast<const char *>(shapesBuf.data()), dimSizes, kFormatInt32, VK_BUFFER_USAGE_STORAGE_BUFFER_BIT);
115 Tensor stepTensor = Tensor(reinterpret_cast<const char *>(actualSteps.data()), dimSizes, kFormatInt32, VK_BUFFER_USAGE_STORAGE_BUFFER_BIT);
116
117 destTypes = {
118 VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, // input1
119 VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, // input2
120 VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, // out
121 VK_DESCRIPTOR_TYPE_UNIFORM_BUFFER, // param
122 VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, // shape
123 VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, // step
124 };
125
126
127 Ptr<Pipeline> pipeline = pipelineFactoryPtr->getPipeline(shader_name, destTypes);
128 Ptr<CommandBuffer> cmdBuffer = cmdPoolPtr->allocBuffer();
129 Ptr<Descriptor> desSet = pipeline->createSet();
130 VkCommandBuffer cmdBufferReal = cmdBuffer->get();
131
132 desSet->writeTensor(ins[0], 0);
133 desSet->writeTensor(ins[1], 1);
134 desSet->writeTensor(outs[0], 2);
135 desSet->writeTensor(paramTensor, 3);
136 desSet->writeTensor(shapeTensor, 4);
137 desSet->writeTensor(stepTensor, 5);
138
139 cmdBuffer->beginRecord();
140 pipeline->bind(cmdBufferReal, desSet->get());
141 vkCmdDispatch(cmdBufferReal, group_x_, group_y_, group_z_);
142 cmdBuffer->endRecord();
143 cmdPoolPtr->submitAndWait(cmdBufferReal);
144
145 return true;
146}

148bool OpNary::forward(std::vector<Tensor>& ins, std::vector<Tensor>& outs)
149{
150
151 firstForward();
152
153 // TODO(VK): Support more dtypes. Currently only kFormatFp32 is supported.
154 for (auto &tensor: ins)
155 {
156 CV_Assert(tensor.getFormat() == kFormatFp32);
157 }
158 for (auto &tensor: outs)
159 {
160 CV_Assert(tensor.getFormat() == kFormatFp32);
161 }
162
163 switch(shaderType) {
164 case kNaryShaderTypeBinary: {
165 return binaryForward(ins, outs);
166 break;
167 }
168 default:
169 CV_Error(Error::StsNotImplemented, "Unsupported shader type invoked.");
170 }
171
172 return true;
173}

175bool OpNary::computeGroupCount()
176{
177 if (shaderType == kNaryShaderTypeBinary)
178 {
179 group_x_ = nplanes; // parallelism at plane level
180 group_y_ = N2;
181 group_z_ = 1;
182 }
183 else
184 {
185 CV_Error(cv::Error::StsNotImplemented, "shader type is not supported at compute GroupCount.");
186 }
187
188 CV_Assert(group_x_ <= MAX_GROUP_COUNT_X);
189 CV_Assert(group_y_ <= MAX_GROUP_COUNT_Y);
190 CV_Assert(group_z_ <= MAX_GROUP_COUNT_Z);
191
192 return true;
193}
