23// out must be pre-allocated
24// pads[] should contains as many elements as inp.dims*2
25static void pad(const Mat& inp, const std::vector<int>& pads_, int mode_, const Mat& value, Mat& out)
26{
27 int inptype = inp.type();
28 MatShape inpshape_ = inp.shape();
29 MatShape outshape_ = out.shape();
30 double buf = 0;
31 Mat vbuf(1, 1, inptype, &buf);
32
33 int inpshape[PAD_MAX_DIMS];
34 int outshape[PAD_MAX_DIMS];
35 int pads[PAD_MAX_DIMS*2];
36 int64_t inpstep[PAD_MAX_DIMS];
37 int64_t outstep[PAD_MAX_DIMS];
38 std::vector<int> tab[PAD_MAX_DIMS];
39
40 int ndims = inp.dims, delta = PAD_MAX_DIMS - ndims;
41 int64_t esz = inp.elemSize();
42
43 CV_Assert(inp.isContinuous());
44 CV_Assert(out.isContinuous());
45 CV_Assert(inp.type() == out.type());
46 CV_Assert(esz == 1 || esz == 2 || esz == 4 || esz == 8);
47 CV_Assert(inp.dims == out.dims);
48 CV_Assert(inp.dims <= PAD_MAX_DIMS);
49
50 if (!value.empty()) {
51 CV_Assert(value.dims <= 2 && value.total() == 1 && value.channels() == 1);
52 tensorToScalar(value, inptype, &buf);
53 }
54
55 for (int i = 0; i < PAD_MAX_DIMS; i++) {
56 inpshape[i] = outshape[i] = 1;
57 pads[i] = pads[i + PAD_MAX_DIMS] = 0;
58 }
59
60 for (int i = 0; i < ndims; i++) {
61 inpshape[i+delta] = inpshape_[i];
62 outshape[i+delta] = outshape_[i];
63 pads[i+delta] = pads_[i];
64 pads[i+delta + PAD_MAX_DIMS] = pads_[i + ndims];
65
66 // initialize lookup table along the corresponding axis
67 int inpsz_i = inpshape_[i];
68 int outsz_i = outshape_[i];
69 tab[i+delta].resize(outsz_i);
70 int* tab_i = tab[i+delta].data();
71 int before = pads_[i];
72 for (int j = 0; j < outsz_i; j++)
73 tab_i[j] = borderInterpolate(j - before, inpsz_i, mode_);
74 }
75
76 for (int i = PAD_MAX_DIMS-1; i >= 0; i--) {
77 if (i == PAD_MAX_DIMS-1)
78 inpstep[i] = outstep[i] = 1;
79 else {
80 inpstep[i] = inpstep[i+1]*inpshape[i+1];
81 outstep[i] = outstep[i+1]*outshape[i+1];
82 }
83 }
84
85 int nplanes = outshape[0]*outshape[1]*outshape[2];
86
87 CV_Assert(!tab[4].empty());
88
89 #undef IMPL_PAD
90 #define IMPL_PAD(T) \
91 parallel_for_(Range(0, nplanes), [&](const Range& r) { \
92 int mode = mode_; \
93 int sz1 = outshape[1], sz2 = outshape[2], sz3 = outshape[3], sz4 = outshape[4]; \
94 const int* tab0 = tab[0].data(); \
95 const int* tab1 = tab[1].data(); \
96 const int* tab2 = tab[2].data(); \
97 const int* tab3 = tab[3].data(); \
98 const int* tab4 = tab[4].data(); \
99 const T* inpdata0 = (const T*)inp.data; \
100 T val0 = *reinterpret_cast<T*>(vbuf.data); \
101 T* outdata0 = (T*)out.data; \
102 int p0 = pads[PAD_MAX_DIMS-1], p1 = pads[PAD_MAX_DIMS*2-1]; \
103 int p0_ = std::max(p0, 0), p1_ = std::max(p1, 0); \
104 for (int plane = r.start; plane < r.end; plane++) { \
105 int plane_ = plane; \
106 int i2 = plane_ % sz2; \
107 plane_ /= sz2; \
108 int i1 = plane_ % sz1; \
109 int i0 = plane_ / sz1; \
110 int ii0 = tab0 ? tab0[i0] : i0; \
111 int ii1 = tab1 ? tab1[i1] : i1; \
112 int ii2 = tab2 ? tab2[i2] : i2; \
113 for (int i3 = 0; i3 < sz3; i3++) { \
114 int ii3 = tab3 ? tab3[i3] : i3; \
115 T* outdata = outdata0 + i0*outstep[0] + i1*outstep[1] + i2*outstep[2] + i3*outstep[3]; \
116 int i4 = 0; \
117 if ((ii0|ii1|ii2|ii3) < 0) { \
118 for (; i4 < sz4; i4++) \
119 outdata[i4] = val0; \
120 continue; \
121 } \
122 const T* inpdata = inpdata0 + ii0*inpstep[0] + ii1*inpstep[1] + ii2*inpstep[2] + ii3*inpstep[3]; \
123 if (mode == BORDER_CONSTANT) {\
124 for (; i4 < p0_; i4++) \
125 outdata[i4] = val0; \
126 } else { \
127 for (; i4 < p0_; i4++) \
128 outdata[i4] = inpdata[tab4[i4]]; \
129 } \
130 for (; i4 < sz4 - p1_; i4++) \
131 outdata[i4] = inpdata[i4 - p0]; \
132 if (mode == BORDER_CONSTANT) { \
133 for (; i4 < sz4; i4++) \
134 outdata[i4] = val0; \
135 } else { \
136 for (; i4 < sz4; i4++) \
137 outdata[i4] = inpdata[tab4[i4]]; \
138 } \
139 } \
140 } \
141 })
142
143 if (esz == 1) {
144 IMPL_PAD(uint8_t);
145 } else if (esz == 2) {
146 IMPL_PAD(uint16_t);
147 } else if (esz == 4) {
148 IMPL_PAD(uint32_t);
149 } else {
150 CV_Assert(esz == 8);
151 IMPL_PAD(uint64_t);
152 }
153}

155// Specialized semantic-padding path for block tensors with shape [N, C1, H, W, C0].
156// It handles semantic channel-axis padding directly in blocked representation.
157static void pad_block5d_semantic(const Mat& inp,
158 const std::vector<int>& semanticPads,
159 int mode_,
160 const Mat& value,
161 DataLayout origLayout,
162 Mat& out)
163{
164 CV_Assert(inp.dims == 5 && out.dims == 5);
165 CV_Assert(inp.shape().layout == DATA_LAYOUT_BLOCK && out.shape().layout == DATA_LAYOUT_BLOCK);
166 CV_Assert(semanticPads.size() == 8);
167 CV_Assert(mode_ == BORDER_CONSTANT);
168
169 int inptype = inp.type();
170 CV_Assert(inp.type() == out.type());
171 CV_Assert(inp.isContinuous() && out.isContinuous());
172
173 double buf = 0;
174 Mat vbuf(1, 1, inptype, &buf);
175 if (!value.empty()) {
176 CV_Assert(value.dims <= 2 && value.total() == 1 && value.channels() == 1);
177 tensorToScalar(value, inptype, &buf);
178 }
179
180 const int C0 = inp.shape()[4];
181 const int semantic_Cin = inp.shape().C;
182 CV_Assert(C0 > 0);
183
184 // Physical layout is always [N, C1, H, W, C0] for BLOCK;
185 // semantic channel axis depends on original model layout.
186 const int semanticChannelAxis = (origLayout == DATA_LAYOUT_NCHW) ? 1 : 3;
187 const int semanticHAxis = (origLayout == DATA_LAYOUT_NCHW) ? 2 : 1;
188 const int semanticWAxis = (origLayout == DATA_LAYOUT_NCHW) ? 3 : 2;
189 const int cPadBefore = semanticPads[semanticChannelAxis];
190
191 const int N = inp.size[0];
192 const int Hin = inp.size[2];
193 const int Win = inp.size[3];
194
195 const int Nout = out.size[0];
196 const int C1out = out.size[1];
197 const int Hout = out.size[2];
198 const int Wout = out.size[3];
199 const int phys_Cout = C1out * C0;
200
201 std::vector<int> tabN(Nout), tabH(Hout), tabW(Wout), tabC(phys_Cout);
202 for (int i = 0; i < Nout; i++)
203 tabN[i] = borderInterpolate(i - semanticPads[0], N, mode_);
204 for (int i = 0; i < Hout; i++)
205 tabH[i] = borderInterpolate(i - semanticPads[semanticHAxis], Hin, mode_);
206 for (int i = 0; i < Wout; i++)
207 tabW[i] = borderInterpolate(i - semanticPads[semanticWAxis], Win, mode_);
208 for (int i = 0; i < phys_Cout; i++)
209 tabC[i] = borderInterpolate(i - cPadBefore, semantic_Cin, mode_);
210
211 std::vector<int> blockL(C1out), blockR(C1out), blockInStart(C1out);
212 std::vector<int> seg0OutOff(C1out), seg0Len(C1out), seg0InC1(C1out), seg0InC0(C1out);
213 std::vector<int> seg1OutOff(C1out), seg1Len(C1out), seg1InC1(C1out);
214 for (int oc1 = 0; oc1 < C1out; oc1++) {
215 int base = oc1 * C0;
216 int l = 0;
217 while (l < C0 && tabC[base + l] < 0)
218 l++;
219 int r = C0;
220 while (r > l && tabC[base + r - 1] < 0)
221 r--;
222 blockL[oc1] = l;
223 blockR[oc1] = r;
224 blockInStart[oc1] = (l < r) ? tabC[base + l] : 0;
225
226 int len = r - l;
227 if (len > 0) {
228 int in_c = blockInStart[oc1];
229 int in_c1 = in_c / C0;
230 int in_c0 = in_c - in_c1 * C0;
231 int len0 = std::min(len, C0 - in_c0);
232 seg0OutOff[oc1] = l;
233 seg0Len[oc1] = len0;
234 seg0InC1[oc1] = in_c1;
235 seg0InC0[oc1] = in_c0;
236
237 int len1 = len - len0;
238 seg1OutOff[oc1] = l + len0;
239 seg1Len[oc1] = len1;
240 seg1InC1[oc1] = in_c1 + 1;
241 } else {
242 seg0OutOff[oc1] = 0;
243 seg0Len[oc1] = 0;
244 seg0InC1[oc1] = 0;
245 seg0InC0[oc1] = 0;
246 seg1OutOff[oc1] = 0;
247 seg1Len[oc1] = 0;
248 seg1InC1[oc1] = 0;
249 }
250 }
251
252 const size_t inStep0 = inp.step.p[0] / inp.elemSize();
253 const size_t inStep1 = inp.step.p[1] / inp.elemSize();
254 const size_t inStep2 = inp.step.p[2] / inp.elemSize();
255 const size_t inStep3 = inp.step.p[3] / inp.elemSize();
256
257 const size_t outStep0 = out.step.p[0] / out.elemSize();
258 const size_t outStep1 = out.step.p[1] / out.elemSize();
259 const size_t outStep2 = out.step.p[2] / out.elemSize();
260 const size_t outStep3 = out.step.p[3] / out.elemSize();
261
262 int wL = 0;
263 while (wL < Wout && tabW[wL] < 0)
264 wL++;
265 int wR = Wout;
266 while (wR > wL && tabW[wR - 1] < 0)
267 wR--;
268
269 const int nrows = Nout * C1out * Hout;
270
271#undef IMPL_PAD_BLOCK5D
272#define IMPL_PAD_BLOCK5D(T) \
273 parallel_for_(Range(0, nrows), [&](const Range& r) { \
274 const T* inpdata0 = reinterpret_cast<const T*>(inp.data); \
275 T* outdata0 = reinterpret_cast<T*>(out.data); \
276 T val0 = *reinterpret_cast<T*>(vbuf.data); \
277 bool val_is_zero = (val0 == (T)0); \
278 for (int row = r.start; row < r.end; row++) { \
279 int t = row; \
280 int oh = t % Hout; \
281 t /= Hout; \
282 int oc1 = t % C1out; \
283 int on = t / C1out; \
284 T* outrow = outdata0 + on*outStep0 + oc1*outStep1 + oh*outStep2; \
285 int in_n = tabN[on]; \
286 int in_h = tabH[oh]; \
287 if ((in_n | in_h) < 0) { \
288 for (int ow = 0; ow < Wout; ow++) { \
289 T* outptr = outrow + ow*outStep3; \
290 if (val_is_zero) \
291 memset(outptr, 0, (size_t)C0 * sizeof(T)); \
292 else { \
293 for (int oc0 = 0; oc0 < C0; oc0++) outptr[oc0] = val0; \
294 } \
295 } \
296 continue; \
297 } \
298 int l = blockL[oc1]; \
299 int rr = blockR[oc1]; \
300 int len0 = seg0Len[oc1]; \
301 int len1 = seg1Len[oc1]; \
302 for (int ow = 0; ow < wL; ow++) { \
303 T* outptr = outrow + ow*outStep3; \
304 if (val_is_zero) \
305 memset(outptr, 0, (size_t)C0 * sizeof(T)); \
306 else { \
307 for (int oc0 = 0; oc0 < C0; oc0++) outptr[oc0] = val0; \
308 } \
309 } \
310 for (int ow = wL; ow < wR; ow++) { \
311 T* outptr = outrow + ow*outStep3; \
312 if (l > 0) { \
313 if (val_is_zero) \
314 memset(outptr, 0, (size_t)l * sizeof(T)); \
315 else { \
316 for (int oc0 = 0; oc0 < l; oc0++) \
317 outptr[oc0] = val0; \
318 } \
319 } \
320 if (l < rr) { \
321 int in_w = tabW[ow]; \
322 const T* inbase = inpdata0 + in_n*inStep0 + seg0InC1[oc1]*inStep1 + in_h*inStep2 + in_w*inStep3; \
323 if (len0 > 0) { \
324 const T* src0 = inbase + seg0InC0[oc1]; \
325 memcpy(outptr + seg0OutOff[oc1], src0, (size_t)len0 * sizeof(T)); \
326 } \
327 if (len1 > 0) { \
328 const T* src1 = inbase + inStep1; \
329 memcpy(outptr + seg1OutOff[oc1], src1, (size_t)len1 * sizeof(T)); \
330 } \
331 } \
332 if (rr < C0) { \
333 if (val_is_zero) \
334 memset(outptr + rr, 0, (size_t)(C0 - rr) * sizeof(T)); \
335 else { \
336 for (int oc0 = rr; oc0 < C0; oc0++) \
337 outptr[oc0] = val0; \
338 } \
339 } \
340 } \
341 for (int ow = wR; ow < Wout; ow++) { \
342 T* outptr = outrow + ow*outStep3; \
343 if (val_is_zero) \
344 memset(outptr, 0, (size_t)C0 * sizeof(T)); \
345 else { \
346 for (int oc0 = 0; oc0 < C0; oc0++) outptr[oc0] = val0; \
347 } \
348 } \
349 } \
350 })
351
352 if (inp.elemSize() == 1) {
353 IMPL_PAD_BLOCK5D(uint8_t);
354 } else if (inp.elemSize() == 2) {
355 IMPL_PAD_BLOCK5D(uint16_t);
356 } else if (inp.elemSize() == 4) {
357 IMPL_PAD_BLOCK5D(uint32_t);
358 } else {
359 CV_Assert(inp.elemSize() == 8);
360 IMPL_PAD_BLOCK5D(uint64_t);
361 }
362}

373{
374 if (origLayout == DATA_LAYOUT_NCHW)

384static bool hasNonZeroChannelPad(const std::vector<int>& semanticPads,
385 int semanticNdims,
386 DataLayout origLayout)
387{
388 if ((int)semanticPads.size() != semanticNdims * 2)
389 return false;
390 int cAxis = origLayout == DATA_LAYOUT_NCHW ? 1 : semanticNdims - 1;
391 if (!(0 <= cAxis && cAxis < semanticNdims))
392 return false;
393 return semanticPads[cAxis] != 0 || semanticPads[cAxis + semanticNdims] != 0;
394}

396static bool canUseBlockWithChannelPad(const MatShape& inpshape,
397 const std::vector<int>& semanticPads,
398 int mode,
399 DataLayout origLayout)
400{
401 if (mode != BORDER_CONSTANT)
402 return false;
403 if ((int)semanticPads.size() != 8)
404 return false;
405
406 int cAxis = origLayout == DATA_LAYOUT_NCHW ? 1 : 3;
407 CV_Assert(cAxis >= 0);
408
409 int cBefore = semanticPads[cAxis];
410 int cAfter = semanticPads[cAxis + 4];
411
412 int semantic_Cin = 0;
413 if (inpshape.layout == DATA_LAYOUT_BLOCK && inpshape.dims == 5) {
414 semantic_Cin = inpshape.C;
415 } else if (inpshape.dims == 4) {
416 semantic_Cin = (origLayout == DATA_LAYOUT_NCHW) ? inpshape[1] : inpshape[3];
417 } else {
418 return false;
419 }
420
421 int semantic_Cout = semantic_Cin + cBefore + cAfter;
422 return semantic_Cout >= 0;
423}

425static MatShape getOutShapeBlockSemantic(const MatShape& inpshape,
426 const std::vector<int>& semanticPads,
427 DataLayout origLayout)
428{
429 CV_Assert(inpshape.layout == DATA_LAYOUT_BLOCK);
430 CV_Assert(inpshape.dims == 5);
431 CV_Assert(semanticPads.size() == 8);
432
433 const int semanticNdims = 4;
434 const int cAxis = origLayout == DATA_LAYOUT_NCHW ? 1 : 3;
435 const int C0 = inpshape[4];
436
437 MatShape outshape = inpshape;
438 for (int i = 0; i < semanticNdims; i++) {
439 int inSz = (i == cAxis) ? inpshape.C : inpshape[semanticToPhysicalAxis(i, semanticNdims, origLayout)];
440 int outSz = inSz + semanticPads[i] + semanticPads[i + semanticNdims];
441 CV_Assert(outSz >= 0);
442
443 if (i == cAxis) {
444 outshape[1] = (outSz + C0 - 1) / C0;
445 } else {
446 int physAxis = semanticToPhysicalAxis(i, semanticNdims, origLayout);
447 outshape[physAxis] = outSz;
448 }
449 }
450 outshape[4] = C0;
451 outshape.C = inpshape.C + semanticPads[cAxis] + semanticPads[cAxis + semanticNdims];
452 return outshape;
453}

455static bool mapPadsToInputLayout(const MatShape& inpshape,
456 const std::vector<int>& semanticPads,
457 std::vector<int>& pads,
458 DataLayout origLayout)
459{
460 int ndims = inpshape.dims;
461 DataLayout layout = inpshape.layout;
462
463 if (layout != DATA_LAYOUT_BLOCK) {
464 if ((int)semanticPads.size() != ndims*2)
465 return false;
466 pads = semanticPads;
467 return true;
468 }
469
470 if (ndims < 2 || ndims > PAD_MAX_DIMS)
471 return false;
472
473 int semanticNdims = ndims - 1;
474 if ((int)semanticPads.size() != semanticNdims*2)
475 return false;
476
477 int cAxis = origLayout == DATA_LAYOUT_NCHW ? 1 : semanticNdims - 1;
478
479 if (semanticPads[cAxis] != 0 || semanticPads[cAxis + semanticNdims] != 0)
480 return false;
481
482 pads.assign(ndims*2, 0);
483
484 for (int i = 0; i < semanticNdims; i++) {
485 int j = i;
486 if (origLayout == DATA_LAYOUT_NHWC) {
487 if (i == 0)
488 j = 0;
489 else if (i == semanticNdims - 1)
490 j = 1;
491 else
492 j = i + 1;
493 }
494 pads[j] = semanticPads[i];
495 pads[j + ndims] = semanticPads[i + semanticNdims];
496 }
497
498 pads[ndims - 1] = 0;
499 pads[ndims*2 - 1] = 0;
500 return true;
501}

553 bool getConstSemanticPads(std::vector<int>& semanticPads) const
554 {
555 size_t ninputs = this->inputs.size();
556 CV_Assert(1 <= ninputs && ninputs <= 4);
557
558 if (ninputs == 1) {
559 semanticPads = pads0;
560 return !semanticPads.empty();
561 }
562
563 Net::Impl* netimpl_ = getNetImpl(this);
564 if (!netimpl_ || !netimpl_->isConstArg(this->inputs[1]))
565 return false;
566
567 Mat padsTensor = netimpl_->argTensor(this->inputs[1]);
568 Mat axesTensor;
569 if (ninputs >= 4) {
570 if (!netimpl_->isConstArg(this->inputs[3]))
571 return false;
572 axesTensor = netimpl_->argTensor(this->inputs[3]);
573 }
574
575 int ndims = netimpl_->argData(this->inputs[0]).shape.dims;
576 if (ndims <= 0) {
577 if (axesTensor.empty())
578 ndims = int(padsTensor.total()/2);
579 else
580 return false;
581 }
582
583 if (ndims < 1 || ndims > PAD_MAX_DIMS)
584 return false;
585
586 getPads(ndims, padsTensor, axesTensor, semanticPads);
587 return true;
588 }

590 int getLayouts(const std::vector<DataLayout>& actualInputs,
591 std::vector<DataLayout>& desiredInputs,
592 const int requiredOutputs,
593 std::vector<DataLayout>& outputs) const CV_OVERRIDE
594 {
595 size_t ninputs = actualInputs.size();
596 CV_Assert(1 <= ninputs && ninputs <= 4);
597
598 desiredInputs.assign(ninputs, DATA_LAYOUT_UNKNOWN);
599 desiredInputs[0] = actualInputs[0];
600 outputs.assign(requiredOutputs, actualInputs[0]);
601
602 if (actualInputs[0] != DATA_LAYOUT_BLOCK)
603 return 0;
604
605 DataLayout origLayout = getOriginalLayout(this);
606 std::vector<int> semanticPads;
607 bool canUseBlock = getConstSemanticPads(semanticPads);
608 if (canUseBlock) {
609 int semanticNdims = int(semanticPads.size() / 2);
610 canUseBlock = (semanticNdims >= 1 && semanticNdims + 1 <= PAD_MAX_DIMS);
611 if (canUseBlock) {
612 bool hasChannelPad = hasNonZeroChannelPad(semanticPads, semanticNdims, origLayout);
613 if (hasChannelPad) {
614 // Channel padding in semantic space requires special block path.
615 // 4D semantic tensors, constant mode.
616 canUseBlock = (mode == BORDER_CONSTANT && semanticNdims == 4);
617 } else {
618 // Non-channel semantic padding can be represented by per-axis
619 // physical pads in block layout.
620 canUseBlock = true;
621 }
622 }
623 } else {
624 canUseBlock = false;
625 }
626
627 if (!canUseBlock) {
628 desiredInputs[0] = origLayout;
629 outputs.assign(requiredOutputs, DATA_LAYOUT_UNKNOWN);
630 } else {
631 desiredInputs[0] = DATA_LAYOUT_BLOCK;
632 outputs.assign(requiredOutputs, DATA_LAYOUT_BLOCK);
633 }
634 return 0;
635 }

637 void getPads(int ndims, const Mat& pads_, const Mat& axes_, std::vector<int>& pads) const
638 {
639 int atype = axes_.type(), ptype = pads_.type();
640 CV_Assert(ndims <= PAD_MAX_DIMS);
641
642 const int32_t* adata_i32 = nullptr;
643 const int64_t* adata_i64 = nullptr;
644 const int32_t* pdata_i32 = nullptr;
645 const int64_t* pdata_i64 = nullptr;
646
647 bool axismask[PAD_MAX_DIMS];
648 int naxes = !axes_.empty() ? (int)axes_.total() : ndims;
649
650 CV_Assert(pads_.dims == 1);
651 CV_Assert(ptype == CV_32S || ptype == CV_64S);
652
653 if (ptype == CV_32S)
654 pdata_i32 = reinterpret_cast<const int32_t*>(pads_.data);
655 else
656 pdata_i64 = reinterpret_cast<const int64_t*>(pads_.data);
657
658 if (!axes_.empty()) {
659 CV_Assert(axes_.dims == 1);
660 CV_Assert(atype == CV_32S || atype == CV_64S);
661 CV_Assert(pads_.total() == axes_.total()*2);
662 CV_Assert(axes_.total() <= (size_t)ndims);
663
664 if (atype == CV_32S)
665 adata_i32 = reinterpret_cast<const int32_t*>(axes_.data);
666 else
667 adata_i64 = reinterpret_cast<const int64_t*>(axes_.data);
668 } else {
669 CV_Assert(pads_.total() == (size_t)ndims*2);
670 }
671
672 pads.resize(ndims*2);
673
674 for (int i = 0; i < ndims; i++) {
675 pads[i] = pads[i+ndims] = 0;
676 axismask[i] = false;
677 }
678
679 for (int i = 0; i < naxes; i++) {
680 int a = adata_i32 ? (int)adata_i32[i] : adata_i64 ? (int)adata_i64[i] : i;
681 a = normalize_axis(a, ndims);
682 if (axismask[a]) {
683 CV_Error_(Error::StsBadArg, ("duplicate axis %d in Pad", a));
684 }
685 axismask[a] = true;
686 int p0 = pdata_i32 ? (int)pdata_i32[i] : pdata_i64 ? (int)pdata_i64[i] : 0;
687 int p1 = pdata_i32 ? (int)pdata_i32[i+naxes] : pdata_i64 ? (int)pdata_i64[i+naxes] : 0;
688 pads[a] = p0;
689 pads[a+ndims] = p1;
690 // p0, p1 can be positive, zero or even negative, according to ONNX specification.
691 // so we don't put any checks here.
692 }
693 }

695 MatShape getOutShape(const MatShape& inpshape, const std::vector<int>& pads) const
696 {
697 MatShape outshape = inpshape;
698 int ndims = inpshape.dims;
699 for (int i = 0; i < ndims; i++) {
700 outshape[i] += pads[i] + pads[i+ndims];
701 CV_Assert(outshape[i] >= 0);
702 }
703 return outshape;
704 }

773

775 Mat value(1, 1, CV_32F, &value0);
776 int inptype = inp.type();

779 std::vector<int> padsbuf;
780 DataLayout origLayout = getOriginalLayout(this);
781
782 if (ninputs >= 2) {

785 Mat axesTensor;

787 axesTensor = inputs_arr.getMat(3);
788 getPads(ndims, padsTensor, axesTensor, semanticPads);
789 if (ninputs >= 3)
790 value = inputs_arr.getMat(2);
791 } else {
792 semanticPads = pads0;
793 }
794
795 bool mapped = mapPadsToInputLayout(inpshape, semanticPads, padsbuf, origLayout);

799 outshape = getOutShape(inpshape, padsbuf);
800 else if (useBlockSemantic)
801 outshape = getOutShapeBlockSemantic(inpshape, semanticPads, origLayout);
802 else
803 CV_Error(Error::StsInternal, "Pad2: unexpected block layout with unsupported padding");
804
805 auto kind = outputs_arr.kind();
806 if (kind == _InputArray::STD_VECTOR_MAT) {
807 std::vector<Mat>& outs = outputs_arr.getMatVecRef();

811 pad(inp, padsbuf, mode, value, outs[0]);

817 outs[0].fit(outshape, inptype);

819 if (mapped)
820 pad(inp, padsbuf, mode, value, temp);
821 else
822 pad_block5d_semantic(inp, semanticPads, mode, value, origLayout, temp);
823 temp.copyTo(outs[0]);
824 } else {
825 CV_Error(Error::StsNotImplemented, "");
826 }
827 }
