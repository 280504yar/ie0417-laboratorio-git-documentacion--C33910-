53static inline void _drawKeypoint( InputOutputArray img, const KeyPoint& p, const Scalar& color, DrawMatchesFlags flags )
54{
55 CV_Assert( !img.empty() );
56 Point center( cvRound(p.pt.x * draw_multiplier), cvRound(p.pt.y * draw_multiplier) );
57
58 if( !!(flags & DrawMatchesFlags::DRAW_RICH_KEYPOINTS) )
59 {
60 int radius = cvRound(p.size/2 * draw_multiplier); // KeyPoint::size is a diameter
61
62 // draw the circles around keypoints with the keypoints size
63 circle( img, center, radius, color, 1, LINE_AA, draw_shift_bits );
64
65 // draw orientation of the keypoint, if it is applicable
66 if( p.angle != -1 )
67 {
68 float srcAngleRad = p.angle*(float)CV_PI/180.f;
69 Point orient( cvRound(cos(srcAngleRad)*radius ),
70 cvRound(sin(srcAngleRad)*radius )
71 );
72 line( img, center, center+orient, color, 1, LINE_AA, draw_shift_bits );
73 }
74#if 0
75 else
76 {
77 // draw center with R=1
78 int radius = 1 * draw_multiplier;
79 circle( img, center, radius, color, 1, LINE_AA, draw_shift_bits );
80 }
81#endif
82 }
83 else
84 {
85 // draw center with R=3
86 int radius = 3 * draw_multiplier;
87 circle( img, center, radius, color, 1, LINE_AA, draw_shift_bits );
88 }
89}

91void drawKeypoints( InputArray image, const std::vector<KeyPoint>& keypoints, InputOutputArray outImage,
92 const Scalar& _color, DrawMatchesFlags flags )
93{
94 CV_INSTRUMENT_REGION();
95
96 if( !(flags & DrawMatchesFlags::DRAW_OVER_OUTIMG) )
97 {
98 if (image.type() == CV_8UC3 || image.type() == CV_8UC4)
99 {
100 image.copyTo(outImage);
101 }
102 else if( image.type() == CV_8UC1 )
103 {
104 cvtColor( image, outImage, COLOR_GRAY2BGR );
105 }
106 else
107 {
108 CV_Error( Error::StsBadArg, "Incorrect type of input image: " + typeToString(image.type()) );
109 }
110 }
111
112 RNG& rng=theRNG();
113 bool isRandColor = _color == Scalar::all(-1);
114
115 CV_Assert( !outImage.empty() );
116 std::vector<KeyPoint>::const_iterator it = keypoints.begin(),
117 end = keypoints.end();
118 for( ; it != end; ++it )
119 {
120 Scalar color = isRandColor ? Scalar( rng(256), rng(256), rng(256), 255 ) : _color;
121 _drawKeypoint( outImage, *it, color, flags );
122 }
123}

125static void _prepareImage(InputArray src, const Mat& dst)
126{
127 CV_CheckType(src.type(), src.type() == CV_8UC1 || src.type() == CV_8UC3 || src.type() == CV_8UC4, "Unsupported source image");
128 CV_CheckType(dst.type(), dst.type() == CV_8UC3 || dst.type() == CV_8UC4, "Unsupported destination image");
129 const int src_cn = src.channels();
130 const int dst_cn = dst.channels();
131
132 if (src_cn == dst_cn)
133 src.copyTo(dst);
134 else if (src_cn == 1)
135 cvtColor(src, dst, dst_cn == 3 ? COLOR_GRAY2BGR : COLOR_GRAY2BGRA);
136 else if (src_cn == 3 && dst_cn == 4)
137 cvtColor(src, dst, COLOR_BGR2BGRA);
138 else if (src_cn == 4 && dst_cn == 3)
139 cvtColor(src, dst, COLOR_BGRA2BGR);
140 else
141 CV_Error(Error::StsInternal, "");
142}

144static void _prepareImgAndDrawKeypoints( InputArray img1, const std::vector<KeyPoint>& keypoints1,
145 InputArray img2, const std::vector<KeyPoint>& keypoints2,
146 InputOutputArray _outImg, Mat& outImg1, Mat& outImg2,
147 const Scalar& singlePointColor, DrawMatchesFlags flags )
148{
149 Mat outImg;
150 Size img1size = img1.size(), img2size = img2.size();
151 Size size( img1size.width + img2size.width, MAX(img1size.height, img2size.height) );
152 if( !!(flags & DrawMatchesFlags::DRAW_OVER_OUTIMG) )
153 {
154 outImg = _outImg.getMat();
155 if( size.width > outImg.cols || size.height > outImg.rows )
156 CV_Error( Error::StsBadSize, "outImg has size less than need to draw img1 and img2 together" );
157 outImg1 = outImg( Rect(0, 0, img1size.width, img1size.height) );
158 outImg2 = outImg( Rect(img1size.width, 0, img2size.width, img2size.height) );
159 }
160 else
161 {
162 const int cn1 = img1.channels(), cn2 = img2.channels();
163 const int out_cn = std::max(3, std::max(cn1, cn2));
164 _outImg.create(size, CV_MAKETYPE(img1.depth(), out_cn));
165 outImg = _outImg.getMat();
166 outImg = Scalar::all(0);
167 outImg1 = outImg( Rect(0, 0, img1size.width, img1size.height) );
168 outImg2 = outImg( Rect(img1size.width, 0, img2size.width, img2size.height) );
169
170 _prepareImage(img1, outImg1);
171 _prepareImage(img2, outImg2);
172 }
173
174 // draw keypoints
175 if( !(flags & DrawMatchesFlags::NOT_DRAW_SINGLE_POINTS) )
176 {
177 Mat _outImg1 = outImg( Rect(0, 0, img1size.width, img1size.height) );
178 drawKeypoints( _outImg1, keypoints1, _outImg1, singlePointColor, flags | DrawMatchesFlags::DRAW_OVER_OUTIMG );
179
180 Mat _outImg2 = outImg( Rect(img1size.width, 0, img2size.width, img2size.height) );
181 drawKeypoints( _outImg2, keypoints2, _outImg2, singlePointColor, flags | DrawMatchesFlags::DRAW_OVER_OUTIMG );
182 }
183}

185static inline void _drawMatch( InputOutputArray outImg, InputOutputArray outImg1, InputOutputArray outImg2 ,
186 const KeyPoint& kp1, const KeyPoint& kp2, const Scalar& matchColor, DrawMatchesFlags flags,
187 const int matchesThickness )
188{
189 RNG& rng = theRNG();
190 bool isRandMatchColor = matchColor == Scalar::all(-1);
191 Scalar color = isRandMatchColor ? Scalar( rng(256), rng(256), rng(256), 255 ) : matchColor;
192
193 _drawKeypoint( outImg1, kp1, color, flags );
194 _drawKeypoint( outImg2, kp2, color, flags );
195
196 Point2f pt1 = kp1.pt,
197 pt2 = kp2.pt,
198 dpt2 = Point2f( std::min(pt2.x+outImg1.size().width, float(outImg.size().width-1)), pt2.y );
199
200 line( outImg,
201 Point(cvRound(pt1.x*draw_multiplier), cvRound(pt1.y*draw_multiplier)),
202 Point(cvRound(dpt2.x*draw_multiplier), cvRound(dpt2.y*draw_multiplier)),
203 color, matchesThickness, LINE_AA, draw_shift_bits );
204}

220void drawMatches( InputArray img1, const std::vector<KeyPoint>& keypoints1,
221 InputArray img2, const std::vector<KeyPoint>& keypoints2,
222 const std::vector<DMatch>& matches1to2, InputOutputArray outImg,
223 const int matchesThickness, const Scalar& matchColor,
224 const Scalar& singlePointColor, const std::vector<char>& matchesMask,
225 DrawMatchesFlags flags )
226{
227 if( !matchesMask.empty() && matchesMask.size() != matches1to2.size() )
228 CV_Error( Error::StsBadSize, "matchesMask must have the same size as matches1to2" );
229
230 Mat outImg1, outImg2;
231 _prepareImgAndDrawKeypoints( img1, keypoints1, img2, keypoints2,
232 outImg, outImg1, outImg2, singlePointColor, flags );
233
234 // draw matches
235 for( size_t m = 0; m < matches1to2.size(); m++ )
236 {
237 if( matchesMask.empty() || matchesMask[m] )
238 {
239 int i1 = matches1to2[m].queryIdx;
240 int i2 = matches1to2[m].trainIdx;
241 CV_Assert(i1 >= 0 && i1 < static_cast<int>(keypoints1.size()));
242 CV_Assert(i2 >= 0 && i2 < static_cast<int>(keypoints2.size()));
243
244 const KeyPoint &kp1 = keypoints1[i1], &kp2 = keypoints2[i2];
245 _drawMatch( outImg, outImg1, outImg2, kp1, kp2, matchColor, flags, matchesThickness );
246 }
247 }

251void drawMatches( InputArray img1, const std::vector<KeyPoint>& keypoints1,
252 InputArray img2, const std::vector<KeyPoint>& keypoints2,
253 const std::vector<std::vector<DMatch> >& matches1to2, InputOutputArray outImg,
254 const Scalar& matchColor, const Scalar& singlePointColor,

259
260 Mat outImg1, outImg2;

262 outImg, outImg1, outImg2, singlePointColor, flags );
263
264 // draw matches
265 for( size_t i = 0; i < matches1to2.size(); i++ )
266 {
267 for( size_t j = 0; j < matches1to2[i].size(); j++ )
268 {
269 int i1 = matches1to2[i][j].queryIdx;
270 int i2 = matches1to2[i][j].trainIdx;
271 if( matchesMask.empty() || matchesMask[i][j] )
272 {
273 const KeyPoint &kp1 = keypoints1[i1], &kp2 = keypoints2[i2];
274 _drawMatch( outImg, outImg1, outImg2, kp1, kp2, matchColor, flags, 1 );
275 }
