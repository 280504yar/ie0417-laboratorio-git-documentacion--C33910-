26AutoPadding getAutoPadding(const LayerParams& params)
27{
28 std::string auto_pad = params.get<std::string>("auto_pad", "NOTSET");
29 std::string pad_mode = params.get<std::string>("pad_mode", "");
30 if (pad_mode == "SAME")
31 return AUTO_PAD_SAME_UPPER;
32 if (pad_mode == "VALID")
33 return AUTO_PAD_VALID;
34 if (auto_pad == "NOTSET")
35 return AUTO_PAD_NONE;
36 if (auto_pad == "SAME_UPPER")
37 return AUTO_PAD_SAME_UPPER;
38 if (auto_pad == "SAME_LOWER")
39 return AUTO_PAD_SAME_LOWER;
40 if (auto_pad != "VALID") {
41 CV_Error_(Error::StsBadArg, ("invalid auto_pad value '%s'", auto_pad.c_str()));
42 }
43 return AUTO_PAD_VALID;
44}

46// computes shape of the output tensor of convolution
47// (including depth-wise convolution), max pooling or average pooling operations
48// computes shape of the output tensor of convolution
49// (including depth-wise convolution), max pooling or average pooling operations
50MatShape convInferShape(const MatShape& inpShape, const MatShape& wshape,
51 const std::vector<int>& kernelShape, int ngroups,
52 const std::vector<int>& strides,
53 const std::vector<int>& dilations,
54 const std::vector<int>& pads,
55 AutoPadding autoPad, bool ceilMode)
56{
57 bool blockLayout = (inpShape.layout == DATA_LAYOUT_BLOCK);
58 int ndims = inpShape.dims;
59 size_t nspatialdims = (size_t)(ndims - 2 - int(blockLayout));
60 MatShape outshape = inpShape;
61 int kshape[MatShape::MAX_DIMS];
62
63 if (!kernelShape.empty()) {
64 size_t kshape_size = kernelShape.size();
65 CV_Assert(kshape_size == nspatialdims || kshape_size == nspatialdims+2);
66 for (size_t i = 0; i < nspatialdims; i++)
67 kshape[i] = kernelShape[kshape_size - nspatialdims + i];
68 } else {
69 CV_Assert(!wshape.empty() && wshape.dims == nspatialdims + 2);
70 for (size_t i = 0; i < nspatialdims; i++)
71 kshape[i] = wshape[wshape.dims - nspatialdims + i];
72 }
73
74 if (ngroups == 0 || wshape.empty()) {
75 outshape[1] = inpShape[1];
76 } else if (blockLayout) {
77 int C0 = inpShape[ndims-1];
78 outshape[1] = (wshape[0] + C0 - 1)/C0;
79 } else {
80 outshape[1] = wshape[0];
81 }
82
83 CV_Assert(strides.empty() || strides.size() == nspatialdims);
84 CV_Assert(dilations.empty() || dilations.size() == nspatialdims);
85 CV_Assert(autoPad == AUTO_PAD_NONE || pads.empty());
86 CV_Assert(pads.empty() || pads.size() == nspatialdims*2);
87
88 for (size_t i = 0; i < nspatialdims; i++) {
89 int inpsz = inpShape[i+2], k_i = kshape[i];
90 int stride = strides.empty() ? 1 : strides[i];
91 int dilation = dilations.empty() ? 1 : dilations[i];
92 int outsz;
93 if (autoPad == AUTO_PAD_NONE || autoPad == AUTO_PAD_VALID) {
94 int pad0 = 0, pad = 0;
95 if (!pads.empty()) {
96 pad0 = pads[i];
97 pad = pads[i] + pads[i + nspatialdims];
98 }
99 outsz = (inpsz + pad - 1 - dilation * (k_i - 1) + (ceilMode ? stride - 1 : 0)) / stride + 1;
100 // ONNX: with ceil_mode a sliding window that starts in the right/bottom
101 // padding region is dropped
102 if (ceilMode && (outsz - 1) * stride >= inpsz + pad0)
103 outsz--;
104 } else {
105 if (ceilMode)
106 outsz = (inpsz + stride - 1)/stride;
107 else
108 outsz = (inpsz - 1)/stride + 1;
109 }
110 outshape[i + 2] = outsz;
111 }
112
113 if (blockLayout) {
114 outshape.C = ngroups == 0 || wshape.empty() ? inpShape.C : wshape[0];
115 } else {
116 outshape.C = 0;
117 }
118
119 return outshape;
120}

122static inline void getPadding(const std::vector<int>& pads,
123 int dim, int nspatialdims, AutoPadding autoPad,
124 int ksize, int& pad0, int& pad1)
125{
126 CV_Assert(0 <= dim && dim < nspatialdims);
127
128 if (autoPad == AUTO_PAD_NONE || autoPad == AUTO_PAD_VALID) {
129 if (!pads.empty()) {
130 pad0 = pads[dim];
131 pad1 = pads[dim + nspatialdims];
132 } else {
133 pad0 = pad1 = 0;
134 }
135 } else {
136 CV_Assert(autoPad == AUTO_PAD_SAME_LOWER || autoPad == AUTO_PAD_SAME_UPPER);
137 pad0 = pad1 = ksize/2;
138 if (pad0*2 == ksize) {
139 pad0 -= autoPad == AUTO_PAD_SAME_UPPER;
140 pad1 -= autoPad == AUTO_PAD_SAME_LOWER;
141 }
142 }

146{
147 for (int i = 0; i < ConvState::MAX_CONV_DIMS; i++) {
148 if (kshape[i] != cs.kshape[i] ||

151 pads[i] != cs.pads[i] ||

159static MatShape getWpackShape(const MatShape& wshape, int ngroups, int C0)
160{
161 CV_Assert(wshape.dims >= 3);
162 int K = wshape[0], Cg = wshape[1];
163 int ksize = int(wshape.total())/(K*Cg);
164 CV_Assert_N(K % ngroups == 0);
165 int Kg = K / ngroups, K0 = C0;
166 int Kblk = (Kg + K0 - 1)/K0;
167 int C1Max = 0;
168 for (int g = 0; g < ngroups; ++g) {
169 int c_start = g * Cg;
170 int c00 = c_start & (C0 - 1);
171 int cblocks = (c00 + Cg + C0 - 1)/C0;
172 C1Max = std::max(C1Max, cblocks);
173 }
174 return MatShape({ngroups, Kblk, ksize, C1Max, C0*K0}, DATA_LAYOUT_UNKNOWN);
175}

177void ConvState::initConv(const MatShape& inpshape_,
178 const MatShape& wshape_,
179 const MatShape& outshape_,
180 int ngroups_,
181 const std::vector<int>& strides_,
182 const std::vector<int>& dilations_,
183 const std::vector<int>& pads_,
184 AutoPadding autoPad, bool ceilMode,
185 FastActivation fastActivation_,
186 ActivationFunc activationFunc_,
187 const std::vector<float>& activParams_)
188{
189 nspatialdims = wshape_.dims - 2;
190 CV_Assert(0 < nspatialdims && nspatialdims <= ConvState::MAX_CONV_DIMS);
191 CV_Assert(strides_.empty() || (strides_.size() == size_t(nspatialdims)));
192 CV_Assert(dilations_.empty() || (dilations_.size() == size_t(nspatialdims)));
193 CV_Assert(pads_.empty() || (pads_.size() == size_t(nspatialdims*2)));
194 CV_Assert(inpshape_.dims == outshape_.dims);
195 CV_Assert(inpshape_.dims == nspatialdims + 2 + int(inpshape_.layout == DATA_LAYOUT_BLOCK));
196 CV_Assert_N(inpshape_.layout == outshape_.layout,
197 inpshape_[0] == outshape_[0]);
198
199 inpshape = inpshape_;
200 outshape = outshape_;
201 ngroups = ngroups_;
202
203 int C = inpshape.channels();
204 int K = outshape.channels();
205
206 depthwise = ngroups == C && ngroups == K;
207
208 if (inpshape.layout == DATA_LAYOUT_BLOCK) {
209 CV_Assert(inpshape.back() == outshape.back());
210 }
211
212 fastActivation = fastActivation_;
213 activation = activationFunc_;
214 activParams = activParams_;
215
216 CV_Assert(wshape_[0] > 0 && wshape_[1] > 0);
217 for (int i = 0; i < MAX_CONV_DIMS; i++) {
218 kshape[i] = strides[i] = dilations[i] = 1;
219 pads[i] = pads[i + MAX_CONV_DIMS] = 0;
220 inner[i] = 0;
221 inner[i + MAX_CONV_DIMS] = 1;
222 }
223
224 for (int i = 0; i < nspatialdims; i++) {
225 int j = i + (MAX_CONV_DIMS - nspatialdims);
226 kshape[j] = wshape_[i+2];
227 CV_Assert(kshape[j] > 0);
228
229 strides[j] = strides_.empty() ? 1 : strides_[i];
230 dilations[j] = dilations_.empty() ? 1 : dilations_[i];
231
232 CV_Assert(strides[j] > 0);
233 CV_Assert(dilations[j] > 0);
234
235 int pad0, pad1;
236 getPadding(pads_, i, nspatialdims, autoPad, kshape[j], pad0, pad1);
237 CV_Assert_N(pad0 >= 0, pad1 >= 0);
238 pads[j] = pad0;
239 pads[j + MAX_CONV_DIMS] = pad1;
240
241 int inner0 = (pad0 + strides[j] - 1)/strides[j];
242 int inner1 = (inpshape[i+2] - (kshape[j] - 1)*dilations[j] + pad0)/strides[j];
243 inner1 += inner1*strides[j] - pad0 + (kshape[j] - 1)*dilations[j] < inpshape[i+2];
244 inner1 = std::min(inner1, outshape[i+2]);
245 if (inner0 >= inner1) {
246 inner0 = inner1 = outshape[i+2];
247 }
248 inner[j] = inner0;
249 inner[j + MAX_CONV_DIMS] = inner1;
250 }
251
252 initOfs();
253 if (!depthwise && inpshape.layout == DATA_LAYOUT_BLOCK) {
254 int C0 = inpshape.back();
255 wshape = getWpackShape(wshape_, ngroups, C0);
256
257 CV_Assert(wshape.dims == 5);
258 }
259}

261void repackConvWeights(const Mat& weights, Mat& Wpack, int outtype, int ngroups, int C0_)
262{
263 CV_Assert(weights.isContinuous());
264 CV_Assert_N(weights.type() == CV_32F, outtype == CV_32F);
265 CV_Assert(ngroups > 0);
266 CV_Assert((C0_ & (C0_ - 1)) == 0 && C0_ >= 4);
267
268 MatShape wshape = weights.shape();
269 CV_Assert(wshape.dims >= 3);
270
271 int K = wshape[0];
272 CV_Assert(K % ngroups == 0);
273
274 if (!Wpack.isContinuous()) {
275 Wpack.release();
276 }
277 MatShape wpackShape = getWpackShape(weights.shape(), ngroups, C0_);
278 Wpack.create(wpackShape, CV_32F);
279 Wpack.setZero();
280
281 parallel_for_(Range(0, K), [&](const Range& range) {
282 int Cg = wshape[1], Kg = K / ngroups;
283 int ksize = wpackShape[2], Kblk = wpackShape[1], C1Max = wpackShape[3];
284 int C0 = C0_, K0 = C0;
285 const float* wdata = weights.ptr<float>();
286 float* Wpackdata = Wpack.ptr<float>();
287
288 for (int k = range.start; k < range.end; ++k) {
289 int g = k / Kg;
290 int kin = k - g * Kg;
291 int kblk = kin / K0;
292 int k0 = kin & (K0 - 1);
293
294 int c_start = g * Cg;
295 int c00 = c_start & (C0 - 1);
296
297 for (int c = 0; c < Cg; ++c) {
298 int ch = c00 + c;
299 int c1 = ch / C0;
300 int c0 = ch & (C0 - 1);
301
302 const float* wptr = wdata + ((k * Cg + c) * ksize);
303 float* wpackptr = Wpackdata + (((g * Kblk + kblk) * ksize * C1Max + c1) * C0 + c0)*K0 + k0;
304 for (int i = 0; i < ksize; ++i) {
305 wpackptr[i*(C1Max*C0*K0)] = wptr[i];
306 }
307 }
308 }
309 });
310}

372static MatShape getDeconvWpackShape(const MatShape& wshape, int ngroups, int C0)
373{
374 CV_Assert(wshape.dims >= 3);
375 int C_in = wshape[0], Kg = wshape[1];
376 CV_Assert(C_in % ngroups == 0);
377 int Cg = C_in / ngroups;
378 int ksize = int(wshape.total()) / (C_in * Kg);
379 int K0 = C0;
380 int Kblk = (Kg + K0 - 1) / K0;
381 int C1Max = 0;
382 for (int g = 0; g < ngroups; ++g) {
383 int c_start = g * Cg;
384 int c00 = c_start & (C0 - 1);
385 int cblocks = (c00 + Cg + C0 - 1) / C0;
386 C1Max = std::max(C1Max, cblocks);
387 }
388 return MatShape({ngroups, Kblk, ksize, C1Max, C0 * K0}, DATA_LAYOUT_UNKNOWN);

391MatShape deconvInferShape(const MatShape& inpShape, const MatShape& wshape,
392 const std::vector<int>& kernelShape, int ngroups,
393 const std::vector<int>& strides,
394 const std::vector<int>& dilations,
395 const std::vector<int>& pads,
396 const std::vector<int>& adjustPads,
397 AutoPadding autoPad)
398{
399 bool blockLayout = (inpShape.layout == DATA_LAYOUT_BLOCK);
400 int ndims = inpShape.dims;
401 int nspatialdims = ndims - 2 - int(blockLayout);
402 CV_Assert(nspatialdims >= 1);
403 MatShape outshape = inpShape;
404
405 int kshape_[MatShape::MAX_DIMS];
406 if (!kernelShape.empty()) {
407 int kshape_size = (int)kernelShape.size();
408 for (int i = 0; i < nspatialdims; i++)
409 kshape_[i] = kernelShape[kshape_size - nspatialdims + i];
410 } else {
411 CV_Assert(!wshape.empty() && wshape.dims == nspatialdims + 2);
412 for (int i = 0; i < nspatialdims; i++)
413 kshape_[i] = wshape[i + 2];
414 }
415
416 int K_out = ngroups * wshape[1];
417 if (blockLayout) {
418 int C0 = inpShape[ndims - 1];
419 outshape[1] = (K_out + C0 - 1) / C0;
420 } else {
421 outshape[1] = K_out;
422 }
423
424 CV_Assert(strides.empty() || (int)strides.size() == nspatialdims);
425 CV_Assert(dilations.empty() || (int)dilations.size() == nspatialdims);
426 CV_Assert(pads.empty() || (int)pads.size() == nspatialdims * 2);
427 CV_Assert(adjustPads.empty() || (int)adjustPads.size() == nspatialdims);
428
429 for (int i = 0; i < nspatialdims; i++) {
430 int inpsz = inpShape[i + 2];
431 int k_i = kshape_[i];
432 int stride = strides.empty() ? 1 : strides[i];
433 int dilation = dilations.empty() ? 1 : dilations[i];
434 int adj = adjustPads.empty() ? 0 : adjustPads[i];
435 int outsz;
436 if (autoPad == AUTO_PAD_NONE || autoPad == AUTO_PAD_VALID) {
437 int pad_total = 0;
438 if (!pads.empty())
439 pad_total = pads[i] + pads[i + nspatialdims];
440 outsz = (inpsz - 1) * stride - pad_total + dilation * (k_i - 1) + 1 + adj;
441 } else {
442 outsz = (inpsz - 1) * stride + 1 + adj;
443 }
444 outshape[i + 2] = outsz;
445 }
446 outshape.C = blockLayout ? K_out : 0;
447 return outshape;
448}

450void repackDeconvWeights(const Mat& weights, Mat& Wpack, int outtype, int ngroups, int C0_)
451{
452 CV_Assert(weights.isContinuous());
453 CV_Assert_N(weights.type() == CV_32F, outtype == CV_32F);
454 CV_Assert(ngroups > 0);
455 CV_Assert((C0_ & (C0_ - 1)) == 0 && C0_ >= 4);
456
457 MatShape wshape = weights.shape();
458 CV_Assert(wshape.dims >= 3);
459 int C_in = wshape[0], Kg = wshape[1];
460 CV_Assert(C_in % ngroups == 0);
461 int K_out = ngroups * Kg;
462
463 if (!Wpack.isContinuous())
464 Wpack.release();
465
466 MatShape wpackShape = getDeconvWpackShape(wshape, ngroups, C0_);
467 Wpack.create(wpackShape, CV_32F);
468 Wpack.setZero();
469
470 parallel_for_(Range(0, K_out), [&](const Range& range) {
471 int Cg = C_in / ngroups;
472 int ksize = wpackShape[2], Kblk = wpackShape[1], C1Max = wpackShape[3];
473 int C0 = C0_, K0 = C0;
474 const float* wdata = weights.ptr<float>();
475 float* Wpackdata = Wpack.ptr<float>();
476
477 for (int k = range.start; k < range.end; ++k) {
478 int g = k / Kg;
479 int kin = k - g * Kg; // output channel within group
480 int kblk = kin / K0;
481 int k0 = kin & (K0 - 1);
482
483 int c_start = g * Cg;
484 int c00 = c_start & (C0 - 1);
485
486 for (int c = 0; c < Cg; ++c) {
487 int ch = c00 + c;
488 int c1 = ch / C0;
489 int c0 = ch & (C0 - 1);
490
491 int c_global = g * Cg + c;
492 const float* wptr = wdata + (c_global * Kg + kin) * ksize;
493 float* wpackptr = Wpackdata + (((g * Kblk + kblk) * ksize * C1Max + c1) * C0 + c0) * K0 + k0;
494 for (int i = 0; i < ksize; ++i) {
495 wpackptr[i * (C1Max * C0 * K0)] = wptr[i];
496 }
497 }
498 }
499 });
500}

549void ConvState::initOfs()
550{
551 CV_Assert(MAX_CONV_DIMS == 3);
552 int sdims = nspatialdims;
553 int KD = kshape[0], KH = kshape[1], KW = kshape[2];
554 int DZ = dilations[0], DY = dilations[1], DX = dilations[2];
555 int Hi = sdims > 1 ? inpshape[sdims] : 1;
556 int Wi = inpshape[sdims + 1];
557 int ksize = KD*KH*KW;
558 int C0 = inpshape.back();
559 coordtab.resize(ksize*MAX_CONV_DIMS);
560 ofstab.resize(ksize);
561 for (int z = 0, k = 0; z < KD; z++) {
562 int dz = z*DZ;
563 for (int y = 0; y < KH; y++) {
564 int dy = y*DY;
