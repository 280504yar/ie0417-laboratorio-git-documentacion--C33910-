100AffineFeature_Impl::AffineFeature_Impl(const Ptr<FeatureDetector>& backend,
101 int maxTilt, int minTilt, float tiltStep, float rotateStepBase)
102 : backend_(backend), maxTilt_(maxTilt), minTilt_(minTilt), tiltStep_(tiltStep), rotateStepBase_(rotateStepBase)
103{
104 int i = minTilt_;
105 if( i == 0 )
106 {
107 tilts_.push_back(1);
108 rolls_.push_back(0);
109 i++;
110 }
111 float tilt = 1;
112 for( ; i <= maxTilt_; i++ )
113 {
114 tilt *= tiltStep_;
115 float rotateStep = rotateStepBase_ / tilt;
116 int rollN = cvFloor(180.0f / rotateStep);
117 if( rollN * rotateStep == 180.0f )
118 rollN--;
119 for( int j = 0; j <= rollN; j++ )
120 {
121 tilts_.push_back(tilt);
122 rolls_.push_back(rotateStep * j);
123 }
124 }
125}

153class skewedDetectAndCompute : public ParallelLoopBody
154{
155public:
156 skewedDetectAndCompute(
157 const std::vector<float>& _tilts,
158 const std::vector<float>& _rolls,
159 std::vector< std::vector<KeyPoint> >& _keypointsCollection,
160 std::vector<Mat>& _descriptorCollection,
161 const Mat& _image,
162 const Mat& _mask,
163 const bool _do_keypoints,
164 const bool _do_descriptors,
165 const Ptr<Feature2D>& _backend)
166 : tilts(_tilts),
167 rolls(_rolls),
168 keypointsCollection(_keypointsCollection),
169 descriptorCollection(_descriptorCollection),
170 image(_image),
171 mask(_mask),
172 do_keypoints(_do_keypoints),
173 do_descriptors(_do_descriptors),
174 backend(_backend) {}
175
176 void operator()( const cv::Range& range ) const CV_OVERRIDE
177 {
178 CV_TRACE_FUNCTION();
179
180 const int begin = range.start;
181 const int end = range.end;
182
183 for( int a = begin; a < end; a++ )
184 {
185 Mat warpedImage, warpedMask;
186 Matx23f pose, invPose;
187 affineSkew(tilts[a], rolls[a], warpedImage, warpedMask, pose);
188 invertAffineTransform(pose, invPose);
189
190 std::vector<KeyPoint> wKeypoints;
191 Mat wDescriptors;
192 if( !do_keypoints )
193 {
194 const std::vector<KeyPoint>& keypointsInView = keypointsCollection[a];
195 if( keypointsInView.size() == 0 ) // when there are no keypoints in this affine view
196 continue;
197
198 std::vector<Point2f> pts_, pts;
199 KeyPoint::convert(keypointsInView, pts_);
200 transform(pts_, pts, pose);
201 wKeypoints.resize(keypointsInView.size());
202 for( size_t wi = 0; wi < wKeypoints.size(); wi++ )
203 {
204 wKeypoints[wi] = keypointsInView[wi];
205 wKeypoints[wi].pt = pts[wi];
206 }
207 }
208 backend->detectAndCompute(warpedImage, warpedMask, wKeypoints, wDescriptors, !do_keypoints);
209 if( do_keypoints )
210 {
211 // KeyPointsFilter::runByPixelsMask( wKeypoints, warpedMask );
212 if( wKeypoints.size() == 0 )
213 {
214 keypointsCollection[a].clear();
215 continue;
216 }
217 std::vector<Point2f> pts_, pts;
218 KeyPoint::convert(wKeypoints, pts_);
219 transform(pts_, pts, invPose);
220
221 keypointsCollection[a].resize(wKeypoints.size());
222 for( size_t wi = 0; wi < wKeypoints.size(); wi++ )
223 {
224 keypointsCollection[a][wi] = wKeypoints[wi];
225 keypointsCollection[a][wi].pt = pts[wi];
226 keypointsCollection[a][wi].class_id = a;
227 }
228 }
229 if( do_descriptors )
230 wDescriptors.copyTo(descriptorCollection[a]);
231 }
232 }
233private:
234 void affineSkew(float tilt, float phi,
235 Mat& warpedImage, Mat& warpedMask, Matx23f& pose) const
236 {
237 int h = image.size().height;
238 int w = image.size().width;
239 Mat rotImage;
240
241 Mat mask0;
242 if( mask.empty() )
243 mask0 = Mat(h, w, CV_8UC1, 255);
244 else
245 mask0 = mask;
246 pose = Matx23f(1,0,0,
247 0,1,0);
248
249 if( phi == 0 )
250 image.copyTo(rotImage);
251 else
252 {
253 phi = phi * (float)CV_PI / 180;
254 float s = std::sin(phi);
255 float c = std::cos(phi);
256 Matx22f A(c, -s, s, c);
257 Matx<float, 4, 2> corners(0, 0, (float)w, 0, (float)w,(float)h, 0, (float)h);
258 Mat tf(corners * A.t());
259 Mat tcorners;
260 tf.convertTo(tcorners, CV_32S);
261 Rect rect = boundingRect(tcorners);
262 h = rect.height; w = rect.width;
263 pose = Matx23f(c, -s, -(float)rect.x,
264 s, c, -(float)rect.y);
265 warpAffine(image, rotImage, pose, Size(w, h), INTER_LINEAR, BORDER_REPLICATE, Scalar(), cv::ALGO_HINT_ACCURATE);
266 }
267 if( tilt == 1 )
268 warpedImage = rotImage;
269 else
270 {
271 float s = 0.8f * sqrt(tilt * tilt - 1);
272 GaussianBlur(rotImage, rotImage, Size(0, 0), s, 0.01);
273 resize(rotImage, warpedImage, Size(0, 0), 1.0/tilt, 1.0, INTER_NEAREST);
274 pose(0, 0) /= tilt;
275 pose(0, 1) /= tilt;
276 pose(0, 2) /= tilt;
277 }
278 if( phi != 0 || tilt != 1 )
279 warpAffine(mask0, warpedMask, pose, warpedImage.size(), INTER_NEAREST, BORDER_CONSTANT, Scalar(), cv::ALGO_HINT_ACCURATE);
280 else
281 warpedMask = mask0;
282 }

296void AffineFeature_Impl::detectAndCompute(InputArray _image, InputArray _mask,
297 std::vector<KeyPoint>& keypoints,
298 OutputArray _descriptors,
299 bool useProvidedKeypoints)
300{
301 CV_TRACE_FUNCTION();
302
303 bool do_keypoints = !useProvidedKeypoints;
304 bool do_descriptors = _descriptors.needed();
305 Mat image = _image.getMat(), mask = _mask.getMat();
306 Mat descriptors;
307
308 if( (!do_keypoints && !do_descriptors) || _image.empty() )
309 return;
310
311 std::vector< std::vector<KeyPoint> > keypointsCollection(tilts_.size());
312 std::vector< Mat > descriptorCollection(tilts_.size());
313
314 if( do_keypoints )
315 keypoints.clear();
316 else
317 splitKeypointsByView(keypoints, keypointsCollection);
318
319 parallel_for_(Range(0, (int)tilts_.size()), skewedDetectAndCompute(tilts_, rolls_, keypointsCollection, descriptorCollection,
320 image, mask, do_keypoints, do_descriptors, backend_));
321
322 if( do_keypoints )
323 for( size_t i = 0; i < keypointsCollection.size(); i++ )
324 {
325 const std::vector<KeyPoint>& keys = keypointsCollection[i];
326 keypoints.insert(keypoints.end(), keys.begin(), keys.end());
327 }
328
329 if( do_descriptors )
330 {
331 _descriptors.create((int)keypoints.size(), backend_->descriptorSize(), backend_->descriptorType());
332 descriptors = _descriptors.getMat();
333 int iter = 0;
334 for( size_t i = 0; i < descriptorCollection.size(); i++ )
335 {
336 const Mat& descs = descriptorCollection[i];
337 if( descs.empty() )
338 continue;
339 Mat roi(descriptors, Rect(0, iter, descriptors.cols, descs.rows));
340 descs.copyTo(roi);
341 iter += descs.rows;
342 }
343 }
344}
