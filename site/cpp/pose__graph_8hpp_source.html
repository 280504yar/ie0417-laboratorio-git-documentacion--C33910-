22// This class provides tools to solve so-called pose graph problem often arisen in SLAM problems
23// The pose graph format, cost function and optimization techniques
24// repeat the ones used in Ceres 3D Pose Graph Optimization:
25// http://ceres-solver.org/nnls_tutorial.html#other-examples, pose_graph_3d.cc bullet
26class CV_EXPORTS_W PoseGraph
27{
28public:
29 static Ptr<PoseGraph> create();
30 virtual ~PoseGraph();
31
32 // Node may have any id >= 0
33 virtual void addNode(size_t _nodeId, const Affine3d& _pose, bool fixed) = 0;
34 virtual bool isNodeExist(size_t nodeId) const = 0;
35 virtual bool setNodeFixed(size_t nodeId, bool fixed) = 0;
36 virtual bool isNodeFixed(size_t nodeId) const = 0;
37 virtual Affine3d getNodePose(size_t nodeId) const = 0;
38 virtual std::vector<size_t> getNodesIds() const = 0;
39 virtual size_t getNumNodes() const = 0;
40
41 // Edges have consequent indices starting from 0
42 virtual void addEdge(size_t _sourceNodeId, size_t _targetNodeId, const Affine3f& _transformation,
43 const Matx66f& _information = Matx66f::eye()) = 0;
44 virtual size_t getEdgeStart(size_t i) const = 0;
45 virtual size_t getEdgeEnd(size_t i) const = 0;
46 virtual Affine3d getEdgePose(size_t i) const = 0;
47 virtual Matx66f getEdgeInfo(size_t i) const = 0;
48 virtual size_t getNumEdges() const = 0;
49
50 // checks if graph is connected and each edge connects exactly 2 nodes
51 virtual bool isValid() const = 0;
52
53 // Calculates an initial pose estimate using the Minimum Spanning Tree (Prim's MST) algorithm.
54 // The result serves as a starting point for further optimization.
55 // Edge weights are calculated as:
56 // weight = translationNorm + lambda * rotationAngle
57 // The default lambda value (0.485) was empirically chosen based on its impact on optimizer performance,
58 // but can/should be tuned for different datasets.
59 virtual void initializePosesWithMST(double lambda = 0.485) = 0;
60
61 // creates an optimizer with user-defined settings and returns a pointer on it
62 virtual Ptr<cv::detail::LevMarqBase> createOptimizer(const LevMarq::Settings& settings) = 0;
63 // creates an optimizer with default settings and returns a pointer on it
64 virtual Ptr<cv::detail::LevMarqBase> createOptimizer() = 0;
65
66 // Creates an optimizer (with default settings) if it wasn't created before and runs it
67 // Returns number of iterations elapsed or -1 if failed to optimize
68 virtual LevMarq::Report optimize() = 0;
69
70 // calculate cost function based on current nodes parameters
71 virtual double calcEnergy() const = 0;
72};
