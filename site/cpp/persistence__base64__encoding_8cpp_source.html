12class base64::Base64ContextEmitter
13{
14public:
15 explicit Base64ContextEmitter(cv::FileStorage::Impl& fs, bool needs_indent_)
16 : file_storage(fs)
17 , needs_indent(needs_indent_)
18 , binary_buffer(BUFFER_LEN)
19 , base64_buffer(base64_encode_buffer_size(BUFFER_LEN))
20 , src_beg(0)
21 , src_cur(0)
22 , src_end(0)
23 {
24 src_beg = binary_buffer.data();
25 src_end = src_beg + BUFFER_LEN;
26 src_cur = src_beg;
27
28 CV_Assert(fs.write_mode);
29
30 if (needs_indent)
31 {
32 file_storage.flush();
33 }
34 }
35
36 ~Base64ContextEmitter()
37 {
38 /* cleaning */
39 if (src_cur != src_beg)
40 flush(); /* encode the rest binary data to base64 buffer */
41 }

43 Base64ContextEmitter & write(const uchar * beg, const uchar * end)
44 {
45 if (beg >= end)
46 return *this;
47
48 while (beg < end) {
49 /* collect binary data and copy to binary buffer */
50 size_t len = std::min(end - beg, src_end - src_cur);
51 std::memcpy(src_cur, beg, len);
52 beg += len;
53 src_cur += len;
54
55 if (src_cur >= src_end) {
56 /* binary buffer is full. */
57 /* encode it to base64 and send result to fs */
58 flush();
59 }
60 }
61
62 return *this;
63 }
64
65 /*
66 * a convertor must provide :
67 * - `operator >> (uchar * & dst)` for writing current binary data to `dst` and moving to next data.
68 * - `operator bool` for checking if current location is valid and not the end.
69 */
70 template<typename _to_binary_convertor_t> inline
71 Base64ContextEmitter & write(_to_binary_convertor_t & convertor)
72 {
73 constexpr size_t BUFFER_MAX_LEN = 1024U;
74
75 uchar buffer[BUFFER_MAX_LEN];
76 uchar * beg = buffer;
77 uchar * end = beg;
78
79 while (convertor) {
80 convertor >> end;
81 write(beg, end);
82 end = beg;
83 }
84
85 return *this;
86 }
87
88 bool flush()
89 {
90 /* control line width, so on. */
91 size_t len = base64_encode(src_beg, base64_buffer.data(), 0U, src_cur - src_beg);
92 if (len == 0U)
93 return false;
94
95 src_cur = src_beg;
96
97 if ( !needs_indent)
98 {
99 file_storage.puts((const char*)base64_buffer.data());
100 }
101 else
102 {
103 const char newline[] = "\n";
104 char space[80];
105 int ident = file_storage.write_stack.back().indent;
106 memset(space, ' ', static_cast<int>(ident));
107 space[ident] = '\0';
108
109 file_storage.puts(space);
110 file_storage.puts((const char*)base64_buffer.data());
111 file_storage.puts(newline);
112 file_storage.flush();
113 }
114
115 return true;
116 }
117
118private:
119 /* because of Base64, we must keep its length a multiple of 3 */
120 static const size_t BUFFER_LEN = 48U;
121 // static_assert(BUFFER_LEN % 3 == 0, "BUFFER_LEN is invalid");

147size_t base64::base64_encode(const uint8_t *src, uint8_t *dst, size_t off, size_t cnt) {
148 if (!src || !dst || !cnt)
149 return 0;
150
151 /* initialize beginning and end */
152 uint8_t * dst_beg = dst;
153 uint8_t * dst_cur = dst_beg;
154
155 uint8_t const * src_beg = src + off;
156 uint8_t const * src_cur = src_beg;
157 uint8_t const * src_end = src_cur + cnt / 3U * 3U;
158
159 /* integer multiples part */
160 while (src_cur < src_end) {
161 uint8_t _2 = *src_cur++;
162 uint8_t _1 = *src_cur++;
163 uint8_t _0 = *src_cur++;
164 *dst_cur++ = base64_mapping[ _2 >> 2U];
165 *dst_cur++ = base64_mapping[(_1 & 0xF0U) >> 4U | (_2 & 0x03U) << 4U];
166 *dst_cur++ = base64_mapping[(_0 & 0xC0U) >> 6U | (_1 & 0x0FU) << 2U];
167 *dst_cur++ = base64_mapping[ _0 & 0x3FU];
168 }
169
170 /* remainder part */
171 size_t rst = src_beg + cnt - src_cur;
172 if (rst == 1U) {
173 uint8_t _2 = *src_cur++;
174 *dst_cur++ = base64_mapping[ _2 >> 2U];
175 *dst_cur++ = base64_mapping[(_2 & 0x03U) << 4U];
176 } else if (rst == 2U) {
177 uint8_t _2 = *src_cur++;
178 uint8_t _1 = *src_cur++;
179 *dst_cur++ = base64_mapping[ _2 >> 2U];
180 *dst_cur++ = base64_mapping[(_2 & 0x03U) << 4U | (_1 & 0xF0U) >> 4U];
181 *dst_cur++ = base64_mapping[(_1 & 0x0FU) << 2U];
182 }
183
184 /* padding */
185 switch (rst)
186 {
187 case 1U: *dst_cur++ = base64_padding;
188 /* fallthrough */
189 case 2U: *dst_cur++ = base64_padding;
190 /* fallthrough */
191 default: *dst_cur = 0;
192 break;
193 }
194
195 return static_cast<size_t>(dst_cur - dst_beg);
196}

198int base64::icvCalcStructSize(const char *dt, int initial_size) {
199 int size = cv::fs::calcElemSize( dt, initial_size );
200 size_t elem_max_size = 0;
201 for ( const char * type = dt; *type != '\0'; type++ ) {
202 switch ( *type )
203 {
204 case 'u': { elem_max_size = std::max( elem_max_size, sizeof(uchar ) ); break; }
205 case 'b': { elem_max_size = std::max( elem_max_size, sizeof(uchar ) ); break; }
206 case 'c': { elem_max_size = std::max( elem_max_size, sizeof(schar ) ); break; }
207 case 'w': { elem_max_size = std::max( elem_max_size, sizeof(ushort) ); break; }
208 case 's': { elem_max_size = std::max( elem_max_size, sizeof(short ) ); break; }
209 case 'i': { elem_max_size = std::max( elem_max_size, sizeof(int ) ); break; }
210 case 'n': { elem_max_size = std::max( elem_max_size, sizeof(unsigned) ); break; }
211 case 'f': { elem_max_size = std::max( elem_max_size, sizeof(float ) ); break; }
212 case 'd': { elem_max_size = std::max( elem_max_size, sizeof(double) ); break; }
213 case 'I': { elem_max_size = std::max( elem_max_size, sizeof(int64_t)); break; }
214 case 'U': { elem_max_size = std::max( elem_max_size, sizeof(uint64_t)); break; }
215 case 'e': { elem_max_size = std::max( elem_max_size, sizeof(uchar) ); break; }
216 case 'E': { elem_max_size = std::max( elem_max_size, sizeof(uchar) ); break; }
217 default: break;
218 }
219 }
220 size = cvAlign( size, static_cast<int>(elem_max_size) );
221 return size;
222}

255void base64::Base64Writer::check_dt(const char* dt)
256{
257 if ( dt == 0 )
258 CV_Error( cv::Error::StsBadArg, "Invalid \'dt\'." );
259 else if (data_type_string.empty()) {
260 data_type_string = dt;
261
262 /* output header */
263 std::string buffer = make_base64_header(dt);
264 const uchar * beg = reinterpret_cast<const uchar *>(buffer.data());
265 const uchar * end = beg + buffer.size();
266
267 emitter->write(beg, end);
268 } else if ( data_type_string != dt )
269 CV_Error( cv::Error::StsBadArg, "\'dt\' does not match." );
270}

291inline base64::RawDataToBinaryConvertor& base64::RawDataToBinaryConvertor::operator >>(uchar * & dst)
292{
293 CV_DbgAssert(*this);
294
295 for (size_t i = 0U, n = to_binary_funcs.size(); i < n; i++) {
296 elem_to_binary_t & pack = to_binary_funcs[i];
297 pack.func(cur + pack.offset, dst + pack.offset_packed);
298 }
299 cur += step;
300 dst += step_packed;
301
302 return *this;
303}

310size_t base64::RawDataToBinaryConvertor::make_to_binary_funcs(const std::string &dt)
311{
312 size_t cnt = 0;
313 size_t offset = 0;
314 size_t offset_packed = 0;
315 char type = '\0';
316
317 std::istringstream iss(dt);
318 while (!iss.eof()) {
319 if (!(iss >> cnt)) {
320 iss.clear();
321 cnt = 1;
322 }
323 CV_Assert(cnt > 0U);
324 if (!(iss >> type))
325 break;
326
327 while (cnt-- > 0)
328 {
329 elem_to_binary_t pack;
330
331 size_t size = 0;
332 switch (type)
333 {
334 case 'u':
335 case 'c':
336 size = sizeof(uchar);
337 pack.func = to_binary<uchar>;
338 break;
339 case 'b':
340 size = sizeof(uchar);
341 pack.func = to_binary<bool>;
342 break;
343 case 'w':
344 case 's':
345 size = sizeof(ushort);
346 pack.func = to_binary<ushort>;
347 break;
348 case 'n':
349 case 'i':
350 size = sizeof(uint);
351 pack.func = to_binary<uint>;
352 break;
353 case 'f':
354 size = sizeof(float);
355 pack.func = to_binary<float>;
356 break;
357 case 'd':
358 size = sizeof(double);
359 pack.func = to_binary<double>;
360 break;
361 case 'I':
362 case 'U':
363 size = sizeof(uint64_t);
364 pack.func = to_binary<uint64_t>;
365 break;
366 case 'e':
367 case 'E':
368 size = sizeof(uchar);
369 pack.func = to_binary<uchar>;
370 break;
371 case 'r':
372 default:
373 CV_Error(cv::Error::StsError, "type is not supported");
374 };
375
376 offset = static_cast<size_t>(cvAlign(static_cast<int>(offset), static_cast<int>(size)));
377 pack.offset = offset;
378 offset += size;
379
380 pack.offset_packed = offset_packed;
381 offset_packed += size;
382
383 to_binary_funcs.push_back(pack);
384 }
385 }
386
387 CV_Assert(iss.eof());
388 return offset_packed;
389}
