34struct BpeTokenizerImpl : public Tokenizer::Impl {
35 Ptr<CoreBPE> coreBPE;
36 std::unordered_set<std::string> allowedSpecial;
37
38 explicit BpeTokenizerImpl(CoreBPE core,
39 std::unordered_set<std::string> special = {})
40 : coreBPE(makePtr<CoreBPE>(std::move(core)))
41 , allowedSpecial(std::move(special)) {}
42
43 std::vector<int> encode(const std::string& text) override {
44 CV_Assert(coreBPE);
45 std::vector<uint32_t> tok = coreBPE->encode(text, allowedSpecial).first;
46 return std::vector<int>(tok.begin(), tok.end());
47 }
48
49 std::string decode(const std::vector<int>& tokens) override {
50 CV_Assert(coreBPE);
51 std::vector<uint32_t> t32(tokens.begin(), tokens.end());
52 const std::vector<std::uint8_t> opt_bytes = coreBPE->decodeBytes(t32);
53 if (opt_bytes.empty())
54 CV_Error(cv::Error::StsError, "Invalid decode.");
55 return std::string(reinterpret_cast<const char*>(opt_bytes.data()), opt_bytes.size());
56 }
57};

76struct SentencePieceTokenizerImpl : public Tokenizer::Impl {
77 CoreGemmaBPE model;
78 std::unordered_set<std::string> allowedSpecial;
79 int bosTokenId;
80
81 explicit SentencePieceTokenizerImpl(CoreGemmaBPE m,
82 std::unordered_set<std::string> special = {},
83 int bos = -1)
84 : model(std::move(m)), allowedSpecial(std::move(special)), bosTokenId(bos) {}
85
86 std::vector<int> encode(const std::string& text) override {
87 std::vector<int> ids = model.encode(text, allowedSpecial);
88 // HuggingFace SentencePiece tokenizers (Gemma2) prepend <bos> automatically
89 if (bosTokenId >= 0) {
90 ids.insert(ids.begin(), bosTokenId);
91 }
92 return ids;
93 }
94
95 std::string decode(const std::vector<int>& tokens) override {
96 // Skip the bos token if present at the beginning
97 if (bosTokenId >= 0 && !tokens.empty() && tokens.front() == bosTokenId) {
98 std::vector<int> stripped(tokens.begin() + 1, tokens.end());
99 return model.decode(stripped);
100 }
101 return model.decode(tokens);
102 }
103};

105static Ptr<GemmaBpeTokenizerImpl> buildGemmaFromJson(
106 const std::string& json_path,
107 std::unordered_set<std::string>* outSpecial = nullptr) {
108
109 cv::FileStorage fs(json_path, cv::FileStorage::READ | cv::FileStorage::FORMAT_JSON);
110 if (!fs.isOpened())
111 CV_Error(cv::Error::StsError, "Failed to open tokenizer.json: " + json_path);
112
113 cv::FileNode model_node = fs["model"];
114 CV_CheckFalse(model_node.empty(), "tokenizer.json missing 'model'");
115
116 std::string model_type;
117 model_node["type"] >> model_type;
118 if (model_type != "BPE")
119 CV_Error(cv::Error::StsError,
120 "Expected BPE model in tokenizer.json for Gemma3, got: " + model_type);
121
122 CoreGemmaBPE gemma;
123
124 cv::FileNode vocab_node = model_node["vocab"];
125 CV_CheckFalse(vocab_node.empty(), "tokenizer.json model missing 'vocab'");
126
127 int maxId = -1;
128 for (auto it = vocab_node.begin(); it != vocab_node.end(); ++it) {
129 cv::FileNode entry = *it;
130 std::string piece = entry.name();
131 int id = (int)entry;
132 if (id > maxId) maxId = id;
133 gemma.pieceToId[piece] = id;
134 }
135
136 gemma.idToPiece.resize(maxId + 1);
137 for (const auto& kv : gemma.pieceToId)
138 gemma.idToPiece[kv.second] = kv.first;
139
140 cv::FileNode merges_node = model_node["merges"];
141 if (!merges_node.empty()) {
142 uint32_t rank = 0;
143 for (auto it = merges_node.begin(); it != merges_node.end(); ++it) {
144 cv::FileNode entry = *it;
145 if (static_cast<int>(entry.size()) != 2) {
146 ++rank;
147 continue;
148 }
149 std::string a, b;
150 entry[0] >> a;
151 entry[1] >> b;
152 gemma.addMerge(a, b, rank);
153 ++rank;
154 }
155 }
156
157 std::unordered_set<std::string> special;
158 cv::FileNode added = fs["added_tokens"];
159 if (!added.empty()) {
160 for (auto it = added.begin(); it != added.end(); ++it) {
161 cv::FileNode t = *it;
162 int id = -1; t["id"] >> id;
163 std::string content; t["content"] >> content;
164 bool is_special = false; t["special"] >> is_special;
165 if (id >= 0 && !content.empty()) {
166 gemma.specialToId[content] = id;
167 gemma.idToSpecial[id] = content;
168 // All added tokens bypass BPE (matching HuggingFace behavior),
169 // not just those marked special.
170 special.insert(content);
171 if (outSpecial) outSpecial->insert(content);
172 }
173 }
174 }
175
176 return makePtr<GemmaBpeTokenizerImpl>(std::move(gemma), std::move(special));
177}

179static Ptr<SentencePieceTokenizerImpl> buildSentencePieceFromJson(
180 const std::string& json_path,
181 std::unordered_set<std::string>* outSpecial = nullptr) {
182
183 cv::FileStorage fs(json_path, cv::FileStorage::READ | cv::FileStorage::FORMAT_JSON);
184 if (!fs.isOpened())
185 CV_Error(cv::Error::StsError, "Failed to open tokenizer.json: " + json_path);
186
187 cv::FileNode model_node = fs["model"];
188 CV_CheckFalse(model_node.empty(), "tokenizer.json missing 'model'");
189
190 std::string model_type;
191 model_node["type"] >> model_type;
192 if (model_type != "BPE")
193 CV_Error(cv::Error::StsError,
194 "Expected BPE model in tokenizer.json for SentencePiece, got: " + model_type);
195
196 CoreGemmaBPE gemma;
197
198 cv::FileNode vocab_node = model_node["vocab"];
199 CV_CheckFalse(vocab_node.empty(), "tokenizer.json model missing 'vocab'");
200
201 int maxId = -1;
202 for (auto it = vocab_node.begin(); it != vocab_node.end(); ++it) {
203 cv::FileNode entry = *it;
204 std::string piece = entry.name();
205 int id = (int)entry;
206 if (id > maxId) maxId = id;
207 gemma.pieceToId[piece] = id;
208 }
209
210 gemma.idToPiece.resize(maxId + 1);
211 for (const auto& kv : gemma.pieceToId)
212 gemma.idToPiece[kv.second] = kv.first;
213
214 cv::FileNode merges_node = model_node["merges"];
215 if (!merges_node.empty()) {
216 uint32_t rank = 0;
217 for (auto it = merges_node.begin(); it != merges_node.end(); ++it) {
218 cv::FileNode entry = *it;
219 std::string a, b;
220 if (entry.isString()) {
221 // SentencePiece format: "a b" (single string with space separator)
222 std::string merge_str;
223 entry >> merge_str;
224 size_t sp = merge_str.find(' ');
225 if (sp == std::string::npos) {
226 ++rank;
227 continue;
228 }
229 a = merge_str.substr(0, sp);
230 b = merge_str.substr(sp + 1);
231 } else if (entry.size() == 2) {
232 // Array format: ["a", "b"]
233 entry[0] >> a;
234 entry[1] >> b;
235 } else {
236 ++rank;
237 continue;
238 }
239 gemma.addMerge(a, b, rank);
240 ++rank;
241 }
242 }
243
244 std::unordered_set<std::string> special;
245 int bosId = -1;
246 cv::FileNode added = fs["added_tokens"];
247 if (!added.empty()) {
248 for (auto it = added.begin(); it != added.end(); ++it) {
249 cv::FileNode t = *it;
250 int id = -1; t["id"] >> id;
251 std::string content; t["content"] >> content;
252 if (id >= 0 && !content.empty()) {
253 gemma.specialToId[content] = id;
254 gemma.idToSpecial[id] = content;
255 special.insert(content);
256 if (outSpecial) outSpecial->insert(content);
257 // Capture <bos> token id for SentencePiece auto-prepend
258 if (content == "<bos>") bosId = id;
259 }
260 }
261 }
262
263 return makePtr<SentencePieceTokenizerImpl>(std::move(gemma), std::move(special), bosId);
264}

266static void registerDefaultTokenizers() {
267 auto& reg = tokenizerRegistry();
268 if (reg.find("BPE") == reg.end()) {
269 reg["BPE"] = [](const FileStorage& cfg, const std::string& dir) -> Ptr<Tokenizer::Impl> {
270 std::string model_type;
271 cfg["model_type"] >> model_type;
272 std::string tok_json = dir + "tokenizer.json";
273
274 CoreBPE core;
275 std::unordered_set<std::string> special;
276 if (model_type == "gpt2" || model_type == "gpt4") {
277 core = buildTokenizerFromJson(model_type, tok_json);
278 } else if (model_type == "qwen2" || model_type == "qwen2.5") {
279 core = buildTokenizerFromJson(model_type, tok_json, &special);
280 } else {
281 CV_Error(cv::Error::StsError, "Unsupported model_type for BPE: " + model_type);
282 }
283 return makePtr<BpeTokenizerImpl>(std::move(core), std::move(special));
284 };
285 }
286
287 if (reg.find("Gemma") == reg.end()) {
288 reg["Gemma"] = [](const FileStorage& /*cfg*/, const std::string& dir) -> Ptr<Tokenizer::Impl> {
289 std::string tok_json = dir + "tokenizer.json";
290 std::unordered_set<std::string> special;
291 return buildGemmaFromJson(tok_json, &special);
292 };
293 }
294
295 if (reg.find("SentencePiece") == reg.end()) {
296 reg["SentencePiece"] = [](const FileStorage& /*cfg*/, const std::string& dir) -> Ptr<Tokenizer::Impl> {
297 std::string tok_json = dir + "tokenizer.json";
298 std::unordered_set<std::string> special;
299 return buildSentencePieceFromJson(tok_json, &special);
300 };
301 }

316CoreBPE buildTokenizerFromJson(const std::string& model_type, const std::string& json_path,
317 std::unordered_set<std::string>* outSpecial) {
318 cv::FileStorage fs(json_path, cv::FileStorage::READ | cv::FileStorage::FORMAT_JSON);
319 if (!fs.isOpened())
320 CV_Error(cv::Error::StsError, "Failed to open tokenizer.json: " + json_path);
321
322 cv::FileNode model = fs["model"];
323 CV_CheckFalse(model.empty(), "tokenizer.json missing 'model'");
324 cv::FileNode vocab = model["vocab"];
325 CV_CheckFalse(vocab.empty(), "tokenizer.json missing model.vocab");
326
327 std::string pattern;
328 std::unordered_set<std::string> skip_tokens;
329 if (model_type == "gpt2" || model_type == "r50k_base") {
330 pattern = R50K_UTF8;
331 skip_tokens.insert("<|endoftext|>");
332 } else if (model_type == "gpt4" || model_type == "cl100k_base") {
333 pattern = CL100K_BASE;
334 } else if (model_type == "qwen2" || model_type == "qwen2.5") {
335 pattern = QWEN2_5;
336 } else {
337 CV_Error(cv::Error::StsError,
338 "Unsupported model_type: " + model_type + " (expected gpt2/r50k_base, gpt4/cl100k_base, or qwen2/qwen2.5)");
339 }
340
341 auto token_to_bytes = [&](const std::string& token_utf8) -> std::vector<uint8_t> {
342 std::vector<std::uint8_t> out;
343 auto cps = unicode_cpts_from_utf8(token_utf8);
344 out.reserve(cps.size());
345 for (uint32_t cp : cps) {
346 const std::string one = unicode_cpt_to_utf8(cp);
347 out.push_back(unicode_utf8_to_byte(one));
348 }
349 return out;
350 };
351
352 ByteVecRankMap mergeableRanks;
353 mergeableRanks.reserve((size_t)vocab.size());
354 int max_id = -1;
355
356 for (cv::FileNodeIterator it = vocab.begin(); it != vocab.end(); ++it) {
357 FileNode val = *it;
358 std::string token = val.name();
359 if (skip_tokens.find(token) != skip_tokens.end()) continue;
360 int id = (int)val;
361 mergeableRanks.emplace(token_to_bytes(token), (uint32_t)id);
362 if (id > max_id) max_id = id;
363 }
364
365 std::unordered_map<std::string, uint32_t> specialTokens;
366 FileNode added = fs["added_tokens"];
367 if (!added.empty()) {
368 for (auto it = added.begin(); it != added.end(); ++it) {
369 cv::FileNode t = *it;
370 bool special = false; t["special"] >> special;
371 int id = -1; t["id"] >> id;
372 std::string content; t["content"] >> content;
373 if (special && id >= 0 && !content.empty()) {
374 specialTokens.emplace(content, (uint32_t)id);
375 if (id > max_id) max_id = id;
376 if (outSpecial) outSpecial->insert(content);
377 }
378 }
379 }
380
381 return CoreBPE(std::move(mergeableRanks), std::move(specialTokens), pattern);
382}

384Tokenizer Tokenizer::load(const std::string& model_config) {
385 cv::FileStorage cfg(model_config, cv::FileStorage::READ | cv::FileStorage::FORMAT_JSON);
386 if (!cfg.isOpened())
387 CV_Error(cv::Error::StsError, "Could not open config.json: " + model_config);
388
389 std::string dir = model_config;
390 size_t pos = dir.find_last_of("/\\");
391 dir = (pos == std::string::npos) ? std::string() : dir.substr(0, pos + 1);
392
393 std::string methodType = "BPE";
394 if (!cfg["method"].empty())
395 cfg["method"] >> methodType;
396
397 registerDefaultTokenizers();
398 auto& reg = tokenizerRegistry();
399 auto it = reg.find(methodType);
400 if (it == reg.end())
401 CV_Error(cv::Error::StsError,
402 "Unsupported tokenizer method: '" + methodType + "'. Supported: BPE, Gemma, SentencePiece");
403
404 Tokenizer tok;
405 tok.impl_ = it->second(cfg, dir);
406 return tok;
407}
