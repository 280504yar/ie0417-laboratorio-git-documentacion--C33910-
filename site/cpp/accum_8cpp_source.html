90inline int getAccTabIdx(int sdepth, int ddepth)
91{
92 return sdepth == CV_8U && ddepth == CV_32F ? 0 :
93 sdepth == CV_8U && ddepth == CV_64F ? 1 :
94 sdepth == CV_16U && ddepth == CV_32F ? 2 :

111static bool ocl_accumulate( InputArray _src, InputArray _src2, InputOutputArray _dst, double alpha,
112 InputArray _mask, int op_type )
113{
114 CV_Assert(op_type == ACCUMULATE || op_type == ACCUMULATE_SQUARE ||
115 op_type == ACCUMULATE_PRODUCT || op_type == ACCUMULATE_WEIGHTED);
116
117 const ocl::Device & dev = ocl::Device::getDefault();
118 bool haveMask = !_mask.empty(), doubleSupport = dev.doubleFPConfig() > 0;
119 int stype = _src.type(), sdepth = CV_MAT_DEPTH(stype), cn = CV_MAT_CN(stype), ddepth = _dst.depth();
120 int kercn = haveMask ? cn : ocl::predictOptimalVectorWidthMax(_src, _src2, _dst), rowsPerWI = dev.isIntel() ? 4 : 1;
121
122 if (!doubleSupport && (sdepth == CV_64F || ddepth == CV_64F))
123 return false;
124
125 const char * const opMap[4] = { "ACCUMULATE", "ACCUMULATE_SQUARE", "ACCUMULATE_PRODUCT",
126 "ACCUMULATE_WEIGHTED" };
127
128 char cvt[50];
129 ocl::Kernel k("accumulate", ocl::imgproc::accumulate_oclsrc,
130 format("-D %s%s -D srcT1=%s -D cn=%d -D dstT1=%s%s -D rowsPerWI=%d -D convertToDT=%s",
131 opMap[op_type], haveMask ? " -D HAVE_MASK" : "",
132 ocl::typeToStr(sdepth), kercn, ocl::typeToStr(ddepth),
133 doubleSupport ? " -D DOUBLE_SUPPORT" : "", rowsPerWI,
134 ocl::convertTypeStr(sdepth, ddepth, 1, cvt, sizeof(cvt))));
135 if (k.empty())
136 return false;
137
138 UMat src = _src.getUMat(), src2 = _src2.getUMat(), dst = _dst.getUMat(), mask = _mask.getUMat();
139
140 ocl::KernelArg srcarg = ocl::KernelArg::ReadOnlyNoSize(src),
141 src2arg = ocl::KernelArg::ReadOnlyNoSize(src2),
142 dstarg = ocl::KernelArg::ReadWrite(dst, cn, kercn),
143 maskarg = ocl::KernelArg::ReadOnlyNoSize(mask);
144
145 int argidx = k.set(0, srcarg);
146 if (op_type == ACCUMULATE_PRODUCT)
147 argidx = k.set(argidx, src2arg);
148 argidx = k.set(argidx, dstarg);
149 if (op_type == ACCUMULATE_WEIGHTED)
150 {
151 if (ddepth == CV_32F)
152 argidx = k.set(argidx, (float)alpha);
153 else
154 argidx = k.set(argidx, alpha);
155 }
156 if (haveMask)
157 k.set(argidx, maskarg);
158
159 size_t globalsize[2] = { (size_t)src.cols * cn / kercn, ((size_t)src.rows + rowsPerWI - 1) / rowsPerWI };
160 return k.run(2, globalsize, NULL, false);
161}

220 else if (ippiAdd_IM)

222 dst.ptr<Ipp32f>(), dststep, ippiSize(size.width, size.height));

225 return true;
226 }
227 }

233void cv::accumulate( InputArray _src, InputOutputArray _dst, InputArray _mask )
234{
235 CV_INSTRUMENT_REGION();
236
237 int stype = _src.type(), sdepth = CV_MAT_DEPTH(stype), scn = CV_MAT_CN(stype);
238 int dtype = _dst.type(), ddepth = CV_MAT_DEPTH(dtype), dcn = CV_MAT_CN(dtype);
239
240 CV_Assert( _src.sameSize(_dst) && dcn == scn );
241 CV_Assert( _mask.empty() || (_src.sameSize(_mask) && (_mask.type() == CV_8U || _mask.type() == CV_Bool)) );
242
243 CV_OCL_RUN(_src.dims() <= 2 && _dst.isUMat(),
244 ocl_accumulate(_src, noArray(), _dst, 0.0, _mask, ACCUMULATE))
245
246 CV_IPP_RUN((_src.dims() <= 2 || (_src.isContinuous() && _dst.isContinuous() && (_mask.empty() || _mask.isContinuous()))),
247 ipp_accumulate(_src, _dst, _mask));
248
249 Mat src = _src.getMat(), dst = _dst.getMat(), mask = _mask.getMat();
250
251
252 int fidx = getAccTabIdx(sdepth, ddepth);
253 AccFunc func = fidx >= 0 ? accTab[fidx] : 0;
254 CV_Assert( func != 0 );
255
256 const Mat* arrays[] = {&src, &dst, &mask, 0};
257 uchar* ptrs[3] = {};
258 NAryMatIterator it(arrays, ptrs);
259 int len = (int)it.size;
260
261 for( size_t i = 0; i < it.nplanes; i++, ++it )
262 func(ptrs[0], ptrs[1], ptrs[2], len, scn);
263}
