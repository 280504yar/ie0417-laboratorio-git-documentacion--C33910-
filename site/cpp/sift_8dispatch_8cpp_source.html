175static inline void
176unpackOctave(const KeyPoint& kpt, int& octave, int& layer, float& scale)
177{
178 octave = kpt.octave & 255;
179 layer = (kpt.octave >> 8) & 255;
180 octave = octave < 128 ? octave : (-128 | octave);
181 scale = octave >= 0 ? 1.f/(1 << octave) : (float)(1 << -octave);
182}

184static Mat createInitialImage( const Mat& img, bool doubleImageSize, float sigma, bool enable_precise_upscale )
185{
186 CV_TRACE_FUNCTION();
187
188 Mat gray, gray_fpt;
189 if( img.channels() == 3 || img.channels() == 4 )
190 {
191 cvtColor(img, gray, COLOR_BGR2GRAY);
192 gray.convertTo(gray_fpt, DataType<sift_wt>::type, SIFT_FIXPT_SCALE, 0);
193 }
194 else
195 img.convertTo(gray_fpt, DataType<sift_wt>::type, SIFT_FIXPT_SCALE, 0);
196
197 float sig_diff;
198
199 if( doubleImageSize )
200 {
201 sig_diff = sqrtf( std::max(sigma * sigma - SIFT_INIT_SIGMA * SIFT_INIT_SIGMA * 4, 0.01f) );
202
203 Mat dbl;
204 if (enable_precise_upscale) {
205 dbl.create(Size(gray_fpt.cols*2, gray_fpt.rows*2), gray_fpt.type());
206 Mat H = Mat::zeros(2, 3, CV_32F);
207 H.at<float>(0, 0) = 0.5f;
208 H.at<float>(1, 1) = 0.5f;
209
210 cv::warpAffine(gray_fpt, dbl, H, dbl.size(), INTER_LINEAR | WARP_INVERSE_MAP, BORDER_REFLECT);
211 } else {
212#if DoG_TYPE_SHORT
213 resize(gray_fpt, dbl, Size(gray_fpt.cols*2, gray_fpt.rows*2), 0, 0, INTER_LINEAR_EXACT);
214#else
215 resize(gray_fpt, dbl, Size(gray_fpt.cols*2, gray_fpt.rows*2), 0, 0, INTER_LINEAR);
216#endif
217 }
218 Mat result;
219 GaussianBlur(dbl, result, Size(), sig_diff, sig_diff);
220 return result;
221 }
222 else
223 {
224 sig_diff = sqrtf( std::max(sigma * sigma - SIFT_INIT_SIGMA * SIFT_INIT_SIGMA, 0.01f) );
225 Mat result;
226 GaussianBlur(gray_fpt, result, Size(), sig_diff, sig_diff);
227 return result;
228 }

232void SIFT_Impl::buildGaussianPyramid( const Mat& base, std::vector<Mat>& pyr, int nOctaves ) const
233{
234 CV_TRACE_FUNCTION();
235
236 AutoBuffer<double> sig(nOctaveLayers + 3);
237 pyr.resize(nOctaves*(nOctaveLayers + 3));
238
239 // precompute Gaussian sigmas using the following formula:
240 // \sigma_{total}^2 = \sigma_{i}^2 + \sigma_{i-1}^2
241 sig[0] = sigma;
242 double k = std::pow( 2, 1. / nOctaveLayers );
243 for( int i = 1; i < nOctaveLayers + 3; i++ )
244 {
245 double sig_prev = (double)std::pow(k, i-1)*sigma;
246 double sig_total = sig_prev*k;
247 sig[i] = std::sqrt(sig_total*sig_total - sig_prev*sig_prev);
248 }
249
250 for( int o = 0; o < nOctaves; o++ )
251 {
252 for( int i = 0; i < nOctaveLayers + 3; i++ )
253 {
254 Mat& dst = pyr[o*(nOctaveLayers + 3) + i];
255 if( o == 0 && i == 0 )
256 dst = base;
257 // base of new octave is halved image from end of previous octave
258 else if( i == 0 )
259 {
260 const Mat& src = pyr[(o-1)*(nOctaveLayers + 3) + nOctaveLayers];
261 resize(src, dst, Size(src.cols/2, src.rows/2),
262 0, 0, INTER_NEAREST);
263 }
264 else
265 {
266 const Mat& src = pyr[o*(nOctaveLayers + 3) + i-1];
267 GaussianBlur(src, dst, Size(), sig[i], sig[i]);
268 }
269 }
270 }

373//
374// Detects features at extrema in DoG scale space. Bad features are discarded
375// based on contrast and ratio of principal curvatures.
376void SIFT_Impl::findScaleSpaceExtrema( const std::vector<Mat>& gauss_pyr, const std::vector<Mat>& dog_pyr,
377 std::vector<KeyPoint>& keypoints ) const
378{
379 CV_TRACE_FUNCTION();
380
381 const int nOctaves = (int)gauss_pyr.size()/(nOctaveLayers + 3);
382 const int threshold = cvFloor(0.5 * contrastThreshold / nOctaveLayers * 255 * SIFT_FIXPT_SCALE);
383
384 keypoints.clear();
385 TLSDataAccumulator<std::vector<KeyPoint> > tls_kpts_struct;
386
387 for( int o = 0; o < nOctaves; o++ )
388 for( int i = 1; i <= nOctaveLayers; i++ )
389 {
390 const int idx = o*(nOctaveLayers+2)+i;
391 const Mat& img = dog_pyr[idx];
392 const int step = (int)img.step1();
393 const int rows = img.rows, cols = img.cols;
394
395 parallel_for_(Range(SIFT_IMG_BORDER, rows-SIFT_IMG_BORDER),
396 findScaleSpaceExtremaComputer(
397 o, i, threshold, idx, step, cols,
398 nOctaveLayers,
399 contrastThreshold,
400 edgeThreshold,
401 sigma,
402 gauss_pyr, dog_pyr, tls_kpts_struct));
403 }
404
405 std::vector<std::vector<KeyPoint>*> kpt_vecs;
406 tls_kpts_struct.gather(kpt_vecs);
407 for (size_t i = 0; i < kpt_vecs.size(); ++i) {
408 keypoints.insert(keypoints.end(), kpt_vecs[i]->begin(), kpt_vecs[i]->end());
409 }
410}

426{
427public:
428 calcDescriptorsComputer(const std::vector<Mat>& _gpyr,
429 const std::vector<KeyPoint>& _keypoints,
430 Mat& _descriptors,
431 int _nOctaveLayers,
432 int _firstOctave)
433 : gpyr(_gpyr),

438
439 void operator()( const cv::Range& range ) const CV_OVERRIDE
440 {
441 CV_TRACE_FUNCTION();
442
443 const int begin = range.start;
444 const int end = range.end;
445
446 static const int d = SIFT_DESCR_WIDTH, n = SIFT_DESCR_HIST_BINS;
447
448 for ( int i = begin; i<end; i++ )

459 float angle = 360.f - kpt.angle;

461 angle = 0.f;
462 calcSIFTDescriptor(img, ptf, angle, size*0.5f, d, n, descriptors, i);
463 }
464 }
465private:
466 const std::vector<Mat>& gpyr;
467 const std::vector<KeyPoint>& keypoints;

469 int nOctaveLayers;

512static void computeGaussianKernel1D(double sigma, std::vector<float>& coeffs, int& radius)
513{
514 radius = (int)std::ceil(sigma * 3.0);
515 coeffs.resize(2 * radius + 1);
516 double sum = 0.0;
517 for (int i = -radius; i <= radius; i++)
518 {
519 double v = std::exp(-0.5 * (double)(i*i) / (sigma * sigma));
520 coeffs[i + radius] = (float)v;
521 sum += v;
522 }
523 for (int i = 0; i < (int)coeffs.size(); i++)
524 coeffs[i] = (float)(coeffs[i] / sum);
525}

527static bool ocl_gaussianBlurSep(const UMat& src, UMat& dst, double sigma)
528{
529 std::vector<float> coeffs;
530 int radius;
531 computeGaussianKernel1D(sigma, coeffs, radius);
532 UMat uc = Mat(coeffs, true).getUMat(ACCESS_READ);
533 dst.create(src.size(), src.type());
534 UMat tmp(src.size(), src.type());
535 size_t localSize[2] = {16, 16};
536 size_t globalSize[2] = {
537 (size_t)((src.cols + 15) / 16) * 16,
538 (size_t)((src.rows + 15) / 16) * 16
539 };
540
541 ocl::Kernel kerH("SIFT_gaussian_blur_h", ocl::features::sift_oclsrc);
542 if (kerH.empty())
543 return false;
544 if (!kerH.args(
545 ocl::KernelArg::ReadOnlyNoSize(src),
546 ocl::KernelArg::WriteOnlyNoSize(tmp),
547 src.cols, src.rows,
548 ocl::KernelArg::PtrReadOnly(uc), radius
549 ).run(2, globalSize, localSize, false))
550 return false;
551
552 ocl::Kernel kerV("SIFT_gaussian_blur_v", ocl::features::sift_oclsrc);
553 if (kerV.empty())
554 return false;
555 return kerV.args(
556 ocl::KernelArg::ReadOnlyNoSize(tmp),
557 ocl::KernelArg::WriteOnlyNoSize(dst),
558 src.cols, src.rows,
559 ocl::KernelArg::PtrReadOnly(uc), radius
560 ).run(2, globalSize, localSize, false);
561}

603static bool ocl_buildGaussianPyramid(const UMat& base, std::vector<UMat>& gauss_packs,
604 int nOctaves, int nOctaveLayers, double sigma)
605{
606 const int levelsPerOctave = nOctaveLayers + 3;
607 std::vector<double> sig(nOctaveLayers + 3);
608 gauss_packs.resize(nOctaves);
609
610 sig[0] = sigma;
611 double k = std::pow(2., 1. / nOctaveLayers);
612 for (int i = 1; i < nOctaveLayers + 3; i++)
613 {
614 double sig_prev = std::pow(k, i-1)*sigma;
615 double sig_total = sig_prev*k;
616 sig[i] = std::sqrt(sig_total*sig_total - sig_prev*sig_prev);
617 }
618
619 int n_kernels = nOctaveLayers + 2;
620 std::vector<std::vector<float> > gk_coeffs(n_kernels);
621 std::vector<int> gk_radius(n_kernels);
622 for (int i = 0; i < n_kernels; i++)
623 computeGaussianKernel1D(sig[i + 1], gk_coeffs[i], gk_radius[i]);
624
625 std::vector<UMat> uc(n_kernels);
626 for (int i = 0; i < n_kernels; i++)
627 uc[i] = Mat(gk_coeffs[i], true).getUMat(ACCESS_READ);
628
629 for (int o = 0; o < nOctaves; o++)
630 {
631 int levelW = base.cols >> o;
632 int levelH = base.rows >> o;
633 gauss_packs[o].create(levelsPerOctave * levelH, levelW, base.type());
634 }
635
636 for (int o = 0; o < nOctaves; o++)
637 {
638 int img_cols = gauss_packs[o].cols;
639 int img_rows = gauss_packs[o].rows / levelsPerOctave;
640
641 UMat level0 = gauss_packs[o].rowRange(0, img_rows);
642 if (o == 0)
643 {
644 base.copyTo(level0);
645 }
646 else
647 {
648 int prev_rows = gauss_packs[o-1].rows / levelsPerOctave;
649 UMat src = gauss_packs[o-1].rowRange(nOctaveLayers * prev_rows,
650 (nOctaveLayers + 1) * prev_rows);
651 resize(src, level0, Size(img_cols, img_rows), 0, 0, INTER_NEAREST);
652 }
653
654 UMat tmp(Size(img_cols, img_rows), base.type());
655 UMat prev = level0;
656 size_t localSize[2] = {16, 16};
657
658 for (int i = 1; i < levelsPerOctave; i++)
659 {
660 int lev_idx = i - 1;
661 UMat dst = gauss_packs[o].rowRange(i * img_rows, (i + 1) * img_rows);
662
663 size_t globalSize[2] = {
664 (size_t)((img_cols + 15) / 16) * 16,
665 (size_t)((img_rows + 15) / 16) * 16
666 };
667
668 ocl::Kernel kerH("SIFT_gaussian_blur_h", ocl::features::sift_oclsrc);
669 if (kerH.empty())
670 return false;
671 if (!kerH.args(
672 ocl::KernelArg::ReadOnlyNoSize(prev),
673 ocl::KernelArg::WriteOnlyNoSize(tmp),
674 img_cols, img_rows,
675 ocl::KernelArg::PtrReadOnly(uc[lev_idx]), gk_radius[lev_idx]
676 ).run(2, globalSize, localSize, false))
677 return false;
678
679 ocl::Kernel kerV("SIFT_gaussian_blur_v", ocl::features::sift_oclsrc);
680 if (kerV.empty())
681 return false;
682 if (!kerV.args(
683 ocl::KernelArg::ReadOnlyNoSize(tmp),
684 ocl::KernelArg::WriteOnlyNoSize(dst),
685 img_cols, img_rows,
686 ocl::KernelArg::PtrReadOnly(uc[lev_idx]), gk_radius[lev_idx]
687 ).run(2, globalSize, localSize, false))
688 return false;
689
690 prev = dst;
691 }
692 }
693 return true;

696static bool ocl_detectAndOrient(
697 const std::vector<UMat>& gauss_packs,
698 std::vector<KeyPoint>& keypoints,
699 int nOctaves, int nOctaveLayers,
700 double contrastThreshold, double edgeThreshold, double sigma)
701{
702 const int threshold = cvFloor(0.5 * contrastThreshold / nOctaveLayers * 255 * SIFT_FIXPT_SCALE);
703 const int max_kpts_per_call = 100000;
704
705 UMat count_umat(1, 1, CV_32S, Scalar::all(0));
706 UMat kp_buf(1, max_kpts_per_call * 6, CV_32F);
707
708 for (int o = 0; o < nOctaves; o++)
709 {
710 int dog_cols = gauss_packs[o].cols;
711 int dog_rows = gauss_packs[o].rows / (nOctaveLayers + 3);
712
713 for (int i = 1; i <= nOctaveLayers; i++)
714 {
715 size_t globalSize[2] = {(size_t)dog_cols, (size_t)dog_rows};
716
717 ocl::Kernel ker("SIFT_detect_and_orient", ocl::features::sift_oclsrc);
718 if (ker.empty())
719 return false;
720
721 bool ok = ker.args(
722 ocl::KernelArg::ReadOnlyNoSize(gauss_packs[o]),
723 dog_cols, dog_rows,
724 threshold,
725 (float)contrastThreshold, (float)edgeThreshold, (float)sigma,
726 nOctaveLayers, o, i,
727 ocl::KernelArg::PtrWriteOnly(count_umat),
728 ocl::KernelArg::PtrWriteOnly(kp_buf),
729 max_kpts_per_call
730 ).run(2, globalSize, 0, false);
731 if (!ok)
732 return false;
733 }
734 }
735
736 ocl::finish();
737
738 Mat count_mat = count_umat.getMat(ACCESS_READ);
739 int total = count_mat.at<int>(0, 0);
740 if (total > max_kpts_per_call)
741 total = max_kpts_per_call;
742
743 Mat kp = kp_buf.getMat(ACCESS_READ);
744 const float* kdata = (const float*)kp.data;
745
746 keypoints.resize(total);
747 for (int i = 0; i < total; i++)
748 {
749 KeyPoint& kpt = keypoints[i];
750 int base = i * 6;
751 kpt.pt.x = kdata[base + 0];
752 kpt.pt.y = kdata[base + 1];
753 kpt.angle = kdata[base + 2];
754 kpt.size = kdata[base + 3];
755 kpt.response = kdata[base + 4];
756 memcpy(&kpt.octave, &kdata[base + 5], sizeof(int));
757 kpt.class_id = -1;
758 }
759
760 return true;
761}

763static bool ocl_computeSIFTDescriptors(
764 const std::vector<UMat>& gauss_packs,
765 const std::vector<KeyPoint>& keypoints,
766 OutputArray _descriptors,
767 int nOctaveLayers, int firstOctave, int descriptor_type)
768{
769 const int dsize = SIFT_DESCR_WIDTH*SIFT_DESCR_WIDTH*SIFT_DESCR_HIST_BINS;
770 const int nkp = (int)keypoints.size();
771 if (nkp == 0)
772 {
773 _descriptors.create(0, dsize, descriptor_type);
774 return true;
775 }
776
777 const int levelsPerOctave = nOctaveLayers + 3;
778 const int nOctaves = (int)gauss_packs.size();
779
780 struct KptGroup { int start; int count; };
781 std::vector<KptGroup> groups(nOctaves, {0, 0});
782
783 for (int i = 0; i < nkp; i++)
784 {
785 int octave, layer; float scale;
786 unpackOctave(keypoints[i], octave, layer, scale);
787 int oct_idx = octave - firstOctave;
788 if (oct_idx < 0 || oct_idx >= nOctaves)
789 return false;
790 groups[oct_idx].count++;
791 }
792
793 int offset = 0;
794 for (int o = 0; o < nOctaves; o++)
795 {
796 groups[o].start = offset;
797 offset += groups[o].count;
798 }
799
800 Mat all_kpts(nkp, 5, CV_32F);
801 Mat all_rows(nkp, 1, CV_32S);
802 std::vector<int> cursors(nOctaves, 0);
803
804 for (int i = 0; i < nkp; i++)
805 {
806 int octave, layer; float scale;
807 unpackOctave(keypoints[i], octave, layer, scale);
808 int oct_idx = octave - firstOctave;
809 float size = keypoints[i].size * scale;
810 Point2f ptf(keypoints[i].pt.x * scale, keypoints[i].pt.y * scale);
811 float angle = 360.f - keypoints[i].angle;
812 if (std::abs(angle - 360.f) < FLT_EPSILON)
813 angle = 0.f;
814 float scl = size * 0.5f;
815 int pos = groups[oct_idx].start + cursors[oct_idx]++;
816 float* dst = all_kpts.ptr<float>(pos);
817 dst[0] = ptf.x; dst[1] = ptf.y; dst[2] = angle; dst[3] = scl;
818 dst[4] = (float)layer;
819 all_rows.ptr<int>(pos)[0] = i;
820 }
821
822 UMat kpts_buf; all_kpts.copyTo(kpts_buf);
823 UMat rows_buf; all_rows.copyTo(rows_buf);
824
825 _descriptors.create(nkp, dsize, descriptor_type);
826 UMat desc_out = _descriptors.getUMat();
827
828 for (int o = 0; o < nOctaves; o++)
829 {
830 int nGroup = groups[o].count;
831 if (nGroup == 0)
832 continue;
833
834 const UMat& packed = gauss_packs[o];
835 int img_cols = packed.cols;
836 int img_rows = packed.rows / levelsPerOctave;
837 float diag = sqrt((float)img_cols * img_cols + (float)img_rows * img_rows);
838
839 UMat kpts_roi = kpts_buf.rowRange(groups[o].start,
840 groups[o].start + nGroup);
841
842 ocl::Kernel ker("SIFT_compute_descriptor", ocl::features::sift_oclsrc);
843 if (ker.empty())
844 return false;
845
846 const int KP_PER_WG = 4;
847 const int WG_SIZE = KP_PER_WG * 16;
848 size_t localsz = WG_SIZE;
849 size_t globalsz = (size_t)(((nGroup + KP_PER_WG - 1) / KP_PER_WG) * WG_SIZE);
850 bool ok = ker.args(
851 ocl::KernelArg::ReadOnlyNoSize(packed), img_cols, img_rows, diag,
852 ocl::KernelArg::ReadOnlyNoSize(kpts_roi),
853 ocl::KernelArg::PtrReadOnly(rows_buf),
854 groups[o].start,
855 nGroup,
856 ocl::KernelArg::WriteOnlyNoSize(desc_out),
857 descriptor_type
858 ).run(1, &globalsz, &localsz, false);
859 if (!ok)
860 return false;
861 }
862
863 return true;
864}

868bool SIFT_Impl::ocl_detectAndCompute(InputArray _image, InputArray _mask,
869 std::vector<KeyPoint>& keypoints,
870 OutputArray _descriptors, bool useProvidedKeypoints)
871{
872 UMat image = _image.getUMat();
873 if (image.empty() || image.depth() != CV_8U)
874 return false;
875
876 int firstOctave = -1;
877 int nOctaves = 0;
878 if (useProvidedKeypoints)
879 {
880 firstOctave = 0;
881 int maxOctave = INT_MIN;
882 int actualNLayers = 0;
883 for (size_t i = 0; i < keypoints.size(); i++)
884 {
885 int octave, layer; float scale;
886 unpackOctave(keypoints[i], octave, layer, scale);
887 firstOctave = std::min(firstOctave, octave);
888 maxOctave = std::max(maxOctave, octave);
889 actualNLayers = std::max(actualNLayers, layer-2);
890 }
891 firstOctave = std::min(firstOctave, 0);
892 if (firstOctave < -1 || actualNLayers > nOctaveLayers)
893 return false;
894 nOctaves = maxOctave - firstOctave + 1;
895 }
896
897 UMat base = ocl_createInitialImage(image, firstOctave < 0, (float)sigma, enable_precise_upscale);
898 if (base.empty())
899 return false;
900 if (!useProvidedKeypoints)
901 nOctaves = cvRound(std::log((double)std::min(base.cols, base.rows)) / std::log(2.) - 2) - firstOctave;
902 if (nOctaves <= 0)
903 return false;
904
905 std::vector<UMat> gauss_packs;
906 if (!ocl_buildGaussianPyramid(base, gauss_packs, nOctaves, nOctaveLayers, sigma))
907 return false;
908
909 if (!useProvidedKeypoints)
910 {
911 if (!ocl_detectAndOrient(gauss_packs, keypoints, nOctaves, nOctaveLayers,
912 contrastThreshold, edgeThreshold, sigma))
913 return false;
914
915 KeyPointsFilter::removeDuplicatedSorted(keypoints);
916 if (nfeatures > 0)
917 KeyPointsFilter::retainBest(keypoints, nfeatures);
918
919 const float scale = 1.f/(float)(1 << -firstOctave);
920 for (size_t i = 0; i < keypoints.size(); i++)
921 {
922 KeyPoint& kpt = keypoints[i];
923 kpt.octave = (kpt.octave & ~255) | ((kpt.octave + firstOctave) & 255);
924 kpt.pt *= scale;
925 kpt.size *= scale;
926 }
927
928 if (!_mask.empty())
929 {
930 Mat mask = _mask.getMat();
931 KeyPointsFilter::runByPixelsMask(keypoints, mask);
932 }
933
934 if (_descriptors.needed())
935 {
936 if (!ocl_computeSIFTDescriptors(gauss_packs, keypoints, _descriptors, nOctaveLayers, firstOctave, descriptor_type))
937 return false;
938 }
939 }
940 else
941 {
942 if (_descriptors.needed())
943 {
944 if (!ocl_computeSIFTDescriptors(gauss_packs, keypoints, _descriptors, nOctaveLayers, firstOctave, descriptor_type))
945 return false;
946 }
947 }
948 return true;
949}
