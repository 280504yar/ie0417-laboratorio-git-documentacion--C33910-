37avifResult CopyToMat(const avifImage *image, int channels, bool useRGB , Mat *mat) {
38 CV_Assert((int)image->height == mat->rows);
39 CV_Assert((int)image->width == mat->cols);
40 if (channels == 1) {
41 const cv::Mat image_wrap =
42 cv::Mat(image->height, image->width,
43 CV_MAKE_TYPE((image->depth == 8) ? CV_8U : CV_16U, 1),
44 image->yuvPlanes[0], image->yuvRowBytes[0]);
45 if ((image->depth == 8 && mat->depth() == CV_8U) ||
46 (image->depth > 8 && mat->depth() == CV_16U)) {
47 image_wrap.copyTo(*mat);
48 } else {
49 CV_Assert(image->depth > 8 && mat->depth() == CV_8U);
50 image_wrap.convertTo(*mat, CV_8U, 1. / (1 << (image->depth - 8)));
51 }
52 return AVIF_RESULT_OK;
53 }
54 avifRGBImage rgba;
55 avifRGBImageSetDefaults(&rgba, image);
56 if (channels == 3) {
57 if (useRGB)
58 rgba.format = AVIF_RGB_FORMAT_RGB;
59 else
60 rgba.format = AVIF_RGB_FORMAT_BGR;
61 } else {
62 CV_Assert(channels == 4);
63 rgba.format = AVIF_RGB_FORMAT_BGRA;
64 }
65 rgba.rowBytes = mat->step[0];
66 rgba.depth = (mat->depth() == CV_16U) ? image->depth : 8;
67 rgba.pixels = reinterpret_cast<uint8_t *>(mat->data);
68 return avifImageYUVToRGB(image, &rgba);
69}

71AvifImageUniquePtr ConvertToAvif(const cv::Mat &img, bool lossless, int bit_depth,
72 const std::vector<std::vector<uchar> >& metadata) {
73 CV_Assert(img.depth() == CV_8U || img.depth() == CV_16U);
74
75 const int width = img.cols;
76 const int height = img.rows;
77
78 avifImage *result;
79
80 if (img.channels() == 1) {
81 result = avifImageCreateEmpty();
82 if (result == nullptr) return nullptr;
83 result->width = width;
84 result->height = height;
85 result->depth = bit_depth;
86 result->yuvFormat = AVIF_PIXEL_FORMAT_YUV400;
87 result->colorPrimaries = AVIF_COLOR_PRIMARIES_UNSPECIFIED;
88 result->transferCharacteristics = AVIF_TRANSFER_CHARACTERISTICS_UNSPECIFIED;
89 result->matrixCoefficients = AVIF_MATRIX_COEFFICIENTS_UNSPECIFIED;
90 result->yuvRange = AVIF_RANGE_FULL;
91 result->yuvPlanes[0] = img.data;
92 result->yuvRowBytes[0] = img.step[0];
93 result->imageOwnsYUVPlanes = AVIF_FALSE;
94 } else if (lossless) {
95 result =
96 avifImageCreate(width, height, bit_depth, AVIF_PIXEL_FORMAT_YUV444);
97 if (result == nullptr) return nullptr;
98 result->colorPrimaries = AVIF_COLOR_PRIMARIES_UNSPECIFIED;
99 result->transferCharacteristics = AVIF_TRANSFER_CHARACTERISTICS_UNSPECIFIED;
100 result->matrixCoefficients = AVIF_MATRIX_COEFFICIENTS_IDENTITY;
101 result->yuvRange = AVIF_RANGE_FULL;
102 } else {
103 result =
104 avifImageCreate(width, height, bit_depth, AVIF_PIXEL_FORMAT_YUV420);
105 if (result == nullptr) return nullptr;
106 result->colorPrimaries = AVIF_COLOR_PRIMARIES_BT709;
107 result->transferCharacteristics = AVIF_TRANSFER_CHARACTERISTICS_SRGB;
108 result->matrixCoefficients = AVIF_MATRIX_COEFFICIENTS_BT601;
109 result->yuvRange = AVIF_RANGE_FULL;
110 }
111
112 if (!metadata.empty()) {
113 const std::vector<uchar>& metadata_exif = metadata[IMAGE_METADATA_EXIF];
114 const std::vector<uchar>& metadata_xmp = metadata[IMAGE_METADATA_XMP];
115 const std::vector<uchar>& metadata_iccp = metadata[IMAGE_METADATA_ICCP];
116#if AVIF_VERSION_MAJOR >= 1
117 if ((!metadata_exif.empty() &&
118 avifImageSetMetadataExif(result, (const uint8_t *)metadata_exif.data(),
119 metadata_exif.size()) != AVIF_RESULT_OK) ||
120 (!metadata_xmp.empty() &&
121 avifImageSetMetadataXMP(result, (const uint8_t *)metadata_xmp.data(),
122 metadata_xmp.size()) != AVIF_RESULT_OK) ||
123 (!metadata_iccp.empty() &&
124 avifImageSetProfileICC(result, (const uint8_t *)metadata_iccp.data(),
125 metadata_iccp.size()) != AVIF_RESULT_OK)) {
126 avifImageDestroy(result);
127 return nullptr;
128 }
129#else
130 if (!metadata_exif.empty())
131 avifImageSetMetadataExif(result, (const uint8_t*)metadata_exif.data(), metadata_exif.size());
132 if (!metadata_xmp.empty())
133 avifImageSetMetadataXMP(result, (const uint8_t*)metadata_xmp.data(), metadata_xmp.size());
134 if (!metadata_iccp.empty())
135 avifImageSetProfileICC(result, (const uint8_t*)metadata_iccp.data(), metadata_iccp.size());
136#endif
137 }
138
139 if (img.channels() > 1) {
140 avifRGBImage rgba;
141 avifRGBImageSetDefaults(&rgba, result);
142 if (img.channels() == 3) {
143 rgba.format = AVIF_RGB_FORMAT_BGR;
144 } else {
145 CV_Assert(img.channels() == 4);
146 rgba.format = AVIF_RGB_FORMAT_BGRA;
147 }
148 rgba.rowBytes = (uint32_t)img.step[0];
149 rgba.depth = bit_depth;
150 rgba.pixels =
151 const_cast<uint8_t *>(reinterpret_cast<const uint8_t *>(img.data));
152
153 if (avifImageRGBToYUV(result, &rgba) != AVIF_RESULT_OK) {
154 avifImageDestroy(result);
155 return nullptr;
156 }
157 }
158 return AvifImageUniquePtr(result);
159}

192bool AvifDecoder::checkSignature(const String &signature) const {
193 std::unique_ptr<avifDecoder, decltype(&avifDecoderDestroy)> decoder(
194 avifDecoderCreate(), avifDecoderDestroy);
195 if (!decoder) return false;
196 decoder->strictFlags = AVIF_STRICT_DISABLED;
197 OPENCV_AVIF_CHECK_STATUS(
198 avifDecoderSetIOMemory(
199 decoder.get(), reinterpret_cast<const uint8_t *>(signature.c_str()),
200 signature.size()),
201 decoder);
202 decoder->io->sizeHint = 1e9;
203 const avifResult status = avifDecoderParse(decoder.get());
204 return (status == AVIF_RESULT_OK || status == AVIF_RESULT_TRUNCATED_DATA);
205}

209bool AvifDecoder::readHeader() {
210 if (decoder_)
211 return true;
212
213 decoder_ = avifDecoderCreate();
214 if (!decoder_) {
215 CV_Error(Error::StsNoMem, "Failed to create AVIF decoder");
216 return false;
217 }
218 decoder_->strictFlags = AVIF_STRICT_DISABLED;
219 if (!m_buf.empty()) {
220 CV_Assert(m_buf.type() == CV_8UC1);
221 CV_Assert(m_buf.rows == 1);
222 }
223
224 OPENCV_AVIF_CHECK_STATUS(
225 m_buf.empty()
226 ? avifDecoderSetIOFile(decoder_, m_filename.c_str())
227 : avifDecoderSetIOMemory(
228 decoder_, reinterpret_cast<const uint8_t *>(m_buf.data),
229 m_buf.total()),
230 decoder_);
231 OPENCV_AVIF_CHECK_STATUS(avifDecoderParse(decoder_), decoder_);
232
233 if (!decoder_->image) {
234 CV_Error(Error::StsParseError, "AVIF image is null after parsing");
235 return false;
236 }
237
238 m_width = decoder_->image->width;
239 m_height = decoder_->image->height;
240 m_frame_count = decoder_->imageCount;
241 channels_ = (decoder_->image->yuvFormat == AVIF_PIXEL_FORMAT_YUV400) ? 1 : 3;
242 if (decoder_->alphaPresent) ++channels_;
243 bit_depth_ = decoder_->image->depth;
244 CV_Assert(bit_depth_ == 8 || bit_depth_ == 10 || bit_depth_ == 12);
245 m_type = CV_MAKETYPE(bit_depth_ == 8 ? CV_8U : CV_16U, channels_);
246 is_first_image_ = true;
247 return true;
248}

250bool AvifDecoder::readData(Mat &img) {
251 CV_CheckGE(m_width, 0, "");
252 CV_CheckGE(m_height, 0, "");
253
254 CV_CheckEQ(img.cols, m_width, "");
255 CV_CheckEQ(img.rows, m_height, "");
256 CV_CheckType(
257 img.type(),
258 (img.channels() == 1 || img.channels() == 3 || img.channels() == 4) &&
259 (img.depth() == CV_8U || img.depth() == CV_16U),
260 "AVIF only supports 1, 3, 4 channels and CV_8U and CV_16U");
261
262 Mat read_img;
263 if (img.channels() == channels_) {
264 read_img = img;
265 } else {
266 // Use the asked depth but keep the number of channels. OpenCV and not
267 // libavif will do the color conversion.
268 read_img.create(m_height, m_width, CV_MAKE_TYPE(img.depth(), channels_));
269 }
270
271 if (is_first_image_) {
272 if (!nextPage()) return false;
273 is_first_image_ = false;
274 }
275
276 if (CopyToMat(decoder_->image, channels_, m_use_rgb, &read_img) != AVIF_RESULT_OK) {
277 CV_Error(Error::StsInternal, "Cannot convert from AVIF to Mat");
278 return false;
279 }
280
281 m_animation.durations.push_back(decoder_->imageTiming.duration * 1000);
282
283 if (decoder_->image->exif.size > 0) {
284 m_exif.parseExif(decoder_->image->exif.data, decoder_->image->exif.size);
285 }
286
287 if (img.channels() == channels_) {
288 // We already wrote to the right buffer.
289 } else {
290 if (channels_ == 1 && img.channels() == 3) {
291 cvtColor(read_img, img, COLOR_GRAY2BGR);
292 } else if (channels_ == 1 && img.channels() == 4) {
293 cvtColor(read_img, img, COLOR_GRAY2BGRA);
294 } else if (channels_ == 3 && img.channels() == 1) {
295 cvtColor(read_img, img, COLOR_BGR2GRAY);
296 } else if (channels_ == 3 && img.channels() == 4) {
297 cvtColor(read_img, img, COLOR_BGR2BGRA);
298 } else if (channels_ == 4 && img.channels() == 1) {
299 cvtColor(read_img, img, COLOR_BGRA2GRAY);
300 } else if (channels_ == 4 && img.channels() == 3) {
301 cvtColor(read_img, img, COLOR_BGRA2BGR);
302 } else {
303 CV_Error(Error::StsInternal, "");
304 }
305 }
306 return true;
307}

309bool AvifDecoder::nextPage() {
310 const avifResult status = avifDecoderNextImage(decoder_);
311 if (status == AVIF_RESULT_NO_IMAGES_REMAINING) return false;
312 if (status != AVIF_RESULT_OK) {
313 const std::string error(decoder_->diag.error);
314 CV_Error(Error::StsParseError, error + " " + avifResultToString(status));
315 return false;
316 }
317 return true;
318}

341bool AvifEncoder::writeanimation(const Animation& animation,
342 const std::vector<int> &params) {
343 int bit_depth = 8;
344 int speed = AVIF_SPEED_FASTEST;
345 for (size_t i = 0; i < params.size(); i += 2) {
346 const int value = params[i + 1];
347 if (params[i] == IMWRITE_AVIF_QUALITY) {
348 const int quality = std::min(std::max(value, AVIF_QUALITY_WORST),
349 AVIF_QUALITY_BEST);
350 if (value != quality) {
351 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_AVIF_QUALITY must be between 0 to 100. It is fallbacked to %d", value, quality));
352 }
353#if CV_AVIF_USE_QUALITY
354 encoder_->quality = quality;
355#else
356 encoder_->minQuantizer = encoder_->maxQuantizer =
357 (AVIF_QUANTIZER_BEST_QUALITY - AVIF_QUANTIZER_WORST_QUALITY) *
358 quality / (AVIF_QUALITY_BEST - AVIF_QUALITY_WORST) +
359 AVIF_QUANTIZER_WORST_QUALITY;
360#endif
361 } else if (params[i] == IMWRITE_AVIF_DEPTH) {
362 bit_depth = value;
363 if ((bit_depth != 8) && (bit_depth !=10) && (bit_depth !=12))
364 {
365 bit_depth = 8;
366 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_AVIF_DEPTH must be 8, 10 or 12. It is fallbacked to %d", value, bit_depth));
367 }
368 } else if (params[i] == IMWRITE_AVIF_SPEED) {
369 speed = std::min(std::max(value,0),10);
370 if (value != speed) {
371 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_AVIF_SPEED must be between 0 to 10. It is fallbacked to %d", value, speed));
372 }
373 }
374 }
375
376 avifRWData output_ori = AVIF_DATA_EMPTY;
377 std::unique_ptr<avifRWData, decltype(&avifRWDataFree)> output(&output_ori,
378 avifRWDataFree);
379#if CV_AVIF_USE_QUALITY
380 const bool do_lossless = (encoder_->quality == AVIF_QUALITY_LOSSLESS);
381#else
382 const bool do_lossless =
383 (encoder_->minQuantizer == AVIF_QUANTIZER_BEST_QUALITY &&
384 encoder_->maxQuantizer == AVIF_QUANTIZER_BEST_QUALITY);
385#endif
386 encoder_->speed = speed;
387
388 const avifAddImageFlags flag = (animation.frames.size() == 1)
389 ? AVIF_ADD_IMAGE_FLAG_SINGLE
390 : AVIF_ADD_IMAGE_FLAG_NONE;
391 std::vector<AvifImageUniquePtr> images;
392 for (const cv::Mat &img : animation.frames) {
393 CV_CheckType(
394 img.type(),
395 (bit_depth == 8 && img.depth() == CV_8U) ||
396 ((bit_depth == 10 || bit_depth == 12) && img.depth() == CV_16U),
397 "AVIF only supports bit depth of 8 with CV_8U input or "
398 "bit depth of 10 or 12 with CV_16U input");
399
400 CV_Check(img.channels(),
401 img.channels() == 1 || img.channels() == 3 || img.channels() == 4,
402 "AVIF only supports 1, 3, 4 channels");
403
404 auto avifImg = ConvertToAvif(img, do_lossless, bit_depth, m_metadata);
405 if (!avifImg) {
406 CV_Error(Error::StsError, "Failed to convert Mat to AVIF image");
407 return false;
408 }
409 images.emplace_back(std::move(avifImg));
410 }
411
412 for (size_t i = 0; i < images.size(); i++)
413 {
414 OPENCV_AVIF_CHECK_STATUS(
415 avifEncoderAddImage(encoder_, images[i].get(), animation.durations[i], flag),
416 encoder_);
417 }
418
419 encoder_->timescale = 1000;
420 OPENCV_AVIF_CHECK_STATUS(avifEncoderFinish(encoder_, output.get()), encoder_);
421
422 if (m_buf) {
423 m_buf->resize(output->size);
424 std::memcpy(m_buf->data(), output->data, output->size);
425 } else {
426 std::ofstream(m_filename, std::ofstream::binary)
427 .write(reinterpret_cast<char *>(output->data), output->size);
428 }
429
430 return (output->size > 0);
431}
