270 template <class ForwardItr>
271 typename std::enable_if<cxx_utils::is_forward_iterator<ForwardItr>::value, void>
272 ::type reshape_(ForwardItr start, ForwardItr end) {
273 CV_Assert(start != end);
274
275 using ItrValueType = typename std::iterator_traits<ForwardItr>::value_type;
276
277 /* the user may leave at most one axis size for deduction by specifying -1 */
278 auto sizes_to_deduce = std::count(start, end, -1);
279 if (sizes_to_deduce > 1) { CV_Error(Error::StsBadArg, "only one axis size can be deduced"); }
280
281 /* sizes must be positive numbers with the exception of -1 */
282 auto invalid_sizes = std::count_if(start, end, [](ItrValueType x) {
283 return !(x > 0 || x == -1);
284 });
285 if (invalid_sizes) { CV_Error(Error::StsBadArg, "invalid axis size"); }
286
287 /* compute the total number of elements in the new tensor */
288 size_type unknown_size = 0;
289 auto total = std::accumulate(start, end, 1, std::multiplies<ItrValueType>());
290 if (total < 0) {
291 /* there is an unknown size */
292 CV_CheckEQ(size() % std::abs(total), static_cast<size_type>(0), "cannot be reshaped"); // must be divisible
293 if (std::abs(total) <= size()) {
294 unknown_size = size() / std::abs(total);
295 total = size();
296 }
297 /* Edge case: if `total` is already more than size(), skip the deduction as it's impossible
298 ** Since `total` is negative, the size check which follows will fail and throw an error
299 */
300 }
301
302 /* the number of elements before and after reshape must be exactly same */
303 if (total != size()) {
304 CV_Error(Error::StsBadArg, "new axes do not preserve the tensor element count");
305 }
306
307 /* copy shape from given iterator and reshape -1 with deduced value */
308 shape.resize(std::distance(start, end));
309 std::copy(start, end, shape.begin());
310 std::replace(std::begin(shape), std::end(shape), size_type(-1), unknown_size);
311 }

767 template <class ForwardItr>
768 typename std::enable_if<cxx_utils::is_forward_iterator<ForwardItr>::value, TensorSpan>
769 ::type subspan_(size_type offset, ForwardItr start, ForwardItr end) const {
770 CV_Assert(start != end);
771 CV_Assert(std::distance(start, end) <= rank());
772
773 auto cur_size = size();
774 CV_Assert(offset < cur_size);
775
776 using ItrValueType = typename std::iterator_traits<ForwardItr>::value_type;
777
778 /* sizes must be positive numbers */
779 auto invalid_sizes = std::count_if(start, end, [](ItrValueType x) {
780 return !(x > 0);
781 });
782 if (invalid_sizes) { CV_Error(Error::StsBadArg, "invalid axis size"); }
783
784 /* the number of elements must be equal to the new size */
785 auto max_size = (cur_size - offset);
786 auto total = std::accumulate(start, end, 1, std::multiplies<ItrValueType>());
787 if (total > max_size) {
788 CV_Error(Error::StsBadArg, "axis lengths lead to OOB accesses");
789 }
790
791 TensorSpan temp;
792 temp.shape.assign(start, end);
793 temp.ptr = ptr + offset;
794 return temp;
795 }

1188 temp.ptr = ptr + offset;
