62float calcSharpness(cv::InputArray _values,float rise_distance)
63{
64 CV_CheckTypeEQ(_values.type(),CV_8UC1, "values must be of the type CV_8UC1");
65 cv::Mat values = normalizeVector(_values);
66 if(values.empty())
67 return 0;
68 if(values.rows != 1 && values.cols != 1)
69 CV_Error(Error::StsBadArg, "values must be 1xn or nx1");
70 if(rise_distance <= 0.0 || rise_distance > 1.0)
71 CV_Error(Error::StsBadArg, "rise_distance must lie in th interval ]0..1]");
72
73 // find global min max
74 cv::Point min_loc,max_loc;
75 double min_val,max_val;
76 cv::minMaxLoc(values,&min_val,&max_val,&min_loc,&max_loc);
77 int max_pos = std::max(max_loc.x,max_loc.y);
78 int min_pos = std::max(min_loc.x,min_loc.y);
79 if(max_pos == min_pos)
80 return 0;
81
82 // calc new interval according to the rise distance
83 double delta = max_val-min_val;
84 double min_val2 = min_val+delta*0.5*(1.0-rise_distance);
85 double max_val2 = max_val-delta*0.5*(1.0-rise_distance);
86
87 // find new max starting at min pos
88 int dt = 1;
89 if(max_pos < min_pos)
90 dt= -1;
91 int max_pos2 = max_pos;
92 for(int i=min_pos+dt;i != max_pos;i+=dt)
93 {
94 uint8_t val = values.at<uint8_t>(i);
95 if(val >= max_val2)
96 {
97 max_pos2 = i;
98 break;
99 }
100 }
101
102 // find new min starting at max pos
103 int min_pos2 = min_pos;
104 for(int i=max_pos-dt;i != min_pos;i-=dt)
105 {
106 uint8_t val = values.at<uint8_t>(i);
107 if(val <= min_val2)
108 {
109 min_pos2 = i;
110 break;
111 }
112 }
113
114 // calc sub pixel max pos
115 double max_pos3 = max_pos2;
116 uint8_t val1 = values.at<uint8_t>(max_pos2-dt); // <= val2
117 uint8_t val2 = values.at<uint8_t>(max_pos2);
118 double m = (val2-val1)/dt;
119 if(m != 0)
120 max_pos3 = max_pos2+(max_val2-val2)/m;
121
122 // calc sub pixel min pos
123 double min_pos3 = min_pos2;
124 val1 = values.at<uint8_t>(min_pos2); // <= val2
125 val2 = values.at<uint8_t>(min_pos2+dt);
126 m = (val2-val1)/dt;
127 if(m != 0)
128 min_pos3 = min_pos2+(min_val2-val1)/m;
129
130 return float(fabs(max_pos3-min_pos3));
131}

134void normalizePoints1D(cv::InputArray _points,cv::OutputArray _T,cv::OutputArray _new_points)
135{
136 cv::Mat points = _points.getMat();
137 if(points.cols > 1 && points.rows == 1)
138 points = points.reshape(1,points.cols);
139 CV_CheckChannelsEQ(points.channels(), 1, "points must have only one channel");
140
141 // calc centroid
142 double centroid= cv::mean(points)[0];
143
144 // shift origin to centroid
145 cv::Mat new_points = points-centroid;
146
147 // calc mean distance
148 double mean_dist = cv::mean(cv::abs(new_points))[0];
149 if(mean_dist<= DBL_EPSILON)
150 CV_Error(Error::StsBadArg, "all given points are identical");
151 double scale = 1.0/mean_dist;
152
153
154 // generate transformation
155 cv::Matx22d Tx(
156 scale, -scale*centroid,
157 0, 1
158 );
159 Mat(Tx, false).copyTo(_T);
160
161 // calc normalized points;
162 _new_points.create(points.rows,1,points.type());
163 new_points = _new_points.getMat();
164 switch(points.type())
165 {
166 case CV_32FC1:
167 for(int i=0;i < points.rows;++i)
168 {
169 cv::Vec2d p(points.at<float>(i), 1.0);
170 p = Tx*p;
171 new_points.at<float>(i) = float(p(0)/p(1));
172 }
173 break;
174 case CV_64FC1:
175 for(int i=0;i < points.rows;++i)
176 {
177 cv::Vec2d p(points.at<double>(i), 1.0);
178 p = Tx*p;
179 new_points.at<double>(i) = p(0)/p(1);
180 }
181 break;
182 default:
183 CV_Error(Error::StsUnsupportedFormat, "unsupported point type");
184 }
185}

187cv::Mat findHomography1D(cv::InputArray _src,cv::InputArray _dst)
188{
189 // check inputs
190 cv::Mat src = _src.getMat();
191 cv::Mat dst = _dst.getMat();
192 if(src.cols > 1 && src.rows == 1)
193 src = src.reshape(1,src.cols);
194 if(dst.cols > 1 && dst.rows == 1)
195 dst = dst.reshape(1,dst.cols);
196 CV_CheckEQ(src.rows, dst.rows, "size mismatch");
197 CV_CheckChannelsEQ(src.channels(), 1, "data with only one channel are supported");
198 CV_CheckChannelsEQ(dst.channels(), 1, "data with only one channel are supported");
199 CV_CheckTypeEQ(src.type(), dst.type(), "src and dst must have the same type");
200 CV_Check(src.rows, src.rows >= 3,"at least three point pairs are needed");
201
202 // normalize points
203 cv::Mat src_T,dst_T, src_n,dst_n;
204 normalizePoints1D(src,src_T,src_n);
205 normalizePoints1D(dst,dst_T,dst_n);
206
207 int count = src_n.rows;
208 cv::Mat A = cv::Mat::zeros(count,3,CV_64FC1);
209 cv::Mat b = cv::Mat::zeros(count,1,CV_64FC1);
210
211 // fill A;b and perform singular value decomposition
212 // it is assumed that w is one for both coordinates
213 // h22 is kept to 1
214 switch(src_n.type())
215 {
216 case CV_32FC1:
217 for(int i=0;i<count;++i)
218 {
219 double s = src_n.at<float>(i);
220 double d = dst_n.at<float>(i);
221 A.at<double>(i,0) = s;
222 A.at<double>(i,1) = 1.0;
223 A.at<double>(i,2) = -s*d;
224 b.at<double>(i) = d;
225 }
226 break;
227 case CV_64FC1:
228 for(int i=0;i<count;++i)
229 {
230 double s = src_n.at<double>(i);
231 double d = dst_n.at<double>(i);
232 A.at<double>(i,0) = s;
233 A.at<double>(i,1) = 1.0;
234 A.at<double>(i,2) = -s*d;
235 b.at<double>(i) = d;
236 }
237 break;
238 default:
239 CV_Error(Error::StsUnsupportedFormat,"unsupported type");
240 }
241
242 cv::Mat u,d,vt;
243 cv::SVD::compute(A,d,u,vt);
244 cv::Mat b_ = u.t()*b;
245
246 cv::Mat y(b_.rows,1,CV_64FC1);
247 for(int i=0;i<b_.rows;++i)
248 y.at<double>(i) = b_.at<double>(i)/d.at<double>(i);
249
250 cv::Mat x = vt.t()*y;
251 cv::Matx22d H_(x.at<double>(0), x.at<double>(1), x.at<double>(2), 1.0);
252
253 // denormalize
254 Mat H = dst_T.inv()*Mat(H_, false)*src_T;
255
256 // enforce frobeniusnorm of one
257 double scale = cv::norm(H);
258 CV_Assert(fabs(scale) > DBL_EPSILON);
259 scale = 1.0 / scale;
260 return H*scale;
261}
262void polyfit(const Mat& src_x, const Mat& src_y, Mat& dst, int order)
263{
264 int npoints = src_x.checkVector(1);
265 int nypoints = src_y.checkVector(1);
266 CV_Assert(npoints == nypoints && npoints >= order+1);
267 Mat_<double> srcX(src_x), srcY(src_y);
268 Mat_<double> A = Mat_<double>::ones(npoints,order + 1);
269 // build A matrix
270 for (int y = 0; y < npoints; ++y)
271 {
272 for (int x = 1; x < A.cols; ++x)
273 A.at<double>(y,x) = srcX.at<double>(y)*A.at<double>(y,x-1);
274 }
275 cv::Mat w;
276 solve(A,srcY.reshape(1, npoints),w,DECOMP_SVD);
277 w.convertTo(dst, ((src_x.depth() == CV_64F || src_y.depth() == CV_64F) ? CV_64F : CV_32F));
278}

296// returns how many tests fails out of 10
297int testPointSymmetry(const cv::Mat &mat,cv::Point2f pt,float dist,float max_error)
298{
299 cv::Rect image_rect(int(0.5*dist),int(0.5*dist),int(mat.cols-0.5*dist),int(mat.rows-0.5*dist));
300 cv::Size size(int(0.5*dist),int(0.5*dist));
301 int count = 0;
302 cv::Mat patch1,patch2;
303 cv::Point2f center1,center2;
304 for (int angle_i = 0; angle_i < 10; angle_i++)
305 {
306 double angle = angle_i * (CV_PI * 0.1);
307 cv::Point2f n(float(std::cos(angle)),float(-std::sin(angle)));
308 center1 = pt+dist*n;
309 if(!image_rect.contains(center1))
310 return false;
311 center2 = pt-dist*n;
312 if(!image_rect.contains(center2))
313 return false;
314 cv::getRectSubPix(mat,size,center1,patch1);
315 cv::getRectSubPix(mat,size,center2,patch2);
316 if(fabs(cv::mean(patch1)[0]-cv::mean(patch2)[0]) > max_error)
317 ++count;
318 }
319 return count;
320}

322inline float calcSubpixel(const float &x_l,const float &x,const float &x_r)
323{
324 // prevent zero values
325 if(x_l <= 0)
326 return 0;
327 if(x <= 0)
328 return 0;
329 if(x_r <= 0)
330 return 0;
331 const float l0 = float(std::log(x_l+1e-6));
332 const float l1 = float(std::log(x+1e-6));
333 const float l2 = float(std::log(x_r+1e-6));
334 float delta = l2-l1-l1+l0;
335 if(delta == 0.0f) // this happens if all values are identical
336 return 0;
337 delta = (l0-l2)/(delta+delta);
338 return delta;
339}

341inline float calcSubPos(const float &x_l,const float &x,const float &x_r)
342{
343 float val = 2.0F *(x_l-2.0F*x+x_r);
344 if(val == 0.0F)
345 return 0.0F;
346 val = (x_l-x_r)/val;
347 if(val > 1.0F)
348 return 1.0F;
349 if(val < -1.0F)
350 return -1.0F;
351 return val;
352}

382void FastX::calcFeatureMap(const Mat &images,Mat& out)const
383{
384 if(images.empty())
385 CV_Error(Error::StsBadArg,"no rotation images");
386 int type = images.type(), depth = CV_MAT_DEPTH(type);
387 CV_CheckType(type,depth == CV_8U,
388 "Only 8-bit grayscale or color images are supported");
389 if(!images.isContinuous())
390 CV_Error(Error::StsBadArg,"image must be continuous");
391
392 float signal,noise,rating;
393 int count1;
394 unsigned char val1,val2,val3;
395 const unsigned char* wrap_around;
396 const unsigned char* pend;
397 const unsigned char* pimages = images.data;
398 const int channels = images.channels();
399 if(channels < 4)
400 CV_Error(Error::StsBadArg,"images must have at least four channels");
401
402 // for each pixel
403 out = cv::Mat::zeros(images.rows,images.cols,CV_32FC1);
404 const float *pout_end = reinterpret_cast<const float*>(out.dataend);
405 for(float *pout=out.ptr<float>(0,0);pout != pout_end;++pout)
406 {
407 //reset values
408 rating = 0.0; count1 = 0;
409 noise = 255; signal = 0;
410
411 //calc rating
412 pend = pimages+channels;
413 val1 = *(pend-1); // wrap around (last value)
414 wrap_around = pimages++; // store for wrap around (first value)
415 val2 = *wrap_around; // first value
416 for(;pimages != pend;++pimages)
417 {
418 val3 = *pimages;
419 if(val1 <= val2)
420 {
421 if(val3 < val2) // maxima
422 {
423 if(signal < val2)
424 signal = val2;
425 ++count1;
426 }
427 }
428 else if(val1 > val2 && val3 >= val2) // minima
429 {
430 if(noise > val2)
431 noise = val2;
432 ++count1;
433 }
434 val1 = val2;
435 val2 = val3;
436 }
437 // wrap around
438 if(val1 <= val2) // maxima
439 {
440 if(*wrap_around < val2)
441 {
442 if(signal < val2)
443 signal = val2;
444 ++count1;
445 }
446 }
447 else if(val1 > val2 && *wrap_around >= val2) // minima
448 {
449 if(noise > val2)
450 noise = val2;
451 ++count1;
452 }
453
454 // store rating
455 if(count1 == parameters.branches)
456 {
457 rating = signal-noise;
458 *pout = rating*rating; //store rating in the feature map
459 }
460 }

463std::vector<std::vector<float> > FastX::calcAngles(const std::vector<cv::Mat> &rotated_images,std::vector<cv::KeyPoint> &keypoints)const
464{
465 // validate rotated_images
466 if(rotated_images.empty())
467 CV_Error(Error::StsBadArg,"no rotated images");
468 std::vector<cv::Mat>::const_iterator iter = rotated_images.begin();
469 for(;iter != rotated_images.end();++iter)
470 {
471 if(iter->empty())
472 CV_Error(Error::StsBadArg,"empty rotated images");
473 if(iter->channels() < 4)
474 CV_Error(Error::StsBadArg,"rotated images must have at least four channels");
475 }
476
477 // assuming all elements of the same channel
478 const int channels = rotated_images.front().channels();
479 const int channels_1 = channels-1;
480 const float resolution = float(CV_PI/channels);
481 const float scale = float(parameters.super_resolution)+1.0F;
482
483 // for each keypoint
484 std::vector<std::vector<float> > angles;
485 angles.resize(keypoints.size());
486 parallel_for_(Range(0,(int)keypoints.size()),[&](const Range& range)
487 {
488 float angle;
489 float val1,val2,val3,wrap_around;
490 const unsigned char *pimages1,*pimages2,*pimages3,*pimages4;
491 std::vector<cv::KeyPoint>::iterator pt_iter = keypoints.begin()+range.start;
492 std::vector<cv::KeyPoint>::iterator pt_end = keypoints.begin()+range.end;
493 for(int id=range.start ;pt_iter != pt_end;++pt_iter,++id)
494 {
495 int scale_id = pt_iter->octave - parameters.min_scale;
496 if(scale_id>= int(rotated_images.size()) ||scale_id < 0)
497 CV_Error(Error::StsBadArg,"no rotated image for requested keypoint octave");
498 const cv::Mat &s_rotated_images = rotated_images[scale_id];
499
500 float x2 = pt_iter->pt.x*scale;
501 float y2 = pt_iter->pt.y*scale;
502 int row = int(y2);
503 int col = int(x2);
504 x2 -= col;
505 y2 -= row;
506 float x1 = 1.0F-x2; float y1 = 1.0F-y2;
507 float a = x1*y1; float b = x2*y1; float c = x1*y2; float d = x2*y2;
508 pimages1 = s_rotated_images.ptr<unsigned char>(row,col);
509 pimages2 = s_rotated_images.ptr<unsigned char>(row,col+1);
510 pimages3 = s_rotated_images.ptr<unsigned char>(row+1,col);
511 pimages4 = s_rotated_images.ptr<unsigned char>(row+1,col+1);
512 std::vector<float> &angles_i = angles[id];
513
514 //calc rating
515 val1 = a**(pimages1+channels_1)+b**(pimages2+channels_1)+
516 c**(pimages3+channels_1)+d**(pimages4+channels_1); // wrap around (last value)
517 wrap_around = a**(pimages1++)+b**(pimages2++)+c**(pimages3++)+d**(pimages4++); // first value
518 val2 = wrap_around; // first value
519 for(int i=0;i<channels-1;++pimages1,++pimages2,++pimages3,++pimages4,++i)
520 {
521 val3 = a**(pimages1)+b**(pimages2)+c**(pimages3)+d**(pimages4);
522 if(val1 <= val2)
523 {
524 if(val3 < val2)
525 {
526 angle = float((calcSubPos(val1,val2,val3)+i)*resolution);
527 if(angle < 0)
528 angle += float(CV_PI);
529 else if(angle > CV_PI)
530 angle -= float(CV_PI);
531 angles_i.push_back(angle);
532 pt_iter->angle = 360.0F-angle*RAD2DEG;
533 }
534 }
535 else if(val1 > val2 && val3 >= val2)
536 {
537 angle = float((calcSubPos(val1,val2,val3)+i)*resolution);
538 if(angle < 0)
539 angle += float(CV_PI);
540 else if(angle > CV_PI)
541 angle -= float(CV_PI);
542 angles_i.push_back(-angle);
543 pt_iter->angle = 360.0F-angle*RAD2DEG;
544 }
545 val1 = val2;
546 val2 = val3;
547 }
548 // wrap around
549 if(val1 <= val2)
550 {
551 if(wrap_around< val2)
552 {
553 angle = float((calcSubPos(val1,val2,wrap_around)+channels-1)*resolution);
554 if(angle < 0)
555 angle += float(CV_PI);
556 else if(angle > CV_PI)
557 angle -= float(CV_PI);
558 angles_i.push_back(angle);
559 pt_iter->angle = 360.0F-angle*RAD2DEG;
560 }
561 }
562 else if(val1 > val2 && wrap_around >= val2)
563 {
564 angle = float((calcSubPos(val1,val2,wrap_around)+channels-1)*resolution);
565 if(angle < 0)
566 angle += float(CV_PI);
567 else if(angle > CV_PI)
568 angle -= float(CV_PI);
569 angles_i.push_back(-angle);
570 pt_iter->angle = 360.0F-angle*RAD2DEG;
571 }
572 }
573 });
574 return angles;
575}

577void FastX::findKeyPoints(const std::vector<cv::Mat> &feature_maps, std::vector<KeyPoint>& keypoints,const Mat& _mask) const
578{
579 //TODO check that all feature_maps have the same size
580 int num_scales = parameters.max_scale-parameters.min_scale;
581 CV_CheckGE(int(feature_maps.size()), num_scales, "missing feature maps");
582 if (!_mask.empty())
583 {
584 CV_CheckTypeEQ(_mask.type(), CV_8UC1, "wrong mask type");
585 CV_CheckEQ(_mask.size(), feature_maps.front().size(),"wrong mask type or size");
586 }
587 keypoints.clear();
588
589 cv::Mat mask;
590 if(!_mask.empty())
591 mask = _mask;
592 else
593 mask = cv::Mat::ones(feature_maps.front().size(),CV_8UC1);
594
595 int super_res = int(parameters.super_resolution);
596 int super_scale = super_res+1;
597 float super_comp = 0.25F*super_res;
598
599 // for each scale
600 float strength = parameters.strength;
601 std::vector<int> windows;
602 cv::Point pt,pt2;
603 double min,max;
604 cv::Mat src;
605 for(int scale=parameters.max_scale;scale>=parameters.min_scale;--scale)
606 {
607 int window_size = (1 << (scale + super_res)) + 1;
608 float window_size2 = 0.5F*window_size;
609 float window_size4 = 0.25F*window_size;
610 int window_size2i = cvRound(window_size2);
611
612 const cv::Mat &feature_map = feature_maps[scale-parameters.min_scale];
613 int y = ((feature_map.rows)/window_size)-2;
614 int x = ((feature_map.cols)/window_size)-2;
615 for(int row=1;row<y;++row)
616 {
617 for(int col=1;col<x;++col)
618 {
619 Rect rect(col*window_size,row*window_size,window_size,window_size);
620 src = feature_map(rect);
621 cv::minMaxLoc(src,&min,&max,NULL,&pt);
622 if(min == max || max < strength)
623 continue;
624
625 cv::Point pos(pt.x+rect.x,pt.y+rect.y);
626 if(mask.at<unsigned char>(pos.y,pos.x) == 0)
627 continue;
628
629 Rect rect2(int(pos.x-window_size2),int(pos.y-window_size2),window_size,window_size);
630 src = feature_map(rect2);
631 cv::minMaxLoc(src,NULL,NULL,NULL,&pt2);
632 if(pos.x == pt2.x+rect2.x && pos.y == pt2.y+rect2.y)
633 {
634 // the point is the best one on the current scale
635 // check all larger scales if there is a stronger one
636 double max2;
637 int scale2= scale-1;
638 //parameters.min_scale;
639 for(;scale2>=parameters.min_scale;--scale2)
640 {
641 cv::minMaxLoc(feature_maps[scale2-parameters.min_scale](rect),NULL,&max2,NULL,NULL);
642 if(max2 > max)
643 break;
644 }
645 if(scale2<parameters.min_scale && pos.x+1 < feature_map.cols && pos.y+1 < feature_map.rows)
646 {
647 float sub_x = float(calcSubpixel(feature_map.at<float>(pos.y,pos.x-1),
648 feature_map.at<float>(pos.y,pos.x),
649 feature_map.at<float>(pos.y,pos.x+1)));
650 float sub_y = float(calcSubpixel(feature_map.at<float>(pos.y-1,pos.x),
651 feature_map.at<float>(pos.y,pos.x),
652 feature_map.at<float>(pos.y+1,pos.x)));
653 cv::KeyPoint kpt(sub_x+pos.x,sub_y+pos.y,float(window_size),0.F,float(max),scale);
654 int x2 = std::max(0,int(kpt.pt.x-window_size4));
655 int y2 = std::max(0,int(kpt.pt.y-window_size4));
656 int w = std::min(int(mask.cols-x2),window_size2i);
657 int h = std::min(int(mask.rows-y2),window_size2i);
658 mask(cv::Rect(x2,y2,w,h)) = 0.0;
659 if(super_scale != 1)
660 {
661 kpt.pt.x /= super_scale;
662 kpt.pt.y /= super_scale;
663 kpt.pt.x -= super_comp;
664 kpt.pt.y -= super_comp;
665 kpt.size /= super_scale;
666 }
667 keypoints.push_back(kpt);
668 }
669 }
670 }
671 }
672 }
673}

697void FastX::detectImpl(const cv::Mat& _gray_image,
698 std::vector<cv::Mat> &rotated_images,
699 std::vector<cv::Mat> &feature_maps,
700 const cv::Mat &_mask)const
701{
702 if(!_mask.empty())
703 CV_Error(Error::StsBadSize, "Mask is not supported");
704 CV_CheckTypeEQ(_gray_image.type(), CV_8UC1, "Unsupported image type");
705
706 // up-sample if needed
707 cv::Mat gray_image;
708 const int super_res = int(parameters.super_resolution);
709 if(super_res)
710 cv::resize(_gray_image,gray_image,cv::Size(),2,2);
711 else
712 _gray_image.copyTo(gray_image);
713
714 //for each scale
715 const int num_scales = parameters.max_scale-parameters.min_scale+1;
716 const int diag = int(sqrt(gray_image.rows*gray_image.rows+gray_image.cols*gray_image.cols));
717 const cv::Size size(diag,diag);
718 const int num = int(0.5001*CV_PI/parameters.resolution);
719
720 rotated_images.resize(num_scales);
721 feature_maps.resize(num_scales);
722
723 parallel_for_(Range(parameters.min_scale,parameters.max_scale+1),[&](const Range& range){
724 for(int scale=range.start;scale < range.end;++scale)
725 {
726 // calc images
727 // for each angle step
728 int scale_id = scale-parameters.min_scale;
729 int scale_size = int(std::pow(2,scale+1+super_res));
730 int scale_size2 = int((scale_size/7)*2+1);
731 std::vector<cv::Mat> images;
732 images.resize(2*num);
733 cv::Mat rotated,filtered_h,filtered_v;
734 cv::boxFilter(gray_image,images[0],-1,cv::Size(scale_size,scale_size2));
735 cv::boxFilter(gray_image,images[num],-1,cv::Size(scale_size2,scale_size));
736 for(int i=1;i<num;++i)
737 {
738 float angle = parameters.resolution*i;
739 rotate(-angle,gray_image,size,rotated);
740 cv::boxFilter(rotated,filtered_h,-1,cv::Size(scale_size,scale_size2));
741 cv::boxFilter(rotated,filtered_v,-1,cv::Size(scale_size2,scale_size));
742
743 // rotate filtered images back
744 rotate(angle,filtered_h,gray_image.size(),images[i]);
745 rotate(angle,filtered_v,gray_image.size(),images[i+num]);
746 }
747 cv::merge(images,rotated_images[scale_id]);
748
749 // calc feature map
750 calcFeatureMap(rotated_images[scale_id],feature_maps[scale_id]);
751
752 // filter feature map to improve impulse responses
753 if(parameters.filter)
754 {
755 cv::Mat high,low;
756 cv::boxFilter(feature_maps[scale_id],low,-1,cv::Size(scale_size,scale_size));
757 int scale2 = int((scale_size/6))*2+1;
758 cv::boxFilter(feature_maps[scale_id],high,-1,cv::Size(scale2,scale2));
759 feature_maps[scale_id] = high-0.8*low;
760 }
761 }
762 });
763}

1285float Chessboard::Board::getAngle()const
1286{
1287 if(isEmpty())
1288 CV_Error(Error::StsBadArg,"Board is empty");
1289 if(colCount() < 3)
1290 CV_Error(Error::StsBadArg,"Board is too small");
1291

1293 cv::Point3f pt(delta.x,delta.y,0);
1294 float val;
1295 if(fabs(pt.x) > fabs(pt.y))
1296 {
1297 cv::Point3f ptx(1,0,0);
1298 val = float(ptx.dot(pt)/cv::norm(pt));
1299 if(val < 0)
1300 val = -std::acos(val);
1301 else
1302 val = std::acos(val);
1303 }
1304 else
1305 {
1306 cv::Point3f ptx(0,1,0);

1616{

1618 Cell *pcell = NULL;

1632 rotateLeft();

1670void Chessboard::Board::normalizeOrientation(bool bblack)
1671{
1672 // fix ordering
1673 cv::Point2f y = getCorner(0,1)-getCorner(2,1);
1674 cv::Point2f x = getCorner(1,2)-getCorner(1,0);
1675 cv::Point3f y3d(y.x,y.y,0);
1676 cv::Point3f x3d(x.x,x.y,0);
1677 if(x3d.cross(y3d).z > 0)
1678 flipHorizontal();
1679
1680 //normalize orientation so that first element is black or white
1681 const Cell* cell = getCell(0,0);
1682 if(cell->black != bblack && colCount()%2 != 0)
1683 rotateLeft();
1684 else if(cell->black != bblack && rowCount()%2 != 0)
1685 {
1686 rotateLeft();
1687 rotateLeft();
1688 }
1689
1690 //find closest point to top left image corner
1691 //in case of symmetric checkerboard
1692 if(colCount() == rowCount())
1693 {
1694 PointIter iter_top_right(top_left,TOP_RIGHT);
1695 while(iter_top_right.right());
1696 PointIter iter_bottom_right(iter_top_right);
1697 while(iter_bottom_right.bottom());
1698 PointIter iter_bottom_left(top_left,BOTTOM_LEFT);

1702 iter_bottom_left.getCell()->empty() || iter_bottom_right.getCell()->empty())

1706 float d2 = (*iter_top_right)->dot(*(*iter_top_right));
1707 float d3 = (*iter_bottom_left)->dot(*(*iter_bottom_left));
1708 float d4 = (*iter_bottom_right)->dot(*(*iter_bottom_right));
1709 if(d2 <= d1 && d2 <= d3 && d2 <= d4) // top left is top right
1710 rotateLeft();
1711 else if(d3 <= d1 && d3 <= d2 && d3 <= d4) // top left is bottom left
1712 rotateRight();
1713 else if(d4 <= d1 && d4 <= d2 && d4 <= d3) // top left is bottom right

1717 }
1718 }

1821// if NaN is returned for a point no point at all was found
1822// if 0 is returned the point lies outside of the ellipse
1823float Chessboard::Board::findMaxPoint(cv::flann::Index &index,const cv::Mat &data,const Ellipse &ellipse,float white_angle,float black_angle,cv::Point2f &point)
1824{
1825 // flann data type enriched with angles (third column)
1826 CV_CheckType(data.type(), CV_32FC1, "type of flann data is not supported");
1827 CV_CheckEQ(data.cols, 4, "4-cols flann data is expected");
1828
1829 std::vector<float> query,dists;
1830 std::vector<int> indices;
1831 query.resize(2);
1832 point = ellipse.getCenter();
1833 query[0] = point.x;
1834 query[1] = point.y;
1835 index.knnSearch(query,indices,dists,4,cv::flann::SearchParams(64));
1836 std::vector<int>::const_iterator iter = indices.begin();
1837 float best_score = -std::numeric_limits<float>::max();
1838 point.x = std::numeric_limits<float>::quiet_NaN();
1839 point.y = std::numeric_limits<float>::quiet_NaN();
1840 for(;iter != indices.end();++iter)
1841 {
1842 const float *val = data.ptr<float>(*iter);
1843 const float &response = *(val+3);
1844 if(response < best_score)
1845 continue;
1846 const float &a0 = *(val+2);
1847 float a1 = std::fabs(a0-white_angle);
1848 float a2 = std::fabs(a0-black_angle);
1849 if(a1 > CV_PI*0.5)
1850 a1 = std::fabs(float(a1-CV_PI));
1851 if(a2> CV_PI*0.5)
1852 a2 = std::fabs(float(a2-CV_PI));
1853 if(a1 < MAX_ANGLE || a2 < MAX_ANGLE )
1854 {
1855 cv::Point2f pt(val[0], val[1]);
1856 if(point.x != point.x) // NaN check
1857 point = pt;
1858 if(best_score < response && ellipse.contains(pt))
1859 {
1860 best_score = response;
1861 point = pt;
1862 }
1863 }
1864 }
1865 if(best_score == -std::numeric_limits<float>::max())
1866 return 0;
1867 else

1869}

1964 return false;

1966 cv::Point2f p12 = p2-p1;

1968 float b = float(cv::norm(p12));
1969 float t = (0.75F*a-0.25F*b);
1970 if(t <= 0)
1971 return false;
1972 float c = 0.25F*b*(a+b)/t;
1973 if(c < 0.1F)
1974 return false;
1975 p01 = p01/a;
1976 p12 = p12/b;
1977 // check angle between p01 and p12 < 25°
1978 if(p01.dot(p12) < 0.9)
1979 return false;

1981 // p12 = (p01+p12)*0.5;
1982 // p3 = p2+p12*c;
1983 p3 = p2+p12*c;
1984
1985 // compensate radial distortion by fitting polynom
1986 std::vector<double> x,y;
1987 x.resize(3,0); y.resize(3,0);
1988 x[1] = b;
1989 x[2] = b+a;
1990 y[2] = calcSignedDistance(-p12,p2,p0);

2002bool Chessboard::Board::estimatePoint(const cv::Point2f &p0,const cv::Point2f &p1,const cv::Point2f &p2, const cv::Point2f &p3, cv::Point2f &p4)

2009 cv::Point2f p02 = p2-p0;

2012 float b = float(cv::norm(p02));
2013 float c = float(cv::norm(p03));
2014 cv::Mat dst = (cv::Mat_<double>(1,4) << 0,a,b,c);
2015 cv::Mat h = findHomography1D(src,dst);

2023
2024 // compensate radial distortion by fitting polynom

2026 x.resize(4,0); y.resize(4,0);
2027 x[1] = c-b;
2028 x[2] = c-a;
2029 x[3] = c;
2030 y[2] = calcSignedDistance(-p23,p3,p1);
2031 y[3] = calcSignedDistance(-p23,p3,p0);
2032 polyfit(cv::Mat(x),cv::Mat(y),dst,2);
2033 d = d-c;
