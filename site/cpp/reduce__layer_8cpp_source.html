19 // set reduce type
20 CV_Assert(params.has("reduce"));
21 String op_type = toLowerCase(params.get<String>("reduce"));
22 if (op_type == "max")
23 reduce_type = ReduceType::MAX;
24 else if (op_type == "min")
25 reduce_type = ReduceType::MIN;
26 else if (op_type == "mean")
27 reduce_type = ReduceType::MEAN;
28 else if (op_type == "sum")
29 reduce_type = ReduceType::SUM;
30 else if (op_type == "sum_square")
31 reduce_type = ReduceType::SUM_SQUARE;
32 else if (op_type == "l1")

39 reduce_type = ReduceType::LOG_SUM_EXP;
40 else if (op_type == "prod")

69 virtual void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr) CV_OVERRIDE {
70 if (axes.empty()) {
71 return;
72 }
73
74 std::vector<Mat> inputs, outputs;
75 inputs_arr.getMatVector(inputs);
76 outputs_arr.getMatVector(outputs);
77
78 auto shape_input = shape(inputs[0]);
79 for (auto i = 0; i < axes.size(); ++i) {
80 auto norm_axis = normalize_axis(axes[i], shape_input);
81 axes[i] = norm_axis;
82 }
83 if (shape_input.empty())
84 return;
85
86 bool do_nothing = true;
87 for (auto axis : axes) {
88 if (shape_input[axis] != 1 || keepdims) {
89 do_nothing = false;
90 }
91 }
92 if (do_nothing) {
93 axes.clear();
94 noop_with_empty_axes = true;
95 }

98 bool getMemoryShapes(const std::vector<MatShape> &inputs,
99 const int requiredOutputs,
100 std::vector<MatShape> &outputs,
101 std::vector<MatShape> &internals) const CV_OVERRIDE
102 {
103 if (inputs[0].empty()){
104 CV_CheckEQ(axes[0], 0, "Axis must be 0 when input is empty.");
105 outputs.assign(1, MatShape());
106 return false;
107 }
108 // empty axes
109 if (axes.empty()) {
110 if (noop_with_empty_axes) {
111 // do nothing
112 outputs.assign(1, inputs[0]);
113 } else {
114 // reduce all axes
115 MatShape shape_output;
116 if (keepdims) {
117 shape_output = inputs[0];
118 for (auto i = 0; i < shape_output.size(); ++i)
119 shape_output[i] = 1;
120 } else {
121 shape_output = MatShape::scalar();
122 }
123 outputs.assign(1, shape_output);
124 }
125 } else {
126 auto shape_output_ = inputs[0];
127 for (size_t i = 0; i < axes.size(); ++i) {
128 auto norm_axis = normalize_axis(axes[i], inputs[0]);
129 shape_output_[norm_axis] = -1;
130 }
131 MatShape shape_output;
132 for (size_t i = 0; i < shape_output_.size(); ++i) {
133 if (shape_output_[i] == -1) {
134 if (keepdims)
135 shape_output.push_back(1);
136 else
137 continue;
138 } else
139 shape_output.push_back(shape_output_[i]);
140 }
141 if (shape_output.empty())
142 shape_output = MatShape::scalar();
143
144 outputs.assign(1, shape_output);
145 }
146
147 return false;
148 }

277 public:
278 using dtype = typename Op::dtype_input;
279
280 const Mat& src;
281 Mat& dst;
282
283 int n_reduce;
284 int loop_size;
285
286 int total;
287 int cost_per_thread;
288
289 ReduceAllInvoker(const Mat& src_, Mat& dst_) : src(src_), dst(dst_) {
290 auto shape_src = shape(src);
291
292 n_reduce = std::accumulate(shape_src.begin(), shape_src.end(), 1, std::multiplies<int>());
293 loop_size = n_reduce;
294
295 total = 1;
296 cost_per_thread = 1;
297 }

299 void operator()(const Range& r) const CV_OVERRIDE {
300 int start = r.start;
301 int end = r.end;
302
303 const dtype* p_src = src.ptr<const dtype>();
304 dtype* p_dst = dst.ptr<dtype>();
305
306 for (int i = start; i < end; ++i) {
307 Op accumulator(n_reduce, *p_src);
308 for (int l = 0; l < loop_size; ++l) {
309 accumulator.update(p_src[l]);
310 }
311 p_dst[i] = accumulator.get_value();
312 }

316 template <typename Op>
317 class ReduceInvoker : public ParallelLoopBody {
318 public:
319 using dtype = typename Op::dtype_input;
320
321 const Mat& src;
322 Mat& dst;
323
324 std::vector<int> reduced_axes; // assume in ascending order
325
326 int n_reduce;
327 int loop_size;
328
329 int last_reduced_dim;
330 int last_reduced_step;
331 std::vector<int> projected_steps;
332
333 int last_unreduced_dim;
334 int last_unreduced_step;
335 std::vector<int> unprojected_steps;
336
337 int total;
338 int cost_per_thread;
339
340 ReduceInvoker(const Mat& src_, Mat& dst_, std::vector<int> axes_) : src(src_), dst(dst_), reduced_axes(axes_) {
341 auto shape_src = shape(src);
342
343 auto steps_src = shape_src;
344 steps_src[steps_src.size() - 1] = 1;
345 for (int i = static_cast<int>(steps_src.size()) - 2; i >= 0; --i)
346 steps_src[i] = steps_src[i + 1] * shape_src[i + 1];
347
348 size_t projection_size = 1;
349 for (auto axis : reduced_axes) {
350 projection_size *= shape_src[axis];
351 }
352 n_reduce = projection_size;
353
354 last_reduced_dim = shape_src[reduced_axes.back()];
355 last_reduced_step = steps_src[reduced_axes.back()];
356 loop_size = last_reduced_dim * last_reduced_step;
357 projection_size /= last_reduced_dim;
358
359 // calculate projected_steps
360 int last_reduced_axis = static_cast<int>(reduced_axes.size()) - 1;
361 if (last_reduced_axis == 0) {
362 projected_steps.resize(1, 0);
363 } else {
364 projected_steps.resize(projection_size);
365 std::vector<int> projected_indices(last_reduced_axis, 0);
366 for (size_t i = 0, current_step = 0; i < projection_size; ++i) {
367 projected_steps[i] = current_step;
368 ++projected_indices[last_reduced_axis - 1];
369 current_step += steps_src[reduced_axes[last_reduced_axis - 1]];
370 for (int j = last_reduced_axis - 1; j > 0; --j) {
371 if (projected_indices[j] < shape_src[reduced_axes[j]]) {
372 break;
373 }
374 projected_indices[j] = 0;
375 ++projected_indices[j - 1];
376 current_step = steps_src[reduced_axes[j - 1]];
377 }
378 }
379 }
380
381 // calculate unprojected_steps
382 std::vector<int> unreduced_axes;
383 for (int i = 0; i < static_cast<int>(shape_src.size()); ++i) {
384 if (std::find(reduced_axes.begin(), reduced_axes.end(), i) == reduced_axes.end()) {
385 unreduced_axes.push_back(i);
386 }
387 }
388 size_t unprojection_size = 1;
389 for (auto axis : unreduced_axes) {
390 unprojection_size *= shape_src[axis];
391 }
392 last_unreduced_dim = shape_src[unreduced_axes.back()];
393 last_unreduced_step = steps_src[unreduced_axes.back()];
394 unprojection_size /= last_unreduced_dim;
395
396 std::vector<int> unprojected_indices(unreduced_axes.size(), 0);
397 unprojected_steps.reserve(unprojection_size);
398 if (unprojected_indices.size() <= 1) {
399 unprojected_steps.push_back(0);
400 } else {
401 for (size_t i = 0, current_step = 0; i < unprojection_size; ++i) {
402 unprojected_steps.push_back(current_step);
403 ++unprojected_indices[unprojected_indices.size() - 2];
404 current_step += steps_src[unreduced_axes[unreduced_axes.size() - 2]];
405 for (int j = static_cast<int>(unreduced_axes.size()) - 2; j > 0; --j) {
406 if (unprojected_indices[j] < shape_src[unreduced_axes[j]]) {
407 break;
408 }
409 unprojected_indices[j] -= shape_src[unreduced_axes[j]];
410 current_step -= shape_src[unreduced_axes[j]] * steps_src[unreduced_axes[j]];
411 ++unprojected_indices[j - 1];
412 current_step += steps_src[unreduced_axes[j - 1]];
413 }
414 }
415 }
416
417 auto shape_dst = shape(dst);
418 total = std::accumulate(shape_dst.begin(), shape_dst.end(), 1, std::multiplies<int>());
419 cost_per_thread = static_cast<int>(projected_steps.size() * last_reduced_step);
420 }

422 static void run(const Mat& src, Mat& dst, std::vector<int> axes, bool noop_with_empty_axes) {
423 CV_Assert(src.isContinuous());
424 CV_Assert(dst.isContinuous());
425 if (shape(src).empty() || (shape(src).size() == 1)){
426 // since there is only one element no need for parallel compute
427 // axis does not matter either (one element)
428 ReduceAllInvoker<Op> p(src, dst);
429 p(Range(0, p.total));
430 return;
431 }
432
433 if (axes.empty()) {
434 if (noop_with_empty_axes) {
435 // copyTo is not used here for the reason that we want a
436 // copy for the case when dims at all axes are 1
437 const auto p_src = src.ptr<const dtype>();
438 auto p_dst = dst.ptr<dtype>();
439 std::memcpy(p_dst, p_src, sizeof(dtype) * dst.total());
440 return;
441 }
442
443 ReduceAllInvoker<Op> p(src, dst);
444 double nstripes = (size_t)p.total * (size_t)p.cost_per_thread * (1 / 1024.0);
445 parallel_for_(Range(0, p.total), p, nstripes);
446 return;
447 }
448
449 ReduceInvoker<Op> p(src, dst, axes);
450 double nstripes = (size_t)p.total * (size_t)p.cost_per_thread * (1 / 1024.0);
451 parallel_for_(Range(0, p.total), p, nstripes);
452 }

454 void operator()(const Range& r) const CV_OVERRIDE {
455 int start = r.start;
456 int end = r.end;
457
458 const dtype* p_src = src.ptr<const dtype>();
459 dtype* p_dst = dst.ptr<dtype>();
460
461 size_t main_index = start / last_unreduced_dim;
462 size_t loop = start % last_unreduced_dim;
463 size_t origin = unprojected_steps[main_index] + loop * last_unreduced_step;
464 for (int i = start; i < end; ++i) {
465 Op accumulator(n_reduce, p_src[origin + projected_steps[0]]);
466 for (auto projected_step : projected_steps) {
467 const dtype* loop_p_src = p_src + origin + projected_step;
468 for (auto l = 0; l < loop_size; l += last_reduced_step) {
469 accumulator.update(loop_p_src[l]);
470 }
471 }
472 p_dst[i] = accumulator.get_value();
473
474 ++loop;
475 if (loop >= last_unreduced_dim) {
476 loop = 0;
477 ++main_index;
478 if (main_index < unprojected_steps.size()) {
479 origin = unprojected_steps[main_index];
480 }
481 } else {
482 origin += last_unreduced_step;
483 }
484 }
485 }

511 case ReduceType::MEAN: ReduceInvoker<ReduceMean<T>>::run(std::forward<Args>(args)...); break;
512 case ReduceType::SUM: ReduceInvoker<ReduceSum<T>>::run(std::forward<Args>(args)...); break;
513 case ReduceType::L1: ReduceInvoker<ReduceL1<T>>::run(std::forward<Args>(args)...); break;
514 case ReduceType::L2: ReduceInvoker<ReduceL2<T>>::run(std::forward<Args>(args)...); break;
515 case ReduceType::PROD: ReduceInvoker<ReduceProd<T>>::run(std::forward<Args>(args)...); break;
516 case ReduceType::SUM_SQUARE: ReduceInvoker<ReduceSumSquare<T>>::run(std::forward<Args>(args)...); break;
517 case ReduceType::LOG_SUM: ReduceInvoker<ReduceLogSum<T>>::run(std::forward<Args>(args)...); break;
518 case ReduceType::LOG_SUM_EXP: ReduceInvoker<ReduceLogSumExp<T>>::run(std::forward<Args>(args)...); break;
