38 def wrap_exceptions_as_warnings(self, original_func=None,
39 ret_type_on_failure=None):
40 def parametrized_wrapper(func):
41 @functools.wraps(func)
42 def wrapped_func(*args, **kwargs):
43 if self.has_failure:
44 if ret_type_on_failure is None:
45 return None
46 return ret_type_on_failure()
47
48 try:
49 ret_type = func(*args, **kwargs)
50 except Exception:
51 self.has_failure = True
52 warnings.warn(
53 "Typing stubs generation has failed.\n{}".format(
54 traceback.format_exc()
55 )
56 )
57 if ret_type_on_failure is None:
58 return None
59 return ret_type_on_failure()
60 return ret_type
61
62 if self.exceptions_as_warnings:
63 return wrapped_func
64 else:
65 return original_func
66
67 if original_func:
68 return parametrized_wrapper(original_func)
69 return parametrized_wrapper

71 @contextmanager
72 def delete_on_failure(self, file_path):
73 # type: (Path) -> Generator[None, None, None]
74 # There is no errors during stubs generation and file doesn't exist
75 if not self.has_failure and not file_path.is_file():
76 file_path.parent.mkdir(parents=True, exist_ok=True)
77 file_path.touch()
78 try:
79 # continue execution
80 yield
81 finally:
82 # If failure is occurred - delete file if exists
83 if self.has_failure and file_path.is_file():
