93bool clipLine( Size2l img_size, Point2l& pt1, Point2l& pt2 )
94{
95 CV_INSTRUMENT_REGION();
96
97 int c1, c2;
98 int64 right = img_size.width-1, bottom = img_size.height-1;
99
100 if( img_size.width <= 0 || img_size.height <= 0 )
101 return false;
102
103 int64 &x1 = pt1.x, &y1 = pt1.y, &x2 = pt2.x, &y2 = pt2.y;
104 c1 = (x1 < 0) + (x1 > right) * 2 + (y1 < 0) * 4 + (y1 > bottom) * 8;
105 c2 = (x2 < 0) + (x2 > right) * 2 + (y2 < 0) * 4 + (y2 > bottom) * 8;
106
107 if( (c1 & c2) == 0 && (c1 | c2) != 0 )
108 {
109 int64 a;
110 if( c1 & 12 )
111 {
112 a = c1 < 8 ? 0 : bottom;
113 x1 += (int64)((double)(a - y1) * (x2 - x1) / (y2 - y1));
114 y1 = a;
115 c1 = (x1 < 0) + (x1 > right) * 2;
116 }
117 if( c2 & 12 )
118 {
119 a = c2 < 8 ? 0 : bottom;
120 x2 += (int64)((double)(a - y2) * (x2 - x1) / (y2 - y1));
121 y2 = a;
122 c2 = (x2 < 0) + (x2 > right) * 2;
123 }
124 if( (c1 & c2) == 0 && (c1 | c2) != 0 )
125 {
126 if( c1 )
127 {
128 a = c1 == 1 ? 0 : right;
129 y1 += (int64)((double)(a - x1) * (y2 - y1) / (x2 - x1));
130 x1 = a;
131 c1 = 0;
132 }
133 if( c2 )
134 {
135 a = c2 == 1 ? 0 : right;
136 y2 += (int64)((double)(a - x2) * (y2 - y1) / (x2 - x1));
137 x2 = a;
138 c2 = 0;
139 }
140 }
141
142 CV_Assert( (c1 & c2) != 0 || (x1 | y1 | x2 | y2) >= 0 );
143 }
144
145 return (c1 | c2) == 0;
146}

160void LineIterator::init( const Mat* img, Rect rect, Point pt1_, Point pt2_, int connectivity, bool leftToRight )
161{
162 CV_Assert( connectivity == 8 || connectivity == 4 );
163
164 count = -1;
165 p = Point(0, 0);
166 ptr0 = ptr = 0;
167 step = elemSize = 0;
168 ptmode = !img;
169
170 Point pt1 = pt1_ - rect.tl();
171 Point pt2 = pt2_ - rect.tl();
172
173 if( (unsigned)pt1.x >= (unsigned)(rect.width) ||
174 (unsigned)pt2.x >= (unsigned)(rect.width) ||
175 (unsigned)pt1.y >= (unsigned)(rect.height) ||
176 (unsigned)pt2.y >= (unsigned)(rect.height) )
177 {
178 if( !clipLine(Size(rect.width, rect.height), pt1, pt2) )
179 {
180 err = plusDelta = minusDelta = plusStep = minusStep = plusShift = minusShift = count = 0;
181 return;
182 }
183 }
184
185 pt1 += rect.tl();
186 pt2 += rect.tl();
187
188 int delta_x = 1, delta_y = 1;
189 int dx = pt2.x - pt1.x;
190 int dy = pt2.y - pt1.y;
191
192 if( dx < 0 )
193 {
194 if( leftToRight )
195 {
196 dx = -dx;
197 dy = -dy;
198 pt1 = pt2;
199 }
200 else
201 {
202 dx = -dx;
203 delta_x = -1;
204 }
205 }
206
207 if( dy < 0 )
208 {
209 dy = -dy;
210 delta_y = -1;
211 }
212
213 bool vert = dy > dx;
214 if( vert )
215 {
216 std::swap(dx, dy);
217 std::swap(delta_x, delta_y);
218 }
219
220 CV_Assert( dx >= 0 && dy >= 0 );
221
222 if( connectivity == 8 )
223 {
224 err = dx - (dy + dy);
225 plusDelta = dx + dx;
226 minusDelta = -(dy + dy);
227 minusShift = delta_x;
228 plusShift = 0;
229 minusStep = 0;
230 plusStep = delta_y;
231 count = dx + 1;
232 }
233 else /* connectivity == 4 */
234 {
235 err = 0;
236 plusDelta = (dx + dx) + (dy + dy);
237 minusDelta = -(dy + dy);
238 minusShift = delta_x;
239 plusShift = -delta_x;
240 minusStep = 0;
241 plusStep = delta_y;
242 count = dx + dy + 1;
243 }
244
245 if( vert )
246 {
247 std::swap(plusStep, plusShift);
248 std::swap(minusStep, minusShift);
249 }
250
251 p = pt1;
252 if( !ptmode )
253 {
254 ptr0 = img->ptr();
255 step = (int)img->step;
256 elemSize = (int)img->elemSize();
257 ptr = (uchar*)ptr0 + (size_t)p.y*step + (size_t)p.x*elemSize;
258 plusStep = plusStep*step + plusShift*elemSize;
259 minusStep = minusStep*step + minusShift*elemSize;
260 }
261}

319 int ecount, scount = 0;
320 int slope;
321 int64 ax, ay;
322 int64 x_step, y_step;
323 int64 i, j;
324 int ep_table[9];
325 int cb = ((uchar*)color)[0], cg = ((uchar*)color)[1], cr = ((uchar*)color)[2], ca = ((uchar*)color)[3];
326 int _cb, _cg, _cr, _ca;
327 int nch = img.channels();
328 uchar* ptr = img.ptr();
329 size_t step = img.step;
330 Size2l size0(img.size()), size = size0;
331
332 if( !((nch == 1 || nch == 3 || nch == 4) && img.depth() == CV_8U) )
333 {
334 Line(img, Point((int)(pt1.x>>XY_SHIFT), (int)(pt1.y>>XY_SHIFT)), Point((int)(pt2.x>>XY_SHIFT), (int)(pt2.y>>XY_SHIFT)), color);
335 return;
336 }
337
338 size.width <<= XY_SHIFT;
339 size.height <<= XY_SHIFT;
340 if( !clipLine( size, pt1, pt2 ))
341 return;
342
343 dx = pt2.x - pt1.x;
344 dy = pt2.y - pt1.y;
345
346 j = dx < 0 ? -1 : 0;
347 ax = (dx ^ j) - j;
348 i = dy < 0 ? -1 : 0;
349 ay = (dy ^ i) - i;
350
351 if( ax > ay )
352 {
353 dy = (dy ^ j) - j;
354 pt1.x ^= pt2.x & j;
355 pt2.x ^= pt1.x & j;
356 pt1.x ^= pt2.x & j;
357 pt1.y ^= pt2.y & j;
358 pt2.y ^= pt1.y & j;
359 pt1.y ^= pt2.y & j;
360
361 x_step = XY_ONE;
362 y_step = (int64)((uint64_t)dy << XY_SHIFT) / (ax | 1);
363 pt2.x += XY_ONE;
364 ecount = (int)((pt2.x >> XY_SHIFT) - (pt1.x >> XY_SHIFT));
365 j = -(pt1.x & (XY_ONE - 1));
366 pt1.y += ((y_step * j) >> XY_SHIFT) + (XY_ONE >> 1);
367 slope = (y_step >> (XY_SHIFT - 5)) & 0x3f;
368 slope ^= (y_step < 0 ? 0x3f : 0);
369
370 /* Get 4-bit fractions for end-point adjustments */
371 i = (pt1.x >> (XY_SHIFT - 7)) & 0x78;
372 j = (pt2.x >> (XY_SHIFT - 7)) & 0x78;
373 }
374 else
375 {
376 dx = (dx ^ i) - i;
377 pt1.x ^= pt2.x & i;
378 pt2.x ^= pt1.x & i;
379 pt1.x ^= pt2.x & i;
380 pt1.y ^= pt2.y & i;
381 pt2.y ^= pt1.y & i;
382 pt1.y ^= pt2.y & i;
383
384 x_step = (int64)((uint64_t)dx << XY_SHIFT) / (ay | 1);
385 y_step = XY_ONE;
386 pt2.y += XY_ONE;
387 ecount = (int)((pt2.y >> XY_SHIFT) - (pt1.y >> XY_SHIFT));
388 j = -(pt1.y & (XY_ONE - 1));
389 pt1.x += ((x_step * j) >> XY_SHIFT) + (XY_ONE >> 1);
390 slope = (x_step >> (XY_SHIFT - 5)) & 0x3f;
391 slope ^= (x_step < 0 ? 0x3f : 0);
392
393 /* Get 4-bit fractions for end-point adjustments */
394 i = (pt1.y >> (XY_SHIFT - 7)) & 0x78;
395 j = (pt2.y >> (XY_SHIFT - 7)) & 0x78;
396 }
397
398 slope = (slope & 0x20) ? 0x100 : SlopeCorrTable[slope];

400 /* Calc end point correction table */
401 {
402 int t0 = slope << 7;
403 int t1 = ((0x78 - (int)i) | 4) * slope;
404 int t2 = ((int)j | 4) * slope;
405
406 ep_table[0] = 0;

973 arc_end += 360;

976 {
977 arc_end -= 360;
978 arc_start -= 360;

1114 int delta1, delta2;
1115
1116 if( line_type < cv::LINE_AA )
1117 delta1 = delta2 = XY_ONE >> 1;
1118 else
1119 delta1 = XY_ONE - 1, delta2 = 0;
1120
1121 p0 = v[npts - 1];
1122 p0.x <<= XY_SHIFT - shift;
1123 p0.y <<= XY_SHIFT - shift;
1124
1125 CV_Assert( 0 <= shift && shift <= XY_SHIFT );
1126 xmin = xmax = v[0].x;
1127 ymin = ymax = v[0].y;
1128
1129 for( i = 0; i < npts; i++ )
1130 {
1131 Point2l p = v[i];
1132 if( p.y < ymin )
1133 {
1134 ymin = p.y;
1135 imin = i;
1136 }
1137
1138 ymax = std::max( ymax, p.y );
1139 xmax = std::max( xmax, p.x );
1140 xmin = MIN( xmin, p.x );
1141
1142 p.x <<= XY_SHIFT - shift;
1143 p.y <<= XY_SHIFT - shift;
1144
1145 if( line_type <= 8 )
1146 {
1147 if( shift == 0 )
1148 {
1149 Point pt0, pt1;
1150 pt0.x = (int)(p0.x >> XY_SHIFT);
1151 pt0.y = (int)(p0.y >> XY_SHIFT);
1152 pt1.x = (int)(p.x >> XY_SHIFT);
1153 pt1.y = (int)(p.y >> XY_SHIFT);
1154 Line( img, pt0, pt1, color, line_type );
1155 }
1156 else
1157 Line2( img, p0, p, color );
1158 }
1159 else
1160 LineAA( img, p0, p, color );
1161 p0 = p;
1162 }
1163
1164 xmin = (xmin + delta) >> shift;
1165 xmax = (xmax + delta) >> shift;
1166 ymin = (ymin + delta) >> shift;
1167 ymax = (ymax + delta) >> shift;
1168
1169 if( npts < 3 || (int)xmax < 0 || (int)ymax < 0 || (int)xmin >= size.width || (int)ymin >= size.height )
1170 return;
1171
1172 ymax = MIN( ymax, size.height - 1 );
1173 edge[0].idx = edge[1].idx = imin;
1174
1175 edge[0].ye = edge[1].ye = y = (int)ymin;
1176 edge[0].di = 1;
1177 edge[1].di = npts - 1;
1178
1179 edge[0].x = edge[1].x = -XY_ONE;
1180 edge[0].dx = edge[1].dx = 0;
1181
1182 ptr += (int64_t)img.step*y;
1183
1184 do
1185 {
1186 if( line_type < cv::LINE_AA || y < (int)ymax || y == (int)ymin )
1187 {
1188 for( i = 0; i < 2; i++ )
1189 {
1190 if( y >= edge[i].ye )
1191 {
1192 int idx0 = edge[i].idx, di = edge[i].di;
1193 int idx = idx0 + di;
1194 if (idx >= npts) idx -= npts;

1196
1197 for (; edges-- > 0; )
1198 {
1199 ty = (int)((v[idx].y + delta) >> shift);
1200 if (ty > y)
1201 {
1202 int64 xs = v[idx0].x;
1203 int64 xe = v[idx].x;
1204 if (shift != XY_SHIFT)
1205 {
1206 xs <<= XY_SHIFT - shift;
1207 xe <<= XY_SHIFT - shift;
1208 }

1210 edge[i].ye = ty;
1211 edge[i].dx = ((xe - xs)*2 + ((int64_t)ty - y)) / (2 * ((int64_t)ty - y));
1212 edge[i].x = xs;
1213 edge[i].idx = idx;
1214 break;
1215 }
1216 idx0 = idx;
1217 idx += di;
1218 if (idx >= npts) idx -= npts;
1219 }
1220 }
1221 }
1222 }
1223
1224 if (edges < 0)
1225 break;
1226
1227 if (y >= 0)
1228 {
1229 int left = 0, right = 1;
1230 if (edge[0].x > edge[1].x)
1231 {
1232 left = 1, right = 0;
1233 }
1234
1235 int xx1 = (int)((edge[left].x + delta1) >> XY_SHIFT);
1236 int xx2 = (int)((edge[right].x + delta2) >> XY_SHIFT);
1237
1238 if( xx2 >= 0 && xx1 < size.width )
1239 {
1240 if( xx1 < 0 )
1241 xx1 = 0;
1242 if( xx2 >= size.width )
1243 xx2 = size.width - 1;
1244 ICV_HLINE( ptr, xx1, xx2, color, pix_size );
1245 }
1246 }
1247 else
1248 {
1249 // TODO optimize scan for negative y
1250 }
1251
1252 edge[0].x += edge[0].dx;
1253 edge[1].x += edge[1].dx;
1254 ptr += img.step;
1255 }
1256 while( ++y <= (int)ymax );
1257}

1261
1262static void
1263CollectPolyEdges( Mat& img, const Point2l* v, int count, std::vector<PolyEdge>& edges,
1264 const void* color, int line_type, int shift, Point offset )
1265{
1266 int i, delta = offset.y + ((1 << shift) >> 1);
1267 Point2l pt0 = v[count-1], pt1;
1268 pt0.x = (pt0.x + offset.x) << (XY_SHIFT - shift);
1269 pt0.y = (pt0.y + delta) >> shift;
1270
1271 edges.reserve( edges.size() + count );
1272
1273 for( i = 0; i < count; i++, pt0 = pt1 )
1274 {
1275 Point2l t0, t1;
1276 PolyEdge edge;
1277
1278 pt1 = v[i];
1279 pt1.x = (pt1.x + offset.x) << (XY_SHIFT - shift);
1280 pt1.y = (pt1.y + delta) >> shift;
1281
1282 Point2l pt0c(pt0), pt1c(pt1);
1283
1284 if (line_type < cv::LINE_AA)
1285 {
1286 t0.y = pt0.y; t1.y = pt1.y;
1287 t0.x = (pt0.x + (XY_ONE >> 1)) >> XY_SHIFT;
1288 t1.x = (pt1.x + (XY_ONE >> 1)) >> XY_SHIFT;
1289 Line(img, t0, t1, color, line_type);
1290
1291 // use clipped endpoints to create a more accurate PolyEdge
1292 if ((unsigned)t0.x >= (unsigned)(img.cols) ||
1293 (unsigned)t1.x >= (unsigned)(img.cols) ||
1294 (unsigned)t0.y >= (unsigned)(img.rows) ||
1295 (unsigned)t1.y >= (unsigned)(img.rows))
1296 {
1297 clipLine(img.size(), t0, t1);
1298
1299 if (t0.y != t1.y)
1300 {
1301 pt0c.y = t0.y; pt1c.y = t1.y;
1302 }
1303 }
1304
1305 pt0c.x = (int64)(t0.x) << XY_SHIFT;
1306 pt1c.x = (int64)(t1.x) << XY_SHIFT;
1307 }
1308 else
1309 {
1310 t0.x = pt0.x; t1.x = pt1.x;
1311 t0.y = pt0.y << XY_SHIFT;
1312 t1.y = pt1.y << XY_SHIFT;
1313 LineAA(img, t0, t1, color);
1314 }
1315
1316 if (pt0.y == pt1.y)
1317 continue;
1318
1319 edge.dx = (pt1c.x - pt0c.x) / (pt1c.y - pt0c.y);
1320 if (pt0.y < pt1.y)
1321 {
1322 edge.y0 = (int)(pt0.y);
1323 edge.y1 = (int)(pt1.y);
1324 edge.x = pt0c.x + (pt0.y - pt0c.y) * edge.dx; // correct starting point for clipped lines
1325 }
1326 else
1327 {
1328 edge.y0 = (int)(pt1.y);
1329 edge.y1 = (int)(pt0.y);
1330 edge.x = pt1c.x + (pt1.y - pt1c.y) * edge.dx; // correct starting point for clipped lines
1331 }
1332 edges.push_back(edge);
1333 }
1334}

1336struct CmpEdges
1337{
1338 bool operator ()(const PolyEdge& e1, const PolyEdge& e2)
1339 {
1340 return e1.y0 - e2.y0 ? e1.y0 < e2.y0 :
1341 e1.x - e2.x ? e1.x < e2.x : e1.dx < e2.dx;
1342 }
1343};

1345/**************** helper macros and functions for sequence/contour processing ***********/
1346
1347static void
1348FillEdgeCollection( Mat& img, std::vector<PolyEdge>& edges, const void* color )
1349{
1350 PolyEdge tmp;
1351 int i, y, total = (int)edges.size();
1352 Size size = img.size();
1353 PolyEdge* e;
1354 int y_max = INT_MIN, y_min = INT_MAX;
1355 int64 x_max = 0xFFFFFFFFFFFFFFFF, x_min = 0x7FFFFFFFFFFFFFFF;
1356 int pix_size = (int)img.elemSize();
1357 int delta = XY_ONE - 1;
1358
1359 if( total < 2 )
1360 return;
1361
1362 for( i = 0; i < total; i++ )
1363 {
1364 PolyEdge& e1 = edges[i];
1365 CV_Assert( e1.y0 < e1.y1 );
1366 // Determine x-coordinate of the end of the edge.
1367 // (This is not necessary x-coordinate of any vertex in the array.)
1368 int64 x1 = e1.x + (e1.y1 - e1.y0) * e1.dx;
1369 y_min = std::min( y_min, e1.y0 );
1370 y_max = std::max( y_max, e1.y1 );
1371 x_min = std::min( x_min, e1.x );
1372 x_max = std::max( x_max, e1.x );
1373 x_min = std::min( x_min, x1 );
1374 x_max = std::max( x_max, x1 );
1375 }
1376
1377 if( y_max < 0 || y_min >= size.height || x_max < 0 || x_min >= ((int64)size.width<<XY_SHIFT) )
1378 return;
1379
1380 std::sort( edges.begin(), edges.end(), CmpEdges() );
1381
1382 // start drawing
1383 tmp.y0 = INT_MAX;
1384 edges.push_back(tmp); // after this point we do not add
1385 // any elements to edges, thus we can use pointers
1386 i = 0;
1387 tmp.next = 0;
1388 e = &edges[i];
1389 y_max = MIN( y_max, size.height );
1390
1391 for( y = e->y0; y < y_max; y++ )
1392 {
1393 PolyEdge *last, *prelast, *keep_prelast;
1394 int draw = 0;
1395 int clipline = y < 0;
1396
1397 prelast = &tmp;
1398 last = tmp.next;
1399 while( last || e->y0 == y )
1400 {
1401 if( last && last->y1 == y )
1402 {
1403 // exclude edge if y reaches its lower point
1404 prelast->next = last->next;
1405 last = last->next;
1406 continue;
1407 }
1408 keep_prelast = prelast;
1409 if( last && (e->y0 > y || last->x < e->x) )
1410 {
1411 // go to the next edge in active list
1412 prelast = last;
1413 last = last->next;
1414 }
1415 else if( i < total )
1416 {
1417 // insert new edge into active list if y reaches its upper point
1418 prelast->next = e;
1419 e->next = last;
1420 prelast = e;
1421 e = &edges[++i];
1422 }
1423 else
1424 break;
1425
1426 if( draw )
1427 {
1428 if( !clipline )
1429 {
1430 // convert x's from fixed-point to image coordinates
1431 uchar *timg = img.ptr(y);
1432 int x1, x2;
1433
1434 if (keep_prelast->x > prelast->x)
1435 {
1436 x1 = (int)((prelast->x + delta) >> XY_SHIFT);
1437 x2 = (int)(keep_prelast->x >> XY_SHIFT);
1438 }
1439 else
1440 {
1441 x1 = (int)((keep_prelast->x + delta) >> XY_SHIFT);
1442 x2 = (int)(prelast->x >> XY_SHIFT);
1443 }
1444
1445 // clip and draw the line
1446 if( x1 < size.width && x2 >= 0 )
1447 {
1448 if( x1 < 0 )
1449 x1 = 0;
1450 if( x2 >= size.width )
1451 x2 = size.width - 1;
1452 ICV_HLINE( timg, x1, x2, color, pix_size );
1453 }
1454 }
1455 keep_prelast->x += keep_prelast->dx;
1456 prelast->x += prelast->dx;
1457 }
1458 draw ^= 1;
1459 }
1460
1461 // sort edges (using bubble sort)
1462 keep_prelast = 0;
1463
1464 do
1465 {
1466 prelast = &tmp;
1467 last = tmp.next;
1468 PolyEdge *last_exchange = 0;
1469
1470 while( last != keep_prelast && last->next != 0 )
1471 {
1472 PolyEdge *te = last->next;
1473
1474 // swap edges
1475 if( last->x > te->x )
1476 {
1477 prelast->next = te;
1478 last->next = te->next;
1479 te->next = last;
1480 prelast = te;
1481 last_exchange = prelast;
1482 }
1483 else
1484 {
1485 prelast = last;
1486 last = te;
1487 }
1488 }
1489 if (last_exchange == NULL)
1490 break;
1491 keep_prelast = last_exchange;
1492 } while( keep_prelast != tmp.next && keep_prelast != &tmp );
1493 }
1494}

1645static void
1646ThickLine( Mat& img, Point2l p0, Point2l p1, const void* color,
1647 int thickness, int line_type, int flags, int shift )
1648{
1649 static const double INV_XY_ONE = 1./static_cast<double>(XY_ONE);
1650
1651 Rect_<int64> boundingRect(Point2l(0, 0), (Size2l)img.size());
1652 if( (thickness > 1) && (shift == 0) && ( !boundingRect.contains(p0) || !boundingRect.contains(p1) ) )
1653 {
1654 const int margin = thickness;
1655 const Point2l offset(margin, margin);
1656 p0 += offset;
1657 p1 += offset;
1658 clipLine(Size2l(boundingRect.width+2*margin, boundingRect.height+2*margin), p0, p1);
1659 p0 -= offset;
1660 p1 -= offset;
1661 }
1662
1663 p0.x <<= XY_SHIFT - shift;
1664 p0.y <<= XY_SHIFT - shift;
1665 p1.x <<= XY_SHIFT - shift;
1666 p1.y <<= XY_SHIFT - shift;
1667
1668 if( thickness <= 1 )
1669 {
1670 if( line_type < cv::LINE_AA )
1671 {
1672 if( line_type == 1 || line_type == 8 || shift == 0 )
1673 {
1674 p0.x = (p0.x + (XY_ONE>>1)) >> XY_SHIFT;
1675 p0.y = (p0.y + (XY_ONE>>1)) >> XY_SHIFT;
1676 p1.x = (p1.x + (XY_ONE>>1)) >> XY_SHIFT;
1677 p1.y = (p1.y + (XY_ONE>>1)) >> XY_SHIFT;
1678 Line( img, p0, p1, color, line_type );
1679 }
1680 else
1681 Line2( img, p0, p1, color );
1682 }
1683 else
1684 LineAA( img, p0, p1, color );
1685 }
1686 else
1687 {
1688 Point2l pt[4], dp = Point2l(0,0);
1689 double dx = (p0.x - p1.x)*INV_XY_ONE, dy = (p1.y - p0.y)*INV_XY_ONE;
1690 double r = dx * dx + dy * dy;
1691 int i, oddThickness = thickness & 1;
1692 thickness <<= XY_SHIFT - 1;
1693
1694 if( fabs(r) > DBL_EPSILON )
1695 {
1696 r = (thickness + oddThickness*XY_ONE*0.5)/std::sqrt(r);
1697 dp.x = cvRound( dy * r );
1698 dp.y = cvRound( dx * r );
1699
1700 pt[0].x = p0.x + dp.x;
1701 pt[0].y = p0.y + dp.y;
1702 pt[1].x = p0.x - dp.x;
1703 pt[1].y = p0.y - dp.y;
1704 pt[2].x = p1.x - dp.x;
1705 pt[2].y = p1.y - dp.y;
1706 pt[3].x = p1.x + dp.x;
1707 pt[3].y = p1.y + dp.y;
1708
1709 FillConvexPoly( img, pt, 4, color, line_type, XY_SHIFT );
1710 }
1711
1712 for( i = 0; i < 2; i++ )
1713 {
1714 if( flags & (i+1) )
1715 {
1716 if( line_type < cv::LINE_AA )
1717 {
1718 Point center;
1719 center.x = (int)((p0.x + (XY_ONE>>1)) >> XY_SHIFT);
1720 center.y = (int)((p0.y + (XY_ONE>>1)) >> XY_SHIFT);
1721 Circle( img, center, (thickness + (XY_ONE>>1)) >> XY_SHIFT, color, 1 );
1722 }
1723 else
1724 {
1725 EllipseEx( img, p0, Size2l(thickness, thickness),
1726 0, 0, 360, color, -1, line_type );
1727 }
1728 }
1729 p0 = p1;
1730 }
1731 }
1732}
