50 void buildTargetShape(const MatShape& inputShape, const Mat& shapeTensor, const Mat& axesTensor,
51 std::vector<int>& targetShape, std::vector<int>& usedAxes) const
52 {
53 const int rank = inputShape.dims;
54 targetShape.assign(inputShape.begin(), inputShape.end());
55 usedAxes.clear();
56
57 CV_Assert(shapeTensor.dims == 1);
58 CV_Assert(shapeTensor.type() == CV_32S || shapeTensor.type() == CV_64S);
59
60 std::vector<int> axes;
61 if (!axesTensor.empty())
62 {
63 CV_Assert(axesTensor.dims == 1);
64 int naxes = (int)axesTensor.total();
65 axes.resize(naxes);
66 if (axesTensor.type() == CV_32S)
67 {
68 const int32_t* a32 = (const int32_t*)axesTensor.data;
69 for (int i = 0; i < naxes; ++i) axes[i] = normalize_axis((int)a32[i], rank);
70 }
71 else
72 {
73 const int64_t* a64 = (const int64_t*)axesTensor.data;
74 for (int i = 0; i < naxes; ++i) axes[i] = normalize_axis((int)a64[i], rank);
75 }
76 }
77 else if (!axes_attr.empty())
78 {
79 axes.resize((int)axes_attr.size());
80 for (size_t i = 0; i < axes_attr.size(); ++i) axes[i] = normalize_axis(axes_attr[i], rank);
81 }
82 else
83 {
84 axes.resize(rank);
85 std::iota(axes.begin(), axes.end(), 0);
86 }
87
88 CV_Assert((int)shapeTensor.total() == (int)axes.size());
89
90 if (shapeTensor.type() == CV_32S)
91 {
92 const int32_t* shp = (const int32_t*)shapeTensor.data;
93 applyShapeFromPtr(shp, (int)axes.size(), axes, targetShape, usedAxes);
94 }
95 else if (shapeTensor.type() == CV_64S)
96 {
97 const int64_t* shp = (const int64_t*)shapeTensor.data;
98 applyShapeFromPtr(shp, (int)axes.size(), axes, targetShape, usedAxes);
99 }
100 }

138 void forward(InputArrayOfArrays inputs_arr,
139 OutputArrayOfArrays outputs_arr,
140 OutputArrayOfArrays) CV_OVERRIDE
141 {
142 CV_TRACE_FUNCTION();
143 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
144
145 Size size = inputs_arr.size();
146 int ninputs = size.area();
147 CV_Assert(ninputs >= 2);
148
149 Mat X = inputs_arr.getMat(0);
150 Mat shapeTensor = inputs_arr.getMat(1);
151 Mat axesTensor;
152 if (ninputs >= 3)
153 axesTensor = inputs_arr.getMat(2);
154
155 const int rank = X.dims;
156 std::vector<int> outShape(rank);
157 std::vector<int> usedAxes;
158
159 MatShape inShape = X.shape();
160 std::vector<int> tgt;
161 buildTargetShape(inShape, shapeTensor, axesTensor, tgt, usedAxes);
162 outShape.assign(tgt.begin(), tgt.end());
163
164 auto kind = outputs_arr.kind();
165 if (kind == _InputArray::STD_VECTOR_MAT)
166 {
167 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
168 outs.resize(1);
169 outs[0].fit(MatShape(outShape.begin(), outShape.end()), X.type());
170 centerCropPad(X, outs[0], usedAxes);
171 }
172 else if (kind == _InputArray::STD_VECTOR_UMAT)
173 {
174 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
175 outs.resize(1);
176 outs[0].fit(MatShape(outShape.begin(), outShape.end()), X.type());
177 Mat temp(MatShape(outShape.begin(), outShape.end()), X.type());
178 centerCropPad(X, temp, usedAxes);
179 temp.copyTo(outs[0]);
180 }
181 else
182 {
183 CV_Error(Error::StsNotImplemented, "");
184 }
185 }

188 template<typename T>
189 static void applyShapeFromPtr(const T* shp, int count, const std::vector<int>& axes,
190 std::vector<int>& targetShape, std::vector<int>& usedAxes)
191 {
192 for (int i = 0; i < count; ++i)
193 {
194 int a = axes[i];
195 int desired = (int)shp[i];
196 CV_Assert(desired >= 0);
197 targetShape[a] = desired;
198 usedAxes.push_back(a);
199 }
200 }

202 void centerCropPad(const Mat& src, Mat& dst, const std::vector<int>& axes) const
203 {
204 CV_Assert(src.dims == dst.dims);
205 const int rank = src.dims;
206
207 std::vector<Range> srcRanges(rank), dstRanges(rank);
208 for (int i = 0; i < rank; ++i)
209 {
210 int s = src.size[i];
211 int d = dst.size[i];
212 if (std::find(axes.begin(), axes.end(), i) == axes.end())
213 {
214 CV_Assert(s == d);
215 srcRanges[i] = Range(0, s);
216 dstRanges[i] = Range(0, d);
217 continue;
218 }
219
220 if (d <= s)
221 {
222 int diff = s - d;
223 int start = diff / 2;
224 int end = start + d;
225 srcRanges[i] = Range(start, end);
226 dstRanges[i] = Range(0, d);
227 }
228 else
229 {
230 int diff = d - s;
231 int start = diff / 2;
232 int end = start + s;
233 srcRanges[i] = Range(0, s);
234 dstRanges[i] = Range(start, end);
235 }
236 }
237
238 dst.setTo(Scalar(0));
239 src(srcRanges.data()).copyTo(dst(dstRanges.data()));
240 }
