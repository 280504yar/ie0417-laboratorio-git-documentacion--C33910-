12// Our three input types have a different value for a depth pixel with no depth
13template<typename DepthDepth>
14inline DepthDepth noDepthSentinelValue()
15{
16 return 0;
17}

21noDepthSentinelValue<float>()

35// Testing for depth pixels with no depth isn't straightforward for NaN values. We
36// need to specialize the equality check for floats and doubles.
37template<typename DepthDepth>

61// When using the unsigned short representation, we'd like to round the values to the nearest
62// integer value. The float/double representations don't need to be rounded
63template<typename DepthDepth>
64inline DepthDepth
65floatToInputDepth(const float& value)
66{
67 return (DepthDepth)value;
68}
69
70template<>
71inline unsigned short
72floatToInputDepth<unsigned short>(const float& value)
73{
74 return (unsigned short)(value + 0.5);

92template<typename DepthDepth>
93void performRegistration(const Mat_<DepthDepth>& unregisteredDepth,
94 const Matx33f& unregisteredCameraMatrix,
95 const Matx33f& registeredCameraMatrix,
96 const Mat_<float>& registeredDistCoeffs,
97 const Matx44f& rbtRgb2Depth,
98 const Size outputImagePlaneSize,
99 const bool depthDilation,
100 const float inputDepthToMetersScale,
101 Mat& registeredDepth)
102{
103 // Create output Mat of the correct type, filled with an initial value indicating no depth
104 registeredDepth = Mat_<DepthDepth>(outputImagePlaneSize, noDepthSentinelValue<DepthDepth>());
105
106 // Figure out whether we'll have to apply a distortion
107 bool hasDistortion = (countNonZero(registeredDistCoeffs) > 0);
108
109 // A point (i,j,1) will have to be converted to 3d first, by multiplying it by K.inv()
110 // It will then be transformed by rbtRgb2Depth.
111 // Finally, it will be projected into the external camera via registeredCameraMatrix and
112 // its distortion coefficients. If there is no distortion in the external camera, we
113 // can linearly chain all three operations together.
114
115 Matx44f K = Matx44f::zeros();
116 for (unsigned char j = 0; j < 3; ++j)
117 for (unsigned char i = 0; i < 3; ++i)
118 {
119 K(j, i) = unregisteredCameraMatrix(j, i);
120 }
121 K(3, 3) = 1;
122
123 Matx44f initialProjection;
124 if (hasDistortion)
125 {
126 // The projection into the external camera will be done separately with distortion
127 initialProjection = rbtRgb2Depth * K.inv();
128 }
129 else
130 {
131 // No distortion, so all operations can be chained
132 initialProjection = Matx44f::zeros();
133 for (unsigned char j = 0; j < 3; ++j)
134 for (unsigned char i = 0; i < 3; ++i)
135 initialProjection(j, i) = registeredCameraMatrix(j, i);
136 initialProjection(3, 3) = 1;
137
138 initialProjection = initialProjection * rbtRgb2Depth * K.inv();
139 }
140
141 // Apply the initial projection to the input depth
142 Mat_<Point3f> transformedCloud;
143 {
144 Mat_<Point3f> point_tmp(outputImagePlaneSize, Point3f(0., 0., 0.));
145 for (int j = 0; j < unregisteredDepth.rows; ++j)
146 {
147 const DepthDepth* unregisteredDepthPtr = unregisteredDepth[j];
148
149 Point3f* point = point_tmp[j];
150 for (int i = 0; i < unregisteredDepth.cols; ++i, ++unregisteredDepthPtr, ++point)
151 {
152 float rescaled_depth = float(*unregisteredDepthPtr) * inputDepthToMetersScale;
153
154 // If the DepthDepth is of type unsigned short, zero is a sentinel value to indicate
155 // no depth. CV_32F and CV_64F should already have NaN for no depth values.
156 if (rescaled_depth == 0)
157 {
158 rescaled_depth = std::numeric_limits<float>::quiet_NaN();
159 }
160
161 point->x = i * rescaled_depth;
162 point->y = j * rescaled_depth;
163 point->z = rescaled_depth;
164 }
165 }
166
167 perspectiveTransform(point_tmp, transformedCloud, initialProjection);
168 }
169
170 std::vector<Point2f> transformedAndProjectedPoints(transformedCloud.cols);
171 const float metersToInputUnitsScale = 1 / inputDepthToMetersScale;
172 const Rect registeredDepthBounds(Point(), outputImagePlaneSize);
173
174 for (int y = 0; y < transformedCloud.rows; y++)
175 {
176 if (hasDistortion)
177 {
178
179 // Project an entire row of points with distortion.
180 // Doing this for the entire image at once would require more memory.
181 projectPoints(transformedCloud.row(y),
182 Vec3f(0, 0, 0),
183 Vec3f(0, 0, 0),
184 registeredCameraMatrix,
185 registeredDistCoeffs,
186 transformedAndProjectedPoints);
187 }
188 else
189 {
190 // With no distortion, we just have to dehomogenize the point since all major transforms
191 // already happened with initialProjection.
192 Point2f* point2d = &transformedAndProjectedPoints[0];
193 const Point2f* point2d_end = point2d + transformedAndProjectedPoints.size();
194 const Point3f* point3d = transformedCloud[y];
195 for (; point2d < point2d_end; ++point2d, ++point3d)
196 {
197 point2d->x = point3d->x / point3d->z;
198 point2d->y = point3d->y / point3d->z;
199 }
200 }
201
202 const Point2f* outputProjectedPoint = &transformedAndProjectedPoints[0];
203 const Point3f* p = transformedCloud[y], * p_end = p + transformedCloud.cols;
204
205 for (; p < p_end; ++outputProjectedPoint, ++p)
206 {
207 // Skip this one if there isn't a valid depth
208 const Point2f projectedPixelFloatLocation = *outputProjectedPoint;
209 if (cvIsNaN(projectedPixelFloatLocation.x))
210 continue;
211
212 //Get integer pixel location
213 const Point2i projectedPixelLocation = projectedPixelFloatLocation;
214
215 // Ensure that the projected point is actually contained in our output image
216 if (!registeredDepthBounds.contains(projectedPixelLocation))
217 continue;
218
219 // Go back to our original scale, since that's what our output will be
220 // The templated function is to ensure that integer values are rounded to the nearest integer
221 const DepthDepth cloudDepth = floatToInputDepth<DepthDepth>(p->z * metersToInputUnitsScale);
222
223 DepthDepth& outputDepth = registeredDepth.at<DepthDepth>(projectedPixelLocation.y, projectedPixelLocation.x);
224
225 // Occlusion check
226 if (isEqualToNoDepthSentinelValue<DepthDepth>(outputDepth) || (outputDepth > cloudDepth))
227 outputDepth = cloudDepth;
228
229 // If desired, dilate this point to avoid holes in the final image
230 if (depthDilation)
231 {
232 // Choosing to dilate in a 2x2 region, where the original projected location is in the bottom right of this
233 // region. This is what's done on PrimeSense devices, but a more accurate scheme could be used.
234 const Point2i dilatedProjectedLocations[3] = { Point2i(projectedPixelLocation.x - 1, projectedPixelLocation.y),
235 Point2i(projectedPixelLocation.x , projectedPixelLocation.y - 1),
236 Point2i(projectedPixelLocation.x - 1, projectedPixelLocation.y - 1) };
237
238 for (int i = 0; i < 3; i++)
239 {
240 const Point2i& dilatedCoordinates = dilatedProjectedLocations[i];
241
242 if (!registeredDepthBounds.contains(dilatedCoordinates))
243 continue;
244
245 DepthDepth& outputDilatedDepth = registeredDepth.at<DepthDepth>(dilatedCoordinates.y, dilatedCoordinates.x);
246
247 // Occlusion check
248 if (isEqualToNoDepthSentinelValue(outputDilatedDepth) || (outputDilatedDepth > cloudDepth))
249 outputDilatedDepth = cloudDepth;
250 }
251
252 } // depthDilation
253
254 } // iterate cols
255 } // iterate rows
256}

260void
261registerDepth(InputArray unregisteredCameraMatrix, InputArray registeredCameraMatrix, InputArray registeredDistCoeffs,
262 InputArray Rt, InputArray unregisteredDepth, const Size& outputImagePlaneSize,
263 OutputArray registeredDepth, bool depthDilation)
264{
265 CV_Assert(unregisteredCameraMatrix.depth() == CV_64F || unregisteredCameraMatrix.depth() == CV_32F);
266 CV_Assert(registeredCameraMatrix.depth() == CV_64F || registeredCameraMatrix.depth() == CV_32F);
267 CV_Assert(registeredDistCoeffs.empty() || registeredDistCoeffs.depth() == CV_64F || registeredDistCoeffs.depth() == CV_32F);
268 CV_Assert(Rt.depth() == CV_64F || Rt.depth() == CV_32F);
269
270 CV_Assert(unregisteredDepth.cols() > 0 && unregisteredDepth.rows() > 0 &&
271 (unregisteredDepth.depth() == CV_32F || unregisteredDepth.depth() == CV_64F || unregisteredDepth.depth() == CV_16U));
272 CV_Assert(outputImagePlaneSize.height > 0 && outputImagePlaneSize.width > 0);
273
274 // Implicitly checking dimensions of the InputArrays
275 Matx33f _unregisteredCameraMatrix = unregisteredCameraMatrix.getMat();
276 Matx33f _registeredCameraMatrix = registeredCameraMatrix.getMat();
277 Mat_<float> _registeredDistCoeffs = registeredDistCoeffs.getMat();
278 Matx44f _rbtRgb2Depth = Rt.getMat();
279
280 Mat& registeredDepthMat = registeredDepth.getMatRef();
281
282 switch (unregisteredDepth.depth())
283 {
284 case CV_16U:
285 {
286 performRegistration<unsigned short>(unregisteredDepth.getMat(), _unregisteredCameraMatrix,
287 _registeredCameraMatrix, _registeredDistCoeffs,
288 _rbtRgb2Depth, outputImagePlaneSize, depthDilation,
289 .001f, registeredDepthMat);
290 break;
291 }
292 case CV_32F:
293 {
294 performRegistration<float>(unregisteredDepth.getMat(), _unregisteredCameraMatrix,
295 _registeredCameraMatrix, _registeredDistCoeffs,
296 _rbtRgb2Depth, outputImagePlaneSize, depthDilation,
297 1.0f, registeredDepthMat);
298 break;
299 }
300 case CV_64F:
301 {
302 performRegistration<double>(unregisteredDepth.getMat(), _unregisteredCameraMatrix,
303 _registeredCameraMatrix, _registeredDistCoeffs,
304 _rbtRgb2Depth, outputImagePlaneSize, depthDilation,
305 1.0f, registeredDepthMat);
306 break;
307 }
308 default:
309 {
310 CV_Error(Error::StsUnsupportedFormat, "Input depth must be unsigned short, float, or double.");
311 }
312 }
313}
