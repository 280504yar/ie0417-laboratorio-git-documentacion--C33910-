15static bool ocl_patchNaNs( InputOutputArray _a, double value )
16{
17 int ftype = _a.depth();
18
19 const ocl::Device d = ocl::Device::getDefault();
20 bool doubleSupport = d.doubleFPConfig() > 0;
21 if (!doubleSupport && ftype == CV_64F)
22 {
23 return false;
24 }
25
26 int rowsPerWI = ocl::Device::getDefault().isIntel() ? 4 : 1;
27 ocl::Kernel k("KF", ocl::core::arithm_oclsrc,
28 format("-D UNARY_OP -D OP_PATCH_NANS -D dstT=%s -D DEPTH_dst=%d -D rowsPerWI=%d %s",
29 ftype == CV_64F ? "double" : "float", ftype, rowsPerWI,
30 doubleSupport ? "-D DOUBLE_SUPPORT" : ""));
31 if (k.empty())
32 return false;
33
34 UMat a = _a.getUMat();
35 int cn = a.channels();
36
37 // to pass float or double to args
38 if (ftype == CV_32F)
39 {
40 k.args(ocl::KernelArg::ReadOnlyNoSize(a), ocl::KernelArg::WriteOnly(a, cn), (float)value);
41 }
42 else // CV_64F
43 {
44 k.args(ocl::KernelArg::ReadOnlyNoSize(a), ocl::KernelArg::WriteOnly(a, cn), value);
45 }
46
47 size_t globalsize[2] = { (size_t)a.cols * cn, ((size_t)a.rows + rowsPerWI - 1) / rowsPerWI };
48 return k.run(2, globalsize, NULL, false);
49}

59void patchNaNs( InputOutputArray _a, double _val )
60{
61 CV_INSTRUMENT_REGION();
62 CV_Assert( _a.depth() == CV_32F || _a.depth() == CV_64F);
63
64 CV_OCL_RUN(_a.isUMat() && _a.dims() <= 2,
65 ocl_patchNaNs(_a, _val))
66
67 Mat a = _a.getMat();
68 const Mat* arrays[] = {&a, 0};
69 uchar* ptrs[1] = {};
70 NAryMatIterator it(arrays, ptrs);
71 size_t len = it.size*a.channels();
72
73 PatchNanFunc func = getPatchNanFunc(_a.depth() == CV_64F);
74
75 for (size_t i = 0; i < it.nplanes; i++, ++it)
76 {
77 func(ptrs[0], len, _val);
78 }
79}

84static bool ocl_finiteMask(const UMat img, UMat mask)
85{
86 int channels = img.channels();
87 int depth = img.depth();
88
89 const ocl::Device d = ocl::Device::getDefault();

93 return false;

95

97 ocl::Kernel k("finiteMask", ocl::core::finitemask_oclsrc,
98 format("-D srcT=%s -D cn=%d -D rowsPerWI=%d %s",

100 doubleSupport ? "-D DOUBLE_SUPPORT" : ""));

104 k.args(ocl::KernelArg::ReadOnlyNoSize(img), ocl::KernelArg::WriteOnly(mask));

118void finiteMask(InputArray _src, OutputArray _mask)
119{
120 CV_INSTRUMENT_REGION();
121
122 int channels = _src.channels();
123 int depth = _src.depth();
124 CV_Assert( channels > 0 && channels <= 4);
125 CV_Assert( depth == CV_32F || depth == CV_64F );
126 std::vector<int> vsz(_src.dims());
127 _src.sizend(vsz.data());
128 _mask.create(_src.dims(), vsz.data(), CV_8UC1);
129
130 CV_OCL_RUN(_src.isUMat() && _mask.isUMat() && _src.dims() <= 2,
131 ocl_finiteMask(_src.getUMat(), _mask.getUMat()));
132
133 Mat src = _src.getMat();
134 Mat mask = _mask.getMat();
135
136 const Mat *arrays[]={&src, &mask, 0};
137 Mat planes[2];
138 NAryMatIterator it(arrays, planes);
139 size_t total = planes[0].total();
140 size_t i, nplanes = it.nplanes;
141
142 FiniteMaskFunc func = getFiniteMaskFunc((depth == CV_64F), channels);
143
144 for( i = 0; i < nplanes; i++, ++it )
145 {
146 const uchar* sptr = planes[0].ptr();
147 uchar* dptr = planes[1].ptr();
148
149 func(sptr, dptr, total);
150 }
151}
