33 getLayer(std::vector<Ptr<LayerInfo> >& newprog, int op_idx) const
34 {
35 return op_idx >= 0 ? dynamic_cast<_LayerType*>(newprog.at(op_idx).get()) : 0;

38 bool fuseGraph(Ptr<Graph>& graph)
39 {
40 vector<Arg> removed_args;
41 bool modified = false;
42 const std::vector<Ptr<LayerInfo> >& prog = graph->prog();
43 size_t i, nargs = netimpl->args.size(), nops = prog.size();
44 std::vector<int> producer_of(nargs, -1);
45 std::vector<Ptr<LayerInfo> > newprog;
46 std::vector<Arg> fused_inputs;
47
48 for (i = 0; i < nops; i++) {
49 const Ptr<LayerInfo>& layer = prog[i];
50 Layer* layer_ptr = (Layer*)layer.get();
51 int fused_layer_idx = -1;
52 std::vector<Ptr<Graph> >* subgraphs = layer->subgraphs();
53 if (subgraphs) {
54 for (Ptr<Graph>& g: *subgraphs) {
55 if (fuseGraph(g))
56 modified = true;
57 }
58 }
59 const std::vector<Arg>& inputs = layer->inputs;
60 const std::vector<Arg>& outputs = layer->outputs;
61 size_t ninputs = inputs.size();
62 removed_args.clear();
63 fused_inputs.clear(); // leave it empty in the merge patterns below to re-use original fused node inputs as-is.
64
65 for(;;) {
66 BatchNorm2Layer* bn = dynamic_cast<BatchNorm2Layer*>(layer_ptr);
67 ActivationLayer* activ = dynamic_cast<ActivationLayer*>(layer_ptr);
68 NaryEltwiseLayer* elemwise = dynamic_cast<NaryEltwiseLayer*>(layer_ptr);
69
70 // merge convolution and batch norm
71 if (bn && ninputs == 1 &&
72 usecounts.at(inputs[0].idx) == 1) {
73 Arg bn_inp = inputs[0];
74 int conv_layer_idx = producer_of.at(bn_inp.idx);
75 Conv2Layer* conv = getLayer<Conv2Layer>(newprog, conv_layer_idx);
76 if (conv) {
77 bool ok = conv->fuseBatchNorm(layer.dynamicCast<Layer>());
78 if (ok) {
79 fused_layer_idx = conv_layer_idx;
80 removed_args.push_back(bn_inp);
81 break;
82 }
83 }
84 }
85
86 // merge residual 'add' into 'conv' node
87 if (elemwise && (elemwise->op == NaryEltwiseLayer::OPERATION::ADD ||
88 elemwise->op == NaryEltwiseLayer::OPERATION::SUM) &&
89 ninputs == 2) {
90
91 int op0 = producer_of.at(inputs[0].idx);
92 int op1 = producer_of.at(inputs[1].idx);
93
94 if (op0 >= 0 && op1 >= 0) {
95 int conv_layer_idx;
96 Arg residual, conv_out;
97
98 if (op0 > op1) { // choose the latter op to ensure that the other component is already computed
99 conv_layer_idx = op0;
100 conv_out = inputs[0];
101 residual = inputs[1];
102 } else {
103 conv_layer_idx = op1;
104 conv_out = inputs[1];
105 residual = inputs[0];
106 }
107
108 Conv2Layer* conv = getLayer<Conv2Layer>(newprog, conv_layer_idx);
109 if (conv && usecounts[conv_out.idx] == 1 &&
110 conv->fuseAddResidual(residual)) {
111 fused_layer_idx = conv_layer_idx;
112 removed_args.push_back(conv_out);
113 break;
114 }
115 }
116 }
117
118 // merge convolution and activation
119 if (activ && ninputs == 1 &&
120 usecounts.at(inputs[0].idx) == 1) {
121 Arg activ_inp = inputs[0];
122 int conv_layer_idx = producer_of.at(activ_inp.idx);
123 Conv2Layer* conv = getLayer<Conv2Layer>(newprog, conv_layer_idx);
124 if (conv) {
125 bool ok = conv->fuseActivation(layer.dynamicCast<Layer>());
126 if (ok) {
127 fused_layer_idx = conv_layer_idx;
128 removed_args.push_back(activ_inp);
129 break;
130 }
131 }
132 }
133
134 // fuse Reshape + InstanceNorm(scale=ones,bias=zeros) + Reshape + Mul + Add
135 if (elemwise && elemwise->op == NaryEltwiseLayer::OPERATION::ADD &&
136 ninputs == 2) {
137 int mul_input_idx = -1;
138 Arg add_bias_arg;
139 for (int k = 0; k < 2; k++) {
140 int pidx = producer_of.at(inputs[k].idx);
141 NaryEltwiseLayer* mul = getLayer<NaryEltwiseLayer>(newprog, pidx);
142 if (mul && mul->op == NaryEltwiseLayer::OPERATION::PROD) {
143 mul_input_idx = k;
144 add_bias_arg = inputs[1 - k];
145 break;
146 }
147 }
148 if (mul_input_idx >= 0 && netimpl->isConstArg(add_bias_arg)) {
149 Arg mul_out = inputs[mul_input_idx];
150 int mul_idx = producer_of.at(mul_out.idx);
151 NaryEltwiseLayer* mul = getLayer<NaryEltwiseLayer>(newprog, mul_idx);
152 if (mul && mul->inputs.size() == 2 &&
153 usecounts.at(mul_out.idx) == 1) {
154 int reshape2_input_idx = -1;
155 Arg mul_scale_arg;
156 for (int k = 0; k < 2; k++) {
157 int pidx = producer_of.at(mul->inputs[k].idx);
158 if (getLayer<Reshape2Layer>(newprog, pidx)) {
159 reshape2_input_idx = k;
160 mul_scale_arg = mul->inputs[1 - k];
161 break;
162 }
163 }
164 if (reshape2_input_idx >= 0 && netimpl->isConstArg(mul_scale_arg)) {
165 Arg reshape2_out = mul->inputs[reshape2_input_idx];
166 int reshape2_idx = producer_of.at(reshape2_out.idx);
167 Reshape2Layer* reshape2_lyr = getLayer<Reshape2Layer>(newprog, reshape2_idx);
168 if (reshape2_lyr && reshape2_lyr->inputs.size() >= 1 &&
169 usecounts.at(reshape2_out.idx) == 1) {
170 Arg reshape2_inp = reshape2_lyr->inputs[0];
171 int instnorm_idx = producer_of.at(reshape2_inp.idx);
172 InstanceNormLayer* instnorm = getLayer<InstanceNormLayer>(newprog, instnorm_idx);
173 if (instnorm && instnorm->inputs.size() == 3 &&
174 usecounts.at(reshape2_inp.idx) == 1) {
175 Arg instnorm_inp = instnorm->inputs[0];
176 int reshape1_idx = producer_of.at(instnorm_inp.idx);
177 Reshape2Layer* reshape1_lyr = getLayer<Reshape2Layer>(newprog, reshape1_idx);
178 if (reshape1_lyr && reshape1_lyr->outputs.size() == 1 &&
179 usecounts.at(instnorm_inp.idx) == 1) {
180 Mat in_scale = netimpl->isConstArg(instnorm->inputs[1]) ?
181 netimpl->argTensor(instnorm->inputs[1]) : Mat();
182 Mat in_bias = netimpl->isConstArg(instnorm->inputs[2]) ?
183 netimpl->argTensor(instnorm->inputs[2]) : Mat();
184 bool valid = !in_scale.empty() && !in_bias.empty() &&
185 in_scale.type() == CV_32F && in_bias.type() == CV_32F;
186 if (valid) {
187 const float* sp = in_scale.ptr<float>();
188 const float* bp = in_bias.ptr<float>();
189 bool all_ones = true, all_zeros = true;
190 for (size_t k = 0; k < in_scale.total() && all_ones; k++)
191 all_ones = (std::abs(sp[k] - 1.f) < 1e-6f);
192 for (size_t k = 0; k < in_bias.total() && all_zeros; k++)
193 all_zeros = (std::abs(bp[k]) < 1e-6f);
194 if (all_ones && all_zeros) {
195 Arg orig_inp = reshape1_lyr->inputs[0];
196 Mat mul_scale_mat = netimpl->argTensor(mul_scale_arg);
197 if (in_scale.total() == mul_scale_mat.total()) {
198 // Channel dim preserved — fuse into InstanceNorm
199 instnorm->inputs[0] = orig_inp;
200 instnorm->inputs[1] = mul_scale_arg;
201 instnorm->inputs[2] = add_bias_arg;
202 } else {
203 // Channel dim changed (e.g. [1,C,H,W]->[1,1,C*H*W]):
204 // original is global norm + per-channel affine.
205 // Replace with GroupNorm(num_groups=reshaped_C).
206 int num_groups = (int)in_scale.total();
207 LayerParams gnparams;
208 gnparams.name = instnorm->name;
209 gnparams.type = "GroupNormalization";
210 gnparams.set("epsilon", instnorm->epsilon);
211 gnparams.set("num_groups", num_groups);
212 Ptr<LayerInfo> gnlayer = GroupNormLayer::create(gnparams);
213 gnlayer->netimpl = netimpl;
214 gnlayer->inputs = {orig_inp, mul_scale_arg, add_bias_arg};
215 newprog[instnorm_idx] = gnlayer;
216 }
217 fused_layer_idx = instnorm_idx;
218 removed_args.push_back(instnorm_inp);
219 removed_args.push_back(reshape2_inp);
220 removed_args.push_back(reshape2_out);
221 removed_args.push_back(mul_out);
222 newprog[reshape1_idx] = Ptr<LayerInfo>();
223 newprog[reshape2_idx] = Ptr<LayerInfo>();
224 newprog[mul_idx] = Ptr<LayerInfo>();
225 break;
226 }
227 }
228 }
229 }
230 }
231 }
232 }
233 }
234 }
235 break;
236 }
237
238 if (fused_layer_idx >= 0) {
239 modified = true;
240 Layer* fused_layer = (Layer*)newprog[fused_layer_idx].get();
241 fused_layer->outputs = outputs;
242 for (Arg new_out: outputs)
243 producer_of[new_out.idx] = fused_layer_idx;
244 for (Arg old_out: removed_args) {
245 usecounts.at(old_out.idx) = 0;
246 producer_of.at(old_out.idx) = -1;
247 }
248 } else {
249 for (auto out: outputs)
250 producer_of[out.idx] = (int)newprog.size();
251 newprog.push_back(layer);
252 }
253 }
254
255 if (modified) {
256 size_t i, j = 0, newops = newprog.size();
257 for (i = 0; i < newops; i++) {
258 if (!newprog[i].empty()) {
259 if (j < i)
260 newprog[j] = newprog[i];
261 j++;
262 }
263 }
264 newprog.resize(j);
265 //printf("fused some ops in graph %s. size before: %zu ops, size after: %zu ops\n",
266 // graph->name().data(), nops, j);
267 graph->setProg(newprog);
268 }
269
270 return modified;
271 }

274 vector<int> usecounts;

283// fold BN scale/bias into the weights of the immediately following Conv2 (pre-constArgs only)
284struct FuseBNPass
285{
286 FuseBNPass(Net::Impl* netimpl_) : netimpl(netimpl_) {}
287
288 void run()
289 {
290 netimpl->useCounts(usecounts);
291 fuseGraph(netimpl->mainGraph);
292 }
293
294 void fuseGraph(Ptr<Graph>& graph)
295 {
296 const std::vector<Ptr<LayerInfo> >& prog = graph->prog();
297 size_t nops = prog.size(), nargs = netimpl->args.size();
298 std::vector<Ptr<LayerInfo> > newprog;
299 newprog.reserve(nops);
300 std::vector<int> producer_of((int)nargs, -1);
301 bool modified = false;
302
303 for (size_t i = 0; i < nops; i++) {
304 const Ptr<LayerInfo>& layer = prog[i];
305 Layer* layer_ptr = (Layer*)layer.get();
306
307 std::vector<Ptr<Graph> >* subgraphs = layer->subgraphs();
308 if (subgraphs)
309 for (Ptr<Graph>& g : *subgraphs) fuseGraph(g);
310
311 const std::vector<Arg>& inputs = layer->inputs;
312 const std::vector<Arg>& outputs = layer->outputs;
313
314 Conv2Layer* conv = dynamic_cast<Conv2Layer*>(layer_ptr);
315 if (conv && !inputs.empty()) {
316 Arg conv_inp0 = inputs[0];
317 int bn_idx = conv_inp0.idx >= 0 && conv_inp0.idx < (int)producer_of.size()
318 ? producer_of[conv_inp0.idx] : -1;
319 if (bn_idx >= 0 && usecounts[conv_inp0.idx] == 1) {
320 BatchNorm2Layer* bn = dynamic_cast<BatchNorm2Layer*>(newprog[bn_idx].get());
321 if (bn && fuseForward(conv, bn)) {
322 Arg bn_inp0 = bn->inputs[0];
323 layer_ptr->inputs[0] = bn_inp0;
324 usecounts[conv_inp0.idx] = 0;
325 if (bn_inp0.idx >= 0)
326 usecounts[bn_inp0.idx]++;
327 newprog[bn_idx] = Ptr<LayerInfo>();
328 modified = true;
329 }
330 }
331 }
332
333 for (Arg out : outputs)
334 if (out.idx >= 0 && out.idx < (int)producer_of.size())
335 producer_of[out.idx] = (int)newprog.size();
336 newprog.push_back(layer);
337 }
338
339 if (modified) {
340 size_t j = 0;
341 for (size_t i = 0; i < newprog.size(); i++) {
342 if (!newprog[i].empty()) {
343 if (j < i) newprog[j] = newprog[i];
344 j++;
345 }
346 }
347 newprog.resize(j);
348 graph->setProg(newprog);
349 }
350 }

352 // fold BN scale/bias into Conv2 weight and bias tensors (raw NCHW fp32 only)
353 bool fuseForward(Conv2Layer* conv, BatchNorm2Layer* bn)
354 {
355 // skip padded convolutions: border taps get incorrect bias from folded BN
356 for (int p : conv->pads)
357 if (p > 0) return false;
358
359 if (conv->inputs.size() < 2 || !netimpl->isConstArg(conv->inputs[1]))
360 return false;
361
362 Mat& Wref = netimpl->argTensor(conv->inputs[1]);
363 if (Wref.empty() || Wref.type() != CV_32F || Wref.dims != 4)
364 return false;
365
366 int OC = Wref.size[0];
367 int IC_pg = Wref.size[1]; // input channels per group
368 int KH = Wref.size[2];
369 int KW = Wref.size[3];
370 int ngroups = conv->ngroups;
371 if (ngroups <= 0 || OC % ngroups != 0)
372 return false;
373 int OC_pg = OC / ngroups;
374
375 // resolve BN effective scale and bias; validate all params are const
376 Mat bn_scale, bn_bias_vec;
377 size_t bn_nin = bn->inputs.size();
378 if (bn_nin == 5) {
379 for (int k = 1; k <= 4; k++)
380 if (!netimpl->isConstArg(bn->inputs[k])) return false;
381 BatchNorm2Layer::getScaleBias(
382 netimpl->argTensor(bn->inputs[1]),
383 netimpl->argTensor(bn->inputs[2]),
384 netimpl->argTensor(bn->inputs[3]),
385 netimpl->argTensor(bn->inputs[4]),
386 bn->epsilon, bn_scale, bn_bias_vec);
387 } else if (bn_nin == 1) {
388 bn->getScaleBias(bn_scale, bn_bias_vec);
389 } else {
390 return false;
391 }
392
393 bn_scale.convertTo(bn_scale, CV_32F);
394 bn_bias_vec.convertTo(bn_bias_vec, CV_32F);
395
396 if ((int)bn_scale.total() != IC_pg * ngroups)
397 return false;
398
399 const float* bn_s = bn_scale.ptr<float>();
400 const float* bn_b = bn_bias_vec.ptr<float>();
401
402 // reshape to 2D [OC, IC_pg*KH*KW]; shares buffer with Wref
403 Mat W2d = Wref.reshape(1, OC);
404
405 // scale weights in-place; accumulate per-output-channel bias correction
406 bias_adj.assign(OC, 0.f);
407 for (int g = 0; g < ngroups; g++) {
408 int ic_off = g * IC_pg;
409 for (int oc = g * OC_pg, oc_end = oc + OC_pg; oc < oc_end; oc++) {
410 float* wrow = W2d.ptr<float>(oc);
411 for (int ic = 0; ic < IC_pg; ic++) {
412 float s = bn_s[ic_off + ic];
413 float b = bn_b[ic_off + ic];
414 float* wic = wrow + ic * KH * KW;
415 float sw = 0.f;
416 for (int k = 0; k < KH * KW; k++) {
417 sw += wic[k];
418 wic[k] *= s;
419 }
420 bias_adj[oc] += b * sw;
421 }
422 }
423 }
424
425 // apply bias correction into existing bias tensor or create a new one
426 size_t conv_nin = conv->inputs.size();
427 if (conv_nin >= 3 && netimpl->isConstArg(conv->inputs[2])) {
428 Mat& Bref = netimpl->argTensor(conv->inputs[2]);
429 if (Bref.type() != CV_32F || (int)Bref.total() != OC)
430 return false;
431 float* bp = Bref.ptr<float>();
432 for (int oc = 0; oc < OC; oc++)
433 bp[oc] += bias_adj[oc];
434 } else {
435 Mat newB(1, OC, CV_32F);
436 std::memcpy(newB.ptr<float>(), bias_adj.data(), OC * sizeof(float));
437 Arg ba = netimpl->newConstArg(
438 "__fused_bn_bias_w" + std::to_string(conv->inputs[1].idx), newB);
439 if (conv_nin == 2) conv->inputs.push_back(ba);
440 else conv->inputs[2] = ba;
441 }
442
443 return true;
444 }

446 Net::Impl* netimpl;
447 std::vector<int> usecounts;
448 std::vector<float> bias_adj;
