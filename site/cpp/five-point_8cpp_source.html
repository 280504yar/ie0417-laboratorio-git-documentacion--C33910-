414// Find essential matrix given undistorted points and two cameras.
415static Mat findEssentialMat_( InputArray _points1, InputArray _points2,
416 InputArray cameraMatrix1, InputArray cameraMatrix2,
417 int method, double prob, double threshold, OutputArray _mask)
418{
419 // Scale the points back. We use "arithmetic mean" between the supplied two camera matrices.
420 // Thanks to such 2-stage procedure RANSAC threshold still makes sense, because the undistorted
421 // and rescaled points have a similar value range to the original ones.
422 Mat _pointsTransformed1, _pointsTransformed2;
423 Mat cm1 = cameraMatrix1.getMat(), cm2 = cameraMatrix2.getMat(), cm0;
424 Mat(cm1 + cm2).convertTo(cm0, CV_64F, 0.5);
425 CV_Assert(cm0.rows == 3 && cm0.cols == 3);
426 CV_Assert(std::abs(cm0.at<double>(2, 0)) < 1e-3 &&
427 std::abs(cm0.at<double>(2, 1)) < 1e-3 &&
428 std::abs(cm0.at<double>(2, 2) - 1.) < 1e-3);
429 Mat affine = cm0.rowRange(0, 2);
430
431 transform(_points1, _pointsTransformed1, affine);
432 transform(_points2, _pointsTransformed2, affine);
433
434 return findEssentialMat(_pointsTransformed1, _pointsTransformed2, cm0, method, prob, threshold, 1000/*maxIters*/, _mask);
435}

552int recoverPose( InputArray E, InputArray _points1, InputArray _points2,
553 InputArray _cameraMatrix, OutputArray _R, OutputArray _t, double distanceThresh,
554 InputOutputArray _mask, OutputArray triangulatedPoints)
555{
556 CV_INSTRUMENT_REGION();
557
558 Mat points1, points2, cameraMatrix;
559 _points1.getMat().convertTo(points1, CV_64F);
560 _points2.getMat().convertTo(points2, CV_64F);
561 _cameraMatrix.getMat().convertTo(cameraMatrix, CV_64F);
562
563 int npoints = points1.checkVector(2);
564 CV_Assert( npoints >= 0 && points2.checkVector(2) == npoints &&
565 points1.type() == points2.type());
566
567 CV_Assert(cameraMatrix.rows == 3 && cameraMatrix.cols == 3 && cameraMatrix.channels() == 1);
568
569 if (points1.channels() > 1)
570 {
571 points1 = points1.reshape(1, npoints);
572 points2 = points2.reshape(1, npoints);
573 }
574
575 double fx = cameraMatrix.at<double>(0,0);
576 double fy = cameraMatrix.at<double>(1,1);
577 double cx = cameraMatrix.at<double>(0,2);
578 double cy = cameraMatrix.at<double>(1,2);
579
580 points1.col(0) = (points1.col(0) - cx) / fx;
581 points2.col(0) = (points2.col(0) - cx) / fx;
582 points1.col(1) = (points1.col(1) - cy) / fy;
583 points2.col(1) = (points2.col(1) - cy) / fy;
584
585 points1 = points1.t();
586 points2 = points2.t();
587
588 Mat R1, R2, t;
589 decomposeEssentialMat(E, R1, R2, t);
590 Mat P0 = Mat::eye(3, 4, R1.type());
591 Mat P1(3, 4, R1.type()), P2(3, 4, R1.type()), P3(3, 4, R1.type()), P4(3, 4, R1.type());
592 P1(Range::all(), Range(0, 3)) = R1 * 1.0; P1.col(3) = t * 1.0;
593 P2(Range::all(), Range(0, 3)) = R2 * 1.0; P2.col(3) = t * 1.0;
594 P3(Range::all(), Range(0, 3)) = R1 * 1.0; P3.col(3) = -t * 1.0;
595 P4(Range::all(), Range(0, 3)) = R2 * 1.0; P4.col(3) = -t * 1.0;
596
597 // Do the chirality check.
598 // Notice here a threshold dist is used to filter
599 // out far away points (i.e. infinite points) since
600 // their depth may vary between positive and negative.
601 std::vector<Mat> allTriangulations(4);
602 Mat Q;
603
604 triangulatePoints(P0, P1, points1, points2, Q);
605 if(triangulatedPoints.needed())
606 Q.copyTo(allTriangulations[0]);
607 Mat mask1 = Q.row(2).mul(Q.row(3)) > 0;
608 Q.row(0) /= Q.row(3);
609 Q.row(1) /= Q.row(3);
610 Q.row(2) /= Q.row(3);
611 Q.row(3) /= Q.row(3);
612 mask1 = (Q.row(2) < distanceThresh) & mask1;
613 Q = P1 * Q;
614 mask1 = (Q.row(2) > 0) & mask1;
615 mask1 = (Q.row(2) < distanceThresh) & mask1;
616
617 triangulatePoints(P0, P2, points1, points2, Q);
618 if(triangulatedPoints.needed())
619 Q.copyTo(allTriangulations[1]);
620 Mat mask2 = Q.row(2).mul(Q.row(3)) > 0;
621 Q.row(0) /= Q.row(3);
622 Q.row(1) /= Q.row(3);
623 Q.row(2) /= Q.row(3);
624 Q.row(3) /= Q.row(3);
625 mask2 = (Q.row(2) < distanceThresh) & mask2;
626 Q = P2 * Q;
627 mask2 = (Q.row(2) > 0) & mask2;
628 mask2 = (Q.row(2) < distanceThresh) & mask2;
629
630 triangulatePoints(P0, P3, points1, points2, Q);
631 if(triangulatedPoints.needed())
632 Q.copyTo(allTriangulations[2]);
633 Mat mask3 = Q.row(2).mul(Q.row(3)) > 0;
634 Q.row(0) /= Q.row(3);
635 Q.row(1) /= Q.row(3);
636 Q.row(2) /= Q.row(3);
637 Q.row(3) /= Q.row(3);
638 mask3 = (Q.row(2) < distanceThresh) & mask3;
639 Q = P3 * Q;
640 mask3 = (Q.row(2) > 0) & mask3;
641 mask3 = (Q.row(2) < distanceThresh) & mask3;
642
643 triangulatePoints(P0, P4, points1, points2, Q);
644 if(triangulatedPoints.needed())
645 Q.copyTo(allTriangulations[3]);
646 Mat mask4 = Q.row(2).mul(Q.row(3)) > 0;
647 Q.row(0) /= Q.row(3);
648 Q.row(1) /= Q.row(3);
649 Q.row(2) /= Q.row(3);
650 Q.row(3) /= Q.row(3);
651 mask4 = (Q.row(2) < distanceThresh) & mask4;
652 Q = P4 * Q;
653 mask4 = (Q.row(2) > 0) & mask4;
654 mask4 = (Q.row(2) < distanceThresh) & mask4;
655
656 mask1 = mask1.t();
657 mask2 = mask2.t();
658 mask3 = mask3.t();
659 mask4 = mask4.t();
660
661 // If _mask is given, then use it to filter outliers.
662 if (!_mask.empty())
663 {
664 Mat mask = _mask.getMat();
665 CV_Assert(npoints == mask.checkVector(1));
666 mask = mask.reshape(1, npoints);
667 bitwise_and(mask, mask1, mask1);
668 bitwise_and(mask, mask2, mask2);
669 bitwise_and(mask, mask3, mask3);
670 bitwise_and(mask, mask4, mask4);
671 }
672 if (_mask.empty() && _mask.needed())
673 {
674 _mask.create(mask1.size(), CV_8U);
675 }
676
677 CV_Assert(_R.needed() && _t.needed());
678 _R.create(3, 3, R1.type());
679 _t.create(3, 1, t.type());
680
681 int good1 = countNonZero(mask1);
682 int good2 = countNonZero(mask2);
683 int good3 = countNonZero(mask3);
684 int good4 = countNonZero(mask4);
685
686 if (good1 >= good2 && good1 >= good3 && good1 >= good4)
687 {
688 if(triangulatedPoints.needed()) allTriangulations[0].copyTo(triangulatedPoints);
689 R1.copyTo(_R);
690 t.copyTo(_t);
691 if (_mask.needed()) mask1.copyTo(_mask);
692 return good1;
693 }
694 else if (good2 >= good1 && good2 >= good3 && good2 >= good4)
695 {
696 if(triangulatedPoints.needed()) allTriangulations[1].copyTo(triangulatedPoints);
697 R2.copyTo(_R);
698 t.copyTo(_t);
699 if (_mask.needed()) mask2.copyTo(_mask);
700 return good2;
701 }
702 else if (good3 >= good1 && good3 >= good2 && good3 >= good4)
703 {
704 if(triangulatedPoints.needed()) allTriangulations[2].copyTo(triangulatedPoints);
705 t = -t;
706 R1.copyTo(_R);
707 t.copyTo(_t);
708 if (_mask.needed()) mask3.copyTo(_mask);
709 return good3;
710 }
711 else
712 {
713 if(triangulatedPoints.needed()) allTriangulations[3].copyTo(triangulatedPoints);
714 t = -t;
715 R2.copyTo(_R);
716 t.copyTo(_t);
717 if (_mask.needed()) mask4.copyTo(_mask);
718 return good4;
719 }
720}

735void decomposeEssentialMat( InputArray _E, OutputArray _R1, OutputArray _R2, OutputArray _t )
736{
737 CV_INSTRUMENT_REGION();
738
739 Mat E = _E.getMat().reshape(1, 3);
740 CV_Assert(E.cols == 3 && E.rows == 3);
741
742 Mat D, U, Vt;
743 SVD::compute(E, D, U, Vt);
744
745 if (determinant(U) < 0) U *= -1.;
746 if (determinant(Vt) < 0) Vt *= -1.;
747
748 Mat W = (Mat_<double>(3, 3) << 0, 1, 0, -1, 0, 0, 0, 0, 1);
749 W.convertTo(W, E.type());
750
751 Mat R1, R2, t;
752 R1 = U * W * Vt;
