23static bool ocl_hasNonZero( InputArray _src, bool & res )
24{
25 int type = _src.type(), depth = CV_MAT_DEPTH(type), kercn = ocl::predictOptimalVectorWidth(_src);
26 bool doubleSupport = ocl::Device::getDefault().doubleFPConfig() > 0;
27
28 if (depth == CV_64F && !doubleSupport)
29 return false;
30
31 int dbsize = ocl::Device::getDefault().maxComputeUnits();
32 size_t wgs = ocl::Device::getDefault().maxWorkGroupSize();
33
34 int wgs2_aligned = 1;
35 while (wgs2_aligned < (int)wgs)
36 wgs2_aligned <<= 1;
37 wgs2_aligned >>= 1;
38
39 ocl::Kernel k("reduce", ocl::core::reduce_oclsrc,
40 format("-D srcT=%s -D srcT1=%s -D cn=1 -D OP_COUNT_NON_ZERO"
41 " -D WGS=%d -D kercn=%d -D WGS2_ALIGNED=%d%s%s",
42 ocl::typeToStr(CV_MAKE_TYPE(depth, kercn)),
43 ocl::typeToStr(depth), (int)wgs, kercn,
44 wgs2_aligned, doubleSupport ? " -D DOUBLE_SUPPORT" : "",
45 _src.isContinuous() ? " -D HAVE_SRC_CONT" : ""));
46 if (k.empty())
47 return false;
48
49 UMat src = _src.getUMat(), db(1, dbsize, CV_32SC1);
50 k.args(ocl::KernelArg::ReadOnlyNoSize(src), src.cols, (int)src.total(),
51 dbsize, ocl::KernelArg::PtrWriteOnly(db));
52
53 size_t globalsize = dbsize * wgs;
54 if (k.run(1, &globalsize, &wgs, true))
55 return res = (saturate_cast<int>(cv::sum(db.getMat(ACCESS_READ))[0])>0), true;
56 return false;
57}

60bool hasNonZero(InputArray _src)
61{
62 CV_INSTRUMENT_REGION();
63
64 int type = _src.type(), cn = CV_MAT_CN(type);
65 CV_Assert( cn == 1 );
66
67 bool res = false;
68
69#ifdef HAVE_OPENCL
70 CV_OCL_RUN_(OCL_PERFORMANCE_CHECK(_src.isUMat()) && _src.dims() <= 2,
71 ocl_hasNonZero(_src, res),
72 res)
73#endif
74
75 Mat src = _src.getMat();
76
77 HasNonZeroFunc func = getHasNonZeroFunc(src.depth());
78 CV_Assert( func != 0 );
79
80 if (src.dims == 2)//fast path to avoid creating planes of single rows
81 {
82 if (src.isContinuous())
83 res |= func(src.ptr<uchar>(0), src.total());
84 else
85 for(int row = 0, rowsCount = src.rows ; !res && (row<rowsCount) ; ++row)
86 res |= func(src.ptr<uchar>(row), src.cols);
87 }
88 else//if (src.dims != 2)
89 {
90 const Mat* arrays[] = {&src, nullptr};
91 Mat planes[1];
92 NAryMatIterator itNAry(arrays, planes, 1);
93 for(size_t p = 0 ; !res && (p<itNAry.nplanes) ; ++p, ++itNAry)
94 {
95 const Mat& plane = itNAry.planes[0];
96 if (plane.isContinuous())
97 res |= func(plane.ptr<uchar>(0), plane.total());
98 else
99 for(int row = 0, rowsCount = plane.rows ; !res && (row<rowsCount) ; ++row)
100 res |= func(plane.ptr<uchar>(row), plane.cols);
101 }
102 }
103
104 return res;
105}
