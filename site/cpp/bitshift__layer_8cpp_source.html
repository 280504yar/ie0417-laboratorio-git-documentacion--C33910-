11// ONNX reference: BitShift operator
12// Spec: https://onnx.ai/onnx/operators/onnx__BitShift.html
13// Supported opsets: ai.onnx opset 11 and newer
14// NOTE: Broadcasting is NOT fully supported. Only two cases are handled:
15// 1) array (input[0]) shifted by array (input[1]) of the SAME SHAPE
16// 2) array (input[0]) shifted by a SCALAR (0-D) shift amount

21template<typename T, typename U>
22static inline T doShift(T inputVal, U shiftVal, int direction, int bitWidth)
23{
24 return (uint64_t)shiftVal >= (uint64_t)bitWidth
25 ? T(0)
26 : T(direction ? (inputVal >> shiftVal) : (inputVal << shiftVal));
27}

29template<typename T, int CvTypeConst, int BitWidth>
30void runBitShift(const Mat& input, const Mat& shift, Mat& output, int direction)
31{
32 output.create(input.size, input.type());
33 const size_t numElements = input.total();
34
35 const T* inputPtr = input.ptr<T>();
36 T* outputPtr = output.ptr<T>();
37
38 if (shift.total() == 1)
39 {
40 T shiftScalar = 0;
41 tensorToScalar(shift, CvTypeConst, &shiftScalar);
42 parallel_for_(Range(0, (int)numElements), [&](const Range& r){
43 for (int i = r.start; i < r.end; ++i)
44 outputPtr[i] = doShift<T,T>(inputPtr[i], shiftScalar, direction, BitWidth);
45 });
46 }
47 else
48 {
49 CV_Assert(shift.size == input.size);
50 CV_Assert(shift.type() == CvTypeConst);
51 const T* shiftPtr = shift.ptr<T>();
52 parallel_for_(Range(0, (int)numElements), [&](const Range& r){
53 for (int i = r.start; i < r.end; ++i)
54 outputPtr[i] = doShift<T,T>(inputPtr[i], shiftPtr[i], direction, BitWidth);
55 });
56 }
57}

96 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr,
97 OutputArrayOfArrays) CV_OVERRIDE
98 {
99 std::vector<Mat> inputs, outputs;
100 inputs_arr.getMatVector(inputs);
101 outputs_arr.getMatVector(outputs);
102
103 CV_Assert(inputs.size() >= 2);
104 const Mat& input = inputs[0];
105 const Mat& shift = inputs[1];
106 Mat& output = outputs[0];
107
108 const int depth = input.depth();
109 if (depth == CV_8U)
110 {
111 runBitShift<uint8_t, CV_8U, 8>(input, shift, output, direction_);
112 }
113 else if (depth == CV_16U)
114 {
115 runBitShift<uint16_t, CV_16U, 16>(input, shift, output, direction_);
116 }
117 else if (depth == CV_32U)
118 {
119 runBitShift<uint32_t, CV_32U, 32>(input, shift, output, direction_);
120 }
121 else if (depth == CV_64U)
122 {
123 runBitShift<uint64_t, CV_64U, 64>(input, shift, output, direction_);
124 }
125 else
126 {
127 CV_Error(Error::StsNotImplemented, "BitShift: unsupported depth");
128 }
129 }
