18/*
19 Implementation of FlexAttention (domain ai.onnx.preview, opset 1).
20 Spec: https://github.com/onnx/onnx/blob/main/docs/Operators-preview.md#aionnxpreviewFlexAttention
21 Supported: score_mod / prob_mod sub-graphs (inlined by the importer).
22 An explicit softmax_precision is rejected; only the spec default is implemented.
23*/
24// Scaled dot-product attention with GQA and optional score_mod / prob_mod
25// sub-graphs that rewrite the whole [B, H, L, S] score / probability tensor:
26// scores = scale * (Q . expand_kv(K)^T); scores = score_mod(scores)
27// probs = softmax(scores, axis=-1); probs = prob_mod(probs)
28// Y = probs . expand_kv(V)
29// The sub-graphs carry no control flow, so the importer inlines them as ordinary
30// graph nodes and cuts this op into three stages executed by this layer:
31// "full" : Q, K, V -> Y (no sub-graphs)
32// "qk" : Q, K -> scores (feeds score_mod / Softmax)
33// "av" : probs, V-> Y (consumes prob_mod / Softmax output)
34
35class FlexAttentionLayerImpl CV_FINAL : public FlexAttentionLayer
36{
37public:
38 FlexAttentionLayerImpl(const LayerParams& params)
39 {
40 setParamsFrom(params);
41 scale = params.get<float>("scale", 0.f);
42 has_scale = params.has("scale");
43 stage = params.get<std::string>("stage", "full");
44 CV_Check(stage, stage == "full" || stage == "qk" || stage == "av", "FlexAttention: bad stage");
45 opt.init();
46 }

50 void getTypes(const std::vector<MatType>& inputs, const int requiredOutputs, const int requiredInternals,
51 std::vector<MatType>& outputs, std::vector<MatType>& internals) const CV_OVERRIDE
52 {
53 CV_CheckGE(inputs.size(), (size_t)2, "FlexAttention needs at least 2 inputs");
54 CV_CheckType(inputs[0], inputs[0] == CV_32F || inputs[0] == CV_64F || inputs[0] == CV_16F,
55 "FlexAttention: only FP32/FP64/FP16 are supported");
56 outputs.assign(requiredOutputs, inputs[0]);
57 internals.assign(requiredInternals, CV_32F); // scores buffer is always fp32
58 }

60 bool getMemoryShapes(const std::vector<MatShape>& inputs, const int /*ro*/,
61 std::vector<MatShape>& outputs, std::vector<MatShape>& internals) const CV_OVERRIDE
62 {
63 CV_CheckEQ(inputs[0].dims, 4, "FlexAttention: inputs must be 4D [batch, heads, seq, head_size]");
64 const int B = inputs[0][0], Hq = inputs[0][1], L = inputs[0][2];
65 internals.clear();
66 if (stage == "qk") // Q[B,Hq,L,D], K[B,Hkv,S,D] -> scores[B,Hq,L,S]
67 outputs.assign(1, MatShape{B, Hq, L, inputs[1][2]});
68 else if (stage == "av") // probs[B,Hq,L,S], V[B,Hkv,S,Dv] -> Y[B,Hq,L,Dv]
69 outputs.assign(1, MatShape{B, Hq, L, inputs[1][3]});
70 else // Q,K,V -> Y[B,Hq,L,Dv]
71 {
72 outputs.assign(1, MatShape{B, Hq, L, inputs[2][3]});
73 // scores buffer, pooled across forward(); the fp64 path uses its own scratch.
74 internals.assign(1, MatShape{B, Hq, L, inputs[1][2]});
75 }
76 return false;
77 }

79 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
80 {
81 std::vector<Mat> inputs, outputs, internals;
82 inputs_arr.getMatVector(inputs);
83 outputs_arr.getMatVector(outputs);
84 internals_arr.getMatVector(internals);
85
86 if (inputs[0].depth() == CV_64F)
87 run<double>(inputs, outputs); // scalar path (fastGemmBatch is float-only)
88 else
89 runFloat(inputs, outputs, internals, inputs[0].depth() == CV_16F);
90 }
91
92 int64 getFLOPS(const std::vector<MatShape>& inputs, const std::vector<MatShape>& /*outputs*/) const CV_OVERRIDE
93 {
94 const int64 B = inputs[0][0], Hq = inputs[0][1], L = inputs[0][2];
95 if (stage == "qk") // Q[B,Hq,L,D] x K[B,Hkv,S,D]^T
96 return CV_BIG_INT(2) * B * Hq * L * inputs[1][2] * inputs[0][3];
97 if (stage == "av") // P[B,Hq,L,S] x V[B,Hkv,S,Dv]
98 return CV_BIG_INT(2) * B * Hq * L * inputs[0][3] * inputs[1][3];
99 const int64 S = inputs[1][2], D = inputs[0][3], Dv = inputs[2][3];
100 return CV_BIG_INT(2) * B * Hq * L * S * (D + Dv) + 4 * B * Hq * L * S; // + softmax
101 }

103private:
104 // fp32/fp16 path: batched GEMM via fastGemmBatch (MLAS-accelerated when built with
105 // HAVE_MLAS). fp16 is computed in fp32. K/V are shared across each GQA group through the
106 // per-head offset arithmetic (n -> n/group), matching the ONNX Attention layer.
107 void runFloat(std::vector<Mat>& rawIn, std::vector<Mat>& rawOut, std::vector<Mat>& internals, bool fp16)
108 {
109 std::vector<Mat> in, out;
110 if (fp16)
111 {
112 in.resize(rawIn.size());
113 for (size_t i = 0; i < rawIn.size(); ++i)
114 if (!rawIn[i].empty()) rawIn[i].convertTo(in[i], CV_32F);
115 out.resize(rawOut.size());
116 out[0].create(rawOut[0].dims, rawOut[0].size.p, CV_32F);
117 }
118 std::vector<Mat>& I = fp16 ? in : rawIn;
119 std::vector<Mat>& O = fp16 ? out : rawOut;
120
121 if (stage == "qk")
122 qkGemm(I[0], I[1], O[0]);
123 else if (stage == "av")
124 avGemm(I[0], I[1], O[0]);
125 else
126 {
127 CV_CheckEQ(internals.size(), (size_t)1, "FlexAttention: missing scores buffer");
128 fullFloat(I[0], I[1], I[2], O[0], internals[0]);
129 }
130
131 if (fp16) O[0].convertTo(rawOut[0], CV_16F);
132 }

134 // scores[B,Hq,L,Skv] = scl * Q * K^T (K^T via ldb0=1,ldb1=D).
135 void qkGemm(const Mat& Q, const Mat& K, Mat& S)
136 {
137 const int B = Q.size[0], Hq = Q.size[1], L = Q.size[2], D = Q.size[3];
138 const int Hkv = K.size[1], Skv = K.size[2], group = Hq / Hkv;
139 const float scl = has_scale ? scale : (float)(1.0 / std::sqrt((double)D));
140 // offset tables assume fully packed tensors
141 CV_Assert(Q.isContinuous() && K.isContinuous() && S.isContinuous());
142
143 const size_t batch = (size_t)B * Hq;
144 std::vector<size_t> qo(batch), ko(batch), so(batch);
145 for (int b = 0; b < B; ++b)
146 for (int n = 0; n < Hq; ++n)
147 {
148 const size_t bn = (size_t)b * Hq + n;
149 qo[bn] = ((size_t)b * Hq + n) * L * D;
150 ko[bn] = ((size_t)b * Hkv + n / group) * Skv * D;
151 so[bn] = ((size_t)b * Hq + n) * L * Skv;
152 }
153 fastGemmBatch(batch, qo.data(), ko.data(), so.data(),
154 L, Skv, D, scl, Q.ptr<float>(), D, 1, K.ptr<float>(), 1, D,
155 0.f, S.ptr<float>(), Skv, opt);
156 }
157
158 // Y[B,Hq,L,Dv] = P * V.
159 void avGemm(const Mat& P, const Mat& V, Mat& Y)
160 {
161 const int B = P.size[0], Hq = P.size[1], L = P.size[2], Skv = P.size[3];
162 const int Hkv = V.size[1], Dv = V.size[3], group = Hq / Hkv;
163 CV_Assert(P.isContinuous() && V.isContinuous() && Y.isContinuous());
164
165 const size_t batch = (size_t)B * Hq;
166 std::vector<size_t> po(batch), vo(batch), yo(batch);
167 for (int b = 0; b < B; ++b)
168 for (int n = 0; n < Hq; ++n)
169 {
170 const size_t bn = (size_t)b * Hq + n;
171 po[bn] = ((size_t)b * Hq + n) * L * Skv;
172 vo[bn] = ((size_t)b * Hkv + n / group) * Skv * Dv;
173 yo[bn] = ((size_t)b * Hq + n) * L * Dv;
174 }
175 fastGemmBatch(batch, po.data(), vo.data(), yo.data(),
176 L, Dv, Skv, 1.f, P.ptr<float>(), Skv, 1, V.ptr<float>(), Dv, 1,
177 0.f, Y.ptr<float>(), Dv, opt);
178 }
179
180 void fullFloat(const Mat& Q, const Mat& K, const Mat& V, Mat& Y, Mat& scores)
181 {
182 qkGemm(Q, K, scores);
183 softmax(scores, scores, 3); // in-place, last axis
184 avGemm(scores, V, Y);
185 }

188 template<typename T>
189 void run(std::vector<Mat>& I, std::vector<Mat>& O)
190 {
191 if (stage == "qk")
192 qk<T>(I[0], I[1], O[0]);
193 else if (stage == "av")
194 av<T>(I[0], I[1], O[0]);
195 else
196 full<T>(I[0], I[1], I[2], O[0]);
197 }

199 // scores[b,n,l,s] = scale * sum_d Q[b,n,l,d] * K[b, n/group, s, d]
200 template<typename T>
201 void qk(const Mat& Q, const Mat& K, Mat& S)
202 {
203 const int B = Q.size[0], Hq = Q.size[1], L = Q.size[2], D = Q.size[3];
204 const int Hkv = K.size[1], Skv = K.size[2], group = Hq / Hkv;
205 const T scl = static_cast<T>(has_scale ? scale : 1.0 / std::sqrt((double)D));
206 parallel_for_(Range(0, B * Hq), [&](const Range& r) {
207 for (int bn = r.start; bn < r.end; ++bn) {
208 const int b = bn / Hq, n = bn % Hq, h = n / group;
209 const T* q = Q.ptr<T>() + ((size_t)b * Hq + n) * L * D;
210 const T* k = K.ptr<T>() + ((size_t)b * Hkv + h) * Skv * D;
211 T* s = S.ptr<T>() + ((size_t)b * Hq + n) * L * Skv;
212 for (int l = 0; l < L; ++l)
213 for (int j = 0; j < Skv; ++j) {
214 T acc = 0;
215 for (int d = 0; d < D; ++d) acc += q[l * D + d] * k[j * D + d];
216 s[l * Skv + j] = acc * scl;
217 }
218 }
219 });
220 }
221
222 // Y[b,n,l,:] = sum_s P[b,n,l,s] * V[b, n/group, s, :]
223 template<typename T>
224 void av(const Mat& P, const Mat& V, Mat& Y)
225 {
226 const int B = P.size[0], Hq = P.size[1], L = P.size[2], Skv = P.size[3];
227 const int Hkv = V.size[1], Dv = V.size[3], group = Hq / Hkv;
228 parallel_for_(Range(0, B * Hq), [&](const Range& r) {
229 for (int bn = r.start; bn < r.end; ++bn) {
230 const int b = bn / Hq, n = bn % Hq, h = n / group;
231 const T* p = P.ptr<T>() + ((size_t)b * Hq + n) * L * Skv;
232 const T* v = V.ptr<T>() + ((size_t)b * Hkv + h) * Skv * Dv;
233 T* y = Y.ptr<T>() + ((size_t)b * Hq + n) * L * Dv;
234 for (int l = 0; l < L; ++l) {
235 for (int c = 0; c < Dv; ++c) y[l * Dv + c] = 0;
236 for (int j = 0; j < Skv; ++j) {
237 const T pw = p[l * Skv + j];
238 for (int c = 0; c < Dv; ++c) y[l * Dv + c] += pw * v[j * Dv + c];
239 }
240 }
241 }
242 });
243 }

245 template<typename T>
246 void full(const Mat& Q, const Mat& K, const Mat& V, Mat& Y)
247 {
248 const int B = Q.size[0], Hq = Q.size[1], L = Q.size[2], D = Q.size[3];
249 const int Hkv = K.size[1], Skv = K.size[2], Dv = V.size[3], group = Hq / Hkv;
250 const T scl = static_cast<T>(has_scale ? scale : 1.0 / std::sqrt((double)D));
251 parallel_for_(Range(0, B * Hq), [&](const Range& r) {
252 std::vector<T> s(Skv);
253 for (int bn = r.start; bn < r.end; ++bn) {
254 const int b = bn / Hq, n = bn % Hq, h = n / group;
255 const T* q = Q.ptr<T>() + ((size_t)b * Hq + n) * L * D;
256 const T* k = K.ptr<T>() + ((size_t)b * Hkv + h) * Skv * D;
257 const T* v = V.ptr<T>() + ((size_t)b * Hkv + h) * Skv * Dv;
258 T* y = Y.ptr<T>() + ((size_t)b * Hq + n) * L * Dv;
259 for (int l = 0; l < L; ++l) {
260 T mx = -std::numeric_limits<T>::infinity();
261 for (int j = 0; j < Skv; ++j) {
262 T acc = 0;
263 for (int d = 0; d < D; ++d) acc += q[l * D + d] * k[j * D + d];
264 s[j] = acc * scl;
265 mx = std::max(mx, s[j]);
266 }
267 T sum = 0;
268 for (int j = 0; j < Skv; ++j) { s[j] = std::exp(s[j] - mx); sum += s[j]; }
269 const T inv = sum > 0 ? (T)1 / sum : 0;
270 for (int c = 0; c < Dv; ++c) y[l * Dv + c] = 0;
271 for (int j = 0; j < Skv; ++j) {
272 const T pw = s[j] * inv;
273 for (int c = 0; c < Dv; ++c) y[l * Dv + c] += pw * v[j * Dv + c];
274 }
275 }
276 }
277 });
278 }
