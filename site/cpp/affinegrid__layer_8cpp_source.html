21template<typename T>
22void computeGrid2D(const Mat& theta, Mat& grid, int N, int H, int W,
23 float xs, float xd, float ys, float yd)
24{
25 const int total = N * H;
26 parallel_for_(Range(0, total), [&](const Range& range){
27 const int H_ = H;
28 const int W_ = W;
29 const float xs_ = xs, xd_ = xd, ys_ = ys, yd_ = yd;
30 const Mat theta_ = theta;
31 Mat grid_ = grid;
32 int lastN = -1;
33 const T* T0 = nullptr;
34 const T* T1 = nullptr;
35 float T0_0 = 0.f, T1_0 = 0.f;
36 for (int nyi = range.start; nyi < range.end; nyi++)
37 {
38 int n = nyi / H_;
39 int y = nyi - n * H_;
40 if (n != lastN)
41 {
42 int i0[3] = {n, 0, 0};
43 int i1[3] = {n, 1, 0};
44 T0 = theta_.ptr<T>(i0);
45 T1 = theta_.ptr<T>(i1);
46 T0_0 = (float)T0[0];
47 T1_0 = (float)T1[0];
48 lastN = n;
49 }
50 float ny = yd_ + ys_ * y;
51 float base0 = (float)T0[1]*ny + (float)T0[2];
52 float base1 = (float)T1[1]*ny + (float)T1[2];
53 float* out = grid_.ptr<float>(n, y, 0);
54 for (int x = 0; x < W_; x++)
55 {
56 float nx = xd_ + xs_ * x;
57 out[2*x + 0] = T0_0*nx + base0;
58 out[2*x + 1] = T1_0*nx + base1;
59 }
60 }
61 });
62}

64template<typename T>
65void computeGrid3D(const Mat& theta, Mat& grid, int N, int D, int H, int W,
66 float xs, float xd, float ys, float yd, float zs, float zd)
67{
68 const int stride = D * H;
69 const int total = N * stride;
70 parallel_for_(Range(0, total), [&](const Range& range){
71 const int H_ = H;
72 const int W_ = W;
73 const int stride_ = stride;
74 const float xs_ = xs, xd_ = xd, ys_ = ys, yd_ = yd, zs_ = zs, zd_ = zd;
75 const Mat theta_ = theta;
76 Mat grid_ = grid;
77 int lastN = -1;
78 const T* T0 = nullptr;
79 const T* T1 = nullptr;
80 const T* T2 = nullptr;
81 float T0_0 = 0.f, T1_0 = 0.f, T2_0 = 0.f;
82 for (int ndz = range.start; ndz < range.end; ndz++)
83 {
84 int n = ndz / stride_;
85 int rem = ndz - n * stride_;
86 int z = rem / H_;
87 int y = rem - z * H_;
88 if (n != lastN)
89 {
90 int i0[3] = {n, 0, 0};
91 int i1[3] = {n, 1, 0};
92 int i2[3] = {n, 2, 0};
93 T0 = theta_.ptr<T>(i0);
94 T1 = theta_.ptr<T>(i1);
95 T2 = theta_.ptr<T>(i2);
96 T0_0 = (float)T0[0];
97 T1_0 = (float)T1[0];
98 T2_0 = (float)T2[0];
99 lastN = n;
100 }
101 float ny = yd_ + ys_ * y;
102 float nz = zd_ + zs_ * z;
103 float base0 = (float)T0[1]*ny + (float)T0[2]*nz + (float)T0[3];
104 float base1 = (float)T1[1]*ny + (float)T1[2]*nz + (float)T1[3];
105 float base2 = (float)T2[1]*ny + (float)T2[2]*nz + (float)T2[3];
106 int idx[5] = {n, z, y, 0, 0};
107 float* out = grid_.ptr<float>(idx);
108 for (int x = 0; x < W_; x++)
109 {
110 float nx = xd_ + xs_ * x;
111 int o = 3*x;
112 out[o + 0] = T0_0*nx + base0;
113 out[o + 1] = T1_0*nx + base1;
114 out[o + 2] = T2_0*nx + base2;
115 }
116 }
117 });
118}

131 void resolveSizeAndShape(int N, bool is3d, const Mat* explicitSz,
132 MatShape& outShape, int& D, int& H, int& W) const
133 {
134 outShape = is3d ? MatShape{N, -1, -1, -1, 3} : MatShape{N, -1, -1, 2};
135 D = 1; H = -1; W = -1;
136
137 Mat sz;
138 if (explicitSz)
139 {
140 sz = *explicitSz;
141 }
142 else
143 {
144 Net::Impl* netimpl_ = getNetImpl(this);
145 if (!netimpl_)
146 return;
147 try
148 {
149 sz = netimpl_->argTensor(this->inputs[1]);
150 }
151 catch (const cv::Exception& e)
152 {
153 CV_Error(cv::Error::StsError,
154 cv::format("DNN/AffineGrid: failed to resolve output shape from 'size' tensor: %s", e.what()));
155 }
156 }
157
158 if (sz.empty())
159 return;
160
161 std::vector<int> sizeVec;
162 tensorToIntVec(sz, sizeVec);
163 CV_CheckTrue((int)sizeVec.size() == (is3d ? 5 : 4), "size input must have 4 (2D) or 5 (3D) elements");
164
165 if (is3d) {
166 D = sizeVec[2];
167 H = sizeVec[3];
168 W = sizeVec[4];
169 if (D > 0 && H > 0 && W > 0)
170 outShape = MatShape{N, D, H, W, 3};
171 } else {
172 H = sizeVec[2];
173 W = sizeVec[3];
174 if (H > 0 && W > 0)
175 outShape = MatShape{N, H, W, 2};
176 }
177 }

179 bool getMemoryShapes(const std::vector<MatShape> &inputs,
180 const int requiredOutputs,
181 std::vector<MatShape> &outputs,
182 std::vector<MatShape> &internals) const CV_OVERRIDE
183 {
184 CV_CheckGE((int)inputs.size(), 2, "AffineGrid requires two inputs: theta and size");
185
186 const MatShape thetaShape = inputs[0];
187 CV_CheckGE((int)thetaShape.size(), 3, "theta must be at least 3D [N, r, c]");
188 int N = thetaShape[0];
189 int r = thetaShape[1];
190 int c = thetaShape[2];
191 CV_CheckTrue((r == 2 && c == 3) || (r == 3 && c == 4), "theta must be [N,2,3] or [N,3,4]");
192 const bool is3d = (r == 3);
193
194 MatShape outShape;
195 const MatShape sizeShape = inputs[1];
196 CV_CheckTrue(sizeShape.size() == 1, "size input must be 1D tensor");
197 const int dims = (int)sizeShape[0];
198 CV_CheckTrue(dims == (is3d ? 5 : 4), "size input must have 4 (2D) or 5 (3D) elements");
199 int D, H, W;
200 resolveSizeAndShape(N, is3d, /*explicitSz*/nullptr, outShape, D, H, W);
201
202 outputs.assign(1, outShape);
203 return false;
204 }

238 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
239 {
240 auto outKind = outputs_arr.kind();
241 CV_Assert(outKind == _InputArray::STD_VECTOR_MAT || outKind == _InputArray::STD_VECTOR_UMAT);
242
243 std::vector<Mat> inMats;
244 inputs_arr.getMatVector(inMats);
245 CV_CheckGE((int)inMats.size(), 2, "AffineGrid requires two inputs: theta and size");
246 Mat theta = inMats[0];
247 Mat sz = inMats[1];
248
249 CV_CheckTrue(theta.dims == 3, "theta must be 3D [N, r, c]");
250 const int N = theta.size[0];
251 const int r = theta.size[1];
252 const bool is3d = (r == 3);
253
254 int D = 1, H = 0, W = 0;
255 MatShape dummy;
256 resolveSizeAndShape(N, is3d, &sz, dummy, D, H, W);
257
258 CV_CheckGT(H, 0, "H must be > 0");
259 CV_CheckGT(W, 0, "W must be > 0");
260 if (is3d) CV_CheckGT(D, 0, "D must be > 0");
261
262 std::vector<int> outShape = is3d ? std::vector<int>{N, D, H, W, 3}
263 : std::vector<int>{N, H, W, 2};
264 MatShape fitShape(outShape.begin(), outShape.end());
265 Mat grid;
266 if (outKind == _InputArray::STD_VECTOR_MAT) {
267 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
268 outs.resize(1);
269 outs[0].fit(fitShape, CV_32F);
270 grid = outs[0];
271 } else {
272 std::vector<UMat>& uouts = outputs_arr.getUMatVecRef();
273 uouts.resize(1);
274 uouts[0].fit(fitShape, CV_32F);
275 grid = uouts[0].getMat(ACCESS_WRITE);
276 }
277
278 float xs, xd, ys, yd, zs=0.f, zd=0.f;
279 if (align_corners) {
280 xs = W > 1 ? 2.f / float(W - 1) : 0.f;
281 xd = -1.f;
282 ys = H > 1 ? 2.f / float(H - 1) : 0.f;
283 yd = -1.f;
284 if (is3d) {
285 zs = D > 1 ? 2.f / float(D - 1) : 0.f;
286 zd = -1.f;
287 }
288 } else {
289 xs = 2.f / float(W);
290 xd = xs * 0.5f - 1.f;
291 ys = 2.f / float(H);
292 yd = ys * 0.5f - 1.f;
293 if (is3d) {
294 zs = 2.f / float(D);
295 zd = zs * 0.5f - 1.f;
296 }
297 }
298
299 int depth = theta.depth();
300 if (!is3d){
301 switch (depth) {
302 case CV_32F: computeGrid2D<float>(theta, grid, N, H, W, xs, xd, ys, yd); break;
303 case CV_64F: computeGrid2D<double>(theta, grid, N, H, W, xs, xd, ys, yd); break;
304 case CV_16F: computeGrid2D<hfloat>(theta, grid, N, H, W, xs, xd, ys, yd); break;
305 case CV_16BF: computeGrid2D<bfloat>(theta, grid, N, H, W, xs, xd, ys, yd); break;
306 default: CV_Error(cv::Error::BadDepth, "AffineGrid: unsupported theta depth");
307 }
308 }
309 else{
310 switch (depth) {
311 case CV_32F: computeGrid3D<float>(theta, grid, N, D, H, W, xs, xd, ys, yd, zs, zd); break;
312 case CV_64F: computeGrid3D<double>(theta, grid, N, D, H, W, xs, xd, ys, yd, zs, zd); break;
313 case CV_16F: computeGrid3D<hfloat>(theta, grid, N, D, H, W, xs, xd, ys, yd, zs, zd); break;
314 case CV_16BF: computeGrid3D<bfloat>(theta, grid, N, D, H, W, xs, xd, ys, yd, zs, zd); break;
315 default: CV_Error(cv::Error::BadDepth, "AffineGrid: unsupported theta depth");
316 }
317 }
318 }
319};
