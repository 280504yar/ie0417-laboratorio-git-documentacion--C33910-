83 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
84 const int requiredOutputs,
85 std::vector<MatShape> &outputs,
86 std::vector<MatShape> &internals) const CV_OVERRIDE
87 {
88 CV_Assert(inputs.size() > 0);
89 outputs.resize(1, inputs[0]);
90 int cAxis = normalize_axis(axis, inputs[0]);
91
92 int axisSum = 0;
93 for (size_t i = 0; i < inputs.size(); i++)
94 {
95 MatShape curShape = inputs[i];
96
97 if (padding)
98 {
99 for (int curAxis = 0; curAxis < outputs[0].size(); curAxis++)
100 {
101 outputs[0][curAxis] = std::max(outputs[0][curAxis], curShape[curAxis]);
102 }
103 }
104 else
105 {
106 CV_Assert(curShape.size() == outputs[0].size());
107 for (int curAxis = 0; curAxis < outputs[0].size(); curAxis++)
108 {
109 if (curAxis != cAxis && outputs[0][curAxis] != curShape[curAxis])
110 CV_Error(Error::StsBadSize, "Inconsistent shape for ConcatLayer");
111 }
112 }
113
114 axisSum += curShape.dims >= cAxis ? curShape[cAxis] : 1;
115 }
116 outputs[0].dims = std::max(outputs[0].dims, 1);
117 outputs[0][cAxis] = axisSum;
118 return false;
119 }

161 template <class T>
162 class ChannelConcatInvoker : public ParallelLoopBody
163 {
164 public:
165 std::vector<Mat>* inputs;
166 Mat* output;
167 int nstripes;
168 std::vector<const T*> chptrs;
169
170 static void run(std::vector<Mat>& inputs, Mat& output, int nstripes)
171 {
172 ChannelConcatInvoker cc;
173 cc.inputs = &inputs;
174 cc.output = &output;
175 cc.nstripes = nstripes;
176
177 size_t i, ninputs = inputs.size();
178 int nchannels = 0, batchsz = output.size[0];
179 for( i = 0; i < ninputs; i++ )
180 {
181 Mat& inp = inputs[i];
182 CV_Assert( inp.isContinuous() && (inp.type() == CV_32F || inp.type() == CV_16F || inp.type() == CV_8S) &&
183 inp.dims == 4 && inp.size[0] == output.size[0] &&
184 inp.size[2] == output.size[2] &&
185 inp.size[3] == output.size[3] );
186 nchannels += inp.size[1];
187 }
188 CV_Assert( nchannels == output.size[1] );
189 CV_Assert( output.isContinuous() && (output.type() == CV_32F || output.type() == CV_16F || output.type() == CV_8S) );
190
191 cc.chptrs.resize(nchannels*batchsz);
192
193 int ofs = 0;
194 for( i = 0; i < ninputs; i++)
195 {
196 Mat& inp = inputs[i];
197 for( int j = 0; j < batchsz; j++ )
198 for( int k = 0; k < inp.size[1]; k++ )
199 {
200 const T* ptr = inp.ptr<T>(j, k);
201 cc.chptrs[ofs + j*nchannels + k] = ptr;
202 }
203 ofs += inp.size[1];
204 }
205
206 parallel_for_(Range(0, nstripes), cc, nstripes);
207 }
208
209 ChannelConcatInvoker() : inputs(0), output(0), nstripes(0) {}
210
211 void operator()(const Range& r) const CV_OVERRIDE
212 {
213 size_t planeSize = (size_t)output->size[2]*output->size[3];
214 size_t nch = chptrs.size();
215 size_t total = nch*planeSize;
216 size_t stripeSize = (total + nstripes - 1)/nstripes;
217 size_t stripeStart = r.start*stripeSize;
218 size_t stripeEnd = std::min(total, r.end*stripeSize);
219 const T** ptrs = (const T**)&chptrs[0];
220 T* outptr = output->ptr<T>();
221 size_t blockSize0 = 1 << 16;
222
223 for( size_t ofs0 = stripeStart; ofs0 < stripeEnd; )
224 {
225 size_t ch = ofs0/planeSize;
226 size_t ofs = ofs0 - ch*planeSize;
227 size_t blockSize = std::min(blockSize0, planeSize - ofs);
228 memcpy(outptr + ofs0, ptrs[ch] + ofs, blockSize*sizeof(outptr[0]));
229 ofs0 += blockSize;
230 }
231 }
232 };

234#ifdef HAVE_OPENCL
235 bool forward_ocl(InputArrayOfArrays inps, OutputArrayOfArrays outs, OutputArrayOfArrays internals)
236 {
237 std::vector<UMat> inputs;
238 std::vector<UMat> outputs;
239 inps.getUMatVector(inputs);
240 outs.getUMatVector(outputs);
241
242 int cAxis = normalize_axis(axis, inputs[0].dims);
243 if (padding)
244 return false;
245
246 int bottom_concat_axis;
247 int concat_size = (int)total(shape(inputs[0]), cAxis + 1);
248 int top_concat_axis = outputs[0].size[cAxis];
249 int num_concats = (int)total(shape(inputs[0]), 0, cAxis);
250 int offset_concat_axis = 0;
251 UMat& outMat = outputs[0];
252 String matType = matTypeToOclType(inputs[0].type());
253 String buildopt = " -DDtype=" + matType;
254 String kname = "concat_" + matType;
255
256 for (size_t i = 0; i < inputs.size(); i++)
257 {
258 ocl::Kernel kernel(kname.c_str(), ocl::dnn::concat_oclsrc, buildopt);
259 if (kernel.empty())
260 return false;
261
262 UMat& inpMat = inputs[i];
263 bottom_concat_axis = inputs[i].size[cAxis];
264 size_t nthreads = inputs[i].total();
265
266 kernel.set(0, (int)nthreads);
267 kernel.set(1, ocl::KernelArg::PtrReadOnly(inpMat));
268 kernel.set(2, (int)num_concats);
269 kernel.set(3, (int)concat_size);
270 kernel.set(4, (int)top_concat_axis);
271 kernel.set(5, (int)bottom_concat_axis);
272 kernel.set(6, (int)offset_concat_axis);
273 kernel.set(7, ocl::KernelArg::PtrWriteOnly(outMat));
274
275 if (!kernel.run(1, &nthreads, NULL, false))
276 return false;
277
278 offset_concat_axis += bottom_concat_axis;
279 }
280
281 return true;
282 }
283#endif

285 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
286 {
287 CV_TRACE_FUNCTION();
288 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
289
290 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
291 forward_ocl(inputs_arr, outputs_arr, internals_arr))
292
293 std::vector<Mat> inputs, outputs;
294 inputs_arr.getMatVector(inputs);
295 outputs_arr.getMatVector(outputs);
296
297 int cAxis = normalize_axis(axis, inputs[0].dims);
298 Mat& outMat = outputs[0];
299
300 if (padding)
301 outMat.setTo(paddingValue);
302
303 if(cAxis == 1 && outMat.dims == 4 && !padding && (inputs[0].depth() == CV_32F || inputs[0].depth() == CV_8S))
304 {
305 int nstripes = getNumThreads();
306 if (outMat.type() == CV_8S)
307 ChannelConcatInvoker<int8_t>::run(inputs, outMat, nstripes);
308 else
309 ChannelConcatInvoker<float>::run(inputs, outMat, nstripes);
310 }
311 else
312 {
313 std::vector<Range> ranges(outputs[0].dims, Range::all());
314
315 ranges[cAxis].start = 0;
316 for (size_t i = 0; i < inputs.size(); i++)
317 {
318 if (inputs[i].empty())
319 continue;
320 ranges[cAxis].end = ranges[cAxis].start + inputs[i].size[cAxis];
321 for (int j = 0; j < outMat.dims; ++j)
322 {
323 if (j == cAxis) continue;
324 ranges[j].start = (outMat.size[j] - inputs[i].size[j]) / 2;
325 ranges[j].end = ranges[j].start + inputs[i].size[j];
326 }
327 inputs[i].copyTo(outMat(&ranges[0]));
328 ranges[cAxis].start = ranges[cAxis].end;
329 }
330 }
331 }

385#ifdef HAVE_DNN_NGRAPH
386 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> >& inputs,
387 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
388 {
389 const int numDims = nodes[0].dynamicCast<InfEngineNgraphNode>()->node.get_shape().size();
390 const int cAxis = normalize_axis(axis, numDims);
391 std::vector<size_t> maxDims(numDims, 0);
392
393 CV_Assert(inputs.size() == nodes.size());
394 ov::OutputVector inp_nodes;
395 for (int i = 0; i < nodes.size(); ++i)
396 {
397 auto inp = nodes[i].dynamicCast<InfEngineNgraphNode>()->node;
398 inp_nodes.push_back(inp);
399
400 std::vector<size_t> inpShape = inp.get_shape();
401 for (int i = 0; i < numDims; ++i)
402 maxDims[i] = std::max(maxDims[i], inpShape[i]);
403 }
404 for (int i = 0; i < inp_nodes.size(); ++i)
405 {
406 bool needPadding = false;
407 std::vector<size_t> inpShape = inp_nodes[i].get_shape();
408 std::vector<int64_t> begins(inpShape.size(), 0), ends(inpShape.size(), 0);
409 for (int j = 0; j < inpShape.size(); ++j)
410 {
411 if (j != cAxis && inpShape[j] != maxDims[j])
412 {
413 needPadding = true;
414 begins[j] = static_cast<int64_t>((maxDims[j] - inpShape[j]) / 2);
415 ends[j] = static_cast<int64_t>(maxDims[j] - inpShape[j] - begins[j]);
416 }
417 }
418 if (needPadding)
419 {
420 inp_nodes[i] = std::make_shared<ov::op::v1::Pad>(
421 inp_nodes[i],
422 std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{begins.size()}, begins.data()),
423 std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{ends.size()}, ends.data()),
424 ov::op::PadMode::CONSTANT);
425 }
426 }
427 auto concat = std::make_shared<ov::op::v0::Concat>(inp_nodes, cAxis);
428 return Ptr<BackendNode>(new InfEngineNgraphNode(concat));
429 }
430#endif // HAVE_DNN_NGRAPH

432#ifdef HAVE_TIMVX
433 virtual Ptr<BackendNode> initTimVX(void* timVXInfo_,
434 const std::vector<Ptr<BackendWrapper> > &inputsWrapper,
435 const std::vector<Ptr<BackendWrapper> > &outputsWrapper,
436 bool isLast) CV_OVERRIDE
437 {
438 // tvGraph Initialization.
439 auto timVxInfo = reinterpret_cast<TimVXInfo *>(timVXInfo_);
440 CV_Assert(timVxInfo);
441 Ptr<TimVXGraph> tvGraph = timVxInfo->getGraph();
442 CV_Assert(tvGraph);
443 Ptr<tim::vx::Graph> graph = tvGraph->graph;
444
445 Ptr<TimVXBackendWrapper> inputWrapper = inputsWrapper[0].dynamicCast<TimVXBackendWrapper>();
446 // convert axis from OpenCV NCHW toTimVX WHCN.
447 Mat blob0 = inputWrapper->getMat();
448
449 // TODO! support TimVX 5 dim in future.
450 if(blob0.dims >4)
451 return Ptr<TimVXBackendNode>();
452
453 int cAxis = normalize_axis(axis, blob0.dims);
454 int tvAxis = blob0.dims - 1 - cAxis;
455 CV_Assert(tvAxis>= 0);
456 std::vector<int> inputsIndex, outputsIndex;
457 int input_index = -1, output_index = -1;
458
459 // Input
460 Ptr<tim::vx::Quantization> tvQuant = Ptr<tim::vx::Quantization>(
461 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, scale, zeropoint));
462
463 for (int i = 0; i<inputsWrapper.size(); i++)
464 {
465 inputWrapper = inputsWrapper[i].dynamicCast<TimVXBackendWrapper>();
466 if (inputWrapper->isTensor())
467 {
468 input_index = tvGraph->getTensorIndex(inputWrapper->getTensor());
469 if (input_index == -1)
470 {
471 // Copy To New inputWrapper
472 Mat tmp = inputWrapper->getMat();
473 inputWrapper = Ptr<TimVXBackendWrapper>(new TimVXBackendWrapper(tmp));
474 }
475 }
476
477 if (!inputWrapper->isTensor())
478 {
479 inputWrapper->createTensor(graph,tim::vx::TensorAttribute::INPUT, tvQuant);
480 input_index = tvGraph->addWrapper(inputWrapper);
481 }
482 inputsIndex.push_back(input_index);
483 }
484
485 //Output
486 CV_Assert(outputsWrapper.size() == 1);
487 Ptr<TimVXBackendWrapper> outputWrapper = outputsWrapper[0].dynamicCast<TimVXBackendWrapper>();
488
489 if (isLast)
490 {
491 auto shapeType = getShapeTypeFromMat(outputWrapper->getMat());
492
493 // For Graph Output tensor, we need to set tensor shape before createTensor().
494 outputWrapper->setTensorShape(shapeType);
495 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::OUTPUT, tvQuant);
496 }
497 else
498 {
499 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::TRANSIENT, tvQuant);
500 }
501 output_index = tvGraph->addWrapper(outputWrapper);
502 outputsIndex.push_back(output_index);
503
504 std::shared_ptr<tim::vx::Operation> tvConcate = graph->CreateOperation<tim::vx::ops::Concat>(tvAxis, inputsWrapper.size());
505
506 Ptr<TimVXBackendNode> tvBackendNode = new TimVXBackendNode(tvGraph, tvConcate, inputsIndex, outputsIndex);
507
508 return tvBackendNode;
509 }
510#endif // HAVE_TIMVX
