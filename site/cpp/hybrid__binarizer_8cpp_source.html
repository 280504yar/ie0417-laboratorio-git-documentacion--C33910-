16// This class uses 5x5 blocks to compute local luminance, where each block is 8x8 pixels.
17// So this is the smallest dimension in each axis we can accept.
18constexpr static int BLOCK_SIZE_POWER = 3;
19constexpr static int BLOCK_SIZE = 1 << BLOCK_SIZE_POWER; // ...0100...00
20constexpr static int BLOCK_SIZE_MASK = BLOCK_SIZE - 1; // ...0011...11
21constexpr static int MINIMUM_DIMENSION = BLOCK_SIZE * 5;
22constexpr static int MIN_DYNAMIC_RANGE = 24;

24void
25calculateThresholdForBlock(const std::vector<uchar> &luminances, int sub_width, int sub_height, int width, int height,
26 const Mat &black_points, Mat &dst)
27{
28 int maxYOffset = height - BLOCK_SIZE;
29 int maxXOffset = width - BLOCK_SIZE;
30 for (int y = 0; y < sub_height; y++)
31 {
32 int yoffset = y << BLOCK_SIZE_POWER;
33 if (yoffset > maxYOffset)
34 {
35 yoffset = maxYOffset;
36 }
37 int top = CLAMP(y, 2, sub_height - 3);
38 for (int x = 0; x < sub_width; x++)
39 {
40 int xoffset = x << BLOCK_SIZE_POWER;
41 if (xoffset > maxXOffset)
42 {
43 xoffset = maxXOffset;
44 }
45 int left = CLAMP(x, 2, sub_width - 3);
46 int sum = 0;
47 const auto *black_row = black_points.ptr<uchar>(top - 2);
48 for (int z = 0; z <= 4; z++)
49 {
50 sum += black_row[left - 2] + black_row[left - 1] + black_row[left] + black_row[left + 1] +
51 black_row[left + 2];
52 black_row += black_points.cols;
53 }
54 int average = sum / 25;
55 int temp_y = 0;
56
57 auto *ptr = dst.ptr<uchar>(yoffset, xoffset);
58 for (int offset = yoffset * width + xoffset; temp_y < 8; offset += width)
59 {
60 for (int temp_x = 0; temp_x < 8; ++temp_x)
61 {
62 *(ptr + temp_x) = (luminances[offset + temp_x] & 255) <= average ? 0 : 255;
63 }
64 ++temp_y;
65 ptr += width;
66 }
67 }
68 }

70}

72Mat calculateBlackPoints(std::vector<uchar> luminances, int sub_width, int sub_height, int width, int height)
73{
74 int maxYOffset = height - BLOCK_SIZE;
75 int maxXOffset = width - BLOCK_SIZE;
76 Mat black_points(Size(sub_width, sub_height), CV_8UC1);
77 for (int y = 0; y < sub_height; y++)
78 {
79 int yoffset = y << BLOCK_SIZE_POWER;
80 if (yoffset > maxYOffset)
81 {
82 yoffset = maxYOffset;
83 }
84 for (int x = 0; x < sub_width; x++)
85 {
86 int xoffset = x << BLOCK_SIZE_POWER;
87 if (xoffset > maxXOffset)
88 {
89 xoffset = maxXOffset;
90 }
91 int sum = 0;
92 int min = 0xFF;
93 int max = 0;
94 for (int yy = 0, offset = yoffset * width + xoffset; yy < BLOCK_SIZE; yy++, offset += width)
95 {
96 for (int xx = 0; xx < BLOCK_SIZE; xx++)
97 {
98 int pixel = luminances[offset + xx] & 0xFF;
99 sum += pixel;
100 // still looking for good contrast
101 if (pixel < min)
102 {
103 min = pixel;
104 }
105 if (pixel > max)
106 {
107 max = pixel;
108 }
109 }
110 // short-circuit min/max tests once dynamic range is met
111 if (max - min > MIN_DYNAMIC_RANGE)
112 {
113 // finish the rest of the rows quickly
114 for (yy++, offset += width; yy < BLOCK_SIZE; yy++, offset += width)
115 {
116 for (int xx = 0; xx < BLOCK_SIZE; xx++)
117 {
118 sum += luminances[offset + xx] & 0xFF;
119 }
120 }
121 }
122 }
123
124 // The default estimate is the average of the values in the block.
125 int average = sum >> (BLOCK_SIZE_POWER * 2);
126 if (max - min <= MIN_DYNAMIC_RANGE)
127 {
128 // If variation within the block is low, assume this is a block with only light or only
129 // dark pixels. In that case we do not want to use the average, as it would divide this
130 // low contrast area into black and white pixels, essentially creating data out of noise.
131 //
132 // The default assumption is that the block is light/background. Since no estimate for
133 // the level of dark pixels exists locally, use half the min for the block.
134 average = min / 2;
135
136 if (y > 0 && x > 0)
137 {
138 // Correct the "white background" assumption for blocks that have neighbors by comparing
139 // the pixels in this block to the previously calculated black points. This is based on
140 // the fact that dark barcode symbology is always surrounded by some amount of light
141 // background for which reasonable black point estimates were made. The bp estimated at
142 // the boundaries is used for the interior.
143
144 // The (min < bp) is arbitrary but works better than other heuristics that were tried.
145 int averageNeighborBlackPoint =
146 (black_points.at<uchar>(y - 1, x) + (2 * black_points.at<uchar>(y, x - 1)) +
147 black_points.at<uchar>(y - 1, x - 1)) / 4;
148 if (min < averageNeighborBlackPoint)
149 {
150 average = averageNeighborBlackPoint;
151 }
152 }
153 }
154 black_points.at<uchar>(y, x) = (uchar) average;
155 }
156 }
157 return black_points;
158
159}

162void hybridBinarization(const Mat &src, Mat &dst)
163{
164 int width = src.cols;
165 int height = src.rows;
166
167 if (width >= MINIMUM_DIMENSION && height >= MINIMUM_DIMENSION)
168 {
169 std::vector<uchar> luminances(src.begin<uchar>(), src.end<uchar>());
170
171 int sub_width = width >> BLOCK_SIZE_POWER;
172 if ((width & BLOCK_SIZE_MASK) != 0)
173 {
174 sub_width++;
175 }
176
177 int sub_height = height >> BLOCK_SIZE_POWER;
178 if ((height & BLOCK_SIZE_MASK) != 0)
179 {
180 sub_height++;
181 }
182
183 Mat black_points = calculateBlackPoints(luminances, sub_width, sub_height, width, height);
184
185 dst.create(src.size(), src.type());
186 calculateThresholdForBlock(luminances, sub_width, sub_height, width, height, black_points, dst);
187 }
188 else
189 {
190 threshold(src, dst, 155, 255, THRESH_OTSU + THRESH_BINARY);
191 }
192
193}
194}
