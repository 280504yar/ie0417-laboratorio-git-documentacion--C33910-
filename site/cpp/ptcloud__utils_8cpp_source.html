13void
14getPointsMatFromInputArray(InputArray input_pts, Mat &mat, int arrangement_of_points, bool clone_data)
15{
16 CV_Check(input_pts.dims(), input_pts.dims() < 3,
17 "Only support data with dimension less than 3.");
18
19 // Guaranteed data can construct N×3 point clouds
20 int rows = input_pts.rows(), cols = input_pts.cols(), channels = input_pts.channels();
21 size_t total = rows * cols * channels;
22 CV_Check(total, total % 3 == 0,
23 "total = input_pts.rows() * input_pts.cols() * input_pts.channels() must be an integer multiple of 3");
24
33 int ori_arrangement = (channels == 1 && rows == 3 && cols != 3) ? 1 : 0;
34
35 // Convert to single channel without copying the data.
36 mat = ori_arrangement == 0 ? input_pts.getMat().reshape(1, (int) (total / 3))
37 : input_pts.getMat();
38
39 if (ori_arrangement != arrangement_of_points)
40 {
41 Mat tmp;
42 transpose(mat, tmp);
43 swap(mat, tmp);
44 }
45
46 if (mat.type() != CV_32F)
47 {
48 Mat tmp;
49 mat.convertTo(tmp, CV_32F); // Use float to store data
50 swap(mat, tmp);
51 }
52
53 if (clone_data || (!mat.isContinuous()))
54 {
55 mat = mat.clone();
56 }
57
58}

60void
61copyPointDataByIdxs(const Mat &src, Mat &dst, const std::vector<int> &idxs, int dst_size,
62 int arrangement_of_points, bool is_parallel)
63{
64 CV_CheckDepthEQ(src.depth(), CV_32F, "Data with only depth CV_32F are supported");
65 CV_CheckChannelsEQ(src.channels(), 1, "Data with only one channel are supported");
66
67 const int idxs_size = (int) idxs.size();
68 dst_size = (dst_size < 0 || dst_size > idxs_size) ? idxs_size : dst_size;
69
70 if (arrangement_of_points == 1)
71 {
72 dst = Mat(3, dst_size, CV_32F);
73 const int src_size = src.rows * src.cols / 3;
74 const float *const src_ptr_x = (float *) src.data;
75 const float *const src_ptr_y = src_ptr_x + src_size;
76 const float *const src_ptr_z = src_ptr_y + src_size;
77
78 float *const dst_ptr_x = (float *) dst.data;
79 float *const dst_ptr_y = dst_ptr_x + dst_size;
80 float *const dst_ptr_z = dst_ptr_y + dst_size;
81
82 int i = 0;
83 if (is_parallel) {
84 int number_of_threads = std::max(getNumThreads(), 1);
85 int block_size = dst_size / number_of_threads;
86 parallel_for_(Range(0, number_of_threads), [&](const Range &range) {
87 int _start = range.start * block_size;
88 int _end = _start + block_size;
89 for (int _i = _start; _i < _end; ++_i) {
90 int _src_idx = idxs[_i];
91 dst_ptr_x[_i] = src_ptr_x[_src_idx];
92 dst_ptr_y[_i] = src_ptr_y[_src_idx];
93 dst_ptr_z[_i] = src_ptr_z[_src_idx];
94 }
95 });
96
97 i = block_size * number_of_threads;
98 }
99
100 for (; i < dst_size; ++i)
101 {
102 int src_idx = idxs[i];
103 dst_ptr_x[i] = src_ptr_x[src_idx];
104 dst_ptr_y[i] = src_ptr_y[src_idx];
105 dst_ptr_z[i] = src_ptr_z[src_idx];
106 }
107
108 }
109 else if (arrangement_of_points == 0)
110 {
111 dst = Mat(dst_size, 3, CV_32F);
112
113 const float *const src_ptr = (float *) src.data;
114 float *const dst_ptr = (float *) dst.data;
115
116 int i = 0;
117 if (is_parallel) {
118 int number_of_threads = std::max(getNumThreads(), 1);
119 int block_size = dst_size / number_of_threads;
120 parallel_for_(Range(0, number_of_threads), [&](const Range &range) {
121 int _start = range.start * block_size;
122 int _end = _start + block_size;
123 for (int _i = _start; _i < _end; ++_i) {
124 const float *_src_ptr_base = src_ptr + 3 * idxs[_i];
125 float *_dst_ptr_base = dst_ptr + 3 * _i;
126 _dst_ptr_base[0] = _src_ptr_base[0];
127 _dst_ptr_base[1] = _src_ptr_base[1];
128 _dst_ptr_base[2] = _src_ptr_base[2];
129 }
130 });
131
132 i = block_size * number_of_threads;
133 }
134
135 for (; i < dst_size; ++i)
136 {
137 const float *src_ptr_base = src_ptr + 3 * idxs[i];
138 float *dst_ptr_base = dst_ptr + 3 * i;
139 dst_ptr_base[0] = src_ptr_base[0];
140 dst_ptr_base[1] = src_ptr_base[1];
141 dst_ptr_base[2] = src_ptr_base[2];
142 }
143 }
144
145}

149 int arrangement_of_points, bool is_parallel)

152 std::vector<int> idxs;

154 {
