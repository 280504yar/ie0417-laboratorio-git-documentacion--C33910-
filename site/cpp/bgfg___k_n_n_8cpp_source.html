347static inline void
348 _cvUpdatePixelBackgroundNP(int x_idx, const uchar* data, int nchannels, int m_nN,
349 uchar* m_aModel,
350 uchar* m_nNextLongUpdate,
351 uchar* m_nNextMidUpdate,
352 uchar* m_nNextShortUpdate,
353 uchar* m_aModelIndexLong,
354 uchar* m_aModelIndexMid,
355 uchar* m_aModelIndexShort,
356 int m_nLongCounter,
357 int m_nMidCounter,
358 int m_nShortCounter,
359 uchar include
360 )
361{
362 // hold the offset
363 int ndata=1+nchannels;
364 long offsetLong = ndata * (m_aModelIndexLong[x_idx] + m_nN * 2);
365 long offsetMid = ndata * (m_aModelIndexMid[x_idx] + m_nN * 1);
366 long offsetShort = ndata * (m_aModelIndexShort[x_idx]);
367
368 // Long update?
369 if (m_nNextLongUpdate[x_idx] == m_nLongCounter)
370 {
371 // add the oldest pixel from Mid to the list of values (for each color)
372 memcpy(&m_aModel[offsetLong],&m_aModel[offsetMid],ndata*sizeof(unsigned char));
373 // increase the index
374 m_aModelIndexLong[x_idx] = (m_aModelIndexLong[x_idx] >= (m_nN-1)) ? 0 : (m_aModelIndexLong[x_idx] + 1);
375 };
376
377 // Mid update?
378 if (m_nNextMidUpdate[x_idx] == m_nMidCounter)
379 {
380 // add this pixel to the list of values (for each color)
381 memcpy(&m_aModel[offsetMid],&m_aModel[offsetShort],ndata*sizeof(unsigned char));
382 // increase the index
383 m_aModelIndexMid[x_idx] = (m_aModelIndexMid[x_idx] >= (m_nN-1)) ? 0 : (m_aModelIndexMid[x_idx] + 1);
384 };
385
386 // Short update?
387 if (m_nNextShortUpdate[x_idx] == m_nShortCounter)
388 {
389 // add this pixel to the list of values (for each color)
390 memcpy(&m_aModel[offsetShort],data,nchannels*sizeof(unsigned char));
391 //set the include flag
392 m_aModel[offsetShort+nchannels]=include;
393 // increase the index
394 m_aModelIndexShort[x_idx] = (m_aModelIndexShort[x_idx] >= (m_nN-1)) ? 0 : (m_aModelIndexShort[x_idx] + 1);
395 };
396}

398static inline int
399 _cvCheckPixelBackgroundNP(const uchar* data, int nchannels,
400 int m_nN,
401 uchar* m_aModel,
402 float m_fTb,
403 int m_nkNN,
404 float tau,
405 bool m_bShadowDetection,
406 uchar& include)
407{
408 int Pbf = 0; // the total probability that this pixel is background
409 int Pb = 0; //background model probability
410 float dData[CV_CN_MAX];
411
412 //uchar& include=data[nchannels];
413 include=0;//do we include this pixel into background model?
414
415 int ndata=nchannels+1;
416 // now increase the probability for each pixel
417 for (int n = 0; n < m_nN*3; n++)
418 {
419 uchar* mean_m = &m_aModel[n*ndata];
420
421 //calculate difference and distance
422 float dist2;
423
424 if( nchannels == 3 )
425 {
426 dData[0] = (float)mean_m[0] - data[0];
427 dData[1] = (float)mean_m[1] - data[1];
428 dData[2] = (float)mean_m[2] - data[2];
429 dist2 = dData[0]*dData[0] + dData[1]*dData[1] + dData[2]*dData[2];
430 }
431 else
432 {
433 dist2 = 0.f;
434 for( int c = 0; c < nchannels; c++ )
435 {
436 dData[c] = (float)mean_m[c] - data[c];
437 dist2 += dData[c]*dData[c];
438 }
439 }
440
441 if (dist2<m_fTb)
442 {
443 Pbf++;//all
444 //background only
445 //if(m_aModel[subPosPixel + nchannels])//indicator
446 if(mean_m[nchannels])//indicator
447 {
448 Pb++;
449 if (Pb >= m_nkNN)//Tb
450 {
451 include=1;//include
452 return 1;//background ->exit
453 };
454 }
455 };
456 };
457
458 //include?
459 if (Pbf>=m_nkNN)//m_nTbf)
460 {
461 include=1;
462 }
463
464 int Ps = 0; // the total probability that this pixel is background shadow
465 // Detected as moving object, perform shadow detection
466 if (m_bShadowDetection)
467 {
468 for (int n = 0; n < m_nN*3; n++)
469 {
470 //long subPosPixel = posPixel + n*ndata;
471 uchar* mean_m = &m_aModel[n*ndata];
472
473 if(mean_m[nchannels])//check only background
474 {
475 float numerator = 0.0f;
476 float denominator = 0.0f;
477 for( int c = 0; c < nchannels; c++ )
478 {
479 numerator += (float)data[c] * mean_m[c];
480 denominator += (float)mean_m[c] * mean_m[c];
481 }
482
483 // no division by zero allowed
484 if( denominator == 0 )
485 return 0;
486
487 // if tau < a < 1 then also check the color distortion
488 if( numerator <= denominator && numerator >= tau*denominator )
489 {
490 float a = numerator / denominator;
491 float dist2a = 0.0f;
492
493 for( int c = 0; c < nchannels; c++ )
494 {
495 float dD= a*mean_m[c] - data[c];
496 dist2a += dD*dD;
497 }
498
499 if (dist2a<m_fTb*a*a)
500 {
501 Ps++;
502 if (Ps >= m_nkNN)//shadow
503 return 2;
504 };
505 };
506 };
507 };
508 }
509 return 0;
510}

555 void operator()(const Range& range) const CV_OVERRIDE
556 {
557 int y0 = range.start, y1 = range.end;
558 int ncols = src->cols, nchannels = src->channels();
559 int ndata=nchannels+1;
560
561 for ( int y = y0; y < y1; y++ )
562 {
563 const uchar* data = src->ptr(y);
564 uchar* m_aModel = m_aModel0 + ncols*m_nN*3*ndata*y;
565 uchar* m_nNextLongUpdate = m_nNextLongUpdate0 + ncols*y;
566 uchar* m_nNextMidUpdate = m_nNextMidUpdate0 + ncols*y;
567 uchar* m_nNextShortUpdate = m_nNextShortUpdate0 + ncols*y;
568 uchar* m_aModelIndexLong = m_aModelIndexLong0 + ncols*y;
569 uchar* m_aModelIndexMid = m_aModelIndexMid0 + ncols*y;
570 uchar* m_aModelIndexShort = m_aModelIndexShort0 + ncols*y;
571 uchar* mask = dst->ptr(y);
572
573 for ( int x = 0; x < ncols; x++ )
574 {
575
576 //update model+ background subtract
577 uchar include=0;
578 int result= _cvCheckPixelBackgroundNP(data, nchannels,
579 m_nN, m_aModel, m_fTb,m_nkNN, m_fTau,m_bShadowDetection,include);
580
581 _cvUpdatePixelBackgroundNP(x,data,nchannels,
582 m_nN, m_aModel,
583 m_nNextLongUpdate,
584 m_nNextMidUpdate,
585 m_nNextShortUpdate,
586 m_aModelIndexLong,
587 m_aModelIndexMid,
588 m_aModelIndexShort,
589 m_nLongCounter,
590 m_nMidCounter,
591 m_nShortCounter,
592 include
593 );
594 // Check that foreground mask exists
595 if (!knownForegroundMask.empty()) {
596 // If input mask states pixel is foreground
597 if (knownForegroundMask.at<uchar>(y, x) > 0)
598 {
599 mask[x] = 255; // ensure output mask marks this pixel as FG
600 data += nchannels;
601 m_aModel += m_nN*3*ndata;
602 continue;
603 }
604 }
605 switch (result)
606 {
607 case 0:
608 //foreground
609 mask[x] = 255;
610 break;
611 case 1:
612 //background
613 mask[x] = 0;
614 break;
615 case 2:
616 //shadow
617 mask[x] = m_nShadowDetection;
618 break;
619 }
620 data += nchannels;
621 m_aModel += m_nN*3*ndata;
622 }
623 }
624 }

648bool BackgroundSubtractorKNNImpl::ocl_apply(InputArray _image, OutputArray _fgmask, double learningRate)
649{
650 bool needToInitialize = nframes == 0 || learningRate >= 1 || _image.size() != frameSize || _image.type() != frameType;
651
652 if( needToInitialize )
653 initialize(_image.size(), _image.type());
654
655 ++nframes;
656 learningRate = learningRate >= 0 && nframes > 1 ? learningRate : 1./std::min( 2*nframes, history );
657 CV_Assert(learningRate >= 0);
658
659 _fgmask.create(_image.size(), CV_8U);
660 UMat fgmask = _fgmask.getUMat();
661
662 UMat frame = _image.getUMat();
663
664 //recalculate update rates - in case alpha is changed
665 // calculate update parameters (using alpha)
666 int Kshort,Kmid,Klong;
667 //approximate exponential learning curve
668 Kshort=(int)(log(0.7)/log(1-learningRate))+1;//Kshort
669 Kmid=(int)(log(0.4)/log(1-learningRate))-Kshort+1;//Kmid
670 Klong=(int)(log(0.1)/log(1-learningRate))-Kshort-Kmid+1;//Klong
671
672 //refresh rates
673 int nShortUpdate = (Kshort/nN)+1;
674 int nMidUpdate = (Kmid/nN)+1;
675 int nLongUpdate = (Klong/nN)+1;
676
677 int idxArg = 0;
678 idxArg = kernel_apply.set(idxArg, ocl::KernelArg::ReadOnly(frame));
679 idxArg = kernel_apply.set(idxArg, ocl::KernelArg::PtrReadOnly(u_nNextLongUpdate));
680 idxArg = kernel_apply.set(idxArg, ocl::KernelArg::PtrReadOnly(u_nNextMidUpdate));
681 idxArg = kernel_apply.set(idxArg, ocl::KernelArg::PtrReadOnly(u_nNextShortUpdate));
682 idxArg = kernel_apply.set(idxArg, ocl::KernelArg::PtrReadWrite(u_aModelIndexLong));
683 idxArg = kernel_apply.set(idxArg, ocl::KernelArg::PtrReadWrite(u_aModelIndexMid));
684 idxArg = kernel_apply.set(idxArg, ocl::KernelArg::PtrReadWrite(u_aModelIndexShort));
685 idxArg = kernel_apply.set(idxArg, ocl::KernelArg::PtrReadWrite(u_flag));
686 idxArg = kernel_apply.set(idxArg, ocl::KernelArg::PtrReadWrite(u_sample));
687 idxArg = kernel_apply.set(idxArg, ocl::KernelArg::WriteOnlyNoSize(fgmask));
688
689 idxArg = kernel_apply.set(idxArg, nLongCounter);
690 idxArg = kernel_apply.set(idxArg, nMidCounter);
691 idxArg = kernel_apply.set(idxArg, nShortCounter);
692 idxArg = kernel_apply.set(idxArg, fTb);
693 idxArg = kernel_apply.set(idxArg, nkNN);
694 idxArg = kernel_apply.set(idxArg, fTau);
695 if (bShadowDetection)
696 kernel_apply.set(idxArg, nShadowDetection);
697
698 size_t globalsize[2] = {(size_t)frame.cols, (size_t)frame.rows};
699 if(!kernel_apply.run(2, globalsize, NULL, true))
700 return false;
701
702 nShortCounter++;//0,1,...,nShortUpdate-1
703 nMidCounter++;
704 nLongCounter++;
705 if (nShortCounter >= nShortUpdate)
706 {
707 nShortCounter = 0;
708 randu(u_nNextShortUpdate, Scalar::all(0), Scalar::all(nShortUpdate));
709 }
710 if (nMidCounter >= nMidUpdate)
711 {
712 nMidCounter = 0;
713 randu(u_nNextMidUpdate, Scalar::all(0), Scalar::all(nMidUpdate));
714 }
715 if (nLongCounter >= nLongUpdate)
716 {
717 nLongCounter = 0;
718 randu(u_nNextLongUpdate, Scalar::all(0), Scalar::all(nLongUpdate));
719 }
720 return true;
721}

752void BackgroundSubtractorKNNImpl::apply(InputArray _image, InputArray _knownForegroundMask, OutputArray _fgmask, double learningRate)
753{
754 CV_INSTRUMENT_REGION();
755
756#ifdef HAVE_OPENCL
757 if (opencl_ON)
758 {
759#ifndef __APPLE__
760 CV_OCL_RUN(_fgmask.isUMat() && OCL_PERFORMANCE_CHECK(!ocl::Device::getDefault().isIntel() || _image.channels() == 1),
761 ocl_apply(_image, _fgmask, learningRate))
762#else
763 CV_OCL_RUN(_fgmask.isUMat() && OCL_PERFORMANCE_CHECK(!ocl::Device::getDefault().isIntel()),
764 ocl_apply(_image, _fgmask, learningRate))
765#endif
766
767 opencl_ON = false;
768 nframes = 0;
769 }
770#endif
771
772 bool needToInitialize = nframes == 0 || learningRate >= 1 || _image.size() != frameSize || _image.type() != frameType;
773
774 if( needToInitialize )
775 initialize(_image.size(), _image.type());
776
777 Mat image = _image.getMat();
778 _fgmask.create( image.size(), CV_8U );
779 Mat fgmask = _fgmask.getMat();
780
781 Mat knownForegroundMask = _knownForegroundMask.getMat();
782
783 if(!knownForegroundMask.empty())
784 {
785 CV_Assert(knownForegroundMask.type() == CV_8UC1);
786 CV_Assert(knownForegroundMask.size() == image.size());
787 }
788
789 ++nframes;
790 learningRate = learningRate >= 0 && nframes > 1 ? learningRate : 1./std::min( 2*nframes, history );
791 CV_Assert(learningRate >= 0);
792
793 //recalculate update rates - in case alpha is changed
794 // calculate update parameters (using alpha)
795 int Kshort,Kmid,Klong;
796 //approximate exponential learning curve
797 Kshort=(int)(log(0.7)/log(1-learningRate))+1;//Kshort
798 Kmid=(int)(log(0.4)/log(1-learningRate))-Kshort+1;//Kmid
799 Klong=(int)(log(0.1)/log(1-learningRate))-Kshort-Kmid+1;//Klong
800
801 //refresh rates
802 int nShortUpdate = (Kshort/nN)+1;
803 int nMidUpdate = (Kmid/nN)+1;
804 int nLongUpdate = (Klong/nN)+1;
805
806 parallel_for_(Range(0, image.rows),
807 KNNInvoker(image, fgmask,
808 bgmodel.ptr(),
809 nNextLongUpdate.ptr(),
810 nNextMidUpdate.ptr(),
811 nNextShortUpdate.ptr(),
812 aModelIndexLong.ptr(),
813 aModelIndexMid.ptr(),
814 aModelIndexShort.ptr(),
815 nLongCounter,
816 nMidCounter,
817 nShortCounter,
818 nN,
819 fTb,
820 nkNN,
821 fTau,
822 bShadowDetection,
823 nShadowDetection,
824 knownForegroundMask),
825 image.total()/(double)(1 << 16));
826
827 nShortCounter++;//0,1,...,nShortUpdate-1
828 nMidCounter++;
829 nLongCounter++;
830 if (nShortCounter >= nShortUpdate)
831 {
832 nShortCounter = 0;
833 randu(nNextShortUpdate, Scalar::all(0), Scalar::all(nShortUpdate));
834 }
835 if (nMidCounter >= nMidUpdate)
836 {
837 nMidCounter = 0;
838 randu(nNextMidUpdate, Scalar::all(0), Scalar::all(nMidUpdate));
839 }
840 if (nLongCounter >= nLongUpdate)
841 {
842 nLongCounter = 0;
843 randu(nNextLongUpdate, Scalar::all(0), Scalar::all(nLongUpdate));
844 }
845}

847void BackgroundSubtractorKNNImpl::getBackgroundImage(OutputArray backgroundImage) const
848{
849 CV_INSTRUMENT_REGION();
850
851#ifdef HAVE_OPENCL
852 if (opencl_ON)
853 {
854 CV_OCL_RUN(opencl_ON, ocl_getBackgroundImage(backgroundImage))
855
856 opencl_ON = false;
857 }
858#endif
859
860 int nchannels = CV_MAT_CN(frameType);
861 //CV_Assert( nchannels == 3 );
862 Mat meanBackground(frameSize, CV_8UC3, Scalar::all(0));
863
864 int ndata=nchannels+1;
865 int modelstep=(ndata * nN * 3);
866
867 const uchar* pbgmodel=bgmodel.ptr(0);
868 for(int row=0; row<meanBackground.rows; row++)
869 {
870 for(int col=0; col<meanBackground.cols; col++)
871 {
872 for (int n = 0; n < nN*3; n++)
873 {
874 const uchar* mean_m = &pbgmodel[n*ndata];
875 if (mean_m[nchannels])
876 {
877 meanBackground.at<Vec3b>(row, col) = Vec3b(mean_m);
878 break;
879 }
880 }
881 pbgmodel=pbgmodel+modelstep;
882 }
883 }
884
885 switch(CV_MAT_CN(frameType))
886 {
887 case 1:
888 {
889 std::vector<Mat> channels;
890 split(meanBackground, channels);
891 channels[0].copyTo(backgroundImage);
892 break;
893 }
894 case 3:
895 {
896 meanBackground.copyTo(backgroundImage);
897 break;
898 }
899 default:
900 CV_Error(Error::StsUnsupportedFormat, "");
901 }
902}
