58 def detectSerial(self):
59 adb_res = self.run(["devices"], silent=True)
60 # assume here that device name may consists of any characters except newline
61 connected_devices = re.findall(r"^[^\n]+[ \t]+device\r?$", adb_res, re.MULTILINE)
62 if not connected_devices:
63 raise Err("Can not find Android device")
64 elif len(connected_devices) != 1:
65 raise Err("Too many (%s) devices are connected. Please specify single device using --serial option:\n\n%s", len(connected_devices), adb_res)
66 else:
67 return connected_devices[0].split("\t")[0]

86 def dump(self, exe):
87 res = ApkInfo()
88 output = self.run(["dump", "xmltree", exe, "AndroidManifest.xml"], silent=True)
89 if not output:
90 raise Err("Can not dump manifest from %s", exe)
91 tags = re.split(r"[ ]+E: ", output)
92 # get package name
93 manifest_tag = [t for t in tags if t.startswith("manifest ")]
94 if not manifest_tag:
95 raise Err("Can not read package name from: %s", exe)
96 res.pkg_name = re.search(r"^[ ]+A: package=\"(?P<pkg>.*?)\" \‍(Raw: \"(?P=pkg)\"\‍)\r?$", manifest_tag[0], flags=re.MULTILINE).group("pkg")
97 # get test instrumentation info
98 instrumentation_tag = [t for t in tags if t.startswith("instrumentation ")]
99 if not instrumentation_tag:
100 raise Err("Can not find instrumentation details in: %s", exe)
101 res.pkg_runner = re.search(r"^[ ]+A: android:name\‍(0x[0-9a-f]{8}\‍)=\"(?P<runner>.*?)\" \‍(Raw: \"(?P=runner)\"\‍)\r?$", instrumentation_tag[0], flags=re.MULTILINE).group("runner")
102 res.pkg_target = re.search(r"^[ ]+A: android:targetPackage\‍(0x[0-9a-f]{8}\‍)=\"(?P<pkg>.*?)\" \‍(Raw: \"(?P=pkg)\"\‍)\r?$", instrumentation_tag[0], flags=re.MULTILINE).group("pkg")
103 if not res.pkg_name or not res.pkg_runner or not res.pkg_target:
104 raise Err("Can not find instrumentation details in: %s", exe)
