78class UIBackendRegistry
79{
80protected:
81 std::vector<BackendInfo> enabledBackends;
82 UIBackendRegistry()
83 {
84 enabledBackends = getBuiltinBackendsInfo();
85 int N = (int)enabledBackends.size();
86 for (int i = 0; i < N; i++)
87 {
88 BackendInfo& info = enabledBackends[i];
89 info.priority = 1000 - i * 10;
90 }
91 CV_LOG_DEBUG(NULL, "UI: Builtin backends(" << N << "): " << dumpBackends());
92 if (readPrioritySettings())
93 {
94 CV_LOG_INFO(NULL, "UI: Updated backends priorities: " << dumpBackends());
95 N = (int)enabledBackends.size();
96 }
97 int enabled = 0;
98 for (int i = 0; i < N; i++)
99 {
100 BackendInfo& info = enabledBackends[enabled];
101 if (enabled != i)
102 info = enabledBackends[i];
103 size_t param_priority = utils::getConfigurationParameterSizeT(cv::format("OPENCV_UI_PRIORITY_%s", info.name.c_str()).c_str(), (size_t)info.priority);
104 CV_Assert(param_priority == (size_t)(int)param_priority); // overflow check
105 if (param_priority > 0)
106 {
107 info.priority = (int)param_priority;
108 enabled++;
109 }
110 else
111 {
112 CV_LOG_INFO(NULL, "UI: Disable backend: " << info.name);
113 }
114 }
115 enabledBackends.resize(enabled);
116 CV_LOG_DEBUG(NULL, "UI: Available backends(" << enabled << "): " << dumpBackends());
117 std::sort(enabledBackends.begin(), enabledBackends.end(), sortByPriority);
118 CV_LOG_INFO(NULL, "UI: Enabled backends(" << enabled << ", sorted by priority): " << (enabledBackends.empty() ? std::string("N/A") : dumpBackends()));
119 }

121 static std::vector<std::string> tokenize_string(const std::string& input, char token)
122 {
123 std::vector<std::string> result;
124 std::string::size_type prev_pos = 0, pos = 0;
125 while((pos = input.find(token, pos)) != std::string::npos)
126 {
127 result.push_back(input.substr(prev_pos, pos-prev_pos));
128 prev_pos = ++pos;
129 }
130 result.push_back(input.substr(prev_pos));
131 return result;
132 }
133 bool readPrioritySettings()
134 {
135 bool hasChanges = false;
136 cv::String prioritized_backends = utils::getConfigurationParameterString("OPENCV_UI_PRIORITY_LIST");
137 if (prioritized_backends.empty())
138 return hasChanges;
139 CV_LOG_INFO(NULL, "UI: Configured priority list (OPENCV_UI_PRIORITY_LIST): " << prioritized_backends);
140 const std::vector<std::string> names = tokenize_string(prioritized_backends, ',');
141 for (size_t i = 0; i < names.size(); i++)
142 {
143 const std::string& name = names[i];
144 int priority = (int)(100000 + (names.size() - i) * 1000);
145 bool found = false;
146 for (size_t k = 0; k < enabledBackends.size(); k++)
147 {
148 BackendInfo& info = enabledBackends[k];
149 if (name == info.name)
150 {
151 info.priority = priority;
152 CV_LOG_DEBUG(NULL, "UI: New backend priority: '" << name << "' => " << info.priority);
153 found = true;
154 hasChanges = true;
155 break;
156 }
157 }
158 if (!found)
159 {
160 CV_LOG_INFO(NULL, "UI: Adding backend (plugin): '" << name << "'");
161 enabledBackends.push_back(BackendInfo{priority, name, createPluginUIBackendFactory(name)});
162 hasChanges = true;
163 }
164 }
165 return hasChanges;
166 }
167public:
168 std::string dumpBackends() const
169 {
170 std::ostringstream os;
171 for (size_t i = 0; i < enabledBackends.size(); i++)
172 {
173 if (i > 0) os << "; ";
174 const BackendInfo& info = enabledBackends[i];
175 os << info.name << '(' << info.priority << ')';
176 }
177#if !defined(OPENCV_HIGHGUI_WITHOUT_BUILTIN_BACKEND)
178 os << " + BUILTIN(" OPENCV_HIGHGUI_BUILTIN_BACKEND_STR ")";
179#endif
180 return os.str();
181 }
