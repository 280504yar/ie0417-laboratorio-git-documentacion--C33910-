34def execute(cmd, silent=False, cwd=".", env=None):
35 try:
36 log.debug("Run: %s", cmd)
37 if env is not None:
38 for k in env:
39 log.debug(" Environ: %s=%s", k, env[k])
40 new_env = os.environ.copy()
41 new_env.update(env)
42 env = new_env
43
44 if sys.platform == 'darwin': # https://github.com/opencv/opencv/issues/14351
45 if env is None:
46 env = os.environ.copy()
47 if 'DYLD_LIBRARY_PATH' in env:
48 env['OPENCV_SAVED_DYLD_LIBRARY_PATH'] = env['DYLD_LIBRARY_PATH']
49
50 if silent:
51 return check_output(cmd, stderr=STDOUT, cwd=cwd, env=env).decode("latin-1")
52 else:
53 return check_call(cmd, cwd=cwd, env=env)
54 except CalledProcessError as e:
55 if silent:
56 log.debug("Process returned: %d", e.returncode)
57 return e.output.decode("latin-1")
58 else:
59 log.error("Process returned: %d", e.returncode)
60 return e.returncode

109 def read(self, path, fname):
110 rx = re.compile(r'^OPENCV_MODULE_opencv_(\w+)_LOCATION:INTERNAL=(.*)$')
111 module_paths = {} # name -> path
112 with open(fname, "rt") as cachefile:
113 for l in cachefile.readlines():
114 ll = l.strip()
115 if not ll or ll.startswith("#"):
116 continue
117 for p in parse_patterns:
118 match = p["pattern"].match(ll)
119 if match:
120 value = match.groups()[0]
121 if value and not value.endswith("-NOTFOUND"):
122 setattr(self, p["name"], value)
123 # log.debug("cache value: %s = %s", p["name"], value)
124
125 match = rx.search(ll)
126 if match:
127 module_paths[match.group(1)] = match.group(2)
128
129 if not self.tests_dir:
130 self.tests_dir = path
131 else:
132 rel = os.path.relpath(self.tests_dir, self.opencv_build)
133 self.tests_dir = os.path.join(path, rel)
134 self.tests_dir = os.path.normpath(self.tests_dir)
135
136 # fix VS test binary path (add Debug or Release)
137 if "Visual Studio" in self.cmake_generator:
138 self.tests_dir = os.path.join(self.tests_dir, self.build_type)
139
140 for module, path in module_paths.items():
141 rel = os.path.relpath(path, self.opencv_home)
142 if ".." not in rel:
143 self.main_modules.append(module)

149 def gatherTests(self, mask, isGood=None):
150 if self.tests_dir and os.path.isdir(self.tests_dir):
151 d = os.path.abspath(self.tests_dir)
152 files = glob.glob(os.path.join(d, mask))
153 if not self.getOS() == "android" and self.withJava():
154 files.append("java")
