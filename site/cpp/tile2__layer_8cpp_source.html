23// out must be pre-allocated
24// repeats_[] should contains as many elements as inp.dims (== out.dims)
25static void tile(const Mat& inp, const int* repeats_, Mat& out)
26{
27 MatShape inpshape_ = inp.shape();
28 MatShape outshape_ = out.shape();
29 const uchar* inpdata0 = inp.data;
30 uchar* outdata0_ = out.data;
31
32 int inpshape[TILE_MAX_DIMS];
33 int outshape[TILE_MAX_DIMS];
34 int repeats[TILE_MAX_DIMS];
35 int64_t inpstep[TILE_MAX_DIMS];
36 int64_t outstep[TILE_MAX_DIMS];
37
38 int ndims = inp.dims, delta = TILE_MAX_DIMS - ndims;
39 int64_t esz = inp.elemSize();
40 int64_t total_size = 1, total_repeats = 1;
41
42 CV_Assert(inp.isContinuous());
43 CV_Assert(out.isContinuous());
44 CV_Assert(inp.type() == out.type());
45 CV_Assert(esz == 1 || esz == 2 || esz == 4 || esz == 8);
46 CV_Assert(inp.dims == out.dims);
47 CV_Assert(inp.dims <= TILE_MAX_DIMS);
48
49 for (int i = 0; i < TILE_MAX_DIMS; i++) {
50 inpshape[i] = outshape[i] = repeats[i] = 1;
51 }
52
53 for (int i = 0; i < ndims; i++) {
54 inpshape[i + delta] = inpshape_[i];
55 outshape[i + delta] = outshape_[i];
56 repeats[i + delta] = repeats_[i];
57
58 CV_Assert(inpshape_[i]*repeats_[i] == outshape_[i]);
59
60 total_size *= outshape_[i];
61 total_repeats *= repeats_[i];
62 }
63
64 for (int i = TILE_MAX_DIMS-1; i >= 0; i--) {
65 if (i == TILE_MAX_DIMS-1)
66 inpstep[i] = outstep[i] = 1;
67 else {
68 inpstep[i] = inpstep[i+1]*inpshape[i+1];
69 outstep[i] = outstep[i+1]*outshape[i+1];
70 }
71 }
72
73 int ntasks = 8;
74 if (ntasks > total_repeats)
75 ntasks = (int)total_repeats;
76 if (total_size < 1000000)
77 ntasks = 1;
78
79 parallel_for_(Range(0, ntasks), [&](const Range& r)
80 {
81 int sz0 = inpshape[0], sz1 = inpshape[1], sz2 = inpshape[2];
82 int sz3 = inpshape[3], sz4 = inpshape[4], sz5 = inpshape[5];
83
84 int64_t outstep_prelast = outstep[TILE_MAX_DIMS-2];
85 int64_t j0 = r.start*total_repeats/ntasks, j1 = r.end*total_repeats/ntasks;
86
87 for (int64_t j = j0; j < j1; j++)
88 {
89 // convert raw tile index into n-dim tile index.
90 // but we don't need this nd-index itself, we just need the
91 // offset of the tile in the output tensor
92 int64_t j_ = j, rawofs = 0;
93 for (int k = TILE_MAX_DIMS-1; k >= 0; k--) {
94 int r = repeats[k];
95 int64_t q = j_ / r;
96 rawofs += (j_ - q*r)*inpshape[k]*outstep[k];
97 j_ = q;
98 }
99
100 #undef IMPL_COPY_TILE
101 #define IMPL_COPY_TILE(T) \
102 T* inpdata = (T*)inpdata0; \
103 T* outdata0 = (T*)outdata0_ + rawofs; \
104 for (int i0 = 0; i0 < sz0; i0++) { \
105 for (int i1 = 0; i1 < sz1; i1++) { \
106 for (int i2 = 0; i2 < sz2; i2++) { \
107 for (int i3 = 0; i3 < sz3; i3++) { \
108 T* outdata = outdata0 + i0*outstep[0] + i1*outstep[1] + i2*outstep[2] + i3*outstep[3]; \
109 for (int i4 = 0; i4 < sz4; i4++, outdata += outstep_prelast, inpdata += sz5) { \
110 for (int i5 = 0; i5 < sz5; i5++) \
111 outdata[i5] = inpdata[i5]; \
112 } \
113 }}}}
114
115 if (esz == 1) {
116 IMPL_COPY_TILE(uint8_t)
117 } else if (esz == 2) {
118 IMPL_COPY_TILE(uint16_t)
119 } else if (esz == 4) {
120 IMPL_COPY_TILE(uint32_t)
121 } else {
122 IMPL_COPY_TILE(uint64_t)
123 }
124 }
125 }
126 , ntasks);
127}

154 void getRepeats(const Mat& repeats_, const Mat& axes_, int ndims, int* repeats) const
155 {
156 int atype = axes_.type(), rtype = repeats_.type();
157 CV_Assert(ndims <= TILE_MAX_DIMS);
158
159 const int32_t* adata_i32 = nullptr;
160 const int64_t* adata_i64 = nullptr;
161 const int32_t* rdata_i32 = nullptr;
162 const int64_t* rdata_i64 = nullptr;
163
164 bool axismask[TILE_MAX_DIMS];
165
166 CV_Assert(repeats_.dims == 1);
167 CV_Assert(rtype == CV_32S || rtype == CV_64S);
168
169 if (rtype == CV_32S)
170 rdata_i32 = reinterpret_cast<const int32_t*>(repeats_.data);
171 else
172 rdata_i64 = reinterpret_cast<const int64_t*>(repeats_.data);
173
174 if (!axes_.empty()) {
175 CV_Assert(axes_.dims == 1);
176 CV_Assert(atype == CV_32S || atype == CV_64S);
177 CV_Assert(repeats_.total() == axes_.total());
178 CV_Assert(axes_.total() <= (size_t)ndims);
179
180 if (atype == CV_32S)
181 adata_i32 = reinterpret_cast<const int32_t*>(axes_.data);
182 else
183 adata_i64 = reinterpret_cast<const int64_t*>(axes_.data);
184 } else {
185 CV_Assert(repeats_.total() == (size_t)ndims);
186 }
187
188 for (int i = 0; i < ndims; i++) {
189 repeats[i] = 1;
190 axismask[i] = false;
191 }
192
193 int nrepeats = (int)repeats_.total();
194 for (int i = 0; i < nrepeats; i++) {
195 int a = adata_i32 ? (int)adata_i32[i] : adata_i64 ? (int)adata_i64[i] : i;
196 a = normalize_axis(a, ndims);
197 if (axismask[a]) {
198 CV_Error_(Error::StsBadArg, ("duplicate axis %d in Tile", a));
199 }
200 axismask[a] = true;
201 int r = rdata_i32 ? (int)rdata_i32[i] : rdata_i64 ? (int)rdata_i64[i] : 1;
202 repeats[a] = r;
203 }
204 }

214 bool getMemoryShapes(const std::vector<MatShape>& inputs,
215 const int,
216 std::vector<MatShape> &outputs,
217 std::vector<MatShape> &internals) const CV_OVERRIDE
218 {
219 CV_Assert(!dynamicOutputShapes());
220
221 size_t ninputs = inputs.size();
222 CV_Assert(ninputs == (size_t)2 || ninputs == (size_t)3);
223 Net::Impl* netimpl_ = getNetImpl(this);
224
225 int repeats[TILE_MAX_DIMS];
226
227 Mat repeatsTensor = netimpl_->argTensor(this->inputs[1]);
228 Mat axesTensor;
229 if (ninputs > 2)
230 axesTensor = netimpl_->argTensor(this->inputs[2]);
231
232 int ndims = inputs[0].dims;
233 getRepeats(repeatsTensor, axesTensor, ndims, repeats);
234
235 outputs.assign(1, getOutShape(inputs[0], repeats));
236 internals.clear();
237 return true;
238 }

257 void forward(InputArrayOfArrays inputs_arr,
258 OutputArrayOfArrays outputs_arr,
259 OutputArrayOfArrays) CV_OVERRIDE
260 {
261 CV_TRACE_FUNCTION();
262 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
263
264 Size size = inputs_arr.size();
265 int ninputs = size.area();
266 CV_Assert(ninputs == 2 || ninputs == 3);
267
268 Mat inp = inputs_arr.getMat(0);
269 Mat repeatsTensor = inputs_arr.getMat(1);
270 Mat axesTensor;
271 int repeats[TILE_MAX_DIMS];
272 int inptype = inp.type();
273 int ndims = inp.dims;
274
275 if (ninputs > 2)
276 axesTensor = inputs_arr.getMat(2);
277
278 getRepeats(repeatsTensor, axesTensor, ndims, repeats);
279 MatShape outshape = getOutShape(inp.shape(), repeats);
280
281 auto kind = outputs_arr.kind();
282 if (kind == _InputArray::STD_VECTOR_MAT) {
283 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
284 outs.resize(1);
285 outs[0].fit(outshape, inptype);
286 tile(inp, repeats, outs[0]);
287 } else if (kind == _InputArray::STD_VECTOR_UMAT) {
288 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
289 outs.resize(1);
290 outs[0].fit(outshape, inptype);
291 Mat temp(outshape, inptype);
292 tile(inp, repeats, temp);
293 temp.copyTo(outs[0]);
294 } else {
295 CV_Error(Error::StsNotImplemented, "");
296 }
297 }
