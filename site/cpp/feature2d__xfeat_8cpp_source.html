34static Mat toGray(InputArray _image)
35{
36 Mat image = _image.getMat();
37 if (image.channels() == 1)
38 return image;
39
40 Mat gray;
41 if (image.channels() == 3)
42 cvtColor(image, gray, COLOR_BGR2GRAY);
43 else if (image.channels() == 4)
44 cvtColor(image, gray, COLOR_BGRA2GRAY);
45 else
46 CV_Error(Error::StsBadArg, "XFeat expects a grayscale, BGR, or BGRA image");
47 return gray;
48}

50static Mat toNCHW(const Mat& blob, int channelsHint)
51{
52 CV_Assert(blob.dims == 4);
53 if (blob.size[1] == channelsHint)
54 return blob;
55
56 CV_Assert(blob.size[3] == channelsHint);
57 Mat out;
58 Mat src = blob.isContinuous() ? blob : blob.clone();
59 transposeND(src, {0, 3, 1, 2}, out);
60 return out;
61}

63static float sampleNearest(const Mat& map, float x, float y, int normW, int normH)
64{
65 CV_Assert(map.type() == CV_32F);
66 if (map.empty() || normW <= 1 || normH <= 1)
67 return 0.0f;
68
69 const float fx = x * static_cast<float>(map.cols - 1) / static_cast<float>(normW - 1);
70 const float fy = y * static_cast<float>(map.rows - 1) / static_cast<float>(normH - 1);
71 const int ix = std::max(0, std::min(map.cols - 1, cvRound(fx)));
72 const int iy = std::max(0, std::min(map.rows - 1, cvRound(fy)));
73 return map.at<float>(iy, ix);
74}

76static float sampleBilinear(const Mat& map, float x, float y, int normW, int normH)
77{
78 CV_Assert(map.type() == CV_32F);
79 if (map.empty() || normW <= 1 || normH <= 1)
80 return 0.0f;
81
82 float fx = x * static_cast<float>(map.cols - 1) / static_cast<float>(normW - 1);
83 float fy = y * static_cast<float>(map.rows - 1) / static_cast<float>(normH - 1);
84 fx = std::max(0.0f, std::min(fx, static_cast<float>(map.cols - 1)));
85 fy = std::max(0.0f, std::min(fy, static_cast<float>(map.rows - 1)));
86
87 const int x0 = cvFloor(fx);
88 const int y0 = cvFloor(fy);
89 const int x1 = std::min(x0 + 1, map.cols - 1);
90 const int y1 = std::min(y0 + 1, map.rows - 1);
91 const float dx = fx - x0;
92 const float dy = fy - y0;
93
94 const float v00 = map.at<float>(y0, x0);
95 const float v01 = map.at<float>(y0, x1);
96 const float v10 = map.at<float>(y1, x0);
97 const float v11 = map.at<float>(y1, x1);
98 return (1.f - dx) * (1.f - dy) * v00 +
99 dx * (1.f - dy) * v01 +
100 (1.f - dx) * dy * v10 +
101 dx * dy * v11;
102}

129 void detectAndCompute(InputArray _image, InputArray _mask,
130 std::vector<KeyPoint>& keypoints,
131 OutputArray _descriptors,
132 bool useProvidedKeypoints) CV_OVERRIDE
133 {
134 CV_Assert(!useProvidedKeypoints && "XFeat does not support providing keypoints externally");
135
136 keypoints.clear();
137
138 Mat image = _image.getMat();
139 if (image.empty())
140 {
141 if (_descriptors.needed())
142 _descriptors.release();
143 return;
144 }
145
146 Mat mask = _mask.getMat();
147 if (!mask.empty())
148 {
149 CV_Assert(mask.type() == CV_8UC1 || mask.type() == CV_BoolC1);
150 CV_Assert(mask.size() == image.size());
151 }
152
153 Mat gray = toGray(image);
154 const float scaleX = static_cast<float>(inputSize_.width) / static_cast<float>(gray.cols);
155 const float scaleY = static_cast<float>(inputSize_.height) / static_cast<float>(gray.rows);
156 const float scale = std::min(scaleX, scaleY);
157 const int resizedW = std::max(1, cvRound(static_cast<float>(gray.cols) * scale));
158 const int resizedH = std::max(1, cvRound(static_cast<float>(gray.rows) * scale));
159 const int padX = (inputSize_.width - resizedW) / 2;
160 const int padY = (inputSize_.height - resizedH) / 2;
161
162 Mat blob;
163 Image2BlobParams blobParams;
164 blobParams.scalefactor = Scalar::all(1.0 / 255.0);
165 blobParams.size = inputSize_;
166 blobParams.mean = Scalar();
167 blobParams.swapRB = false;
168 blobParams.ddepth = CV_32F;
169 blobParams.datalayout = DNN_LAYOUT_NCHW;
170 blobParams.paddingmode = DNN_PMODE_LETTERBOX;
171 blobParams.borderValue = Scalar();
172 blobFromImageWithParams(gray, blob, blobParams);
173 net_.setInput(blob);
174
175 std::vector<Mat> outs;
176 net_.forward(outs, kXFeatOutputNames);
177 CV_Assert(outs.size() == 3);
178
179 Mat featBlob = toNCHW(outs[0], kXFeatDescriptorSize);
180 Mat kptBlob = outs[1];
181 CV_Assert(kptBlob.dims == 4);
182 if (!(kptBlob.size[1] == 64 || kptBlob.size[1] == 65))
183 {
184 CV_Assert(kptBlob.size[3] == 64 || kptBlob.size[3] == 65);
185 Mat src = kptBlob.isContinuous() ? kptBlob : kptBlob.clone();
186 transposeND(src, {0, 3, 1, 2}, kptBlob);
187 }
188 Mat relBlob = toNCHW(outs[2], 1);
189 CV_Assert(featBlob.dims == 4 && kptBlob.dims == 4 && relBlob.dims == 4);
190 if (!featBlob.isContinuous())
191 featBlob = featBlob.clone();
192 if (!kptBlob.isContinuous())
193 kptBlob = kptBlob.clone();
194 if (!relBlob.isContinuous())
195 relBlob = relBlob.clone();
196
197 const int featH = featBlob.size[2];
198 const int featW = featBlob.size[3];
199 const int kptC = kptBlob.size[1];
200 const int kptH = kptBlob.size[2];
201 const int kptW = kptBlob.size[3];
202 CV_Assert(featBlob.size[1] == kXFeatDescriptorSize && (kptC == 64 || kptC == 65));
203
204 Mat reliability(relBlob.size[2], relBlob.size[3], CV_32F, relBlob.ptr<float>());
205 Mat heatmap = Mat::zeros(kptH * 8, kptW * 8, CV_32F);
206 const float* kptPtr = kptBlob.ptr<float>();
207 const int kptHW = kptH * kptW;
208
209 parallel_for_(Range(0, kptH), [&](const Range& range)
210 {
211 for (int y = range.start; y < range.end; ++y)
212 {
213 for (int x = 0; x < kptW; ++x)
214 {
215 const int offset = y * kptW + x;
216 float maxLogit = -FLT_MAX;
217 for (int ch = 0; ch < kptC; ++ch)
218 maxLogit = std::max(maxLogit, kptPtr[ch * kptHW + offset]);
219
220 float sumExp = 0.f;
221 float logits[64];
222 float probs[64];
223 for (int ch = 0; ch < 64; ++ch)
224 logits[ch] = kptPtr[ch * kptHW + offset] - maxLogit;
225#if (defined(CV_SIMD) && CV_SIMD) || (defined(CV_SIMD_SCALABLE) && CV_SIMD_SCALABLE)
226 const int vlanes = VTraits<v_float32>::vlanes();
227 int ch = 0;
228 v_float32 vSum = vx_setzero_f32();
229 for (; ch <= 64 - vlanes; ch += vlanes)
230 {
231 v_float32 v = vx_load(logits + ch);
232 v_float32 e = v_exp(v);
233 vx_store(probs + ch, e);
234 vSum = v_add(vSum, e);
235 }
236 sumExp = v_reduce_sum(vSum);
237 for (; ch < 64; ++ch)
238 {
239 probs[ch] = std::exp(logits[ch]);
240 sumExp += probs[ch];
241 }
242#else
243 for (int ch = 0; ch < 64; ++ch)
244 {
245 probs[ch] = std::exp(logits[ch]);
246 sumExp += probs[ch];
247 }
248#endif
249 if (kptC == 65)
250 sumExp += std::exp(kptPtr[64 * kptHW + offset] - maxLogit);
251 if (sumExp <= 0.f)
252 continue;
253
254 for (int ch = 0; ch < 64; ++ch)
255 {
256 const int dy = ch / 8;
257 const int dx = ch % 8;
258 heatmap.at<float>(y * 8 + dy, x * 8 + dx) = probs[ch] / sumExp;
259 }
260 }
261 }
262 });
263
264 Mat localMax;
265 dilate(heatmap, localMax, getStructuringElement(MORPH_RECT, Size(5, 5)));
266
267 std::vector<XFeatCandidate> candidates;
268 candidates.reserve(4096);
269
270 for (int y = 0; y < heatmap.rows; ++y)
271 {
272 const float* hm = heatmap.ptr<float>(y);
273 const float* mx = localMax.ptr<float>(y);
274 for (int x = 0; x < heatmap.cols; ++x)
275 {
276 const float h = hm[x];
277 if (h <= scoreThreshold_ || h != mx[x])
278 continue;
279
280 const float xp = static_cast<float>(x);
281 const float yp = static_cast<float>(y);
282 const float score = sampleNearest(heatmap, xp, yp, inputSize_.width, inputSize_.height) *
283 sampleBilinear(reliability, xp, yp, inputSize_.width, inputSize_.height);
284 if (score <= 0.f)
285 continue;
286
287 const float px = (xp - static_cast<float>(padX)) / scale;
288 const float py = (yp - static_cast<float>(padY)) / scale;
289 const int ix = cvFloor(px);
290 const int iy = cvFloor(py);
291 if (ix < 0 || iy < 0 || ix >= image.cols || iy >= image.rows)
292 continue;
293 if (!mask.empty() && mask.at<uchar>(iy, ix) == 0)
294 continue;
295
296 candidates.push_back({Point2f(xp, yp), Point2f(px, py), score});
297 }
298 }
299
300 if (maxKeypoints_ > 0 && static_cast<int>(candidates.size()) > maxKeypoints_)
301 {
302 std::partial_sort(candidates.begin(), candidates.begin() + maxKeypoints_, candidates.end(),
303 [](const XFeatCandidate& a, const XFeatCandidate& b)
304 {
305 return a.score > b.score;
306 });
307 candidates.resize(maxKeypoints_);
308 }
309
310 keypoints.reserve(candidates.size());
311 for (const XFeatCandidate& c : candidates)
312 keypoints.emplace_back(c.pt, 1.0f, -1.0f, c.score);
313
314 if (_descriptors.needed())
315 {
316 if (candidates.empty())
317 {
318 _descriptors.release();
319 return;
320 }
321
322 _descriptors.create(static_cast<int>(candidates.size()), kXFeatDescriptorSize, CV_32F);
323 Mat descriptors = _descriptors.getMat();
324 const float* featPtr = featBlob.ptr<float>();
325 const int featHW = featH * featW;
326
327 parallel_for_(Range(0, static_cast<int>(candidates.size())),
328 [&](const Range& range)
329 {
330 for (int i = range.start; i < range.end; ++i)
331 {
332 float* dst = descriptors.ptr<float>(i);
333 const XFeatCandidate& c = candidates[i];
334 for (int ch = 0; ch < kXFeatDescriptorSize; ++ch)
335 {
336 Mat channel(featH, featW, CV_32F,
337 const_cast<float*>(featPtr + ch * featHW));
338 dst[ch] = sampleBilinear(channel, c.ptPadded.x, c.ptPadded.y,
339 inputSize_.width, inputSize_.height);
340 }
341 normalize(descriptors.row(i), descriptors.row(i), 1.0, 0.0, NORM_L2);
342 }
343 });
344 }
345 }

369 void initNet(const Net& net, int backendId, int targetId)
370 {
371 net_ = net;
372 net_.setPreferableBackend(backendId);
373 net_.setPreferableTarget(targetId);
374
375 // Check output names once and fail early.
376 const std::vector<String> modelOutNames = net_.getUnconnectedOutLayersNames();
377 if (modelOutNames.size() != kXFeatOutputNames.size())
378 {
379 String msg = "XFeat ONNX output count mismatch: expected " +
380 std::to_string(kXFeatOutputNames.size()) + " outputs (";
381 for (size_t i = 0; i < kXFeatOutputNames.size(); ++i)
382 {
383 msg += kXFeatOutputNames[i];
384 if (i + 1 < kXFeatOutputNames.size())
385 msg += ", ";
386 }
387 msg += "), but model has " + std::to_string(modelOutNames.size()) + " outputs: ";
388 for (size_t i = 0; i < modelOutNames.size(); ++i)
389 {
390 msg += modelOutNames[i];
391 if (i + 1 < modelOutNames.size())
392 msg += ", ";
393 }
394 CV_Error(Error::StsError, msg);
395 }
396
397 for (size_t i = 0; i < kXFeatOutputNames.size(); ++i)
398 {
399 bool found = false;
400 for (size_t j = 0; j < modelOutNames.size(); ++j)
401 {
402 if (modelOutNames[j] == kXFeatOutputNames[i])
403 {
404 found = true;
405 break;
406 }
407 }
408
409 if (!found)
410 {
411 String msg = "XFeat ONNX output name mismatch: expected output '" + kXFeatOutputNames[i] +
412 "'. Please check model outputs. Available outputs: ";
413 for (size_t j = 0; j < modelOutNames.size(); ++j)
414 {
415 msg += modelOutNames[j];
416 if (j + 1 < modelOutNames.size())
417 msg += ", ";
418 }
419 CV_Error(Error::StsError, msg);
420 }
421 }
422 }
