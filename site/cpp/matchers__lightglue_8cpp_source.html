159void LightGlueMatcherImpl::lightglueMatch(const Mat& queryDesc, const Mat& trainDesc,
160 const Mat& queryKpts, const Mat& trainKpts,
161 Size queryImgSize, Size trainImgSize,
162 std::vector<DMatch>& matches)
163{
164 int N = queryDesc.rows;
165 int M = trainDesc.rows;
166
167 // Normalize keypoints to [-1, 1] if in pixel coordinates
168 Mat kpts0 = queryKpts.clone();
169 Mat kpts1 = trainKpts.clone();
170
171 if (queryImgSize.width > 0 && queryImgSize.height > 0)
172 {
173 for (int i = 0; i < kpts0.rows; i++)
174 {
175 kpts0.at<float>(i, 0) = kpts0.at<float>(i, 0) / (float)queryImgSize.width * 2.0f - 1.0f;
176 kpts0.at<float>(i, 1) = kpts0.at<float>(i, 1) / (float)queryImgSize.height * 2.0f - 1.0f;
177 }
178 }
179 if (trainImgSize.width > 0 && trainImgSize.height > 0)
180 {
181 for (int i = 0; i < kpts1.rows; i++)
182 {
183 kpts1.at<float>(i, 0) = kpts1.at<float>(i, 0) / (float)trainImgSize.width * 2.0f - 1.0f;
184 kpts1.at<float>(i, 1) = kpts1.at<float>(i, 1) / (float)trainImgSize.height * 2.0f - 1.0f;
185 }
186 }
187
188 // Prepare blobs: [1, N, 2] and [1, N, D]
189 int descDim = queryDesc.cols;
190 int szK0[] = {1, N, 2};
191 int szK1[] = {1, M, 2};
192 int szD0[] = {1, N, descDim};
193 int szD1[] = {1, M, descDim};
194 Mat kpts0blob = kpts0.reshape(0, 3, szK0);
195 Mat kpts1blob = kpts1.reshape(0, 3, szK1);
196 Mat desc0blob = queryDesc.reshape(0, 3, szD0);
197 Mat desc1blob = trainDesc.reshape(0, 3, szD1);
198
199 net.setInput(kpts0blob, "kpts0");
200 net.setInput(kpts1blob, "kpts1");
201 net.setInput(desc0blob, "desc0");
202 net.setInput(desc1blob, "desc1");
203
204 std::vector<String> outNames = {"matches0", "mscores0"};
205 std::vector<Mat> outs;
206 net.forward(outs, outNames);
207
208 CV_Assert(outs.size() == 2);
209
210 // matches0: [M, 2] int64 - pair indices (kpt0_idx, kpt1_idx)
211 // mscores0: [M] float32 - confidence per pair
212 Mat matchesMat = outs[0];
213 Mat scoresMat = outs[1];
214
215 matches.clear();
216 int nMatches = matchesMat.rows;
217 matches.reserve(nMatches);
218
219 for (int i = 0; i < nMatches; i++)
220 {
221 int qIdx = (int)matchesMat.at<int64_t>(i, 0);
222 int tIdx = (int)matchesMat.at<int64_t>(i, 1);
223 if (qIdx >= 0 && tIdx >= 0 && qIdx < N && tIdx < M)
224 {
225 float score = scoresMat.at<float>(i);
226 if (score >= scoreThreshold)
227 {
228 matches.push_back(DMatch(qIdx, tIdx, 1.0f - score));
229 }
230 }
231 }

234void LightGlueMatcherImpl::knnMatchImpl(InputArray _queryDescriptors,
235 std::vector<std::vector<DMatch>>& matches,
236 int k, InputArrayOfArrays, bool)
237{
238 CV_INSTRUMENT_REGION();
239
240 if (k != 1)
241 CV_Error(cv::Error::StsBadArg, "LightGlueMatcher only supports k=1");
242
243 Mat queryKpts, trainKpts;
244 Size queryImgSize, trainImgSize;
245 if (!resolveContext(queryKpts, trainKpts, queryImgSize, trainImgSize))
246 {
247 CV_Error(cv::Error::StsBadArg,
248 "LightGlueMatcher: no valid context. Call setPairInfo() before matching.");
249 }
250
251 CV_Assert(!trainDescCollection.empty());
252 const Mat& trainDesc = trainDescCollection[0];
253 Mat queryDesc = _queryDescriptors.getMat();
254
255 std::vector<DMatch> flatMatches;
256 lightglueMatch(queryDesc, trainDesc, queryKpts, trainKpts,
257 queryImgSize, trainImgSize, flatMatches);
258
259 matches.clear();
260 matches.resize(queryDesc.rows);
261 for (const auto& m : flatMatches)
262 {
263 matches[m.queryIdx].push_back(m);
264 }
265
266 clearPairInfo();
267}
