59inline std::vector<UMat> ToUMat(const std::vector<Mat>& src)
60{
61 std::vector<UMat> dst;
62 dst.resize(src.size());
63 for (size_t i = 0; i < src.size(); ++i)
64 {
65 src[i].copyTo(dst[i]);

67 return dst;

234 int randomInt(int minVal, int maxVal)

244 double randomDoubleLog(double minVal, double maxVal)
245 {
246 double logMin = log((double)minVal + 1);
247 double logMax = log((double)maxVal + 1);
248 double pow = rng.uniform(logMin, logMax);
249 double v = exp(pow) - 1;
250 CV_Assert(v >= minVal && (v < maxVal || (v == minVal && v == maxVal)));
251 return v;
252 }

320 static inline double checkNormRelative(InputArray m1, InputArray m2, InputArray mask = noArray())
321 {
322 return cvtest::norm(m1.getMat(), m2.getMat(), cv::NORM_INF, mask) /
323 std::max((double)std::numeric_limits<float>::epsilon(),
324 (double)std::max(cvtest::norm(m1.getMat(), cv::NORM_INF), cvtest::norm(m2.getMat(), cv::NORM_INF)));
325 }
326
327 static inline double checkNormRelativeSparse(InputArray m1, InputArray m2, InputArray mask = noArray())
328 {
329 double norm_inf = cvtest::norm(m1.getMat(), m2.getMat(), cv::NORM_INF, mask);
330 double norm_rel = norm_inf /
331 std::max((double)std::numeric_limits<float>::epsilon(),
332 (double)std::max(cvtest::norm(m1.getMat(), cv::NORM_INF), cvtest::norm(m2.getMat(), cv::NORM_INF)));
333 return std::min(norm_inf, norm_rel);
334 }
