40 BackendRegistry()
41 {
42 bool haveBackendOpenVINO = false;
43#ifdef HAVE_INF_ENGINE
44 haveBackendOpenVINO = true;
45#elif defined(ENABLE_PLUGINS)
46 {
47 auto factory = dnn_backend::createPluginDNNBackendFactory("openvino");
48 if (factory)
49 {
50 auto backend = factory->createNetworkBackend();
51 if (backend)
52 haveBackendOpenVINO = true;
53 }
54 }
55#endif
56
57 bool haveBackendCPU_FP16 = false;
58#if defined(__arm64__) && __arm64__
59 haveBackendCPU_FP16 = true;
60#endif
61
62 if (haveBackendOpenVINO && openvino::checkTarget(DNN_TARGET_CPU))
63 {
64 backends.push_back(std::make_pair(DNN_BACKEND_INFERENCE_ENGINE_NGRAPH, DNN_TARGET_CPU));
65 }
66 if (haveBackendOpenVINO && openvino::checkTarget(DNN_TARGET_MYRIAD))
67 {
68 backends.push_back(std::make_pair(DNN_BACKEND_INFERENCE_ENGINE_NGRAPH, DNN_TARGET_MYRIAD));
69 }
70 if (haveBackendOpenVINO && openvino::checkTarget(DNN_TARGET_HDDL))
71 {
72 backends.push_back(std::make_pair(DNN_BACKEND_INFERENCE_ENGINE_NGRAPH, DNN_TARGET_HDDL));
73 }
74#ifdef HAVE_OPENCL
75 if (cv::ocl::useOpenCL() && ocl::Device::getDefault().isIntel())
76 {
77 if (haveBackendOpenVINO && openvino::checkTarget(DNN_TARGET_OPENCL))
78 {
79 backends.push_back(std::make_pair(DNN_BACKEND_INFERENCE_ENGINE_NGRAPH, DNN_TARGET_OPENCL));
80 }
81 if (haveBackendOpenVINO && openvino::checkTarget(DNN_TARGET_OPENCL_FP16))
82 {
83 backends.push_back(std::make_pair(DNN_BACKEND_INFERENCE_ENGINE_NGRAPH, DNN_TARGET_OPENCL_FP16));
84 }
85 }
86#endif // HAVE_OPENCL
87
88#ifdef HAVE_WEBNN
89 if (haveWebnn())
90 {
91 backends.push_back(std::make_pair(DNN_BACKEND_WEBNN, DNN_TARGET_CPU));
92 }
93#endif // HAVE_WEBNN
94
95#ifdef HAVE_OPENCL
96 if (cv::ocl::useOpenCL())
97 {
98 backends.push_back(std::make_pair(DNN_BACKEND_OPENCV, DNN_TARGET_OPENCL));
99 backends.push_back(std::make_pair(DNN_BACKEND_OPENCV, DNN_TARGET_OPENCL_FP16));
100 }
101#endif
102
103 backends.push_back(std::make_pair(DNN_BACKEND_OPENCV, DNN_TARGET_CPU));
104
105 if (haveBackendCPU_FP16)
106 backends.push_back(std::make_pair(DNN_BACKEND_OPENCV, DNN_TARGET_CPU_FP16));
107
108#ifdef HAVE_VULKAN
109 if (haveVulkan())
110 backends.push_back(std::make_pair(DNN_BACKEND_VKCOM, DNN_TARGET_VULKAN));
111#endif
112
113#ifdef HAVE_CUDA
114 cuda4dnn::checkVersions();
115
116 bool hasCudaCompatible = false;
117 bool hasCudaFP16 = false;
118 for (int i = 0; i < cuda4dnn::getDeviceCount(); i++)
119 {
120 if (cuda4dnn::isDeviceCompatible(i))
121 {
122 hasCudaCompatible = true;
123 if (cuda4dnn::doesDeviceSupportFP16(i))
124 {
125 hasCudaFP16 = true;
126 break; // we already have all we need here
127 }
128 }
129 }
130
131 if (hasCudaCompatible)
132 {
133 backends.push_back(std::make_pair(DNN_BACKEND_CUDA, DNN_TARGET_CUDA));
134 if (hasCudaFP16)
135 backends.push_back(std::make_pair(DNN_BACKEND_CUDA, DNN_TARGET_CUDA_FP16));
136 }
137#endif
138
139#ifdef HAVE_TIMVX
140 if (haveTimVX())
141 {
142 backends.push_back(std::make_pair(DNN_BACKEND_TIMVX, DNN_TARGET_NPU));
143 }
144#endif
145
146#ifdef HAVE_CANN
147 backends.push_back(std::make_pair(DNN_BACKEND_CANN, DNN_TARGET_NPU));
148#endif
149 }
150
151 BackendsList backends;
152};

160std::vector<Target> getAvailableTargets(Backend be)
161{
162 if (be == DNN_BACKEND_DEFAULT)
163 be = (Backend)getParam_DNN_BACKEND_DEFAULT();
164
165 if (be == DNN_BACKEND_INFERENCE_ENGINE)
166 be = DNN_BACKEND_INFERENCE_ENGINE_NGRAPH;
167
168 std::vector<Target> result;
169 const BackendRegistry::BackendsList all_backends = getAvailableBackends();
170 for (BackendRegistry::BackendsList::const_iterator i = all_backends.begin(); i != all_backends.end(); ++i)
171 {
172 if (i->first == be)
173 result.push_back(i->second);
174 }
175 return result;
176}
