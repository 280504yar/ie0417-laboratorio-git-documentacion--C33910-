59bool getParameter(const LayerParams &params, const std::string& nameBase, const std::string& nameAll,
60 std::vector<size_t>& parameter, bool hasDefault = false, const std::vector<size_t>& defaultValue = std::vector<size_t>(2, 0))
61{
62 std::string nameH = makeName(nameBase, std::string("_h"));
63 std::string nameW = makeName(nameBase, std::string("_w"));
64 std::string nameAll_ = nameAll;
65 if (nameAll_ == "")
66 nameAll_ = nameBase;
67
68 if (params.has(nameH) && params.has(nameW))
69 {
70 CV_Assert(params.get<int>(nameH) >= 0 && params.get<int>(nameW) >= 0);
71 parameter.push_back(params.get<int>(nameH));
72 parameter.push_back(params.get<int>(nameW));
73 return true;
74 }
75 else
76 {
77 if (params.has(nameAll_))
78 {
79 DictValue param = params.get(nameAll_);
80 for (int i = 0; i < param.size(); i++) {
81 CV_Assert(param.get<int>(i) >= 0);
82 parameter.push_back(param.get<int>(i));
83 }
84 if (parameter.size() == 1)
85 parameter.resize(2, parameter[0]);
86 return true;
87 }
88 else
89 {
90 if (hasDefault)
91 {
92 parameter = defaultValue;
93 return true;
94 }
95 else
96 {
97 return false;
98 }
99 }
100 }

112void getStrideAndPadding(const LayerParams &params, std::vector<size_t>& pads_begin, std::vector<size_t>& pads_end,
113 std::vector<size_t>& strides, cv::String& padMode, size_t kernel_size = 2)
114{
115 if (params.has("pad_l") && params.has("pad_t") && params.has("pad_r") && params.has("pad_b")) {
116 CV_Assert(params.get<int>("pad_t") >= 0 && params.get<int>("pad_l") >= 0 &&
117 params.get<int>("pad_b") >= 0 && params.get<int>("pad_r") >= 0);
118 pads_begin.push_back(params.get<int>("pad_t"));
119 pads_begin.push_back(params.get<int>("pad_l"));
120 pads_end.push_back(params.get<int>("pad_b"));
121 pads_end.push_back(params.get<int>("pad_r"));
122 }
123 else {
124 util::getParameter(params, "pad", "pad", pads_begin, true, std::vector<size_t>(kernel_size, 0));
125 if (pads_begin.size() < 4)
126 pads_end = pads_begin;
127 else
128 {
129 pads_end = std::vector<size_t>(pads_begin.begin() + pads_begin.size() / 2, pads_begin.end());
130 pads_begin.resize(pads_begin.size() / 2);
131 }
132 CV_Assert(pads_begin.size() == pads_end.size());
133 }
134 util::getParameter(params, "stride", "stride", strides, true, std::vector<size_t>(kernel_size, 1));
135
136 padMode = "";
137 if (params.has("pad_mode"))
138 {
139 padMode = params.get<String>("pad_mode");
140 }
141
142 for (int i = 0; i < strides.size(); i++)
143 CV_Assert(strides[i] > 0);
144}

147void getPoolingKernelParams(const LayerParams &params, std::vector<size_t>& kernel, std::vector<bool>& globalPooling,
148 std::vector<size_t>& pads_begin, std::vector<size_t>& pads_end,
149 std::vector<size_t>& strides, cv::String &padMode)
150{
151 bool is_global = params.get<bool>("global_pooling", false);
152 globalPooling.assign({
153 params.get<bool>("global_pooling_d", is_global),
154 params.get<bool>("global_pooling_h", is_global),
155 params.get<bool>("global_pooling_w", is_global)
156 });
157
158 if (globalPooling[0] || globalPooling[1] || globalPooling[2])
159 {
160 util::getStrideAndPadding(params, pads_begin, pads_end, strides, padMode);
161 if ((globalPooling[0] && params.has("kernel_d")) ||
162 (globalPooling[1] && params.has("kernel_h")) ||
163 (globalPooling[2] && params.has("kernel_w")) ||
164 params.has("kernel_size")) {
165 CV_Error(cv::Error::StsBadArg, "In global_pooling mode, kernel_size (or kernel_h and kernel_w) cannot be specified");
166 }
167
168 kernel.resize(3);
169 kernel[0] = params.get<int>("kernel_d", 1);
170 kernel[1] = params.get<int>("kernel_h", 1);
171 kernel[2] = params.get<int>("kernel_w", 1);
172
173 for (int i = 0, j = globalPooling.size() - pads_begin.size(); i < pads_begin.size(); i++, j++) {
174 if ((pads_begin[i] != 0 || pads_end[i] != 0) && globalPooling[j])
175 CV_Error(cv::Error::StsBadArg, "In global_pooling mode, pads must be = 0");
176 }
177 for (int i = 0, j = globalPooling.size() - strides.size(); i < strides.size(); i++, j++) {
178 if (strides[i] != 1 && globalPooling[j])
179 CV_Error(cv::Error::StsBadArg, "In global_pooling mode, strides must be = 1");
180 }
181 }
182 else
183 {
184 util::getKernelSize(params, kernel);
185 util::getStrideAndPadding(params, pads_begin, pads_end, strides, padMode, kernel.size());
186 }

190 std::vector<size_t>& pads_end, std::vector<size_t>& strides,

204// From TensorFlow code:
205// Total padding on rows and cols is
206// Pr = (R' - 1) * S + Kr - R
207// Pc = (C' - 1) * S + Kc - C
208// where (R', C') are output dimensions, (R, C) are input dimensions, S
209// is stride, (Kr, Kc) are filter dimensions.
210// We pad Pr/2 on the left and Pr - Pr/2 on the right, Pc/2 on the top
211// and Pc - Pc/2 on the bottom. When Pr or Pc is odd, this means
212// we pad more on the right and bottom than on the top and left.
213void getConvPoolOutParams(const std::vector<int>& inp, const std::vector<size_t>& kernel,
214 const std::vector<size_t>& stride, const String &padMode,
215 const std::vector<size_t>& dilation, std::vector<int>& out)
216{
217 if (padMode == "VALID")
218 {
219 for (int i = 0; i < inp.size(); i++)
220 out.push_back((inp[i] - dilation[i] * (kernel[i] - 1) - 1 + stride[i]) / stride[i]);
221 }
222 else if (padMode == "SAME")
223 {
224 for (int i = 0; i < inp.size(); i++)
225 out.push_back((inp[i] - 1 + stride[i]) / stride[i]);
226 }
227 else
228 {
229 CV_Error(Error::StsError, "Unsupported padding mode");
230 }
231}

233void getConvPoolPaddings(const std::vector<int>& inp, const std::vector<size_t>& kernel,
234 const std::vector<size_t>& strides, const String &padMode,
235 std::vector<size_t>& pads_begin, std::vector<size_t>& pads_end)
236{
237 if (padMode == "SAME" || padMode == "VALID")
238 {
239 pads_begin.assign(kernel.size(), 0);
240 pads_end.assign(kernel.size(), 0);
241 }
242 if (padMode == "SAME")
243 {
244 CV_Assert_N(kernel.size() == strides.size(), kernel.size() == inp.size());
245 for (int i = 0; i < pads_begin.size(); i++) {
246 // There are test cases with stride > kernel.
247 if (strides[i] <= kernel[i])
248 {
249 int pad = (kernel[i] - 1 - (inp[i] - 1 + strides[i]) % strides[i]) / 2;
250 pads_begin[i] = pads_end[i] = pad;
251 }
252 }
253 }

256double getWeightScale(const Mat& weightsMat)
257{
258 double realMin, realMax;
259
260 cv::minMaxIdx(weightsMat, &realMin, &realMax);
261 realMin = std::min(realMin, 0.0);
262 realMax = std::max(realMax, 0.0);
263
264 return (realMax == realMin) ? 1.0 : std::max(-realMin, realMax)/127;
265}

267void tensorToIntVec(const Mat& tensor, std::vector<int>& vec)
268{
269 if (tensor.empty()) {
270 vec.clear();
271 } else {
272 int type = tensor.type();
273 CV_Assert(type == CV_32S || type == CV_64S);
274 // Accept tensors of any dimensionality; treat them as a flat vector.
275 CV_Assert(tensor.isContinuous());
276 int size = (int)tensor.total();
277 vec.resize(size);
278 if (type == CV_32S) {
279 const int* p = tensor.ptr<int>();
280 for (int i = 0; i < size; i++)
281 vec[i] = p[i];
282 } else {
283 const int64_t* p = tensor.ptr<int64_t>();
284 for (int i = 0; i < size; i++)
285 vec[i] = saturate_cast<int>(p[i]);
286 }
287 }

314void reshapeAndCopyFirst(InputArrayOfArrays inputs,
315 OutputArrayOfArrays outputs,
316 const MatShape& shape)
317{
318 int inpKind = inputs.kind(), outKind = outputs.kind();
319 CV_Assert(inpKind == outKind);
320 CV_Assert(inpKind == _InputArray::STD_VECTOR_MAT ||
321 inpKind == _InputArray::STD_VECTOR_UMAT);
322 CV_Assert(inputs.isContinuous(0));
323 int inpType = inputs.type(0);
324 if (inpKind == _InputArray::STD_VECTOR_MAT) {
325 Mat inp = inputs.getMat(0);
326 MatShape inpShape = inp.shape();
327 const size_t inpTotal = inpShape.total();
328 const size_t outTotal = shape.total();
329 std::vector<Mat>& outref = outputs.getMatVecRef();
330 outref.resize(1);
331 outref[0].fit(shape, inpType);
332 CV_Assert(outref[0].isContinuous());
333 if (inpTotal == 0 && outTotal == 0)
334 return;
335 Mat inp_ = inp.reshape(0, shape);
336 if (inp_.data != outref[0].data) {
337 // Parallel memcpy for large buffers to avoid single-thread bottleneck
338 // on reshape-style layers that don't get in-place-allocated.
339 CV_Assert(inp_.isContinuous());
340 CV_Assert(outref[0].isContinuous());
341 size_t bytes = inpTotal * inp_.elemSize();
342 const size_t CHUNK_BYTES = 64 * 1024;
343 if (bytes > 2 * CHUNK_BYTES) {
344 const uchar* src = inp_.data;
345 uchar* dst = outref[0].data;
346 int nChunks = (int)((bytes + CHUNK_BYTES - 1) / CHUNK_BYTES);
347 parallel_for_(Range(0, nChunks), [&](const Range& r) {
348 for (int i = r.start; i < r.end; i++) {
349 size_t off = (size_t)i * CHUNK_BYTES;
350 size_t len = std::min(CHUNK_BYTES, bytes - off);
351 memcpy(dst + off, src + off, len);
352 }
353 });
354 } else {
355 inp_.copyTo(outref[0]);
356 }
357 }
358 }
359 else {
360 UMat inp = inputs.getUMat(0);
361 MatShape inpShape = inputs.shape(0);
362 const size_t inpTotal = inpShape.total();
363 const size_t outTotal = shape.total();
364 std::vector<UMat>& outref = outputs.getUMatVecRef();
365 outref.resize(1);
366 outref[0].fit(shape, inpType);
367 CV_Assert(outref[0].isContinuous());
368 if (inpTotal == 0 && outTotal == 0)
369 return;
370 UMat inp_ = inp.reshape(0, shape);
371 inp_.copyTo(outref[0]);
372 }
373}

382void tensorToScalar(const Mat& tensor, int type, void* value)
383{
384 CV_Assert(tensor.total() == 1);
385 int type0 = tensor.type();
386 int depth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type);
387 CV_Assert(cn == 1);
388 double v = 0;
389 int64_t iv = 0;
390 bool isflt = type0 == CV_32F || type0 == CV_64F || type0 == CV_16F || type0 == CV_16BF;
391
392 if (type0 == CV_8U)
393 iv = *tensor.ptr<uint8_t>();
394 else if (type0 == CV_8S)
395 iv = *tensor.ptr<uint8_t>();
396 else if (type0 == CV_16U)
397 iv = *tensor.ptr<uint8_t>();
398 else if (type0 == CV_16S)
399 iv = *tensor.ptr<int16_t>();
400 else if (type0 == CV_32U)
401 iv = *tensor.ptr<uint32_t>();
402 else if (type0 == CV_32S)
403 iv = *tensor.ptr<int32_t>();
404 else if (type0 == CV_64S)
405 iv = *tensor.ptr<int64_t>();
406 else if (type0 == CV_32F)
407 v = *tensor.ptr<float>();
408 else if (type0 == CV_64F)
409 v = *tensor.ptr<double>();
410 else if (type0 == CV_16F)
411 v = (float)*tensor.ptr<hfloat>();
412 else if (type0 == CV_16BF)
413 v = (float)*tensor.ptr<bfloat>();
414 else if (type0 == CV_Bool)
415 iv = *tensor.ptr<uint8_t>() != 0;
416 else {
417 CV_Error_(Error::StsNotImplemented, ("type %s is not supported", typeToString(type0).c_str()));
418 }
419
420 if (depth == CV_8U)
421 *reinterpret_cast<uint8_t*>(value) = isflt ? saturate_cast<uint8_t>(v) : saturate_cast<uint8_t>(iv);
422 else if (depth == CV_8S)
423 *reinterpret_cast<int8_t*>(value) = isflt ? saturate_cast<int8_t>(v) : saturate_cast<int8_t>(iv);
424 else if (depth == CV_16U)
425 *reinterpret_cast<uint16_t*>(value) = isflt ? saturate_cast<uint16_t>(v) : saturate_cast<uint16_t>(iv);
426 else if (depth == CV_16S)
427 *reinterpret_cast<int16_t*>(value) = isflt ? saturate_cast<int16_t>(v) : saturate_cast<int16_t>(iv);
428 else if (depth == CV_32U)
429 *reinterpret_cast<uint32_t*>(value) = isflt ? saturate_cast<uint32_t>(v) : saturate_cast<uint32_t>(iv);
430 else if (depth == CV_32S)
431 *reinterpret_cast<int32_t*>(value) = isflt ? saturate_cast<int32_t>(v) : saturate_cast<int32_t>(iv);
432 else if (depth == CV_64U)
433 *reinterpret_cast<uint64_t*>(value) = isflt ? saturate_cast<uint64_t>(v) : saturate_cast<uint64_t>(iv);
434 else if (depth == CV_64S)
435 *reinterpret_cast<int64_t*>(value) = isflt ? saturate_cast<int64_t>(v) : iv;
436 else if (depth == CV_32F)
437 *reinterpret_cast<float*>(value) = isflt ? (float)v : saturate_cast<float>(iv);
438 else if (depth == CV_64F)
439 *reinterpret_cast<double*>(value) = isflt ? v : saturate_cast<double>(iv);
440 else if (depth == CV_16F)
441 *reinterpret_cast<hfloat*>(value) = isflt ? saturate_cast<hfloat>(v) : saturate_cast<hfloat>(iv);
442 else if (depth == CV_16BF)
443 *reinterpret_cast<bfloat*>(value) = isflt ? saturate_cast<bfloat>(v) : saturate_cast<bfloat>(iv);
444 else if (depth == CV_Bool)
445 *reinterpret_cast<uint8_t*>(value) = isflt ? (uint8_t)(v != 0) : (uint8_t)(iv != 0);
446 else {
447 CV_Error_(Error::StsNotImplemented, ("type %s is not supported", typeToString(depth).c_str()));
448 }
449}
