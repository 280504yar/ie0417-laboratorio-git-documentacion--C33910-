85static bool sumTemplate(InputArray _src, UMat & result)
86{
87 int type = _src.type(), depth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type);
88 int wdepth = CV_32F, wtype = CV_MAKE_TYPE(wdepth, cn);
89 size_t wgs = ocl::Device::getDefault().maxWorkGroupSize();
90
91 int wgs2_aligned = 1;
92 while (wgs2_aligned < (int)wgs)
93 wgs2_aligned <<= 1;
94 wgs2_aligned >>= 1;
95
96 char cvt[50];
97 ocl::Kernel k("calcSum", ocl::imgproc::match_template_oclsrc,
98 format("-D CALC_SUM -D T=%s -D T1=%s -D WT=%s -D cn=%d -D convertToWT=%s -D WGS=%d -D WGS2_ALIGNED=%d",
99 ocl::typeToStr(type), ocl::typeToStr(depth), ocl::typeToStr(wtype), cn,
100 ocl::convertTypeStr(depth, wdepth, cn, cvt, sizeof(cvt)),
101 (int)wgs, wgs2_aligned));
102 if (k.empty())
103 return false;
104
105 UMat src = _src.getUMat();
106 result.create(1, 1, CV_32FC1);
107
108 ocl::KernelArg srcarg = ocl::KernelArg::ReadOnlyNoSize(src),
109 resarg = ocl::KernelArg::PtrWriteOnly(result);
110
111 k.args(srcarg, src.cols, (int)src.total(), resarg);
112
113 size_t globalsize = wgs;
114 return k.run(1, &globalsize, &wgs, false);
115}

136void ConvolveBuf::create(Size image_size, Size templ_size)
137{
138 result_size = Size(image_size.width - templ_size.width + 1,
139 image_size.height - templ_size.height + 1);
140
141 const double blockScale = 4.5;
142 const int minBlockSize = 256;
143
144 block_size.width = cvRound(templ_size.width*blockScale);
145 block_size.width = std::max( block_size.width, minBlockSize - templ_size.width + 1 );
146 block_size.width = std::min( block_size.width, result_size.width );
147 block_size.height = cvRound(templ_size.height*blockScale);
148 block_size.height = std::max( block_size.height, minBlockSize - templ_size.height + 1 );
149 block_size.height = std::min( block_size.height, result_size.height );
150
151 dft_size.width = std::max(getOptimalDFTSize(block_size.width + templ_size.width - 1), 2);
152 dft_size.height = getOptimalDFTSize(block_size.height + templ_size.height - 1);
153 if( dft_size.width <= 0 || dft_size.height <= 0 )
154 CV_Error( cv::Error::StsOutOfRange, "the input arrays are too big" );
155
156 // recompute block size
157 block_size.width = dft_size.width - templ_size.width + 1;
158 block_size.width = std::min( block_size.width, result_size.width);
159 block_size.height = dft_size.height - templ_size.height + 1;
160 block_size.height = std::min( block_size.height, result_size.height );
161
162 image_block.create(dft_size, CV_32F);
163 templ_block.create(dft_size, CV_32F);
164 result_data.create(dft_size, CV_32F);
165
166 image_spect.create(dft_size.height, dft_size.width / 2 + 1, CV_32FC2);
167 templ_spect.create(dft_size.height, dft_size.width / 2 + 1, CV_32FC2);
168 result_spect.create(dft_size.height, dft_size.width / 2 + 1, CV_32FC2);
169
170 // Use maximum result matrix block size for the estimated DFT block size
171 block_size.width = std::min(dft_size.width - templ_size.width + 1, result_size.width);
172 block_size.height = std::min(dft_size.height - templ_size.height + 1, result_size.height);
173}

175static bool convolve_dft(InputArray _image, InputArray _templ, OutputArray _result)
176{
177 ConvolveBuf buf;
178 CV_Assert(_image.type() == CV_32F);
179 CV_Assert(_templ.type() == CV_32F);
180
181 buf.create(_image.size(), _templ.size());
182 _result.create(buf.result_size, CV_32F);
183
184 UMat image = _image.getUMat();
185 UMat templ = _templ.getUMat();
186
187 UMat result = _result.getUMat();
188
189 Size& block_size = buf.block_size;
190 Size& dft_size = buf.dft_size;
191
192 UMat& image_block = buf.image_block;
193 UMat& templ_block = buf.templ_block;
194 UMat& result_data = buf.result_data;
195
196 UMat& image_spect = buf.image_spect;
197 UMat& templ_spect = buf.templ_spect;
198 UMat& result_spect = buf.result_spect;
199
200 UMat templ_roi = templ;
201 copyMakeBorder(templ_roi, templ_block, 0, templ_block.rows - templ_roi.rows, 0,
202 templ_block.cols - templ_roi.cols, BORDER_ISOLATED);
203
204 dft(templ_block, templ_spect, 0, templ.rows);
205
206 // Process all blocks of the result matrix
207 for (int y = 0; y < result.rows; y += block_size.height)
208 {
209 for (int x = 0; x < result.cols; x += block_size.width)
210 {
211 Size image_roi_size(std::min(x + dft_size.width, image.cols) - x,
212 std::min(y + dft_size.height, image.rows) - y);
213 Rect roi0(x, y, image_roi_size.width, image_roi_size.height);
214
215 UMat image_roi(image, roi0);
216
217 copyMakeBorder(image_roi, image_block, 0, image_block.rows - image_roi.rows,
218 0, image_block.cols - image_roi.cols, BORDER_ISOLATED);
219
220 dft(image_block, image_spect, 0);
221
222 mulSpectrums(image_spect, templ_spect, result_spect, 0, true);
223
224 dft(result_spect, result_data, cv::DFT_INVERSE | cv::DFT_REAL_OUTPUT | cv::DFT_SCALE);
225
226 Size result_roi_size(std::min(x + block_size.width, result.cols) - x,
227 std::min(y + block_size.height, result.rows) - y);
228
229 Rect roi1(x, y, result_roi_size.width, result_roi_size.height);
230 Rect roi2(0, 0, result_roi_size.width, result_roi_size.height);
231
232 UMat result_roi(result, roi1);
233 UMat result_block(result_data, roi2);
234
235 result_block.copyTo(result_roi);
236 }
237 }
238 return true;
239}

241static bool convolve_32F(InputArray _image, InputArray _templ, OutputArray _result)
242{
243 _result.create(matchResultSize(_image.size(), _templ.size()), CV_32F);
244
245 if (_image.channels() == 1)
246 return(convolve_dft(_image, _templ, _result));
247 else
248 {
249 UMat image = _image.getUMat();
250 UMat templ = _templ.getUMat();
251 UMat result_(image.rows-templ.rows+1,(image.cols-templ.cols+1)*image.channels(), CV_32F);
252 bool ok = convolve_dft(image.reshape(1), templ.reshape(1), result_);
253 if (ok==false)
254 return false;
255 UMat result = _result.getUMat();
256 return (extractFirstChannel_32F(result_, _result, _image.channels()));
257 }

432static bool matchTemplate_CCOEFF_NORMED(InputArray _image, InputArray _templ, OutputArray _result)
433{
434 matchTemplate(_image, _templ, _result, cv::TM_CCORR);
435
436 UMat temp, image_sums, image_sqsums;
437 integral(_image, image_sums, image_sqsums, CV_32F, CV_32F);
438
439 int type = image_sums.type(), depth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type);
440 CV_Assert(cn >= 1 && cn <= 4);
441
442 ocl::Kernel k("matchTemplate_CCOEFF_NORMED", ocl::imgproc::match_template_oclsrc,
443 format("-D CCOEFF_NORMED -D T=%s -D T1=%s -D cn=%d", ocl::typeToStr(type), ocl::typeToStr(depth), cn));
444 if (k.empty())
445 return false;
446
447 UMat templ = _templ.getUMat();
448 Size size = _image.size(), tsize = templ.size();
449 _result.create(matchResultSize(size, tsize), CV_32F);
450 UMat result = _result.getUMat();
451
452 float scale = 1.f / tsize.area();
453
454 if (cn == 1)
455 {
456 float templ_sum = (float)sum(templ)[0];
457
458 multiply(templ, templ, temp, 1, CV_32F);
459 float templ_sqsum = (float)sum(temp)[0];
460
461 templ_sqsum -= scale * templ_sum * templ_sum;
462 templ_sum *= scale;
463
464 if (templ_sqsum < DBL_EPSILON)
465 {
466 result = Scalar::all(1);
467 return true;
468 }
469
470 k.args(ocl::KernelArg::ReadOnlyNoSize(image_sums), ocl::KernelArg::ReadOnlyNoSize(image_sqsums),
471 ocl::KernelArg::ReadWrite(result), templ.rows, templ.cols, scale, templ_sum, templ_sqsum);
472 }
473 else
474 {
475 Vec4f templ_sum = Vec4f::all(0), templ_sqsum = Vec4f::all(0);
476 templ_sum = sum(templ);
477
478 multiply(templ, templ, temp, 1, CV_32F);
479 templ_sqsum = sum(temp);
480
481 float templ_sqsum_sum = 0;
482 for (int i = 0; i < cn; i ++)
483 templ_sqsum_sum += templ_sqsum[i] - scale * templ_sum[i] * templ_sum[i];
484
485 templ_sum *= scale;
486
487 if (templ_sqsum_sum < DBL_EPSILON)
488 {
489 result = Scalar::all(1);
490 return true;
491 }
492
493 k.args(ocl::KernelArg::ReadOnlyNoSize(image_sums), ocl::KernelArg::ReadOnlyNoSize(image_sqsums),
494 ocl::KernelArg::ReadWrite(result), templ.rows, templ.cols, scale,
495 templ_sum, templ_sqsum_sum); }
496
497 size_t globalsize[2] = { (size_t)result.cols, (size_t)result.rows };
498 return k.run(2, globalsize, NULL, false);
499}

526void crossCorr( const Mat& img, const Mat& _templ, Mat& corr,
527 Point anchor, double delta, int borderType )
528{
529 const double blockScale = 4.5;
530 const int minBlockSize = 256;
531
532 Mat templ = _templ;
533 int depth = img.depth(), cn = img.channels();
534 int tdepth = templ.depth(), tcn = templ.channels();
535 int cdepth = corr.depth(), ccn = corr.channels();
536
537 CV_Assert( img.dims <= 2 && templ.dims <= 2 && corr.dims <= 2 );
538
539 if( depth != tdepth && tdepth != std::max(CV_32F, depth) )
540 {
541 _templ.convertTo(templ, std::max(CV_32F, depth));
542 tdepth = templ.depth();
543 }
544
545 CV_Assert( depth == tdepth || tdepth == CV_32F);
546 CV_Assert( corr.rows <= img.rows + templ.rows - 1 &&
547 corr.cols <= img.cols + templ.cols - 1 );
548
549 CV_Assert( ccn == 1 || delta == 0 );
550
551 int maxDepth = depth > CV_8S ? CV_64F : std::max(std::max(CV_32F, tdepth), cdepth);
552 Size blocksize, dftsize;
553
554 blocksize.width = cvRound(templ.cols*blockScale);
555 blocksize.width = std::max( blocksize.width, minBlockSize - templ.cols + 1 );
556 blocksize.width = std::min( blocksize.width, corr.cols );
557 blocksize.height = cvRound(templ.rows*blockScale);
558 blocksize.height = std::max( blocksize.height, minBlockSize - templ.rows + 1 );
559 blocksize.height = std::min( blocksize.height, corr.rows );
560
561 dftsize.width = std::max(getOptimalDFTSize(blocksize.width + templ.cols - 1), 2);
562 dftsize.height = getOptimalDFTSize(blocksize.height + templ.rows - 1);
563 if( dftsize.width <= 0 || dftsize.height <= 0 )
564 CV_Error( cv::Error::StsOutOfRange, "the input arrays are too big" );
565
566 // recompute block size

617 int tileCountY = (corr.rows + blocksize.height - 1)/blocksize.height;

689 src = dftImg(Rect(0, 0, bsz.width, bsz.height));

691 if( ccn > 1 )
692 {
693 if( cdepth != maxDepth )

695 Mat plane(bsz, cdepth, &buf[0]);

699 int pairs[] = {0, k};

706 else
707 {
708 if( maxDepth != cdepth )

710 Mat plane(bsz, cdepth, &buf[0]);
711 src.convertTo(plane, cdepth);
712 src = plane;
713 }
714 add(src, cdst, cdst);
715 }
716 }

735static void matchTemplateMask( InputArray _img, InputArray _templ, OutputArray _result, int method, InputArray _mask )
736{
737 CV_Assert(_mask.depth() == CV_8U || _mask.depth() == CV_Bool || _mask.depth() == CV_32F);
738 CV_Assert(_mask.channels() == _templ.channels() || _mask.channels() == 1);
739 CV_Assert(_templ.size() == _mask.size());
740 CV_Assert(_img.size().height >= _templ.size().height &&
741 _img.size().width >= _templ.size().width);
742
743 Mat img = _img.getMat(), templ = _templ.getMat(), mask = _mask.getMat();
744
745 if (img.depth() == CV_8U)
746 {
747 img.convertTo(img, CV_32F);
748 }
749 if (templ.depth() == CV_8U)
750 {
751 templ.convertTo(templ, CV_32F);
752 }
753 if (mask.depth() == CV_Bool)
754 {
755 mask.convertTo(mask, CV_32F);
756 }
757 else if (mask.depth() == CV_8U)
758 {
759 Mat maskBin;
760 threshold(mask, maskBin, 0/*threshold*/, 1.0/*maxVal*/, THRESH_BINARY);
761 maskBin.convertTo(mask, CV_32F);
762 }
763
764 Size corrSize = matchResultSize(img.size(), templ.size());
765 _result.create(corrSize, CV_32F);
766 Mat result = _result.getMat();
767
768 // If mask has only one channel, we repeat it for every image/template channel
769 if (templ.type() != mask.type())
770 {
771 // Assertions above ensured, that depth is the same and only number of channel differ
772 std::vector<Mat> maskChannels(templ.channels(), mask);
773 merge(maskChannels.data(), templ.channels(), mask);
774 }
775
776 if (method == cv::TM_SQDIFF || method == cv::TM_SQDIFF_NORMED)
777 {
778 Mat temp_result(corrSize, CV_32F);
779 Mat mask2 = mask.mul(mask);
780 // If the mul() is ever unnested, declare MatExpr, *not* Mat, to be more efficient.
781 // NORM_L2SQR calculates sum of squares
782 double templ2_mask2_sum = norm(templ.mul(mask), NORM_L2SQR);
783 crossCorrImgSq(img, mask2, temp_result);
784 crossCorr(img, templ.mul(mask2), result, Point(0,0), 0, 0);
785 // result and temp_result should not be switched, because temp_result is potentially needed
786 // for normalization.
787 result = -2 * result + temp_result + templ2_mask2_sum;
788
789 if (method == cv::TM_SQDIFF_NORMED)
790 {
791 sqrt(templ2_mask2_sum * temp_result, temp_result);
792 result /= temp_result;
793 }
794 }
795 else if (method == cv::TM_CCORR || method == cv::TM_CCORR_NORMED)
796 {
797 // If the mul() is ever unnested, declare MatExpr, *not* Mat, to be more efficient.
798 Mat templ_mask2 = templ.mul(mask.mul(mask));
799 crossCorr(img, templ_mask2, result, Point(0,0), 0, 0);
800
801 if (method == cv::TM_CCORR_NORMED)
802 {
803 Mat temp_result(corrSize, CV_32F);
804 Mat mask2 = mask.mul(mask);
805 // NORM_L2SQR calculates sum of squares
806 double templ2_mask2_sum = norm(templ.mul(mask), NORM_L2SQR);
807 crossCorrImgSq(img, mask2, temp_result);
808 sqrt(templ2_mask2_sum * temp_result, temp_result);
809 result /= temp_result;
810 }
811 }
812 else if (method == cv::TM_CCOEFF || method == cv::TM_CCOEFF_NORMED)
813 {
814 // Do mul() inline or declare MatExpr where possible, *not* Mat, to be more efficient.
815
816 Scalar mask_sum = sum(mask);
817 // T' * M where T' = M * (T - 1/sum(M)*sum(M*T))
818 Mat templx_mask = mask.mul(mask.mul(templ - sum(mask.mul(templ)).div(mask_sum)));
819 Mat img_mask_corr(corrSize, img.type()); // Needs separate channels
820
821 // CCorr(I, T'*M)
822 crossCorr(img, templx_mask, result, Point(0, 0), 0, 0);
823 // CCorr(I, M)
824 crossCorr(img, mask, img_mask_corr, Point(0, 0), 0, 0);
825
826 // CCorr(I', T') = CCorr(I, T'*M) - sum(T'*M)/sum(M)*CCorr(I, M)
827 // It does not matter what to use Mat/MatExpr, it should be evaluated to perform assign subtraction
828 Mat temp_res;
829 multiply(img_mask_corr, sum(templx_mask).div(mask_sum), temp_res);
830 result -= sumChannels(temp_res, result.rows);
831 if (method == cv::TM_CCOEFF_NORMED)
832 {
833 // norm(T')
834 double norm_templx = norm(mask.mul(templ - sum(mask.mul(templ)).div(mask_sum)),
835 NORM_L2);
836 // norm(I') = sqrt{ CCorr(I^2, M^2) - 2*CCorr(I, M^2)/sum(M)*CCorr(I, M)
837 // + sum(M^2)*CCorr(I, M)^2/sum(M)^2 }
838 // = sqrt{ CCorr(I^2, M^2)
839 // + CCorr(I, M)/sum(M)*{ sum(M^2) / sum(M) * CCorr(I,M)
840 // - 2 * CCorr(I, M^2) } }
841 Mat norm_imgx(corrSize, CV_32F);
842 Mat mask2 = mask.mul(mask);
843 Scalar mask2_sum = sum(mask2);
844 Mat img_mask2_corr(corrSize, img.type());
845 crossCorrImgSq(img, mask2, norm_imgx);
846 crossCorr(img, mask2, img_mask2_corr, Point(0,0), 0, 0);
847 Mat temp_res1;
848 multiply(img_mask_corr, Scalar(1.0, 1.0, 1.0, 1.0).div(mask_sum), temp_res1);
849 Mat temp_res2;
850 multiply(img_mask_corr, mask2_sum.div(mask_sum), temp_res2);
851 temp_res = temp_res1.mul(temp_res2 - 2 * img_mask2_corr);
852 norm_imgx += sumChannels(temp_res, result.rows);
853 sqrt(norm_imgx, norm_imgx);
854 result /= norm_imgx * norm_templx;
855 }
856 }

859static void common_matchTemplate( Mat& img, Mat& templ, Mat& result, int method, int cn )
860{
861 if( method == cv::TM_CCORR )
862 return;
863
864 int numType = method == cv::TM_CCORR || method == cv::TM_CCORR_NORMED ? 0 :
865 method == cv::TM_CCOEFF || method == cv::TM_CCOEFF_NORMED ? 1 : 2;
866 bool isNormed = method == cv::TM_CCORR_NORMED ||
867 method == cv::TM_SQDIFF_NORMED ||
868 method == cv::TM_CCOEFF_NORMED;
869
870 double invArea = 1./((double)templ.rows * templ.cols);
871
872 Mat sum, sqsum;
873 Scalar templMean, templSdv;
874 double *q0 = 0, *q1 = 0, *q2 = 0, *q3 = 0;
875 double templNorm = 0, templSum2 = 0;
876
877 if( method == cv::TM_CCOEFF )
878 {
879 integral(img, sum, CV_64F);
880 templMean = mean(templ);
881 }
882 else
883 {
884 integral(img, sum, sqsum, CV_64F);
885 meanStdDev( templ, templMean, templSdv );
886
887 templNorm = templSdv[0]*templSdv[0] + templSdv[1]*templSdv[1] + templSdv[2]*templSdv[2] + templSdv[3]*templSdv[3];
888
889 if( templNorm < DBL_EPSILON && method == cv::TM_CCOEFF_NORMED )
890 {
891 result = Scalar::all(1);
892 return;
893 }
894
895 templSum2 = templNorm + templMean[0]*templMean[0] + templMean[1]*templMean[1] + templMean[2]*templMean[2] + templMean[3]*templMean[3];
896
897 if( numType != 1 )
898 {
899 templMean = Scalar::all(0);
900 templNorm = templSum2;
901 }
902
903 templSum2 /= invArea;
904 templNorm = std::sqrt(templNorm);
905 templNorm /= std::sqrt(invArea); // care of accuracy here
906
907 CV_Assert(sqsum.data != NULL);
908 q0 = (double*)sqsum.data;
909 q1 = q0 + templ.cols*cn;
910 q2 = (double*)(sqsum.data + templ.rows*sqsum.step);
911 q3 = q2 + templ.cols*cn;
912 }
913
914 CV_Assert(sum.data != NULL);
915 double* p0 = (double*)sum.data;
916 double* p1 = p0 + templ.cols*cn;
917 double* p2 = (double*)(sum.data + templ.rows*sum.step);
918 double* p3 = p2 + templ.cols*cn;
919
920 int sumstep = sum.data ? (int)(sum.step / sizeof(double)) : 0;
921 int sqstep = sqsum.data ? (int)(sqsum.step / sizeof(double)) : 0;
922
923 int i, j, k;
924
925 for( i = 0; i < result.rows; i++ )
926 {
927 float* rrow = result.depth() == CV_32F ? result.ptr<float>(i) : nullptr;
928 double* drow = result.depth() == CV_64F ? result.ptr<double>(i) : nullptr;
929 int idx = i * sumstep;
930 int idx2 = i * sqstep;
931
932 for( j = 0; j < result.cols; j++, idx += cn, idx2 += cn )
933 {
934 double num = rrow ? (double)rrow[j] : drow[j], t;
935 double wndMean2 = 0, wndSum2 = 0;
936
937 if( numType == 1 )
938 {
939 for( k = 0; k < cn; k++ )
940 {
941 t = p0[idx+k] - p1[idx+k] - p2[idx+k] + p3[idx+k];
942 wndMean2 += t*t;
943 num -= t*templMean[k];
944 }
945
946 wndMean2 *= invArea;
947 }
948
949 if( isNormed || numType == 2 )
950 {
951 for( k = 0; k < cn; k++ )
952 {
953 t = q0[idx2+k] - q1[idx2+k] - q2[idx2+k] + q3[idx2+k];
954 wndSum2 += t;
955 }
956
957 if( numType == 2 )
958 {
959 num = wndSum2 - 2*num + templSum2;
960 num = MAX(num, 0.);
961 }
962 }
963
964 if( isNormed )
965 {
966 double diff2 = MAX(wndSum2 - wndMean2, 0);
967 if (diff2 <= std::min(0.5, 10 * FLT_EPSILON * wndSum2))
968 t = 0; // avoid rounding errors
969 else
970 t = std::sqrt(diff2)*templNorm;
971
972 if( fabs(num) < t )
973 num /= t;
974 else if( fabs(num) < t*1.125 )
975 num = num > 0 ? 1 : -1;
976 else
977 num = method != cv::TM_SQDIFF_NORMED ? 0 : 1;
978 }
979
980 if (rrow) rrow[j] = (float)num;
981 else drow[j] = num;
982 }
983 }
984}
