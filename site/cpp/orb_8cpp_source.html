130static void
131HarrisResponses(const Mat& img, const std::vector<Rect>& layerinfo,
132 std::vector<KeyPoint>& pts, int blockSize, float harris_k)
133{
134 CV_CheckTypeEQ(img.type(), CV_8UC1, "");
135 CV_CheckGT(blockSize, 0, "");
136 CV_CheckLE(blockSize*blockSize, 2048, "");
137
138 size_t ptidx, ptsize = pts.size();
139
140 const uchar* ptr00 = img.ptr<uchar>();
141 size_t size_t_step = img.step;
142 CV_CheckLE(size_t_step * blockSize + blockSize + 1, (size_t)INT_MAX, ""); // ofs computation, step+1
143 int step = static_cast<int>(size_t_step);
144
145 int r = blockSize/2;
146
147 float scale = 1.f/((1 << 2) * blockSize * 255.f);
148 float scale_sq_sq = scale * scale * scale * scale;
149
150 AutoBuffer<int> ofsbuf(blockSize*blockSize);
151 int* ofs = ofsbuf.data();
152 for( int i = 0; i < blockSize; i++ )
153 for( int j = 0; j < blockSize; j++ )
154 ofs[i*blockSize + j] = (int)(i*step + j);
155
156 for( ptidx = 0; ptidx < ptsize; ptidx++ )
157 {
158 int x0 = cvRound(pts[ptidx].pt.x);
159 int y0 = cvRound(pts[ptidx].pt.y);
160 int z = pts[ptidx].octave;
161
162 const uchar* ptr0 = ptr00 + (y0 - r + layerinfo[z].y)*size_t_step + (x0 - r + layerinfo[z].x);
163 int a = 0, b = 0, c = 0;
164
165 for( int k = 0; k < blockSize*blockSize; k++ )
166 {
167 const uchar* ptr = ptr0 + ofs[k];
168 int Ix = (ptr[1] - ptr[-1])*2 + (ptr[-step+1] - ptr[-step-1]) + (ptr[step+1] - ptr[step-1]);
169 int Iy = (ptr[step] - ptr[-step])*2 + (ptr[step-1] - ptr[-step-1]) + (ptr[step+1] - ptr[-step+1]);
170 a += Ix*Ix;
171 b += Iy*Iy;
172 c += Ix*Iy;
173 }
174 pts[ptidx].response = ((float)a * b - (float)c * c -
175 harris_k * ((float)a + b) * ((float)a + b))*scale_sq_sq;
176 }
177}

181static void ICAngles(const Mat& img, const std::vector<Rect>& layerinfo,
182 std::vector<KeyPoint>& pts, const std::vector<int> & u_max, int half_k)
183{
184 int step = (int)img.step1();
185 size_t ptidx, ptsize = pts.size();
186
187 for( ptidx = 0; ptidx < ptsize; ptidx++ )
188 {
189 const Rect& layer = layerinfo[pts[ptidx].octave];
190 const uchar* center = &img.at<uchar>(cvRound(pts[ptidx].pt.y) + layer.y, cvRound(pts[ptidx].pt.x) + layer.x);
191
192 int m_01 = 0, m_10 = 0;
193
194 // Treat the center line differently, v=0
195 for (int u = -half_k; u <= half_k; ++u)
196 m_10 += u * center[u];
197
198 // Go line by line in the circular patch
199 for (int v = 1; v <= half_k; ++v)
200 {
201 // Proceed over the two lines
202 int v_sum = 0;
203 int d = u_max[v];
204 for (int u = -d; u <= d; ++u)
205 {
206 int val_plus = center[u + v*step], val_minus = center[u - v*step];
207 v_sum += (val_plus - val_minus);
208 m_10 += u * (val_plus + val_minus);
209 }
210 m_01 += v * v_sum;
211 }
212
213 pts[ptidx].angle = fastAtan2((float)m_01, (float)m_10);
214 }
215}

219static void
220computeOrbDescriptors( const Mat& imagePyramid, const std::vector<Rect>& layerInfo,
221 const std::vector<float>& layerScale, std::vector<KeyPoint>& keypoints,
222 Mat& descriptors, const std::vector<Point>& _pattern, int dsize, int wta_k )
223{
224 int step = (int)imagePyramid.step;
225 int j, i, nkeypoints = (int)keypoints.size();
226
227 for( j = 0; j < nkeypoints; j++ )
228 {
229 const KeyPoint& kpt = keypoints[j];
230 const Rect& layer = layerInfo[kpt.octave];
231 float scale = 1.f/layerScale[kpt.octave];
232 float angle = kpt.angle;
233
234 angle *= (float)(CV_PI/180.f);
235 float a = (float)cos(angle), b = (float)sin(angle);
236
237 const uchar* center = &imagePyramid.at<uchar>(cvRound(kpt.pt.y*scale) + layer.y,
238 cvRound(kpt.pt.x*scale) + layer.x);
239 float x, y;
240 int ix, iy;
241 const Point* pattern = &_pattern[0];
242 uchar* desc = descriptors.ptr<uchar>(j);
243
244 #if 1
245 #define GET_VALUE(idx) \
246 (x = pattern[idx].x*a - pattern[idx].y*b, \
247 y = pattern[idx].x*b + pattern[idx].y*a, \
248 ix = cvRound(x), \
249 iy = cvRound(y), \
250 *(center + iy*step + ix) )
251 #else
252 #define GET_VALUE(idx) \
253 (x = pattern[idx].x*a - pattern[idx].y*b, \
254 y = pattern[idx].x*b + pattern[idx].y*a, \
255 ix = cvFloor(x), iy = cvFloor(y), \
256 x -= ix, y -= iy, \
257 cvRound(center[iy*step + ix]*(1-x)*(1-y) + center[(iy+1)*step + ix]*(1-x)*y + \
258 center[iy*step + ix+1]*x*(1-y) + center[(iy+1)*step + ix+1]*x*y))
259 #endif
260
261 if( wta_k == 2 )
262 {
263 for (i = 0; i < dsize; ++i, pattern += 16)
264 {
265 int t0, t1, val;
266 t0 = GET_VALUE(0); t1 = GET_VALUE(1);
267 val = t0 < t1;
268 t0 = GET_VALUE(2); t1 = GET_VALUE(3);
269 val |= (t0 < t1) << 1;
270 t0 = GET_VALUE(4); t1 = GET_VALUE(5);
271 val |= (t0 < t1) << 2;
272 t0 = GET_VALUE(6); t1 = GET_VALUE(7);
273 val |= (t0 < t1) << 3;
274 t0 = GET_VALUE(8); t1 = GET_VALUE(9);
275 val |= (t0 < t1) << 4;
276 t0 = GET_VALUE(10); t1 = GET_VALUE(11);
277 val |= (t0 < t1) << 5;
278 t0 = GET_VALUE(12); t1 = GET_VALUE(13);
279 val |= (t0 < t1) << 6;
280 t0 = GET_VALUE(14); t1 = GET_VALUE(15);
281 val |= (t0 < t1) << 7;
282
283 desc[i] = (uchar)val;
284 }
285 }
286 else if( wta_k == 3 )
287 {
288 for (i = 0; i < dsize; ++i, pattern += 12)
289 {
290 int t0, t1, t2, val;
291 t0 = GET_VALUE(0); t1 = GET_VALUE(1); t2 = GET_VALUE(2);
292 val = t2 > t1 ? (t2 > t0 ? 2 : 0) : (t1 > t0);
293
294 t0 = GET_VALUE(3); t1 = GET_VALUE(4); t2 = GET_VALUE(5);
295 val |= (t2 > t1 ? (t2 > t0 ? 2 : 0) : (t1 > t0)) << 2;
296
297 t0 = GET_VALUE(6); t1 = GET_VALUE(7); t2 = GET_VALUE(8);
298 val |= (t2 > t1 ? (t2 > t0 ? 2 : 0) : (t1 > t0)) << 4;
299
300 t0 = GET_VALUE(9); t1 = GET_VALUE(10); t2 = GET_VALUE(11);
301 val |= (t2 > t1 ? (t2 > t0 ? 2 : 0) : (t1 > t0)) << 6;
302
303 desc[i] = (uchar)val;
304 }
305 }
306 else if( wta_k == 4 )
307 {
308 for (i = 0; i < dsize; ++i, pattern += 16)
309 {
310 int t0, t1, t2, t3, u, v, k, val;
311 t0 = GET_VALUE(0); t1 = GET_VALUE(1);
312 t2 = GET_VALUE(2); t3 = GET_VALUE(3);
313 u = 0, v = 2;
314 if( t1 > t0 ) t0 = t1, u = 1;
315 if( t3 > t2 ) t2 = t3, v = 3;
316 k = t0 > t2 ? u : v;
317 val = k;
318
319 t0 = GET_VALUE(4); t1 = GET_VALUE(5);
320 t2 = GET_VALUE(6); t3 = GET_VALUE(7);
321 u = 0, v = 2;
322 if( t1 > t0 ) t0 = t1, u = 1;
323 if( t3 > t2 ) t2 = t3, v = 3;
324 k = t0 > t2 ? u : v;
325 val |= k << 2;
326
327 t0 = GET_VALUE(8); t1 = GET_VALUE(9);
328 t2 = GET_VALUE(10); t3 = GET_VALUE(11);
329 u = 0, v = 2;
330 if( t1 > t0 ) t0 = t1, u = 1;
331 if( t3 > t2 ) t2 = t3, v = 3;
332 k = t0 > t2 ? u : v;
333 val |= k << 4;
334
335 t0 = GET_VALUE(12); t1 = GET_VALUE(13);
336 t2 = GET_VALUE(14); t3 = GET_VALUE(15);
337 u = 0, v = 2;
338 if( t1 > t0 ) t0 = t1, u = 1;
339 if( t3 > t2 ) t2 = t3, v = 3;
340 k = t0 > t2 ? u : v;
341 val |= k << 6;
342
343 desc[i] = (uchar)val;
344 }
345 }
346 else
347 CV_Error( Error::StsBadSize, "Wrong wta_k. It can be only 2, 3 or 4." );
348 #undef GET_VALUE
349 }
350}

353static void initializeOrbPattern( const Point* pattern0, std::vector<Point>& pattern, int ntuples, int tupleSize, int poolSize )
354{
355 RNG rng(0x12345678);
356 int i, k, k1;
357 pattern.resize(ntuples*tupleSize);
358
359 for( i = 0; i < ntuples; i++ )
360 {
361 for( k = 0; k < tupleSize; k++ )
362 {
363 for(;;)
364 {
365 int idx = rng.uniform(0, poolSize);
366 Point pt = pattern0[idx];
367 for( k1 = 0; k1 < k; k1++ )
368 if( pattern[tupleSize*i + k1] == pt )
369 break;
370 if( k1 == k )
371 {
372 pattern[tupleSize*i + k] = pt;
373 break;
374 }
375 }
376 }
377 }
378}

826static void computeKeyPoints(const Mat& imagePyramid,
827 const UMat& uimagePyramid,
828 const Mat& maskPyramid,
829 const std::vector<Rect>& layerInfo,
830 const UMat& ulayerInfo,
831 const std::vector<float>& layerScale,
832 std::vector<KeyPoint>& allKeypoints,
833 int nfeatures, double scaleFactor,
834 int edgeThreshold, int patchSize, ORB::ScoreType scoreType,
835 bool useOCL, int fastThreshold )
836{
837#ifndef HAVE_OPENCL
838 CV_UNUSED(uimagePyramid);CV_UNUSED(ulayerInfo);CV_UNUSED(useOCL);
839#endif
840
841 int i, nkeypoints, level, nlevels = (int)layerInfo.size();
842 AutoBuffer<int> nfeaturesPerLevel(nlevels);
843
844 // fill the extractors and descriptors for the corresponding scales
845 float factor = (float)(1.0 / scaleFactor);
846 float ndesiredFeaturesPerScale = nfeatures*(1 - factor)/(1 - (float)std::pow((double)factor, (double)nlevels));
847
848 int sumFeatures = 0;
849 for( level = 0; level < nlevels-1; level++ )
850 {
851 nfeaturesPerLevel[level] = cvRound(ndesiredFeaturesPerScale);
852 sumFeatures += nfeaturesPerLevel[level];
853 ndesiredFeaturesPerScale *= factor;
854 }
855 nfeaturesPerLevel[nlevels-1] = std::max(nfeatures - sumFeatures, 0);
856
857 // Make sure we forget about what is too close to the boundary
858 //edge_threshold_ = std::max(edge_threshold_, patch_size_/2 + kKernelWidth / 2 + 2);
859
860 // pre-compute the end of a row in a circular patch
861 int halfPatchSize = patchSize / 2;
862 std::vector<int> umax(halfPatchSize + 2);
863
864 int v, v0, vmax = cvFloor(halfPatchSize * std::sqrt(2.f) / 2 + 1);
865 int vmin = cvCeil(halfPatchSize * std::sqrt(2.f) / 2);
866 for (v = 0; v <= vmax; ++v)
867 umax[v] = cvRound(std::sqrt((double)halfPatchSize * halfPatchSize - v * v));
868
869 // Make sure we are symmetric
870 for (v = halfPatchSize, v0 = 0; v >= vmin; --v)
871 {
872 while (umax[v0] == umax[v0 + 1])
873 ++v0;
874 umax[v] = v0;
875 ++v0;
876 }
877
878 allKeypoints.clear();
879 std::vector<KeyPoint> keypoints;
880 AutoBuffer<int> counters(nlevels);
881 keypoints.reserve(nfeaturesPerLevel[0]*2);
882
883 for( level = 0; level < nlevels; level++ )
884 {
885 int featuresNum = nfeaturesPerLevel[level];
886 Mat img = imagePyramid(layerInfo[level]);
887 Mat mask = maskPyramid.empty() ? Mat() : maskPyramid(layerInfo[level]);
888
889 // Detect FAST features, 20 is a good threshold
890 {
891 Ptr<FastFeatureDetector> fd = FastFeatureDetector::create(fastThreshold, true);
892 fd->detect(img, keypoints, mask);
893 }
894
895 // Remove keypoints very close to the border
896 KeyPointsFilter::runByImageBorder(keypoints, img.size(), edgeThreshold);
897
898 // Keep more points than necessary as FAST does not give amazing corners
899 KeyPointsFilter::retainBest(keypoints, scoreType == ORB_Impl::HARRIS_SCORE ? 2 * featuresNum : featuresNum);
900
901 nkeypoints = (int)keypoints.size();
902 counters[level] = nkeypoints;
903
904 float sf = layerScale[level];
905 for( i = 0; i < nkeypoints; i++ )
906 {
907 keypoints[i].octave = level;
908 keypoints[i].size = patchSize*sf;
909 }
910
911 std::copy(keypoints.begin(), keypoints.end(), std::back_inserter(allKeypoints));
912 }
913
914 std::vector<Vec3i> ukeypoints_buf;
915
916 nkeypoints = (int)allKeypoints.size();
917 if(nkeypoints == 0)
918 {
919 return;
920 }
921 Mat responses;
922 UMat ukeypoints, uresponses(1, nkeypoints, CV_32F);
923
924 // Select best features using the Harris cornerness (better scoring than FAST)
925 if( scoreType == ORB_Impl::HARRIS_SCORE )
926 {
927#ifdef HAVE_OPENCL
928 if( useOCL )
929 {
930 uploadORBKeypoints(allKeypoints, ukeypoints_buf, ukeypoints);
931 useOCL = ocl_HarrisResponses( uimagePyramid, ulayerInfo, ukeypoints,
932 uresponses, nkeypoints, 7, HARRIS_K );
933 if( useOCL )
934 {
935 CV_IMPL_ADD(CV_IMPL_OCL);
936 uresponses.copyTo(responses);
937 for( i = 0; i < nkeypoints; i++ )
938 allKeypoints[i].response = responses.at<float>(i);
939 }
940 }
941
942 if( !useOCL )
943#endif
944 HarrisResponses(imagePyramid, layerInfo, allKeypoints, 7, HARRIS_K);
945
946 std::vector<KeyPoint> newAllKeypoints;
947 newAllKeypoints.reserve(nfeaturesPerLevel[0]*nlevels);
948
949 int offset = 0;
950 for( level = 0; level < nlevels; level++ )
951 {
952 int featuresNum = nfeaturesPerLevel[level];
953 nkeypoints = counters[level];
954 keypoints.resize(nkeypoints);
955 std::copy(allKeypoints.begin() + offset,
956 allKeypoints.begin() + offset + nkeypoints,
957 keypoints.begin());
958 offset += nkeypoints;
959
960 //cull to the final desired level, using the new Harris scores.
961 KeyPointsFilter::retainBest(keypoints, featuresNum);
962
963 std::copy(keypoints.begin(), keypoints.end(), std::back_inserter(newAllKeypoints));
