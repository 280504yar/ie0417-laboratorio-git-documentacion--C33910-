13static String cat_string(const String& str)
14{
15 int left = 0, right = (int)str.length();
16 while( left < right && str[left] == ' ' )
17 left++;
18 while( right > left && str[right-1] == ' ' )
19 right--;
20 return left >= right ? String("") : str.substr(left, right-left);
21}
22}

75static bool parse_bool(std::string str)
76{
77 std::transform(str.begin(), str.end(), str.begin(), details::char_tolower);
78 std::istringstream is(str);
79 bool b;
80 is >> (str.size() > 1 ? std::boolalpha : std::noboolalpha) >> b;

84static void from_str(const String& str, Param type, void* dst)
85{
86 std::stringstream ss(str.c_str());
87 if( type == Param::INT )
88 ss >> *(int*)dst;
89 else if( type == Param::BOOLEAN )
90 {
91 std::string temp;
92 ss >> temp;
93 *(bool*) dst = parse_bool(temp);
94 }
95 else if( type == Param::UNSIGNED_INT )
96 ss >> *(unsigned*)dst;
97 else if( type == Param::UINT64 )
98 ss >> *(uint64*)dst;
99 else if( type == Param::FLOAT )
100 ss >> *(float*)dst;
101 else if( type == Param::REAL )
102 ss >> *(double*)dst;
103 else if( type == Param::STRING )
104 *(String*)dst = str;
105 else if( type == Param::SCALAR)
106 {
107 Scalar& scalar = *(Scalar*)dst;
108 for (int i = 0; i < 4 && !ss.eof(); ++i)
109 ss >> scalar[i];
110 }
111 else
112 CV_Error(Error::StsBadArg, "unknown/unsupported parameter type");
113
114 if (ss.fail())
115 {
116 CV_Error_(Error::StsBadArg, ("can not convert: [%s] to [%s]", str.c_str(), get_type_name(type)));
117 }

120void CommandLineParser::getByName(const String& name, bool space_delete, Param type, void* dst) const
121{
122 try
123 {
124 for (size_t i = 0; i < impl->data.size(); i++)
125 {
126 for (size_t j = 0; j < impl->data[i].keys.size(); j++)
127 {
128 if (name == impl->data[i].keys[j])
129 {
130 String v = impl->data[i].def_value;
131 if (space_delete)
132 v = cat_string(v);
133
134 // the key was neither specified nor has a default value
135 if((v.empty() && type != Param::STRING) || v == noneValue) {
136 impl->error = true;
137 impl->error_message = impl->error_message + "Missing parameter: '" + name + "'\n";
138 return;
139 }
140
141 from_str(v, type, dst);
142 return;
143 }
144 }
145 }
146 }
147 catch (const Exception& e)
148 {
149 impl->error = true;
150 impl->error_message = impl->error_message + "Parameter '"+ name + "': " + e.err + "\n";
151 return;
152 }
153
154 CV_Error_(Error::StsBadArg, ("undeclared key '%s' requested", name.c_str()));
155}

190static bool cmp_params(const CommandLineParserParams & p1, const CommandLineParserParams & p2)
191{
192 if (p1.number < p2.number)
193 return true;
194
195 if (p1.number > p2.number)
196 return false;
197
198 return p1.keys[0].compare(p2.keys[0]) < 0;
199}

201CommandLineParser::CommandLineParser(int argc, const char* const argv[], const String& keys)
202{
203 impl = new Impl;
204 impl->refcount = 1;
205
206 // path to application
207 size_t pos_s = String(argv[0]).find_last_of("/\\");
208 if (pos_s == String::npos)
209 {
210 impl->path_to_app = "";
211 impl->app_name = String(argv[0]);
212 }
213 else
214 {
215 impl->path_to_app = String(argv[0]).substr(0, pos_s);
216 impl->app_name = String(argv[0]).substr(pos_s + 1, String(argv[0]).length() - pos_s);
217 }
218
219 impl->error = false;
220 impl->error_message = "";
221
222 // parse keys
223 std::vector<String> k = impl->split_range_string(keys, '{', '}');
224
225 int jj = 0;
226 for (size_t i = 0; i < k.size(); i++)
227 {
228 std::vector<String> l = impl->split_string(k[i], '|', true);
229 CommandLineParserParams p;
230 p.keys = impl->split_string(l[0]);
231 p.def_value = l[1];
232 p.help_message = cat_string(l[2]);
233 p.number = -1;
234 if (p.keys.size() <= 0)
235 {
236 impl->error = true;
237 impl->error_message = "Field KEYS could not be empty\n";
238 }
239 else
240 {
241 if (p.keys[0][0] == '@')
242 {
243 p.number = jj;
244 jj++;
245 }
246
247 impl->data.push_back(p);
248 }
249 }
250
251 // parse argv
252 jj = 0;
253 for (int i = 1; i < argc; i++)
254 {
255 String s(argv[i]);
256 bool hasSingleDash = s.length() > 1 && s[0] == '-';
257
258 if (hasSingleDash)
259 {
260 bool hasDoubleDash = s.length() > 2 && s[1] == '-';
261 String key = s.substr(hasDoubleDash ? 2 : 1);
262 String value = "true";
263 size_t equalsPos = key.find('=');
264
265 if(equalsPos != String::npos) {
266 value = key.substr(equalsPos + 1);
267 key = key.substr(0, equalsPos);
268 }
269 impl->apply_params(key, value);
270 }
271 else
272 {
273 impl->apply_params(jj, s);
274 jj++;
275 }
276 }
277
278 impl->sort_params();
279}

399 printf("\n\n");
400
401 for (size_t i = 0; i < impl->data.size(); i++)
402 {
403 if (impl->data[i].number == -1)

405 printf("\t");
406 for (size_t j = 0; j < impl->data[i].keys.size(); j++)
407 {

409 if (k.length() > 1)

455{
456 String str = _str;
457 std::vector<String> vec;
458 String word = "";
459 bool begin = false;
460
461 while (!str.empty())
462 {
463 if (str[0] == fs)
464 {
465 if (begin == true)

481 }
482

500 }
501
502 if (begin == true)
503 {
504 word = word + str[0];

508

526std::vector<String> CommandLineParser::Impl::split_string(const String& _str, char symbol, bool create_empty_item) const
527{
528 String str = _str;
529 std::vector<String> vec;
530 String word = "";
531
532 while (!str.empty())
533 {
534 if (str[0] == symbol)
535 {
536 if (!word.empty() || create_empty_item)
537 {
538 vec.push_back(word);
539 word = "";
540 }
541 }
542 else
543 {
544 word = word + str[0];
545 }
546 str = str.substr(1, str.length() - 1);
547 }
548
549 if (word != "" || create_empty_item)
550 {
551 vec.push_back(word);
552 }
553
554 return vec;
555}
