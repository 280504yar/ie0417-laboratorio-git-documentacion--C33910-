795 const int ch = img.channels();
796 CV_Assert(!img.empty() && ch <= 4);
797
798 Scalar s;
799 int n = 100/delta;
800 int nR = std::min(n, (img.rows+1)/2), nC = std::min(n, (img.cols+1)/2);
801
802 int r, c, i;
803 for(r=0; r<nR; r++)
804 {
805 double k1 = r*delta/100., k2 = 1-k1;
806 for(c=0; c<img.cols; c++)
807 {
808 auto *p = img.ptr<T>(r, c);
809 for(i=0; i<ch; i++) s[i] = p[i];
810 s = s * k1 + color * k2;
811 for(i=0; i<ch; i++) p[i] = static_cast<T>((s[i]));
812 }
813 for(c=0; c<img.cols; c++)
814 {
815 auto *p = img.ptr<T>(img.rows-r-1, c);
816 for(i=0; i<ch; i++) s[i] = p[i];
817 s = s * k1 + color * k2;
818 for(i=0; i<ch; i++) p[i] = static_cast<T>((s[i]));
819 }
820 }
821
822 for(r=0; r<img.rows; r++)
823 {
824 for(c=0; c<nC; c++)
825 {
826 double k1 = c*delta/100., k2 = 1-k1;
827 auto *p = img.ptr<T>(r, c);

829 s = s * k1 + color * k2;
830 for(i=0; i<ch; i++) p[i] = static_cast<T>((s[i]));

832 for(c=0; c<n; c++)
