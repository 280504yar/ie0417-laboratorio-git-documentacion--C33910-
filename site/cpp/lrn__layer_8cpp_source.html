117 bool forward_ocl(InputArrayOfArrays inps, OutputArrayOfArrays outs, OutputArrayOfArrays internals)
118 {
119 std::vector<UMat> inputs;
120 std::vector<UMat> outputs;
121
122 bool use_half = (inps.depth() == CV_16F);
123 inps.getUMatVector(inputs);
124 outs.getUMatVector(outputs);
125
126 if (lrnOp.empty())
127 {
128 OCL4DNNLRNConfig config;
129 config.lrn_type = type == CHANNEL_NRM ?
130 LRNParameter_NormRegion_ACROSS_CHANNELS :
131 LRNParameter_NormRegion_WITHIN_CHANNEL;
132
133 CHECK_EQ(size % 2, 1)<< "LRN only supports odd values for local_size";
134 config.local_size = size;
135 config.alpha = alpha;
136 config.beta = beta;
137 config.k = bias;
138 CHECK_EQ(4, inputs[0].dims) << "Input must have 4 axes, "
139 << "corresponding to (num, channels, height, width)";
140 config.batch_size = inputs[0].size[0];
141 config.channels = inputs[0].size[1];
142 config.height = inputs[0].size[2];
143 config.width = inputs[0].size[3];
144 config.norm_by_size = normBySize;
145 config.use_half = use_half;
146
147 lrnOp = Ptr<OCL4DNNLRN<float> >(new OCL4DNNLRN<float>(config));
148 }
149
150 if (!lrnOp->Forward(inputs[0], outputs[0]))
151 return false;
152
153 return true;
154 }

157 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
158 {
159 CV_TRACE_FUNCTION();
160 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
161
162 CV_Assert(inputs_arr.total() == outputs_arr.total());
163
164 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
165 forward_ocl(inputs_arr, outputs_arr, internals_arr))
166
167 if (inputs_arr.depth() == CV_16F)
168 {
169 forward_fallback(inputs_arr, outputs_arr, internals_arr);
170 return;
171 }
172
173 std::vector<Mat> inputs, outputs;
174 inputs_arr.getMatVector(inputs);
175 outputs_arr.getMatVector(outputs);
176
177 CV_Assert(inputs.size() == outputs.size());
178
179 for (int i = 0; i < inputs.size(); i++)
180 {
181 CV_Assert(inputs[i].dims == 4);
182
183 Mat &src = inputs[i];
184 Mat &dst = outputs[i];
185
186 switch (type)
187 {
188 case CHANNEL_NRM:
189 channelNormalization(src, dst);
190 break;
191 case SPATIAL_NRM:
192 spatialNormalization(src, dst);
193 break;
194 default:
195 CV_Error(Error::StsNotImplemented, "Unimplemented mode of LRN layer");
196 break;
197 }
198 }

201 class ChannelLRN : public ParallelLoopBody
202 {
203 public:
204 ChannelLRN(const float* src, float* dst, int channels, int ksize,
205 float alpha1, float bias1, float beta1,
206 size_t planeSize, int nsamples, int nstripes)
207 {
208 src_ = src; dst_ = dst;
209 channels_ = channels;
210 ksize_ = ksize;
211 alpha1_ = alpha1; bias1_ = bias1; beta1_ = beta1;
212 planeSize_ = planeSize; nsamples_ = nsamples; nstripes_ = nstripes;
213 }
214
215 void operator()(const Range& r) const CV_OVERRIDE
216 {
217 int nsamples = nsamples_, nstripes = nstripes_;
218 size_t planeSize = planeSize_, planeSize_n = planeSize * nsamples;
219 size_t elemsPerStripe = (planeSize_n + nstripes - 1)/nstripes;
220 size_t rstart = r.start*elemsPerStripe;
221 size_t rend = r.end == nstripes ? planeSize_n : r.end*elemsPerStripe;
222 rstart = std::min(rstart, planeSize_n);
223 rend = std::min(rend, planeSize_n);
224 float alpha1 = alpha1_, bias1 = bias1_, beta1 = beta1_;
225 int k, channels = channels_, ksize = ksize_;
226
227 AutoBuffer<float> buf_((channels + ksize + 1)*2);
228 float* acc = buf_.data();
229 float* buf = acc + channels + ksize + 1;
230 for( k = 0; k <= ksize; k++ )
231 buf[-k-1] = buf[channels + k] = 0.f;
232
233 for( size_t ofs = rstart; ofs < rend; )
234 {
235 int sampleIdx = (int)(ofs/planeSize);
236 if( sampleIdx >= nsamples )
237 break;
238 size_t ofs0 = ofs - sampleIdx*planeSize;
239 size_t ofs1 = std::min(planeSize - ofs0, rend - ofs) + ofs;
240 const float* src = src_ + sampleIdx*planeSize*channels + ofs0;
241 float* dst = dst_ + sampleIdx*planeSize*channels + ofs0;
242
243 for( ; ofs < ofs1; ofs++, src++, dst++ )
244 {
245 for( k = 0; k < channels; k++ )
246 buf[k] = src[k*planeSize];
247 float s = 0;
248 for( k = 0; k < ksize; k++ )
249 s += buf[k]*buf[k];
250 for( k = 0; k < channels; k++ )
251 {
252 float x1 = buf[k + ksize];
253 float x0 = buf[k - ksize - 1];
254 s = std::max(s + (x1 + x0)*(x1 - x0), 0.f);
255 acc[k] = (float)(alpha1*s + bias1);
256 }
257
258 hal::log32f(acc, acc, channels);
259 for( k = 0; k < channels; k++ )
260 acc[k] *= beta1;
261 hal::exp32f(acc, acc, channels);
262
263 for( k = 0; k < channels; k++ )
264 dst[k*planeSize] = buf[k]*acc[k];
265 }
266 }
267 }
268
269 const float* src_;
270 float* dst_;
271 float alpha1_, bias1_, beta1_;
272 size_t planeSize_;
273 int channels_, ksize_, nsamples_, nstripes_;
274 };

276 void channelNormalization(Mat &srcBlob, Mat &dstBlob)
277 {
278 int num = srcBlob.size[0];
279 int channels = srcBlob.size[1];
280 int ksize = (size - 1) / 2;
281 int sizeNormFactor = normBySize ? size : 1;
282 size_t planeSize = srcBlob.size[2]*srcBlob.size[3];
283
284 int nstripes = std::max(getNumThreads(), 1);
285
286 ChannelLRN clrn(srcBlob.ptr<float>(), dstBlob.ptr<float>(), channels,
287 ksize, alpha/sizeNormFactor, bias, -beta, planeSize, num, nstripes);
288 parallel_for_(Range(0, nstripes), clrn, nstripes);
289 }
290
291 void sqrBoxFilter_(const Mat &src, Mat &dst)
292 {
293 Mat srcRawWrapper(src.rows, src.cols, src.type(), src.data, src.step[0]);
294 cv::sqrBoxFilter(srcRawWrapper, dst, dst.depth(), Size(size, size), Point(-1, -1), false, BORDER_CONSTANT);
295 }
296
297 void spatialNormalization(Mat &srcBlob, Mat &dstBlob)
298 {
299 int num = srcBlob.size[0];
300 int channels = srcBlob.size[1];
301 int sizeNormFactor = normBySize ? size*size : 1;
302
303 Mat srcMat = srcBlob;
304 Mat dstMat = dstBlob;
305
306 for (int n = 0; n < num; n++)
307 {
308 for (int cn = 0; cn < channels; cn++)
309 {
310 Mat src = getPlane(srcMat, n, cn);
311 Mat dst = getPlane(dstMat, n, cn);
312
313 sqrBoxFilter_(src, dst);
314
315 dst.convertTo(dst, dst.type(), alpha/sizeNormFactor, bias);
316 cv::pow(dst, beta, dst);
317 cv::divide(src, dst, dst);
318 }
319 }

323 Ptr<BackendNode> initCUDA(
324 void *context_,
325 const std::vector<Ptr<BackendWrapper>>& inputs,
326 const std::vector<Ptr<BackendWrapper>>& outputs
327 ) override
328 {
329 auto context = reinterpret_cast<csl::CSLContext*>(context_);
330
331 cuda4dnn::LRNType type_;
332 if (type == CHANNEL_NRM)
333 type_ = cuda4dnn::LRNType::ACROSS_CHANNELS;
334 else if (type == SPATIAL_NRM)
335 type_ = cuda4dnn::LRNType::WITHIN_CHANNEL;
336 else
337 CV_Error(Error::StsNotImplemented, "Unknown normalization region");
338
339 float alphaSize = alpha;
340 if (!normBySize) {
341 switch (type) {
342 case CHANNEL_NRM: alphaSize = alpha * size; break;
343 case SPATIAL_NRM: alphaSize = alpha * size * size; break;
344 }
345 }
346
347 std::size_t largestInputSize = 0;
348 for(auto& wrapper : inputs) {
349 auto input_wrapper = wrapper.dynamicCast<CUDABackendWrapper>();
350 auto shape = input_wrapper->getShape();
351 largestInputSize = std::max<std::size_t>(
352 largestInputSize,
353 std::accumulate(std::begin(shape), std::end(shape), 1, std::multiplies<int>())
354 );
355 }
356
357 return make_cuda_node<cuda4dnn::LRNOp>(preferableTarget,
358 std::move(context->cudnn_handle), type_, size, alphaSize, beta, bias, largestInputSize);
359 }

417 virtual int64 getFLOPS(const std::vector<MatShape> &inputs,
418 const std::vector<MatShape> &outputs) const CV_OVERRIDE
419 {
420 CV_UNUSED(outputs); // suppress unused variable warning
421 CV_Assert(inputs.size() > 0);
422 long flops = 0;
423
424 for(int i = 0; i < inputs.size(); i++)
425 {
426 if (type == CHANNEL_NRM)
427 {
428 int channels = inputs[i][1];
429 int ksize = (size - 1) / 2;
430
431 flops += inputs[i][0]*(std::min(ksize, channels)*2*total(inputs[i], 2) + channels*4*total(inputs[i], 2));
432
433 if (ksize < channels)
434 {
435 flops += (size + 2*(channels - size))*total(inputs[i], 2);
436 }
437 }
438 else
439 {
440 flops += total(inputs[i])*(2*size*size + 2);
441 }
442 }
443 return flops;
444 }
