15void
16normalEstimate(OutputArray normals, OutputArray curvatures, InputArray input_pts_,
17 InputArrayOfArrays nn_idx_, int max_neighbor_num_)
18{
19 Mat ori_pts;
20 getPointsMatFromInputArray(input_pts_, ori_pts, 0);
21 int pts_size = ori_pts.rows;
22
23 std::vector<Mat> nn_idx;
24 nn_idx_.getMatVector(nn_idx);
25
26 CV_CheckEQ((int) nn_idx.size(), pts_size,
27 "The point number of NN search result should be equal to the size of the point cloud.");
28
29 if (normals.channels() == 3 && normals.isVector())
30 {
31 // std::vector<cv::Point3f>
32 normals.create(1, pts_size, CV_32FC3);
33 }
34 else
35 {
36 // cv::Mat
37 normals.create(pts_size, 3, CV_32F);
38 }
39
40 curvatures.create(pts_size, 1, CV_32F);
41
42 int max_neighbor_num = max_neighbor_num_ <= 0 ? INT_MAX : max_neighbor_num_;
43
44 float *normals_ptr = (float *) normals.getMat().data;
45 float *curvatures_ptr = (float *) curvatures.getMat().data;
46
47 parallel_for_(Range(0, pts_size), [&](const Range &range) {
48 // Index of current nearest neighbor point
49 int cur_nei_idx;
50 for (int i = range.start; i < range.end; ++i)
51 {
52 // The maximum size that may be used for this row
53 int bound = max_neighbor_num > nn_idx[i].cols ? nn_idx[i].cols : max_neighbor_num;
54 const int *nn_idx_ptr_base = (int *) nn_idx[i].data;
55 // The first point should be itself
56 Mat pt_set(ori_pts.row(i));
57 // Push the nearest neighbor points to pt_set
58 for (int j = 1; j < bound; ++j)
59 {
60 cur_nei_idx = nn_idx_ptr_base[j];
61 // If the index is less than 0,
62 // the nn_idx of this row will no longer have any information.
63 if (cur_nei_idx < 0) break;
64 pt_set.push_back(ori_pts.row(cur_nei_idx));
65 }
66
67 Mat mean;
68 // Calculate the mean of point set, use "reduce()" is faster than default method in PCA
69 reduce(pt_set, mean, 0, REDUCE_AVG);
70 // Use PCA to get eigenvalues and eigenvectors of pt_set
71 PCA pca(pt_set, mean, PCA::DATA_AS_ROW);
72
73 const float *eigenvectors_ptr = (float *) pca.eigenvectors.data;
74 float *normals_ptr_base = normals_ptr + 3 * i;
75 normals_ptr_base[0] = eigenvectors_ptr[6];
76 normals_ptr_base[1] = eigenvectors_ptr[7];
77 normals_ptr_base[2] = eigenvectors_ptr[8];
78
79 const float *eigenvalues_ptr = (float *) pca.eigenvalues.data;
80 curvatures_ptr[i] = eigenvalues_ptr[2] /
81 (eigenvalues_ptr[0] + eigenvalues_ptr[1] + eigenvalues_ptr[2]);
82 }
83 });
84}
