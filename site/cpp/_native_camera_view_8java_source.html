92 private boolean initializeCamera(int width, int height) {
93 synchronized (this) {
94 Camera.CameraInfo cameraInfo = new Camera.CameraInfo();
95 int localCameraIndex = mCameraIndex;
96 if (mCameraIndex == CAMERA_ID_ANY) {
97 Log.d(TAG, "Try to open default camera");
98 localCameraIndex = 0;
99 } else if (mCameraIndex == CAMERA_ID_BACK) {
100 Log.i(TAG, "Trying to open back camera");
101 for (int camIdx = 0; camIdx < Camera.getNumberOfCameras(); ++camIdx) {
102 Camera.getCameraInfo( camIdx, cameraInfo );
103 if (cameraInfo.facing == Camera.CameraInfo.CAMERA_FACING_BACK) {
104 localCameraIndex = camIdx;
105 break;
106 }
107 }
108 } else if (mCameraIndex == CAMERA_ID_FRONT) {
109 Log.i(TAG, "Trying to open front camera");
110 for (int camIdx = 0; camIdx < Camera.getNumberOfCameras(); ++camIdx) {
111 Camera.getCameraInfo( camIdx, cameraInfo );
112 if (cameraInfo.facing == Camera.CameraInfo.CAMERA_FACING_FRONT) {
113 localCameraIndex = camIdx;
114 break;
115 }
116 }
117 }
118
119 if (localCameraIndex == CAMERA_ID_BACK) {
120 Log.e(TAG, "Back camera not found!");
121 return false;
122 } else if (localCameraIndex == CAMERA_ID_FRONT) {
123 Log.e(TAG, "Front camera not found!");
124 return false;
125 }
126
127 MatOfInt params = new MatOfInt(Videoio.CAP_PROP_FRAME_WIDTH, width,
128 Videoio.CAP_PROP_FRAME_HEIGHT, height);
129
130 Log.d(TAG, "Try to open camera with index " + localCameraIndex);
131 mCamera = new VideoCapture(localCameraIndex, Videoio.CAP_ANDROID, params);
132
133 if (mCamera == null)
134 return false;
135 if (mCamera.isOpened() == false)
136 return false;
137
138 if (mCameraIndex != CAMERA_ID_BACK && mCameraIndex != CAMERA_ID_FRONT)
139 Camera.getCameraInfo(localCameraIndex, cameraInfo);
140 int frameRotation = getFrameRotation(
141 cameraInfo.facing == Camera.CameraInfo.CAMERA_FACING_FRONT,
142 cameraInfo.orientation);
143
144 mFrame = new RotatedCameraFrame(new NativeCameraFrame(mCamera), frameRotation);
145
146 if (frameRotation % 180 == 0) {
147 mFrameWidth = (int) mCamera.get(Videoio.CAP_PROP_FRAME_WIDTH);
148 mFrameHeight = (int) mCamera.get(Videoio.CAP_PROP_FRAME_HEIGHT);
149 } else {
150 mFrameWidth = (int) mCamera.get(Videoio.CAP_PROP_FRAME_HEIGHT);
151 mFrameHeight = (int) mCamera.get(Videoio.CAP_PROP_FRAME_WIDTH);
152 }
153
154 if ((getLayoutParams().width == LayoutParams.MATCH_PARENT) && (getLayoutParams().height == LayoutParams.MATCH_PARENT))
155 mScale = Math.min(((float)height)/mFrameHeight, ((float)width)/mFrameWidth);
156 else
157 mScale = 0;
158
159 if (mFpsMeter != null) {
160 mFpsMeter.setResolution(mFrameWidth, mFrameHeight);
161 }
162
163 AllocateCache();
164 }
165
166 Log.i(TAG, "Selected camera frame size = (" + mFrameWidth + ", " + mFrameHeight + ")");
167
168 return true;
169 }
