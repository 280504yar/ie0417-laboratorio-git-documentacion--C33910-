7// Folds a Transpose-of-last-two-dims into the consuming MatMul's trans_a / trans_b.
8//
9// Pattern:
10// X -> Transpose(perm=[0,..,n-3, n-1, n-2]) -> MatMul(., Y)
11// => X -> MatMul(trans_a=true, ., Y)

13// Symmetric on the second operand -> trans_b.

30 static bool isLastTwoSwap(const vector<int>& perm)
31 {
32 int n = (int)perm.size();
33 if (n < 2) return false;
34 for (int i = 0; i < n - 2; i++)
35 if (perm[i] != i) return false;
36 return perm[n - 2] == n - 1 && perm[n - 1] == n - 2;
37 }

39 bool fuseGraph(Ptr<Graph>& graph)
40 {
41 const vector<Ptr<LayerInfo>>& prog = graph->prog();
42 size_t nops = prog.size();
43 bool modified = false;
44
45 for (size_t i = 0; i < nops; i++) {
46 if (!prog[i]) continue;
47 vector<Ptr<Graph>>* subgraphs = prog[i]->subgraphs();
48 if (subgraphs) {
49 for (Ptr<Graph>& g : *subgraphs)
50 if (fuseGraph(g)) modified = true;
51 }
52 }
53
54 vector<int> usecounts;
55 netimpl->useCounts(usecounts);
56
57 std::set<int> externalArgs;
58 for (Arg out : graph->outputs())
59 externalArgs.insert(out.idx);
60
61 std::map<int, int> producer;
62 for (size_t i = 0; i < nops; i++) {
63 if (!prog[i]) continue;
64 for (Arg out : prog[i]->outputs)
65 producer[out.idx] = (int)i;
66 }
67
68 vector<bool> dropped(nops, false);
69
70 for (size_t i = 0; i < nops; i++) {
71 const Ptr<LayerInfo>& layer = prog[i];
72 if (!layer || dropped[i]) continue;
73
74 MatMulLayer* mm = dynamic_cast<MatMulLayer*>(layer.get());
75 if (!mm) continue;
76
77 if (layer->inputs.size() != 2) continue;
78
79 for (int slot = 0; slot < 2; slot++) {
80 Arg in = layer->inputs[slot];
81 auto it = producer.find(in.idx);
82 if (it == producer.end()) continue;
83 int prod_idx = it->second;
84 if (prod_idx < 0 || dropped[prod_idx]) continue;
85
86 const Ptr<LayerInfo>& pl = prog[prod_idx];
87 TransposeLayer* tr = dynamic_cast<TransposeLayer*>(pl.get());
88 if (!tr || pl->outputs.size() != 1) continue;
89 if (!isLastTwoSwap(tr->perm)) continue;
90
91 bool single_consumer = usecounts[in.idx] == 1
92 && externalArgs.count(in.idx) == 0;
93 if (!single_consumer) continue;
94
95 if (slot == 0) mm->trans_a = !mm->trans_a;
96 else mm->trans_b = !mm->trans_b;
97
98 layer->inputs[slot] = pl->inputs[0];
99 dropped[prod_idx] = true;
100 usecounts[in.idx] = 0;
101 modified = true;
102 }
103 }
104
105 if (modified) {
106 vector<Ptr<LayerInfo>> newprog;
107 newprog.reserve(nops);
108 for (size_t i = 0; i < nops; i++) {
109 if (!dropped[i] && prog[i])
110 newprog.push_back(prog[i]);
111 }
112 graph->setProg(newprog);
113 }
114 return modified;
115 }
