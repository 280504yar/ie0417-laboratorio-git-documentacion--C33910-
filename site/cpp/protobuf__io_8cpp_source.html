111static const int kProtoReadBytesLimit = INT_MAX; // Max size of 2 GB minus 1 byte.

113static bool ReadProtoFromBinary(ZeroCopyInputStream* input, MessageLite *proto) {
114 CodedInputStream coded_input(input);
115#if defined(GOOGLE_PROTOBUF_VERSION) && GOOGLE_PROTOBUF_VERSION < 3006000
116 coded_input.SetTotalBytesLimit(kProtoReadBytesLimit, 536870912);
117#else
118 coded_input.SetTotalBytesLimit(kProtoReadBytesLimit);
119#endif
120
121 return proto->ParseFromCodedStream(&coded_input);
122}
