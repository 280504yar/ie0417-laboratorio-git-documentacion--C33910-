26def getParameter(pos, tokens):
27 deep = 0
28 p = []
29 while True:
30 if pos >= len(tokens):
31 break
32 if (tokens[pos] == ')' or tokens[pos] == ',') and deep == 0:
33 if tokens[pos] == ')':
34 pos = len(tokens)
35 else:
36 pos += 1
37 break
38 if tokens[pos] == '(':
39 deep += 1
40 if tokens[pos] == ')':
41 deep -= 1
42 p.append(tokens[pos])
43 pos += 1
44 return (' '.join(p), pos)

47def getParameters(i, tokens):
48 assert tokens[i] == '('
49 i += 1
50
51 params = []
52 while True:
53 if i >= len(tokens) or tokens[i] == ')':
54 break
55
56 (param, i) = getParameter(i, tokens)
57 if len(param) > 0:
58 params.append(param)
59 else:
60 assert False

68def postProcessParameters(fns):
69 fns.sort(key=lambda x: x['name'])

73 p = fn['params'][i]
