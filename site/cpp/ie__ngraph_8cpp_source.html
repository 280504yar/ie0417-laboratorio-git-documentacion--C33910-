109class NgraphCustomOp: public ov::op::Op {
110public:
111 OPENVINO_OP(kOpenCVLayersType);
112
113 NgraphCustomOp(const ov::OutputVector& inputs, Ptr<Layer>& cvLayer, const std::vector<Mat>& outputs, const std::vector<Mat>& internals):
114 Op(inputs), cvLayer(cvLayer), outputs(outputs), internals(internals)
115 {
116 constructor_validate_and_infer_types();
117 }
118
119 void validate_and_infer_types() override
120 {
121 std::vector<MatType> inputTypes(get_input_size());
122 std::vector<MatType> internalTypes;
123 std::vector<MatType> outputTypes;
124 for (int i = 0; i < get_input_size(); ++i)
125 {
126 inputTypes[i] = ovTypeToCvType(get_input_element_type(i));
127 }
128 cvLayer->getTypes(inputTypes, outputs.size(), internals.size(), outputTypes, internalTypes);
129 for (int i = 0; i < internals.size(); ++i) {
130 if (internals[i].depth() != internalTypes[i])
131 internals[i] = cv::Mat(shape(internals[i]), internalTypes[i]);
132 }
133
134 set_output_size(outputs.size());
135 for (int i = 0; i < outputs.size(); ++i)
136 {
137 ov::PartialShape shape;
138 for (int j = 0; j < outputs[i].dims; ++j) {
139 shape.push_back(outputs[i].size[j]);
140 }
141 set_output_type(i, cvTypeToOvType(outputTypes[i]), shape);
142 }
143 }
144
145 std::shared_ptr<ov::Node> clone_with_new_inputs(const ov::OutputVector& new_args) const override
146 {
147 return std::make_shared<NgraphCustomOp>(new_args, cvLayer, outputs, internals);
148 }
149
150 bool has_evaluate() const {
151 return true;
152 }
153
154 bool evaluate(ov::TensorVector& outputs, const ov::TensorVector& inputs) const override {
155 std::vector<Mat> inpMats, outMats;
156 infEngineBlobsToMats(inputs, inpMats);
157 infEngineBlobsToMats(outputs, outMats);
158 try
159 {
160 cvLayer->forward(inpMats, outMats, internals);
161 return true;
162 }
163 catch (...)
164 {
165 return false;
166 }
167 }
168
169 Ptr<Layer>& cvLayer;
170 std::vector<Mat> outputs, internals;
171};

225void InfEngineNgraphNet::createNet(Target targetId) {
226 if (!hasNetOwner)
227 {
228 CV_Assert(!requestedOutputs.empty());
229 ov::ResultVector outs;
230
231 for (auto output_node_it = requestedOutputs.begin(); output_node_it != requestedOutputs.end(); ++output_node_it)
232 {
233 CV_LOG_DEBUG(NULL, "DNN/NGRAPH: Add 'Result' output: " << output_node_it->first);
234 CV_Assert(output_node_it->second);
235 auto out = std::make_shared<ov::op::v0::Result>(output_node_it->second->node);
236 std::string name = output_node_it->first + (output_node_it->second->node.get_node()->get_output_size() == 1 ? "" : ".0");
237 CV_LOG_DEBUG(NULL, "DNN-IE: Change friendly name from " << out->get_friendly_name() << " to " << name);
238 out->set_friendly_name(name);
239 outs.push_back(out);
240 }
241 CV_Assert_N(!inputs_vec.empty(), !outs.empty());
242 ngraph_function = std::make_shared<ov::Model>(outs, inputs_vec);
243 init(targetId);
244 }

247void InfEngineNgraphNet::init(Target targetId)
248{
249 if (!hasNetOwner)
250 {
251 if (targetId == DNN_TARGET_OPENCL_FP16)
252 {
253 ov::pass::ConvertFP32ToFP16().run_on_model(ngraph_function);
254 }
255 cnn = ngraph_function;
256
257 if (DNN_IE_SERIALIZE)
258 {
259#ifndef OPENCV_DNN_DISABLE_NETWORK_AUTO_DUMP
260 std::string dumpFileNameBase = netImpl_.getDumpFileNameBase();
261 try
262 {
263 ov::pass::Serialize(dumpFileNameBase + "_ngraph.xml", dumpFileNameBase + "_ngraph.bin").run_on_model(cnn);
264 }
265 catch (const std::exception& e)
266 {
267 std::ofstream out((dumpFileNameBase + "_ngraph.error").c_str(), std::ios::out);
268 out << "Exception: " << e.what() << std::endl;
269 }
270 catch (...)
271 {
272 std::ofstream out((dumpFileNameBase + "_ngraph.error").c_str(), std::ios::out);
273 out << "Can't dump: unknown exception" << std::endl;
274 }
275#endif
276 }
277 }
278
279 switch (targetId)
280 {
281 case DNN_TARGET_CPU:
282 device_name = "CPU";
283 break;
284 case DNN_TARGET_OPENCL:
285 case DNN_TARGET_OPENCL_FP16:
286 device_name = "GPU";
287 break;
288 case DNN_TARGET_MYRIAD:
289 device_name = "MYRIAD";
290 break;
291 case DNN_TARGET_HDDL:
292 device_name = "HDDL";
293 break;
294 case DNN_TARGET_FPGA:
295 device_name = "FPGA";
296 break;
297 case DNN_TARGET_NPU:
298 device_name = "NPU";
299 break;
300 default:
301 CV_Error(Error::StsNotImplemented, "Unknown target");
302 };
303
304 ov::preprocess::PrePostProcessor ppp(cnn);
305 int i = 0;
306 for (const auto& inp : cnn->inputs()) { // TODO: not sure why but ngraph_function->inputs() here causes segfault.
307 const std::string& name = inp.get_node()->get_friendly_name();
308 auto blobIt = allBlobs.find(name);
309 CV_Assert(blobIt != allBlobs.end());
310
311 auto srcT = blobIt->second.get_element_type();
312 if (srcT != inp.get_node()->get_element_type()) {
313 ppp.input(i++).tensor().set_element_type(srcT);
314 }
315 }
316
317 i = 0;
318 for (const auto& it : cnn->outputs())
319 {
320 const std::string& name = it.get_node()->get_friendly_name();
321 auto blobIt = allBlobs.find(name);
322 CV_Assert(blobIt != allBlobs.end());
323 const auto& src = blobIt->second;
324
325 // A workaround for single dimension output for which OpenCV allocates 2d Mat.
326 // For example, face-detection-0105 with Result of shape {200} while output blob is {200, 1}
327 auto outShape = it.get_partial_shape().get_max_shape();
328 if (outShape != src.get_shape()) {
329 size_t sz = std::accumulate(outShape.begin(), outShape.end(), 1, std::multiplies<size_t>());
330 CV_Assert(sz == src.get_size());
331 allBlobs[name] = ov::Tensor(src.get_element_type(), outShape, src.data());
332 }
333
334 ppp.output(i++).tensor().set_element_type(src.get_element_type());
335 }
336
337 ppp.build();
338
339 initPlugin(cnn);
340}

342ov::ParameterVector InfEngineNgraphNet::setInputs(const std::vector<cv::Mat>& inputs,
343 const std::vector<std::string>& names) {
344 CV_Assert_N(inputs.size() == names.size());
345 ov::ParameterVector current_inp;
346 for (size_t i = 0; i < inputs.size(); i++)
347 {
348 std::vector<size_t> shape = getShape<size_t>(inputs[i]);
349 auto inp = std::make_shared<ov::op::v0::Parameter>(cvTypeToOvType(inputs[i]), ov::Shape(shape));
350 inp->set_friendly_name(names[i]);
351
352 auto it = std::find_if(inputs_vec.begin(), inputs_vec.end(),
353 [&inp](const std::shared_ptr<ov::op::v0::Parameter>& a) {
354 return a->get_friendly_name() == inp->get_friendly_name();
355 });
356 if (it == inputs_vec.end()) {
357 inputs_vec.push_back(inp);
358 current_inp.push_back(inp);
359 } else {
360 current_inp.push_back(*it);
361 }
362 }
363 return current_inp;
364}

367void InfEngineNgraphNet::initPlugin(std::shared_ptr<ov::Model>& net)
368{
369 CV_Assert(!isInitialized());
370
371 try
372 {
373 AutoLock lock(getInitializationMutex());
374 ov::Core& ie = getCore(device_name);
375 {
376 isInit = true;
377 std::vector<std::string> candidates;
378 std::string param_pluginPath = utils::getConfigurationParameterString("OPENCV_DNN_IE_EXTRA_PLUGIN_PATH", "");
379 if (!param_pluginPath.empty())
380 {
381 candidates.push_back(param_pluginPath);
382 }
383 bool found = false;
384 for (size_t i = 0; i != candidates.size(); ++i)
385 {
386 const std::string& libName = candidates[i];
387 try
388 {
389 ie.add_extension(libName);
390 CV_LOG_INFO(NULL, "DNN-IE: Loaded extension plugin: " << libName);
391 found = true;
392 break;
393 }
394 catch(...) {}
395 }
396 if (!found && !candidates.empty())
397 {
398 CV_LOG_WARNING(NULL, "DNN-IE: Can't load extension plugin (extra layers for some networks). Specify path via OPENCV_DNN_IE_EXTRA_PLUGIN_PATH parameter");
399 }
400#ifndef _WIN32
401 // Limit the number of CPU threads.
402 if (device_name == "CPU")
403 ie.set_property(device_name, ov::inference_num_threads(getNumThreads()));
404#endif
405 if (device_name.find("GPU") == 0)
406 {
407#if OPENCV_HAVE_FILESYSTEM_SUPPORT
408 std::string cache_path = utils::fs::getCacheDirectory((std::string("dnn_ie_cache_") + device_name).c_str(), "OPENCV_DNN_IE_GPU_CACHE_DIR");
409#else
410 std::string cache_path = utils::getConfigurationParameterString("OPENCV_DNN_IE_GPU_CACHE_DIR", "");
411#endif
412 if (!cache_path.empty() && cache_path != "disabled")
413 {
414 CV_LOG_INFO(NULL, "OpenCV/nGraph: using GPU kernels cache: " << cache_path);
415 ie.set_property(device_name, ov::cache_dir(cache_path));
416 }
417 }
418 }
419 ov::AnyMap config;
420 if (device_name == "MYRIAD" || device_name == "HDDL") {
421 config.emplace("MYRIAD_DETECT_NETWORK_BATCH", "NO");
422 }
423
424 bool isHetero = device_name == "FPGA";
425 // It is actual only for non-CPU targets and networks built in runtime using nGraph.
426 // We do not check IR models because they can be with version less than IRv10
427 if (!isHetero && device_name != "CPU" && !hasNetOwner)
428 {
429 for (auto& node : net->get_ops())
430 {
431 if (node->description() == kOpenCVLayersType)
432 {
433 isHetero = true;
434 break;
435 }
436 }
437 }
438
439 std::string ieDevice = isHetero ? ("HETERO:" + device_name + ",CPU") : device_name;
440 CV_LOG_INFO(NULL, "DNN/IE: Calling LoadNetwork(device=" << ieDevice << ")...");
441 netExec = ie.compile_model(net, ieDevice, config);
442 }
443 catch (const std::exception& ex)
444 {
445 CV_Error(Error::StsError, format("Failed to initialize Inference Engine backend (device = %s): %s", device_name.c_str(), ex.what()));
446 }
447}

454bool NgraphBackendLayer::getMemoryShapes(const std::vector<MatShape> &inputs,
455 const int requiredOutputs,
456 std::vector<MatShape> &outputs,
457 std::vector<MatShape> &internals) const
458{
459 bool equal_flag = true;
460 std::map<std::string, ov::PartialShape> inShapes;
461 int i = 0;
462 for (const auto& inp : t_net->get_parameters())
463 {
464 ov::Shape oldShape = inp->get_shape();
465 ov::Shape newShape(inputs[i].begin(), inputs[i].end());
466 inShapes.insert({inp->get_friendly_name(), newShape});
467 if (oldShape != newShape)
468 {
469 equal_flag = false;
470 }
471 i++;
472 }
473
474 if (!equal_flag)
475 {
476 std::shared_ptr<ov::Model> curr_t_net(t_net);
477 curr_t_net->reshape(inShapes);
478 }
479 std::vector<size_t> dims;
480 for (const auto& it : t_net->outputs()) {
481 if (it.get_node()->get_friendly_name() == name) {
482 dims = it.get_partial_shape().get_max_shape();
483 }
484 }
485 if (dims.empty())
486 CV_Error(Error::StsError, format("Unable find result with name %s", name.c_str()));
487 outputs.push_back(MatShape(dims.begin(), dims.end()));
488 return false;
489}

585void InfEngineNgraphNet::forward(const std::vector<Ptr<BackendWrapper> >& outBlobsWrappers, bool isAsync)
586{
587 CV_LOG_DEBUG(NULL, "InfEngineNgraphNet::forward(" << (isAsync ? "async" : "sync") << ")");
588
589 // Look for finished requests.
590 Ptr<NgraphReqWrapper> reqWrapper;
591 for (auto& wrapper : infRequests)
592 {
593 if (wrapper->isReady)
594 {
595 reqWrapper = wrapper;
596 break;
597 }
598 }
599 if (reqWrapper.empty())
600 {
601 reqWrapper = Ptr<NgraphReqWrapper>(new NgraphReqWrapper());
602 try
603 {
604 reqWrapper->req = netExec.create_infer_request();
605 }
606 catch (const std::exception& ex)
607 {
608 CV_Error(Error::StsAssert, format("Failed to initialize Inference Engine backend: %s", ex.what()));
609 }
610 infRequests.push_back(reqWrapper);
611
612 int i = 0;
613 for (const auto& it : netExec.inputs())
614 {
615 const std::string& name = it.get_node()->get_friendly_name();
616 auto blobIt = allBlobs.find(name);
617 if (blobIt == allBlobs.end())
618 {
619 CV_Error(Error::StsAssert, format("Input blob with name %s not found", name.c_str()));
620 }
621 reqWrapper->req.set_input_tensor(i++, isAsync ? copyBlob(blobIt->second) : blobIt->second);
622 }
623
624 i = 0;
625 for (const auto& it : cnn->outputs()) // Starts from OpenVINO 2024 CompiledModel changes output frindly names
626 {
627 const std::string& name = it.get_node()->get_friendly_name();
628 auto blobIt = allBlobs.find(name);
629 if (blobIt == allBlobs.end())
630 {
631 CV_Error(Error::StsAssert, format("Output blob with name %s not found", name.c_str()));
632 }
633 reqWrapper->req.set_output_tensor(i++, isAsync ? copyBlob(blobIt->second) : blobIt->second);
634 }
635
636 if (isAsync) {
637 bool* isReady = &reqWrapper->isReady;
638 auto* promises = &reqWrapper->outProms;
639 auto* req = &reqWrapper->req;
640 reqWrapper->req.set_callback([isReady, promises, req](std::exception_ptr ex) {
641 CV_LOG_DEBUG(NULL, "DNN(nGraph): completionCallback()");
642
643 size_t processedOutputs = 0;
644 try
645 {
646 for (; processedOutputs < promises->size(); ++processedOutputs)
647 {
648 Mat m = infEngineBlobToMat(req->get_output_tensor(processedOutputs));
649
650 try
651 {
652 (*promises)[processedOutputs].setValue(m.clone());
653 }
654 catch (...)
655 {
656 try {
657 (*promises)[processedOutputs].setException(std::current_exception());
658 } catch(...) {
659 CV_LOG_ERROR(NULL, "DNN: Exception occurred during async inference exception propagation");
660 }
661 }
662 }
663 }
664 catch (...)
665 {
666 std::exception_ptr e = std::current_exception();
667 for (; processedOutputs < promises->size(); ++processedOutputs)
668 {
669 try {
670 (*promises)[processedOutputs].setException(e);
671 } catch(...) {
672 CV_LOG_ERROR(NULL, "DNN: Exception occurred during async inference exception propagation");
673 }
674 }
675 }
676 *isReady = true;
677 });
678 }
679 }
680
681 if (isAsync)
682 {
683 // Copy actual data to infer request's input blobs.
684 int i = 0;
