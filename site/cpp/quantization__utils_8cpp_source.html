15static void broadcast1D2TargetMat(Mat& data, const MatShape& targetShape, int axis)
16{
17 // The data is the 1-D scales or zeropoints.
18 CV_CheckGE(axis, 0, "Quantization axis must be non-negative.");
19 CV_CheckGT((int)targetShape.size(),axis,"Quantization axis must be within the valid range of target shape dimensions.");
20 CV_CheckEQ((int)data.total(), (int)targetShape[axis], "Data total size must match the size of the specified target dimension.");
21
22 std::vector<int> broadcast_axes;
23 for (int i = 0; i < targetShape.size(); i++)
24 {
25 if (i != axis)
26 broadcast_axes.push_back(i);
27 }
28
29 MatShape subTargetShape = shape(data);
30
31 // convert std::vector to 1D Mat.
32 for (auto broadcast_axis : broadcast_axes)
33 {
34 subTargetShape[broadcast_axis] = targetShape[broadcast_axis];
35 data = data.reshape(0, total(data, 0, broadcast_axis));
36 Mat tmp = cv::repeat(data, 1, subTargetShape[broadcast_axis]);
37 data = tmp.reshape(0, subTargetShape);
38 }
39}

41static void block_repeat(InputArray src, const MatShape& srcShape, int axis, int repetitions, OutputArray dst)
42{
43 CV_Assert(src.getObj() != dst.getObj());
44 CV_Check(axis, axis >= 0 && (axis < src.dims() || (src.dims()==1 && axis==1)), "axis is out of range"); // (src.dims()==1 && axis==1) has been added as a temporary fix for quantized models. Refer issue https://github.com/opencv/opencv_zoo/issues/273
45 CV_CheckGT(repetitions, 1, "More than one repetition expected");
46
47 Mat src_mat = src.getMat();
48 Mat dst_mat;
49
50 if (src_mat.depth() != CV_32F)
51 src_mat.convertTo(src_mat, CV_32F);
52
53 MatShape sshape = srcShape;
54 MatShape dshape = srcShape;
55
56 size_t dtype_bytes = src_mat.elemSize();
57 int chunk_size = dtype_bytes;
58 int num_chunks = 1;
59
60 dshape[axis] *= repetitions;
61
62 for (int i = axis+1; i < sshape.size(); ++i)
63 chunk_size*=sshape[i];
64
65 for (int i = 0; i <= axis; ++i)
66 num_chunks*=sshape[i];
67
68 dst.create(dshape.size(), dshape.data(), src_mat.type());
69 dst_mat = dst.getMat();
70
71 CV_Assert(dst_mat.isContinuous());
72 CV_Assert(src_mat.isContinuous());
73
74 for (int i = 0; i < repetitions; ++i) {
75 size_t src_offset = 0;
76 size_t dst_offset = i * chunk_size;
77
78 for (int j = 0; j < num_chunks; ++j) {
79 memcpy(dst_mat.data + dst_offset, src_mat.data + src_offset, chunk_size);
80 src_offset += chunk_size;
81 dst_offset += chunk_size * repetitions;
82 }
83 }
84}

95template <typename T>
96static void broadcastBlockedMatrix(Mat& mat, const std::vector<T>& data, const MatShape& targetShape, int axis, int block_size){
97 CV_Check(block_size, targetShape[axis] % block_size == 0 && block_size <= targetShape[axis], "Block size must be a divisor of the target dimension size and not exceed it.");
98
99 MatShape subTargetShape(targetShape);
100 subTargetShape[axis] = static_cast<int>(subTargetShape[axis] / block_size);
101
102 block_repeat(data, subTargetShape, axis, block_size, mat);
103}

105template <typename T>
106static void broadcastStandardMatrix(Mat& mat, const std::vector<T>& data, const MatShape& targetShape, int axis)
107{
108 MatShape subTargetShape(targetShape.size(), 1);
109 subTargetShape[axis] = data.size();
110 mat.create(subTargetShape.size(), subTargetShape.data(), CV_32FC1);
111
112 copyVecToMat(mat,data);
113
114 broadcast1D2TargetMat(mat, targetShape, axis);
115}

118static void broadcastScaleAndZeropoint(Mat& scalesMat, Mat& zeropointsMat, const std::vector<float>& scales,
119 const std::vector<int>& zeropoints, const MatShape& targetShape, int axis, int block_size)
120{
121 // broad cast the scales and zeropoint to the input shape.
122
123 if (block_size == 0)
124 {
125 broadcastStandardMatrix(zeropointsMat, zeropoints, targetShape, axis);
126 broadcastStandardMatrix(scalesMat, scales, targetShape, axis);
127 }
128 else
129 {
130 broadcastBlockedMatrix(zeropointsMat, zeropoints, targetShape, axis, block_size);
131 broadcastBlockedMatrix(scalesMat, scales, targetShape, axis, block_size);
132 }

136class QuantizeLayerImpl CV_FINAL : public QuantizeLayer
137{
138public:

141 int outDepth; // CV_8S (legacy default) or CV_8U; driven by ONNX y_zero_point dtype
142 bool is1D;
143 Mat scalesMat, zeropointsMat; // Saving the broadcasted scales data.
144 bool quantParamExternal = true; // Indicates if the quantization parameters (scale and zero point) are provided as inputs to the node.
145
146 QuantizeLayerImpl(const LayerParams& params)
147 {
148 is1D = params.get<bool>("is1D", false);
149 axis = params.get<int>("axis", 1);
150 block_size = params.get<int>("block_size", 0);
151 outDepth = params.get<int>("depth", CV_8S);
152
153 if (!is1D)
154 {
155 scales.push_back(params.get<float>("scales", 1.0f));
156 zeropoints.push_back(params.get<int>("zeropoints", 0));
157 }
158 else
159 {
160 DictValue paramScales = params.get("scales");
161 int i, n = paramScales.size();
162
163 CV_CheckGT(n, 0, "Scale missing.");
164 scales.resize(n, 0.);
165 for (i = 0; i < n; i++)
166 scales[i] = paramScales.get<float>(i);
167
168 zeropoints.resize(n, 0);
169 DictValue paramZp = params.get("zeropoints");
170 n = paramZp.size();
171
172 for (i = 0; i < n; i++)
173 zeropoints[i] = paramZp.get<int>(i);
174 }
175 setParamsFrom(params);
176 }

239 {
240 CV_Check(inputs.size(), inputs.size() >= 1 && inputs.size() <= 3, "Number of inputs must be between 1 and 3 inclusive.");

242

252
253
254 if (inputs.size() > 2)

263 MatShape inputShape = shape(inputs[0]);
264 broadcastScaleAndZeropoint(scalesMat, zeropointsMat, scales, zeropoints, inputShape, axis, block_size);
265 }
266 }

269 outputs[0].convertTo(outputs[0], outDepth);

518 virtual Ptr<BackendNode> initTimVX(void* timVXInfo_,
519 const std::vector<Ptr<BackendWrapper> > &inputsWrapper,
520 const std::vector<Ptr<BackendWrapper> > &outputsWrapper,
521 bool isLast) CV_OVERRIDE
522 {
523#ifdef HAVE_TIMVX
524 // preprocessing
525 // Check if data is 8-bit.
526 CV_Assert(inputsWrapper.size() == 1 && outputsWrapper.size() == 1);
527 Ptr<TimVXBackendWrapper> inputWrapper = inputsWrapper[0].dynamicCast<TimVXBackendWrapper>();
528
529 if (!inputWrapper->isTensor())
530 {
531 return Ptr<BackendNode>();
532 }
533
534 auto timVxInfo = reinterpret_cast<TimVXInfo *>(timVXInfo_);
535 CV_Assert(timVxInfo);
536 Ptr<TimVXGraph> tvGraph = timVxInfo->getGraph();
537 CV_Assert(tvGraph);
538 Ptr<tim::vx::Graph> graph = tvGraph->graph;
539
540 std::vector<int> inputsIndex, outputsIndex;
541 int input_index = -1, output_index = -1;
542
543 // Input
544 std::shared_ptr<tim::vx::Tensor> inputTensor = inputWrapper->getTensor();
545 input_index = tvGraph->getTensorIndex(inputTensor);
546 if (input_index == -1)
547 return Ptr<BackendNode>();
548
549 inputsIndex.push_back(input_index);
550
551 Ptr<tim::vx::Quantization> inputQuant = inputWrapper->getTensorQuantization();
552
553 tim::vx::QuantType quanType = inputQuant->Type();
554 CV_Assert(quanType == tim::vx::QuantType::ASYMMETRIC);
555
556 std::vector<float> scales = inputQuant->Scales();
557 std::vector<int32_t> zeropoints = inputQuant->ZeroPoints();
558 CV_Assert(!scales.empty() && !zeropoints.empty());
559 int input_zp = int(zeropoints[0]);
560 float input_scale = scales[0];
561
562 float tmpOut_sc = input_scale/scale;
563 int tmpOut_zp = int(shift + scale * input_zp);
564
565 // Output
566 Ptr<TimVXBackendWrapper> outputWrapper = outputsWrapper[0].dynamicCast<TimVXBackendWrapper>();
567 Ptr<tim::vx::Quantization> outputQuant = Ptr<tim::vx::Quantization>(
568 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, tmpOut_sc, tmpOut_zp));
569
570 if (isLast)
571 {
572 auto shapeType = getShapeTypeFromMat(outputWrapper->getMat());
573
574 // For Graph Output tensor, we need to set tensor shape before createTensor().
575 outputWrapper->setTensorShape(shapeType);
576 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::OUTPUT, outputQuant);
577 }
578 else
579 {
580 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::TRANSIENT, outputQuant);
581 }
582 output_index = tvGraph->addWrapper(outputWrapper);
583 outputsIndex.push_back(output_index);
584
585 std::shared_ptr<tim::vx::Operation> tvRequantize = graph->CreateOperation<tim::vx::ops::DataConvert>();
586
587 Ptr<TimVXBackendNode> tvBackendNode = new TimVXBackendNode(tvGraph, tvRequantize, inputsIndex, outputsIndex);
588
589 return tvBackendNode;
590#endif // HAVE_TIMVX
591 return Ptr<BackendNode>();
592 }
