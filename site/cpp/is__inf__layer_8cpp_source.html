21template <typename T, typename WT = T>
22static inline void computeIsInfMask(const T* src, uchar* dst, const size_t count, const bool detectPositive, const bool detectNegative)
23{
24 if (detectPositive && detectNegative)
25 {
26 parallel_for_(Range(0, (int)count), [&](const Range& r){
27 for (int i = r.start; i < r.end; ++i)
28 {
29 WT v = (WT)src[i];
30 dst[i] = static_cast<uchar>(cvIsInf(v));
31 }
32 });
33 }
34 else if (detectPositive)
35 {
36 parallel_for_(Range(0, (int)count), [&](const Range& r){
37 for (int i = r.start; i < r.end; ++i)
38 {
39 WT v = (WT)src[i];
40 dst[i] = static_cast<uchar>(cvIsInf(v) && (v > 0));
41 }
42 });
43 }
44 else if (detectNegative)
45 {
46 parallel_for_(Range(0, (int)count), [&](const Range& r){
47 for (int i = r.start; i < r.end; ++i)
48 {
49 WT v = (WT)src[i];
50 dst[i] = static_cast<uchar>(cvIsInf(v) && (v < 0));
51 }
52 });
53 }
54 else
55 {
56 CV_Error_(Error::StsError, ("IsInf: Unsupported mode"));
57 }
58}

94 void forward(InputArrayOfArrays in, OutputArrayOfArrays out, OutputArrayOfArrays) CV_OVERRIDE
95 {
96 std::vector<Mat> inputs, outputs; in.getMatVector(inputs); out.getMatVector(outputs);
97 CV_Assert(inputs.size() == 1 && outputs.size() == 1);
98
99 const Mat& X = inputs[0];
100 Mat& Y = outputs[0];
101
102 const int defaultOutType = CV_BoolC1;
103 const int outType = Y.empty() ? defaultOutType : Y.type();
104 Y.create(X.size, outType);
105
106 const int depth = CV_MAT_DEPTH(X.type());
107 const size_t total = X.total();
108 uchar* dst = Y.ptr<uchar>();
109
110 switch (depth) {
111 case CV_32F: computeIsInfMask<float>(X.ptr<float>(), dst, total, detect_pos, detect_neg); break;
112 case CV_64F: computeIsInfMask<double>(X.ptr<double>(), dst, total, detect_pos, detect_neg); break;
113 case CV_16F: computeIsInfMask<hfloat, float>(X.ptr<hfloat>(), dst, total, detect_pos, detect_neg); break;
114 case CV_16BF: computeIsInfMask<bfloat, float>(X.ptr<bfloat>(), dst, total, detect_pos, detect_neg); break;
115 default: CV_Error_(Error::StsError, ("IsInf: Unsupported type depth=%d", depth));
116 }
117 }
118};
