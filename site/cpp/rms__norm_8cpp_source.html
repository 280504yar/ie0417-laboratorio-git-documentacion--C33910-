37 std::vector<MatShape> &outputs,
38 std::vector<MatShape> &internals) const CV_OVERRIDE
39 {
40 const int n_inputs = inputs.size();
41 CV_Check(n_inputs, n_inputs == 2, "RMSNorm: require two (x, scale) inputs");
42
43 auto x_shape = inputs[0];
44 auto scale_shape = inputs[1];
45 const int normalized_axis = normalize_axis(axis, static_cast<int>(x_shape.size()));
46 const int x_ndims = static_cast<int>(x_shape.size());
47
48 CV_CheckTrue(scale_shape.size() == x_ndims - normalized_axis,
49 "RMSNorm: scale shape should match normalized shape");
50
51 for (int i = 0; i + normalized_axis < x_ndims; ++i)
52 {
53 CV_CheckTrue(
54 x_shape[i + normalized_axis] == scale_shape[i],
55 "RMSNorm: scale shape should match normalized shape");
56 }
57
58 outputs.assign(1, inputs[0]);
59 return true;
60 }
