148class PlaneABC : public PlaneBase
149{
150public:
151 PlaneABC(const Vec3f& m, const Vec3f& n_in, int index, float sensor_error_a, float sensor_error_b, float sensor_error_c) :
152 PlaneBase(m, n_in, index),
153 sensor_error_a_(sensor_error_a),
154 sensor_error_b_(sensor_error_b),
155 sensor_error_c_(sensor_error_c)
156 {
157 }
158
160 inline float distance(const Vec3f& p_j) const CV_OVERRIDE
161 {
162 float cst = p_j.dot(n_) + d_;
163 float err = sensor_error_a_ * p_j[2] * p_j[2] + sensor_error_b_ * p_j[2] + sensor_error_c_;
164 if (((cst - n_[2] * err <= 0) && (cst + n_[2] * err >= 0)) || ((cst + n_[2] * err <= 0) && (cst - n_[2] * err >= 0)))
165 return 0;
166 return std::min(std::abs(cst - err), std::abs(cst + err));
167 }
168private:
169 float sensor_error_a_;
170 float sensor_error_b_;
171 float sensor_error_c_;

178class PlaneGrid
179{
180public:
181 PlaneGrid(const Mat_<Vec4f>& points3d, int block_size) :
182 block_size_(block_size)
183 {
184 // Figure out some dimensions
185 int mini_rows = points3d.rows / block_size;
186 if (points3d.rows % block_size != 0)
187 ++mini_rows;
188
189 int mini_cols = points3d.cols / block_size;
190 if (points3d.cols % block_size != 0)
191 ++mini_cols;
192
193 // Compute all the interesting quantities
194 m_.create(mini_rows, mini_cols);
195 n_.create(mini_rows, mini_cols);
196 Q_.create(points3d.rows, points3d.cols);
197 mse_.create(mini_rows, mini_cols);
198 for (int y = 0; y < mini_rows; ++y)
199 for (int x = 0; x < mini_cols; ++x)
200 {
201 // Update the tiles
202 Matx33f Q = Matx33f::zeros();
203 Vec3f m = Vec3f(0, 0, 0);
204 int K = 0;
205 for (int j = y * block_size; j < std::min((y + 1) * block_size, points3d.rows); ++j)
206 {
207 const Vec4f* vec = points3d.ptr < Vec4f >(j, x * block_size), * vec_end;
208 float* pointpointt = reinterpret_cast<float*>(Q_.ptr < Vec<float, 9> >(j, x * block_size));
209 if (x == mini_cols - 1)
210 vec_end = points3d.ptr < Vec4f >(j, points3d.cols - 1) + 1;
211 else
212 vec_end = vec + block_size;
213 for (; vec != vec_end; ++vec, pointpointt += 9)
214 {
215 if (cvIsNaN(vec->val[0]))
216 continue;
217 // Fill point*point.t()
218 *pointpointt = vec->val[0] * vec->val[0];
219 *(pointpointt + 1) = vec->val[0] * vec->val[1];
220 *(pointpointt + 2) = vec->val[0] * vec->val[2];
221 *(pointpointt + 3) = *(pointpointt + 1);
222 *(pointpointt + 4) = vec->val[1] * vec->val[1];
223 *(pointpointt + 5) = vec->val[1] * vec->val[2];
224 *(pointpointt + 6) = *(pointpointt + 2);
225 *(pointpointt + 7) = *(pointpointt + 5);
226 *(pointpointt + 8) = vec->val[2] * vec->val[2];
227
228 Q += *reinterpret_cast<Matx33f*>(pointpointt);
229 m += Vec3f((*vec)[0], (*vec)[1], (*vec)[2]);
230 ++K;
231 }
232 }
233 if (K == 0)
234 {
235 mse_(y, x) = std::numeric_limits<float>::max();
236 continue;
237 }
238
239 m /= K;
240 m_(y, x) = m;
241
242 // Compute C
243 Matx33f C = Q - K * m * m.t();
244
245 // Compute n
246 SVD svd(C);
247 n_(y, x) = Vec3f(svd.vt.at<float>(2, 0), svd.vt.at<float>(2, 1), svd.vt.at<float>(2, 2));
248 mse_(y, x) = svd.w.at<float>(2) / K;
249 }
250 }
251
253 int block_size_;
254 Mat_<Vec3f> m_;
255 Mat_<Vec3f> n_;
256 Mat_<Vec<float, 9> > Q_;
257 Mat_<float> mse_;
258};

282
283 TileQueue(const PlaneGrid& plane_grid)
284 {
285 done_tiles_ = Mat_<unsigned char>::zeros(plane_grid.mse_.rows, plane_grid.mse_.cols);
286 tiles_.clear();
287 for (int y = 0; y < plane_grid.mse_.rows; ++y)
288 for (int x = 0; x < plane_grid.mse_.cols; ++x)
289 if (plane_grid.mse_(y, x) != std::numeric_limits<float>::max())
290 // Update the tiles
291 tiles_.push_back(PlaneTile(x, y, plane_grid.mse_(y, x)));
292 // Sort tiles by MSE
293 tiles_.sort();
294 }
295
296 bool empty()
297 {
298 while (!tiles_.empty())
299 {
300 const PlaneTile& tile = tiles_.front();
301 if (done_tiles_(tile.y_, tile.x_))
302 tiles_.pop_front();
303 else
304 break;
305 }
306 return tiles_.empty();
307 }
308
309 const PlaneTile& front() const
310 {
311 return tiles_.front();
312 }
313
314 void remove(int y, int x)
315 {
316 done_tiles_(y, x) = 1;
317 }
318private:
320 std::list<PlaneTile> tiles_;
322 Mat_<unsigned char> done_tiles_;
323};

327class InlierFinder
328{
329public:
330 InlierFinder(float err, const Mat_<Vec4f>& points3d, const Mat_<Vec4f>& normals,
331 unsigned char plane_index, int block_size) :
332 err_(err),
333 points3d_(points3d),
334 normals_(normals),
335 plane_index_(plane_index),
336 block_size_(block_size)
337 {
338 }
339
340 void Find(const PlaneGrid& plane_grid, Ptr<PlaneBase>& plane, TileQueue& tile_queue,
341 std::set<TileQueue::PlaneTile>& neighboring_tiles, Mat_<unsigned char>& overall_mask,
342 Mat_<unsigned char>& plane_mask)
343 {
344 // Do not use reference as we pop the from later on
345 TileQueue::PlaneTile tile = *(neighboring_tiles.begin());
346
347 // Figure the part of the image to look at
348 Range range_x, range_y;
349 int x = tile.x_ * block_size_, y = tile.y_ * block_size_;
350
351 if (tile.x_ == plane_mask.cols - 1)
352 range_x = Range(x, overall_mask.cols);
353 else
354 range_x = Range(x, x + block_size_);
355
356 if (tile.y_ == plane_mask.rows - 1)
357 range_y = Range(y, overall_mask.rows);
358 else
359 range_y = Range(y, y + block_size_);
360
361 int n_valid_points = 0;
362 for (int yy = range_y.start; yy != range_y.end; ++yy)
363 {
364 uchar* data = overall_mask.ptr(yy, range_x.start), * data_end = data + range_x.size();
365 const Vec4f* point = points3d_.ptr < Vec4f >(yy, range_x.start);
366 const Matx33f* Q_local = reinterpret_cast<const Matx33f*>(plane_grid.Q_.ptr < Vec<float, 9>
367 >(yy, range_x.start));
368
369 // Depending on whether you have a normal, check it
370 if (!normals_.empty())
371 {
372 const Vec4f* normal = normals_.ptr < Vec4f >(yy, range_x.start);
373 for (; data != data_end; ++data, ++point, ++normal, ++Q_local)
374 {
375 // Don't do anything if the point already belongs to another plane
376 if (cvIsNaN(point->val[0]) || ((*data) != 255))
377 continue;
378
379 // If the point is close enough to the plane
380 Vec3f _p = Vec3f((*point)[0], (*point)[1], (*point)[2]);
381 if (plane->distance(_p) < err_)
382 {
383 // make sure the normals are similar to the plane
384 Vec3f _n = Vec3f((*normal)[0], (*normal)[1], (*normal)[2]);
385 if (std::abs(plane->n().dot(_n)) > 0.3)
386 {
387 // The point now belongs to the plane
388 plane->UpdateStatistics(_p, *Q_local);
389 *data = plane_index_;
390 ++n_valid_points;
391 }
392 }
393 }
394 }
395 else
396 {
397 for (; data != data_end; ++data, ++point, ++Q_local)
398 {
399 // Don't do anything if the point already belongs to another plane
400 if (cvIsNaN(point->val[0]) || ((*data) != 255))
401 continue;
402
403 // If the point is close enough to the plane
404 Vec3f _p = Vec3f((*point)[0], (*point)[1], (*point)[2]);
405 if (plane->distance(_p) < err_)
406 {
407 // The point now belongs to the plane
408 plane->UpdateStatistics(_p, *Q_local);
409 *data = plane_index_;
410 ++n_valid_points;
411 }
412 }
413 }
414 }
415
416 plane->UpdateParameters();
417
418 // Mark the front as being done and pop it
419 if (n_valid_points > (range_x.size() * range_y.size()) / 2)
420 tile_queue.remove(tile.y_, tile.x_);
421 plane_mask(tile.y_, tile.x_) = 1;
422 neighboring_tiles.erase(neighboring_tiles.begin());
423
424 // Add potential neighbors of the tile
425 std::vector<std::pair<int, int> > pairs;
426 if (tile.x_ > 0)
427 for (unsigned char* val = overall_mask.ptr<unsigned char>(range_y.start, range_x.start), *val_end = val
428 + range_y.size() * overall_mask.step; val != val_end; val += overall_mask.step)
429 if (*val == plane_index_)
430 {
431 pairs.push_back(std::pair<int, int>(tile.x_ - 1, tile.y_));
432 break;
433 }
434 if (tile.x_ < plane_mask.cols - 1)
435 for (unsigned char* val = overall_mask.ptr<unsigned char>(range_y.start, range_x.end - 1), *val_end = val
436 + range_y.size() * overall_mask.step; val != val_end; val += overall_mask.step)
437 if (*val == plane_index_)
438 {
439 pairs.push_back(std::pair<int, int>(tile.x_ + 1, tile.y_));
440 break;
441 }
442 if (tile.y_ > 0)
443 for (unsigned char* val = overall_mask.ptr<unsigned char>(range_y.start, range_x.start), *val_end = val
444 + range_x.size(); val != val_end; ++val)
445 if (*val == plane_index_)
446 {
447 pairs.push_back(std::pair<int, int>(tile.x_, tile.y_ - 1));
448 break;
449 }
450 if (tile.y_ < plane_mask.rows - 1)
451 for (unsigned char* val = overall_mask.ptr<unsigned char>(range_y.end - 1, range_x.start), *val_end = val
452 + range_x.size(); val != val_end; ++val)
453 if (*val == plane_index_)
454 {
455 pairs.push_back(std::pair<int, int>(tile.x_, tile.y_ + 1));
456 break;
457 }
458
459 for (unsigned char i = 0; i < pairs.size(); ++i)
460 if (!plane_mask(pairs[i].second, pairs[i].first))
461 neighboring_tiles.insert(
462 TileQueue::PlaneTile(pairs[i].first, pairs[i].second, plane_grid.mse_(pairs[i].second, pairs[i].first)));
463 }
464
465private:
466 float err_;
467 const Mat_<Vec4f>& points3d_;
468 const Mat_<Vec4f>& normals_;
469 unsigned char plane_index_;
471 int block_size_;
472
473 const InlierFinder& operator = (const InlierFinder&);
474};

478void findPlanes(InputArray points3d_in, InputArray normals_in, OutputArray mask_out, OutputArray plane_coefficients_out,
479 int block_size, int min_size, double threshold, double sensor_error_a, double sensor_error_b, double sensor_error_c,
480 RgbdPlaneMethod method)
481{
482 CV_Assert(method == RGBD_PLANE_METHOD_DEFAULT);
483
484 Mat_<Vec4f> points3d, normals;
485 if (points3d_in.depth() == CV_32F)
486 points3d = points3d_in.getMat();
487 else
488 points3d_in.getMat().convertTo(points3d, CV_32F);
489 if (!normals_in.empty())
490 {
491 if (normals_in.depth() == CV_32F)
492 normals = normals_in.getMat();
493 else
494 normals_in.getMat().convertTo(normals, CV_32F);
495 }
496
497 // Pre-computations
498 mask_out.create(points3d.size(), CV_8U);
499 Mat mask_out_mat = mask_out.getMat();
500 Mat_<unsigned char> mask_out_uc = (Mat_<unsigned char>&) mask_out_mat;
501 mask_out_uc.setTo(255);
502 PlaneGrid plane_grid(points3d, block_size);
503 TileQueue plane_queue(plane_grid);
504 size_t index_plane = 0;
505
506 std::vector<Vec4f> plane_coefficients;
507 float mse_min = (float)(threshold * threshold);
508
509 while (!plane_queue.empty())
510 {
511 // Get the first tile if it's good enough
512 const TileQueue::PlaneTile front_tile = plane_queue.front();
513 if (front_tile.mse_ > mse_min)
514 break;
515
516 InlierFinder inlier_finder((float)threshold, points3d, normals, (unsigned char)index_plane, block_size);
517
518 // Construct the plane for the first tile
519 int x = front_tile.x_, y = front_tile.y_;
520 const Vec3f& n = plane_grid.n_(y, x);
521 Ptr<PlaneBase> plane;
522 if ((sensor_error_a == 0) && (sensor_error_b == 0) && (sensor_error_c == 0))
523 plane = Ptr<PlaneBase>(new Plane(plane_grid.m_(y, x), n, (int)index_plane));
524 else
525 plane = Ptr<PlaneBase>(new PlaneABC(plane_grid.m_(y, x), n, (int)index_plane,
526 (float)sensor_error_a, (float)sensor_error_b, (float)sensor_error_c));
527
528 Mat_<unsigned char> plane_mask = Mat_<unsigned char>::zeros(divUp(points3d.rows, block_size),
529 divUp(points3d.cols, block_size));
530 std::set<TileQueue::PlaneTile> neighboring_tiles;
531 neighboring_tiles.insert(front_tile);
532 plane_queue.remove(front_tile.y_, front_tile.x_);
533
534 // Process all the neighboring tiles
535 while (!neighboring_tiles.empty())
536 inlier_finder.Find(plane_grid, plane, plane_queue, neighboring_tiles, mask_out_uc, plane_mask);
537
538 // Don't record the plane if it's empty
539 if (plane->empty())
540 continue;
541 // Don't record the plane if it's smaller than asked
542 if (plane->K() < min_size)
543 {
544 // Reset the plane index in the mask
545 for (y = 0; y < plane_mask.rows; ++y)
546 for (x = 0; x < plane_mask.cols; ++x)
547 {
548 if (!plane_mask(y, x))
549 continue;
550 // Go over the tile
551 for (int yy = y * block_size;
552 yy < std::min((y + 1) * block_size, mask_out_uc.rows); ++yy)
553 {
554 uchar* data = mask_out_uc.ptr(yy, x * block_size);
555 uchar* data_end = data + std::min(block_size, mask_out_uc.cols - x * block_size);
556 for (; data != data_end; ++data)
557 {
558 if (*data == index_plane)
559 *data = 255;
560 }
561 }
562 }
563 continue;
564 }
565
566 ++index_plane;
567 if (index_plane >= 255)
568 break;
569 Vec4f coeffs(plane->n()[0], plane->n()[1], plane->n()[2], plane->d());
570 if (coeffs(2) > 0)
571 coeffs = -coeffs;
572 plane_coefficients.push_back(coeffs);
573 };
574
575 // Fill the plane coefficients
576 if (plane_coefficients.empty())
577 return;
578 plane_coefficients_out.create((int)plane_coefficients.size(), 1, CV_32FC4);
579 Mat plane_coefficients_mat = plane_coefficients_out.getMat();
580 float* data = plane_coefficients_mat.ptr<float>(0);
581 for (size_t i = 0; i < plane_coefficients.size(); ++i)
582 for (uchar j = 0; j < 4; ++j, ++data)
583 *data = plane_coefficients[i][j];
584}
