31 LayerNormOp(csl::Stream stream_, const Mat &scale, const Mat &bias, int normalized_axis, float epsilon_, size_t loops)
32 : stream(std::move(stream_)), epsilon(epsilon_) {
33 CV_CheckGE(normalized_axis, 0, "LayerNorm/CUDA: axis needs to be normalized");
34 axis = static_cast<size_t>(normalized_axis);
35
36 if (!scale.empty()) {
37 input_scale_tensor = csl::makeTensorHeader<T>(scale);
38 csl::copyMatToTensor<T>(scale, input_scale_tensor, stream);
39 }
40 if (!bias.empty()) {
41 input_bias_tensor = csl::makeTensorHeader<T>(bias);
42 csl::copyMatToTensor<T>(bias, input_bias_tensor, stream);
43 }

45 csl::WorkspaceBuilder builder;

51 void forward(const std::vector<cv::Ptr<BackendWrapper>>& inputs,
52 const std::vector<cv::Ptr<BackendWrapper>>& outputs,
53 csl::Workspace& workspace) override {
54 auto input_wrapper = inputs[0].dynamicCast<wrapper_type>();
55 auto input = input_wrapper->getView();
56
57 csl::TensorView<T> scale;
58 if (input_scale_tensor.empty()) {
59 auto scale_wrapper = inputs[1].dynamicCast<wrapper_type>();
60 scale = scale_wrapper->getView();
61 } else {
62 scale = csl::TensorView<T>(input_scale_tensor);
63 }
64
65 csl::TensorView<T> bias;
66 if (input_bias_tensor.empty()) {
67 if (inputs.size() >= 3) {
68 auto bias_wrapper = inputs[2].dynamicCast<wrapper_type>();
69 bias = bias_wrapper->getView();
70 }
71 } else {
72 bias = csl::TensorView<T>(input_bias_tensor);
73 }
74
75 auto output_wrapper = outputs[0].dynamicCast<wrapper_type>();
76 auto output = output_wrapper->getSpan();
77
78 auto loops = input.size_range(0, axis);
79 auto norm_size = input.size_range(axis, input.rank());
80 if (norm_size == 1) {
81 kernels::fill<T>(stream, output, 0.f);
82 return;
83 } else {
84 auto ws_allocator = csl::WorkspaceAllocator(workspace);
85
86 auto mean = ws_allocator.get_span<float>(loops);
87 kernels::fill<float>(stream, mean, 0.f);
88
89 auto inv_stddev = ws_allocator.get_span<float>(loops);
90 kernels::fill<float>(stream, inv_stddev, 0.f);
91
92 kernels::reduce_mean_sqr_sum<T>(stream, mean, inv_stddev, input, norm_size);
93 kernels::compute_normalization_scale(stream, inv_stddev, mean, inv_stddev, norm_size, epsilon);
94 if (!bias.empty()) {
95 kernels::normalize_mean_variance_layernorm<T>(stream, output, input, scale, bias, mean, inv_stddev, norm_size);
96 } else {
97 kernels::normalize_mean_variance_layernorm<T>(stream, output, input, scale, mean, inv_stddev, norm_size);
98 }
99 }
100 }
