335// New element-wise engine scalar handling. A genuine number / Scalar / Vec / Matx operand to an
336// arithmetic op arrives via _InputArray::MATX (its data is inline in the caller's object;
337// getObj() points straight at it). In addition, the EXACT Scalar materialization that the
338// python/java bindings and operator-(Mat, Matx) produce - a 2-D 4x1 CV_64F single-channel Mat or
339// UMat - is a scalar UNCONDITIONALLY: by coincidence it can be broadcast-COMPATIBLE with the array
340// (a 4-row array, a 1-D array make (4,1) legal numpy-wise), and the 4.x per-channel-scalar
341// semantics must win there for binding users. Any other real Mat/UMat rides normal broadcasting
342// (but see isScalarLikeMat below for the shape-incompatible compat fallback).
343inline bool isScalarArg(const _InputArray& sc, int cn)
344{
345 const _InputArray::KindFlag kind = sc.kind();
346 if (kind == _InputArray::MATX)
347 {
348 Size sz = sc.getSz();
349 int scn0 = sz.width * sz.height;
350 // A genuine scalar is a 1D MATX (a Vec/Scalar/number: one of width/height is 1). A 2D MATX
351 // (e.g. a Matx33) is a real matrix operand and rides broadcasting - never a scalar.
352 if (scn0 != sz.width + sz.height - 1)
353 return false;
354 // Per-channel match (incl. Vec<_,N> for an N-channel array), a 4-elem Scalar on a <4-channel
355 // array, or a single broadcast value. No 4-channel cap: a multichannel scalar rides as a 0-dim
356 // per-channel CONST over the caller's data (not squeezed into a 4-slot Scalar).
357 return scn0 == cn || (cn < 4 && scn0 == 4) || scn0 == 1;
358 }
359 // the bindings-style Scalar column. dims must be exactly 2: a 1-D [4] CV_64F array is an honest
360 // broadcast operand. Direct field reads (no _InputArray getter dispatch) - this runs on EVERY
361 // engine call with Mat operands, and `rows == 4` alone rejects almost every real array.
362 if (kind == _InputArray::MAT)
363 {
364 const Mat& m = *(const Mat*)sc.getObj();
365 return m.rows == 4 && m.cols == 1 && cn <= 4 && m.dims == 2 &&
366 m.type() == CV_64F && m.isContinuous();
367 }
368 if (kind == _InputArray::UMAT)
369 {
370 const UMat& m = *(const UMat*)sc.getObj();
371 return m.rows == 4 && m.cols == 1 && cn <= 4 && m.dims == 2 &&
372 m.type() == CV_64F && m.isContinuous();
373 }
374 return false;
375}

377// The remaining old arithm_op checkScalar geometry: a real Mat/UMat that LOOKS like a scalar - 1x1
378// or a 1xcn/cnx1 vector (the 4x1 CV_64F column is handled unconditionally by isScalarArg above).
379// arithm_op treats such an operand as a per-channel scalar ONLY as a fallback, when the shapes are
380// not broadcast-compatible - a call that is valid under numpy rules keeps its numpy meaning, one
381// that would throw gets the 4.x scalar semantics instead.
382inline bool isScalarLikeMat(const _InputArray& sc, int cn)
383{
384 // direct field reads, like isScalarArg: this probe also runs per engine call (see arithm_op's
385 // compat fallback). `rows != 1 && cols != 1` exits in two inline compares for ordinary arrays.
386 int rows, cols, nval;
387 bool continuous;
388 const _InputArray::KindFlag kind = sc.kind();
389 if (kind == _InputArray::MAT)
390 {
391 const Mat& m = *(const Mat*)sc.getObj();
392 if ((m.rows != 1 && m.cols != 1) || m.dims > 2) return false;
393 rows = m.rows; cols = m.cols; nval = rows * cols * m.channels(); continuous = m.isContinuous();
394 }
395 else if (kind == _InputArray::UMAT)
396 {
397 const UMat& m = *(const UMat*)sc.getObj();
398 if ((m.rows != 1 && m.cols != 1) || m.dims > 2) return false;
399 rows = m.rows; cols = m.cols; nval = rows * cols * m.channels(); continuous = m.isContinuous();
400 }
401 else
402 return false;
403 if (!continuous || nval > 4)
404 return false;
405 return (rows == 1 && cols == 1) || (rows == cn && cols == 1) || (rows == 1 && cols == cn);
406}

408// The size of the caller-provided stack buffer for scalarArgElems (4 slots of the widest depth).
409// A scalar is <= 4 values by contract, so it NEVER touches the heap.
410enum { EW_SCALAR_BUF_SIZE = 4 * sizeof(double) };
411
412// A scalar operand's raw payload: the MATX inline storage or the Mat data, both returned in place.
413// A UMAT scalar's values are copied device->host into `scbuf` (>= EW_SCALAR_BUF_SIZE bytes, on the
414// CALLER's stack - no heap, no UMat::getMat mapping machinery for 32 bytes of data).
415// p/d receive the data pointer and depth; returns the value count (elems x channels).
416inline int scalarArgElems(const _InputArray& sc, const uchar*& p, int& d, uchar* scbuf)
417{
418 const _InputArray::KindFlag kind = sc.kind();
419 if (kind == _InputArray::MAT)
420 {
421 const Mat& m = *(const Mat*)sc.getObj();
422 p = m.data;
423 d = m.depth();
424 return (int)m.total() * m.channels();
425 }
426 if (kind == _InputArray::UMAT)
427 {
428 const UMat& u = *(const UMat*)sc.getObj();
429 d = u.depth();
430 int n = (int)u.total() * u.channels();
431 CV_Assert(n * (int)CV_ELEM_SIZE1(d) <= (int)EW_SCALAR_BUF_SIZE);
432 Mat header(u.dims, u.size.p, u.type(), scbuf); // header over the caller's stack buffer
433 u.copyTo(header); // create() is a no-op (exact match) -> the
434 p = scbuf; // copy lands straight in scbuf
435 return n;
436 }
437 p = (const uchar*)sc.getObj();
438 d = sc.depth();
439 Size sz = sc.getSz();
440 return sz.width * sz.height;
441}

443// Read one element of depth `d` at p as a double (no Mat, no convertTo, no dispatcher).
444inline double elemToDouble(int d, const uchar* p)
445{
446 switch (d)
447 {
448 case CV_8U: return *(const uchar*)p;
449 case CV_8S: return *(const schar*)p;
450 case CV_16U: return *(const ushort*)p;
451 case CV_16S: return *(const short*)p;
452 case CV_32U: return *(const unsigned*)p;
453 case CV_32S: return *(const int*)p;
454 case CV_64U: return (double)*(const uint64_t*)p;
455 case CV_64S: return (double)*(const int64_t*)p;
456 case CV_16F: return (float)*(const hfloat*)p;
457 case CV_16BF: return (float)*(const bfloat*)p;
458 case CV_32F: return *(const float*)p;
459 case CV_64F: return *(const double*)p;
460 default: CV_Error(Error::StsUnsupportedFormat, "unsupported scalar depth");
461 }
462}

464// Extract a scalar operand's values (see isScalarArg) as up to 4 doubles, straight from its
465// storage. Returns the element count.
466inline int readScalarArg(const _InputArray& sc, Scalar& out)
467{
468 out = Scalar(); // unused channels stay 0 (independent of the caller's Scalar)
469 const uchar* p; int d; uchar scbuf[EW_SCALAR_BUF_SIZE];
470 int n = scalarArgElems(sc, p, d, scbuf);
471 CV_Assert(n <= 4); // Scalar holds 4 slots; isScalarArg admits more only for MATX
472 size_t esz = CV_ELEM_SIZE1(d);
473 for (int i = 0; i < n; i++)
474 out[i] = elemToDouble(d, p + (size_t)i * esz);
475 return n;
476}
