35 // ONNX Attention optional-input layout (Q, K, V are always inputs 0..2):
36 // inputs.size() == 4 : Q, K, V, attn_mask
37 // inputs.size() == 5 : Q, K, V, past_key, past_value (no mask)
38 // inputs.size() == 6 : Q, K, V, attn_mask, past_key, past_value
39 static bool hasMaskInput(size_t n_inputs) { return n_inputs == 4 || n_inputs == 6; }
40 static int pastKeyIdx(size_t n_inputs) { return n_inputs == 5 ? 3 : n_inputs == 6 ? 4 : -1; }
41 static int pastValueIdx(size_t n_inputs) { return n_inputs == 5 ? 4 : n_inputs == 6 ? 5 : -1; }

43 virtual void getTypes(const std::vector<MatType>&inputs,
44 const int requiredOutputs,
45 const int requiredInternals,
46 std::vector<MatType>&outputs,
47 std::vector<MatType>&internals) const CV_OVERRIDE {
48 // type checks
49 CV_CheckTrue(inputs.size() >= 3, "At least three inputs (query, key, value) are required");
50
51 for (int i = 0; i < 3; i++) {
52 CV_CheckType(inputs[i], inputs[i] == CV_16F || inputs[i] == CV_32F, "");
53 }
54
55 CV_CheckType(inputs[0], inputs[0] == inputs[1] && inputs[0] == inputs[2], "");
56
57 if (hasMaskInput(inputs.size())) {
58 CV_CheckType(inputs[3], inputs[3] == CV_8U || inputs[3] == CV_8S ||
59 inputs[3] == CV_16U || inputs[3] == CV_16S ||
60 inputs[3] == CV_32S || inputs[3] == CV_64S ||
61 inputs[3] == CV_64U || inputs[3] == CV_Bool ||
62 inputs[3] == inputs[2], ""); // attention_mask
63 }
64
65 const int past_k_idx = pastKeyIdx(inputs.size());
66 if (past_k_idx >= 0) {
67 CV_CheckType(inputs[past_k_idx], inputs[past_k_idx] == inputs[0], ""); // past_key
68 CV_CheckType(inputs[past_k_idx+1], inputs[past_k_idx+1] == inputs[0], ""); // past_value
69 }
70
71 outputs.assign(requiredOutputs, inputs[0]);
72
73 // internals:
74 internals.clear();
75
76 // 1. attention_prob
77 internals.push_back(inputs[0]);
78 }

80 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
81 const int requiredOutputs,
82 std::vector<MatShape> &outputs,
83 std::vector<MatShape> &internals) const CV_OVERRIDE {
84 CV_CheckTrue(inputs.size() >= 3, "At least three inputs (query, key, value) are required");
85 CV_CheckTrue(inputs[0].dims == inputs[1].dims &&
86 inputs[0].dims == inputs[2].dims,
87 "Query, key and value must have the same number of dimensions");
88
89 const int input_dims = inputs[0].dims;
90 CV_CheckTrue(
91 input_dims == 4 || (q_num_heads > 0 && kv_num_heads > 0 && input_dims == 3),
92 "Input dimensions must be 4D or 3D (in the latter case, q_num_heads and kv_num_heads must be set)"
93 );
94
95 const int seq_dim = input_dims - 2;
96 const int batch_size = inputs[0][0];
97 const int seq_len_q = inputs[0][seq_dim];
98 int seq_len_k = inputs[1][seq_dim];
99 int seq_len_v = inputs[2][seq_dim];
100
101 int past_seq_kv = 0;
102 Net::Impl* netimpl = getNetImpl(const_cast<AttentionOnnxAiLayerImpl*>(this));
103 if (netimpl && netimpl->useKVCache) {
104 KVCacheManager& kvCacheManager = netimpl->kvCacheManager;
105 if (kvCacheManager.isInitialized) {
106 auto it_k = kvCacheManager.kData.find(name);
107 CV_Assert(it_k != kvCacheManager.kData.end());
108 if (it_k != kvCacheManager.kData.end())
109 seq_len_k += it_k->second.getNumTokens();
110
111 auto it_v = kvCacheManager.vData.find(name);
112 CV_Assert(it_v != kvCacheManager.vData.end());
113 if (it_v != kvCacheManager.vData.end())
114 seq_len_v += it_v->second.getNumTokens();
115 }
116 } else {
117 const int past_k_idx = pastKeyIdx(inputs.size());
118 if (past_k_idx >= 0) {
119 const int past_v_idx = pastValueIdx(inputs.size());
120 CV_CheckTrue(past_v_idx >= 0 && past_v_idx < (int)inputs.size(),
121 "past_key and past_value must be provided as a pair");
122
123 // past_key/past_value are always 4D [batch, nhkv, past_seq, head] (even for 3D Q/K/V).
124 const MatShape& pk = inputs[past_k_idx];
125 const MatShape& pv = inputs[past_v_idx];
126 CV_CheckEQ(pk.dims, 4, "past_key must be 4D [batch, nhkv, past_seq, head]");
127 CV_CheckEQ(pv.dims, 4, "past_value must be 4D [batch, nhkv, past_seq, head]");
128 CV_CheckEQ(pk[0], batch_size, "past_key batch dimension must match query batch");
129 CV_CheckEQ(pv[0], batch_size, "past_value batch dimension must match query batch");
130
131 past_seq_kv = pk[pk.dims - 2];
132 seq_len_k += past_seq_kv;
133 seq_len_v += past_seq_kv;
134 }
135 }
136
137 const int q_hn = input_dims == 4 ?
138 inputs[0][1] : q_num_heads;
139 const int kv_hn =input_dims == 4 ?
140 inputs[1][1] : kv_num_heads;
141
142 CV_CheckTrue(seq_len_v == seq_len_k,
143 "Key and value sequence lengths must be equal");
144 const int nhq = input_dims == 4 ? inputs[0][1] : q_num_heads;
145 const int total_seq_kv = seq_len_k;
146
147 CV_CheckTrue(q_hn % kv_hn == 0,
148 "q_num_heads must be divisible by kv_num_heads");
149
150 if (input_dims == 3)
151 {
152 CV_CheckTrue(kv_hn > 0,
153 "For 3D input, kv_num_heads must be greater than 0 (this normally means that kv_num_heads is not set)");
154 CV_CheckTrue(q_hn > 0,
155 "For 3D input, q_num_heads must be greater than 0 (this normally means that q_num_heads is not set)");
156
157 int v_head_size = inputs[2][2] / kv_hn;
158 MatShape output_shape{batch_size, seq_len_q, v_head_size * q_num_heads};
159 outputs.push_back(output_shape);
160 }
161 else
162 {
163 int v_head_size = inputs[2][3];
164 MatShape output_shape{batch_size, nhq, seq_len_q, v_head_size};
165 outputs.push_back(output_shape);
166 }
167
168 const int qk_head_size = input_dims == 4 ? inputs[1][3] : inputs[1][2] / kv_hn;
169 const int v_head_size = input_dims == 4 ? inputs[2][3] : inputs[2][2] / kv_hn;
170 if (requiredOutputs > 1)
171 outputs.push_back(MatShape{batch_size, kv_hn, total_seq_kv, qk_head_size}); // present_key
172 if (requiredOutputs > 2)
173 outputs.push_back(MatShape{batch_size, kv_hn, total_seq_kv, v_head_size}); // present_value
174 if (requiredOutputs > 3)
175 outputs.push_back(MatShape{batch_size, nhq, seq_len_q, total_seq_kv}); // qk_matmul_output
176
177 MatShape attention_prob_shape{batch_size , nhq, seq_len_q, total_seq_kv};
178 internals.push_back(attention_prob_shape);
179
180 return false;
181 }

183 virtual int64 getFLOPS(const std::vector<MatShape> &inputs,
184 const std::vector<MatShape> &outputs) const CV_OVERRIDE
185 {
186 const int input_dims = inputs[0].dims;
187 int64 B = inputs[0][0];
188 int64 Sq = inputs[0][input_dims - 2];
189 int64 Skv = inputs[1][input_dims - 2];
190 int64 nhq = input_dims == 4 ? inputs[0][1] : q_num_heads;
191 int64 qk_head = input_dims == 4 ? inputs[0][3] : inputs[0][2] / nhq;
192 int64 nhkv = input_dims == 4 ? inputs[1][1] : kv_num_heads;
193 int64 v_head = input_dims == 4 ? inputs[2][3] : inputs[2][2] / nhkv;
194
195 // QK^T: batch * nhq * (2 * Sq * Skv * qk_head)
196 int64 flops = B * nhq * CV_BIG_INT(2) * Sq * Skv * qk_head;
197 // Softmax: ~4 ops per element
198 flops += B * nhq * 4 * Sq * Skv;
199 // Attention * V: batch * nhq * (2 * Sq * v_head * Skv)
200 flops += B * nhq * CV_BIG_INT(2) * Sq * v_head * Skv;
201 return flops;
202 }

204 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE {
205 opt.init();
206
207 Net::Impl* netimpl = getNetImpl(this);
208 bool with_kv_cache = false;
209
210 if (netimpl && netimpl->useKVCache) {
211 with_kv_cache = netimpl->kvCacheManager.isInitialized;
212 }
213
214 if (inputs_arr.depth() == CV_16F)
215 {
216 forward_fallback(inputs_arr, outputs_arr, internals_arr);
217 return;
218 }
219
220 std::vector<Mat> inputs, outputs, internals;
221 inputs_arr.getMatVector(inputs);
222 outputs_arr.getMatVector(outputs);
223 internals_arr.getMatVector(internals);
224 Mat &attention_prob = internals[internals.size() - 1];
225
226 const int input_dims = inputs[0].dims;
227 const int seq_dim = input_dims - 2;
228
229 const int batch_size = inputs[0].size[0];
230
231 const int seq_len_q = inputs[0].size[seq_dim];
232 int seq_len_kv = inputs[1].size[seq_dim];
233
234 const int nhq = input_dims == 3 ?
235 q_num_heads :
236 inputs[0].size[1];
237
238 const int qk_head_size = input_dims == 3 ?
239 inputs[0].size[2] / nhq :
240 inputs[0].size[3];
241
242 const int nhkv = input_dims == 3 ?
243 kv_num_heads :
244 inputs[1].size[1];
245
246 const int v_head_size = input_dims == 3 ?
247 inputs[2].size[2] / nhkv :
248 inputs[2].size[3];
249
250 const bool has_mask_input = hasMaskInput(inputs.size());
251 const Mat mask_mat = has_mask_input ? inputs[3] : Mat();
252
253 scale = is_scale_set ? scale : 1.0f / std::sqrt(static_cast<float>(qk_head_size));
254
255 std::vector<size_t> _q_offsets, _k_offsets, _v_offsets,
256 _a_offsets, _o_offsets;
257
258 int past_seq_kv = 0;
259
260 if (with_kv_cache){
261 KVCacheManager& kvCacheManager = netimpl->kvCacheManager;
262
263 auto it_k = kvCacheManager.kData.find(name);
264 CV_Assert(it_k != kvCacheManager.kData.end());
265 KCache&kData = it_k->second;
266
267 kData.grow(inputs[1]);
268
269 const std::vector<Mat> kCachePages = kData.getActivePages();
270 seq_len_kv = kData.getNumTokens();
271
272 pagedAttnQKGemm(
273 inputs[0], kCachePages, attention_prob,
274 seq_len_q, nhq, nhkv, kData.getPageSize(),
275 qk_head_size, seq_len_kv,
276 scale, opt
277 );
278
279 past_seq_kv = seq_len_kv - seq_len_q;
280
281 fused_softmax_softcap_mask(
282 attention_prob, mask_mat,
283 softcap, softcap > 0.f, 9.f, -FLT_MAX,
284 has_mask_input, is_causal, past_seq_kv
285 );
286
287 auto it_v = kvCacheManager.vData.find(name);
288 CV_Assert(it_v != kvCacheManager.vData.end());
289 VCache& vData = it_v->second;
290
291 vData.grow(inputs[2]);
292 seq_len_kv = vData.getNumTokens();
293
294 pagedAttnAVGemm(
295 attention_prob, vData.getActivePages(), outputs[0],
296 seq_len_q, nhq, nhkv, vData.getPageSize() , v_head_size, seq_len_kv,
297 opt
298 );
299 return;
300 }
301
302 // Standard (non-paged) path, with optional past_key/past_value graph inputs
303 const int past_k_idx = pastKeyIdx(inputs.size());
304 const int past_v_idx = pastValueIdx(inputs.size());
305 const bool use_past = past_k_idx >= 0;
306 // past_key/past_value are always 4D [batch, nhkv, past_seq, head]; use their own rank.
307 past_seq_kv = use_past ? inputs[past_k_idx].size[inputs[past_k_idx].dims - 2] : 0;
308 const int total_seq_kv = past_seq_kv + seq_len_kv;
309
310 Mat K_eff, V_eff;
311 if (use_past && past_seq_kv > 0) {
312 if (input_dims == 4) {
313 // 4D [batch, nhkv, seq, head]: concat on axis 2.
314 int dims_k[4] = {batch_size, nhkv, total_seq_kv, qk_head_size};
315 int dims_v[4] = {batch_size, nhkv, total_seq_kv, v_head_size};
316 K_eff.create(4, dims_k, CV_32F);
317 V_eff.create(4, dims_v, CV_32F);
318 const std::vector<Range> past_r{Range::all(), Range::all(), Range(0, past_seq_kv), Range::all()};
319 const std::vector<Range> cur_r {Range::all(), Range::all(), Range(past_seq_kv, total_seq_kv), Range::all()};
320 inputs[past_k_idx].copyTo(K_eff(past_r)); inputs[1].copyTo(K_eff(cur_r));
321 inputs[past_v_idx].copyTo(V_eff(past_r)); inputs[2].copyTo(V_eff(cur_r));
322 } else {
323 const int k_elem = nhkv * qk_head_size;
324 const int v_elem = nhkv * v_head_size;
325 int dims_k[3] = {batch_size, total_seq_kv, k_elem};
326 int dims_v[3] = {batch_size, total_seq_kv, v_elem};
327 int pk_sz[3] = {batch_size, past_seq_kv, k_elem};
328 int pv_sz[3] = {batch_size, past_seq_kv, v_elem};
329 K_eff.create(3, dims_k, CV_32F);
330 V_eff.create(3, dims_v, CV_32F);
331 const std::vector<Range> past_r{Range::all(), Range(0, past_seq_kv), Range::all()};
332 const std::vector<Range> cur_r {Range::all(), Range(past_seq_kv, total_seq_kv), Range::all()};
333 Mat pk, pv;
334 cv::transposeND(inputs[past_k_idx], {0, 2, 1, 3}, pk);
335 cv::transposeND(inputs[past_v_idx], {0, 2, 1, 3}, pv);
336 pk.reshape(1, 3, pk_sz).copyTo(K_eff(past_r)); inputs[1].copyTo(K_eff(cur_r));
337 pv.reshape(1, 3, pv_sz).copyTo(V_eff(past_r)); inputs[2].copyTo(V_eff(cur_r));
338 }
339 } else {
340 K_eff = inputs[1];
341 V_eff = inputs[2];
342 }
343
344 const auto* Q = inputs[0].ptr<const float>();
345 const auto* K = K_eff.ptr<const float>();
346 const auto* V = V_eff.ptr<const float>();
347
348 const int num_gq_groups = nhq / nhkv;
349 const auto seq_len_square = seq_len_q * total_seq_kv;
350
351 _q_offsets.resize(nhq * batch_size);
352 _k_offsets.resize(nhq * batch_size);
353 _a_offsets.resize(nhq * batch_size);
354 _v_offsets.resize(nhq * batch_size);
355 _o_offsets.resize(nhq * batch_size);
356
357 for (int b = 0; b < batch_size; b++)
358 for (int n = 0; n < nhq; n++){
359 _q_offsets[b * nhq + n] =
360 b * seq_len_q * qk_head_size * nhq +
361 (input_dims == 3 ? n * qk_head_size : n * qk_head_size * seq_len_q);
362 _k_offsets[b * nhq + n] =
363 b * total_seq_kv * qk_head_size * nhkv +
364 (n / num_gq_groups * qk_head_size) * (input_dims == 3 ? 1 : total_seq_kv);
365 _v_offsets[b * nhq + n] =
366 b * total_seq_kv * v_head_size * nhkv +
367 (n / num_gq_groups * v_head_size) * (input_dims == 3 ? 1 : total_seq_kv);
368 _a_offsets[b * nhq + n] =
369 b * seq_len_square * nhq +
370 n * seq_len_square;
371 _o_offsets[b * nhq + n] =
372 b * seq_len_q * v_head_size * nhq +
373 (input_dims == 3 ? n * v_head_size : n * v_head_size * seq_len_q);
374 }
375
376 const int ldq0 = input_dims == 3 ? qk_head_size * nhq : qk_head_size;
377 const int ldk0 = input_dims == 3 ? qk_head_size * nhkv : qk_head_size;
378
379 fastGemmBatch(
380 batch_size * nhq,
381 _q_offsets.data(), _k_offsets.data(), _a_offsets.data(),
382 seq_len_q, total_seq_kv, qk_head_size , scale,
383 Q, ldq0, 1,
384 K, 1, ldk0,
385 0.f,
386 attention_prob.ptr<float>(), total_seq_kv,
387 opt
388 );
389
390 // qk_matmul_output (optional 4th output), per qk_matmul_output_mode:
391 // 0 = raw scaled QK^T, 1 = + attention bias, 2 = + softcap, 3 = post-softmax.
392 const bool want_qk = outputs.size() > 3 && !outputs[3].empty();
393 if (want_qk && qk_matmul_output_mode == 0) {
394 attention_prob.copyTo(outputs[3]);
395 } else if (want_qk && (qk_matmul_output_mode == 1 || qk_matmul_output_mode == 2)) {
396 attention_prob.copyTo(outputs[3]);
397 fused_softmax_softcap_mask(
398 outputs[3], mask_mat,
399 softcap, (qk_matmul_output_mode == 2) && (softcap > 0.f), 9.f,
400 -std::numeric_limits<float>::infinity(),
401 has_mask_input, is_causal, past_seq_kv, /*do_softmax=*/false
402 );
403 }
404
405 fused_softmax_softcap_mask(
406 attention_prob, mask_mat,
407 softcap, softcap > 0.f, 9.f, -FLT_MAX,
408 has_mask_input, is_causal, past_seq_kv
409 );
410
411 if (want_qk && qk_matmul_output_mode == 3)
412 attention_prob.copyTo(outputs[3]);
413
414 const int ldv0 = input_dims == 3 ? v_head_size * nhkv : v_head_size;
415 const int ldout = input_dims == 3 ? v_head_size * nhq : v_head_size;
416
417 fastGemmBatch(
418 batch_size * nhq,
419 _a_offsets.data(), _v_offsets.data(), _o_offsets.data(),
420 seq_len_q, v_head_size, total_seq_kv, 1.f,
421 attention_prob.ptr<float>(), total_seq_kv, 1,
422 V, ldv0, 1,
423 0.f,
424 outputs[0].ptr<float>(), ldout,
425 opt
426 );
427
428 auto writePresent = [&](Mat& out, const Mat& eff, int head_size) {
429 if (out.empty()) return;
430 if (input_dims == 4) {
431 eff.reshape(1, out.dims, out.size.p).copyTo(out);
432 } else {
433 int sz[4] = {batch_size, total_seq_kv, nhkv, head_size};
434 cv::transposeND(eff.reshape(1, 4, sz), {0, 2, 1, 3}, out);
435 }
436 };
437 if (outputs.size() > 1) writePresent(outputs[1], K_eff, qk_head_size);
438 if (outputs.size() > 2) writePresent(outputs[2], V_eff, v_head_size);
439 }
440
441 private:
442 bool is_causal;
443 int q_num_heads;
444 int qk_matmul_output_mode;
445 float scale;
446 bool is_scale_set = false;
447 float softcap;
448 int softmax_precision;
449 FastGemmOpt opt;
450};
