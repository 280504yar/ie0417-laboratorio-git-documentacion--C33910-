57double cv::fisheye::calibrate(InputArrayOfArrays objectPoints, InputArrayOfArrays imagePoints, const Size& image_size,
58 InputOutputArray K, InputOutputArray D, OutputArrayOfArrays rvecs, OutputArrayOfArrays tvecs,
59 int flags , cv::TermCriteria criteria)
60{
61 CV_INSTRUMENT_REGION();
62
63 CV_Assert(!objectPoints.empty() && !imagePoints.empty() && objectPoints.total() == imagePoints.total());
64 CV_Assert(objectPoints.type() == CV_32FC3 || objectPoints.type() == CV_64FC3);
65 CV_Assert(imagePoints.type() == CV_32FC2 || imagePoints.type() == CV_64FC2);
66 CV_Assert(K.empty() || (K.size() == Size(3,3)));
67 CV_Assert(D.empty() || (D.total() == 4));
68 CV_Assert(rvecs.empty() || (rvecs.channels() == 3));
69 CV_Assert(tvecs.empty() || (tvecs.channels() == 3));
70
71 CV_Assert((!K.empty() && !D.empty()) || !(flags & CALIB_USE_INTRINSIC_GUESS));
72
73 using namespace cv::internal;
74 //-------------------------------Initialization
75 IntrinsicParams finalParam;
76 IntrinsicParams currentParam;
77 IntrinsicParams errors;
78
79 finalParam.isEstimate[0] = flags & CALIB_FIX_FOCAL_LENGTH ? 0 : 1;
80 finalParam.isEstimate[1] = flags & CALIB_FIX_FOCAL_LENGTH ? 0 : 1;
81 finalParam.isEstimate[2] = flags & CALIB_FIX_PRINCIPAL_POINT ? 0 : 1;
82 finalParam.isEstimate[3] = flags & CALIB_FIX_PRINCIPAL_POINT ? 0 : 1;
83 finalParam.isEstimate[4] = flags & CALIB_FIX_SKEW ? 0 : 1;
84 finalParam.isEstimate[5] = flags & CALIB_FIX_K1 ? 0 : 1;
85 finalParam.isEstimate[6] = flags & CALIB_FIX_K2 ? 0 : 1;
86 finalParam.isEstimate[7] = flags & CALIB_FIX_K3 ? 0 : 1;
87 finalParam.isEstimate[8] = flags & CALIB_FIX_K4 ? 0 : 1;
88
89 const int recompute_extrinsic = flags & CALIB_RECOMPUTE_EXTRINSIC ? 1: 0;
90 const int check_cond = flags & CALIB_CHECK_COND ? 1 : 0;
91
92 const double alpha_smooth = 0.4;
93 const double thresh_cond = 1e6;
94 double change = 1;
95 Vec2d err_std;
96
97 Matx33d _K;
98 Vec4d _D;
99 if (flags & CALIB_USE_INTRINSIC_GUESS)
100 {
101 K.getMat().convertTo(_K, CV_64FC1);
102 D.getMat().convertTo(_D, CV_64FC1);
103 finalParam.Init(Vec2d(_K(0,0), _K(1, 1)),
104 Vec2d(_K(0,2), _K(1, 2)),
105 Vec4d(flags & CALIB_FIX_K1 ? 0 : _D[0],
106 flags & CALIB_FIX_K2 ? 0 : _D[1],
107 flags & CALIB_FIX_K3 ? 0 : _D[2],
108 flags & CALIB_FIX_K4 ? 0 : _D[3]),
109 _K(0, 1) / _K(0, 0));
110 }
111 else
112 {
113 finalParam.Init(Vec2d(max(image_size.width, image_size.height) / 2., max(image_size.width, image_size.height) / 2.),
114 Vec2d(image_size.width / 2.0 - 0.5, image_size.height / 2.0 - 0.5));
115 }
116
117 errors.isEstimate = finalParam.isEstimate;
118
119 std::vector<Vec3d> omc(objectPoints.total()), Tc(objectPoints.total());
120
121 CalibrateExtrinsics(objectPoints, imagePoints, finalParam, check_cond, thresh_cond, omc, Tc);
122
123
124 //-------------------------------Optimization
125 for(int iter = 0; iter < std::numeric_limits<int>::max(); ++iter)
126 {
127 if ((criteria.type == 1 && iter >= criteria.maxCount) ||
128 (criteria.type == 2 && change <= criteria.epsilon) ||
129 (criteria.type == 3 && (change <= criteria.epsilon || iter >= criteria.maxCount)))
130 break;
131
132 double alpha_smooth2 = 1 - std::pow(1 - alpha_smooth, iter + 1.0);
133
134 Mat JJ2, ex3;
135 ComputeJacobians(objectPoints, imagePoints, finalParam, omc, Tc, check_cond,thresh_cond, JJ2, ex3);
136
137 Mat G;
138 solve(JJ2, ex3, G);
139 currentParam = finalParam + alpha_smooth2*G;
140
141 change = norm(Vec4d(currentParam.f[0], currentParam.f[1], currentParam.c[0], currentParam.c[1]) -
142 Vec4d(finalParam.f[0], finalParam.f[1], finalParam.c[0], finalParam.c[1]))
143 / norm(Vec4d(currentParam.f[0], currentParam.f[1], currentParam.c[0], currentParam.c[1]));
144
145 finalParam = currentParam;
146
147 if (recompute_extrinsic)
148 {
149 CalibrateExtrinsics(objectPoints, imagePoints, finalParam, check_cond,
150 thresh_cond, omc, Tc);
151 }
152 }
153
154 //-------------------------------Validation
155 double rms;
156 EstimateUncertainties(objectPoints, imagePoints, finalParam, omc, Tc, errors, err_std, thresh_cond,
157 check_cond, rms);
158
159 //-------------------------------
160 _K = Matx33d(finalParam.f[0], finalParam.f[0] * finalParam.alpha, finalParam.c[0],
161 0, finalParam.f[1], finalParam.c[1],
162 0, 0, 1);
163
164 if (K.needed()) Mat(_K).convertTo(K, K.empty() ? CV_64FC1 : K.type());
165 if (D.needed()) Mat(finalParam.k).convertTo(D, D.empty() ? CV_64FC1 : D.type());
166 if (rvecs.isMatVector())
167 {
168 int N = (int)objectPoints.total();
169
170 if(rvecs.empty())
171 rvecs.create(N, 1, CV_64FC3);
172
173 if(tvecs.empty())
174 tvecs.create(N, 1, CV_64FC3);
175
176 for(int i = 0; i < N; i++ )
177 {
178 rvecs.create(3, 1, CV_64F, i, true);
179 tvecs.create(3, 1, CV_64F, i, true);
180 memcpy(rvecs.getMat(i).ptr(), omc[i].val, sizeof(Vec3d));
181 memcpy(tvecs.getMat(i).ptr(), Tc[i].val, sizeof(Vec3d));
182 }
183 }
184 else
185 {
186 if (rvecs.needed()) Mat(omc).reshape(3, (int)omc.size()).convertTo(rvecs, rvecs.empty() ? CV_64FC3 : rvecs.type());
187 if (tvecs.needed()) Mat(Tc).reshape(3, (int)Tc.size()).convertTo(tvecs, tvecs.empty() ? CV_64FC3 : tvecs.type());
188 }
189
190 return rms;
191}

252 calibrate(objectPoints, imagePoints1, imageSize, _K1, _D1, rvecs1, tvecs1, flags, TermCriteria(3, 20, 1e-6));

256 intrinsicLeft.Init(Vec2d(_K1(0,0), _K1(1, 1)), Vec2d(_K1(0,2), _K1(1, 2)),

260 Vec4d(_D2[0], _D2[1], _D2[2], _D2[3]), _K2(0, 1) / _K2(0, 0));

271 intrinsicLeft.isEstimate[3] = flags & CALIB_FIX_INTRINSIC ? 0 : 1;
272 intrinsicLeft.isEstimate[4] = flags & (CALIB_FIX_SKEW | CALIB_FIX_INTRINSIC) ? 0 : 1;
273 intrinsicLeft.isEstimate[5] = flags & (CALIB_FIX_K1 | CALIB_FIX_INTRINSIC) ? 0 : 1;
274 intrinsicLeft.isEstimate[6] = flags & (CALIB_FIX_K2 | CALIB_FIX_INTRINSIC) ? 0 : 1;
275 intrinsicLeft.isEstimate[7] = flags & (CALIB_FIX_K3 | CALIB_FIX_INTRINSIC) ? 0 : 1;
276 intrinsicLeft.isEstimate[8] = flags & (CALIB_FIX_K4 | CALIB_FIX_INTRINSIC) ? 0 : 1;
277
278 intrinsicRight.isEstimate[0] = flags & CALIB_FIX_INTRINSIC ? 0 : 1;
279 intrinsicRight.isEstimate[1] = flags & CALIB_FIX_INTRINSIC ? 0 : 1;
280 intrinsicRight.isEstimate[2] = flags & CALIB_FIX_INTRINSIC ? 0 : 1;
281 intrinsicRight.isEstimate[3] = flags & CALIB_FIX_INTRINSIC ? 0 : 1;
282 intrinsicRight.isEstimate[4] = flags & (CALIB_FIX_SKEW | CALIB_FIX_INTRINSIC) ? 0 : 1;
283 intrinsicRight.isEstimate[5] = flags & (CALIB_FIX_K1 | CALIB_FIX_INTRINSIC) ? 0 : 1;
284 intrinsicRight.isEstimate[6] = flags & (CALIB_FIX_K2 | CALIB_FIX_INTRINSIC) ? 0 : 1;
285 intrinsicRight.isEstimate[7] = flags & (CALIB_FIX_K3 | CALIB_FIX_INTRINSIC) ? 0 : 1;
286 intrinsicRight.isEstimate[8] = flags & (CALIB_FIX_K4 | CALIB_FIX_INTRINSIC) ? 0 : 1;

288 intrinsicLeft_errors.isEstimate = intrinsicLeft.isEstimate;
289 intrinsicRight_errors.isEstimate = intrinsicRight.isEstimate;
290
291 std::vector<uchar> selectedParams;
292 std::vector<uchar> tmp(6 * (n_images + 1), 1);
293 selectedParams.insert(selectedParams.end(), intrinsicLeft.isEstimate.begin(), intrinsicLeft.isEstimate.end());
294 selectedParams.insert(selectedParams.end(), intrinsicRight.isEstimate.begin(), intrinsicRight.isEstimate.end());
295 selectedParams.insert(selectedParams.end(), tmp.begin(), tmp.end());

297 //Init values for rotation and translation between two views
298 Mat om_list(1, n_images, CV_64FC3), T_list(1, n_images, CV_64FC3);
299 Mat om_ref, R_ref, T_ref, R1, R2;
300 for (int image_idx = 0; image_idx < n_images; ++image_idx)
301 {
302 Rodrigues(rvecs1[image_idx], R1);
303 Rodrigues(rvecs2[image_idx], R2);
304 R_ref = R2 * R1.t();
305 T_ref = Mat(tvecs2[image_idx]) - R_ref * Mat(tvecs1[image_idx]);
306 Rodrigues(R_ref, om_ref);
307 om_ref.reshape(3, 1).copyTo(om_list.col(image_idx));
308 T_ref.reshape(3, 1).copyTo(T_list.col(image_idx));
309 }
310 Vec3d omcur = cv::internal::median3d(om_list);
311 Vec3d Tcur = cv::internal::median3d(T_list);
312
313 Mat J = Mat::zeros(4 * n_points * n_images, 18 + 6 * (n_images + 1), CV_64FC1),
314 e = Mat::zeros(4 * n_points * n_images, 1, CV_64FC1), Jkk, ekk;

316 for(int iter = 0; ; ++iter)
317 {
318 if ((criteria.type == 1 && iter >= criteria.maxCount) ||
319 (criteria.type == 2 && change <= criteria.epsilon) ||
320 (criteria.type == 3 && (change <= criteria.epsilon || iter >= criteria.maxCount)))
321 break;

323 J.create(4 * n_points * n_images, 18 + 6 * (n_images + 1), CV_64FC1);
324 e.create(4 * n_points * n_images, 1, CV_64FC1);
325 Jkk.create(4 * n_points, 18 + 6 * (n_images + 1), CV_64FC1);
326 ekk.create(4 * n_points, 1, CV_64FC1);
327
328 Mat omr, Tr, domrdomckk, domrdTckk, domrdom, domrdT, dTrdomckk, dTrdTckk, dTrdom, dTrdT;
329
330 for (int image_idx = 0; image_idx < n_images; ++image_idx)
331 {
332 Jkk = Mat::zeros(4 * n_points, 18 + 6 * (n_images + 1), CV_64FC1);

334 Mat object = objectPoints.getMat(image_idx).clone();

336 Mat imageRight = imagePoints2.getMat(image_idx).clone();

341 Mat tvec = Mat(tvecs1[image_idx]);

348 jacobians.colRange(0, 2).copyTo(Jkk.colRange(0, 2).rowRange(0, 2 * n_points));

353 //right camera jacobian

355 rvec = Mat(rvecs2[image_idx]);
356 tvec = Mat(tvecs2[image_idx]);
357
358 cv::internal::projectPoints(object, projected, omr, Tr, intrinsicRight, jacobians);
359
360 pt_diff = imageRight.reshape(1, n_points*2) -
361 projected.reshape(1, n_points*2);
362 pt_diff.copyTo(ekk.rowRange(2 * n_points, 4 * n_points));
363 Mat dxrdom = jacobians.colRange(8, 11) * domrdom + jacobians.colRange(11, 14) * dTrdom;
364 Mat dxrdT = jacobians.colRange(8, 11) * domrdT + jacobians.colRange(11, 14)* dTrdT;

371 dxrdTckk.copyTo(Jkk.colRange(27 + image_idx * 6, 30 + image_idx * 6).rowRange(2 * n_points, 4 * n_points));

373 jacobians.colRange(2, 4).copyTo(Jkk.colRange(9 + 2, 9 + 4).rowRange(2 * n_points, 4 * n_points));
374 jacobians.colRange(4, 8).copyTo(Jkk.colRange(9 + 5, 9 + 9).rowRange(2 * n_points, 4 * n_points));

378 double abs_max = 0;
379 for (int i = 0; i < 4 * n_points; i++)
380 {

382 {

384 }
385 }

388
389 Jkk.copyTo(J.rowRange(image_idx * 4 * n_points, (image_idx + 1) * 4 * n_points));
390 ekk.copyTo(e.rowRange(image_idx * 4 * n_points, (image_idx + 1) * 4 * n_points));
391 }
392
393 Vec6d oldTom(Tcur[0], Tcur[1], Tcur[2], omcur[0], omcur[1], omcur[2]);
394
395 //update all parameters
396 cv::subMatrix(J, J, selectedParams, std::vector<uchar>(J.rows, 1));
397 int a = cv::countNonZero(intrinsicLeft.isEstimate);
398 int b = cv::countNonZero(intrinsicRight.isEstimate);
399 Mat deltas;

401 if (a > 0)
402 intrinsicLeft = intrinsicLeft + deltas.rowRange(0, a);
403 if (b > 0)
404 intrinsicRight = intrinsicRight + deltas.rowRange(a, a + b);

406 Tcur = Tcur + Vec3d(deltas.rowRange(a + b + 3, a + b + 6));
407 for (int image_idx = 0; image_idx < n_images; ++image_idx)
408 {
409 rvecs1[image_idx] = Mat(Mat(rvecs1[image_idx]) + deltas.rowRange(a + b + 6 + image_idx * 6, a + b + 9 + image_idx * 6));
410 tvecs1[image_idx] = Mat(Mat(tvecs1[image_idx]) + deltas.rowRange(a + b + 9 + image_idx * 6, a + b + 12 + image_idx * 6));
411 }
412
413 Vec6d newTom(Tcur[0], Tcur[1], Tcur[2], omcur[0], omcur[1], omcur[2]);
414 change = cv::norm(newTom - oldTom) / cv::norm(newTom);
415 }
416

418 const Vec2d* ptr_e = e.ptr<Vec2d>();
419 for (size_t i = 0; i < e.total() / 2; i++)
420 {
421 rms += ptr_e[i][0] * ptr_e[i][0] + ptr_e[i][1] * ptr_e[i][1];
422 }
423
424 rms /= ((double)e.total() / 2.0);
425 rms = sqrt(rms);

427 _K1 = Matx33d(intrinsicLeft.f[0], intrinsicLeft.f[0] * intrinsicLeft.alpha, intrinsicLeft.c[0],
428 0, intrinsicLeft.f[1], intrinsicLeft.c[1],
429 0, 0, 1);
430
431 _K2 = Matx33d(intrinsicRight.f[0], intrinsicRight.f[0] * intrinsicRight.alpha, intrinsicRight.c[0],
432 0, intrinsicRight.f[1], intrinsicRight.c[1],
433 0, 0, 1);
434
435 Mat _R;
436 Rodrigues(omcur, _R);
437
438 if (K1.needed()) Mat(_K1).convertTo(K1, K1.empty() ? CV_64FC1 : K1.type());
439 if (K2.needed()) Mat(_K2).convertTo(K2, K2.empty() ? CV_64FC1 : K2.type());
440 if (D1.needed()) Mat(intrinsicLeft.k).convertTo(D1, D1.empty() ? CV_64FC1 : D1.type());
441 if (D2.needed()) Mat(intrinsicRight.k).convertTo(D2, D2.empty() ? CV_64FC1 : D2.type());
442 if (R.needed()) _R.convertTo(R, R.empty() ? CV_64FC1 : R.type());
443 if (T.needed()) Mat(Tcur).convertTo(T, T.empty() ? CV_64FC1 : T.type());

570void cv::internal::ComputeExtrinsicRefine(const Mat& imagePoints, const Mat& objectPoints, Mat& rvec,
571 Mat& tvec, Mat& J, const int MaxIter,
572 const IntrinsicParams& param, const double thresh_cond)
573{
574 CV_Assert(!objectPoints.empty() && objectPoints.type() == CV_64FC3);
575 CV_Assert(!imagePoints.empty() && imagePoints.type() == CV_64FC2);
576 CV_Assert(rvec.total() > 2 && tvec.total() > 2);
577 Vec6d extrinsics(rvec.at<double>(0), rvec.at<double>(1), rvec.at<double>(2),
578 tvec.at<double>(0), tvec.at<double>(1), tvec.at<double>(2));
579 double change = 1;
580 int iter = 0;
581
582 while (change > 1e-10 && iter < MaxIter)
583 {
584 std::vector<Point2d> x;
585 Mat jacobians;
586 projectPoints(objectPoints, x, rvec, tvec, param, jacobians);
587
588 Mat ex = imagePoints - Mat(x);
589 ex = ex.reshape(1, 2);
590
591 J = jacobians.colRange(8, 14).clone();
592
593 SVD svd(J, SVD::NO_UV);
594 double condJJ = svd.w.at<double>(0)/svd.w.at<double>(5);
595
596 if (condJJ > thresh_cond)
597 change = 0;
598 else
599 {
600 Vec6d param_innov;
601 solve(J, ex.reshape(1, (int)ex.total()), param_innov, DECOMP_SVD + DECOMP_NORMAL);
602
603 Vec6d param_up = extrinsics + param_innov;
604 change = norm(param_innov)/norm(param_up);
605 extrinsics = param_up;
606 iter = iter + 1;
607
608 rvec = Mat(Vec3d(extrinsics.val));
609 tvec = Mat(Vec3d(extrinsics.val+3));
610 }
611 }

614cv::Mat cv::internal::ComputeHomography(Mat m, Mat M)
615{
616 CV_INSTRUMENT_REGION();
617
618 int Np = m.cols;
619
620 if (m.rows < 3)
621 {
622 vconcat(m, Mat::ones(1, Np, CV_64FC1), m);
623 }
624 if (M.rows < 3)
625 {
626 vconcat(M, Mat::ones(1, Np, CV_64FC1), M);
627 }
628
629 divide(m, Mat::ones(3, 1, CV_64FC1) * m.row(2), m);
630 divide(M, Mat::ones(3, 1, CV_64FC1) * M.row(2), M);
631
632 Mat ax = m.row(0).clone();
633 Mat ay = m.row(1).clone();
634
635 double mxx = mean(ax)[0];
636 double myy = mean(ay)[0];
637
638 ax = ax - mxx;
639 ay = ay - myy;
640
641 double scxx = mean(abs(ax))[0];
642 double scyy = mean(abs(ay))[0];
643
644 Mat Hnorm (Matx33d( 1/scxx, 0.0, -mxx/scxx,
645 0.0, 1/scyy, -myy/scyy,
646 0.0, 0.0, 1.0 ));
647
648 Mat inv_Hnorm (Matx33d( scxx, 0, mxx,
649 0, scyy, myy,
650 0, 0, 1 ));
651 Mat mn = Hnorm * m;
652
653 Mat L = Mat::zeros(2*Np, 9, CV_64FC1);
654
655 for (int i = 0; i < Np; ++i)
656 {
657 for (int j = 0; j < 3; j++)
658 {
659 L.at<double>(2 * i, j) = M.at<double>(j, i);
660 L.at<double>(2 * i + 1, j + 3) = M.at<double>(j, i);
661 L.at<double>(2 * i, j + 6) = -mn.at<double>(0,i) * M.at<double>(j, i);
662 L.at<double>(2 * i + 1, j + 6) = -mn.at<double>(1,i) * M.at<double>(j, i);
663 }
664 }
665
666 if (Np > 4) L = L.t() * L;
667 SVD svd(L);
668 Mat hh = svd.vt.row(8) / svd.vt.row(8).at<double>(8);
669 Mat Hrem = hh.reshape(1, 3);
670 Mat H = inv_Hnorm * Hrem;
671
672 if (Np > 4)
673 {
674 Mat hhv = H.reshape(1, 9)(Rect(0, 0, 1, 8)).clone();
675 for (int iter = 0; iter < 10; iter++)
676 {
677 Mat mrep = H * M;
678 Mat J = Mat::zeros(2 * Np, 8, CV_64FC1);
679 Mat MMM;
680 divide(M, Mat::ones(3, 1, CV_64FC1) * mrep(Rect(0, 2, mrep.cols, 1)), MMM);
681 divide(mrep, Mat::ones(3, 1, CV_64FC1) * mrep(Rect(0, 2, mrep.cols, 1)), mrep);
682 Mat m_err = m(Rect(0,0, m.cols, 2)) - mrep(Rect(0,0, mrep.cols, 2));
683 m_err = Mat(m_err.t()).reshape(1, m_err.cols * m_err.rows);
684 Mat MMM2, MMM3;
685 multiply(Mat::ones(3, 1, CV_64FC1) * mrep(Rect(0, 0, mrep.cols, 1)), MMM, MMM2);
686 multiply(Mat::ones(3, 1, CV_64FC1) * mrep(Rect(0, 1, mrep.cols, 1)), MMM, MMM3);
687
688 for (int i = 0; i < Np; ++i)
689 {
690 for (int j = 0; j < 3; ++j)
691 {
692 J.at<double>(2 * i, j) = -MMM.at<double>(j, i);
693 J.at<double>(2 * i + 1, j + 3) = -MMM.at<double>(j, i);
694 }
695
696 for (int j = 0; j < 2; ++j)
697 {
698 J.at<double>(2 * i, j + 6) = MMM2.at<double>(j, i);
699 J.at<double>(2 * i + 1, j + 6) = MMM3.at<double>(j, i);
700 }
701 }
702 divide(M, Mat::ones(3, 1, CV_64FC1) * mrep(Rect(0,2,mrep.cols,1)), MMM);
703 Mat hh_innov = (J.t() * J).inv() * (J.t()) * m_err;
704 Mat hhv_up = hhv - hh_innov;
705 Mat tmp;
706 vconcat(hhv_up, Mat::ones(1,1,CV_64FC1), tmp);
707 Mat H_up = tmp.reshape(1,3);
708 hhv = hhv_up;
709 H = H_up;
710 }
711 }
712 return H;
713}

716{
717 CV_INSTRUMENT_REGION();
718
719 CV_Assert(!imagePoints.empty() && imagePoints.type() == CV_64FC2);

733void cv::internal::InitExtrinsics(const Mat& _imagePoints, const Mat& _objectPoints, const IntrinsicParams& param, Mat& omckk, Mat& Tckk)
734{
735 CV_Assert(!_objectPoints.empty() && _objectPoints.type() == CV_64FC3);
736 CV_Assert(!_imagePoints.empty() && _imagePoints.type() == CV_64FC2);
737
738 Mat imagePointsNormalized = NormalizePixels(_imagePoints, param).reshape(1).t();
739 Mat objectPoints = _objectPoints.reshape(1).t();
740 Mat objectPointsMean, covObjectPoints;
741 Mat Rckk;
742 int Np = imagePointsNormalized.cols;
743 calcCovarMatrix(objectPoints, covObjectPoints, objectPointsMean, COVAR_NORMAL | COVAR_COLS);
744 SVD svd(covObjectPoints);
745 Mat R(svd.vt);
746 if (norm(R(Rect(2, 0, 1, 2))) < 1e-6)
747 R = Mat::eye(3,3, CV_64FC1);
748 if (determinant(R) < 0)
749 R = -R;
750 Mat T = -R * objectPointsMean;
751 Mat X_new = R * objectPoints + T * Mat::ones(1, Np, CV_64FC1);
752 Mat H = ComputeHomography(imagePointsNormalized, X_new(Rect(0,0,X_new.cols,2)));
753 double sc = .5 * (norm(H.col(0)) + norm(H.col(1)));
754 H = H / sc;
755 Mat u1 = H.col(0).clone();
756 double norm_u1 = norm(u1);
757 CV_Assert(fabs(norm_u1) > 0);
758 u1 = u1 / norm_u1;
759 Mat u2 = H.col(1).clone() - u1.dot(H.col(1).clone()) * u1;
760 double norm_u2 = norm(u2);
761 CV_Assert(fabs(norm_u2) > 0);
762 u2 = u2 / norm_u2;
763 Mat u3 = u1.cross(u2);
764 Mat RRR;
765 hconcat(u1, u2, RRR);
766 hconcat(RRR, u3, RRR);
767 Rodrigues(RRR, omckk);
768 Rodrigues(omckk, Rckk);
769 Tckk = H.col(2).clone();
770 Tckk = Tckk + Rckk * T;
771 Rckk = Rckk * R;
772 Rodrigues(Rckk, omckk);
773}

775void cv::internal::CalibrateExtrinsics(InputArrayOfArrays objectPoints, InputArrayOfArrays imagePoints,
776 const IntrinsicParams& param, const int check_cond,
777 const double thresh_cond, InputOutputArray omc, InputOutputArray Tc)
778{
779 CV_Assert(!objectPoints.empty() && (objectPoints.type() == CV_32FC3 || objectPoints.type() == CV_64FC3));
780 CV_Assert(!imagePoints.empty() && (imagePoints.type() == CV_32FC2 || imagePoints.type() == CV_64FC2));
781 CV_Assert(omc.type() == CV_64FC3 || Tc.type() == CV_64FC3);
782
783 if (omc.empty()) omc.create(1, (int)objectPoints.total(), CV_64FC3);
784 if (Tc.empty()) Tc.create(1, (int)objectPoints.total(), CV_64FC3);
785
786 const int maxIter = 20;
787
788 for(int image_idx = 0; image_idx < (int)imagePoints.total(); ++image_idx)
789 {
790 Mat omckk, Tckk, JJ_kk;
791 Mat image, object;
792
793 objectPoints.getMat(image_idx).convertTo(object, CV_64FC3);
794 imagePoints.getMat (image_idx).convertTo(image, CV_64FC2);
795
796 bool imT = image.rows < image.cols;
797 bool obT = object.rows < object.cols;
798
799 InitExtrinsics(imT ? image.t() : image, obT ? object.t() : object, param, omckk, Tckk);
800
801 ComputeExtrinsicRefine(!imT ? image.t() : image, !obT ? object.t() : object, omckk, Tckk, JJ_kk, maxIter, param, thresh_cond);
802 if (check_cond)
803 {
804 SVD svd(JJ_kk, SVD::NO_UV);
805 if(svd.w.at<double>(0) / svd.w.at<double>((int)svd.w.total() - 1) > thresh_cond )
806 CV_Error( cv::Error::StsInternal, format("CALIB_CHECK_COND - Ill-conditioned matrix for input array %d",image_idx));
807 }
808 omckk.reshape(3,1).copyTo(omc.getMat().col(image_idx));
809 Tckk.reshape(3,1).copyTo(Tc.getMat().col(image_idx));
810 }

813void cv::internal::ComputeJacobians(InputArrayOfArrays objectPoints, InputArrayOfArrays imagePoints,
814 const IntrinsicParams& param, InputArray omc, InputArray Tc,
815 const int& check_cond, const double& thresh_cond, Mat& JJ2, Mat& ex3)
816{
817 CV_Assert(!objectPoints.empty() && (objectPoints.type() == CV_32FC3 || objectPoints.type() == CV_64FC3));
818 CV_Assert(!imagePoints.empty() && (imagePoints.type() == CV_32FC2 || imagePoints.type() == CV_64FC2));
819
820 CV_Assert(!omc.empty() && omc.type() == CV_64FC3);
821 CV_Assert(!Tc.empty() && Tc.type() == CV_64FC3);
822
823 int n = (int)objectPoints.total();
824
825 JJ2 = Mat::zeros(9 + 6 * n, 9 + 6 * n, CV_64FC1);
826 ex3 = Mat::zeros(9 + 6 * n, 1, CV_64FC1 );
827
828 for (int image_idx = 0; image_idx < n; ++image_idx)
829 {
830 Mat image, object;
831 objectPoints.getMat(image_idx).convertTo(object, CV_64FC3);
832 imagePoints.getMat (image_idx).convertTo(image, CV_64FC2);
833
834 bool imT = image.channels() == 1 && image.rows > image.cols;
835 Mat om(omc.getMat().col(image_idx)), T(Tc.getMat().col(image_idx));
836
837 std::vector<Point2d> x;
838 Mat jacobians;
839 projectPoints(object, x, om, T, param, jacobians);
840 Mat exkk = (imT ? image.t() : image) - Mat(x);
841
842 Mat A(jacobians.rows, 9, CV_64FC1);
843 jacobians.colRange(0, 4).copyTo(A.colRange(0, 4));
844 jacobians.col(14).copyTo(A.col(4));
845 jacobians.colRange(4, 8).copyTo(A.colRange(5, 9));
846
847 A = A.t();
848
849 Mat B = jacobians.colRange(8, 14).clone();
850 B = B.t();
851
852 JJ2(Rect(0, 0, 9, 9)) += A * A.t();
853 JJ2(Rect(9 + 6 * image_idx, 9 + 6 * image_idx, 6, 6)) = B * B.t();
854
855 JJ2(Rect(9 + 6 * image_idx, 0, 6, 9)) = A * B.t();
856 JJ2(Rect(0, 9 + 6 * image_idx, 9, 6)) = JJ2(Rect(9 + 6 * image_idx, 0, 6, 9)).t();
857
858 Mat exkk_col = exkk.reshape(1, 2 * (int)exkk.total());
859 ex3.rowRange(0, 9) += A * exkk_col;
860 ex3.rowRange(9 + 6 * image_idx, 9 + 6 * (image_idx + 1)) = B * exkk_col;
861
862 if (check_cond)
863 {
864 Mat JJ_kk = B.t();
865 SVD svd(JJ_kk, SVD::NO_UV);
866 CV_Assert(svd.w.at<double>(0) / svd.w.at<double>(svd.w.rows - 1) < thresh_cond);
867 }
868 }
869
870 std::vector<uchar> idxs(param.isEstimate);
871 idxs.insert(idxs.end(), 6 * n, 1);
872
873 subMatrix(JJ2, JJ2, idxs, idxs);
874 subMatrix(ex3, ex3, std::vector<uchar>(1, 1), idxs);
875}

877void cv::internal::EstimateUncertainties(InputArrayOfArrays objectPoints, InputArrayOfArrays imagePoints,
878 const IntrinsicParams& params, InputArray omc, InputArray Tc,
879 IntrinsicParams& errors, Vec2d& std_err, double thresh_cond, int check_cond, double& rms)
880{
881 CV_Assert(!objectPoints.empty() && (objectPoints.type() == CV_32FC3 || objectPoints.type() == CV_64FC3));
882 CV_Assert(!imagePoints.empty() && (imagePoints.type() == CV_32FC2 || imagePoints.type() == CV_64FC2));
883
884 CV_Assert(!omc.empty() && omc.type() == CV_64FC3);
885 CV_Assert(!Tc.empty() && Tc.type() == CV_64FC3);
886
887 int total_ex = 0;
888 for (int image_idx = 0; image_idx < (int)objectPoints.total(); ++image_idx)
889 {
890 total_ex += (int)objectPoints.getMat(image_idx).total();
891 }
892 Mat ex(total_ex, 1, CV_64FC2);
893 int insert_idx = 0;
894 for (int image_idx = 0; image_idx < (int)objectPoints.total(); ++image_idx)
895 {
896 Mat image, object;
897 objectPoints.getMat(image_idx).convertTo(object, CV_64FC3);
898 imagePoints.getMat (image_idx).convertTo(image, CV_64FC2);
899
900 bool imT = image.channels() == 1 && image.rows > image.cols;
901
902 Mat om(omc.getMat().col(image_idx)), T(Tc.getMat().col(image_idx));
903
904 std::vector<Point2d> x;
905 projectPoints(object, x, om, T, params, noArray());
906 Mat ex_ = (imT ? image.t() : image) - Mat(x);
907 ex_ = ex_.reshape(2, (int)ex_.total());
908 ex_.copyTo(ex.rowRange(insert_idx, insert_idx + ex_.rows));
909 insert_idx += ex_.rows;
910 }
911
912 meanStdDev(ex, noArray(), std_err);
913 std_err *= sqrt((double)ex.total()/((double)ex.total() - 1.0));
914
915 Vec<double, 1> sigma_x;
916 meanStdDev(ex.reshape(1, 1), noArray(), sigma_x);
917
918 Mat JJ2, ex3;
919 ComputeJacobians(objectPoints, imagePoints, params, omc, Tc, check_cond, thresh_cond, JJ2, ex3);
920
921 sqrt(JJ2.inv(), JJ2);
922
923 int nParams = JJ2.rows;
924 // an explanation of that denominator correction can be found here:
925 // R. Hartley, A. Zisserman, Multiple View Geometry in Computer Vision, 2004, section 5.1.3, page 134
926 // see the discussion for more details: https://github.com/opencv/opencv/pull/22992
927 sigma_x *= sqrt(2.0 * (double)ex.total()/(2.0 * (double)ex.total() - nParams));
928
929 errors = 3 * sigma_x(0) * JJ2.diag();
930 rms = sqrt(norm(ex, NORM_L2SQR)/ex.total());
931}
