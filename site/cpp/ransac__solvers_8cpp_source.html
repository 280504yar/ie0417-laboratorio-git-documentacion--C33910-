95double getLambda (std::vector<int> &supports, double cdf_thr, int points_size,
96 int sample_size, bool is_independent, int &min_non_random_inliers) {
97 std::sort(supports.begin(), supports.end());
98 double lambda = supports.size() % 2 ? (supports[supports.size()/2] + supports[supports.size()/2+1])*0.5 : supports[supports.size()/2];
99 const double cdf = lambda + cdf_thr*sqrt(lambda * (1 - lambda / (is_independent ? points_size - sample_size : points_size)));
100 int lower_than_cdf = 0; lambda = 0;
101 for (const auto &inl : supports)
102 if (inl < cdf) {
103 lambda += inl; lower_than_cdf++;
104 } else break; // list is sorted
105 lambda /= lower_than_cdf;
106 if (lambda < 1 || lower_than_cdf == 0) lambda = 1;
107 // use 0.9999 quantile https://keisan.casio.com/exec/system/14060745333941
108 if (! is_independent) // do not calculate it for all inliers
109 min_non_random_inliers = (int)(lambda + 2.32*sqrt(lambda * (1 - lambda / points_size))) + 1;
110 return lambda;
111}

281 const int min_sample_size = params->getSampleSize(), prosac_termination_length = std::min((int)(.5*points_size), 100);
282 // inner inlier threshold will be used in LO to obtain inliers

289 weight_fnc = MagsacWeightFunction::create(_gamma_generator, params->getDegreesOfFreedom(), params->getUpperIncompleteOfSigmaQuantile(), params->getC(), params->getMaximumThreshold());
290 else weight_fnc = nullptr;
291
292 switch (params->getError()) {
293 case ErrorMetric::SYMM_REPR_ERR:
294 error = ReprojectionErrorSymmetric::create(points); break;
295 case ErrorMetric::FORW_REPR_ERR:
296 if (params->getEstimator() == EstimationMethod::AFFINE)
297 error = ReprojectionErrorAffine::create(points);
298 else error = ReprojectionErrorForward::create(points);
299 break;
300 case ErrorMetric::SAMPSON_ERR:
301 error = SampsonError::create(points); break;
302 case ErrorMetric::SGD_ERR:

304 case ErrorMetric::RERPOJ:
305 error = ReprojectionErrorPmatrix::create(points); break;
306 case ErrorMetric::POINT_TO_PLANE:
307 error = PlaneModelError::create(points); break;
308 case ErrorMetric::POINT_TO_SPHERE:
309 error = SphereModelError::create(points); break;

313 const double k_mlesac = params->getKmlesac ();
314 switch (params->getScore()) {
315 case ScoreMethod::SCORE_METHOD_RANSAC :
316 quality = RansacQuality::create(points_size, threshold, error); break;
317 case ScoreMethod::SCORE_METHOD_MSAC :
318 quality = MsacQuality::create(points_size, threshold, error, k_mlesac); break;

320 quality = MagsacQuality::create(max_thr, points_size, error, _gamma_generator,

322 params->getUpperIncompleteOfSigmaQuantile()); break;

324 quality = LMedsQuality::create(points_size, threshold, error); break;

331 min_solver = HomographyMinimalSolver4pts::create(points, is_ge_solver);

335 if (params->getFinalPolisher() == COV_POLISHER)
336 _fo_solver = CovarianceHomographySolver::create(norm_points, T1, T2);
337 else _fo_solver = HomographyNonMinimalSolver::create(points);
338 }
339 } else if (params->isFundamental()) {
340 if (K1.empty() || K2.empty()) {
341 degeneracy = FundamentalDegeneracy::create(state++, quality, points, min_sample_size,
342 params->getPlaneAndParallaxIters(), std::max(threshold, 8.) /*sqr homogr thr*/, inner_inlier_thr_sqr, K1, K2);

345 min_solver = FundamentalMinimalSolver7pts::create(points, is_ge_solver);
346 } else min_solver = FundamentalMinimalSolver8pts::create(points);

348 non_min_solver = LarssonOptimizer::create(calib_points, K1, K2, params->getLevMarqItersLO(), true/*F*/);

365 min_solver = EssentialMinimalSolver5pts::create(points, !is_ge_solver, true/*Nister*/);
366 degeneracy = EssentialDegeneracy::create(points, min_sample_size);

371 _fo_solver = LarssonOptimizer::create(calib_points, K1, K2, params->getLevMarqIters(), false/*E*/);

373 degeneracy = makePtr<Degeneracy>();
374 if (min_sample_size == 3) {
375 min_solver = P3PSolver::create(points, calib_points, K1);
376 non_min_solver = DLSPnP::create(points, calib_points, K1);
377 } else {
378 if (is_ge_solver)
379 min_solver = PnPMinimalSolver6Pts::create(points);
380 else min_solver = PnPSVDSolver::create(points);
381 non_min_solver = PnPNonMinimalSolver::create(points);
382 }
383 estimator = PnPEstimator::create(min_solver, non_min_solver);
384 if (!parallel_call && params->getFinalPolisher() != NONE_POLISHER) _fo_solver = non_min_solver;
385 } else if (params->getEstimator() == EstimationMethod::AFFINE) {
386 degeneracy = makePtr<Degeneracy>();
387 min_solver = AffineMinimalSolver::create(points);
388 non_min_solver = AffineNonMinimalSolver::create(points, cv::noArray(), cv::noArray());
389 estimator = AffineEstimator::create(min_solver, non_min_solver);
390 if (!parallel_call && params->getFinalPolisher() != NONE_POLISHER) {
391 if (params->getFinalPolisher() == COV_POLISHER)
392 _fo_solver = CovarianceAffineSolver::create(points);
393 else _fo_solver = non_min_solver;
394 }
395 } else if (params->getEstimator() == EstimationMethod::PLANE || params->getEstimator() == EstimationMethod::SPHERE) {
396 degeneracy = makePtr<Degeneracy>();
397 using ModelConstraintFunction = std::function<bool(const std::vector<double> &/*model_coefficients*/)>;
398 ModelConstraintFunction custom_model_constraints;
399 const ModelConstraintFunction &_custom_constraint = custom_model_constraints;
400 ModelConstraintFunction &constraint_func = custom_model_constraints;
401 if (params->getEstimator() == EstimationMethod::PLANE) {
402 min_solver = PlaneModelMinimalSolver::create(points);
403 non_min_solver = PlaneModelNonMinimalSolver::create(points);
404 _fo_solver = PlaneModelNonMinimalSolver::create(points);
405 } else {
406 min_solver = SphereModelMinimalSolver::create(points);
407 non_min_solver = SphereModelNonMinimalSolver::create(points);
408 _fo_solver = SphereModelNonMinimalSolver::create(points);
409 const double _radius_min = DBL_MIN, _radius_max = DBL_MAX;
410 constraint_func = [_radius_min, _radius_max, _custom_constraint]
411 (const std::vector<double> &model_coeffs) {
412 double radius = model_coeffs[3];
413 return radius >= _radius_min && radius <= _radius_max &&
414 (!_custom_constraint || _custom_constraint(model_coeffs));
415 };
416 }
417 estimator = PointCloudModelEstimator::create(min_solver, non_min_solver, constraint_func);
418 } else CV_Error(cv::Error::StsNotImplemented, "Estimator not implemented!");
419
420 switch (params->getSampler()) {
421 case SamplingMethod::SAMPLING_UNIFORM:
422 sampler = UniformSampler::create(state++, min_sample_size, points_size);
423 break;
424 case SamplingMethod::SAMPLING_PROSAC:
425 if (!parallel_call) // for parallel only one PROSAC sampler
426 sampler = ProsacSampler::create(state++, points_size, min_sample_size, params->getProsacMaxSamples());
427 break;
428 case SamplingMethod::SAMPLING_PROGRESSIVE_NAPSAC:
429 sampler = ProgressiveNapsac::create(state++, points_size, min_sample_size, layers, 20); break;
430 case SamplingMethod::SAMPLING_NAPSAC:
431 sampler = NapsacSampler::create(state++, points_size, min_sample_size, graph); break;
432 default: CV_Error(cv::Error::StsNotImplemented, "Sampler is not implemented!");
433 }

435 const bool is_sprt = params->getVerifier() == VerificationMethod::SPRT_VERIFIER || params->getVerifier() == VerificationMethod::ASPRT;
436 if (is_sprt)
437 verifier = AdaptiveSPRT::create(state++, quality, points_size, params->getScore() == ScoreMethod ::SCORE_METHOD_MAGSAC ? max_thr : threshold,
438 params->getSPRTepsilon(), params->getSPRTdelta(), params->getTimeForModelEstimation(),
439 params->getSPRTavgNumModels(), params->getScore(), k_mlesac, params->getVerifier() == VerificationMethod::ASPRT);
440 else if (params->getVerifier() == VerificationMethod::NULL_VERIFIER)
441 verifier = ModelVerifier::create(quality);
442 else CV_Error(cv::Error::StsNotImplemented, "Verifier is not imeplemented!");
443
444 if (params->getSampler() == SamplingMethod::SAMPLING_PROSAC) {
445 if (parallel_call) {
446 termination = ProsacTerminationCriteria::create(nullptr, error,
447 points_size, min_sample_size, params->getConfidence(), params->getMaxIters(), prosac_termination_length, 0.05, 0.05, threshold,

513int UniversalRANSAC::getIndependentInliers (const Mat &model_, const std::vector<int> &sample,
514 std::vector<int> &inliers, const int num_inliers_) {
515 bool is_F = params->isFundamental();
516 Mat model = model_;
517 int sample_size = 0;
518 if (is_F) sample_size = 7;
519 else if (params->isHomography()) sample_size = 4;
520 else if (params->isEssential()) {
521 is_F = true;
522 // convert E to F
523 model = Mat(Matx33d(K2).inv().t() * Matx33d(model) * Matx33d(K1).inv());
524 sample_size = 5;
525 } else if (params->isPnP() || params->getEstimator() == EstimationMethod::AFFINE) sample_size = 3;
526 else
527 CV_Error(cv::Error::StsNotImplemented, "Method for independent inliers is not implemented for this problem");
528 if (num_inliers_ <= sample_size) return 0; // minimal sample size generates model
529 model.convertTo(model, CV_32F);
530 int num_inliers = num_inliers_, num_pts_near_ep = 0,
531 num_pts_validatin_or_constr = 0, pt1 = 0;
532 const auto * const pts = params->isEssential() ? (float *) image_points.data : (float *) points.data;
533 // scale for thresholds should be used
534 const float ep_thr_sqr = 0.000001f, line_thr = 0.01f, neigh_thr = 4.0f;
535 float sign1=0,a1=0, b1=0, c1=0, a2=0, b2=0, c2=0, ep1_x, ep1_y, ep2_x, ep2_y;
536 const auto * const m = (float *) model.data;
537 Vec3f ep1;
538 bool do_or_test = false, ep1_inf = false, ep2_inf = false;
539 if (is_F) { // compute epipole and sign of the first point for orientation test
540 model *= (1/norm(model));
541 ep1 = Utils::getRightEpipole(model);
542 const Vec3f ep2 = Utils::getLeftEpipole(model);
543 if (fabsf(ep1[2]) < DBL_EPSILON) {
544 ep1_inf = true;
545 } else {
546 ep1_x = ep1[0] / ep1[2];
547 ep1_y = ep1[1] / ep1[2];
548 }
549 if (fabsf(ep2[2]) < DBL_EPSILON) {
550 ep2_inf = true;
551 } else {
552 ep2_x = ep2[0] / ep2[2];
553 ep2_y = ep2[1] / ep2[2];
554 }
555 }
556 const auto * const e1 = ep1.val; // of size 3x1
557
558 // we move sample points to the end, so every inlier will be checked by sample point
559 int num_sample_in_inliers = 0;
560 if (!sample.empty()) {
561 num_sample_in_inliers = 0;
562 int temp_idx = num_inliers;
563 for (int i = 0; i < temp_idx; i++) {
564 const int inl = inliers[i];
565 for (int s : sample) {
566 if (inl == s) {
567 std::swap(inliers[i], inliers[--temp_idx]);
568 i--; // we need to check inlier that we just swapped
569 num_sample_in_inliers++;
570 break;
571 }
572 }
573 }
574 }
575
576 if (is_F) {
577 int MIN_TEST = std::min(15, num_inliers);
578 for (int i = 0; i < MIN_TEST; i++) {
579 pt1 = 4*inliers[i];
580 sign1 = (m[0]*pts[pt1+2]+m[3]*pts[pt1+3]+m[6])*(e1[1]-e1[2]*pts[pt1+1]);
581 int validate = 0;
582 for (int j = 0; j < MIN_TEST; j++) {
583 if (i == j) continue;
584 const int inl_idx = 4*inliers[j];
585 if (sign1*(m[0]*pts[inl_idx+2]+m[3]*pts[inl_idx+3]+m[6])*(e1[1]-e1[2]*pts[inl_idx+1])<0)
586 validate++;
587 }
588 if (validate < MIN_TEST/2) {
589 do_or_test = true; break;
590 }
591 }
592 }
593
594 // verification does not include sample points as they are surely random
595 const int max_verify = num_inliers - num_sample_in_inliers;
596 if (max_verify <= 0)
597 return 0;
598 int num_non_random_inliers = num_inliers - sample_size;
599 auto removeDependentPoints = [&] (bool do_orient_test, bool check_epipoles) {
600 for (int i = 0; i < max_verify; i++) {
601 // checks over inliers if they are dependent to other inliers
602 const int inl_idx = 4*inliers[i];
603 const auto x1 = pts[inl_idx], y1 = pts[inl_idx+1], x2 = pts[inl_idx+2], y2 = pts[inl_idx+3];
604 if (is_F) {
605 // epipolar line on image 2 = l2
606 a2 = m[0] * x1 + m[1] * y1 + m[2];
607 b2 = m[3] * x1 + m[4] * y1 + m[5];
608 c2 = m[6] * x1 + m[7] * y1 + m[8];
609 // epipolar line on image 1 = l1
610 a1 = m[0] * x2 + m[3] * y2 + m[6];
611 b1 = m[1] * x2 + m[4] * y2 + m[7];
612 c1 = m[2] * x2 + m[5] * y2 + m[8];
613 if ((!ep1_inf && fabsf(x1-ep1_x)+fabsf(y1-ep1_y) < neigh_thr) ||
614 (!ep2_inf && fabsf(x2-ep2_x)+fabsf(y2-ep2_y) < neigh_thr)) {
615 num_non_random_inliers--;
616 num_pts_near_ep++;
617 continue; // is dependent, continue to the next point
618 } else if (check_epipoles) {
619 if (a2 * a2 + b2 * b2 + c2 * c2 < ep_thr_sqr ||
620 a1 * a1 + b1 * b1 + c1 * c1 < ep_thr_sqr) {
621 num_non_random_inliers--;
622 num_pts_near_ep++;
623 continue; // is dependent, continue to the next point
624 }
625 }
626 else if (do_orient_test && pt1 != inl_idx && sign1*(m[0]*x2+m[3]*y2+m[6])*(e1[1]-e1[2]*y1)<0) {
627 num_non_random_inliers--;
628 num_pts_validatin_or_constr++;
629 continue;
630 }
631 const auto mag2 = 1 / sqrt(a2 * a2 + b2 * b2), mag1 = 1/sqrt(a1 * a1 + b1 * b1);
632 a2 *= mag2; b2 *= mag2; c2 *= mag2;
633 a1 *= mag1; b1 *= mag1; c1 *= mag1;
634 }
635
636 for (int j = i+1; j < num_inliers; j++) {// verify through all including sample points
637 const int inl_idx_j = 4*inliers[j];
638 const auto X1 = pts[inl_idx_j], Y1 = pts[inl_idx_j+1], X2 = pts[inl_idx_j+2], Y2 = pts[inl_idx_j+3];
639 // use L1 norm instead of L2 for faster evaluation
640 if (fabsf(X1-x1) + fabsf(Y1 - y1) < neigh_thr || fabsf(X2-x2) + fabsf(Y2 - y2) < neigh_thr) {
641 num_non_random_inliers--;
642 // num_pts_bad_conditioning++;
643 break; // is dependent stop verification
644 } else if (is_F) {
645 if (fabsf(a2 * X2 + b2 * Y2 + c2) < line_thr && //|| // xj'^T F xi
646 fabsf(a1 * X1 + b1 * Y1 + c1) < line_thr) { // xj^T F^T xi'
647 num_non_random_inliers--;
648 break; // is dependent stop verification
649 }
650 }
651 }
652 }
653 };
654 if (params->isPnP()) {
655 for (int i = 0; i < max_verify; i++) {
656 const int inl_idx = 5*inliers[i];
657 const auto x = pts[inl_idx], y = pts[inl_idx+1], X = pts[inl_idx+2], Y = pts[inl_idx+3], Z = pts[inl_idx+4];
658 for (int j = i+1; j < num_inliers; j++) {
659 const int inl_idx_j = 5*inliers[j];
660 if (fabsf(x-pts[inl_idx_j ]) + fabsf(y-pts[inl_idx_j+1]) < neigh_thr ||
661 fabsf(X-pts[inl_idx_j+2]) + fabsf(Y-pts[inl_idx_j+3]) + fabsf(Z-pts[inl_idx_j+4]) < neigh_thr) {
662 num_non_random_inliers--;
663 break;
664 }
665 }
666 }
667 } else {
668 removeDependentPoints(do_or_test, !ep1_inf && !ep2_inf);
669 if (is_F) {
670 const bool is_pts_vald_constr_normal = (double)num_pts_validatin_or_constr / num_inliers < 0.6;
671 const bool is_pts_near_ep_normal = (double)num_pts_near_ep / num_inliers < 0.6;
672 if (!is_pts_near_ep_normal || !is_pts_vald_constr_normal) {
673 num_non_random_inliers = num_inliers-sample_size;
674 num_pts_near_ep = 0; num_pts_validatin_or_constr = 0;
675 removeDependentPoints(is_pts_vald_constr_normal, is_pts_near_ep_normal);
676 }
677 }
678 }
679 return num_non_random_inliers;
680}

691bool UniversalRANSAC::run(Ptr<RansacOutput> &ransac_output) {
692 if (points_size < params->getSampleSize())
693 return false;
694 const bool LO = params->getLO() != LocalOptimMethod::LOCAL_OPTIM_NULL,
695 IS_FUNDAMENTAL = params->isFundamental(), IS_NON_RAND_TEST = params->isNonRandomnessTest();
696 const int MAX_MODELS_ADAPT = 21, MAX_ITERS_ADAPT = MAX_MODELS_ADAPT/*assume at least 1 model from 1 sample*/,
697 sample_size = params->getSampleSize();
698 const double IOU_SIMILARITY_THR = 0.80;
699 std::vector<int> non_degen_sample, best_sample;
700
701 double lambda_non_random_all_inliers = -1;
702 int final_iters, num_total_tested_models = 0;
703
704 // non-random
705 const int MAX_TEST_MODELS_NONRAND = IS_NON_RAND_TEST ? MAX_MODELS_ADAPT : 0;
706 std::vector<Mat> models_for_random_test; models_for_random_test.reserve(MAX_TEST_MODELS_NONRAND);
707 std::vector<std::vector<int>> samples_for_random_test; samples_for_random_test.reserve(MAX_TEST_MODELS_NONRAND);
708
709 bool last_model_from_LO = false;
710 Mat best_model, best_model_not_from_LO, K1_approx, K2_approx;
711 Score best_score, best_score_model_not_from_LO;
712 std::vector<bool> best_inliers_mask(points_size);
713 if (! parallel) {
714 // adaptive sprt test
715 double IoU = 0, mean_num_est_models = 0;
716 bool adapt = IS_NON_RAND_TEST || params->getVerifier() == VerificationMethod ::ASPRT, was_LO_run = false;
717 int min_non_random_inliers = 30, iters = 0, num_estimations = 0, max_iters = params->getMaxIters();
718 Mat non_degenerate_model, lo_model;
719 Score current_score, non_degenerate_model_score, best_score_sample;
720 std::vector<bool> model_inliers_mask (points_size);
721 std::vector<Mat> models(_estimator->getMaxNumSolutions());
722 std::vector<int> sample(_estimator->getMinimalSampleSize()), supports;
723 supports.reserve(3*MAX_MODELS_ADAPT); // store model supports during adaption
724
725 auto update_best = [&] (const Mat &new_model, const Score &new_score, bool from_lo=false) {
726 _quality->getInliers(new_model, model_inliers_mask);
727 IoU = Utils::intersectionOverUnion(best_inliers_mask, model_inliers_mask);
728 best_inliers_mask = model_inliers_mask;
729
730 if (!best_model.empty() && (int)models_for_random_test.size() < MAX_TEST_MODELS_NONRAND && IoU < IOU_SIMILARITY_THR && !from_lo) { // use IoU to not save similar models
731 // save old best model for non-randomness test if necessary
732 models_for_random_test.emplace_back(best_model.clone());
733 samples_for_random_test.emplace_back(best_sample);
734 }
735
736 // update score, model, inliers and max iterations
737 best_score = new_score;
738 new_model.copyTo(best_model);
739
740 if (!from_lo) {
741 best_sample = sample;
742 if (IS_FUNDAMENTAL) { // avoid degeneracy after LO run
743 // save last model not from LO
744 best_model.copyTo(best_model_not_from_LO);
745 best_score_model_not_from_LO = best_score;
746 }
747 }
748
749 _model_verifier->update(best_score, iters);
750 max_iters = _termination->update(best_model, best_score.inlier_number);
751 // max_iters = std::max(max_iters, std::min(10, params->getMaxIters()));
752 if (!adapt) // update quality and verifier to save evaluation time of a model
753 _quality->setBestScore(best_score.score);
754 last_model_from_LO = from_lo;
755 };
756
757 auto run_lo = [&] (const Mat &_model, const Score &_score, bool force_lo) {
758 was_LO_run = true;
759 _local_optimization->setCurrentRANSACiter(force_lo ? iters : -1);
760 Score lo_score;
761 if (_local_optimization->refineModel(_model, _score, lo_model, lo_score) && lo_score.isBetter(best_score))
762 update_best(lo_model, lo_score, true);
763 };
764
765 for (; iters < max_iters; iters++) {
766 _sampler->generateSample(sample);
767 int number_of_models;
768 if (adapt) {
769 number_of_models = _estimator->estimateModels(sample, models);
770 mean_num_est_models += number_of_models;
771 num_estimations++;
772 } else {
773 number_of_models = _estimator->estimateModels(sample, models);
774 }
775 for (int i = 0; i < number_of_models; i++) {
776 num_total_tested_models++;
777
778 if (adapt) {
779 current_score = _quality->getScore(models[i]);
780 supports.emplace_back(current_score.inlier_number);
781 if (IS_NON_RAND_TEST && best_score_sample.isBetter(current_score)) {
782 models_for_random_test.emplace_back(models[i].clone());
783 samples_for_random_test.emplace_back(sample);
784 }
785 } else {
786 if (! _model_verifier->isModelGood(models[i], current_score))
787 continue;
788 }
789 if (current_score.isBetter(best_score_sample)) {
790 if (_degeneracy->recoverIfDegenerate(sample, models[i], current_score,
791 non_degenerate_model, non_degenerate_model_score)) {
792 // check if best non degenerate model is better than so far the best model
793 if (non_degenerate_model_score.isBetter(best_score)) {
794 update_best(non_degenerate_model, non_degenerate_model_score);
795 best_score_sample = current_score.isBetter(best_score) ? best_score : current_score;
796 } else continue;
797 } else {
798 best_score_sample = current_score;
799 update_best(models[i], current_score);
800 }
801
802 if (LO && ((iters < max_iters && best_score.inlier_number > min_non_random_inliers && IoU < IOU_SIMILARITY_THR)))
803 run_lo(best_model, best_score, false);
804
805 } // end of if so far the best score
806 } // end loop of number of models
807 if (adapt && iters >= MAX_ITERS_ADAPT && num_total_tested_models >= MAX_MODELS_ADAPT) {
808 adapt = false;
809 lambda_non_random_all_inliers = getLambda(supports, 2.32, points_size, sample_size, false, min_non_random_inliers);
810 _model_verifier->updateSPRT(params->getTimeForModelEstimation(), 1.0, mean_num_est_models/num_estimations, lambda_non_random_all_inliers/points_size,(double)std::max(min_non_random_inliers, best_score.inlier_number)/points_size, best_score);
811 }
812 } // end main while loop
813 final_iters = iters;
814 if (! was_LO_run && !best_model.empty() && LO)
815 run_lo(best_model, best_score, true);
816 } else { // parallel VSAC
817 const int MAX_THREADS = getNumThreads(), growth_max_samples = params->getProsacMaxSamples();
818 const bool is_prosac = params->getSampler() == SamplingMethod::SAMPLING_PROSAC;
819 std::atomic_bool success(false);
820 std::atomic_int num_hypothesis_tested(0), thread_cnt(0), max_number_inliers(0), subset_size, termination_length;
821 std::atomic<float> best_score_all(std::numeric_limits<float>::max());
822 std::vector<Score> best_scores(MAX_THREADS), best_scores_not_LO;
823 std::vector<Mat> best_models(MAX_THREADS), best_models_not_LO, K1_apx, K2_apx;
824 std::vector<int> num_tested_models_threads(MAX_THREADS), growth_function, non_random_inliers;
825 std::vector<std::vector<Mat>> tested_models_threads(MAX_THREADS);
826 std::vector<std::vector<std::vector<int>>> tested_samples_threads(MAX_THREADS);
827 std::vector<std::vector<int>> best_samples_threads(MAX_THREADS);
828 std::vector<bool> last_model_from_LO_vec;
829 std::vector<double> lambda_non_random_all_inliers_vec(MAX_THREADS);
830 if (IS_FUNDAMENTAL) {
831 last_model_from_LO_vec = std::vector<bool>(MAX_THREADS);
832 best_models_not_LO = std::vector<Mat>(MAX_THREADS);
833 best_scores_not_LO = std::vector<Score>(MAX_THREADS);
834 K1_apx = std::vector<Mat>(MAX_THREADS);
835 K2_apx = std::vector<Mat>(MAX_THREADS);
836 }
837 if (is_prosac) {
838 growth_function = _sampler.dynamicCast<ProsacSampler>()->getGrowthFunction();
839 subset_size = 2*sample_size; // n, size of the current sampling pool
840 termination_length = points_size;
841 }
843 parallel_for_(Range(0, MAX_THREADS), [&](const Range & /*range*/) {
844 if (!success) { // cover all if not success to avoid thread creating new variables
845 const int thread_rng_id = thread_cnt++;
846 bool adapt = params->getVerifier() == VerificationMethod ::ASPRT || IS_NON_RAND_TEST;
847 int thread_state = _state + thread_rng_id, min_non_random_inliers = 0, num_tested_models = 0,
848 num_estimations = 0, mean_num_est_models = 0, iters, max_iters = params->getMaxIters();
849 double IoU = 0, lambda_non_random_all_inliers_thread = -1;
850 std::vector<Mat> tested_models_thread; tested_models_thread.reserve(MAX_TEST_MODELS_NONRAND);
851 std::vector<std::vector<int>> tested_samples_thread; tested_samples_thread.reserve(MAX_TEST_MODELS_NONRAND);
852 Ptr<UniformRandomGenerator> random_gen;
853 if (is_prosac) random_gen = UniformRandomGenerator::create(thread_state);
854 Ptr<Error> error;
855 Ptr<Estimator> estimator;
856 Ptr<Degeneracy> degeneracy;
857 Ptr<Quality> quality;
858 Ptr<ModelVerifier> model_verifier;
859 Ptr<Sampler> sampler;
860 Ptr<RandomGenerator> lo_sampler;
861 Ptr<Termination> termination;
862 Ptr<LocalOptimization> local_optimization;
863 Ptr<MinimalSolver> min_solver;
864 Ptr<NonMinimalSolver> non_min_solver;
865 Ptr<WeightFunction> weight_fnc;
866 initialize (thread_state, min_solver, non_min_solver, error, estimator, degeneracy, quality,
867 model_verifier, local_optimization, termination, sampler, lo_sampler, weight_fnc, true);
868 bool is_last_from_LO_thread = false;
869 Mat best_model_thread, non_degenerate_model, lo_model, best_not_LO_thread;
870 Score best_score_thread, current_score, non_denegenerate_model_score, lo_score, best_score_all_threads, best_not_LO_score_thread;
871 std::vector<int> sample(estimator->getMinimalSampleSize()), best_sample_thread, supports;
872 supports.reserve(3*MAX_MODELS_ADAPT); // store model supports
873 std::vector<bool> best_inliers_mask_local(points_size, false), model_inliers_mask(points_size, false);
874 std::vector<Mat> models(estimator->getMaxNumSolutions());
875 auto update_best = [&] (const Score &new_score, const Mat &new_model, bool from_LO=false) {
876 // update best score of all threads
877 if (max_number_inliers < new_score.inlier_number) max_number_inliers = new_score.inlier_number;
878 if (best_score_all > new_score.score)
879 best_score_all = new_score.score;
880 best_score_all_threads = Score(max_number_inliers, best_score_all);
881 //
882 quality->getInliers(new_model, model_inliers_mask);
883 IoU = Utils::intersectionOverUnion(best_inliers_mask_local, model_inliers_mask);

886 tested_samples_thread.emplace_back(best_sample_thread);

888 if (!adapt) { // update quality and verifier
889 quality->setBestScore(best_score_all);
890 model_verifier->update(best_score_all_threads, iters);
891 }

893 best_score_thread = new_score;
894 best_sample_thread = sample;
895 best_inliers_mask_local = model_inliers_mask;
896 // remember best model
897 new_model.copyTo(best_model_thread);
898
899 // update upper bound of iterations
900 if (is_prosac) {
901 int new_termination_length;
902 max_iters = termination.dynamicCast<ProsacTerminationCriteria>()->
903 updateTerminationLength(best_model_thread, best_score_thread.inlier_number, new_termination_length);
904 // update termination length
905 if (new_termination_length < termination_length)
906 termination_length = new_termination_length;
907 } else max_iters = termination->update(best_model_thread, max_number_inliers);
908 if (IS_FUNDAMENTAL) {
909 is_last_from_LO_thread = from_LO;
910 if (!from_LO) {
911 best_model_thread.copyTo(best_not_LO_thread);
912 best_not_LO_score_thread = best_score_thread;
913 }
914 }
915 };

918 was_LO_run = true;
919 local_optimization->setCurrentRANSACiter(current_ransac_iters);
920 if (local_optimization->refineModel(best_model_thread, best_score_thread, lo_model,
921 lo_score) && lo_score.isBetter(best_score_thread))
922 update_best(lo_score, lo_model, true);
923 };
924 for (iters = 0; iters < max_iters && !success; iters++) {
925 success = num_hypothesis_tested++ > max_iters;
926 if (iters % 10 && !adapt) {
927 // Synchronize threads. just to speed verification of model.
928 quality->setBestScore(std::min(best_score_thread.score, (float)best_score_all));
929 model_verifier->update(best_score_thread.inlier_number > max_number_inliers ? best_score_thread : best_score_all_threads, iters);
930 }
931
932 if (is_prosac) {
933 if (num_hypothesis_tested > growth_max_samples) {
934 // if PROSAC has not converged to solution then do uniform sampling.
935 random_gen->generateUniqueRandomSet(sample, sample_size, points_size);
936 } else {
937 if (num_hypothesis_tested >= growth_function[subset_size-1] && subset_size < termination_length-MAX_THREADS) {
938 subset_size++;
939 if (subset_size >= points_size) subset_size = points_size-1;
940 }
941 if (growth_function[subset_size-1] < num_hypothesis_tested) {
942 // The sample contains m-1 points selected from U_(n-1) at random and u_n
943 random_gen->generateUniqueRandomSet(sample, sample_size-1, subset_size-1);
944 sample[sample_size-1] = subset_size-1;
945 } else
946 // Select m points from U_n at random.
947 random_gen->generateUniqueRandomSet(sample, sample_size, subset_size);
948 }
949 } else sampler->generateSample(sample); // use local sampler
950
951 const int number_of_models = estimator->estimateModels(sample, models);
952 if (adapt) {
953 num_estimations++; mean_num_est_models += number_of_models;
954 }
955 for (int i = 0; i < number_of_models; i++) {
956 num_tested_models++;
957 if (adapt) {
958 current_score = quality->getScore(models[i]);
959 supports.emplace_back(current_score.inlier_number);
960 } else if (! model_verifier->isModelGood(models[i], current_score))
961 continue;
962
963 if (current_score.isBetter(best_score_all_threads)) {
964 if (degeneracy->recoverIfDegenerate(sample, models[i], current_score,
965 non_degenerate_model, non_denegenerate_model_score)) {
966 // check if best non degenerate model is better than so far the best model
967 if (non_denegenerate_model_score.isBetter(best_score_thread))
968 update_best(non_denegenerate_model_score, non_degenerate_model);
969 else continue;
970 } else update_best(current_score, models[i]);
971 if (LO && num_hypothesis_tested < max_iters && IoU < IOU_SIMILARITY_THR &&
972 best_score_thread.inlier_number > min_non_random_inliers)
973 runLO(iters);
974 } // end of if so far the best score
975 else if ((int)tested_models_thread.size() < MAX_TEST_MODELS_NONRAND) {
976 tested_models_thread.emplace_back(models[i].clone());
977 tested_samples_thread.emplace_back(sample);
978 }
979 if (num_hypothesis_tested > max_iters) {
980 success = true; break;
981 }
982 } // end loop of number of models
983 if (adapt && iters >= MAX_ITERS_ADAPT && num_tested_models >= MAX_MODELS_ADAPT) {
984 adapt = false;
985 lambda_non_random_all_inliers_thread = getLambda(supports, 2.32, points_size, sample_size, false, min_non_random_inliers);
986 model_verifier->updateSPRT(params->getTimeForModelEstimation(), 1, (double)mean_num_est_models/num_estimations, lambda_non_random_all_inliers_thread/points_size,
987 (double)std::max(min_non_random_inliers, best_score.inlier_number)/points_size, best_score_all_threads);
988 }
989 if (!adapt && LO && num_hypothesis_tested < max_iters && !was_LO_run && !best_model_thread.empty() &&
990 best_score_thread.inlier_number > min_non_random_inliers)
991 runLO(iters);
992 } // end of loop over iters
993 if (! was_LO_run && !best_model_thread.empty() && LO)
994 runLO(-1 /*use full iterations of LO*/);
995 best_model_thread.copyTo(best_models[thread_rng_id]);
996 best_scores[thread_rng_id] = best_score_thread;
997 num_tested_models_threads[thread_rng_id] = num_tested_models;
998 tested_models_threads[thread_rng_id] = tested_models_thread;
999 tested_samples_threads[thread_rng_id] = tested_samples_thread;
1000 best_samples_threads[thread_rng_id] = best_sample_thread;
1001 if (IS_FUNDAMENTAL) {
1002 best_scores_not_LO[thread_rng_id] = best_not_LO_score_thread;
1003 best_not_LO_thread.copyTo(best_models_not_LO[thread_rng_id]);
1004 last_model_from_LO_vec[thread_rng_id] = is_last_from_LO_thread;
1005 }
1006 lambda_non_random_all_inliers_vec[thread_rng_id] = lambda_non_random_all_inliers_thread;
1007 }}); // end parallel
1009 // find best model from all threads' models
1010 best_score = best_scores[0];
1011 int best_thread_idx = 0;
1012 for (int i = 1; i < MAX_THREADS; i++)
1013 if (best_scores[i].isBetter(best_score)) {
1014 best_score = best_scores[i];
1015 best_thread_idx = i;
1016 }
1017 best_model = best_models[best_thread_idx];
1018 if (IS_FUNDAMENTAL) {
1019 last_model_from_LO = last_model_from_LO_vec[best_thread_idx];
1020 K1_approx = K1_apx[best_thread_idx];
1021 K2_approx = K2_apx[best_thread_idx];

1024 best_sample = best_samples_threads[best_thread_idx];

1026 double avg_lambda = 0;
1027 for (int i = 0; i < MAX_THREADS; i++) {
1028 if (IS_FUNDAMENTAL && best_scores_not_LO[i].isBetter(best_score_model_not_from_LO)) {
1029 best_score_model_not_from_LO = best_scores_not_LO[i];
1030 best_models_not_LO[i].copyTo(best_model_not_from_LO);
1031 }
1032 if (IS_NON_RAND_TEST && lambda_non_random_all_inliers_vec[i] > 0) {
1033 num_lambdas ++;
1034 avg_lambda += lambda_non_random_all_inliers_vec[i];
1035 }
1036 num_total_tested_models += num_tested_models_threads[i];
1037 if ((int)models_for_random_test.size() < MAX_TEST_MODELS_NONRAND) {
1038 for (int m = 0; m < (int)tested_models_threads[i].size(); m++) {
1039 models_for_random_test.emplace_back(tested_models_threads[i][m].clone());
1040 samples_for_random_test.emplace_back(tested_samples_threads[i][m]);
1041 if ((int)models_for_random_test.size() == MAX_TEST_MODELS_NONRAND)
1042 break;
1043 }
1044 }
1045 }
1046 if (IS_NON_RAND_TEST && num_lambdas > 0 && avg_lambda > 0)
1047 lambda_non_random_all_inliers = avg_lambda / num_lambdas;
1048 }
1049 if (best_model.empty()) {
1050 ransac_output = RansacOutput::create(best_model, std::vector<bool>(), best_score.inlier_number, final_iters, ModelConfidence::RANDOM, std::vector<float>());
1051 return false;
1052 }
1053 if (last_model_from_LO && IS_FUNDAMENTAL && K1.empty() && K2.empty()) {
1054 Score new_score; Mat new_model;
1055 const double INL_THR = 0.80;
1056 if (parallel)
1057 _quality->getInliers(best_model, best_inliers_mask);
1058 // run additional degeneracy check for F:
1059 if (_degeneracy.dynamicCast<FundamentalDegeneracy>()->verifyFundamental(best_model, best_score, best_inliers_mask, new_model, new_score)) {
1060 // so-far-the-best F is degenerate
1061 // Update best F using non-degenerate F or the one which is not from LO
1062 if (new_score.isBetter(best_score_model_not_from_LO) && new_score.inlier_number > INL_THR*best_score.inlier_number) {
1063 best_score = new_score;
1064 new_model.copyTo(best_model);
1065 } else if (best_score_model_not_from_LO.inlier_number > INL_THR*best_score.inlier_number) {
1066 best_score = best_score_model_not_from_LO;
1067 best_model_not_from_LO.copyTo(best_model);
1068 }
1069 } else { // so-far-the-best F is not degenerate
1070 if (new_score.isBetter(best_score)) {
1071 // if new model is better then update
1072 best_score = new_score;
1073 new_model.copyTo(best_model);
1074 }
1075 }
1076 }
1077 if (params->getFinalPolisher() != PolishingMethod::NONE_POLISHER) {
1078 Mat polished_model;
1079 Score polisher_score;
1080 if (polisher->polishSoFarTheBestModel(best_model, best_score, // polish final model
1081 polished_model, polisher_score) && polisher_score.isBetter(best_score)) {
1082 best_score = polisher_score;
1083 polished_model.copyTo(best_model);
1084 }
1085 }
1086
1088 std::vector<bool> inliers_mask; std::vector<float> residuals;
1089 if (params->isMaskRequired()) {
1090 inliers_mask = std::vector<bool>(points_size);
1091 residuals = _error->getErrors(best_model);
1092 _quality->getInliers(residuals, inliers_mask, threshold);
1093 }
1094
1095 ModelConfidence model_conf = ModelConfidence::UNKNOWN;
1096 if (IS_NON_RAND_TEST) {
1097 std::vector<int> temp_inliers(points_size);
1098 const int non_random_inls_best_model = getIndependentInliers(best_model, best_sample, temp_inliers,
1099 _quality->getInliers(best_model, temp_inliers));
1100 // quick test on lambda from all inliers (= upper bound of independent inliers)
1101 // if model with independent inliers is not random for Poisson with all inliers then it is not random using independent inliers too
1102 if (std::pow(Utils::getPoissonCDF(lambda_non_random_all_inliers, non_random_inls_best_model), num_total_tested_models) < 0.9999) {
1103 std::vector<int> inliers_list(models_for_random_test.size());
1104 for (int m = 0; m < (int)models_for_random_test.size(); m++)
1105 inliers_list[m] = getIndependentInliers(models_for_random_test[m], samples_for_random_test[m],
1106 temp_inliers, _quality->getInliers(models_for_random_test[m], temp_inliers));
1107 int min_non_rand_inliers;
1108 const double lambda = getLambda(inliers_list, 1.644, points_size, sample_size, true, min_non_rand_inliers);
1109 const double cdf_lambda = Utils::getPoissonCDF(lambda, non_random_inls_best_model), cdf_N = std::pow(cdf_lambda, num_total_tested_models);
1110 model_conf = cdf_N < 0.9999 ? ModelConfidence ::RANDOM : ModelConfidence ::NON_RANDOM;
1111 } else model_conf = ModelConfidence ::NON_RANDOM;
1112 }
1113 ransac_output = RansacOutput::create(best_model, inliers_mask, best_score.inlier_number, final_iters, model_conf, residuals);
1114 return true;
1115}

1117/*
1118 * pts1, pts2 are matrices either N x a, N x b or a x N or b x N, where N > a and N > b
1119 * pts1 are image points, if pnp pts2 are object points otherwise - image points as well.
1120 * output is matrix of size N x (a + b)
1121 * return points_size = N
1122 */
1123int mergePoints (InputArray pts1_, InputArray pts2_, Mat &pts, bool ispnp) {
1124 Mat pts1 = pts1_.getMat(), pts2 = pts2_.getMat();
1125 auto convertPoints = [] (Mat &points, int pt_dim) {
1126 points.convertTo(points, CV_32F); // convert points to have float precision
1127 if (points.channels() > 1)
1128 points = points.reshape(1, (int)points.total()); // convert point to have 1 channel
1129 if (points.rows < points.cols)
1130 transpose(points, points); // transpose so points will be in rows
1131 CV_CheckGE(points.cols, pt_dim, "Invalid dimension of point");
1132 if (points.cols != pt_dim) // in case when image points are 3D convert them to 2D
1133 points = points.colRange(0, pt_dim);
1134 };
1135
1136 convertPoints(pts1, 2); // pts1 are always image points
1137 convertPoints(pts2, ispnp ? 3 : 2); // for PnP points are 3D
1138
1139 // points are of size [Nx2 Nx2] = Nx4 for H, F, E
1140 // points are of size [Nx2 Nx3] = Nx5 for PnP
1141 hconcat(pts1, pts2, pts);
1142 return pts.rows;
1143}
