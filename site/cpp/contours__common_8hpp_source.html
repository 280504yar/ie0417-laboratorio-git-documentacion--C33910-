34private:

64template <typename T>
65class Tree
66{
67public:
68 Tree() {}
69 Tree(const Tree&) = delete;
70 Tree(Tree&&) = delete;
71 Tree& operator=(const Tree&) = delete;
72 Tree& operator=(Tree&&) = delete;
73 ~Tree() = default;
74private:
75 std::vector<TreeNode<T>> nodes;
76
77public:
78 TreeNode<T>& newElem(T && body_)
79 {
80 const size_t idx = nodes.size();
81 CV_DbgAssert(idx < (size_t)std::numeric_limits<int>::max());
82 nodes.emplace_back(std::move(TreeNode<T>((int)idx, std::move(body_))));
83 return nodes[idx];
84 }
85 TreeNode<T>& elem(int idx)
86 {
87 CV_DbgAssert(idx >= 0 && (size_t)idx < nodes.size());
88 return nodes[(size_t)idx];
89 }
90 const TreeNode<T>& elem(int idx) const
91 {
92 CV_DbgAssert(idx >= 0 && (size_t)idx < nodes.size());
93 return nodes[(size_t)idx];
94 }
95 int lastSibling(int e) const
96 {
97 if (e != -1)
98 {
99 while (true)
100 {
101 const TreeNode<T>& cur_elem = elem(e);
102 if (cur_elem.next == -1)
103 break;
104 e = cur_elem.next;
105 }
106 }
107 return e;
108 }
109 void addSiblingAfter(int prev, int idx)
110 {
111 TreeNode<T>& prev_item = nodes[prev];
112 TreeNode<T>& child = nodes[idx];
113 child.parent = prev_item.parent;
114 if (prev_item.next != -1)
115 {
116 ((TreeNode<T>&)nodes[prev_item.next]).prev = idx;
117 child.next = prev_item.next;
118 }
119 child.prev = prev;
120 prev_item.next = idx;
121 }
122 void addChild(int parent_idx, int child_idx)
123 {
124 TreeNode<T>& parent = nodes[parent_idx];
125 TreeNode<T>& child = nodes[child_idx];
126 if (parent.first_child != -1)
127 {
128 TreeNode<T>& fchild_ = nodes[parent.first_child];
129 fchild_.prev = child_idx;
130 child.next = parent.first_child;
131 }
132 parent.first_child = child_idx;
133 child.parent = parent_idx;
134 child.prev = -1;
135 }
136 bool isEmpty() const
137 {
138 return nodes.size() == 0;
139 }
140 size_t size() const
141 {
142 return nodes.size();
143 }
144};

146template <typename T>
147class TreeIterator
148{
149public:
150 TreeIterator(Tree<T>& tree_) : tree(tree_)
151 {
152 CV_Assert(!tree.isEmpty());
153 levels.push(0);
154 }
155 bool isDone() const
156 {
157 return levels.empty();
158 }
159 const TreeNode<T>& getNext_s()
160 {
161 int idx = levels.top();
162 levels.pop();
163 const TreeNode<T>& res = tree.elem(idx);
164 int cur = tree.lastSibling(res.first_child);
165 while (cur != -1)
166 {
167 levels.push(cur);
168 cur = tree.elem(cur).prev;
169 }
170 return res;
171 }
172
173private:
174 Tree<T>& tree;
175 std::stack<int> levels;
176};
