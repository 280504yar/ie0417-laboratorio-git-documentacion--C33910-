94 bool getMemoryShapes(const std::vector<MatShape> &inputs,
95 const int requiredOutputs,
96 std::vector<MatShape> &outputs,
97 std::vector<MatShape> &internals) const CV_OVERRIDE
98 {
99 CV_Assert(inputs.size() > 0);
100 for (size_t i = 1; i < inputs.size(); i++)
101 {
102 CV_Assert(inputs[i] == inputs[0]);
103 }
104
105 MatShape outputShapeVec;
106
107 int numAxes = (int)inputs[0].size();
108 /*
109 Ticket: https://github.com/opencv/opencv/issues/26197
110 [TODO] this is not quite correct,
111 in ONNX Flatten valid range is [0, numAxes],
112 not [0, numAxes-1] which normalize_axis() produces.
113 But if we fix it, flatten_const.onnx from opencv_extra
114 is not processed correctly.
115 libprotobuf-c reads it correctly,
116 but the current version of libprotobuf does not
117 */
118 int startAxis = normalize_axis(_startAxis, numAxes);
119 int endAxis = normalize_axis(_endAxis, numAxes);
120
121 CV_Assert(startAxis >= 0 && startAxis <= numAxes);
122
123 if (_onnxMode) {
124 int onnxAxis = _startAxis;
125 if (onnxAxis < 0) onnxAxis += numAxes;
126 onnxAxis = std::max(0, std::min(onnxAxis, numAxes));
127 size_t outer = 1, inner = 1;
128 int i = 0;
129 for (; i < onnxAxis; i++)
130 outer *= inputs[0][i];
131 for (; i < numAxes; i++)
132 inner *= inputs[0][i];
133
134 CV_Assert_N(inner <= (size_t)INT_MAX, outer < (size_t)INT_MAX);
135 outputShapeVec.push_back((int)outer);
136 outputShapeVec.push_back((int)inner);
137 }
138 else {
139 CV_Assert(endAxis >= startAxis && endAxis <= numAxes);
140
141 size_t flattenedDimensionSize = total(inputs[0], startAxis, endAxis + 1);
142
143 for (int i = 0; i < startAxis; i++)
144 {
145 outputShapeVec.push_back(inputs[0][i]);
146 }
147 outputShapeVec.push_back(flattenedDimensionSize);
148 for (size_t i = endAxis + 1; i < numAxes; i++)
149 {
150 outputShapeVec.push_back(inputs[0][i]);
151 }
152 }
153
154 outputs.resize(inputs.size(), outputShapeVec);
155
156 return true;
157 }

184 std::vector<UMat> inpvec;
185 std::vector<UMat> outputs;
186
187 inputs_arr.getUMatVector(inpvec);
188 outputs_arr.getUMatVector(outputs);
189
190 std::vector<UMat*> inputs(inpvec.size());

198 inputs[i]->reshape(1, (int)outShape.size(), &outShape[0]).copyTo(output);

250#ifdef HAVE_DNN_NGRAPH
251 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> >& inputs,
252 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
253 {
254 auto& ieInpNode = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
255 std::vector<size_t> dims = ieInpNode.get_shape();
256
257 int numAxes = dims.size();
258 int startAxis = _startAxis;
259 if (startAxis < 0) startAxis += numAxes;
260 startAxis = std::max(0, std::min(startAxis, numAxes));
261 int endAxis = _endAxis;
262 if (endAxis < 0) endAxis += numAxes;
263 endAxis = std::max(0, std::min(endAxis, numAxes - 1));
264
265 CV_Assert(startAxis >= 0);
266 int64_t flattenedDimensionSize = std::accumulate(dims.begin() + startAxis,
267 dims.begin() + endAxis + 1, 1, std::multiplies<size_t>());
268
269 std::vector<int64_t> outputShapeVec(dims.begin(), dims.begin() + startAxis);
270 outputShapeVec.push_back(flattenedDimensionSize);
271 outputShapeVec.insert(outputShapeVec.end(), dims.begin() + endAxis + 1, dims.end());
272
273 auto shape = std::make_shared<ov::op::v0::Constant>(ov::element::i64,
274 ov::Shape({outputShapeVec.size()}), outputShapeVec.data());
275 auto reshape = std::make_shared<ov::op::v1::Reshape>(ieInpNode, shape, true);
276 return Ptr<BackendNode>(new InfEngineNgraphNode(reshape));
277 }
278#endif // HAVE_DNN_NGRAPH
