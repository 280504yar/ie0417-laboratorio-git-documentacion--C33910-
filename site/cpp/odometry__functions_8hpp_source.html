13enum class OdometryTransformType
14{
15 // rotation, translation, rotation+translation
16 ROTATION = 1, TRANSLATION = 2, RIGID_TRANSFORMATION = 4

19static inline int getTransformDim(OdometryTransformType transformType)
20{
21 switch(transformType)
22 {
23 case OdometryTransformType::RIGID_TRANSFORMATION:
24 return 6;
25 case OdometryTransformType::ROTATION:
26 case OdometryTransformType::TRANSLATION:
27 return 3;
28 default:
29 CV_Error(Error::StsBadArg, "Incorrect transformation type");
30 }
31}

34static inline
35Vec6d calcRgbdEquationCoeffs(double dIdx, double dIdy, const Point3d& p3d, double fx, double fy)
36{
37 double invz = 1. / p3d.z,
38 v0 = dIdx * fx * invz,
39 v1 = dIdy * fy * invz,
40 v2 = -(v0 * p3d.x + v1 * p3d.y) * invz;
41 Point3d v(v0, v1, v2);
42 Point3d pxv = p3d.cross(v);
43
44 return Vec6d(pxv.x, pxv.y, pxv.z, v0, v1, v2);
45}

47static inline
48Vec3d calcRgbdEquationCoeffsRotation(double dIdx, double dIdy, const Point3d& p3d, double fx, double fy)
49{
50 double invz = 1. / p3d.z,
51 v0 = dIdx * fx * invz,
52 v1 = dIdy * fy * invz,
53 v2 = -(v0 * p3d.x + v1 * p3d.y) * invz;
54
55 Point3d v(v0, v1, v2);
56 Point3d pxv = p3d.cross(v);
57
58 return Vec3d(pxv);
59}

61static inline
62Vec3d calcRgbdEquationCoeffsTranslation(double dIdx, double dIdy, const Point3d& p3d, double fx, double fy)
63{
64 double invz = 1. / p3d.z,
65 v0 = dIdx * fx * invz,
66 v1 = dIdy * fy * invz,
67 v2 = -(v0 * p3d.x + v1 * p3d.y) * invz;
68
69 return Vec3d(v0, v1, v2);
70}

72static inline void rgbdCoeffsFunc(OdometryTransformType transformType,
73 double* C, double dIdx, double dIdy, const Point3d& p3d, double fx, double fy)
74{
75 int dim = getTransformDim(transformType);
76 Vec6d ret;
77 switch(transformType)
78 {
79 case OdometryTransformType::RIGID_TRANSFORMATION:
80 {
81 ret = calcRgbdEquationCoeffs(dIdx, dIdy, p3d, fx, fy);
82 break;
83 }
84 case OdometryTransformType::ROTATION:
85 {
86 Vec3d r = calcRgbdEquationCoeffsRotation(dIdx, dIdy, p3d, fx, fy);
87 ret = Vec6d(r[0], r[1], r[2], 0, 0, 0);
88 break;
89 }
90 case OdometryTransformType::TRANSLATION:
91 {
92 Vec3d r = calcRgbdEquationCoeffsTranslation(dIdx, dIdy, p3d, fx, fy);
93 ret = Vec6d(r[0], r[1], r[2], 0, 0, 0);
94 break;
95 }
96 default:
97 CV_Error(Error::StsBadArg, "Incorrect transformation type");
98 }
99 for (int i = 0; i < dim; i++)
100 C[i] = ret[i];
101}

105Vec6d calcICPEquationCoeffs(const Point3d& psrc, const Vec3d& ndst)
106{
107 Point3d pxv = psrc.cross(Point3d(ndst));

126static inline
127void icpCoeffsFunc(OdometryTransformType transformType, double* C, const Point3d& p0, const Point3d& /*p1*/, const Vec3d& n1)
128{
129 int dim = getTransformDim(transformType);
130 Vec6d ret;
131 switch(transformType)
132 {
133 case OdometryTransformType::RIGID_TRANSFORMATION:
134 {
135 ret = calcICPEquationCoeffs(p0, n1);
136 break;
137 }
138 case OdometryTransformType::ROTATION:
139 {
140 Vec3d r = calcICPEquationCoeffsRotation(p0, n1);
141 ret = Vec6d(r[0], r[1], r[2], 0, 0, 0);
142 break;
143 }
144 case OdometryTransformType::TRANSLATION:
145 {
146 Vec3d r = calcICPEquationCoeffsTranslation(p0, n1);
147 ret = Vec6d(r[0], r[1], r[2], 0, 0, 0);
148 break;
149 }
150 default:
151 CV_Error(Error::StsBadArg, "Incorrect transformation type");
152 }
153 for (int i = 0; i < dim; i++)
154 C[i] = ret[i];
155}
