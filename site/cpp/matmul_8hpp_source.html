23 template <class T>
24 class MatMulOp final : public CUDABackendNode {
25 public:
26 using wrapper_type = GetCUDABackendWrapperType<T>;
27
28 MatMulOp(csl::Stream stream_, csl::cublas::Handle handle, const Mat& constInp, const Mat& bias, bool _transA, bool _transB)
29 : stream(std::move(stream_)), cublasHandle(std::move(handle))
30 {
31 if (!constInp.empty())
32 {
33 constTensor = csl::makeTensorHeader<T>(constInp);
34 csl::copyMatToTensor<T>(constInp, constTensor, stream);
35 }
36
37 if (!bias.empty())
38 {
39 biasTensor = csl::makeTensorHeader<T>(bias);
40 csl::copyMatToTensor<T>(bias, biasTensor, stream);
41 }
42
43 transA = _transA;
44 transB = _transB;
45 }
46
47 void forward(
48 const std::vector<cv::Ptr<BackendWrapper>>& inputs,
49 const std::vector<cv::Ptr<BackendWrapper>>& outputs,
50 csl::Workspace& workspace) override
51 {
52 CV_Assert(((inputs.size() == 2 && constTensor.empty()) ||
53 (inputs.size() == 1 && !constTensor.empty())) && outputs.size() == 1);
54
55 auto input1_wrapper = inputs[0].dynamicCast<wrapper_type>();
56 auto input1 = input1_wrapper->getView();
57
58 csl::TensorView<T> input2;
59 if (constTensor.empty())
60 {
61 auto input2_wrapper = inputs[1].dynamicCast<wrapper_type>();
62 input2 = input2_wrapper->getView();
63 }
64 else
65 input2 = csl::TensorView<T>(constTensor);
66
67 auto output_wrapper = outputs[0].dynamicCast<wrapper_type>();
68 auto output = output_wrapper->getSpan();
69
70 auto rank = output.rank();
71 CV_Assert(rank == input1.rank());
72 CV_Assert(rank == input2.rank());
73 CV_Assert(rank >= 2); // 1D MatMul not supported
74
75 for (int i = 0; i < rank - 2; i++)
76 {
77 // broadcasting not supported
78 auto size = output.get_axis_size(i);
79 CV_Assert(input1.get_axis_size(i) == size);
80 CV_Assert(input2.get_axis_size(i) == size);
81 }
82
83 int m1, n1, b1, m2, n2, b2;
84 if (transA)
85 {
86 m1 = input1.get_axis_size(-1);
87 n1 = input1.get_axis_size(-2);
88 }
89 else
90 {
91 m1 = input1.get_axis_size(-2);
92 n1 = input1.get_axis_size(-1);
93 }
94
95 if (transB)
96 {
97 m2 = input2.get_axis_size(-1);
98 n2 = input2.get_axis_size(-2);
99 }
100 else
101 {
102 m2 = input2.get_axis_size(-2);
103 n2 = input2.get_axis_size(-1);
104 }
105
106 b1 = input1.size() / m1 / n1;
107 b2 = input2.size() / m2 / n2;
108 CV_Assert(b1 == b2);
109 CV_Assert(n1 == m2);
110 CV_Assert(output.get_axis_size(-2) == m1);
111 CV_Assert(output.get_axis_size(-1) == n2);
112
113 if (get_effective_rank(output) <= 2)
114 {
115 CV_Assert(b2 == 1);
116 CV_Assert(get_effective_rank(input1) <= 2);
117 CV_Assert(get_effective_rank(input2) <= 2);
118 csl::tensor_ops::gemm<T>(cublasHandle, 0.0, output, 1.0, transA, input1, transB, input2);
119 // used for GEMM
120 if (!biasTensor.empty())
121 kernels::biasN<T>(stream, output, output, 1, biasTensor);
122 }
123 else
124 {
125 CV_Assert(rank >= 3);
126 if (transA)
127 input1.reshape(b1, n1, m1);
128 else
129 input1.reshape(b1, m1, n1);
130
131 if (transB)
132 input2.reshape(b2, n2, m2);
133 else
134 input2.reshape(b2, m2, n2);
135
136 output.reshape(b1, m1, n2);
137 input1.squeeze_to(3);
138 input2.squeeze_to(3);
139 output.squeeze_to(3);
140 csl::tensor_ops::gemmStridedBatched<T>(cublasHandle, 0.0, output, 1.0, transA, input1, transB, input2);
141 }
142 }
143
144 private:
145 csl::Stream stream;
146 csl::cublas::Handle cublasHandle;
147 csl::Tensor<T> constTensor, biasTensor;
148 bool transA, transB;
149 };
