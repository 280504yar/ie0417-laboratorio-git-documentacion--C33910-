124bool CvCaptureCAM_XIMEA::_open()
125{
126#define HandleXiResult(res) if (res!=XI_OK) goto error;
127 int width = 0;
128 int height = 0;
129 int isColor = 0;
130
131 // always use auto exposure/gain
132 int mvret = xiSetParamInt( hmv, XI_PRM_AEAG, 1);
133 HandleXiResult(mvret);
134
135 mvret = xiGetParamInt( hmv, XI_PRM_WIDTH, &width);
136 HandleXiResult(mvret);
137
138 mvret = xiGetParamInt( hmv, XI_PRM_HEIGHT, &height);
139 HandleXiResult(mvret);
140
141 mvret = xiGetParamInt(hmv, XI_PRM_IMAGE_IS_COLOR, &isColor);
142 HandleXiResult(mvret);
143
144 if(isColor) // for color cameras
145 {
146 // default image format RGB24
147 mvret = xiSetParamInt( hmv, XI_PRM_IMAGE_DATA_FORMAT, XI_RGB24);
148 HandleXiResult(mvret);
149
150 // always use auto white balance for color cameras
151 mvret = xiSetParamInt( hmv, XI_PRM_AUTO_WB, 1);
152 HandleXiResult(mvret);
153
154 // allocate frame buffer for RGB24 image
155 frame.create(Size(width, height), CV_8UC3);
156 }
157 else // for mono cameras
158 {
159 // default image format MONO8
160 mvret = xiSetParamInt( hmv, XI_PRM_IMAGE_DATA_FORMAT, XI_MONO8);
161 HandleXiResult(mvret);
162
163 // allocate frame buffer for MONO8 image
164 frame.create(Size(width, height), CV_8UC1);
165 }
166
167 //default capture timeout 10s
168 timeout = 10000;
169
170 mvret = xiStartAcquisition(hmv);
171 if(mvret != XI_OK)
172 {
173 errMsg("StartAcquisition XI_DEVICE failed", mvret);
174 goto error;
175 }
176 return true;
177
178error:
179 errMsg("Open XI_DEVICE failed", mvret);
180 xiCloseDevice(hmv);
181 hmv = NULL;
182 return false;
183}

206 int mvret = xiGetImage( hmv, timeout, &image);

210 xiStartAcquisition(hmv);

212 }
213
214 if(mvret != XI_OK)
215 {
216 errMsg("Error during GetImage", mvret);
217 return false;
218 }
219
220 return true;

233 case XI_RGB_PLANAR : return CV_8UC3;
234 case XI_RGB32 : return CV_8UC4;

258void CvCaptureCAM_XIMEA::resetCvImage()
259{
260 bool do_reset = false;
261
262 // first check basic image resolution
263 if((int)image.width != frame.size().width || (int)image.height != frame.size().height)
264 do_reset = true;
265
266 const int image_type = getMatType(image.frm);
267 CV_Assert(image_type != -1);
268 if (frame.type() != image_type)
269 do_reset = true;
270
271 if(do_reset)
272 frame = Mat::zeros(Size(image.width, image.height), image_type);
273}
