191def RGB2LAB(r,g,b):
192 if max(r,g,b):
193 r /= 255.
194 g /= 255.
195 b /= 255.
196
197 X = (0.412453 * r + 0.357580 * g + 0.180423 * b) / 0.950456
198 Y = (0.212671 * r + 0.715160 * g + 0.072169 * b)
199 Z = (0.019334 * r + 0.119193 * g + 0.950227 * b) / 1.088754
200
201 #[X * 0.950456] [0.412453 0.357580 0.180423] [R]
202 #[Y ] = [0.212671 0.715160 0.072169] * [G]
203 #[Z * 1.088754] [0.019334 0.119193 0.950227] [B]
204
205 T = 0.008856 #threshold
206
207 if X > T:
208 fX = math.pow(X, 1./3.)
209 else:
210 fX = 7.787 * X + 16./116.
211
212 # Compute L
213 if Y > T:
214 Y3 = math.pow(Y, 1./3.)
215 fY = Y3
216 L = 116. * Y3 - 16.0
217 else:
218 fY = 7.787 * Y + 16./116.
219 L = 903.3 * Y
220
221 if Z > T:
222 fZ = math.pow(Z, 1./3.)
223 else:
224 fZ = 7.787 * Z + 16./116.
225
226 # Compute a and b
227 a = 500. * (fX - fY)
228 b = 200. * (fY - fZ)
229
230 return (L,a,b)

232def colorDistance(r1,g1,b1 = None, r2 = None, g2 = None,b2 = None):
233 if type(r1) == tuple and type(g1) == tuple and b1 is None and r2 is None and g2 is None and b2 is None:
234 (l1,a1,b1) = RGB2LAB(*r1)
235 (l2,a2,b2) = RGB2LAB(*g1)
236 else:
237 (l1,a1,b1) = RGB2LAB(r1,g1,b1)
238 (l2,a2,b2) = RGB2LAB(r2,g2,b2)
239 #CIE94
240 dl = l1-l2
241 C1 = math.sqrt(a1*a1 + b1*b1)
242 C2 = math.sqrt(a2*a2 + b2*b2)
243 dC = C1 - C2
244 da = a1-a2
245 db = b1-b2
246 dH = math.sqrt(max(0, da*da + db*db - dC*dC))
247 Kl = 1
248 K1 = 0.045
249 K2 = 0.015
250
251 s1 = dl/Kl
252 s2 = dC/(1. + K1 * C1)
253 s3 = dH/(1. + K2 * C1)
254 return math.sqrt(s1*s1 + s2*s2 + s3*s3)

256def parseHexColor(col):
257 if len(col) != 4 and len(col) != 7 and not col.startswith("#"):
258 return (0,0,0)
259 if len(col) == 4:
260 r = col[1]*2
261 g = col[2]*2
262 b = col[3]*2
263 else:
264 r = col[1:3]
265 g = col[3:5]
