12class HomographyMinimalSolver4ptsImpl : public HomographyMinimalSolver4pts {
13private:
14 Mat points_mat;
15 const bool use_ge;
16public:
17 explicit HomographyMinimalSolver4ptsImpl (const Mat &points_, bool use_ge_) :
18 points_mat(points_), use_ge(use_ge_)
19 {
20 CV_DbgAssert(!points_mat.empty() && points_mat.isContinuous());
21 }
22
23 int estimate (const std::vector<int>& sample, std::vector<Mat> &models) const override {
24 const float * points = points_mat.ptr<float>();
25 int m = 8, n = 9;
26 std::vector<double> A(72, 0);
27 int cnt = 0;
28 for (int i = 0; i < 4; i++) {
29 const int smpl = 4*sample[i];
30 const auto x1 = points[smpl], y1 = points[smpl+1], x2 = points[smpl+2], y2 = points[smpl+3];
31
32 A[cnt++] = -x1;
33 A[cnt++] = -y1;
34 A[cnt++] = -1;
35 cnt += 3; // skip zeros
36 A[cnt++] = x2*x1;
37 A[cnt++] = x2*y1;
38 A[cnt++] = x2;
39
40 cnt += 3;
41 A[cnt++] = -x1;
42 A[cnt++] = -y1;
43 A[cnt++] = -1;
44 A[cnt++] = y2*x1;
45 A[cnt++] = y2*y1;
46 A[cnt++] = y2;
47 }
48
49 if (use_ge) {
50 if (!Math::eliminateUpperTriangular(A, m, n))
51 return 0;
52
53 models = std::vector<Mat>{ Mat_<double>(3,3) };
54 auto * h = (double *) models[0].data;
55 h[8] = 1.;
56
57 // start from the last row
58 for (int i = m-1; i >= 0; i--) {
59 double acc = 0;
60 for (int j = i+1; j < n; j++)
61 acc -= A[i*n+j]*h[j];
62
63 h[i] = acc / A[i*n+i];
64 // due to numerical errors return 0 solutions
65 if (std::isnan(h[i]))
66 return 0;
67 }
68 } else {
69 Mat U, Vt, D;
70 cv::Matx<double, 8, 9> A_svd(&A[0]);
71 SVD::compute(A_svd, D, U, Vt, SVD::FULL_UV+SVD::MODIFY_A);
72 models = std::vector<Mat> { Vt.row(Vt.rows-1).reshape(0, 3) };
73 }
74 return 1;
75 }
76
77 int getMaxNumberOfSolutions () const override { return 1; }
78 int getSampleSize() const override { return 4; }

104 if (sample_size < getMinimumRequiredSampleSize())
105 return 0;
106
107 Matx33d T1, T2;
108 Mat norm_points_;
109 if (do_norm)
110 normTr->getNormTransformation(norm_points_, sample, sample_size, T1, T2);
111 const float * const npts = do_norm ? norm_points_.ptr<float>() : points_mat.ptr<float>();
112
113 Mat H;
114 if (use_ge) {
115 double a1[8] = {0, 0, -1, 0, 0, 0, 0, 0},
116 a2[8] = {0, 0, 0, 0, 0, -1, 0, 0};
117 std::vector<double> AtAb(72, 0); // 8x9
118 if (weights.empty()) {
119 for (int i = 0; i < sample_size; i++) {
120 const int idx = do_norm ? 4*i : 4*sample[i];
121 const double x1 = npts[idx], y1 = npts[idx+1], x2 = npts[idx+2], y2 = npts[idx+3];
122 a1[0] = -x1;
123 a1[1] = -y1;
124 a1[6] = x2*x1;
125 a1[7] = x2*y1;
126
127 a2[3] = -x1;
128 a2[4] = -y1;
129 a2[6] = y2*x1;
130 a2[7] = y2*y1;
131
132 // calculate covariance for eigen
133 for (int j = 0; j < 8; j++) {
134 for (int z = j; z < 8; z++)
135 AtAb[j * 9 + z] += a1[j]*a1[z] + a2[j]*a2[z];
136 AtAb[j * 9 + 8] += a1[j]*x2 + a2[j]*y2;
137 }
138 }
139 } else { // use weights
140 for (int i = 0; i < sample_size; i++) {
141 const double weight = weights[i];
142 if (weight < FLT_EPSILON) continue;
143 const int idx = do_norm ? 4*i : 4*sample[i];
144 const double x1 = npts[idx], y1 = npts[idx+1], x2 = npts[idx+2], y2 = npts[idx+3];
145 const double minus_weight_times_x1 = -weight * x1,
146 minus_weight_times_y1 = -weight * y1,
147 weight_times_x2 = weight * x2,
148 weight_times_y2 = weight * y2;
149
150 a1[0] = minus_weight_times_x1;
151 a1[1] = minus_weight_times_y1;
152 a1[2] = -weight;
153 a1[6] = weight_times_x2 * x1;
154 a1[7] = weight_times_x2 * y1;
155
156 a2[3] = minus_weight_times_x1;
157 a2[4] = minus_weight_times_y1;
158 a2[5] = -weight;
159 a2[6] = weight_times_y2 * x1;
160 a2[7] = weight_times_y2 * y1;
161
162 for (int j = 0; j < 8; j++) {
163 for (int z = j; z < 8; z++)
164 AtAb[j * 9 + z] += a1[j]*a1[z] + a2[j]*a2[z];
165 AtAb[j * 9 + 8] += a1[j]*weight_times_x2 + a2[j]*weight_times_y2;
166 }
167 }
168 }
169 for (int j = 1; j < 8; j++)
170 for (int z = 0; z < j; z++)
171 AtAb[j*9+z] = AtAb[z*9+j];
172 if (!Math::eliminateUpperTriangular(AtAb, 8, 9))
173 return 0;
174 H = Mat_<double>(3,3);
175 auto * h = (double *) H.data;
176 h[8] = 1.;
177 const int m = 8, n = 9;
178 // start from the last row
179 for (int i = m-1; i >= 0; i--) {
180 double acc = 0;
181 for (int j = i+1; j < n; j++)
182 acc -= AtAb[i*n+j]*h[j];
183 h[i] = acc / AtAb[i*n+i];
184 if (std::isnan(h[i]))
185 return 0; // numerical imprecision
186 }
187 } else {
188 double a1[9] = {0, 0, -1, 0, 0, 0, 0, 0, 0},
189 a2[9] = {0, 0, 0, 0, 0, -1, 0, 0, 0}, AtA[81] = {0};
190 if (weights.empty()) {
191 for (int i = 0; i < sample_size; i++) {
192 const int smpl = do_norm ? 4*i : 4*sample[i];
193 const auto x1 = npts[smpl], y1 = npts[smpl+1], x2 = npts[smpl+2], y2 = npts[smpl+3];
194
195 a1[0] = -x1;
196 a1[1] = -y1;
197 a1[6] = x2*x1;
198 a1[7] = x2*y1;
199 a1[8] = x2;
200
201 a2[3] = -x1;
202 a2[4] = -y1;
203 a2[6] = y2*x1;
204 a2[7] = y2*y1;
205 a2[8] = y2;
206
207 for (int j = 0; j < 9; j++)
208 for (int z = j; z < 9; z++)
209 AtA[j*9+z] += a1[j]*a1[z] + a2[j]*a2[z];
210 }
211 } else { // use weights
212 for (int i = 0; i < sample_size; i++) {
213 const double weight = weights[i];
214 if (weight < FLT_EPSILON) continue;
215 const int smpl = do_norm ? 4*i : 4*sample[i];
216 const auto x1 = npts[smpl], y1 = npts[smpl+1], x2 = npts[smpl+2], y2 = npts[smpl+3];
217 const double minus_weight_times_x1 = -weight * x1,
218 minus_weight_times_y1 = -weight * y1,
219 weight_times_x2 = weight * x2,
220 weight_times_y2 = weight * y2;
221
222 a1[0] = minus_weight_times_x1;
223 a1[1] = minus_weight_times_y1;
224 a1[2] = -weight;
225 a1[6] = weight_times_x2 * x1;
226 a1[7] = weight_times_x2 * y1;
227 a1[8] = weight_times_x2;
228
229 a2[3] = minus_weight_times_x1;
230 a2[4] = minus_weight_times_y1;
231 a2[5] = -weight;
232 a2[6] = weight_times_y2 * x1;
233 a2[7] = weight_times_y2 * y1;
234 a2[8] = weight_times_y2;
235
236 for (int j = 0; j < 9; j++)
237 for (int z = j; z < 9; z++)
238 AtA[j*9+z] += a1[j]*a1[z] + a2[j]*a2[z];
239 }
240 }
241 // copy symmetric part of covariance matrix
242 for (int j = 1; j < 9; j++)
243 for (int z = 0; z < j; z++)
244 AtA[j*9+z] = AtA[z*9+j];
245
246#ifdef HAVE_EIGEN
247 H = Mat_<double>(3,3);
248 // extract the last null-vector
249 Eigen::Map<Eigen::Matrix<double, 9, 1>>((double *)H.data) = Eigen::Matrix<double, 9, 9>
250 (Eigen::HouseholderQR<Eigen::Matrix<double, 9, 9>> (
251 (Eigen::Matrix<double, 9, 9> (AtA))).householderQ()).col(8);
252#else
253 Matx<double, 9, 9> Vt;
254 Vec<double, 9> D;
255 if (! eigen(Matx<double, 9, 9>(AtA), D, Vt)) return 0;
256 H = Mat_<double>(3, 3, Vt.val + 72/*=8*9*/).clone();
257#endif
258 }
259 const auto * const h = (double *) H.data;
260 const auto * const t1 = do_norm ? T1.val : _T1.val, * const t2 = do_norm ? T2.val : _T2.val;
261 // H = T2^-1 H T1
262 models = std::vector<Mat>{ Mat(Matx33d(t1[0]*(h[0]/t2[0] - (h[6]*t2[2])/t2[0]),
263 t1[0]*(h[1]/t2[0] - (h[7]*t2[2])/t2[0]), h[2]/t2[0] + t1[2]*(h[0]/t2[0] -
264 (h[6]*t2[2])/t2[0]) + t1[5]*(h[1]/t2[0] - (h[7]*t2[2])/t2[0]) - (h[8]*t2[2])/t2[0],

266 h[5]/t2[0] + t1[2]*(h[3]/t2[0] - (h[6]*t2[5])/t2[0]) + t1[5]*(h[4]/t2[0] -
267 (h[7]*t2[5])/t2[0]) - (h[8]*t2[5])/t2[0], t1[0]*h[6], t1[0]*h[7],
268 h[8] + h[6]*t1[2] + h[7]*t1[5])) };
269 return 1;
270 }

273 return 0;

286class CovarianceHomographySolverImpl : public CovarianceHomographySolver {

319 /*
320 * Find homography using 4-point algorithm with covariance matrix and PCA
321 */
322 int estimate (const std::vector<bool> &new_mask, std::vector<Mat> &models,
323 const std::vector<double> &/*weights*/) override {
324 double a1[9] = {0, 0, -1, 0, 0, 0, 0, 0, 0},
325 a2[9] = {0, 0, 0, 0, 0, -1, 0, 0, 0};
326
327 for (int i = 0; i < points_size; i++) {
328 if (mask[i] != new_mask[i]) {
329 const int smpl = 4*i;
330 const double x1 = norm_points[smpl ], y1 = norm_points[smpl+1],
331 x2 = norm_points[smpl+2], y2 = norm_points[smpl+3];
332
333 a1[0] = -x1;
334 a1[1] = -y1;
335 a1[6] = x2*x1;
336 a1[7] = x2*y1;
337 a1[8] = x2;
338
339 a2[3] = -x1;
340 a2[4] = -y1;
341 a2[6] = y2*x1;
342 a2[7] = y2*y1;
343 a2[8] = y2;
344
345 if (mask[i]) // if mask[i] is true then new_mask[i] must be false
346 for (int j = 0; j < 9; j++)
347 for (int z = j; z < 9; z++)
348 covariance[j*9+z] +=-a1[j]*a1[z] - a2[j]*a2[z];
349 else
350 for (int j = 0; j < 9; j++)
351 for (int z = j; z < 9; z++)
352 covariance[j*9+z] += a1[j]*a1[z] + a2[j]*a2[z];
353 }
354 }
355 mask = new_mask;
356
357 // copy symmetric part of covariance matrix
358 for (int j = 1; j < 9; j++)
359 for (int z = 0; z < j; z++)
360 covariance[j*9+z] = covariance[z*9+j];
361
362#ifdef HAVE_EIGEN
363 Mat H = Mat_<double>(3,3);
364 // extract the last null-vector
365 Eigen::Map<Eigen::Matrix<double, 9, 1>>((double *)H.data) = Eigen::Matrix<double, 9, 9>
366 (Eigen::HouseholderQR<Eigen::Matrix<double, 9, 9>> (
367 (Eigen::Matrix<double, 9, 9> (covariance))).householderQ()).col(8);
368#else
369 Matx<double, 9, 9> Vt;
370 Vec<double, 9> D;
371 if (! eigen(Matx<double, 9, 9>(covariance), D, Vt)) return 0;
372 Mat H = Mat_<double>(3, 3, Vt.val + 72/*=8*9*/);
373#endif
374
375 const auto * const h = (double *) H.data;
376 // H = T2^-1 H T1
377 models = std::vector<Mat>{ Mat(Matx33d(t1[0]*(h[0]/t2[0] - (h[6]*t2[2])/t2[0]),
378 t1[0]*(h[1]/t2[0] - (h[7]*t2[2])/t2[0]), h[2]/t2[0] + t1[2]*(h[0]/t2[0] -
379 (h[6]*t2[2])/t2[0]) + t1[5]*(h[1]/t2[0] - (h[7]*t2[2])/t2[0]) - (h[8]*t2[2])/t2[0],
380 t1[0]*(h[3]/t2[0] - (h[6]*t2[5])/t2[0]), t1[0]*(h[4]/t2[0] - (h[7]*t2[5])/t2[0]),
381 h[5]/t2[0] + t1[2]*(h[3]/t2[0] - (h[6]*t2[5])/t2[0]) + t1[5]*(h[4]/t2[0] -
382 (h[7]*t2[5])/t2[0]) - (h[8]*t2[5])/t2[0], t1[0]*h[6], t1[0]*h[7],
383 h[8] + h[6]*t1[2] + h[7]*t1[5])) };
384
385 return 1;
386 }
387 void enforceRankConstraint (bool /*enforce*/) override {}
388 int getMinimumRequiredSampleSize() const override { return 4; }
389 int getMaxNumberOfSolutions () const override { return 1; }

398class AffineMinimalSolverImpl : public AffineMinimalSolver {
399private:
400 Mat points_mat;
401public:
402 explicit AffineMinimalSolverImpl (const Mat &points_) :
403 points_mat(points_)
404 {
405 CV_DbgAssert(!points_mat.empty() && points_mat.isContinuous());
406 }
407 /*
408 Affine transformation
409 x1 y1 1 0 0 0 a u1
410 0 0 0 x1 y1 1 b v1
411 x2 y2 1 0 0 0 c u2
412 0 0 0 x2 y2 1 * d = v2
413 x3 y3 1 0 0 0 e u3
414 0 0 0 x3 y3 1 f v3
415 */
416 int estimate (const std::vector<int> &sample, std::vector<Mat> &models) const override {
417 const int smpl1 = 4*sample[0], smpl2 = 4*sample[1], smpl3 = 4*sample[2];
418 const float * points = points_mat.ptr<float>();
419 const auto
420 x1 = points[smpl1], y1 = points[smpl1+1], u1 = points[smpl1+2], v1 = points[smpl1+3],
421 x2 = points[smpl2], y2 = points[smpl2+1], u2 = points[smpl2+2], v2 = points[smpl2+3],
422 x3 = points[smpl3], y3 = points[smpl3+1], u3 = points[smpl3+2], v3 = points[smpl3+3];
423
424 // covers degeneracy test when all 3 points are collinear.
425 // In this case denominator will be 0
426 double denominator = x1*y2 - x2*y1 - x1*y3 + x3*y1 + x2*y3 - x3*y2;
427 if (fabs(denominator) < FLT_EPSILON) // check if denominator is zero
428 return 0;
429 denominator = 1. / denominator;
430
431 double a = (u1*y2 - u2*y1 - u1*y3 + u3*y1 + u2*y3 - u3*y2) * denominator;
432 double b = -(u1*x2 - u2*x1 - u1*x3 + u3*x1 + u2*x3 - u3*x2) * denominator;
433 double c = u1 - a * x1 - b * y1; // ax1 + by1 + c = u1
434 double d = (v1*y2 - v2*y1 - v1*y3 + v3*y1 + v2*y3 - v3*y2) * denominator;
435 double e = -(v1*x2 - v2*x1 - v1*x3 + v3*x1 + v2*x3 - v3*x2) * denominator;
436 double f = v1 - d * x1 - e * y1; // dx1 + ey1 + f = v1
437
438 models[0] = Mat(Matx33d(a, b, c, d, e, f, 0, 0, 1));
439 return 1;
440 }
441 int getSampleSize() const override { return 3; }
442 int getMaxNumberOfSolutions () const override { return 1; }
443};

448class AffineNonMinimalSolverImpl : public AffineNonMinimalSolver {
449private:
450 Mat points_mat;
451 Ptr<NormTransform> normTr;
452 Matx33d _T1, _T2;
453 bool do_norm;
454public:
455 explicit AffineNonMinimalSolverImpl (const Mat &points_, InputArray T1, InputArray T2) :
456 points_mat(points_) {
457 CV_DbgAssert(!points_mat.empty() && points_mat.isContinuous());
458 if (!T1.empty() && !T2.empty()) {
459 do_norm = false;
460 _T1 = T1.getMat();
461 _T2 = T2.getMat();
462 } else {
463 do_norm = true;
464 normTr = NormTransform::create(points_);
465 }
466 }
467
468 int estimate (const std::vector<int> &sample, int sample_size, std::vector<Mat> &models,
469 const std::vector<double> &weights) const override {
470 if (sample_size < getMinimumRequiredSampleSize())
471 return 0;
472 Matx33d T1, T2;
473 Mat norm_points_;
474 if (do_norm)
475 normTr->getNormTransformation(norm_points_, sample, sample_size, T1, T2);
476 const float * const pts = normTr ? norm_points_.ptr<float>() : points_mat.ptr<float>();
477
478 // do Least Squares
479 // Ax = b -> A^T Ax = A^T b
480 // x = (A^T A)^-1 A^T b
481 double AtA[36] = {0}, Ab[6] = {0};
482 double r1[6] = {0, 0, 1, 0, 0, 0}; // row 1 of A
483 double r2[6] = {0, 0, 0, 0, 0, 1}; // row 2 of A
484
485 if (weights.empty())
486 for (int p = 0; p < sample_size; p++) {
487 const int idx = do_norm ? 4*p : 4*sample[p];
488 const auto x1=pts[idx], y1=pts[idx+1], x2=pts[idx+2], y2=pts[idx+3];
489 r1[0] = x1;
490 r1[1] = y1;
491
492 r2[3] = x1;
493 r2[4] = y1;
494
495 for (int j = 0; j < 6; j++) {
496 for (int z = j; z < 6; z++)
497 AtA[j * 6 + z] += r1[j] * r1[z] + r2[j] * r2[z];
498 Ab[j] += r1[j]*x2 + r2[j]*y2;
499 }
500 }
501 else
502 for (int p = 0; p < sample_size; p++) {
503 const auto weight = weights[p];
504 if (weight < FLT_EPSILON) continue;
505 const int idx = do_norm ? 4*p : 4*sample[p];
506 const double weight_times_x1 = weight * pts[idx ],
507 weight_times_y1 = weight * pts[idx+1],
508 weight_times_x2 = weight * pts[idx+2],
509 weight_times_y2 = weight * pts[idx+3];
510
511 r1[0] = weight_times_x1;
512 r1[1] = weight_times_y1;
513 r1[2] = weight;
514
515 r2[3] = weight_times_x1;
516 r2[4] = weight_times_y1;
517 r2[5] = weight;
518
519 for (int j = 0; j < 6; j++) {
520 for (int z = j; z < 6; z++)
521 AtA[j * 6 + z] += r1[j] * r1[z] + r2[j] * r2[z];
522 Ab[j] += r1[j]*weight_times_x2 + r2[j]*weight_times_y2;
523 }
524 }
525
526 // copy symmetric part
527 for (int j = 1; j < 6; j++)
528 for (int z = 0; z < j; z++)
529 AtA[j*6+z] = AtA[z*6+j];
530
531 Vec6d aff;
532 if (!solve(Matx66d(AtA), Vec6d(Ab), aff))
533 return 0;
534 const double h[9] = {aff(0), aff(1), aff(2),
535 aff(3), aff(4), aff(5),
536 0, 0, 1};
537 const auto * const t1 = normTr ? T1.val : _T1.val, * const t2 = normTr ? T2.val : _T2.val;
538 // A = T2^-1 A T1
539 models = std::vector<Mat>{ Mat(Matx33d(t1[0]*(h[0]/t2[0] - (h[6]*t2[2])/t2[0]),
540 t1[0]*(h[1]/t2[0] - (h[7]*t2[2])/t2[0]), h[2]/t2[0] + t1[2]*(h[0]/t2[0] -
541 (h[6]*t2[2])/t2[0]) + t1[5]*(h[1]/t2[0] - (h[7]*t2[2])/t2[0]) - (h[8]*t2[2])/t2[0],
542 t1[0]*(h[3]/t2[0] - (h[6]*t2[5])/t2[0]), t1[0]*(h[4]/t2[0] - (h[7]*t2[5])/t2[0]),
543 h[5]/t2[0] + t1[2]*(h[3]/t2[0] - (h[6]*t2[5])/t2[0]) + t1[5]*(h[4]/t2[0] -
544 (h[7]*t2[5])/t2[0]) - (h[8]*t2[5])/t2[0], t1[0]*h[6], t1[0]*h[7],
545 h[8] + h[6]*t1[2] + h[7]*t1[5])) };
546 return 1;
547 }
548 int estimate (const std::vector<bool> &/*mask*/, std::vector<Mat> &/*models*/,
549 const std::vector<double> &/*weights*/) override {
550 return 0;
551 }
552 void enforceRankConstraint (bool /*enforce*/) override {}

554 int getMinimumRequiredSampleSize() const override { return 3; }
555 int getMaxNumberOfSolutions () const override { return 1; }

595 */

599 double r2[6] = {0, 0, 0, 0, 0, 1}; // row 2 of A
600 for (int i = 0; i < points_size; i++) {
601 if (mask[i] != new_mask[i]) {
602 const int smpl = 4*i;
603 const double x1 = norm_points[smpl ], y1 = norm_points[smpl+1],
604 x2 = norm_points[smpl+2], y2 = norm_points[smpl+3];
605
606 r1[0] = x1;
607 r1[1] = y1;
608
609 r2[3] = x1;
610 r2[4] = y1;
611
612 if (mask[i]) // if mask[i] is true then new_mask[i] must be false
613 for (int j = 0; j < 6; j++) {
614 for (int z = j; z < 6; z++)
615 covariance[j*6+z] +=-r1[j]*r1[z] - r2[j]*r2[z];
616 Ab[j] +=-r1[j]*x2 - r2[j]*y2;
617 }
618 else
619 for (int j = 0; j < 6; j++) {
620 for (int z = j; z < 6; z++)
621 covariance[j*6+z] += r1[j]*r1[z] + r2[j]*r2[z];
622 Ab[j] += r1[j]*x2 + r2[j]*y2;
623 }
624 }
625 }
626 mask = new_mask;
627
628 // copy symmetric part of covariance matrix
629 for (int j = 1; j < 6; j++)
630 for (int z = 0; z < j; z++)
631 covariance[j*6+z] = covariance[z*6+j];
632
633 Vec6d aff;
634 if (!solve(Matx66d(covariance), Vec6d(Ab), aff))
635 return 0;
636 double a[9] = { aff(0), aff(1), aff(2), aff(3), aff(4), aff(5), 0, 0, 1 };
637 models = std::vector<Mat>{ Mat(Matx33d(t1[0]*(a[0]/t2[0] - (a[6]*t2[2])/t2[0]),
638 t1[0]*(a[1]/t2[0] - (a[7]*t2[2])/t2[0]), a[2]/t2[0] + t1[2]*(a[0]/t2[0] -
639 (a[6]*t2[2])/t2[0]) + t1[5]*(a[1]/t2[0] - (a[7]*t2[2])/t2[0]) - (a[8]*t2[2])/t2[0],
640 t1[0]*(a[3]/t2[0] - (a[6]*t2[5])/t2[0]), t1[0]*(a[4]/t2[0] - (a[7]*t2[5])/t2[0]),
641 a[5]/t2[0] + t1[2]*(a[3]/t2[0] - (a[6]*t2[5])/t2[0]) + t1[5]*(a[4]/t2[0] -
642 (a[7]*t2[5])/t2[0]) - (a[8]*t2[5])/t2[0], t1[0]*a[6], t1[0]*a[7],
643 a[8] + a[6]*t1[2] + a[7]*t1[5])) };
644 return 1;
645 }
