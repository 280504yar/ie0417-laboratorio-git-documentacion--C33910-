84static inline int calcType(int type, int flags)
85{
86 if(flags != IMREAD_UNCHANGED)
87 {
88 CV_CheckNE(flags & (IMREAD_COLOR_BGR | IMREAD_COLOR_RGB),
89 IMREAD_COLOR_BGR | IMREAD_COLOR_RGB,
90 "IMREAD_COLOR_BGR (IMREAD_COLOR) and IMREAD_COLOR_RGB can not be set at the same time.");
91 }
92
93 if ( (flags & (IMREAD_COLOR | IMREAD_ANYCOLOR | IMREAD_ANYDEPTH)) == (IMREAD_COLOR | IMREAD_ANYCOLOR | IMREAD_ANYDEPTH))
94 return type;
95
96 if( (flags & IMREAD_LOAD_GDAL) != IMREAD_LOAD_GDAL && flags != IMREAD_UNCHANGED )
97 {
98 if( (flags & IMREAD_ANYDEPTH) == 0 )
99 type = CV_MAKETYPE(CV_8U, CV_MAT_CN(type));
100
101 //if( (flags & IMREAD_ANYCOLOR) != 0 /*&& CV_MAT_CN(type) > 1*/ )
102 // type = CV_MAKETYPE(CV_MAT_DEPTH(type), CV_MAT_CN(type));
103 //else if( (flags & IMREAD_COLOR) != 0 || (flags & IMREAD_COLOR_RGB) != 0 )
104 if( (flags & IMREAD_COLOR) != 0 || (flags & IMREAD_COLOR_RGB) != 0 ||
105 ((flags & IMREAD_ANYCOLOR) != 0 && CV_MAT_CN(type) > 1) )
106 type = CV_MAKETYPE(CV_MAT_DEPTH(type), 3);
107 else
108 type = CV_MAKETYPE(CV_MAT_DEPTH(type), 1);
109 }
110 return type;
111}

115class ByteStreamBuffer: public std::streambuf
116{
117public:
118 ByteStreamBuffer(char* base, size_t length)
119 {
120 setg(base, base, base + length);
121 }
122
123protected:
124 virtual pos_type seekoff( off_type offset,
125 std::ios_base::seekdir dir,
126 std::ios_base::openmode ) CV_OVERRIDE
127 {
128 char* whence = eback();
129 if (dir == std::ios_base::cur)
130 {
131 whence = gptr();
132 }
133 else if (dir == std::ios_base::end)
134 {
135 whence = egptr();
136 }
137 char* to = whence + offset;
138
139 // check limits
140 if (to >= eback() && to <= egptr())
141 {
142 setg(eback(), to, egptr());
143 return gptr() - eback();
144 }
145
146 return -1;
147 }
148};

261static ImageDecoder findDecoder( const String& filename ) {
262
263 size_t i, maxlen = 0;
264
266 ImageCodecInitializer& codecs = getCodecs();
267 for( i = 0; i < codecs.decoders.size(); i++ )
268 {
269 size_t len = codecs.decoders[i]->signatureLength();
270 maxlen = std::max(maxlen, len);
271 }
272
274 FILE* f= fopen( filename.c_str(), "rb" );
275
277 if( !f ) {
278 CV_LOG_WARNING(NULL, "imread_('" << filename << "'): can't open/read file: check file path/integrity");
279 return ImageDecoder();
280 }
281
282 // read the file signature
283 String signature(maxlen, ' ');
284 maxlen = fread( (void*)signature.c_str(), 1, maxlen, f );
285 fclose(f);
286 signature = signature.substr(0, maxlen);
287
289 for( i = 0; i < codecs.decoders.size(); i++ )
290 {
291 if( codecs.decoders[i]->checkSignature(signature) )
292 return codecs.decoders[i]->newDecoder();
293 }
294
296 return ImageDecoder();
297}

299static ImageDecoder findDecoder( const Mat& buf )
300{
301 size_t i, maxlen = 0;
302
303 if( buf.rows*buf.cols < 1 || !buf.isContinuous() )
304 return ImageDecoder();
305
306 ImageCodecInitializer& codecs = getCodecs();
307 for( i = 0; i < codecs.decoders.size(); i++ )
308 {
309 size_t len = codecs.decoders[i]->signatureLength();
310 maxlen = std::max(maxlen, len);
311 }
312
313 String signature(maxlen, ' ');
314 size_t bufSize = buf.rows*buf.cols*buf.elemSize();
315 maxlen = std::min(maxlen, bufSize);
316 memcpy( (void*)signature.c_str(), buf.data, maxlen );
317
318 for( i = 0; i < codecs.decoders.size(); i++ )
319 {
320 if( codecs.decoders[i]->checkSignature(signature) )
321 return codecs.decoders[i]->newDecoder();
322 }
323
324 return ImageDecoder();
325}

327static ImageEncoder findEncoder( const String& _ext )
328{
329 if( _ext.size() <= 1 )
330 return ImageEncoder();
331
332 const char* ext = strrchr( _ext.c_str(), '.' );
333 if( !ext )
334 return ImageEncoder();
335 int len = 0;
336 for( ext++; len < 128 && isalnum(ext[len]); len++ )
337 ;
338
339 ImageCodecInitializer& codecs = getCodecs();
340 for( size_t i = 0; i < codecs.encoders.size(); i++ )
341 {
342 String description = codecs.encoders[i]->getDescription();
343 const char* descr = strchr( description.c_str(), '(' );
344
345 while( descr )
346 {
347 descr = strchr( descr + 1, '.' );
348 if( !descr )
349 break;
350 int j = 0;
351 for( descr++; j < len && isalnum(descr[j]) ; j++ )
352 {
353 int c1 = tolower(ext[j]);
354 int c2 = tolower(descr[j]);
355 if( c1 != c2 )
356 break;
357 }
358 if( j == len && !isalnum(descr[j]))
359 return codecs.encoders[i]->newEncoder();
360 descr += j;
361 }
362 }
363
364 return ImageEncoder();
365}

382static void ExifTransform(int orientation, OutputArray img)
383{
384 switch( orientation )
385 {
386 case IMAGE_ORIENTATION_TL: //0th row == visual top, 0th column == visual left-hand side
387 //do nothing, the image already has proper orientation
388 break;
389 case IMAGE_ORIENTATION_TR: //0th row == visual top, 0th column == visual right-hand side
390 flip(img, img, 1); //flip horizontally
391 break;
392 case IMAGE_ORIENTATION_BR: //0th row == visual bottom, 0th column == visual right-hand side
393 flip(img, img, -1);//flip both horizontally and vertically
394 break;
395 case IMAGE_ORIENTATION_BL: //0th row == visual bottom, 0th column == visual left-hand side
396 flip(img, img, 0); //flip vertically
397 break;
398 case IMAGE_ORIENTATION_LT: //0th row == visual left-hand side, 0th column == visual top
399 transpose(img, img);
400 break;
401 case IMAGE_ORIENTATION_RT: //0th row == visual right-hand side, 0th column == visual top
402 transpose(img, img);
403 flip(img, img, 1); //flip horizontally
404 break;
405 case IMAGE_ORIENTATION_RB: //0th row == visual right-hand side, 0th column == visual bottom
406 transpose(img, img);
407 flip(img, img, -1); //flip both horizontally and vertically
408 break;
409 case IMAGE_ORIENTATION_LB: //0th row == visual left-hand side, 0th column == visual bottom
410 transpose(img, img);
411 flip(img, img, 0); //flip vertically
412 break;
413 default:
414 //by default the image read has normal (JPEG_ORIENTATION_TL) orientation
415 break;
416 }
417}

419static void ApplyExifOrientation(ExifEntry_t orientationTag, OutputArray img)
420{
421 int orientation = IMAGE_ORIENTATION_TL;
422
423 if (orientationTag.tag != INVALID_TAG)
424 {
425 orientation = orientationTag.field_u16; //orientation is unsigned short, so check field_u16
426 ExifTransform(orientation, img);
427 }

430static void readMetadata(ImageDecoder& decoder,
431 std::vector<int>* metadata_types,
432 OutputArrayOfArrays metadata)
433{
434 if (!metadata_types)
435 return;
436 int kind = metadata.kind();
437 void* obj = metadata.getObj();
438 std::vector<Mat>* matvector = nullptr;
439 std::vector<std::vector<uchar> >* vecvector = nullptr;
440 if (kind == _InputArray::STD_VECTOR_MAT) {
441 matvector = (std::vector<Mat>*)obj;
442 } else if (kind == _InputArray::STD_VECTOR_VECTOR) {
443 int elemtype = metadata.type(0);
444 CV_Assert(elemtype == CV_8UC1 || elemtype == CV_8SC1);
445 vecvector = (std::vector<std::vector<uint8_t> >*)obj;
446 } else {
447 CV_Error(Error::StsBadArg,
448 "unsupported metadata type, should be a vector of matrices or vector of byte vectors");
449 }
450 std::vector<Mat> src_metadata;
451 for (int m = (int)IMAGE_METADATA_EXIF; m <= (int)IMAGE_METADATA_MAX; m++) {
452 Mat mm = decoder->getMetadata((ImageMetadataType)m);
453 if (!mm.empty()) {
454 CV_Assert(mm.isContinuous());
455 CV_Assert(mm.elemSize() == 1u);
456 metadata_types->push_back(m);
457 src_metadata.push_back(mm);
458 }
459 }
460 size_t nmetadata = metadata_types->size();
461 if (matvector) {
462 matvector->resize(nmetadata);
463 for (size_t m = 0; m < nmetadata; m++)
464 src_metadata[m].copyTo(matvector->at(m));
465 } else {
466 vecvector->resize(nmetadata);
467 for (size_t m = 0; m < nmetadata; m++) {
468 const Mat& mm = src_metadata[m];
469 const uchar* data = (uchar*)mm.data;
470 vecvector->at(m).assign(data, data + mm.total());
471 }
472 }
473}

509static bool
510imread_( const String& filename, int flags, OutputArray mat,
511 std::vector<int>* metadata_types, OutputArrayOfArrays metadata)
512{
514 ImageDecoder decoder;
515
516#ifdef HAVE_GDAL
517 if(flags != IMREAD_UNCHANGED && (flags & IMREAD_LOAD_GDAL) == IMREAD_LOAD_GDAL ){
518 decoder = GdalDecoder().newDecoder();
519 }else{
520#endif
521 decoder = findDecoder( filename );
522#ifdef HAVE_GDAL
523 }
524#endif
525
527 if( !decoder ){
528 return 0;
529 }
530
531 int scale_denom = 1;
532 if( flags > IMREAD_LOAD_GDAL )
533 {
534 if( flags & IMREAD_REDUCED_GRAYSCALE_2 )
535 scale_denom = 2;
536 else if( flags & IMREAD_REDUCED_GRAYSCALE_4 )
537 scale_denom = 4;
538 else if( flags & IMREAD_REDUCED_GRAYSCALE_8 )
539 scale_denom = 8;
540 }
541
542 // Try to decode image by RGB instead of BGR.
543 if (flags & IMREAD_COLOR_RGB && flags != IMREAD_UNCHANGED)
544 {
545 decoder->setRGB(true);
546 }
547
549 decoder->setScale( scale_denom );
550
552 decoder->setSource( filename );
553
554 if (metadata_types)
555 {
556 metadata_types->clear();
557 decoder->setReadOptions(1);
558 }
559
560 try
561 {
562 // read the header to make sure it succeeds
563 if( !decoder->readHeader() )
564 return 0;
565 }
566 catch (const cv::Exception& e)
567 {
568 CV_LOG_ERROR(NULL, "imread_('" << filename << "'): can't read header: " << e.what());
569 return 0;
570 }
571 catch (...)
572 {
573 CV_LOG_ERROR(NULL, "imread_('" << filename << "'): can't read header: unknown exception");
574 return 0;
575 }
576
577
578 // established the required input image size
579 Size size = validateInputImageSize(Size(decoder->width(), decoder->height()));
580
581 // grab the decoded type
582 const int type = calcType(decoder->type(), flags);
583
584 if (mat.empty())
585 {
586 mat.create( size.height, size.width, type );
587 }
588 else
589 {
590 CV_CheckEQ(size, mat.size(), "");
591 CV_CheckTypeEQ(type, mat.type(), "");
592 CV_Assert(mat.isContinuous());
593 }
594
595 // read the image data
596 Mat real_mat = mat.getMat();
597 const void * original_ptr = real_mat.data;
598 bool success = false;
599 decoder->resetFrameCount(); // this is needed for PngDecoder. it should be called before decoder->readData()
600 try
601 {
602 if (decoder->readData(real_mat))
603 {
604 CV_CheckTrue(original_ptr == real_mat.data, "Internal imread issue");
605 success = true;
606 }
607
608 readMetadata(decoder, metadata_types, metadata);
609 }
610 catch (const cv::Exception& e)
611 {
612 CV_LOG_ERROR(NULL, "imread_('" << filename << "'): can't read data: " << e.what());
613 }
614 catch (...)
615 {
616 CV_LOG_ERROR(NULL, "imread_('" << filename << "'): can't read data: unknown exception");
617 }
618 if (!success)
619 {
620 mat.release();
621 return false;
622 }
623
624 if( decoder->setScale( scale_denom ) > 1 ) // if decoder is JpegDecoder then decoder->setScale always returns 1
625 {
626 resize( mat, mat, Size( size.width / scale_denom, size.height / scale_denom ), 0, 0, INTER_LINEAR_EXACT);
627 }
628
630 if (!mat.empty() && (flags & IMREAD_IGNORE_ORIENTATION) == 0 && flags != IMREAD_UNCHANGED )
631 {
632 ApplyExifOrientation(decoder->getExifTag(ORIENTATION), mat);
633 }
634
635 return true;
636}

639static bool
640imreadmulti_(const String& filename, int flags, std::vector<Mat>& mats, int start, int count)
641{
643 ImageDecoder decoder;
644
645 CV_CheckGE(start, 0, "Start index cannont be < 0");
646
647#ifdef HAVE_GDAL
648 if (flags != IMREAD_UNCHANGED && (flags & IMREAD_LOAD_GDAL) == IMREAD_LOAD_GDAL) {
649 decoder = GdalDecoder().newDecoder();
650 }
651 else {
652#endif
653 decoder = findDecoder(filename);
654#ifdef HAVE_GDAL
655 }
656#endif
657
659 if (!decoder) {
660 return 0;
661 }
662
663 if (count < 0) {
664 count = std::numeric_limits<int>::max();
665 }
666
667 if (flags & IMREAD_COLOR_RGB && flags != IMREAD_UNCHANGED)
668 decoder->setRGB(true);
669
671 decoder->setSource(filename);
672
673 // read the header to make sure it succeeds
674 try
675 {
676 // read the header to make sure it succeeds
677 if (!decoder->readHeader())
678 return 0;
679 }
680 catch (const cv::Exception& e)
681 {
682 CV_LOG_ERROR(NULL, "imreadmulti_('" << filename << "'): can't read header: " << e.what());
683 return 0;
684 }
685 catch (...)
686 {
687 CV_LOG_ERROR(NULL, "imreadmulti_('" << filename << "'): can't read header: unknown exception");
688 return 0;
689 }
690
691 int current = start;
692
693 while (current > 0)
694 {
695 if (!decoder->nextPage())
696 {
697 return false;
698 }
699 --current;
700 }
701
702 while (current < count)
703 {
704 // grab the decoded type
705 const int type = calcType(decoder->type(), flags);
706
707 // established the required input image size
708 Size size = validateInputImageSize(Size(decoder->width(), decoder->height()));
709
710 // read the image data
711 Mat mat(size.height, size.width, type);
712 bool success = false;
713 try
714 {
715 if (decoder->readData(mat))
716 success = true;
717 }
718 catch (const cv::Exception& e)
719 {
720 CV_LOG_ERROR(NULL, "imreadmulti_('" << filename << "'): can't read data: " << e.what());
721 }
722 catch (...)
723 {
724 CV_LOG_ERROR(NULL, "imreadmulti_('" << filename << "'): can't read data: unknown exception");
725 }
726 if (!success)
727 break;
728
729 // optionally rotate the data if EXIF' orientation flag says so
730 if ((flags & IMREAD_IGNORE_ORIENTATION) == 0 && flags != IMREAD_UNCHANGED)
731 {
732 ApplyExifOrientation(decoder->getExifTag(ORIENTATION), mat);
733 }
734
735 mats.push_back(mat);
736 if (!decoder->nextPage())
737 {
738 break;
739 }
740 ++current;
741 }
742
743 return !mats.empty();
744}

1072static bool imwrite_( const String& filename, const std::vector<Mat>& img_vec,
1073 const std::vector<int>& metadata_types,
1074 InputArrayOfArrays metadata,
1075 const std::vector<int>& params, bool flipv )
1076{
1077 bool isMultiImg = img_vec.size() > 1;
1078 std::vector<Mat> write_vec;
1079
1080 ImageEncoder encoder = findEncoder( filename );
1081 if( !encoder )
1082 CV_Error( Error::StsError, "could not find a writer for the specified extension" );
1083
1084 for (size_t page = 0; page < img_vec.size(); page++)
1085 {
1086 Mat image = img_vec[page];
1087 CV_Assert(!image.empty());
1088
1089#ifdef HAVE_OPENEXR
1090 CV_Assert( image.channels() == 1 || image.channels() == 3 || image.channels() == 4 || encoder.dynamicCast<ExrEncoder>() );
1091#else
1092 CV_Assert( image.channels() == 1 || image.channels() == 3 || image.channels() == 4 );
1093#endif
1094
1095
1096 Mat temp;
1097 if( !encoder->isFormatSupported(image.depth()) )
1098 {
1099 CV_LOG_ONCE_WARNING(NULL, "Unsupported depth image for selected encoder is fallbacked to CV_8U.");
1100 CV_Assert( encoder->isFormatSupported(CV_8U) );
1101 fallbackToCV8U(image, temp);
1102 }
1103
1104 if( flipv )
1105 {
1106 flip(image, temp, 0);
1107 image = temp;
1108 }
1109
1110 write_vec.push_back(image);
1111 }

1113 encoder->setDestination( filename );
1114 addMetadata(encoder, metadata_types, metadata);
1115

1118
1119 for(size_t v = 0; v < params.size(); v+= 2)
1120 {
1121 const int key = params[v];
1122 if(encoder->isValidEncodeKey(key))
1123 {
1124 // Current encoder supports specified key.
1125 // Do nothing.
1126 }
1127 else if(isValidEncodeKeyAtAll(key))
1128 {
1129 // Current encoder does not support specified key, but some encoder supports it.
1130 CV_LOG_WARNING(nullptr, cv::format("An unsupported key(%d) was specified and has been ignored.", key));
1131 }
1132 else
1133 {
1134 // No encoder supports specified key.
1135 CV_LOG_WARNING(nullptr, cv::format("An unknown key(%d) was specified and has been ignored.", key));
1136 }
1137 }
1138
1139 bool code = false;
1140 try
1141 {
1142 if (!isMultiImg)
1143 code = encoder->write( write_vec[0], params );
1144 else
1145 code = encoder->writemulti( write_vec, params ); //to be implemented
1146
1147 if (!code)
1148 {
1149 FILE* f = fopen( filename.c_str(), "wb" );
1150 if ( !f )
1151 {
1152 if (errno == EACCES)
1153 {
1154 CV_LOG_WARNING(NULL, "imwrite_('" << filename << "'): can't open file for writing: permission denied");
1155 }
1156 }
1157 else
1158 {
1159 fclose(f);

1161 }
1162 }
1163 }
1164 catch (const cv::Exception& e)
1165 {
1166 CV_LOG_ERROR(NULL, "imwrite_('" << filename << "'): can't write data: " << e.what());
1167 code = false;
1168 }
1169 catch (...)
1170 {
1171 CV_LOG_ERROR(NULL, "imwrite_('" << filename << "'): can't write data: unknown exception");
1172 code = false;
1173 }

1304
1305 String filename;
1306

1308 if( !decoder )
1309 return false;

1314 if( flags & IMREAD_REDUCED_GRAYSCALE_2 )

1325 decoder->setRGB(true);

1332 {

1624bool imencodeWithMetadata( const String& ext, InputArray _img,
1625 const std::vector<int>& metadata_types,
1626 InputArrayOfArrays metadata,
1627 std::vector<uchar>& buf, const std::vector<int>& params )
1628{
1629 CV_TRACE_FUNCTION();
1630
1631 ImageEncoder encoder = findEncoder( ext );
1632 if( !encoder )
1633 CV_Error( Error::StsError, "could not find encoder for the specified extension" );
1634
1635 std::vector<Mat> img_vec;
1636 CV_Assert(!_img.empty());
1637 if (_img.isMatVector() || _img.isUMatVector())
1638 _img.getMatVector(img_vec);
1639 else
1640 img_vec.push_back(_img.getMat());
1641
1642 CV_Assert(!img_vec.empty());
1643 const bool isMultiImg = img_vec.size() > 1;
1644
1645 std::vector<Mat> write_vec;
1646 for (size_t page = 0; page < img_vec.size(); page++)
1647 {
1648 Mat image = img_vec[page];
1649 CV_Assert(!image.empty());
1650
1651 const int channels = image.channels();
1652#ifdef HAVE_OPENEXR
1653 CV_Assert( channels == 1 || channels == 3 || channels == 4 || encoder.dynamicCast<ExrEncoder>() );
1654#else
1655 CV_Assert( channels == 1 || channels == 3 || channels == 4 );
1656#endif
1657
1658 Mat temp;
1659 if( !encoder->isFormatSupported(image.depth()) )
1660 {
1661 CV_LOG_ONCE_WARNING(NULL, "Unsupported depth image for selected encoder is fallbacked to CV_8U.");
1662 CV_Assert( encoder->isFormatSupported(CV_8U) );
1663 fallbackToCV8U(image, temp);
1664 }
1665
1666 write_vec.push_back(image);
1667 }
1668
1669 CV_Check(params.size(), (params.size() & 1) == 0, "Encoding 'params' must be key-value pairs");
1670 CV_CheckLE(params.size(), (size_t)(CV_IO_MAX_IMAGE_PARAMS*2), "");
1671
1672 for(size_t v = 0; v < params.size(); v+= 2)
1673 {
1674 const int key = params[v];
1675 if(encoder->isValidEncodeKey(key))
1676 {
1677 // Current encoder supports specified key.
1678 // Do nothing.
1679 }
1680 else if(isValidEncodeKeyAtAll(key))
1681 {
1682 // Current encoder does not support specified key, but some encoder supports it.
1683 CV_LOG_WARNING(nullptr, cv::format("An unsupported key(%d) was specified and has been ignored.", key));
1684 }
1685 else
1686 {
1687 // No encoder supports specified key.
1688 CV_LOG_WARNING(nullptr, cv::format("An unknown key(%d) was specified and has been ignored.", key));
1689 }
1690 }
1691
1692 bool code = false;
1693 String filename;
1694 if( !encoder->setDestination(buf) )
1695 {
1696 filename = tempfile();
1697 code = encoder->setDestination(filename);
1698 CV_Assert( code );
1699 }
1700 addMetadata(encoder, metadata_types, metadata);
1701
1702 try {
1703 if (!isMultiImg)
1704 code = encoder->write(write_vec[0], params);
1705 else
1706 code = encoder->writemulti(write_vec, params);
1707
1708 encoder->throwOnError();
1709 CV_Assert( code );
1710 }
1711 catch (const cv::Exception& e)
1712 {
1713 CV_LOG_ERROR(NULL, "imencode(): can't encode data: " << e.what());
1714 code = false;
1715 }
1716 catch (...)
1717 {
1718 CV_LOG_ERROR(NULL, "imencode(): can't encode data: unknown exception");
1719 code = false;
1720 }
1721
1722 if( !filename.empty() && code )
1723 {
1724 FILE* f = fopen( filename.c_str(), "rb" );
1725 CV_Assert(f != 0);
1726 fseek( f, 0, SEEK_END );
1727 long pos = ftell(f);
1728 buf.resize((size_t)pos);
1729 fseek( f, 0, SEEK_SET );
1730 buf.resize(fread( &buf[0], 1, buf.size(), f ));
1731 fclose(f);
1732 remove(filename.c_str());
1733 }
1734 return code;
1735}
