61 const Ptr<ImportNodeWrapper>& node,
62 int inpId)
63{
64 CV_Assert(inpId < node->getNumInputs());
65 std::string name = node->getInputName(inpId);
66 const int numNodes = net->getNumNodes();
67 for (int i = 0; i < numNodes; ++i)
68 {
69 const int numOutputs = net->getNumOutputs(i);
70 for (int j = 0; j < numOutputs; j++)
71 {
72 if (net->getOutputName(i, j) == name)

79bool Subgraph::match(const Ptr<ImportGraphWrapper>& net, int nodeId,
80 std::vector<int>& matchedNodesIds)
81{
82 matchedNodesIds.clear();
83
84 // Collection of all matchings states across branching.
85 // If there is no commutative ops in the subgraph - there would be just a single map.
86 std::vector<std::shared_ptr<std::map<int, int>>> matchCandidates;
87 matchCandidates.push_back(makePtr<std::map<int, int>>());
88
89 struct State
90 {
91 int nodeToMatch;
92 int targetNodeId;
93 // Every state refers to current matchings pairs as well as
94 // matchings from parent branches produced by commutative ops.
95 std::vector<std::shared_ptr<std::map<int, int>>> matchings;
96
97 // When we register a matching pair we should register it in every parent branch.
98 // This is actual for branching in case of commutative ops only.
99 void addMatch(std::pair<int, int> match)
100 {
101 for (auto& m : matchings)
102 m->insert(match);
103 }
104 };
105
106 std::queue<State> states;
107 states.push({nodeId, (int)nodes.size() - 1, matchCandidates});
108
109 while (!states.empty())
110 {
111 auto state = states.front();
112 states.pop();
113 int nodeToMatch = state.nodeToMatch;
114 int targetNodeId = state.targetNodeId;
115 auto matchings = state.matchings.back();
116
117 if (matchings->find(targetNodeId) != matchings->end())
118 continue;
119
120 // Empty placeholder matches with any input type
121 if (nodes[targetNodeId].empty()) {
122 state.addMatch({targetNodeId, nodeToMatch});
123 continue;
124 }
125
126 const Ptr<ImportNodeWrapper> node = net->getNode(nodeToMatch);
127 if (node->getType() != nodes[targetNodeId])
128 continue;
129
130 std::vector<int>& inputNodes = inputs[targetNodeId];
131 if (inputNodes.size() != node->getNumInputs())
132 continue;
133
134 state.addMatch({targetNodeId, nodeToMatch});
135
136 bool isCommutative = net->isCommutativeOp(node->getType());
137 if (isCommutative)
138 {
139 if (inputNodes.size() != 2)
140 CV_Error(Error::StsNotImplemented, "Commutative op fusion with more than 2 inputs");
141
142 int inp0 = getInputNodeId(net, node, 0);
143 int inp1 = getInputNodeId(net, node, 1);
144 auto newMatchings = makePtr<std::map<int, int>>(*matchings);
145 matchCandidates.push_back(newMatchings);
146 state.matchings.push_back(newMatchings);
147 if (inp0 >= 0)
148 states.push({inp0, inputNodes[0], state.matchings});
149 if (inp1 >= 0)
150 states.push({inp1, inputNodes[1], state.matchings});
151 state.matchings.pop_back();
152
153 newMatchings = makePtr<std::map<int, int>>(*matchings);
154 matchCandidates.push_back(newMatchings);
155 state.matchings.push_back(newMatchings);
156 if (inp0 >= 0)
157 states.push({inp0, inputNodes[1], state.matchings});
158 if (inp1 >= 0)
159 states.push({inp1, inputNodes[0], state.matchings});
160 state.matchings.pop_back();
161 }
162 else
163 {
164 for (int j = 0; j < inputNodes.size(); ++j)
165 {
166 nodeId = getInputNodeId(net, node, j);
167 if (nodeId < 0)
168 continue;
169 states.push({nodeId, inputNodes[j], state.matchings});
170 }
171 }
172 }
173 for (auto& matchings : matchCandidates)
174 {
175 if (matchings->size() != nodes.size())
176 continue;
177 matchedNodesIds.resize(matchings->size());
178 for (int i = 0; i < matchings->size(); ++i)
179 {
180 CV_Assert(matchings->find(i) != matchings->end());
181 matchedNodesIds[i] = matchings->at(i);
182 }
183 return true;
184 }
185 return false;
186}

188void Subgraph::replace(const Ptr<ImportGraphWrapper>& net, const std::vector<int>& matchedNodesIds)
189{
190 // Extract names of input nodes.
191 std::vector<std::string> inputsNames(fusedNodeInputs.size());
192 for (int i = 0; i < fusedNodeInputs.size(); ++i)
193 {
194 std::string inpName;
195 // Find input node name looking at inputs of fused nodes.
196 for (int j = 0; j < matchedNodesIds.size() && inpName.empty(); ++j)
197 {
198 Ptr<ImportNodeWrapper> node = net->getNode(matchedNodesIds[j]);
199 std::vector<int>& inpIndices = inputs[j];
200
201 CV_Assert(inpIndices.empty() || node->getNumInputs() == inpIndices.size());
202 for (int k = 0; k < inpIndices.size(); ++k)
203 {
204 if (inpIndices[k] == fusedNodeInputs[i])
205 {
206 inpName = node->getInputName(k);
207 break;
208 }
209 }
210 }
211 CV_Assert(!inpName.empty());
212 inputsNames[i] = inpName;
213 }
214
215 Ptr<ImportNodeWrapper> node = net->getNode(matchedNodesIds.back());
216
217 // Modify the last node to be a fused one.
218 node->setType(fusedNodeOp);
219 node->setInputNames(inputsNames);
220
221 std::vector<Ptr<ImportNodeWrapper> > inputNodes(inputsNames.size());
222 for (int i = 0; i < inputsNames.size(); ++i)
223 {
224 inputNodes[i] = net->getNode(getInputNodeId(net, node, i));
225 }
226 finalize(net, node, inputNodes);
227}

233void simplifySubgraphs(const Ptr<ImportGraphWrapper>& net,
234 const std::vector<Ptr<Subgraph> >& patterns)
235{
236 int numNodes = net->getNumNodes();
237 std::vector<int> matchedNodesIds;
238 std::vector<int> nodesToRemove;
239 for (int j = 0; j < patterns.size(); ++j)
240 {
241 for (int i = 0; i < numNodes; ++i)
242 {
243 if (patterns[j]->match(net, i, matchedNodesIds))
244 {
245 patterns[j]->replace(net, matchedNodesIds);
246 // Remove matched nodes except the last one.
247 nodesToRemove.insert(nodesToRemove.end(), matchedNodesIds.begin(), matchedNodesIds.end() - 1);
248 }
249 }
250 }
251
252 if (nodesToRemove.empty())
253 return;
254
255 // Collect reference counts for every node
256 std::vector<int> refcounts(net->getNumNodes(), 0);
257 std::map<std::string, int> nodeIds;
258
259 // Register node outputs.
260 // Every usage of one of the node's outputs should be counted.
261 for (int nodeId = 0; nodeId < refcounts.size(); ++nodeId) {
262 for (int i = 0; i < net->getNumOutputs(nodeId); ++i) {
263 std::string name = net->getOutputName(nodeId, i);
264 nodeIds[name] = nodeId;
265 }
266 }
267
268 for (int nodeId = 0; nodeId < refcounts.size(); ++nodeId) {
269 // Increase counters for node's inputs
270 auto node = net->getNode(nodeId);
271 for (int i = 0; i < node->getNumInputs(); ++i) {
272 std::string inpName = node->getInputName(i);
273 if (inpName.empty())
274 continue;
275 auto it = nodeIds.find(inpName);
276 if (it == nodeIds.end())
277 continue;
278 refcounts[it->second] += 1;
279 }
280 }
281
282 // Remove all fused nodes. Indices expected to be in descending order.
283 std::sort(nodesToRemove.begin(), nodesToRemove.end(), [](int a, int b) { return a > b; });
284 for (int nodeId : nodesToRemove) {
285 if (refcounts[nodeId] == 0) {
286 // Decrease references to node's inputs and remove node itself
287 auto node = net->getNode(nodeId);
288 for (int i = 0; i < node->getNumInputs(); ++i) {
289 std::string inpName = node->getInputName(i);
290 auto it = nodeIds.find(inpName);
291 if (it == nodeIds.end())
292 continue;
293 refcounts[it->second] -= 1;
294 }
295 net->removeNode(nodeId);
296 refcounts[nodeId] = -1; // Same node cannot be removed twice
297 }
298 }
299}
