29void ScharrDerivInvoker_SIMD(const Mat& src, Mat& dst, const Range& range)
30{
31 typedef short deriv_type;
32 int rows = src.rows, cols = src.cols, cn = src.channels(), colsn = cols*cn;
33
34 int x, y, delta = (int)alignSize((cols + 2)*cn, 16);
35 AutoBuffer<deriv_type> _tempBuf(delta*2 + 64);
36 deriv_type *trow0 = alignPtr(_tempBuf.data() + cn, 16), *trow1 = alignPtr(trow0 + delta, 16);
37
38#if (CV_SIMD)
39 const int vlanes = VTraits<v_int16>::vlanes();
40 v_int16 c3 = vx_setall_s16(3), c10 = vx_setall_s16(10);
41#endif
42
43 for( y = range.start; y < range.end; y++ )
44 {
45 const uchar* srow0 = src.ptr<uchar>(y > 0 ? y-1 : rows > 1 ? 1 : 0);
46 const uchar* srow1 = src.ptr<uchar>(y);
47 const uchar* srow2 = src.ptr<uchar>(y < rows-1 ? y+1 : rows > 1 ? rows-2 : 0);
48 deriv_type* drow = (deriv_type *)dst.ptr<deriv_type>(y);
49
50 // do vertical convolution
51 x = 0;
52#if (CV_SIMD)
53 {
54 for( ; x <= colsn - vlanes; x += vlanes )
55 {
56 v_int16 s0 = v_reinterpret_as_s16(vx_load_expand(srow0 + x));
57 v_int16 s1 = v_reinterpret_as_s16(vx_load_expand(srow1 + x));
58 v_int16 s2 = v_reinterpret_as_s16(vx_load_expand(srow2 + x));
59
60 v_int16 t1 = v_sub(s2, s0);
61 v_int16 t0 = v_add(v_mul_wrap(v_add(s0, s2), c3), v_mul_wrap(s1, c10));
62
63 v_store(trow0 + x, t0);
64 v_store(trow1 + x, t1);
65 }
66 }
67#endif
68
69 for( ; x < colsn; x++ )
70 {
71 int t0 = (srow0[x] + srow2[x])*3 + srow1[x]*10;
72 int t1 = srow2[x] - srow0[x];
73 trow0[x] = (deriv_type)t0;
74 trow1[x] = (deriv_type)t1;
75 }
76
77 // make border
78 int x0 = (cols > 1 ? 1 : 0)*cn, x1 = (cols > 1 ? cols-2 : 0)*cn;
79 for( int k = 0; k < cn; k++ )
80 {
81 trow0[-cn + k] = trow0[x0 + k]; trow0[colsn + k] = trow0[x1 + k];
82 trow1[-cn + k] = trow1[x0 + k]; trow1[colsn + k] = trow1[x1 + k];
83 }
84
85 // do horizontal convolution, interleave the results and store them to dst
86 x = 0;
87#if (CV_SIMD)
88 {
89 for( ; x <= colsn - vlanes; x += vlanes )
90 {
91 v_int16 s0 = vx_load(trow0 + x - cn);
92 v_int16 s1 = vx_load(trow0 + x + cn);
93 v_int16 s2 = vx_load(trow1 + x - cn);
94 v_int16 s3 = vx_load(trow1 + x);
95 v_int16 s4 = vx_load(trow1 + x + cn);
96
97 v_int16 t0 = v_sub(s1, s0);
98 v_int16 t1 = v_add(v_mul_wrap(v_add(s2, s4), c3), v_mul_wrap(s3, c10));
99
100 v_store_interleave((drow + x*2), t0, t1);
101 }
102 }
103#endif
104 for( ; x < colsn; x++ )
105 {
106 deriv_type t0 = (deriv_type)(trow0[x+cn] - trow0[x-cn]);
107 deriv_type t1 = (deriv_type)((trow1[x+cn] + trow1[x-cn])*3 + trow1[x]*10);
108 drow[x*2] = t0; drow[x*2+1] = t1;
109 }
110 }
111}
