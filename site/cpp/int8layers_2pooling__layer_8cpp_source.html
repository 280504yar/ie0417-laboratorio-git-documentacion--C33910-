107 virtual bool supportBackend(int backendId) CV_OVERRIDE
108 {
109 if (backendId == DNN_BACKEND_OPENCV)
110 {
111 if (kernel_size.size() == 3)
112 return preferableTarget == DNN_TARGET_CPU;
113 if (kernel_size.size() <= 2)
114 return true;
115 else
116 return false;
117 }
118 else if (backendId == DNN_BACKEND_TIMVX && haveTimVX())
119 {
120 // Only pool 2d and pool 1d were supported.
121 if (kernel_size.size() == 3)
122 {
123 // fallback to CPU implementation.
124 preferableTarget = DNN_TARGET_CPU;
125 return false;
126 }
127 if (!avePoolPaddedArea) // TimVX does not support exclude padding.
128 return false;
129 if (globalPooling) // TODO support globalPooling in TimVX backend.
130 return false;
131 if (kernel_size.size() == 2)
132 return type == MAX || type == AVE;
133 return false;
134 }
135 else if (backendId == DNN_BACKEND_INFERENCE_ENGINE_NGRAPH)
136 {
137 return true;
138 }
139
140 return false;
141 }

154 virtual Ptr<BackendNode> initTimVX(void* timVXInfo_,
155 const std::vector<Ptr<BackendWrapper> > &inputsWrapper,
156 const std::vector<Ptr<BackendWrapper> > &outputsWrapper,
157 bool isLast) CV_OVERRIDE
158 {
159#ifdef HAVE_TIMVX
160 // tvGraph Initialization.
161 auto timVxInfo = reinterpret_cast<TimVXInfo *>(timVXInfo_);
162 CV_Assert(timVxInfo);
163 Ptr<TimVXGraph> tvGraph = timVxInfo->getGraph();
164 CV_Assert(tvGraph);
165 Ptr<tim::vx::Graph> graph = tvGraph->graph;
166
167 tim::vx::PoolType tvPoolType;
168 tim::vx::RoundType tvRoundType;
169 size_t ksize = kernel_size.size();
170 if (ksize != 2)
171 return Ptr<BackendNode>();
172
173 // type Change from OpenCV to TimVX only MAX and AVG are supported.
174 switch (type) {
175 case MAX: {
176 tvPoolType = tim::vx::PoolType::MAX;
177 break;
178 }
179 case AVE:{
180 tvPoolType = tim::vx::PoolType::AVG;
181 break;
182 }
183 default:
184 CV_Error(Error::StsNotImplemented, "Not implemented Pooling type in TimVX Backend.");
185 }
186
187 // Padding Type
188 tim::vx::PadType tvPadType;
189 if (padMode.empty())
190 {
191 tvPadType = tim::vx::PadType::AUTO; // TODO! check the padding type.
192 }
193 else if(padMode == "VALID")
194 {
195 tvPadType = tim::vx::PadType::VALID;
196 }
197 else if (padMode == "SAME")
198 {
199 tvPadType = tim::vx::PadType::SAME;
200 }
201 else
202 {
203 CV_Error(Error::StsError, "Unsupported padding mode in TimVXBackend!");
204 }
205
206 if (ceilMode)
207 tvRoundType = tim::vx::RoundType::CEILING;
208 else
209 tvRoundType = tim::vx::RoundType::FLOOR;
210
211 auto input = inputsWrapper[0];
212 std::vector<int> inputsIndex;
213 std::vector<int> outputsIndex;
214
215 // input Tensor
216 auto inputWrapper = inputsWrapper[0].dynamicCast<TimVXBackendWrapper>();
217 int input_index, output_index;
218
219 if (inputWrapper->isTensor())
220 {
221 input_index = tvGraph->getTensorIndex(inputWrapper->getTensor());
222 if (input_index == -1)
223 {
224 // Copy To New inputWrapper
225 Mat tmp = inputWrapper->getMat();
226 inputWrapper = Ptr<TimVXBackendWrapper>(new TimVXBackendWrapper(tmp));
227 }
228 }
229
230 if (!inputWrapper->isTensor())
231 {
232 Ptr<tim::vx::Quantization> tvInputQuant = Ptr<tim::vx::Quantization>(
233 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, input_sc, input_zp));
234 inputWrapper->createTensor(graph,tim::vx::TensorAttribute::INPUT, tvInputQuant);
235 input_index = tvGraph->addWrapper(inputWrapper);
236 }
237 inputsIndex.push_back(input_index);
238
239 // Output tensor
240 CV_Assert(outputsWrapper.size() == 1);
241 auto outputWrapper = outputsWrapper[0].dynamicCast<TimVXBackendWrapper>();
242 Ptr<tim::vx::Quantization> outputQuant = Ptr<tim::vx::Quantization>(
243 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, output_sc, output_zp));
244
245 if (isLast)
246 {
247 auto shapeType = getShapeTypeFromMat(outputWrapper->getMat());
248 // For Graph Output tensor, we need to set tensor shape before createTensor().
249 outputWrapper->setTensorShape(shapeType);
250 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::OUTPUT, outputQuant);
251 }
252 else
253 {
254 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::TRANSIENT, outputQuant);
255 }
256
257 output_index = tvGraph->addWrapper(outputWrapper);
258 outputsIndex.push_back(output_index);
259 std::shared_ptr<tim::vx::Operation> tvPool;
260
261 if (tvPadType == tim::vx::PadType::AUTO)
262 {
263 tvPool = graph->CreateOperation<tim::vx::ops::Pool2d>( tvPoolType,
264 std::array<uint32_t, 4>({(uint32_t) pads_begin[1], (uint32_t) pads_end[1],
265 (uint32_t) pads_begin[0], (uint32_t) pads_end[0]}),
266 std::array<uint32_t, 2>({(uint32_t)kernel_size[1], (uint32_t)kernel_size[0]}),
267 std::array<uint32_t, 2>({(uint32_t)strides[1], (uint32_t)strides[0]}),
268 tvRoundType);
269 }
270 else
271 {
272 tvPool = graph->CreateOperation<tim::vx::ops::Pool2d>(
273 tvPoolType, tvPadType,
274 std::array<uint32_t, 2>({(uint32_t)kernel_size[1], (uint32_t)kernel_size[0]}),
275 std::array<uint32_t, 2>({(uint32_t)strides[1], (uint32_t)strides[0]}),
276 tvRoundType);
277 }
278
279 Ptr<TimVXBackendNode> tvBackendNode = new TimVXBackendNode(tvGraph, tvPool, inputsIndex, outputsIndex);
280
281 return tvBackendNode;
282#endif // HAVE_TIMVX
283 return Ptr<BackendNode>();
284 }

286#ifdef HAVE_DNN_NGRAPH
287 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> > &inputs,
288 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
289 {
290 auto input = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
291
292 input = ngraphDequantize(input, input_sc, input_zp);
293
294 ov::op::PadType pad_type = ov::op::PadType::EXPLICIT;
295 if (!padMode.empty())
296 pad_type = padMode == "VALID" ? ov::op::PadType::VALID : ov::op::PadType::SAME_UPPER;
297
298 auto rounding_type = ceilMode ? ov::op::RoundingType::CEIL : ov::op::RoundingType::FLOOR;
299 ov::Output<ov::Node> pool;
300 if (type == MAX) {
301 pool = std::make_shared<ov::op::v1::MaxPool>(input, ov::Strides(strides),
302 ov::Shape(pads_begin), ov::Shape(pads_end), ov::Shape(kernel_size),
303 rounding_type, pad_type);
304 } else if (type == AVE) {
305#if defined(INF_ENGINE_RELEASE) && INF_ENGINE_VER_MAJOR_GE(2025030000)
306 std::vector<size_t> dilations(kernel_size.size(), 1);
307 pool = std::make_shared<ov::op::v16::AvgPool>(input, ov::Strides(strides), ov::Strides(dilations),
308 ov::Shape(pads_begin), ov::Shape(pads_end), ov::Shape(kernel_size),
309 !avePoolPaddedArea, rounding_type, pad_type);
310#else
311 pool = std::make_shared<ov::op::v1::AvgPool>(input, ov::Strides(strides),
312 ov::Shape(pads_begin), ov::Shape(pads_end), ov::Shape(kernel_size),
313 !avePoolPaddedArea, rounding_type, pad_type);
314#endif
315 } else if (type == SUM) {
316 ov::Shape inpShape = input.get_shape();
317 CV_Assert(inpShape.size() == 2 + kernel_size.size());
318 std::vector<int64_t> axes;
319 for (size_t i = 0; i < kernel_size.size(); i++)
320 {
321 if (inpShape[2 + i] == kernel_size[i])
322 axes.push_back(2 + i);
323 }
324 auto reduction_axes = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{axes.size()}, axes);
325 pool = std::make_shared<ov::op::v1::ReduceSum>(input, reduction_axes, true);
326 } else {
327 CV_Error(Error::StsNotImplemented, format("INT8 Pooling type: %d", type));
328 }
329
330 pool = ngraphQuantize(pool, output_sc, output_zp);
331
332 return new InfEngineNgraphNode(pool);
333 }
334#endif // HAVE_DNN_NGRAPH

336 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
337 {
338 CV_TRACE_FUNCTION();
339 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
340
341 std::vector<Mat> inputs, outputs;
342 inputs_arr.getMatVector(inputs);
343 outputs_arr.getMatVector(outputs);
344
345 switch (type)
346 {
347 case MAX:
348 {
349 CV_Assert_N(inputs.size() == 1, outputs.size() == 1);
350 maxPooling(inputs[0], outputs[0]);

359 break;
360 }

363 class PoolingInvoker : public ParallelLoopBody
364 {
365 public:
366 const Mat* src, *rois;
367 Mat *dst;
368 int pad_l, pad_t, pad_r, pad_b;
369 bool avePoolPaddedArea;
370 int nstripes, inpZp, outZp;
371 std::vector<int> ofsbuf;
372 int poolingType;
373 float inputScaleRatio;
374 float spatialScale;
375
376 std::vector<size_t> pads_begin, pads_end;
377 std::vector<size_t> kernel_size;
378 std::vector<size_t> strides;
379
380 PoolingInvoker() : src(0), rois(0), dst(0), pad_l(0), pad_t(0), pad_r(0), pad_b(0),
381 avePoolPaddedArea(false), nstripes(0), inpZp(0), outZp(0),
382 poolingType(MAX), inputScaleRatio(1.f), spatialScale(0){}
383
384 static void run(const Mat& src, const Mat& rois, Mat& dst,
385 std::vector<size_t> kernel_size, std::vector<size_t> strides,
386 std::vector<size_t> pads_begin, std::vector<size_t> pads_end,
387 bool avePoolPaddedArea, int poolingType, float spatialScale,
388 float inputScaleRatio, int inpZp, int outZp, int nstripes)
389 {
390 CV_Assert_N(
391 src.isContinuous(), dst.isContinuous(),
392 src.type() == CV_8S, src.type() == dst.type(),
393 src.dims == 3 || src.dims == 4 || src.dims == 5, dst.dims == 3 || dst.dims == 4 || dst.dims == 5,
394 src.size[0] == dst.size[0], src.size[1] == dst.size[1], rois.empty());
395
396 PoolingInvoker p;
397
398 bool isPool1D = src.dims == 3;
399 bool isPool3D = src.dims == 5;
400
401 p.src = &src;
402 p.rois = &rois;
403 p.dst = &dst;
404
405 p.kernel_size = kernel_size;
406 p.strides = strides;
407 p.pads_begin = pads_begin;
408 p.pads_end = pads_end;
409
410 p.pad_l = pads_begin.back();
411 p.pad_t = isPool1D ? 0 : pads_begin[pads_begin.size() - 2];
412 p.pad_r = pads_end.back();
413 p.pad_b = isPool1D ? 0 : pads_end[pads_end.size() - 2];
414
415 p.avePoolPaddedArea = avePoolPaddedArea;
416 p.nstripes = nstripes;
417 p.inpZp = inpZp;
418 p.outZp = outZp;
419 p.poolingType = poolingType;
420 p.spatialScale = spatialScale;
421 p.inputScaleRatio = inputScaleRatio;
422
423 int height = isPool1D ? 1 : src.size[src.dims - 2];
424 int width = src.size[src.dims - 1];
425
426 int kernel_d = isPool3D ? kernel_size[0] : 1;
427 int kernel_h = isPool1D ? 1 : kernel_size[kernel_size.size() - 2];
428 int kernel_w = kernel_size.back();
429
430 p.ofsbuf.resize(kernel_d * kernel_h * kernel_w);
431 for (int i = 0; i < kernel_d; ++i) {
432 for (int j = 0; j < kernel_h; ++j) {
433 for (int k = 0; k < kernel_w; ++k) {
434 p.ofsbuf[i * kernel_h * kernel_w + j * kernel_w + k] = width * height * i + width * j + k;
435 }
436 }
437 }
438
439 parallel_for_(Range(0, nstripes), p, nstripes);
440 }
441
442 void operator()(const Range& r) const CV_OVERRIDE
443 {
444 int channels = dst->size[1];
445
446 bool isPool3D = src->dims == 5;
447 bool isPool2D = src->dims == 4;
448 bool isPool1D = src->dims == 3;
449 int depth = isPool3D? dst->size[2] : 1;
450 int height = isPool1D? 1 : dst->size[dst->dims - 2];
451 int width = dst->size[dst->dims - 1];
452
453 int inp_depth = isPool3D? src->size[2] : 1;
454 int inp_height = isPool1D? 1 : src->size[src->dims - 2];
455 int inp_width = src->size[src->dims - 1];
456
457 size_t total = dst->total();
458 size_t stripeSize = (total + nstripes - 1)/nstripes;
459 size_t stripeStart = r.start*stripeSize;
460 size_t stripeEnd = std::min(r.end*stripeSize, total);
461
462 int kernel_d = isPool3D? kernel_size[0] : 1;
463 int kernel_h = isPool1D? 1 : kernel_size[kernel_size.size() - 2];
464 int kernel_w = kernel_size.back();
465
466 int stride_d = isPool3D? strides[0] : 0;
467 int stride_h = isPool1D? 1 :strides[strides.size() - 2];
468 int stride_w = strides.back();
469
470#if CV_SIMD128
471 const int* ofsptr = (const int*)&ofsbuf[0];
472 if (poolingType == MAX && !ofsptr)
473 CV_Error(Error::StsBadArg, "ofsbuf should be initialized in this mode");
474#endif
475
476 for( size_t ofs0 = stripeStart; ofs0 < stripeEnd; )
477 {
478 size_t ofs = ofs0;
479 int x0 = (int)(ofs % width);
480 ofs /= width;
481 int y0 = (int)(ofs % height);
482 ofs /= height;
483
484 int d0 = (int)(ofs % depth);
485 ofs /= depth;
486
487 int c = (int)(ofs % channels);
488 int n = (int)(ofs / channels);
489 int ystart, yend;
490 int dstart = 0, dend = 1;
491
492 const int8_t *srcData = 0;
493 int pad_d_begin = (pads_begin.size() == 3) ? pads_begin[0] : 0;
494 dstart = d0 * stride_d - pad_d_begin;
495 dend = min(dstart + kernel_d, (int)(inp_depth + pads_end[0]));
496
497 ystart = y0 * stride_h - pad_t;
498 yend = min(ystart + kernel_h, inp_height + pad_b);
499 srcData = src->ptr<int8_t>(n, c);
500
501 int ddelta = dend - dstart;
502 dstart = max(dstart, 0);
503 dend = min(dend, inp_depth);
504 int ydelta = yend - ystart;
505 ystart = max(ystart, 0);
506 yend = min(yend, inp_height);
507 int8_t *dstData = &dst->ptr<int8_t>(n, c, d0)[y0 * width];
508
509 int delta = std::min((int)(stripeEnd - ofs0), width - x0);
510 ofs0 += delta;
511 int x1 = x0 + delta;
512
513 if( poolingType == MAX )
514 for( ; x0 < x1; x0++ )
515 {
516 int xstart = x0 * stride_w - pad_l;
517 int xend = min(xstart + kernel_w, inp_width);
518 xstart = max(xstart, 0);
519 if (xstart >= xend || ystart >= yend)
520 {
521 dstData[x0] = (int8_t)outZp;
522 continue;
523 }
524#if CV_SIMD128
525 if( isPool2D && xstart > 0 && x0 + 15 < x1 && (x0 + 15) * stride_w - pad_l + kernel_w < inp_width )
526 {
527 v_int8x16 max_val0 = v_setall_s8(-128);
528 if( yend - ystart == kernel_h )
529 {
530 const int8_t* srcData1 = srcData + ystart*inp_width + xstart;
531 if( stride_w == 1 )
532 for (int k = 0; k < kernel_w*kernel_h; k++)
533 {
534 int index = ofsptr[k];
535 v_int8x16 v0 = v_load(srcData1 + index);
536 max_val0 = v_max(max_val0, v0);
537 }
538 else if( stride_w == 2 )
539 for (int k = 0; k < kernel_w*kernel_h; k++)
540 {
541 int index = ofsptr[k];
542 v_int8x16 v0, dummy;
543 v_load_deinterleave(srcData1 + index, v0, dummy);
544 max_val0 = v_max(max_val0, v0);
545 }
546 else
547 for (int k = 0; k < kernel_w*kernel_h; k++)
548 {
549 int index = ofsptr[k];
550 v_int8x16 v0(srcData1[index], srcData1[index + stride_w],
551 srcData1[index + stride_w*2], srcData1[index + stride_w*3],
552 srcData1[index + stride_w*4], srcData1[index + stride_w*5],
553 srcData1[index + stride_w*6], srcData1[index + stride_w*7],
554 srcData1[index + stride_w*8], srcData1[index + stride_w*9],
555 srcData1[index + stride_w*10], srcData1[index + stride_w*11],
556 srcData1[index + stride_w*12], srcData1[index + stride_w*13],
557 srcData1[index + stride_w*14], srcData1[index + stride_w*15]);
558 max_val0 = v_max(max_val0, v0);
559 }
560 }
561 else
562 {
563 for (int y = ystart; y < yend; ++y)
564 {
565 for (int x = xstart; x < xend; ++x)
566 {
567 const int index = y * inp_width + x;
568 v_int8x16 v0(srcData[index], srcData[index + stride_w],
569 srcData[index + stride_w*2], srcData[index + stride_w*3],
570 srcData[index + stride_w*4], srcData[index + stride_w*5],
571 srcData[index + stride_w*6], srcData[index + stride_w*7],
572 srcData[index + stride_w*8], srcData[index + stride_w*9],
573 srcData[index + stride_w*10], srcData[index + stride_w*11],
574 srcData[index + stride_w*12], srcData[index + stride_w*13],
575 srcData[index + stride_w*14], srcData[index + stride_w*15]);
576 max_val0 = v_max(max_val0, v0);
577 }
578 }
579 }
580 v_store(dstData + x0, max_val0);
581 x0 += 15;
582 }
583 else
584#else
585 CV_UNUSED(isPool2D);
586#endif
587 if( isPool1D )
588 {
589 const int8_t* first = srcData + xstart;
590 const int8_t* last = srcData + xend;
591 const int8_t* max_elem = std::max_element(first, last);
592 if (max_elem != last)
593 dstData[x0] = *max_elem;
594 }
595 else
596 {
597 int8_t max_val = -128;
598 for (int d = dstart; d < dend; ++d) {
599 for (int y = ystart; y < yend; ++y) {
600 for (int x = xstart; x < xend; ++x) {
601 const int index = d * inp_width * inp_height + y * inp_width + x;
602 int8_t val = srcData[index];
603 max_val = std::max(max_val, val);
604 }
605 }
606 }
607 dstData[x0] = max_val;
608 }
609 }
610 else if (poolingType == AVE || poolingType == SUM)
611 {
612 for( ; x0 < x1; ++x0)
613 {
614 int xstart = x0 * stride_w - pad_l;
615 int xend = min(xstart + kernel_w, inp_width + pad_r);
616 int xdelta = xend - xstart;
617 xstart = max(xstart, 0);
618 xend = min(xend, inp_width);
619
620 int real_kernel_area = (dend - dstart) * (yend - ystart) * (xend - xstart);
621 int padded_kernel_area = xdelta * ydelta * ddelta;
622 int kernel_area = avePoolPaddedArea ? padded_kernel_area : real_kernel_area;
623
624 int bias = (avePoolPaddedArea ? (padded_kernel_area - real_kernel_area) * inpZp : 0)
625 - (inpZp * kernel_area);
626 float inv_kernel_area = poolingType == AVE ? inputScaleRatio / kernel_area : inputScaleRatio;
627#if CV_SIMD128
628 if( isPool2D && xstart > 0 && x0 + 15 < x1 && (x0 + 15) * stride_w - pad_l + kernel_w < inp_width )
629 {
630 v_int32x4 sum_val0 = v_setall_s32(bias), sum_val1 = v_setall_s32(bias),
631 sum_val2 = v_setall_s32(bias), sum_val3 = v_setall_s32(bias),
632 voutzp = v_setall_s32(outZp);
633 v_float32x4 ikarea = v_setall_f32(inv_kernel_area);
634
635 for (int y = ystart; y < yend; ++y)
636 {
637 for (int x = xstart; x < xend; ++x)
638 {
639 const int index = y * inp_width + x;
640 v_int32x4 v0((int)srcData[index], (int)srcData[index + stride_w],
641 (int)srcData[index + stride_w*2], (int)srcData[index + stride_w*3]);
642 v_int32x4 v1((int)srcData[index + stride_w*4], (int)srcData[index + stride_w*5],
643 (int)srcData[index + stride_w*6], (int)srcData[index + stride_w*7]);
644 v_int32x4 v2((int)srcData[index + stride_w*8], (int)srcData[index + stride_w*9],
645 (int)srcData[index + stride_w*10], (int)srcData[index + stride_w*11]);
646 v_int32x4 v3((int)srcData[index + stride_w*12], (int)srcData[index + stride_w*13],
647 (int)srcData[index + stride_w*14], (int)srcData[index + stride_w*15]);
648 sum_val0 = v_add(sum_val0, v0);
649 sum_val1 = v_add(sum_val1, v1);
650 sum_val2 = v_add(sum_val2, v2);
651 sum_val3 = v_add(sum_val3, v3);
652 }
653 }
654
655 sum_val0 = v_add(v_round(v_mul(v_cvt_f32(sum_val0), ikarea)), voutzp);
656 sum_val1 = v_add(v_round(v_mul(v_cvt_f32(sum_val1), ikarea)), voutzp);
657 sum_val2 = v_add(v_round(v_mul(v_cvt_f32(sum_val2), ikarea)), voutzp);
658 sum_val3 = v_add(v_round(v_mul(v_cvt_f32(sum_val3), ikarea)), voutzp);
659
660 v_store(dstData + x0, v_pack(v_pack(sum_val0, sum_val1), v_pack(sum_val2, sum_val3)));
661 x0 += 15;
662 }
663 else
664#endif
665 if( isPool1D )
666 {
667 const int8_t* first = srcData + xstart;
668 const int8_t* last = srcData + xend;
669 int sum_val = bias + std::accumulate(first, last, 0);
670 dstData[x0] = saturate_cast<int8_t>(outZp + std::round(sum_val*inv_kernel_area));
671 }
672 else
673 {
674 int sum_val = bias;
675 for (int d = dstart; d < dend; ++d) {
676 for (int y = ystart; y < yend; ++y) {
677 for (int x = xstart; x < xend; ++x) {
678 const int index = d * inp_width * inp_height + y * inp_width + x;
679 int8_t val = srcData[index];
680 sum_val += (int)val;
681 }
682 }
683 }
684 dstData[x0] = saturate_cast<int8_t>(outZp + std::round(sum_val*inv_kernel_area));
685 }
686 }
687 }
688 }
689 }
690 };

710 bool getMemoryShapes(const std::vector<MatShape> &inputs,
711 const int requiredOutputs,
712 std::vector<MatShape> &outputs,
713 std::vector<MatShape> &internals) const CV_OVERRIDE
714 {
715 CV_Assert(inputs.size() != 0);
716
717 bool isPool1D = inputs[0].size() == 3;
718 std::vector<int> inpShape(inputs[0].begin() + 2, inputs[0].end());
719 std::vector<int> outShape(inputs[0].begin(), inputs[0].begin() + 2);
720
721 std::vector<size_t> local_kernel;
722 if (globalPooling) {
723 for (int i = 0; i < inpShape.size(); i++) {
724 size_t idx = isGlobalPooling.size() - inpShape.size() + i;
725 local_kernel.push_back(isGlobalPooling[idx] ? inpShape[i] : kernel_size[idx]);
726 }
727 } else {
728 local_kernel = kernel_size;
729 }
730
731 if (hasDynamicShapes && !shapesInitialized)
732 {
733 //Just copy input shapes for width and height to prevent errors on loading stage
734 for (int i = 0; i < inpShape.size(); i++)
735 outShape.push_back(inpShape[i]);
736 }
737 else if (padMode.empty())
738 {
739 int addedDims = isPool1D? inpShape.size() : local_kernel.size();
740 for (int i = 0; i < addedDims; i++) {
741 float dst = (float) (inpShape[i] + pads_begin[i] + pads_end[i] - local_kernel[i]) / strides[i];
742 outShape.push_back(1 + (ceilMode ? ceil(dst) : floor(dst)));
743 }
744
745 // If we have padding, ensure that the last pooling starts strictly
746 // inside the image (instead of at the padding); otherwise clip the last.
747 for (int i = 0; i < addedDims; i++) {
748 if (pads_end[i] && (outShape[2 + i] - 1) * strides[i] >= inpShape[i] + pads_end[i]) {
749 --outShape[2 + i];
750 CV_Assert((outShape[2 + i] - 1) * strides[i] < inpShape[i] + pads_end[i]);
751 }
752 }
753 }
754 else {
755 getConvPoolOutParams(inpShape, local_kernel, strides, padMode,
756 std::vector<size_t>(local_kernel.size(), 1), outShape);
757 }
758
759 outputs.assign(1, MatShape(outShape));
760 return false;
761 }
762
763 bool updateMemoryShapes(const std::vector<MatShape> &inputs) CV_OVERRIDE
764 {
765 int dims = inputs[0].size();
766 CV_Assert(inputs[0][dims - 1] > 0 && inputs[0][dims - 2] > 0);
767 shapesInitialized = true;
768 return true;
769 }

771 virtual int64 getFLOPS(const std::vector<MatShape> &inputs,
772 const std::vector<MatShape> &outputs) const CV_OVERRIDE
773 {
774 CV_UNUSED(inputs); // suppress unused variable warning
775 long flops = 0;
776 bool isPool1D = inputs[0].size() == 3;
777 size_t karea = std::accumulate(kernel_size.begin(), isPool1D? kernel_size.begin() + 1 : kernel_size.end(),
778 1, std::multiplies<size_t>());
779 for(int i = 0; i < outputs.size(); i++)
780 {
781 if (type == MAX)
782 {
783 if (i%2 == 0)
784 flops += total(outputs[i])*karea;
785 }
786 else
787 {
788 flops += total(outputs[i])*(karea + 1);
789 }
790 }
791 return flops;
792 }

801 PSROI // Position-sensitive RoI pooling, https://arxiv.org/pdf/1605.06409.pdf
802 };
803 bool hasDynamicShapes;
804 bool shapesInitialized;
