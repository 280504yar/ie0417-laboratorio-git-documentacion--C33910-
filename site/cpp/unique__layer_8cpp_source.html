63 void forward(InputArrayOfArrays inputs_arr,
64 OutputArrayOfArrays outputs_arr,
65 OutputArrayOfArrays /*internals_arr*/) CV_OVERRIDE
66 {
67 std::vector<Mat> inputs;
68 inputs_arr.getMatVector(inputs);
69 CV_Assert(inputs.size() == 1);
70
71 const Mat& Xin = inputs[0];
72 CV_Assert(Xin.dims >= 1);
73
74 int ax = hasAxis_ ? normalize_axis(axis_, Xin.dims) : -1;
75
76 auto kind = outputs_arr.kind();
77 if (kind == _InputArray::STD_VECTOR_MAT)
78 {
79 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
80 CV_Assert(!outs.empty());
81 dispatchByDepth(Xin, outs, ax, sorted_);
82 }
83 else if (kind == _InputArray::STD_VECTOR_UMAT)
84 {
85 std::vector<UMat>& uouts = outputs_arr.getUMatVecRef();
86 CV_Assert(!uouts.empty());
87 std::vector<Mat> tmp(uouts.size());
88 dispatchByDepth(Xin, tmp, ax, sorted_);
89 for (size_t i = 0; i < uouts.size(); ++i)
90 {
91 if (tmp[i].empty())
92 continue;
93 MatShape sh = shape(tmp[i]);
94 uouts[i].fit(sh, tmp[i].type());
95 tmp[i].copyTo(uouts[i]);
96 }
97 }
98 else
99 {
100 CV_Error(cv::Error::StsBadArg, cv::format("Unsupported output array kind: %d", kind));
101 }
102 }

105 template<typename Key, typename LessKey, typename EqualKey>
106 void sortAndGroupKeys(std::vector<std::pair<int, Key>>& keyed,
107 const LessKey& lessKey,
108 const EqualKey& equalKey,
109 std::vector<int>& groupRepIndex,
110 std::vector<int>& firstJ,
111 std::vector<int64>& counts,
112 std::vector<int>& inv) const
113 {
114 std::sort(keyed.begin(), keyed.end(), [&](const std::pair<int, Key>& a, const std::pair<int, Key>& b){
115 if (lessKey(a.second, b.second)) return true;
116 if (lessKey(b.second, a.second)) return false;
117 return a.first < b.first;
118 });
119
120 int g = 0;
121 const int A = (int)keyed.size();
122 for (int pos = 0; pos < A; )
123 {
124 int start = pos;
125 int minJ = keyed[pos].first;
126 int64 cnt = 0;
127 while (pos < A && equalKey(keyed[start].second, keyed[pos].second)) {
128 inv[ keyed[pos].first ] = g;
129 minJ = std::min(minJ, keyed[pos].first);
130 ++cnt; ++pos;
131 }
132 groupRepIndex.push_back(keyed[start].first);
133 firstJ.push_back(minJ);
134 counts.push_back(cnt);
135 ++g;
136 }
137 }

139 template<typename T, typename WT = T>
140 void uniqueAxisImpl(const Mat& X, std::vector<Mat>& outs, int ax, bool sorted) const
141 {
142 const int r = X.dims;
143
144 MatShape inShape = shape(X);
145 int dimAxis;
146 size_t outer = 1, inner = 1;
147 if (ax < 0) {
148 dimAxis = (int)X.total();
149 } else {
150 dimAxis = inShape[ax];
151 outer = std::accumulate(inShape.begin(), inShape.begin() + ax, (size_t)1, std::multiplies<int>());
152 inner = std::accumulate(inShape.begin() + ax + 1, inShape.end(), (size_t)1, std::multiplies<int>());
153 }
154
155 const int A = dimAxis;
156 const int64 sliceElems = (ax < 0) ? 1 : (int64)outer * (int64)inner;
157
158 const T* inPtr = X.ptr<const T>();
159
160 std::vector<int> groupRepIndex; groupRepIndex.reserve(A);
161 std::vector<int> firstJ; firstJ.reserve(A);
162 std::vector<int64> counts; counts.reserve(A);
163 std::vector<int> inv(A, -1);
164
165 if (ax < 0)
166 {
167 using Pair = std::pair<int, WT>; // (original index, value)
168 std::vector<Pair> keyed(A);
169 for (int u = 0; u < A; ++u)
170 keyed[u] = {u, static_cast<WT>(inPtr[(size_t)u])};
171
172 auto lessPair = [](const WT& a, const WT& b){ return a < b; };
173 auto equalPair = [](const WT& a, const WT& b){ return a == b; };
174 sortAndGroupKeys(keyed, lessPair, equalPair, groupRepIndex, firstJ, counts, inv);
175 }
176 else
177 {
178 const size_t S = (size_t)sliceElems;
179 std::vector<WT> flat((size_t)A * S);
180 for (int u = 0; u < A; ++u)
181 {
182 size_t wrote = 0;
183 for (size_t ob = 0; ob < outer; ++ob)
184 {
185 for (size_t ij = 0; ij < inner; ++ij)
186 {
187 size_t srcOff = ob * (size_t)dimAxis * inner + (size_t)u * inner + ij;
188 flat[(size_t)u * S + wrote] = static_cast<WT>(inPtr[srcOff]);
189 ++wrote;
190 }
191 }
192 }
193
194 using Pair = std::pair<int, size_t>; // (original index, offset into flat)
195 std::vector<Pair> keyed(A);
196 for (int u = 0; u < A; ++u)
197 keyed[u] = {u, (size_t)u * S};
198
199 auto lessLex = [&](size_t aoff, size_t boff){
200 const WT* pa = &flat[aoff];
201 const WT* pb = &flat[boff];
202 for (size_t t = 0; t < S; ++t) {
203 if (pa[t] < pb[t]) return true;
204 if (pa[t] > pb[t]) return false;
205 }
206 return false;
207 };
208 auto equalLex = [&](size_t aoff, size_t boff){
209 const WT* pa = &flat[aoff];
210 const WT* pb = &flat[boff];
211 for (size_t t = 0; t < S; ++t) {
212 if (pa[t] != pb[t]) return false;
213 }
214 return true;
215 };
216
217 sortAndGroupKeys(keyed, lessLex, equalLex, groupRepIndex, firstJ, counts, inv);
218 }
219
220 std::vector<int> order((int)groupRepIndex.size());
221 std::iota(order.begin(), order.end(), 0);
222 if (!sorted) {
223 auto firstOccurLess = [&](int ga, int gb){ return firstJ[ga] < firstJ[gb]; };
224 std::sort(order.begin(), order.end(), firstOccurLess);
225 }
226
227 std::vector<int> remap(order.size());
228 for (int newi = 0; newi < (int)order.size(); ++newi)
229 remap[ order[newi] ] = newi;
230
231 if (ax < 0)
232 {
233 MatShape yshape(1); yshape[0] = (int)order.size();
234 outs[0].fit(yshape, X.type());
235 T* yp = outs[0].ptr<T>();
236 for (int yi = 0; yi < (int)order.size(); ++yi)
237 {
238 int u = groupRepIndex[ order[yi] ];
239 yp[yi] = inPtr[(size_t)u];
240 }
241 }
242 else
243 {
244 std::vector<int> ysz(r);
245 for (int k = 0; k < r; ++k) ysz[k] = X.size[k];
246 ysz[ax] = (int)order.size();
247 MatShape yshape(ysz);
248 outs[0].fit(yshape, X.type());
249
250 parallel_for_(Range(0, (int)order.size()), [&](const Range& rq){
251 std::vector<int> yidx(r, 0);
252 std::vector<size_t> ystepB(r);
253 for (int k = 0; k < r; ++k) ystepB[k] = outs[0].step[k];
254
255 for (int q = rq.start; q < rq.end; ++q) {
256 int oldq = order[q];
257 int oldu = groupRepIndex[oldq];
258
259 std::fill(yidx.begin(), yidx.end(), 0);
260 int64 wrote = 0;
261 for (int64 t = 0; t < sliceElems; ++t)
262 {
263 yidx[ax] = q;
264 size_t yoff = 0;
265 for (int k = 0; k < r; ++k) yoff += (size_t)yidx[k] * ystepB[k];
266 T* yptrT = reinterpret_cast<T*>(outs[0].ptr() + yoff);
267
268 size_t ob = (size_t)(wrote / (int64)inner);
269 size_t ij = (size_t)(wrote % (int64)inner);
270 size_t srcOff = ob * (size_t)dimAxis * inner + (size_t)oldu * inner + ij;
271 const T vs = inPtr[srcOff];
272 *yptrT = vs;
273 wrote++;
274
275 for (int k = r - 1; k >= 0; --k) {
276 if (k == ax) continue;
277 if (++yidx[k] < X.size[k]) break;
278 yidx[k] = 0;
279 }
280 }
281 }
282 });
283 }
284
285 if (outs.size() > 1) {
286 MatShape ishape(1); ishape[0] = (int)order.size();
287 outs[1].fit(ishape, CV_64S);
288 auto ip = outs[1].ptr<int64_t>();
289 for (int yi = 0; yi < (int)order.size(); ++yi)
290 ip[yi] = firstJ[ order[yi] ];
291 }
292
293 if (outs.size() > 2) {
294 MatShape invshape(1); invshape[0] = A;
295 outs[2].fit(invshape, CV_64S);
296 auto invp = outs[2].ptr<int64_t>();
297 for (int j = 0; j < A; ++j)
298 invp[j] = remap[ inv[j] ];
299 }
300
301 if (outs.size() > 3) {
302 MatShape cshape(1); cshape[0] = (int)order.size();
303 outs[3].fit(cshape, CV_64S);
304 auto cp = outs[3].ptr<int64_t>();
305 for (int yi = 0; yi < (int)order.size(); ++yi)
306 cp[yi] = counts[ order[yi] ];
307 }
308 }

310 void dispatchByDepth(const Mat& X, std::vector<Mat>& outs, int ax, bool sorted) const
311 {
312 switch (X.depth())
313 {
314 case CV_8U: uniqueAxisImpl<uint8_t >(X, outs, ax, sorted); break;
315 case CV_8S: uniqueAxisImpl<int8_t >(X, outs, ax, sorted); break;
316 case CV_16U: uniqueAxisImpl<uint16_t >(X, outs, ax, sorted); break;
317 case CV_16S: uniqueAxisImpl<int16_t >(X, outs, ax, sorted); break;
318 case CV_16F: uniqueAxisImpl<hfloat, float >(X, outs, ax, sorted); break;
319 case CV_16BF: uniqueAxisImpl<bfloat, float >(X, outs, ax, sorted); break;
320 case CV_32U: uniqueAxisImpl<uint32_t >(X, outs, ax, sorted); break;
321 case CV_32S: uniqueAxisImpl<int32_t >(X, outs, ax, sorted); break;
322 case CV_32F: uniqueAxisImpl<float >(X, outs, ax, sorted); break;
323 case CV_64U: uniqueAxisImpl<uint64_t >(X, outs, ax, sorted); break;
324 case CV_64S: uniqueAxisImpl<int64_t >(X, outs, ax, sorted); break;
325 case CV_64F: uniqueAxisImpl<double >(X, outs, ax, sorted); break;
326 default:
327 CV_Error(Error::StsUnsupportedFormat, "Unsupported data type for Unique");
328 }
329 }
