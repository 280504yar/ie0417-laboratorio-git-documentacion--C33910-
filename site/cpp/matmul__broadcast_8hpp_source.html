24 class MatMulBroadcastOp final : public CUDABackendNode {
25 public:
26 using wrapper_type = GetCUDABackendWrapperType<T>;

28 MatMulBroadcastOp(csl::Stream stream_, csl::cublas::Handle handle, const Mat &B, const Mat &bias, bool _transA, bool _transB,
29 const std::vector<size_t> &A_offsets_, const std::vector<size_t> &B_offsets_, std::vector<size_t> &C_offsets_,
30 size_t batch_)
31 : stream(std::move(stream_)), cublasHandle(std::move(handle)), A_offsets(A_offsets_), B_offsets(B_offsets_), C_offsets(C_offsets_), batch(batch_)
32 {
33 if (!B.empty()) {
34 input_B_tensor = csl::makeTensorHeader<T>(B);
35 csl::copyMatToTensor<T>(B, input_B_tensor, stream);
36 }
37
38 if (!bias.empty()) {
39 bias_tensor = csl::makeTensorHeader<T>(bias);
40 csl::copyMatToTensor<T>(bias, bias_tensor, stream);
41 }
42
43 transA = _transA;
44 transB = _transB;
45 }

47 void forward(
48 const std::vector<cv::Ptr<BackendWrapper>>& inputs,
49 const std::vector<cv::Ptr<BackendWrapper>>& outputs,
50 csl::Workspace& workspace) override
51 {
52 auto input_A_wrapper = inputs[0].dynamicCast<wrapper_type>();
53 auto input_A = input_A_wrapper->getView();
54
55 csl::TensorView<T> input_B;
56 if (input_B_tensor.empty()) {
57 auto input_B_wrapper = inputs[1].dynamicCast<wrapper_type>();
58 input_B = input_B_wrapper->getView();
59 } else {
60 input_B = csl::TensorView<T>(input_B_tensor);
61 }
62
63 auto output_wrapper = outputs[0].dynamicCast<wrapper_type>();
64 auto output = output_wrapper->getSpan();
65
66 csl::tensor_ops::gemmBatched<T>(cublasHandle, batch, 0.f, output, C_offsets, 1.f, transA, input_A, A_offsets, transB, input_B, B_offsets);
67
68 // add bias if exists
69 if (!bias_tensor.empty() || inputs.size() >= 3) {
70 csl::TensorView<T> bias;
71 if (bias_tensor.empty()) {
72 auto bias_wrapper = inputs[2].dynamicCast<wrapper_type>();
73 bias = bias_wrapper->getView();
74 } else {
75 bias = csl::TensorView<T>(bias_tensor);
76 }
77
78 kernels::eltwise_sum_2<T>(stream, output, output, bias);
79 }
80 }

83 csl::Stream stream;
84 csl::cublas::Handle cublasHandle;
85 csl::Tensor<T> input_B_tensor;
86 csl::Tensor<T> bias_tensor;
87 bool transA, transB;
88
89 std::vector<size_t> A_offsets;
90 std::vector<size_t> B_offsets;
91 std::vector<size_t> C_offsets;
92 size_t batch;
93 };
