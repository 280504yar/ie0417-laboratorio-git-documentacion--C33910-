35 struct ConvolutionConfiguration {
36 /* the size of the following vectors must be equal to the kernel size */
37 std::vector<std::size_t> kernel_size;
38 std::vector<std::size_t> dilations, strides;
39
40 enum class PaddingMode {
41 MANUAL, /* uses explicit padding values provided in `pads_begin` and `pads_end` */
42 VALID, /* no padding is added */
43 SAME /* TensorFlow logic is used for same padding */
44 };
45
46 /* explicit paddings are used if and only if padMode is set to manual */
47 PaddingMode padMode;
48 std::vector<std::size_t> pads_begin, pads_end;
49
50 /* full shape inclusive of channel and batch axis */
51 std::vector<std::size_t> input_shape;
52 std::vector<std::size_t> output_shape;
53
54 /* group count for grouped convolution */
55 std::size_t groups;
56
57 enum class FusionMode {
58 NONE,
59 ACTIVATION, /* act(conv) */
60 ELTWISE_SUM, /* eltwise + conv */ /* eltwise tensor is passed as second input to forward */
61 ELTWISE_SUM_THEN_ACTIVATION, /* act(conv + eltwise) */
62 ACTIVATION_THEN_ELTWISE_SUM, /* act(conv) + eltwise */
63 };
64
65 FusionMode fusion_mode;
66
67 enum class ActivationType {
68 IDENTITY,
69 RELU, /* uses value provided in `relu_negative_slope` */
70 CLIPPED_RELU, /* uses values provided in `crelu_floor` and `crelu_ceil` */
71 POWER,
72 TANH,
73 SIGMOID,
74 SWISH,
75 MISH
76 };
77
78 ActivationType activation_type;
79 float relu_negative_slope, crelu_floor, crelu_ceil;
80 float power_exp, power_scale, power_shift;
81 };

83 template <class T>
84 class ConvolutionOp final : public CUDABackendNode {
85 public:
86 using wrapper_type = GetCUDABackendWrapperType<T>;
87
88 ConvolutionOp(csl::Stream stream_, csl::cudnn::Handle handle_, const ConvolutionConfiguration& config, const Mat& filters, const Mat& bias)
89 : stream(std::move(stream_)), cudnnHandle(std::move(handle_))
90 {
91 const auto& kernel_size = config.kernel_size;
92 const auto& dilations = config.dilations;
93 const auto& strides = config.strides;
94
95 const auto convolution_order = kernel_size.size();
96 CV_Assert(convolution_order == dilations.size());
97 CV_Assert(convolution_order == strides.size());
98
99 const auto& input_shape = config.input_shape;
100 const auto& output_shape = config.output_shape;
101 CV_Assert(input_shape.size() == output_shape.size());
102 CV_Assert(input_shape.size() == convolution_order + 2);
103
104 const auto groups = config.groups;
105
106 CV_Assert (1 <= convolution_order && convolution_order <= 3);
107
108 const auto rank = input_shape.size();
109 const auto output_feature_maps = output_shape[1];
110 const auto input_feature_maps = input_shape[1];
111 const auto input_feature_maps_per_group = input_feature_maps / groups;
112 CV_Assert(input_feature_maps % groups == 0);
113
114 filtersTensor = csl::makeTensorHeader<T>(filters);
115 csl::copyMatToTensor<T>(filters, filtersTensor, stream);
116
117 if (!bias.empty())
118 {
119 biasTensor = csl::makeTensorHeader<T>(bias);
120 csl::copyMatToTensor<T>(bias, biasTensor, stream);
121 }
122
123 /* left and right are misleading as the padding is applicable for any number of dimensions
124 * but we use those identifiers to avoid confusion with `pads_begin` and `pads_end`
125 *
126 * `common_padding` contains the amount of padding that has to be added to both sides
127 * `padding_left` and `padding_right` contains the amount of padding that needs to be added
128 * to a particular side in addition to the common padding
129 */
130 std::vector<std::size_t> common_padding(rank, 0);
131 std::vector<std::size_t> padding_left(rank, 0), padding_right(rank, 0);
132 if (config.padMode == ConvolutionConfiguration::PaddingMode::MANUAL)
133 {
134 const auto& pads_begin = config.pads_begin;
135 const auto& pads_end = config.pads_end;
136
137 CV_Assert(convolution_order == pads_begin.size());
138 CV_Assert(convolution_order == pads_end.size());
139
140 for (int i = 2; i < common_padding.size(); i++)
141 {
142 common_padding[i] = std::min(pads_begin[i - 2], pads_end[i - 2]);
143 padding_left[i] = pads_begin[i - 2] - common_padding[i];
144 padding_right[i] = pads_end[i - 2] - common_padding[i];
145 }
146 }
147 else if (config.padMode == ConvolutionConfiguration::PaddingMode::VALID)
148 {
149 /* nothing to do as the paddings are already preset to zero */
150 }
151 else if (config.padMode == ConvolutionConfiguration::PaddingMode::SAME)
152 {
153 /* TensorFlow Logic:
154 * total_padding[i] = (o[i] - 1) * s[i] + effective_k[i] - i[i]
155 *
156 * if total padding is odd, the extra is added towards the end
157 */
158 for (int i = 2; i < rank; i++)
159 {
160 const auto j = i - 2; /* filter index */
161 const auto effective_kernel_size = dilations[j] * (kernel_size[j] - 1) + 1;
162 const auto required_total_padding =
163 std::max<std::int64_t>(0, (output_shape[i] - 1) * strides[j] + effective_kernel_size - input_shape[i]);
164
165 common_padding[i] = required_total_padding / 2;
166 padding_left[i] = 0;
167 padding_right[i] = required_total_padding % 2;
168 }
169 }
170
171 /* in some scenarios, the extra padding at the end may not change the output at all */
172 for (int i = 2; i < rank; i++) {
173 const auto j = i - 2; /* filter idx */
174 const auto total_padding = common_padding[i] * 2 + padding_left[i] + padding_right[i];
175 const auto effective_kernel_size = dilations[j] * (kernel_size[j] - 1) + 1;
176 std::int64_t rem = (input_shape[i] + total_padding - effective_kernel_size) % strides[j];
177
178 /* the output shape doesn't change if we decrease the total padding by at most `rem`
179 * provided that we decrease from the right
180 */
181 if (rem && padding_right[i] > 0)
182 padding_right[i] = std::max<std::int64_t>(0, padding_right[i] - rem);
183 }
184
185 auto is_not_zero = [](std::size_t i) { return i != 0; };
186 if(std::any_of(std::begin(padding_left), std::end(padding_left), is_not_zero) ||
187 std::any_of(std::begin(padding_right), std::end(padding_right), is_not_zero))
188 {
189 /* csl::Convolution supports symmetric padding only; hence, we deal with asymmetric padding by
190 * copying the input to a bigger tensor and padding the ends manually
191 */
192 transformed_shape = input_shape;
193 for (int i = 0; i < rank; i++)
194 transformed_shape[i] += padding_left[i] + padding_right[i];
195
196 inputTransformer = csl::TensorTransform<T>(cudnnHandle, padding_left, padding_right);
197 }
198
199 typename csl::Convolution<T>::params_type params;
200 if (transformed_shape.empty())
201 {
202 params.input_shape.assign(std::begin(input_shape), std::end(input_shape));
203 }
204 else
205 {
206 /* the convolution operation will be seeing the transformed input */
207 params.input_shape.assign(std::begin(transformed_shape), std::end(transformed_shape));
208 }
209
210 auto& fshape = params.filter_shape;
211 fshape.resize(rank);
212 fshape[0] = output_feature_maps;
213 fshape[1] = input_feature_maps_per_group;
214 std::copy(std::begin(kernel_size), std::end(kernel_size), std::begin(fshape) + 2);
215 CV_Assert(fshape.size() == kernel_size.size() + 2);
216
217 params.padding.assign(std::begin(common_padding) + 2, std::end(common_padding));
218 params.stride = strides;
219 params.dilation = dilations;
220 params.groups = config.groups;
221
222 fusion_mode = config.fusion_mode;
223 activation = config.activation_type;
224 relu_negative_slope = config.relu_negative_slope;
225 crelu_floor = config.crelu_floor;
226 crelu_ceil = config.crelu_ceil;
227 power_exp = config.power_exp;
228 power_scale = config.power_scale;
229 power_shift = config.power_shift;
230
231 /* we normally use cuDNN for convolution and perform bias, activation and eltwise ops ourselves
232 * hence, the activation for cuDNN is IDENTITY by default
233 */
234 fusion_location = InternalFusionLocation::NATIVE; /* i.e. we perform bias, act and eltwise */
235 params.eltwise = false;
236 params.activation_type = csl::Convolution<T>::ActivationType::IDENTITY;
237
238 /* cuDNN can fuse the operations with convolution in some cases; try if it's possible */
239 if (!biasTensor.empty() && 0 &&
240 biasTensor.size() == output_feature_maps && /* cuDNN requirement */
241 activation == ConvolutionConfiguration::ActivationType::RELU && /* cuDNN requirement */
242 relu_negative_slope == 0.0 && /* cuDNN requirement */
243 (fusion_mode == ConvolutionConfiguration::FusionMode::ACTIVATION || /* act(conv + bias) */
244 fusion_mode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM_THEN_ACTIVATION) /* act(conv + bias + eltwise) */
245 )
246 {
247 bool do_not_fuse = false;
248 if(std::is_same<T, half>::value)
249 {
250 /* performance degrades if fused with tensor core based convolutions in most cases */
251 int device;
252 CUDA4DNN_CHECK_CUDA(cudaGetDevice(&device));
253
254 int cc_major;
255 CUDA4DNN_CHECK_CUDA(cudaDeviceGetAttribute(&cc_major, cudaDevAttrComputeCapabilityMajor, device));
256
257 if (cc_major >= 7)
258 do_not_fuse = true;
259 }
260
261 if (!do_not_fuse)
262 {
263 fusion_location = InternalFusionLocation::CUDNN;
264 auto bias_shape = std::vector<std::size_t>(rank, 1);
265 bias_shape[1] = output_feature_maps;
266 params.bias_shape = bias_shape;
267
268 if (config.fusion_mode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM_THEN_ACTIVATION)
269 params.eltwise = true;
270
271 params.activation_type = csl::Convolution<T>::ActivationType::RELU;
272 }
273 }
274
275 convoluter = csl::Convolution<T>(cudnnHandle, params);
276
277 csl::WorkspaceBuilder builder;
278 if (!transformed_shape.empty())
279 {
280 auto& shape = transformed_shape;
281 auto sz = std::accumulate(std::begin(shape), std::end(shape), 1, std::multiplies<std::size_t>());
282 builder.require<T>(sz);
283 }
284 builder.require(convoluter.get_workspace_size());
285 scratch_mem_in_bytes = builder.required_workspace_size();
286 }
287
288 void forward(
289 const std::vector<cuda::GpuMatND>& inputs,
290 const std::vector<cuda::GpuMatND>& outputs,
291 csl::Workspace& workspace) override
292 {
293 /* input[0] = conv input, input[1] = bias (from fused eltwise layer) */
294 CV_Assert(inputs.size() == 1 || inputs.size() == 2);
295 CV_Assert(outputs.size() == 1);
296
297 csl::WorkspaceAllocator allocator(workspace);
298
299 auto input = csl::viewOf<T>(inputs[0]);
300
301 if (!transformed_shape.empty())
302 {
303 auto& shape = transformed_shape;
304 auto transformed_input = allocator.get_tensor_span<T>(std::begin(shape), std::end(shape));
305 inputTransformer.transform(input, transformed_input);
306 input = transformed_input;
307 }
308
309 auto conv_scratchpad = allocator.get_instance();
310
311 auto output = csl::spanOf<T>(outputs[0]);
312
313 if (fusion_location == InternalFusionLocation::CUDNN)
314 {
315 try
316 {
317 if (fusion_mode == ConvolutionConfiguration::FusionMode::ACTIVATION)
318 convoluter.convolve_with_bias_activation(output, input, filtersTensor, biasTensor, conv_scratchpad);
319 else if (fusion_mode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM_THEN_ACTIVATION)
320 {
321 auto eltwise = csl::viewOf<T>(inputs[1]);
322 CV_Assert(is_shape_same(eltwise, output));
323
324 convoluter.convolve_with_bias_eltwise_activation(output, input, filtersTensor, biasTensor, eltwise, conv_scratchpad);
325 }
326 }
327 catch(const csl::cudnn::cuDNNException& ex)
328 {
329 if (ex.getCUDNNStatus() == CUDNN_STATUS_NOT_SUPPORTED)
330 {
331 /* drop cuDNN fusion and use the native fusion path */
332 fusion_location = InternalFusionLocation::NATIVE;
333 }
334 else
335 throw;
336 }
337 }
338
339 if (fusion_location == InternalFusionLocation::NATIVE)
340 {
341 convoluter.convolve(output, input, filtersTensor, conv_scratchpad);
342
343 if (fusion_mode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM ||
344 fusion_mode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM_THEN_ACTIVATION ||
345 fusion_mode == ConvolutionConfiguration::FusionMode::ACTIVATION_THEN_ELTWISE_SUM)
346 {
347 CV_Assert(inputs.size() == 2);
348 }
349
350 if (!biasTensor.empty() && inputs.size() == 2)
351 {
352 /* bias and eltwise */
353 CV_Assert(fusion_mode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM ||
354 fusion_mode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM_THEN_ACTIVATION ||
355 fusion_mode == ConvolutionConfiguration::FusionMode::ACTIVATION_THEN_ELTWISE_SUM);
356
357 auto eltwise = csl::viewOf<T>(inputs[1]);
358 CV_Assert(is_shape_same(eltwise, output));
359
360 std::size_t inner_size = output.size_range(2, output.rank());
361
362 if (fusion_mode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM)
363 {
364 kernels::biasN_eltwise_sum_2_identity_inplace<T>(stream, output, inner_size, biasTensor, eltwise);
365 }
366 else if (fusion_mode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM_THEN_ACTIVATION)
367 {
368 /* activation(conv + bias + eltwise) */
369 switch (activation)
370 {
371 case ConvolutionConfiguration::ActivationType::IDENTITY:
372 kernels::biasN_eltwise_sum_2_identity_inplace<T>(stream, output, inner_size, biasTensor, eltwise);
373 break;
374 case ConvolutionConfiguration::ActivationType::RELU:
375 kernels::biasN_eltwise_sum_2_relu_inplace<T>(stream, output, inner_size, biasTensor, eltwise, relu_negative_slope);
376 break;
377 case ConvolutionConfiguration::ActivationType::CLIPPED_RELU:
378 kernels::biasN_eltwise_sum_2_clipped_relu_inplace<T>(stream, output, inner_size, biasTensor, eltwise, crelu_floor, crelu_ceil);
379 break;
380 case ConvolutionConfiguration::ActivationType::POWER:
381 kernels::biasN_eltwise_sum_2_power_inplace<T>(stream, output, inner_size, biasTensor, eltwise, power_exp, power_scale, power_shift);
382 break;
383 case ConvolutionConfiguration::ActivationType::TANH:
384 kernels::biasN_eltwise_sum_2_tanh_inplace<T>(stream, output, inner_size, biasTensor, eltwise);
385 break;
386 case ConvolutionConfiguration::ActivationType::SIGMOID:
387 kernels::biasN_eltwise_sum_2_sigmoid_inplace<T>(stream, output, inner_size, biasTensor, eltwise);
388 break;
389 case ConvolutionConfiguration::ActivationType::SWISH:
390 kernels::biasN_eltwise_sum_2_swish_inplace<T>(stream, output, inner_size, biasTensor, eltwise);
391 break;
392 case ConvolutionConfiguration::ActivationType::MISH:
393 kernels::biasN_eltwise_sum_2_mish_inplace<T>(stream, output, inner_size, biasTensor, eltwise);
394 break;
395 }
396 }
397 else if (fusion_mode == ConvolutionConfiguration::FusionMode::ACTIVATION_THEN_ELTWISE_SUM)
398 {
399 /* activation(conv + bias) + eltwise */
400 switch (activation)
401 {
402 case ConvolutionConfiguration::ActivationType::IDENTITY:
403 kernels::biasN_eltwise_sum_2_identity_inplace<T>(stream, output, inner_size, biasTensor, eltwise);
404 break;
405 case ConvolutionConfiguration::ActivationType::RELU:
406 kernels::biasN_relu_eltwise_sum_2_inplace<T>(stream, output, inner_size, biasTensor, eltwise, relu_negative_slope);
407 break;
408 case ConvolutionConfiguration::ActivationType::CLIPPED_RELU:
409 kernels::biasN_clipped_relu_eltwise_sum_2_inplace<T>(stream, output, inner_size, biasTensor, eltwise, crelu_floor, crelu_ceil);
410 break;
411 case ConvolutionConfiguration::ActivationType::POWER:
412 kernels::biasN_power_eltwise_sum_2_inplace<T>(stream, output, inner_size, biasTensor, eltwise, power_exp, power_scale, power_shift);
413 break;
414 case ConvolutionConfiguration::ActivationType::TANH:
415 kernels::biasN_tanh_eltwise_sum_2_inplace<T>(stream, output, inner_size, biasTensor, eltwise);
416 break;
417 case ConvolutionConfiguration::ActivationType::SIGMOID:
418 kernels::biasN_sigmoid_eltwise_sum_2_inplace<T>(stream, output, inner_size, biasTensor, eltwise);
419 break;
420 case ConvolutionConfiguration::ActivationType::SWISH:
421 kernels::biasN_swish_eltwise_sum_2_inplace<T>(stream, output, inner_size, biasTensor, eltwise);
422 break;
423 case ConvolutionConfiguration::ActivationType::MISH:
424 kernels::biasN_mish_eltwise_sum_2_inplace<T>(stream, output, inner_size, biasTensor, eltwise);
425 break;
426 }
427 }
428 }
429 else if (!biasTensor.empty() && inputs.size() == 1)
430 {
431 /* bias but no eltwise */
432 CV_Assert(fusion_mode == ConvolutionConfiguration::FusionMode::NONE ||
433 fusion_mode == ConvolutionConfiguration::FusionMode::ACTIVATION);
434
435 std::size_t inner_size = output.size_range(2, output.rank());
436 switch(activation)
437 {
438 case ConvolutionConfiguration::ActivationType::IDENTITY:
439 kernels::biasN<T>(stream, output, output, inner_size, biasTensor);
440 break;
441 case ConvolutionConfiguration::ActivationType::RELU:
442 kernels::biasN_relu_inplace<T>(stream, output, inner_size, biasTensor, relu_negative_slope);
443 break;
444 case ConvolutionConfiguration::ActivationType::CLIPPED_RELU:
445 kernels::biasN_clipped_relu_inplace<T>(stream, output, inner_size, biasTensor, crelu_floor, crelu_ceil);
446 break;
447 case ConvolutionConfiguration::ActivationType::POWER:
448 kernels::biasN_power_inplace<T>(stream, output, inner_size, biasTensor, power_exp, power_scale, power_shift);
449 break;
450 case ConvolutionConfiguration::ActivationType::TANH:
451 kernels::biasN_tanh_inplace<T>(stream, output, inner_size, biasTensor);
452 break;
453 case ConvolutionConfiguration::ActivationType::SIGMOID:
454 kernels::biasN_sigmoid_inplace<T>(stream, output, inner_size, biasTensor);
455 break;
456 case ConvolutionConfiguration::ActivationType::SWISH:
457 kernels::biasN_swish_inplace<T>(stream, output, inner_size, biasTensor);
458 break;
459 case ConvolutionConfiguration::ActivationType::MISH:
460 kernels::biasN_mish_inplace<T>(stream, output, inner_size, biasTensor);
461 break;
462 }
463 }
464 else if (biasTensor.empty() && inputs.size() == 2)
465 {
466 /* no bias but eltwise */
467 CV_Assert(fusion_mode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM ||
468 fusion_mode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM_THEN_ACTIVATION ||
469 fusion_mode == ConvolutionConfiguration::FusionMode::ACTIVATION_THEN_ELTWISE_SUM);
470
471 auto eltwise = csl::viewOf<T>(inputs[1]);
472 CV_Assert(is_shape_same(eltwise, output));
473
474 /* we pass `eltwise` as `bias` (with `inner_size` as one) to bias-activation kernels */
475
476 if (fusion_mode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM)
477 {
478 kernels::eltwise_sum_2<T>(stream, output, output, eltwise);
479 }
480 else if (fusion_mode == ConvolutionConfiguration::FusionMode::ELTWISE_SUM_THEN_ACTIVATION)
481 {
482 switch (activation)
483 {
484 case ConvolutionConfiguration::ActivationType::IDENTITY:
485 kernels::eltwise_sum_2<T>(stream, output, output, eltwise);
486 break;
487 case ConvolutionConfiguration::ActivationType::RELU:
488 kernels::eltwise_sum_2_relu<T>(stream, output, output, eltwise, relu_negative_slope);
489 break;
490 case ConvolutionConfiguration::ActivationType::CLIPPED_RELU:
491 kernels::eltwise_sum_2_clipped_relu<T>(stream, output, output, eltwise, crelu_floor, crelu_ceil);
492 break;
493 case ConvolutionConfiguration::ActivationType::POWER:
494 kernels::eltwise_sum_2_power<T>(stream, output, output, eltwise, power_exp, power_scale, power_shift);
495 break;
496 case ConvolutionConfiguration::ActivationType::TANH:
497 kernels::eltwise_sum_2_tanh<T>(stream, output, output, eltwise);
498 break;
499 case ConvolutionConfiguration::ActivationType::SIGMOID:
500 kernels::eltwise_sum_2_sigmoid<T>(stream, output, output, eltwise);
501 break;
502 case ConvolutionConfiguration::ActivationType::SWISH:
503 kernels::eltwise_sum_2_swish<T>(stream, output, output, eltwise);
504 break;
505 case ConvolutionConfiguration::ActivationType::MISH:
506 kernels::eltwise_sum_2_mish<T>(stream, output, output, eltwise);
507 break;
508 }
509 }
510 else if (fusion_mode == ConvolutionConfiguration::FusionMode::ACTIVATION_THEN_ELTWISE_SUM)
511 {
512 switch (activation)
513 {
514 case ConvolutionConfiguration::ActivationType::IDENTITY:
515 kernels::eltwise_sum_2<T>(stream, output, output, eltwise);
516 break;
517 case ConvolutionConfiguration::ActivationType::RELU:
518 kernels::relu_eltwise_sum_2_inplace<T>(stream, output, eltwise, relu_negative_slope);
519 break;
520 case ConvolutionConfiguration::ActivationType::CLIPPED_RELU:
521 kernels::clipped_relu_eltwise_sum_2_inplace<T>(stream, output, eltwise, crelu_floor, crelu_ceil);
522 break;
523 case ConvolutionConfiguration::ActivationType::POWER:
524 kernels::power_eltwise_sum_2_inplace<T>(stream, output, eltwise, power_exp, power_scale, power_shift);
525 break;
526 case ConvolutionConfiguration::ActivationType::TANH:
527 kernels::tanh_eltwise_sum_2_inplace<T>(stream, output, eltwise);
528 break;
529 case ConvolutionConfiguration::ActivationType::SIGMOID:
530 kernels::sigmoid_eltwise_sum_2_inplace<T>(stream, output, eltwise);
531 break;
532 case ConvolutionConfiguration::ActivationType::SWISH:
533 kernels::swish_eltwise_sum_2_inplace<T>(stream, output, eltwise);
534 break;
535 case ConvolutionConfiguration::ActivationType::MISH:
536 kernels::mish_eltwise_sum_2_inplace<T>(stream, output, eltwise);
537 break;
538 }
539 }
540 }
541 else if(biasTensor.empty() && inputs.size() == 1)
542 {
543 /* no bias and no eltwise */
544 CV_Assert(fusion_mode == ConvolutionConfiguration::FusionMode::NONE ||
545 fusion_mode == ConvolutionConfiguration::FusionMode::ACTIVATION);
546
547 switch(activation)
548 {
549 case ConvolutionConfiguration::ActivationType::IDENTITY:
550 break;
551 case ConvolutionConfiguration::ActivationType::RELU:
552 kernels::relu<T>(stream, output, output, relu_negative_slope);
553 break;
554 case ConvolutionConfiguration::ActivationType::CLIPPED_RELU:
555 kernels::clipped_relu<T>(stream, output, output, crelu_floor, crelu_ceil);
556 break;
557 case ConvolutionConfiguration::ActivationType::POWER:
558 kernels::power<T>(stream, output, output, power_exp, power_scale, power_shift);
559 break;
560 case ConvolutionConfiguration::ActivationType::TANH:
561 kernels::tanh<T>(stream, output, output);
562 break;
563 case ConvolutionConfiguration::ActivationType::SIGMOID:
564 kernels::sigmoid<T>(stream, output, output);
565 break;
566 case ConvolutionConfiguration::ActivationType::SWISH:
567 kernels::swish<T>(stream, output, output);
