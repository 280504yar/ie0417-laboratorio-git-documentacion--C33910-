19String dumpInputArray(InputArray argument)
20{
21 if (&argument == &noArray())
22 return "InputArray: noArray()";
23 std::ostringstream ss;
24 ss << "InputArray:";
25 try {
26 do {
27 ss << (argument.empty() ? " empty()=true" : " empty()=false");
28 ss << cv::format(" kind=0x%08llx", (long long int)argument.kind());
29 ss << cv::format(" flags=0x%08llx", (long long int)argument.getFlags());
30 if (argument.getObj() == NULL)
31 {
32 ss << " obj=NULL";
33 break; // done
34 }
35 ss << cv::format(" total(-1)=%lld", (long long int)argument.total(-1));
36 int dims = argument.dims(-1);
37 ss << cv::format(" dims(-1)=%d", dims);
38 if (dims <= 2)
39 {
40 Size size = argument.size(-1);
41 ss << cv::format(" size(-1)=%dx%d", size.width, size.height);
42 }
43 else
44 {
45 int sz[CV_MAX_DIM] = {0};
46 argument.sizend(sz, -1);
47 ss << " size(-1)=[";
48 for (int i = 0; i < dims; i++)
49 {
50 if (i > 0)
51 ss << ' ';
52 ss << sz[i];
53 }
54 ss << "]";
55 }
56 ss << " type(-1)=" << cv::typeToString(argument.type(-1));
57 } while (0);
58 }
59 catch (const std::exception& e)
60 {
61 ss << " ERROR: exception occurred: " << e.what();
62 }
63 catch (...)
64 {
65 ss << " ERROR: unknown exception occurred, dump is non-complete";
66 }
67 return ss.str();
68}

235template <class T, class Formatter>
236static inline String dumpVector(const std::vector<T>& vec, Formatter format)
237{
238 std::ostringstream oss("[", std::ios::ate);
239 if (!vec.empty())
240 {
241 format(oss) << vec[0];
242 for (std::size_t i = 1; i < vec.size(); ++i)
243 {
244 oss << ", ";
245 format(oss) << vec[i];
246 }
247 }
248 oss << "]";
249 return oss.str();
250}
