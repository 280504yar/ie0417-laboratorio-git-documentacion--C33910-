133 float logistic_activate(float x) { return 1.F / (1.F + exp(-x)); }

135 void softmax_activate(const float* input, const int n, const float temp, float* output)
136 {
137 int i;
138 float sum = 0;
139 float largest = -FLT_MAX;
140 for (i = 0; i < n; ++i) {
141 if (input[i] > largest) largest = input[i];
142 }
143 for (i = 0; i < n; ++i) {
144 float e = exp((input[i] - largest) / temp);
145 sum += e;
146 output[i] = e;
147 }
148 for (i = 0; i < n; ++i) {
149 output[i] /= sum;
150 }
151 }

153#ifdef HAVE_OPENCL
154 bool forward_ocl(InputArrayOfArrays inps, OutputArrayOfArrays outs, OutputArrayOfArrays internals)
155 {
156 if (blob_umat.empty())
157 blobs[0].copyTo(blob_umat);
158
159 std::vector<UMat> inputs;
160 std::vector<UMat> outputs;
161
162 // TODO: implement a logistic activation to classification scores.
163 if (useLogistic || inps.depth() == CV_16F)
164 return false;
165
166 inps.getUMatVector(inputs);
167 outs.getUMatVector(outputs);
168
169 CV_Assert(inputs.size() >= 1);
170 int const cell_size = classes + coords + 1;
171
172 for (size_t ii = 0; ii < outputs.size(); ii++)
173 {
174 UMat& inpBlob = inputs[ii];
175 UMat& outBlob = outputs[ii];
176
177 int batch_size = inpBlob.size[0];
178 int rows = inpBlob.size[1];
179 int cols = inpBlob.size[2];
180
181 // channels == cell_size*anchors, see l. 94
182 int sample_size = cell_size*rows*cols*anchors;
183
184 ocl::Kernel logistic_kernel("logistic_activ", ocl::dnn::region_oclsrc);
185 size_t nanchors = rows*cols*anchors*batch_size;
186 logistic_kernel.set(0, (int)nanchors);
187 logistic_kernel.set(1, ocl::KernelArg::PtrReadOnly(inpBlob));
188 logistic_kernel.set(2, (int)cell_size);
189 logistic_kernel.set(3, ocl::KernelArg::PtrWriteOnly(outBlob));
190 logistic_kernel.run(1, &nanchors, NULL, false);
191
192 if (useSoftmax)
193 {
194 // Yolo v2
195 // softmax activation for Probability, for each grid cell (X x Y x Anchor-index)
196 ocl::Kernel softmax_kernel("softmax_activ", ocl::dnn::region_oclsrc);
197 size_t nanchors = rows*cols*anchors*batch_size;
198 softmax_kernel.set(0, (int)nanchors);
199 softmax_kernel.set(1, ocl::KernelArg::PtrReadOnly(inpBlob));
200 softmax_kernel.set(2, ocl::KernelArg::PtrReadOnly(blob_umat));
201 softmax_kernel.set(3, (int)cell_size);
202 softmax_kernel.set(4, (int)classes);
203 softmax_kernel.set(5, (int)classfix);
204 softmax_kernel.set(6, (int)rows);
205 softmax_kernel.set(7, (int)cols);
206 softmax_kernel.set(8, (int)anchors);
207 softmax_kernel.set(9, (float)thresh);
208 softmax_kernel.set(10, ocl::KernelArg::PtrWriteOnly(outBlob));
209 if (!softmax_kernel.run(1, &nanchors, NULL, false))
210 return false;
211 }
212
213 if (nmsThreshold > 0) {
214 Mat mat = outBlob.getMat(ACCESS_WRITE);
215 float *dstData = mat.ptr<float>();
216 for (int b = 0; b < batch_size; ++b)
217 do_nms_sort(dstData + b*sample_size, rows*cols*anchors, thresh, nmsThreshold);
218 }
219
220 }
221
222 return true;
223 }

226 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
227 {
228 CV_TRACE_FUNCTION();
229 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
230
231 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
232 forward_ocl(inputs_arr, outputs_arr, internals_arr))
233
234 if (inputs_arr.depth() == CV_16F)
235 {
236 forward_fallback(inputs_arr, outputs_arr, internals_arr);
237 return;
238 }
239
240 std::vector<Mat> inputs, outputs, internals;
241 inputs_arr.getMatVector(inputs);
242 outputs_arr.getMatVector(outputs);
243 internals_arr.getMatVector(internals);
244
245 CV_Assert(inputs.size() >= 1);
246 CV_Assert(outputs.size() == 1);
247 int const cell_size = classes + coords + 1;
248
249 const float* biasData = blobs[0].ptr<float>();
250
251 for (size_t ii = 0; ii < outputs.size(); ii++)
252 {
253 Mat &inpBlob = inputs[ii];
254 Mat &outBlob = outputs[ii];
255
256 int batch_size = inpBlob.size[0];
257 int rows = inpBlob.size[1];
258 int cols = inpBlob.size[2];
259
260 // address length for one image in batch, both for input and output
261 int sample_size = cell_size*rows*cols*anchors;
262
263 // assert that the comment above is true
264 CV_Assert(sample_size*batch_size == inpBlob.total());
265 CV_Assert(sample_size*batch_size == outBlob.total());
266
267 CV_Assert(inputs.size() < 2 || inputs[1].dims == 4);
268 int hNorm = inputs.size() > 1 ? inputs[1].size[2] : rows;
269 int wNorm = inputs.size() > 1 ? inputs[1].size[3] : cols;
270
271 const float *srcData = inpBlob.ptr<float>();
272 float *dstData = outBlob.ptr<float>();
273
274 if (new_coords == 0) {
275 // logistic activation for t0, for each grid cell (X x Y x Anchor-index)
276 for (int i = 0; i < batch_size*rows*cols*anchors; ++i) {
277 int index = cell_size*i;
278 float x = srcData[index + 4];
279 dstData[index + 4] = logistic_activate(x); // logistic activation
280 }
281
282 if (useSoftmax) { // Yolo v2
283 Mat _inpBlob = inpBlob.reshape(0, outBlob.size);
284 softmax(outBlob, _inpBlob, -1, 5, classes);
285 }
286 else if (useLogistic) { // Yolo v3
287 for (int i = 0; i < batch_size*rows*cols*anchors; ++i){
288 int index = cell_size*i;
289 const float* input = srcData + index + 5;
290 float* output = dstData + index + 5;
291 for (int c = 0; c < classes; ++c)
292 output[c] = logistic_activate(input[c]);
293 }
294 }
295 }
296 for (int b = 0; b < batch_size; ++b)
297 for (int x = 0; x < cols; ++x)
298 for(int y = 0; y < rows; ++y)
299 for (int a = 0; a < anchors; ++a) {
300 // relative start address for image b within the batch data
301 int index_sample_offset = sample_size*b;
302 int index = (y*cols + x)*anchors + a; // index for each grid-cell & anchor
303 int p_index = index_sample_offset + index * cell_size + 4;
304 float scale = dstData[p_index];
305 if (classfix == -1 && scale < .5)
306 {
307 scale = 0; // if(t0 < 0.5) t0 = 0;
308 }
309 int box_index = index_sample_offset + index * cell_size;
310
311 if (new_coords == 1) {
312 float x_tmp = (srcData[box_index + 0] - 0.5f) * scale_x_y + 0.5f;
313 float y_tmp = (srcData[box_index + 1] - 0.5f) * scale_x_y + 0.5f;
314 dstData[box_index + 0] = (x + x_tmp) / cols;
315 dstData[box_index + 1] = (y + y_tmp) / rows;
316 dstData[box_index + 2] = (srcData[box_index + 2]) * (srcData[box_index + 2]) * 4 * biasData[2 * a] / wNorm;
317 dstData[box_index + 3] = (srcData[box_index + 3]) * (srcData[box_index + 3]) * 4 * biasData[2 * a + 1] / hNorm;
318 dstData[box_index + 4] = srcData[p_index];
319
320 scale = srcData[p_index];
321 if (classfix == -1 && scale < thresh)
322 {
323 scale = 0; // if(t0 < 0.5) t0 = 0;
324 }
325
326 int class_index = index_sample_offset + index * cell_size + 5;
327 for (int j = 0; j < classes; ++j) {
328 float prob = scale*srcData[class_index + j]; // prob = IoU(box, object) = t0 * class-probability
329 dstData[class_index + j] = (prob > thresh) ? prob : 0; // if (IoU < threshold) IoU = 0;
330 }
331 }
332 else
333 {
334 float x_tmp = (logistic_activate(srcData[box_index + 0]) - 0.5f) * scale_x_y + 0.5f;
335 float y_tmp = (logistic_activate(srcData[box_index + 1]) - 0.5f) * scale_x_y + 0.5f;
336 dstData[box_index + 0] = (x + x_tmp) / cols;
337 dstData[box_index + 1] = (y + y_tmp) / rows;
338 dstData[box_index + 2] = exp(srcData[box_index + 2]) * biasData[2 * a] / wNorm;
339 dstData[box_index + 3] = exp(srcData[box_index + 3]) * biasData[2 * a + 1] / hNorm;
340
341 int class_index = index_sample_offset + index * cell_size + 5;
342 for (int j = 0; j < classes; ++j) {
343 float prob = scale*dstData[class_index + j]; // prob = IoU(box, object) = t0 * class-probability
344 dstData[class_index + j] = (prob > thresh) ? prob : 0; // if (IoU < threshold) IoU = 0;
345 }
346 }
347 }
348 if (nmsThreshold > 0) {
349 for (int b = 0; b < batch_size; ++b){
350 do_nms_sort(dstData+b*sample_size, rows*cols*anchors, thresh, nmsThreshold);
351 }
352 }
353 }
354 }

356 void do_nms_sort(float *detections, int total, float score_thresh, float nms_thresh)
357 {
358 std::vector<Rect2d> boxes(total);
359 std::vector<float> scores(total);
360
361 for (int i = 0; i < total; ++i)
362 {
363 Rect2d &b = boxes[i];
364 int box_index = i * (classes + coords + 1);
365 b.width = detections[box_index + 2];
366 b.height = detections[box_index + 3];
367 b.x = detections[box_index + 0] - b.width / 2;
368 b.y = detections[box_index + 1] - b.height / 2;
369 }
370
371 std::vector<int> indices;
372 for (int k = 0; k < classes; ++k)
373 {
374 for (int i = 0; i < total; ++i)
375 {
376 int box_index = i * (classes + coords + 1);
377 int class_index = box_index + 5;
378 scores[i] = detections[class_index + k];
379 detections[class_index + k] = 0;
380 }
381 NMSBoxes(boxes, scores, score_thresh, nms_thresh, indices);
382 for (int i = 0, n = indices.size(); i < n; ++i)
383 {
384 int box_index = indices[i] * (classes + coords + 1);
385 int class_index = box_index + 5;
386 detections[class_index + k] = scores[indices[i]];
387 }
388 }
389 }

391#ifdef HAVE_CUDA
392 Ptr<BackendNode> initCUDA(
393 void *context_,
394 const std::vector<Ptr<BackendWrapper>>& inputs,
395 const std::vector<Ptr<BackendWrapper>>& outputs
396 ) override
397 {
398 auto context = reinterpret_cast<csl::CSLContext*>(context_);
399
400 if (coords != 4)
401 CV_Error(Error::StsNotImplemented, "Only upright rectangular boxes are supported in RegionLayer.");
402
403 std::size_t height_norm, width_norm;
404 if (inputs.size() == 1)
405 {
406 auto input_wrapper = inputs[0].dynamicCast<CUDABackendWrapper>();
407 auto input_shape = input_wrapper->getShape();
408 height_norm = input_shape[1];
409 width_norm = input_shape[2];
410 }
411 else
412 {
413 auto input_wrapper = inputs[1].dynamicCast<CUDABackendWrapper>();
414 auto input_shape = input_wrapper->getShape();
415 CV_Assert(input_shape.size() == 4);
416 height_norm = input_shape[2];
417 width_norm = input_shape[3];
418 }
419
420 cuda4dnn::SquashMethod squash_method;
421 if(useLogistic)
422 squash_method = cuda4dnn::SquashMethod::SIGMOID;
423 else if (useSoftmax)
424 squash_method = cuda4dnn::SquashMethod::SOFTMAX;
425
426 /* exactly one must be true */
427 CV_Assert((useLogistic || useSoftmax) && !(useLogistic && useSoftmax));
428
429 cuda4dnn::RegionConfiguration<float> config;
430 config.squash_method = squash_method;
431 config.classes = classes;
432 config.boxes_per_cell = anchors;
433
434 config.height_norm = height_norm;
435 config.width_norm = width_norm;
436
437 config.scale_x_y = scale_x_y;
438
439 config.object_prob_cutoff = (classfix == -1) ? thresh : 0.f;
440 config.class_prob_cutoff = thresh;
441
442 config.nms_iou_threshold = nmsThreshold;
443
444 config.new_coords = (new_coords == 1);
445 return make_cuda_node<cuda4dnn::RegionOp>(preferableTarget, std::move(context->stream), blobs[0], config);
446 }

462#ifdef HAVE_DNN_NGRAPH
463 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> > &inputs,
464 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
465 {
466 auto& input = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
467 auto parent_shape = input.get_shape();
468 int64_t b = parent_shape[0];
469 int64_t h = parent_shape[1];
470 int64_t w = parent_shape[2];
471 int64_t c = parent_shape[3];
472
473 int64_t cols = b * h * w * anchors;
474 int64_t rows = c / anchors;
475 auto shape_node = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{2}, std::vector<int64_t>{cols, rows});
476 auto tr_axes = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{2}, std::vector<int64_t>{1, 0});
477
478 std::shared_ptr<ov::Node> input2d;
479 {
480 input2d = std::make_shared<ov::op::v1::Reshape>(input, shape_node, true);
481 input2d = std::make_shared<ov::op::v1::Transpose>(input2d, tr_axes);
482 }
483
484 std::shared_ptr<ov::Node> region;
485 {
486 auto new_axes = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{4}, std::vector<int64_t>{0, 3, 1, 2});
487 auto tr_input = std::make_shared<ov::op::v1::Transpose>(input, new_axes);
488
489 std::vector<float> anchors_vec(blobs[0].ptr<float>(), blobs[0].ptr<float>() + blobs[0].total());
490 std::vector<int64_t> mask(anchors, 1);
491 region = std::make_shared<ov::op::v0::RegionYolo>(tr_input, coords, classes, anchors, useSoftmax, mask, 1, 3, anchors_vec);
492
493 auto tr_shape = tr_input->get_shape();
494 auto shape_as_inp = std::make_shared<ov::op::v0::Constant>(ov::element::i64,
495 ov::Shape{tr_shape.size()},
496 std::vector<int64_t>(tr_shape.begin(), tr_shape.end()));
497
498 region = std::make_shared<ov::op::v1::Reshape>(region, shape_as_inp, true);
499 new_axes = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{4}, std::vector<int64_t>{0, 2, 3, 1});
500 region = std::make_shared<ov::op::v1::Transpose>(region, new_axes);
501
502 region = std::make_shared<ov::op::v1::Reshape>(region, shape_node, true);
503 region = std::make_shared<ov::op::v1::Transpose>(region, tr_axes);
504 }
505
506 auto strides = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{2}, std::vector<int64_t>{1, 1});
507 std::vector<int64_t> boxes_shape{b, anchors, h, w};
508 auto shape_3d = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{boxes_shape.size()}, boxes_shape.data());
509
510 ov::Shape box_broad_shape{1, (size_t)anchors, (size_t)h, (size_t)w};
511 auto scale_x_y_node = std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{1}, &scale_x_y);
512 auto shift_node = std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{1}, std::vector<float>{0.5});
513
514 auto axis = ov::op::v0::Constant::create<int64_t>(ov::element::i64, ov::Shape{}, {0});
515 auto splits = ov::op::v0::Constant::create<int64_t>(ov::element::i64, ov::Shape{5}, {1, 1, 1, 1, rows - 4});
516 auto split = std::make_shared<ov::op::v1::VariadicSplit>(input2d, axis, splits);
517 std::shared_ptr<ov::Node> box_x;
518 {
519 box_x = std::make_shared<ov::op::v0::Sigmoid>(split->output(0));
520 box_x = std::make_shared<ov::op::v1::Subtract>(box_x, shift_node, ov::op::AutoBroadcastType::NUMPY);
521 box_x = std::make_shared<ov::op::v1::Multiply>(box_x, scale_x_y_node, ov::op::AutoBroadcastType::NUMPY);
522 box_x = std::make_shared<ov::op::v1::Add>(box_x, shift_node, ov::op::AutoBroadcastType::NUMPY);
523 box_x = std::make_shared<ov::op::v1::Reshape>(box_x, shape_3d, true);
524
525 std::vector<float> x_indices(w * h * anchors);
526 auto begin = x_indices.begin();
527 for (int i = 0; i < w; i++)
528 {
529 std::fill(begin + i * anchors, begin + (i + 1) * anchors, i);
530 }
531
532 for (int j = 1; j < h; j++)
533 {
534 std::copy(begin, begin + w * anchors, begin + j * w * anchors);
535 }
536 auto horiz = std::make_shared<ov::op::v0::Constant>(ov::element::f32, box_broad_shape, x_indices.data());
537 box_x = std::make_shared<ov::op::v1::Add>(box_x, horiz, ov::op::AutoBroadcastType::NUMPY);
538
539 auto cols_node = std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{1}, std::vector<float>{float(w)});
540 box_x = std::make_shared<ov::op::v1::Divide>(box_x, cols_node, ov::op::AutoBroadcastType::NUMPY);
541 }
542
543 std::shared_ptr<ov::Node> box_y;
544 {
545 box_y = std::make_shared<ov::op::v0::Sigmoid>(split->output(1));
546 box_y = std::make_shared<ov::op::v1::Subtract>(box_y, shift_node, ov::op::AutoBroadcastType::NUMPY);
547 box_y = std::make_shared<ov::op::v1::Multiply>(box_y, scale_x_y_node, ov::op::AutoBroadcastType::NUMPY);
548 box_y = std::make_shared<ov::op::v1::Add>(box_y, shift_node, ov::op::AutoBroadcastType::NUMPY);
549 box_y = std::make_shared<ov::op::v1::Reshape>(box_y, shape_3d, true);
550
551 std::vector<float> y_indices(h * anchors);
552 for (int i = 0; i < h; i++)
553 {
554 std::fill(y_indices.begin() + i * anchors, y_indices.begin() + (i + 1) * anchors, i);
555 }
556
557 auto vert = std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{1, (size_t)anchors, (size_t)h, 1}, y_indices.data());
558 box_y = std::make_shared<ov::op::v1::Add>(box_y, vert, ov::op::AutoBroadcastType::NUMPY);
559 auto rows_node = std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{1}, std::vector<float>{float(h)});
560 box_y = std::make_shared<ov::op::v1::Divide>(box_y, rows_node, ov::op::AutoBroadcastType::NUMPY);
561 }
562
563 std::shared_ptr<ov::Node> box_w, box_h;
564 {
565 int hNorm, wNorm;
566 if (nodes.size() > 1)
567 {
568 auto node_1_shape = nodes[1].dynamicCast<InfEngineNgraphNode>()->node.get_shape();
569 hNorm = node_1_shape[2];
570 wNorm = node_1_shape[3];
571 }
572 else
573 {
574 hNorm = h;
575 wNorm = w;
576 }
577
578 std::vector<float> anchors_w(anchors), anchors_h(anchors);
579 for (size_t a = 0; a < anchors; ++a)
580 {
581 anchors_w[a] = blobs[0].at<float>(0, 2 * a) / wNorm;
582 anchors_h[a] = blobs[0].at<float>(0, 2 * a + 1) / hNorm;
583 }
584
585 std::vector<float> bias_w(w * h * anchors), bias_h(w * h * anchors);
586 for (int j = 0; j < h; j++)
587 {
588 std::copy(anchors_w.begin(), anchors_w.end(), bias_w.begin() + j * anchors);
589 std::copy(anchors_h.begin(), anchors_h.end(), bias_h.begin() + j * anchors);
590 }
591
592 for (int i = 1; i < w; i++)
593 {
594 std::copy(bias_w.begin(), bias_w.begin() + h * anchors, bias_w.begin() + i * h * anchors);
595 std::copy(bias_h.begin(), bias_h.begin() + h * anchors, bias_h.begin() + i * h * anchors);
596 }
597
598 box_w = std::make_shared<ov::op::v0::Exp>(split->output(2));
599 box_w = std::make_shared<ov::op::v1::Reshape>(box_w, shape_3d, true);
600 auto anchor_w_node = std::make_shared<ov::op::v0::Constant>(ov::element::f32, box_broad_shape, bias_w.data());
601 box_w = std::make_shared<ov::op::v1::Multiply>(box_w, anchor_w_node, ov::op::AutoBroadcastType::NUMPY);
602
603 box_h = std::make_shared<ov::op::v0::Exp>(split->output(3));
604 box_h = std::make_shared<ov::op::v1::Reshape>(box_h, shape_3d, true);
605 auto anchor_h_node = std::make_shared<ov::op::v0::Constant>(ov::element::f32, box_broad_shape, bias_h.data());
606 box_h = std::make_shared<ov::op::v1::Multiply>(box_h, anchor_h_node, ov::op::AutoBroadcastType::NUMPY);
607 }
608
609 auto region_splits = ov::op::v0::Constant::create<int64_t>(ov::element::i64, ov::Shape{3}, {4, 1, rows - 5});
610 auto region_split = std::make_shared<ov::op::v1::VariadicSplit>(region, axis, region_splits);
611
612 std::shared_ptr<ov::Node> scale;
613 {
614 float thr = classfix == -1 ? 0.5 : 0;
615 auto thresh_node = std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{1}, std::vector<float>{thr});
616 auto mask = std::make_shared<ov::op::v1::Less>(region_split->output(1), thresh_node);
617 auto zero_node = std::make_shared<ov::op::v0::Constant>(ov::element::f32, mask->get_shape(), std::vector<float>(cols, 0));
618 scale = std::make_shared<ov::op::v1::Select>(mask, zero_node, region_split->output(1));
619 }
620
621 std::shared_ptr<ov::Node> probs;
622 {
623 probs = std::make_shared<ov::op::v1::Multiply>(region_split->output(2), scale, ov::op::AutoBroadcastType::NUMPY);
624 auto thresh_node = std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{1}, &thresh);
625 auto mask = std::make_shared<ov::op::v1::Greater>(probs, thresh_node);
626 auto zero_node = std::make_shared<ov::op::v0::Constant>(ov::element::f32, mask->get_shape(), std::vector<float>((rows - 5) * cols, 0));
627 probs = std::make_shared<ov::op::v1::Select>(mask, probs, zero_node);
628 }
629
630
631 auto concat_shape = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{2}, std::vector<int64_t>{1, cols});
632 box_x = std::make_shared<ov::op::v1::Reshape>(box_x, concat_shape, true);
633 box_y = std::make_shared<ov::op::v1::Reshape>(box_y, concat_shape, true);
634 box_w = std::make_shared<ov::op::v1::Reshape>(box_w, concat_shape, true);
635 box_h = std::make_shared<ov::op::v1::Reshape>(box_h, concat_shape, true);
636
637 ov::NodeVector inp_nodes{box_x, box_y, box_w, box_h, scale, probs};
638 std::shared_ptr<ov::Node> result = std::make_shared<ov::op::v0::Concat>(inp_nodes, 0);
639 result = std::make_shared<ov::op::v1::Transpose>(result, tr_axes);
640 if (b > 1)
641 {
642 std::vector<int64_t> sizes{b, static_cast<int64_t>(result->get_shape()[0]) / b, static_cast<int64_t>(result->get_shape()[1])};
643 auto shape_node = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{sizes.size()}, sizes.data());
644 result = std::make_shared<ov::op::v1::Reshape>(result, shape_node, true);
645 }
646
647 return Ptr<BackendNode>(new InfEngineNgraphNode(result));
648 }
649#endif // HAVE_DNN_NGRAPH
