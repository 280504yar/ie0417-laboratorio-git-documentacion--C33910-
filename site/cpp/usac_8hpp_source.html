266 std::vector<int> &inliers, double threshold);
267 static int getInliers (const std::vector<float> &errors, std::vector<bool> &inliers,
268 double threshold);
269 static int getInliers (const std::vector<float> &errors, std::vector<int> &inliers,
270 double threshold);
271 Score selectBest (const std::vector<Mat> &models, int num_models, Mat &best) {
272 if (num_models == 0) return {};
273 int best_idx = 0;
274 Score best_score = getScore(models[0]);
275 for (int i = 1; i < num_models; i++) {
276 const auto sc = getScore(models[i]);
277 if (sc.isBetter(best_score)) {
278 best_score = sc;
279 best_idx = i;
280 }
281 }
282 models[best_idx].copyTo(best);
283 return best_score;
284 }

466struct SPRT_history {
467 /*
468 * delta:
469 * The probability of a data point being consistent
470 * with a 'bad' model is modeled as a probability of
471 * a random event with Bernoulli distribution with parameter
472 * delta : p(1|Hb) = delta.
473
474 * epsilon:
475 * The probability p(1|Hg) = epsilon
476 * that any randomly chosen data point is consistent with a 'good' model
477 * is approximated by the fraction of inliers epsilon among the data
478 * points
479
480 * A is the decision threshold, the only parameter of the Adapted SPRT
481 */
482 double epsilon, delta, A;
483 // number of samples processed by test
484 int tested_samples; // k
485 SPRT_history () : epsilon(0), delta(0), A(0) {
486 tested_samples = 0;
487 }
488};
