190 Ptr<Layer> getLayerInstance(LayerData& ld) const
191 {
192 CV_TRACE_FUNCTION();
193 CV_TRACE_ARG_VALUE(type, "type", ld.type.c_str());
194
195 if (ld.layerInstance)
196 return ld.layerInstance;
197
198 ld.layerInstance = createLayerInstance(ld);
199 if (!ld.layerInstance && basePtr_)
200 {
201 ld.layerInstance = basePtr_->createLayerInstance(ld);
202 CV_LOG_IF_DEBUG(NULL, ld.layerInstance, "Created layer \"" + ld.name + "\" of type \"" + ld.type + "\" from upstream layers registry");
203 }
204 if (!ld.layerInstance)
205 {
206 CV_Error(Error::StsError, "Can't create layer \"" + ld.name + "\" of type \"" + ld.type + "\"");
207 }
208
209 return ld.layerInstance;
210 }
