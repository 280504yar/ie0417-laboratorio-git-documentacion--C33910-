786 template<typename _Tp, int numflag> class VecWriterProxy
787 {
788 public:
789 VecWriterProxy( FileStorage* _fs ) : fs(_fs) {}
790 void operator()(const std::vector<_Tp>& vec) const
791 {
792 size_t count = vec.size();
793 for (size_t i = 0; i < count; i++)
794 write(*fs, vec[i]);
795 }
796 private:

800 template<typename _Tp> class VecWriterProxy<_Tp, 1>
801 {
802 public:
803 VecWriterProxy( FileStorage* _fs ) : fs(_fs) {}
804 void operator()(const std::vector<_Tp>& vec) const
805 {
806 int _fmt = traits::SafeFmt<_Tp>::fmt;
807 char fmt[] = { (char)((_fmt >> 8) + '1'), (char)_fmt, '\0' };
808 fs->writeRaw(fmt, !vec.empty() ? (uchar*)&vec[0] : 0, vec.size() * sizeof(_Tp));
809 }
810 private:
811 FileStorage* fs;
812 };
813
814 template<typename _Tp, int numflag> class VecReaderProxy
815 {
816 public:
817 VecReaderProxy( FileNodeIterator* _it ) : it(_it) {}
818 void operator()(std::vector<_Tp>& vec, size_t count) const
819 {
820 count = std::min(count, it->remaining());
821 vec.resize(count);
822 for (size_t i = 0; i < count; i++, ++(*it))
823 read(**it, vec[i], _Tp());
824 }
825 private:
826 FileNodeIterator* it;
827 };
828
829 template<typename _Tp> class VecReaderProxy<_Tp, 1>
830 {
831 public:
832 VecReaderProxy( FileNodeIterator* _it ) : it(_it) {}
833 void operator()(std::vector<_Tp>& vec, size_t count) const
834 {
835 size_t remaining = it->remaining();
836 size_t cn = DataType<_Tp>::channels;
837 int _fmt = traits::SafeFmt<_Tp>::fmt;
838 CV_Assert((_fmt >> 8) < 9);
839 char fmt[] = { (char)((_fmt >> 8)+'1'), (char)_fmt, '\0' };
840 CV_Assert((remaining % cn) == 0);
841 size_t remaining1 = remaining / cn;
842 count = count > remaining1 ? remaining1 : count;
843 vec.resize(count);
844 it->readRaw(fmt, !vec.empty() ? (uchar*)&vec[0] : 0, count*sizeof(_Tp));
845 }
846 private:
847 FileNodeIterator* it;
848 };
