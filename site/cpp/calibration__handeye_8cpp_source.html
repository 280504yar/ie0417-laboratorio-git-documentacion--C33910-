9static Mat homogeneousInverse(const Mat& T)
10{
11 CV_Assert(T.rows == 4 && T.cols == 4);
12
13 Mat R = T(Rect(0, 0, 3, 3));
14 Mat t = T(Rect(3, 0, 1, 3));
15 Mat Rt = R.t();
16 Mat tinv = -Rt * t;
17 Mat Tinv = Mat::eye(4, 4, T.type());
18 Rt.copyTo(Tinv(Rect(0, 0, 3, 3)));
19 tinv.copyTo(Tinv(Rect(3, 0, 1, 3)));
20
21 return Tinv;
22}

24// q = rot2quatMinimal(R)
25//
26// R - 3x3 rotation matrix, or 4x4 homogeneous matrix
27// q - 3x1 unit quaternion <qx, qy, qz>
28// q = sin(theta/2) * v
29// theta - rotation angle
30// v - unit rotation axis, |v| = 1
31// Reference: http://www.euclideanspace.com/maths/geometry/rotations/conversions/matrixToQuaternion/
32static Mat rot2quatMinimal(const Mat& R)
33{
34 CV_Assert(R.type() == CV_64FC1 && R.rows >= 3 && R.cols >= 3);
35
36 double m00 = R.at<double>(0,0), m01 = R.at<double>(0,1), m02 = R.at<double>(0,2);
37 double m10 = R.at<double>(1,0), m11 = R.at<double>(1,1), m12 = R.at<double>(1,2);
38 double m20 = R.at<double>(2,0), m21 = R.at<double>(2,1), m22 = R.at<double>(2,2);
39 double trace = m00 + m11 + m22;
40
41 double qx, qy, qz;
42 if (trace > 0) {
43 double S = sqrt(trace + 1.0) * 2; // S=4*qw
44 qx = (m21 - m12) / S;
45 qy = (m02 - m20) / S;
46 qz = (m10 - m01) / S;
47 } else if (m00 > m11 && m00 > m22) {
48 double S = sqrt(1.0 + m00 - m11 - m22) * 2; // S=4*qx
49 qx = 0.25 * S;
50 qy = (m01 + m10) / S;
51 qz = (m02 + m20) / S;
52 } else if (m11 > m22) {
53 double S = sqrt(1.0 + m11 - m00 - m22) * 2; // S=4*qy
54 qx = (m01 + m10) / S;
55 qy = 0.25 * S;
56 qz = (m12 + m21) / S;
57 } else {
58 double S = sqrt(1.0 + m22 - m00 - m11) * 2; // S=4*qz
59 qx = (m02 + m20) / S;
60 qy = (m12 + m21) / S;
61 qz = 0.25 * S;
62 }
63
64 return (Mat_<double>(3,1) << qx, qy, qz);
65}

67static Mat skew(const Mat& v)
68{
69 CV_Assert(v.type() == CV_64FC1 && v.rows == 3 && v.cols == 1);
70
71 double vx = v.at<double>(0,0);
72 double vy = v.at<double>(1,0);
73 double vz = v.at<double>(2,0);
74 return (Mat_<double>(3,3) << 0, -vz, vy,
75 vz, 0, -vx,
76 -vy, vx, 0);
77}

79// R = quatMinimal2rot(q)
80//
81// q - 3x1 unit quaternion <qx, qy, qz>
82// R - 3x3 rotation matrix
83// q = sin(theta/2) * v
84// theta - rotation angle
85// v - unit rotation axis, |v| = 1
86static Mat quatMinimal2rot(const Mat& q)
87{
88 CV_Assert(q.type() == CV_64FC1 && q.rows == 3 && q.cols == 1);
89
90 Mat p = q.t()*q;
91 double w = sqrt(1 - p.at<double>(0,0));
92
93 Mat diag_p = Mat::eye(3,3,CV_64FC1)*p.at<double>(0,0);
94 return 2*q*q.t() + 2*w*skew(q) + Mat::eye(3,3,CV_64FC1) - 2*diag_p;
95}

97// q = rot2quat(R)
98//
99// q - 4x1 unit quaternion <qw, qx, qy, qz>
100// R - 3x3 rotation matrix
101// Reference: http://www.euclideanspace.com/maths/geometry/rotations/conversions/matrixToQuaternion/
102static Mat rot2quat(const Mat& R)
103{
104 CV_Assert(R.type() == CV_64FC1 && R.rows >= 3 && R.cols >= 3);
105
106 double m00 = R.at<double>(0,0), m01 = R.at<double>(0,1), m02 = R.at<double>(0,2);
107 double m10 = R.at<double>(1,0), m11 = R.at<double>(1,1), m12 = R.at<double>(1,2);
108 double m20 = R.at<double>(2,0), m21 = R.at<double>(2,1), m22 = R.at<double>(2,2);
109 double trace = m00 + m11 + m22;
110
111 double qw, qx, qy, qz;
112 if (trace > 0) {
113 double S = sqrt(trace + 1.0) * 2; // S=4*qw
114 qw = 0.25 * S;
115 qx = (m21 - m12) / S;
116 qy = (m02 - m20) / S;
117 qz = (m10 - m01) / S;
118 } else if (m00 > m11 && m00 > m22) {
119 double S = sqrt(1.0 + m00 - m11 - m22) * 2; // S=4*qx
120 qw = (m21 - m12) / S;
121 qx = 0.25 * S;
122 qy = (m01 + m10) / S;
123 qz = (m02 + m20) / S;
124 } else if (m11 > m22) {
125 double S = sqrt(1.0 + m11 - m00 - m22) * 2; // S=4*qy
126 qw = (m02 - m20) / S;
127 qx = (m01 + m10) / S;
128 qy = 0.25 * S;
129 qz = (m12 + m21) / S;
130 } else {
131 double S = sqrt(1.0 + m22 - m00 - m11) * 2; // S=4*qz
132 qw = (m10 - m01) / S;
133 qx = (m02 + m20) / S;
134 qy = (m12 + m21) / S;
135 qz = 0.25 * S;
136 }
137
138 return (Mat_<double>(4,1) << qw, qx, qy, qz);
139}

141// R = quat2rot(q)
142//
143// q - 4x1 unit quaternion <qw, qx, qy, qz>
144// R - 3x3 rotation matrix
145static Mat quat2rot(const Mat& q)
146{
147 CV_Assert(q.type() == CV_64FC1 && q.rows == 4 && q.cols == 1);
148
149 double qw = q.at<double>(0,0);
150 double qx = q.at<double>(1,0);
151 double qy = q.at<double>(2,0);
152 double qz = q.at<double>(3,0);
153
154 Mat R(3, 3, CV_64FC1);
155 R.at<double>(0, 0) = 1 - 2*qy*qy - 2*qz*qz;
156 R.at<double>(0, 1) = 2*qx*qy - 2*qz*qw;
157 R.at<double>(0, 2) = 2*qx*qz + 2*qy*qw;
158
159 R.at<double>(1, 0) = 2*qx*qy + 2*qz*qw;
160 R.at<double>(1, 1) = 1 - 2*qx*qx - 2*qz*qz;
161 R.at<double>(1, 2) = 2*qy*qz - 2*qx*qw;
162
163 R.at<double>(2, 0) = 2*qx*qz - 2*qy*qw;
164 R.at<double>(2, 1) = 2*qy*qz + 2*qx*qw;
165 R.at<double>(2, 2) = 1 - 2*qx*qx - 2*qy*qy;
166
167 return R;
168}

170// Kronecker product or tensor product
171// https://stackoverflow.com/a/36552682
172static Mat kron(const Mat& A, const Mat& B)
173{
174 CV_Assert(A.channels() == 1 && B.channels() == 1);
175
176 Mat1d Ad, Bd;
177 A.convertTo(Ad, CV_64F);
178 B.convertTo(Bd, CV_64F);
179
180 Mat1d Kd(Ad.rows * Bd.rows, Ad.cols * Bd.cols, 0.0);
181 for (int ra = 0; ra < Ad.rows; ra++)
182 {
183 for (int ca = 0; ca < Ad.cols; ca++)
184 {
185 Kd(Range(ra*Bd.rows, (ra + 1)*Bd.rows), Range(ca*Bd.cols, (ca + 1)*Bd.cols)) = Bd.mul(Ad(ra, ca));
186 }
187 }
188
189 Mat K;
190 Kd.convertTo(K, A.type());
191 return K;
192}

194// quaternion multiplication
195static Mat qmult(const Mat& s, const Mat& t)
196{
197 CV_Assert(s.type() == CV_64FC1 && t.type() == CV_64FC1);
198 CV_Assert(s.rows == 4 && s.cols == 1);
199 CV_Assert(t.rows == 4 && t.cols == 1);
200
201 double s0 = s.at<double>(0,0);
202 double s1 = s.at<double>(1,0);
203 double s2 = s.at<double>(2,0);
204 double s3 = s.at<double>(3,0);
205
206 double t0 = t.at<double>(0,0);
207 double t1 = t.at<double>(1,0);
208 double t2 = t.at<double>(2,0);
209 double t3 = t.at<double>(3,0);
210
211 Mat q(4, 1, CV_64FC1);
212 q.at<double>(0,0) = s0*t0 - s1*t1 - s2*t2 - s3*t3;
213 q.at<double>(1,0) = s0*t1 + s1*t0 + s2*t3 - s3*t2;
214 q.at<double>(2,0) = s0*t2 - s1*t3 + s2*t0 + s3*t1;
215 q.at<double>(3,0) = s0*t3 + s1*t2 - s2*t1 + s3*t0;
216
217 return q;
218}

220// dq = homogeneous2dualQuaternion(H)
221//
222// H - 4x4 homogeneous transformation: [R | t; 0 0 0 | 1]
223// dq - 8x1 dual quaternion: <q (rotation part), qprime (translation part)>
224static Mat homogeneous2dualQuaternion(const Mat& H)
225{
226 CV_Assert(H.type() == CV_64FC1 && H.rows == 4 && H.cols == 4);
227
228 Mat dualq(8, 1, CV_64FC1);
229 Mat R = H(Rect(0, 0, 3, 3));
230 Mat t = H(Rect(3, 0, 1, 3));
231
232 Mat q = rot2quat(R);
233 Mat qt = Mat::zeros(4, 1, CV_64FC1);
234 t.copyTo(qt(Rect(0, 1, 1, 3)));
235 Mat qprime = 0.5 * qmult(qt, q);
236
237 q.copyTo(dualq(Rect(0, 0, 1, 4)));
238 qprime.copyTo(dualq(Rect(0, 4, 1, 4)));
239
240 return dualq;
241}

243// H = dualQuaternion2homogeneous(dq)
244//
245// H - 4x4 homogeneous transformation: [R | t; 0 0 0 | 1]
246// dq - 8x1 dual quaternion: <q (rotation part), qprime (translation part)>
247static Mat dualQuaternion2homogeneous(const Mat& dualq)
248{
249 CV_Assert(dualq.type() == CV_64FC1 && dualq.rows == 8 && dualq.cols == 1);
250
251 Mat q = dualq(Rect(0, 0, 1, 4));
252 Mat qprime = dualq(Rect(0, 4, 1, 4));
253
254 Mat R = quat2rot(q);
255 q.at<double>(1,0) = -q.at<double>(1,0);
256 q.at<double>(2,0) = -q.at<double>(2,0);
257 q.at<double>(3,0) = -q.at<double>(3,0);
258
259 Mat qt = 2*qmult(qprime, q);
260 Mat t = qt(Rect(0, 1, 1, 3));
261
262 Mat H = Mat::eye(4, 4, CV_64FC1);
263 R.copyTo(H(Rect(0, 0, 3, 3)));
264 t.copyTo(H(Rect(3, 0, 1, 3)));
265
266 return H;
267}

269//Reference:
270//R. Y. Tsai and R. K. Lenz, "A new technique for fully autonomous and efficient 3D robotics hand/eye calibration."
271//In IEEE Transactions on Robotics and Automation, vol. 5, no. 3, pp. 345-358, June 1989.
272//C++ code converted from Zoran Lazarevic's Matlab code:
273//http://lazax.com/www.cs.columbia.edu/~laza/html/Stewart/matlab/handEye.m
274static void calibrateHandEyeTsai(const std::vector<Mat>& Hg, const std::vector<Mat>& Hc,
275 Mat& R_cam2gripper, Mat& t_cam2gripper)
276{
277 //Number of unique camera position pairs
278 int K = static_cast<int>((Hg.size()*Hg.size() - Hg.size()) / 2.0);
279 //Will store: skew(Pgij+Pcij)
280 Mat A(3*K, 3, CV_64FC1);
281 //Will store: Pcij - Pgij
282 Mat B(3*K, 1, CV_64FC1);
283
284 std::vector<Mat> vec_Hgij, vec_Hcij;
285 vec_Hgij.reserve(static_cast<size_t>(K));
286 vec_Hcij.reserve(static_cast<size_t>(K));
287
288 int idx = 0;
289 for (size_t i = 0; i < Hg.size(); i++)
290 {
291 for (size_t j = i+1; j < Hg.size(); j++)
292 {
293 //Defines coordinate transformation from Gi to Gj
294 //Hgi is from Gi (gripper) to RW (robot base)
295 //Hgj is from Gj (gripper) to RW (robot base)
296 Mat Hgij = homogeneousInverse(Hg[j]) * Hg[i]; //eq 6
297 //Rotation axis for Rgij which is the 3D rotation from gripper coordinate frame Gi to Gj
298 Mat Pgij = 2*rot2quatMinimal(Hgij);
299
300 //Defines coordinate transformation from Ci to Cj
301 //Hci is from CW (calibration target) to Ci (camera)
302 //Hcj is from CW (calibration target) to Cj (camera)
303 Mat Hcij = Hc[j] * homogeneousInverse(Hc[i]); //eq 7
304 //Rotation axis for Rcij
305 Mat Pcij = 2*rot2quatMinimal(Hcij);
306
307 // Discard motions with rotation too small or too close to pi radians
308 // The limits 1.7 and 0.3 correspond to angles less than 17 degrees or greater than 120 degrees. They are
309 // based on verifying equation 12 from the source paper using data generated with a known hand-eye
310 // calibration. The data contained 25 poses, so 300 motions were considered. Of these, 188 satisfied
311 // equation 12, and the remaining 112 all had Pcij or Pgij with norms greater than 1.7. Although errors
312 // from small rotations were not observed, it is known that these motions are less informative (see
313 // section II.B.3, and figure 6).
314 double Pgij_norm = cv::norm(Pgij);
315 double Pcij_norm = cv::norm(Pcij);
316 if (Pgij_norm < 0.3 || Pcij_norm < 0.3 || Pgij_norm > 1.7 || Pcij_norm > 1.7) {
317 continue;
318 }
319
320 vec_Hgij.push_back(Hgij);
321 vec_Hcij.push_back(Hcij);
322
323 //Left-hand side: skew(Pgij+Pcij)
324 skew(Pgij+Pcij).copyTo(A(Rect(0, idx*3, 3, 3)));
325 //Right-hand side: Pcij - Pgij
326 Mat diff = Pcij - Pgij;
327 diff.copyTo(B(Rect(0, idx*3, 1, 3)));
328 idx++;
329 }
330 }
331
332 // insufficient data
333 if (idx < 2) {
334 CV_LOG_ERROR(NULL, "Hand-eye calibration failed! Not enough informative motions--include larger rotations.");
335 return;
336 }
337 A.resize(3*idx);
338 B.resize(3*idx);
339
340 Mat Pcg_;
341 //Rotation from camera to gripper is obtained from the set of equations:
342 // skew(Pgij+Pcij) * Pcg_ = Pcij - Pgij (eq 12)
343 solve(A, B, Pcg_, DECOMP_SVD);
344
345 Mat Pcg_norm = Pcg_.t() * Pcg_;
346 //Obtained non-unit quaternion is scaled back to unit value that
347 //designates camera-gripper rotation
348 Mat Pcg = 2 * Pcg_ / sqrt(1 + Pcg_norm.at<double>(0,0)); //eq 14
349
350 Mat Rcg = quatMinimal2rot(Pcg/2.0);
351
352 for (size_t i = 0; i < vec_Hgij.size(); i++)
353 {
354 //Defines coordinate transformation from Gi to Gj
355 //Hgi is from Gi (gripper) to RW (robot base)
356 //Hgj is from Gj (gripper) to RW (robot base)
357 Mat Hgij = vec_Hgij[i];
358 //Defines coordinate transformation from Ci to Cj
359 //Hci is from CW (calibration target) to Ci (camera)
360 //Hcj is from CW (calibration target) to Cj (camera)
361 Mat Hcij = vec_Hcij[i];
362
363 //Left-hand side: (Rgij - I)
364 Mat diff = Hgij(Rect(0,0,3,3)) - Mat::eye(3,3,CV_64FC1);
365 diff.copyTo(A(Rect(0, static_cast<int>(i)*3, 3, 3)));
366
367 //Right-hand side: Rcg*Tcij - Tgij
368 diff = Rcg*Hcij(Rect(3, 0, 1, 3)) - Hgij(Rect(3, 0, 1, 3));
369 diff.copyTo(B(Rect(0, static_cast<int>(i)*3, 1, 3)));
370 }
371
372 Mat Tcg;
373 //Translation from camera to gripper is obtained from the set of equations:
374 // (Rgij - I) * Tcg = Rcg*Tcij - Tgij (eq 15)
375 solve(A, B, Tcg, DECOMP_SVD);
376
377 R_cam2gripper = Rcg;
378 t_cam2gripper = Tcg;
379}

381//Reference:
382//F. Park, B. Martin, "Robot Sensor Calibration: Solving AX = XB on the Euclidean Group."
383//In IEEE Transactions on Robotics and Automation, 10(5): 717-721, 1994.
384//Matlab code: http://math.loyola.edu/~mili/Calibration/
385static void calibrateHandEyePark(const std::vector<Mat>& Hg, const std::vector<Mat>& Hc,
386 Mat& R_cam2gripper, Mat& t_cam2gripper)
387{
388 Mat M = Mat::zeros(3, 3, CV_64FC1);
389
390 for (size_t i = 0; i < Hg.size(); i++)
391 {
392 for (size_t j = i+1; j < Hg.size(); j++)
393 {
394 Mat Hgij = homogeneousInverse(Hg[j]) * Hg[i];
395 Mat Hcij = Hc[j] * homogeneousInverse(Hc[i]);
396
397 Mat Rgij = Hgij(Rect(0, 0, 3, 3));
398 Mat Rcij = Hcij(Rect(0, 0, 3, 3));
399
400 Mat a, b;
401 Rodrigues(Rgij, a);
402 Rodrigues(Rcij, b);
403
404 M += b * a.t();
405 }
406 }
407
408 Mat eigenvalues, eigenvectors;
409 eigen(M.t()*M, eigenvalues, eigenvectors);
410
411 Mat v = Mat::zeros(3, 3, CV_64FC1);
412 for (int i = 0; i < 3; i++) {
413 v.at<double>(i,i) = 1.0 / sqrt(eigenvalues.at<double>(i,0));
414 }
415
416 Mat R = eigenvectors.t() * v * eigenvectors * M.t();
417 R_cam2gripper = R;
418
419 int K = static_cast<int>((Hg.size()*Hg.size() - Hg.size()) / 2.0);
420 Mat C(3*K, 3, CV_64FC1);
421 Mat d(3*K, 1, CV_64FC1);
422 Mat I3 = Mat::eye(3, 3, CV_64FC1);
423
424 int idx = 0;
425 for (size_t i = 0; i < Hg.size(); i++)
426 {
427 for (size_t j = i+1; j < Hg.size(); j++, idx++)
428 {
429 Mat Hgij = homogeneousInverse(Hg[j]) * Hg[i];
430 Mat Hcij = Hc[j] * homogeneousInverse(Hc[i]);
431
432 Mat Rgij = Hgij(Rect(0, 0, 3, 3));
433
434 Mat tgij = Hgij(Rect(3, 0, 1, 3));
435 Mat tcij = Hcij(Rect(3, 0, 1, 3));
436
437 Mat I_tgij = I3 - Rgij;
438 I_tgij.copyTo(C(Rect(0, 3*idx, 3, 3)));
439
440 Mat A_RB = tgij - R*tcij;
441 A_RB.copyTo(d(Rect(0, 3*idx, 1, 3)));
442 }
443 }
444
445 Mat t;
446 solve(C, d, t, DECOMP_SVD);
447 t_cam2gripper = t;
448}

450//Reference:
451//R. Horaud, F. Dornaika, "Hand-Eye Calibration"
452//In International Journal of Robotics Research, 14(3): 195-210, 1995.
453//Matlab code: http://math.loyola.edu/~mili/Calibration/
454static void calibrateHandEyeHoraud(const std::vector<Mat>& Hg, const std::vector<Mat>& Hc,
455 Mat& R_cam2gripper, Mat& t_cam2gripper)
456{
457 Mat A = Mat::zeros(4, 4, CV_64FC1);
458
459 for (size_t i = 0; i < Hg.size(); i++)
460 {
461 for (size_t j = i+1; j < Hg.size(); j++)
462 {
463 Mat Hgij = homogeneousInverse(Hg[j]) * Hg[i];
464 Mat Hcij = Hc[j] * homogeneousInverse(Hc[i]);
465
466 Mat Rgij = Hgij(Rect(0, 0, 3, 3));
467 Mat Rcij = Hcij(Rect(0, 0, 3, 3));
468
469 Mat qgij = rot2quat(Rgij);
470 if (qgij.at<double>(0, 0) < 0) {
471 qgij *= -1;
472 }
473 double r0 = qgij.at<double>(0,0);
474 double rx = qgij.at<double>(1,0);
475 double ry = qgij.at<double>(2,0);
476 double rz = qgij.at<double>(3,0);
477
478 // Q(r) Appendix A
479 Matx44d Qvi(r0, -rx, -ry, -rz,
480 rx, r0, -rz, ry,
481 ry, rz, r0, -rx,
482 rz, -ry, rx, r0);
483
484 Mat qcij = rot2quat(Rcij);
485 if (qcij.at<double>(0, 0) < 0) {
486 qcij *= -1;
487 }
488 r0 = qcij.at<double>(0,0);
489 rx = qcij.at<double>(1,0);
490 ry = qcij.at<double>(2,0);
491 rz = qcij.at<double>(3,0);
492
493 // W(r) Appendix A
494 Matx44d Wvi(r0, -rx, -ry, -rz,
495 rx, r0, rz, -ry,
496 ry, -rz, r0, rx,
497 rz, ry, -rx, r0);
498
499 // Ai = (Q(vi') - W(vi))^T (Q(vi') - W(vi))
500 A += (Qvi - Wvi).t() * (Qvi - Wvi);
501 }
502 }
503
504 Mat eigenvalues, eigenvectors;
505 eigen(A, eigenvalues, eigenvectors);
506
507 Mat R = quat2rot(eigenvectors.row(3).t());
508 R_cam2gripper = R;
509
510 int K = static_cast<int>((Hg.size()*Hg.size() - Hg.size()) / 2.0);
511 Mat C(3*K, 3, CV_64FC1);
512 Mat d(3*K, 1, CV_64FC1);
513 Mat I3 = Mat::eye(3, 3, CV_64FC1);
514
515 int idx = 0;
516 for (size_t i = 0; i < Hg.size(); i++)
517 {
518 for (size_t j = i+1; j < Hg.size(); j++, idx++)
519 {
520 Mat Hgij = homogeneousInverse(Hg[j]) * Hg[i];
521 Mat Hcij = Hc[j] * homogeneousInverse(Hc[i]);
522
523 Mat Rgij = Hgij(Rect(0, 0, 3, 3));
524
525 Mat tgij = Hgij(Rect(3, 0, 1, 3));
526 Mat tcij = Hcij(Rect(3, 0, 1, 3));
527
528 Mat I_tgij = I3 - Rgij;
529 I_tgij.copyTo(C(Rect(0, 3*idx, 3, 3)));
530
531 Mat A_RB = tgij - R*tcij;
532 A_RB.copyTo(d(Rect(0, 3*idx, 1, 3)));
533 }
534 }
535
536 Mat t;
537 solve(C, d, t, DECOMP_SVD);
538 t_cam2gripper = t;
539}

541static Mat_<double> normalizeRotation(const Mat_<double>& R_)
542{
543 // Make R unit determinant
544 Mat_<double> R = R_.clone();
545 double det = determinant(R);
546 if (std::fabs(det) < FLT_EPSILON)
547 {
548 CV_Error(Error::StsNoConv, "Rotation normalization issue: determinant(R) is null");
549 }
550 R = std::cbrt(std::copysign(1, det) / std::fabs(det)) * R;
551
552 // Make R orthogonal
553 Mat w, u, vt;
554 SVDecomp(R, w, u, vt);
555 R = u*vt;
556
557 // Handle reflection case
558 if (determinant(R) < 0)
559 {
560 Matx33d diag(1.0, 0.0, 0.0,
561 0.0, 1.0, 0.0,
562 0.0, 0.0, -1.0);
563 R = u*diag*vt;
564 }
565
566 return R;
567}

571//In Second International Conference on 3-D Digital Imaging and Modeling (3DIM'99), pages 430-436, 1999.

573static void calibrateHandEyeAndreff(const std::vector<Mat>& Hg, const std::vector<Mat>& Hc,
574 Mat& R_cam2gripper, Mat& t_cam2gripper)

581 Mat I3 = Mat::eye(3, 3, CV_64FC1);
582 Mat O9x3 = Mat::zeros(9, 3, CV_64FC1);
583 Mat O9x1 = Mat::zeros(9, 1, CV_64FC1);

585 int idx = 0;
586 for (size_t i = 0; i < Hg.size(); i++)
587 {
588 for (size_t j = i+1; j < Hg.size(); j++, idx++)
589 {

594 Mat Rcij = Hcij(Rect(0, 0, 3, 3));

596 Mat tgij = Hgij(Rect(3, 0, 1, 3));
597 Mat tcij = Hcij(Rect(3, 0, 1, 3));
598
599 //Eq 10
600 Mat a00 = I9 - kron(Rgij, Rcij);
601 Mat a01 = O9x3;
602 Mat a10 = kron(I3, tcij.t());
603 Mat a11 = I3 - Rgij;

605 a00.copyTo(A(Rect(0, idx*12, 9, 9)));
606 a01.copyTo(A(Rect(9, idx*12, 3, 9)));
607 a10.copyTo(A(Rect(0, idx*12 + 9, 9, 3)));
608 a11.copyTo(A(Rect(9, idx*12 + 9, 3, 3)));
609
610 O9x1.copyTo(B(Rect(0, idx*12, 1, 9)));
611 tgij.copyTo(B(Rect(0, idx*12 + 9, 1, 3)));
612 }
613 }

616 solve(A, B, X, DECOMP_SVD);

626//Reference:
627//K. Daniilidis, "Hand-Eye Calibration Using Dual Quaternions."
628//In The International Journal of Robotics Research,18(3): 286-298, 1998.
629//Matlab code: http://math.loyola.edu/~mili/Calibration/
630static void calibrateHandEyeDaniilidis(const std::vector<Mat>& Hg, const std::vector<Mat>& Hc,
631 Mat& R_cam2gripper, Mat& t_cam2gripper)
632{
633 int K = static_cast<int>((Hg.size()*Hg.size() - Hg.size()) / 2.0);
634 Mat T = Mat::zeros(6*K, 8, CV_64FC1);
635
636 int idx = 0;
637 for (size_t i = 0; i < Hg.size(); i++)
638 {
639 for (size_t j = i+1; j < Hg.size(); j++, idx++)
640 {
641 Mat Hgij = homogeneousInverse(Hg[j]) * Hg[i];
642 Mat Hcij = Hc[j] * homogeneousInverse(Hc[i]);
643
644 Mat dualqa = homogeneous2dualQuaternion(Hgij);
645 if (dualqa.at<double>(0, 0) < 0) {
646 dualqa *= -1;
647 }
648 Mat dualqb = homogeneous2dualQuaternion(Hcij);
649 if (dualqb.at<double>(0, 0) < 0) {
650 dualqb *= -1;
651 }
652
653 Mat a = dualqa(Rect(0, 1, 1, 3));
654 Mat b = dualqb(Rect(0, 1, 1, 3));
655
656 Mat aprime = dualqa(Rect(0, 5, 1, 3));
657 Mat bprime = dualqb(Rect(0, 5, 1, 3));
658
659 //Eq 31
660 Mat s00 = a - b;
661 Mat s01 = skew(a + b);
662 Mat s10 = aprime - bprime;
663 Mat s11 = skew(aprime + bprime);
664 Mat s12 = a - b;
665 Mat s13 = skew(a + b);
666
667 s00.copyTo(T(Rect(0, idx*6, 1, 3)));
668 s01.copyTo(T(Rect(1, idx*6, 3, 3)));
669 s10.copyTo(T(Rect(0, idx*6 + 3, 1, 3)));
670 s11.copyTo(T(Rect(1, idx*6 + 3, 3, 3)));
671 s12.copyTo(T(Rect(4, idx*6 + 3, 1, 3)));
672 s13.copyTo(T(Rect(5, idx*6 + 3, 3, 3)));
673 }
674 }
675
676 Mat w, u, vt;
677 SVDecomp(T, w, u, vt);
678 Mat v = vt.t();
679
680 Mat u1 = v(Rect(6, 0, 1, 4));
681 Mat v1 = v(Rect(6, 4, 1, 4));
682 Mat u2 = v(Rect(7, 0, 1, 4));
683 Mat v2 = v(Rect(7, 4, 1, 4));
684
685 //Solves Eq 34, Eq 35
686 Mat ma = u1.t()*v1;
687 Mat mb = u1.t()*v2 + u2.t()*v1;
688 Mat mc = u2.t()*v2;
689
690 double a = ma.at<double>(0,0);
691 double b = mb.at<double>(0,0);
692 double c = mc.at<double>(0,0);
693
694 double s1 = (-b + sqrt(b*b - 4*a*c)) / (2*a);
695 double s2 = (-b - sqrt(b*b - 4*a*c)) / (2*a);
696
697 Mat sol1 = s1*s1*u1.t()*u1 + 2*s1*u1.t()*u2 + u2.t()*u2;
698 Mat sol2 = s2*s2*u1.t()*u1 + 2*s2*u1.t()*u2 + u2.t()*u2;
699 double s, val;
700 if (sol1.at<double>(0,0) > sol2.at<double>(0,0))
701 {
702 s = s1;
703 val = sol1.at<double>(0,0);
704 }
705 else
706 {
707 s = s2;
708 val = sol2.at<double>(0,0);
709 }
710
711 double lambda2 = sqrt(1.0 / val);
712 double lambda1 = s * lambda2;
713
714 Mat dualq = lambda1 * v(Rect(6, 0, 1, 8)) + lambda2*v(Rect(7, 0, 1, 8));
715 Mat X = dualQuaternion2homogeneous(dualq);
716
717 Mat R = X(Rect(0, 0, 3, 3));
718 Mat t = X(Rect(3, 0, 1, 3));
719 R_cam2gripper = R;
720 t_cam2gripper = t;
721}

826static void calibrateRobotWorldHandEyeShah(const std::vector<Mat_<double>>& cRw, const std::vector<Mat_<double>>& ctw,
