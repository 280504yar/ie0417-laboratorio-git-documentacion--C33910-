29 DropoutDescriptor(const Handle &handle, float dropout)
30 {
31 CUDA4DNN_CHECK_CUDNN(cudnnCreateDropoutDescriptor(&descriptor));
32
33 // we need additional memory for dropout descriptor
34 size_t stateSize;
35 CUDA4DNN_CHECK_CUDNN(cudnnDropoutGetStatesSize(handle.get(), &stateSize));
36 states.reset(stateSize);
37
38 try
39 {
40 auto seed = 1234ull; // Pick a seed.
41 CUDA4DNN_CHECK_CUDNN(cudnnSetDropoutDescriptor(descriptor, handle.get(), dropout,
42 states.get().get(), stateSize, seed));
43 }
44 catch (...)
45 {
46 CUDA4DNN_CHECK_CUDNN(cudnnDestroyDropoutDescriptor(descriptor));
47 throw;
48 }
49 }
50
51 ~DropoutDescriptor() noexcept
52 {
53 if (descriptor)
54 {
55 CUDA4DNN_CHECK_CUDNN(cudnnDestroyDropoutDescriptor(descriptor));
56 }
57 }
58
59 DropoutDescriptor &operator=(const DropoutDescriptor &) = delete;
60 DropoutDescriptor &operator=(DropoutDescriptor &&other) noexcept
61 {
62 descriptor = other.descriptor;
63 states = std::move(other.states);
64 other.descriptor = nullptr;
65 return *this;
66 };
67
68 cudnnDropoutDescriptor_t get() const noexcept { return descriptor; }
69
70private:
71 cudnnDropoutDescriptor_t descriptor{nullptr};
72
73 using value_type = typename ManagedPtr<char>::element_type;
74 ManagedPtr<value_type> states;
75};

100 RNNDescriptor(const Handle &handle, RNNMode mode, int input_size, int hidden_size, int num_layers,
101 bool bidirectional, const DropoutDescriptor &dropoutDesc)
102 {
103 CUDA4DNN_CHECK_CUDNN(cudnnCreateRNNDescriptor(&descriptor));
104 const auto rnn_mode = [mode] {
105 switch (mode)
106 {
107 case RNNMode::RNN_RELU:
108 return CUDNN_RNN_RELU;
109 case RNNMode::RNN_TANH:
110 return CUDNN_RNN_TANH;
111 case RNNMode::LSTM:
112 return CUDNN_LSTM;
113 case RNNMode::GRU:
114 return CUDNN_GRU;
115 default:
116 return CUDNN_LSTM;
117 }
118 }();
119
120 try
121 {
122#if CUDNN_MAJOR >= 9
123 CUDA4DNN_CHECK_CUDNN(cudnnSetRNNDescriptor_v8(
124 descriptor,
125 algo,
126 rnn_mode,
127 CUDNN_RNN_DOUBLE_BIAS,
128 bidirectional ? CUDNN_BIDIRECTIONAL : CUDNN_UNIDIRECTIONAL,
129 CUDNN_LINEAR_INPUT, detail::get_data_type<T>(),
130 detail::get_data_type<T>(),
131 detail::get_data_type<T>() == CUDNN_DATA_HALF ? CUDNN_TENSOR_OP_MATH : CUDNN_DEFAULT_MATH,
132 input_size,
133 hidden_size,
134 hidden_size,
135 num_layers,
136 dropoutDesc.get(),
137 0)); // What other flags do we might want here?
138#else
139 CUDA4DNN_CHECK_CUDNN(cudnnSetRNNDescriptor_v6(
140 handle.get(),
141 descriptor,
142 hidden_size,
143 num_layers,
144 dropoutDesc.get(),
145 CUDNN_LINEAR_INPUT,
146 bidirectional ? CUDNN_BIDIRECTIONAL : CUDNN_UNIDIRECTIONAL,
147 rnn_mode,
148 algo,
149 detail::get_data_type<T>()));
150#endif
151 }
152 catch (...)
