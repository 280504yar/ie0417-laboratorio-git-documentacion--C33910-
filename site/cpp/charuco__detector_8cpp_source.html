27 bool checkBoard(InputArrayOfArrays markerCorners, InputArray markerIds, InputArray charucoCorners, InputArray charucoIds) {
28 vector<Mat> mCorners;
29 markerCorners.getMatVector(mCorners);
30 const Mat mIds = markerIds.getMat();
31 const Mat chCorners = charucoCorners.getMat();
32 const Mat chIds = charucoIds.getMat();
33 const vector<int>& boardIds = board.getIds();
34
35 const vector<vector<int> > nearestMarkerIdx = board.getNearestMarkerIdx(); // only copy the vectors once
36 const vector<vector<int> > nearestMarkerCorners = board.getNearestMarkerCorners();
37
38 vector<Point2f> distance(nearestMarkerIdx.size(), Point2f(0.f, std::numeric_limits<float>::max()));
39 // distance[i].x: max distance from the i-th charuco corner to charuco corner-forming markers.
40 // The two charuco corner-forming markers of i-th charuco corner are defined in getNearestMarkerIdx()[i]
41 // distance[i].y: min distance from the charuco corner to other markers.
42 for (size_t i = 0ull; i < chIds.total(); i++) {
43 int chId = chIds.ptr<int>(0)[i];
44 Point2f charucoCorner(chCorners.ptr<Point2f>(0)[i]);
45 for (size_t j = 0ull; j < mIds.total(); j++) {
46 int idMaker = mIds.ptr<int>(0)[j];
47 // skip the check if the marker is not in the current board.
48 if (find(boardIds.begin(), boardIds.end(), idMaker) == boardIds.end())
49 continue;
50 Point2f centerMarker((mCorners[j].ptr<Point2f>(0)[0] + mCorners[j].ptr<Point2f>(0)[1] +
51 mCorners[j].ptr<Point2f>(0)[2] + mCorners[j].ptr<Point2f>(0)[3]) / 4.f);
52 float dist = sqrt(normL2Sqr<float>(centerMarker - charucoCorner));
53 // nearestMarkerIdx contains for each charuco corner, nearest marker index in ids array
54 const int nearestMarkerId1 = boardIds[nearestMarkerIdx[chId][0]];
55 const int nearestMarkerId2 = boardIds[nearestMarkerIdx[chId][1]];
56 if (nearestMarkerId1 == idMaker || nearestMarkerId2 == idMaker) {
57 int nearestCornerId = nearestMarkerId1 == idMaker ? nearestMarkerCorners[chId][0] : nearestMarkerCorners[chId][1];
58 Point2f nearestCorner = mCorners[j].ptr<Point2f>(0)[nearestCornerId];
59 // distToNearest: distance from the charuco corner to charuco corner-forming markers
60 float distToNearest = sqrt(normL2Sqr<float>(nearestCorner - charucoCorner));
61 distance[chId].x = max(distance[chId].x, distToNearest);
62 // check that nearestCorner is nearest point
63 {
64 Point2f mid1 = (mCorners[j].ptr<Point2f>(0)[(nearestCornerId + 1) % 4]+nearestCorner)*0.5f;
65 Point2f mid2 = (mCorners[j].ptr<Point2f>(0)[(nearestCornerId + 3) % 4]+nearestCorner)*0.5f;
66 float tmpDist = min(sqrt(normL2Sqr<float>(mid1 - charucoCorner)), sqrt(normL2Sqr<float>(mid2 - charucoCorner)));
67 if (tmpDist < distToNearest)
68 return false;
69 }
70 }
71 // check distance from the charuco corner to other markers
72 else
73 distance[chId].y = min(distance[chId].y, dist);
74 }
75 // if distance from the charuco corner to charuco corner-forming markers more then distance from the charuco corner to other markers,
76 // then a false board is found.
77 if (distance[chId].x > 0.f && distance[chId].y < std::numeric_limits<float>::max() && distance[chId].x > distance[chId].y)
78 return false;
79 }
80 return true;
81 }

85 vector<Size> getMaximumSubPixWindowSizes(InputArrayOfArrays markerCorners, InputArray markerIds,
86 InputArray charucoCorners) {
87 size_t nCharucoCorners = charucoCorners.getMat().total();
88
89 const vector<vector<int> > nearestMarkerIdx = board.getNearestMarkerIdx(); // only copy the vectors once
90 const vector<vector<int> > nearestMarkerCorners = board.getNearestMarkerCorners();
91
92 CV_Assert(nearestMarkerIdx.size() == nCharucoCorners);
93
94 vector<Size> winSizes(nCharucoCorners, Size(-1, -1));
95 for(size_t i = 0ull; i < nCharucoCorners; i++) {
96 if(charucoCorners.getMat().at<Point2f>((int)i) == Point2f(-1.f, -1.f)) continue;
97 if(nearestMarkerIdx[i].empty()) continue;
98 double minDist = -1;
99 int counter = 0;
100 // calculate the distance to each of the closest corner of each closest marker
101 for(size_t j = 0; j < nearestMarkerIdx[i].size(); j++) {
102 // find marker
103 int markerId = board.getIds()[nearestMarkerIdx[i][j]];
104 int markerIdx = -1;
105 for(size_t k = 0; k < markerIds.getMat().total(); k++) {
106 if(markerIds.getMat().at<int>((int)k) == markerId) {
107 markerIdx = (int)k;
108 break;
109 }
110 }
111 if(markerIdx == -1) continue;
112 Point2f markerCorner =
113 markerCorners.getMat(markerIdx).at<Point2f>(nearestMarkerCorners[i][j]);
114 Point2f charucoCorner = charucoCorners.getMat().at<Point2f>((int)i);
115 double dist = norm(markerCorner - charucoCorner);
116 if(minDist == -1) minDist = dist; // if first distance, just assign it
117 minDist = min(dist, minDist);
118 counter++;
119 }
120 // if this is the first closest marker, don't do anything
121 if(counter == 0)
122 continue;
123 else {
124 // else, calculate the maximum window size
125 int winSizeInt = int(minDist - 2); // remove 2 pixels for safety
126 if(winSizeInt < 1) winSizeInt = 1; // minimum size is 1
127 if(winSizeInt > 10) winSizeInt = 10; // maximum size is 10
128 winSizes[i] = Size(winSizeInt, winSizeInt);
129 }
130 }
131 return winSizes;
132 }

135 void selectAndRefineChessboardCorners(InputArray allCorners, InputArray image, OutputArray selectedCorners,
136 OutputArray selectedIds, const vector<Size> &winSizes) {
137 const int minDistToBorder = 2; // minimum distance of the corner to the image border
138 // remaining corners, ids and window refinement sizes after removing corners outside the image
139 vector<Point2f> filteredChessboardImgPoints;
140 vector<Size> filteredWinSizes;
141 vector<int> filteredIds;
142 // filter corners outside the image
143 Rect innerRect(minDistToBorder, minDistToBorder, image.getMat().cols - 2 * minDistToBorder,
144 image.getMat().rows - 2 * minDistToBorder);
145 for(unsigned int i = 0; i < allCorners.getMat().total(); i++) {
146 if(innerRect.contains(allCorners.getMat().at<Point2f>(i))) {
147 filteredChessboardImgPoints.push_back(allCorners.getMat().at<Point2f>(i));
148 filteredIds.push_back(i);
149 filteredWinSizes.push_back(winSizes[i]);
150 }
151 }
152 // if none valid, return 0
153 if(filteredChessboardImgPoints.empty()) return;
154 // corner refinement, first convert input image to grey
155 Mat grey;
156 if(image.type() == CV_8UC3)
157 cvtColor(image, grey, COLOR_BGR2GRAY);
158 else
159 grey = image.getMat();
161 parallel_for_(Range(0, (int)filteredChessboardImgPoints.size()), [&](const Range& range) {
162 const int begin = range.start;
163 const int end = range.end;
164 for (int i = begin; i < end; i++) {
165 vector<Point2f> in;
166 in.push_back(filteredChessboardImgPoints[i]);
167 Size winSize = filteredWinSizes[i];
168 if (winSize.height == -1 || winSize.width == -1)
169 winSize = Size(arucoDetector.getDetectorParameters().cornerRefinementWinSize,
170 arucoDetector.getDetectorParameters().cornerRefinementWinSize);
171 cornerSubPix(grey, in, winSize, Size(),
172 TermCriteria(TermCriteria::MAX_ITER | TermCriteria::EPS,
173 arucoDetector.getDetectorParameters().cornerRefinementMaxIterations,
174 arucoDetector.getDetectorParameters().cornerRefinementMinAccuracy));
175 filteredChessboardImgPoints[i] = in[0];
176 }
177 });
178 // parse output
179 Mat(filteredChessboardImgPoints).copyTo(selectedCorners);
180 Mat(filteredIds).copyTo(selectedIds);
181 }

184 void interpolateCornersCharucoApproxCalib(InputArrayOfArrays markerCorners, InputArray markerIds,
185 InputArray image, OutputArray charucoCorners, OutputArray charucoIds) {
186 CV_Assert(image.getMat().channels() == 1 || image.getMat().channels() == 3);
187 CV_Assert(markerCorners.total() == markerIds.getMat().total());
188
189 // approximated pose estimation using marker corners
190 Mat approximatedRvec, approximatedTvec;
191 Mat objPoints, imgPoints; // object and image points for the solvePnP function
192 Board simpleBoard(board.getObjPoints(), board.getDictionary(), board.getIds());
193 simpleBoard.matchImagePoints(markerCorners, markerIds, objPoints, imgPoints);
194 if (objPoints.total() < 4ull) // need, at least, 4 corners
195 return;
196
197 solvePnP(objPoints, imgPoints, charucoParameters.cameraMatrix, charucoParameters.distCoeffs, approximatedRvec, approximatedTvec);
198
199 // project chessboard corners
200 vector<Point2f> allChessboardImgPoints;
201 projectPoints(board.getChessboardCorners(), approximatedRvec, approximatedTvec, charucoParameters.cameraMatrix,
202 charucoParameters.distCoeffs, allChessboardImgPoints);
203 // calculate maximum window sizes for subpixel refinement. The size is limited by the distance
204 // to the closes marker corner to avoid erroneous displacements to marker corners
205 vector<Size> subPixWinSizes = getMaximumSubPixWindowSizes(markerCorners, markerIds, allChessboardImgPoints);
206 // filter corners outside the image and subpixel-refine charuco corners
207 selectAndRefineChessboardCorners(allChessboardImgPoints, image, charucoCorners, charucoIds, subPixWinSizes);
208 }

211 void interpolateCornersCharucoLocalHom(InputArrayOfArrays markerCorners, InputArray markerIds, InputArray image,
212 OutputArray charucoCorners, OutputArray charucoIds) {
213 CV_Assert(image.getMat().channels() == 1 || image.getMat().channels() == 3);
214 CV_Assert(markerCorners.total() == markerIds.getMat().total());
215 size_t nMarkers = markerIds.getMat().total();
216 // calculate local homographies for each marker
217 vector<Mat> transformations(nMarkers);
218 vector<bool> validTransform(nMarkers, false);
219 const auto& ids = board.getIds();
220 const vector<vector<int> > nearestMarkerIdx = board.getNearestMarkerIdx(); // only copy the vectors once
221
222 for(size_t i = 0ull; i < nMarkers; i++) {
223 vector<Point2f> markerObjPoints2D;
224 int markerId = markerIds.getMat().at<int>((int)i);
225 auto it = find(ids.begin(), ids.end(), markerId);
226 if(it == ids.end()) continue;
227 auto boardIdx = it - ids.begin();
228 markerObjPoints2D.resize(4ull);
229 for(size_t j = 0ull; j < 4ull; j++)
230 markerObjPoints2D[j] =
231 Point2f(board.getObjPoints()[boardIdx][j].x, board.getObjPoints()[boardIdx][j].y);
232 transformations[i] = getPerspectiveTransform(markerObjPoints2D, markerCorners.getMat((int)i));
233 // set transform as valid if transformation is non-singular
234 double det = determinant(transformations[i]);
235 validTransform[i] = std::abs(det) > 1e-6;
236 }
237 const vector<Point3f> chessboardCorners = board.getChessboardCorners();
238 size_t nCharucoCorners = (size_t)chessboardCorners.size();
239 vector<Point2f> allChessboardImgPoints(nCharucoCorners, Point2f(-1, -1));
240 // for each charuco corner, calculate its interpolation position based on the closest markers
241 // homographies
242 for(size_t i = 0ull; i < nCharucoCorners; i++) {
243 Point2f objPoint2D = Point2f(chessboardCorners[i].x, chessboardCorners[i].y);
244 vector<Point2f> interpolatedPositions;
245 for(size_t j = 0ull; j < nearestMarkerIdx[i].size(); j++) {
246 int markerId = board.getIds()[nearestMarkerIdx[i][j]];
247 int markerIdx = -1;
248 for(size_t k = 0ull; k < markerIds.getMat().total(); k++) {
249 if(markerIds.getMat().at<int>((int)k) == markerId) {
250 markerIdx = (int)k;
251 break;
252 }
253 }
254 if (markerIdx != -1 &&
255 validTransform[markerIdx])
256 {
257 vector<Point2f> in, out;
258 in.push_back(objPoint2D);
259 perspectiveTransform(in, out, transformations[markerIdx]);
260 interpolatedPositions.push_back(out[0]);
261 }
262 }
263 // none of the closest markers detected
264 if(interpolatedPositions.empty()) continue;
265 // more than one closest marker detected, take middle point
266 if(interpolatedPositions.size() > 1ull) {
267 allChessboardImgPoints[i] = (interpolatedPositions[0] + interpolatedPositions[1]) / 2.;
268 }
269 // a single closest marker detected
270 else allChessboardImgPoints[i] = interpolatedPositions[0];
271 }
272 // calculate maximum window sizes for subpixel refinement. The size is limited by the distance
273 // to the closes marker corner to avoid erroneous displacements to marker corners
274 vector<Size> subPixWinSizes = getMaximumSubPixWindowSizes(markerCorners, markerIds, allChessboardImgPoints);
275 // filter corners outside the image and subpixel-refine charuco corners
276 selectAndRefineChessboardCorners(allChessboardImgPoints, image, charucoCorners, charucoIds, subPixWinSizes);
277 }

280 int filterCornersWithoutMinMarkers(InputArray _allCharucoCorners, InputArray allCharucoIds, InputArray allArucoIds,
281 OutputArray _filteredCharucoCorners, OutputArray _filteredCharucoIds) {
282 CV_Assert(charucoParameters.minMarkers >= 0 && charucoParameters.minMarkers <= 2);
283 vector<Point2f> filteredCharucoCorners;
284 vector<int> filteredCharucoIds;
285 const vector<vector<int> > nearestMarkerIdx = board.getNearestMarkerIdx(); // only copy the vectors once
286 // for each charuco corner
287 for(unsigned int i = 0; i < allCharucoIds.getMat().total(); i++) {
288 int currentCharucoId = allCharucoIds.getMat().at<int>(i);
289 int totalMarkers = 0; // nomber of closest marker detected
290 // look for closest markers
291 for(unsigned int m = 0; m < nearestMarkerIdx[currentCharucoId].size(); m++) {
292 int markerId = board.getIds()[nearestMarkerIdx[currentCharucoId][m]];
293 bool found = false;
294 for(unsigned int k = 0; k < allArucoIds.getMat().total(); k++) {
295 if(allArucoIds.getMat().at<int>(k) == markerId) {
296 found = true;
297 break;
298 }
299 }
300 if(found) totalMarkers++;
301 }
302 // if enough markers detected, add the charuco corner to the final list
303 if(totalMarkers >= charucoParameters.minMarkers) {
304 filteredCharucoIds.push_back(currentCharucoId);
305 filteredCharucoCorners.push_back(_allCharucoCorners.getMat().at<Point2f>(i));
306 }
307 }
308 // parse output
309 Mat(filteredCharucoCorners).copyTo(_filteredCharucoCorners);
310 Mat(filteredCharucoIds).copyTo(_filteredCharucoIds);
311 return (int)_filteredCharucoIds.total();
312 }

314 void detectBoard(InputArray image, OutputArray charucoCorners, OutputArray charucoIds,
315 InputOutputArrayOfArrays markerCorners, InputOutputArray markerIds) {
316 CV_Assert((markerCorners.empty() && markerIds.empty() && !image.empty()) || (markerCorners.total() == markerIds.total()));
317 vector<vector<Point2f>> tmpMarkerCorners;
318 vector<int> tmpMarkerIds;
319 InputOutputArrayOfArrays _markerCorners = markerCorners.needed() ? markerCorners : _InputOutputArray(tmpMarkerCorners);
320 InputOutputArray _markerIds = markerIds.needed() ? markerIds : _InputOutputArray(tmpMarkerIds);
321
322 if (markerCorners.empty() && markerIds.empty()) {
323 vector<vector<Point2f> > rejectedMarkers;
324 arucoDetector.detectMarkers(image, _markerCorners, _markerIds, rejectedMarkers);
325 if (charucoParameters.tryRefineMarkers)
326 arucoDetector.refineDetectedMarkers(image, board, _markerCorners, _markerIds, rejectedMarkers);
327 if (_markerCorners.empty() && _markerIds.empty())
328 return;
329 }
330 // if camera parameters are avaible, use approximated calibration
331 if(!charucoParameters.cameraMatrix.empty())
332 interpolateCornersCharucoApproxCalib(_markerCorners, _markerIds, image, charucoCorners, charucoIds);
333 // else use local homography
334 else
335 interpolateCornersCharucoLocalHom(_markerCorners, _markerIds, image, charucoCorners, charucoIds);
336 // to return a charuco corner, its closest aruco markers should have been detected
337 filterCornersWithoutMinMarkers(charucoCorners, charucoIds, _markerIds, charucoCorners, charucoIds);
338 }

398void CharucoDetector::detectDiamonds(InputArray image, OutputArrayOfArrays _diamondCorners, OutputArray _diamondIds,
399 InputOutputArrayOfArrays inMarkerCorners, InputOutputArray inMarkerIds) const {
400 CV_Assert(getBoard().getChessboardSize() == Size(3, 3));
401 CV_Assert((inMarkerCorners.empty() && inMarkerIds.empty() && !image.empty()) || (inMarkerCorners.total() == inMarkerIds.total()));
402
403 vector<vector<Point2f>> tmpMarkerCorners;
404 vector<int> tmpMarkerIds;
405 InputOutputArrayOfArrays _markerCorners = inMarkerCorners.needed() ? inMarkerCorners : _InputOutputArray(tmpMarkerCorners);
406 InputOutputArray _markerIds = inMarkerIds.needed() ? inMarkerIds : _InputOutputArray(tmpMarkerIds);
407 if (_markerCorners.empty() && _markerIds.empty()) {
408 charucoDetectorImpl->arucoDetector.detectMarkers(image, _markerCorners, _markerIds);
409 }
410
411 const float minRepDistanceRate = 1.302455f;
412 vector<vector<Point2f>> diamondCorners;
413 vector<Vec4i> diamondIds;
414
415 // stores if the detected markers have been assigned or not to a diamond
416 vector<bool> assigned(_markerIds.total(), false);
417 if(_markerIds.total() < 4ull)
418 {
419 if (_diamondCorners.needed()) _diamondCorners.release();
420 if (_diamondIds.needed()) _diamondIds.release();
421 return; // a diamond need at least 4 markers
422 }
423
424 // convert input image to grey
425 Mat grey;
426 if(image.type() == CV_8UC3)
427 cvtColor(image, grey, COLOR_BGR2GRAY);
428 else
429 grey = image.getMat();
430 auto board = getBoard();
431
432 unsigned int nmarkers = (unsigned int)_markerCorners.total();
433 std::vector<std::vector<Point2f>> markerCorners(nmarkers);
434 for(unsigned int i = 0; i < nmarkers; i++)
435 _markerCorners.getMat((int)i).copyTo(markerCorners[i]);
436
437 // for each of the detected markers, try to find a diamond
438 for(unsigned int i = 0; i < (unsigned int)_markerIds.total(); i++) {
439 if(assigned[i]) continue;
440
441 // calculate marker perimeter
442 float perimeterSq = 0;
443 for(int c = 0; c < 4; c++) {
444 Point2f edge = markerCorners[i][c] - markerCorners[i][(c + 1) % 4];
445 perimeterSq += edge.x*edge.x + edge.y*edge.y;
446 }
447 // maximum reprojection error relative to perimeter
448 float minRepDistance = sqrt(perimeterSq) * minRepDistanceRate;
449
450 int currentId = _markerIds.getMat().at<int>(i);
451
452 // prepare data to call refineDetectedMarkers()
453 // detected markers (only the current one)
454 vector<vector<Point2f> > currentMarker;
455 vector<int> currentMarkerId;
456 currentMarker.push_back(markerCorners[i]);
457 currentMarkerId.push_back(currentId);
458
459 // marker candidates (the rest of markers if they have not been assigned)
460 vector<vector<Point2f> > candidates;
461 vector<int> candidatesIdxs;
462 for(unsigned int k = 0; k < assigned.size(); k++) {
463 if(k == i) continue;
464 if(!assigned[k]) {
465 candidates.push_back(markerCorners[k]);
466 candidatesIdxs.push_back(k);
467 }
468 }
469 if(candidates.size() < 3ull) break; // we need at least 3 free markers
470 // modify charuco layout id to make sure all the ids are different than current id
471 vector<int> tmpIds(4ull);
472 for(int k = 1; k < 4; k++)
473 tmpIds[k] = currentId + 1 + k;
474 // current id is assigned to [0], so it is the marker on the top
475 tmpIds[0] = currentId;
476
477 // create Charuco board layout for diamond (3x3 layout)
478 charucoDetectorImpl->board = CharucoBoard(Size(3, 3), board.getSquareLength(),
479 board.getMarkerLength(), board.getDictionary(), tmpIds);
480
481 // try to find the rest of markers in the diamond
482 vector<int> acceptedIdxs;
483 if (currentMarker.size() != 4ull)
484 {
485 RefineParameters refineParameters(minRepDistance, -1.f, false);
486 RefineParameters tmp = charucoDetectorImpl->arucoDetector.getRefineParameters();
487 charucoDetectorImpl->arucoDetector.setRefineParameters(refineParameters);
488 charucoDetectorImpl->arucoDetector.refineDetectedMarkers(grey, getBoard(), currentMarker, currentMarkerId,
489 candidates,
490 noArray(), noArray(), acceptedIdxs);
491 charucoDetectorImpl->arucoDetector.setRefineParameters(tmp);
492 }
493
494 // if found, we have a diamond
495 if(currentMarker.size() == 4ull) {
496 assigned[i] = true;
497 // calculate diamond id, acceptedIdxs array indicates the markers taken from candidates array
498 Vec4i markerId;
499 markerId[0] = currentId;
500 for(int k = 1; k < 4; k++) {
501 int currentMarkerIdx = candidatesIdxs[acceptedIdxs[k - 1]];
502 markerId[k] = _markerIds.getMat().at<int>(currentMarkerIdx);
503 assigned[currentMarkerIdx] = true;
504 }
505
506 // interpolate the charuco corners of the diamond
507 vector<Point2f> currentMarkerCorners;
508 Mat aux;
509 charucoDetectorImpl->detectBoardWithCheck(grey, currentMarkerCorners, aux, currentMarker, currentMarkerId);
510
511 // if everything is ok, save the diamond
