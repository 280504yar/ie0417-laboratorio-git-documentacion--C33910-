53 MatShape getOutShape(const MatShape& inpShape, const std::vector<int>& axes_) const
54 {
55 bool squeezeMask[MatShape::MAX_DIMS];
56
57 if (axes_.empty()) {
58 // remove all 1's
59 for (int i = 0; i < inpShape.dims; i++)
60 squeezeMask[i] = inpShape[i] == 1;
61 } else {
62 for (int i = 0; i < inpShape.dims; i++)
63 squeezeMask[i] = false;
64 for (int a: axes_) {
65 int a_ = normalize_axis(a, inpShape.dims);
66 if (squeezeMask[a_]) {
67 CV_Error_(Error::StsBadArg, ("duplicate squeezed axis #%d", a));
68 }
69 if (inpShape[a_] != 1) {
70 CV_Error_(Error::StsBadArg, ("squeezed axis #%d (== %d) != 1", a, inpShape[a_]));
71 }
72 squeezeMask[a_] = true;
73 }
74 }
75
76 MatShape outShape(inpShape.dims);
77 int j = 0;
78 for (int i = 0; i < inpShape.dims; i++) {
79 if (!squeezeMask[i])
80 outShape[j++] = inpShape[i];
81 }
82 outShape.dims = j;
83 return outShape;
84 }

92 {
93 CV_Assert(inputs.size() == 1 || inputs.size() == 2);

95 std::vector<int> tempAxes;

102 Net::Impl* netimpl_ = getNetImpl(this);
103 Mat axesTensor = netimpl_->argTensor(this->inputs[1]);
104 tensorToIntVec(axesTensor, tempAxes);
105 axes_ = &tempAxes;
106 }
107 outputs.assign(1, getOutShape(inputs[0], *axes_));
108 internals.clear();
109 return true;

132 {
133 CV_TRACE_FUNCTION();
134 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
135
136 Size size = inputs_arr.size();
137 int ninputs = size.area();
138 CV_Assert(ninputs == 1 || ninputs == 2);
139
140 MatShape inpShape = inputs_arr.shape(0);
141 std::vector<int> tempAxes;
142 const std::vector<int>* axes_ = &axes;
143
144 if (ninputs == 2)
145 {
146 CV_Assert(axes.empty()); // if we have a dedicated 'axes' input,
147 // we should not have 'axes' attribute at the same time
148 Mat axesTensor = inputs_arr.getMat(1);
149 tensorToIntVec(axesTensor, tempAxes);
150 axes_ = &tempAxes;
151 }
152 MatShape outShape = getOutShape(inpShape, *axes_);
153 reshapeAndCopyFirst(inputs_arr, outputs_arr, outShape);
154 }
