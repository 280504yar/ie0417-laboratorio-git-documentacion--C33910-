24template<typename Ti, typename T> static void
25LUT_( const Ti* src, const T* lut, T* dst, const int len, const int cn, const int lutcn )
26{
27 if( lutcn == 1 )
28 {
29 for( int i = 0; i < len*cn; i++ )
30 dst[i] = lut[src[i]];
31 }
32 else
33 {
34 for( int i = 0; i < len*cn; i += cn )
35 for( int k = 0; k < cn; k++ )
36 dst[i+k] = lut[src[i+k]*cn+k];
37 }
38}

42static LUTFunc getLUTFunc(const int srcDepth, const int dstDepth)
43{
44 LUTFunc ret = nullptr;
45 if((srcDepth == CV_8U) || (srcDepth == CV_8S))
46 {
47 switch(dstDepth)
48 {
49 case CV_8U: ret = (LUTFunc)LUT8u_dispatch; break;
50 case CV_8S: ret = (LUTFunc)LUT_<uint8_t, int8_t>; break;
51 case CV_16U: ret = (LUTFunc)LUT16u_dispatch; break;
52 case CV_16S: ret = (LUTFunc)LUT_<uint8_t, int16_t>; break;
53 case CV_32S: ret = (LUTFunc)LUT_<uint8_t, int32_t>; break;
54 case CV_32F: ret = (LUTFunc)LUT_<uint8_t, int32_t>; break; // float
55 case CV_64F: ret = (LUTFunc)LUT_<uint8_t, int64_t>; break; // double
56 case CV_16F: ret = (LUTFunc)LUT_<uint8_t, int16_t>; break; // hfloat
57 case CV_16BF: ret = (LUTFunc)LUT_<uint8_t, int16_t>; break; // bfloat
58 case CV_Bool: ret = (LUTFunc)LUT_<uint8_t, uint8_t>; break; // bool
59 case CV_64U: ret = (LUTFunc)LUT_<uint8_t, uint64_t>; break;
60 case CV_64S: ret = (LUTFunc)LUT_<uint8_t, int64_t>; break;
61 case CV_32U: ret = (LUTFunc)LUT_<uint8_t, uint32_t>; break;
62 default: ret = nullptr; break;
63 }
64 }
65 else if((srcDepth == CV_16U) || (srcDepth == CV_16S))
66 {
67 switch(dstDepth)
68 {
69 case CV_8U: ret = (LUTFunc)LUT_<uint16_t, uint8_t>; break;
70 case CV_8S: ret = (LUTFunc)LUT_<uint16_t, int8_t>; break;
71 case CV_16U: ret = (LUTFunc)LUT_<uint16_t, uint16_t>; break;
72 case CV_16S: ret = (LUTFunc)LUT_<uint16_t, int16_t>; break;
73 case CV_32S: ret = (LUTFunc)LUT_<uint16_t, int32_t>; break;
74 case CV_32F: ret = (LUTFunc)LUT_<uint16_t, int32_t>; break; // float
75 case CV_64F: ret = (LUTFunc)LUT_<uint16_t, int64_t>; break; // double
76 case CV_16F: ret = (LUTFunc)LUT_<uint16_t, int16_t>; break; // hfloat
77 case CV_16BF: ret = (LUTFunc)LUT_<uint16_t, int16_t>; break; // bfloat
78 case CV_Bool: ret = (LUTFunc)LUT_<uint16_t, uint8_t>; break; // bool
79 case CV_64U: ret = (LUTFunc)LUT_<uint16_t, uint64_t>; break;
80 case CV_64S: ret = (LUTFunc)LUT_<uint16_t, int64_t>; break;
81 case CV_32U: ret = (LUTFunc)LUT_<uint16_t, uint32_t>; break;
82 default: ret = nullptr; break;
83 }
84 }
85
86 CV_CheckTrue(ret != nullptr, "An unexpected type combination was specified.");
87 return ret;
88}

92static bool ocl_LUT(InputArray _src, InputArray _lut, OutputArray _dst)
93{
94 int lcn = _lut.channels(), dcn = _src.channels(), ddepth = _lut.depth();
95
96 UMat src = _src.getUMat(), lut = _lut.getUMat();
97 _dst.createSameSize(src, CV_MAKETYPE(ddepth, dcn));
98 UMat dst = _dst.getUMat();
99 int kercn = lcn == 1 ? std::min(4, ocl::predictOptimalVectorWidth(_src, _dst)) : dcn;
100
101 ocl::Kernel k("LUT", ocl::core::lut_oclsrc,
102 format("-D dcn=%d -D lcn=%d -D srcT=%s -D dstT=%s", kercn, lcn,
103 ocl::typeToStr(src.depth()), ocl::memopTypeToStr(ddepth)));
104 if (k.empty())
105 return false;
106
107 k.args(ocl::KernelArg::ReadOnlyNoSize(src), ocl::KernelArg::ReadOnlyNoSize(lut),
108 ocl::KernelArg::WriteOnly(dst, dcn, kercn));
109
110 size_t globalSize[2] = { (size_t)dst.cols * dcn / kercn, ((size_t)dst.rows + 3) / 4 };
111 return k.run(2, globalSize, NULL, false);
112}

156void cv::LUT( InputArray _src, InputArray _lut, OutputArray _dst )
157{
158 CV_INSTRUMENT_REGION();
159
160 int cn = _src.channels(), depth = _src.depth();
161 int lutcn = _lut.channels();
162 const size_t lut_size = _lut.total();
163
164 CV_Assert( (lutcn == cn || lutcn == 1) && _lut.isContinuous() &&
165 (
166 ((lut_size == 256) && ((depth == CV_8U)||(depth == CV_8S))) ||
167 ((lut_size == 65536) && ((depth == CV_16U)||(depth == CV_16S)))
168 )
169 );
170
171 CV_OCL_RUN(_dst.isUMat() && _src.dims() <= 2 && (lut_size == 256),
172 ocl_LUT(_src, _lut, _dst))
173
174 Mat src = _src.getMat(), lut = _lut.getMat();
175 _dst.createSameSize(_src, CV_MAKETYPE(_lut.depth(), cn));
176 Mat dst = _dst.getMat();
177
178 if(lut_size == 256)
179 {
180 CALL_HAL(LUT, cv_hal_lut, src.data, src.step, src.type(), lut.data,
181 lut.elemSize1(), lutcn, dst.data, dst.step, src.cols, src.rows);
182 }
183 else
184 {
185 CALL_HAL(LUT16, cv_hal_lut16, src.ptr<ushort>(), src.step, src.type(), lut.data,
186 lut.elemSize1(), lutcn, dst.data, dst.step, src.cols, src.rows);
187 }
188
189 const LUTFunc func = getLUTFunc(src.depth(), dst.depth());
190 CV_Assert( func != nullptr );
191
192 if (_src.dims() <= 2)
193 {
194 LUTParallelBody body(src, lut, dst, func);
195 Range all(0, dst.rows);
196 if (dst.total() >= (size_t)(1<<18))
197 parallel_for_(all, body, (double)std::max((size_t)1, dst.total()>>16));
198 else
199 body(all);
200
201 return;
202 }
203
204 const Mat* arrays[] = {&src, &dst, 0};
205 uchar* ptrs[2] = {};
206 NAryMatIterator it(arrays, ptrs);
207 int len = (int)it.size;
208
209 for( size_t i = 0; i < it.nplanes; i++, ++it )
210 func(ptrs[0], lut.ptr(), ptrs[1], len, cn, lutcn);
211}
