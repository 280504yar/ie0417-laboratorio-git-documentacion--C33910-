62inline float LogAdd(float x, float y) {
63 float diff;
64 if (x < y) {
65 diff = x - y;
66 x = y;
67 } else {
68 diff = y - x;
69 }
70 // diff is negative. x is now the larger one.
71
72 if (diff >= kMinLogDiffFloat) {
73 float res;
74 res = x + Log1p(Exp(diff));
75 return res;
76 } else {
77 return x; // return the larger one.
78 }
79}
