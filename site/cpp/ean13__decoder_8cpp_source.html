23Result Ean13Decoder::decode(const vector<uchar> &data) const
24{
25 string result;
26 char decode_result[EAN13DIGIT_NUM + 1]{'\0'};
27 if (data.size() < EAN13BITS_NUM)
28 {
29 return Result("Wrong Size", Result::BARCODE_NONE);
30 }
31 pair<uint, uint> pattern;
32 if (!findStartGuardPatterns(data, pattern))
33 {
34 return Result("Begin Pattern Not Found", Result::BARCODE_NONE);
35 }
36 uint start = pattern.second;
37 Counter counter(vector<int>{0, 0, 0, 0});
38 size_t end = data.size();
39 int first_char_bit = 0;
40 // [1,6] are left part of EAN, [7,12] are right part, index 0 is calculated by left part
41 for (int i = 1; i < 7 && start < end; ++i)
42 {
43 int bestMatch = decodeDigit(data, counter, start, get_AB_Patterns());
44 if (bestMatch == -1)
45 {
46 return Result("Decode Error", Result::BARCODE_NONE);
47 }
48 decode_result[i] = static_cast<char>('0' + bestMatch % 10);
49 start = counter.sum + start;
50 first_char_bit += (bestMatch >= 10) << i;
51 }
52 decode_result[0] = static_cast<char>(FIRST_CHAR_ARRAY()[first_char_bit >> 2] + '0');
53 // why there need >> 2?
54 // first, the i in for-cycle is begin in 1
55 // second, the first i = 1 is always
56 Counter middle_counter(vector<int>(MIDDLE_PATTERN().size()));
57 if (!findGuardPatterns(data, start, true, MIDDLE_PATTERN(), middle_counter, pattern))
58 {
59 return Result("Middle Pattern Not Found", Result::BARCODE_NONE);
60
61 }
62 start = pattern.second;
63 for (int i = 0; i < 6 && start < end; ++i)
64 {
65 int bestMatch = decodeDigit(data, counter, start, get_A_or_C_Patterns());
66 if (bestMatch == -1)
67 {
68 return Result("Decode Error", Result::BARCODE_NONE);
69 }
70 decode_result[i + 7] = static_cast<char>('0' + bestMatch);
71 start = counter.sum + start;
72 }
73 Counter end_counter(vector<int>(BEGIN_PATTERN().size()));
74 if (!findGuardPatterns(data, start, false, BEGIN_PATTERN(), end_counter, pattern))
75 {
76 return Result("End Pattern Not Found", Result::BARCODE_NONE);
77 }
78 result = string(decode_result);
79 if (!isValid(result))
80 {
81 return Result("Wrong: " + result.append(string(EAN13DIGIT_NUM - result.size(), ' ')), Result::BARCODE_NONE);
82 }
83 return Result(result, Result::BARCODE_EAN_13);
84}
