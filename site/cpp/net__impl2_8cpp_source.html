24struct OrtNamesCache
25{
26 std::vector<std::string> input_names;
27 std::vector<std::string> output_names;
28 std::unordered_map<std::string, int> input_name_to_index;
29 std::unordered_map<std::string, int> output_name_to_index;
30
31 explicit OrtNamesCache(Ort::Session& session)
32 {
33 Ort::AllocatorWithDefaultOptions allocator;
34
35 const size_t ninputs = session.GetInputCount();
36 input_names.reserve(ninputs);
37 for (size_t i = 0; i < ninputs; ++i)
38 {
39 Ort::AllocatedStringPtr in = session.GetInputNameAllocated(i, allocator);
40 std::string n = in ? std::string(in.get()) : std::string();
41 input_name_to_index[n] = (int)i;
42 input_names.push_back(std::move(n));
43 }
44
45 const size_t noutputs = session.GetOutputCount();
46 output_names.reserve(noutputs);
47 for (size_t i = 0; i < noutputs; ++i)
48 {
49 Ort::AllocatedStringPtr out = session.GetOutputNameAllocated(i, allocator);
50 std::string n = out ? std::string(out.get()) : std::string();
51 output_name_to_index[n] = (int)i;
52 output_names.push_back(std::move(n));
53 }
54 }
55};

59void Net::Impl::applyStagedOrtInputs()
60{
61 if (!ort_session || ort_staged_inputs.empty())
62 return;
63
64 if (!ort_names_cache)
65 ort_names_cache = std::make_shared<OrtNamesCache>(*ort_session);
66
67 OrtNamesCache& names = *ort_names_cache;
68 const size_t ninputs = names.input_names.size();
69 if (ninputs == 0)
70 CV_Error(Error::StsError, "DNN/ORT: ORT session has no inputs");
71
72 if (!netInputLayer)
73 {
74 netInputLayer = Ptr<DataLayer>(new DataLayer());
75 netInputLayer->name = "ort_data_layer";
76 netInputLayer->type = "Data";
77 }
78 if (netInputLayer->blobs.size() != ninputs)
79 netInputLayer->blobs.resize(ninputs);
80
81 for (size_t k = 0; k < ort_staged_inputs.size(); ++k)
82 {
83 const std::string& inpname = ort_staged_inputs[k].first;
84 const Mat& inputMat = ort_staged_inputs[k].second;
85 if (inputMat.empty())
86 CV_Error(Error::StsBadArg, "DNN/ORT: Input blob is empty");
87
88 size_t inputIdx = 0;
89 if (inpname.empty())
90 {
91 if (ninputs != 1)
92 CV_Error(Error::StsBadArg, "DNN/ORT: input name must be specified for models with multiple inputs");
93 inputIdx = 0;
94 }
95 else
96 {
97 auto it = names.input_name_to_index.find(inpname);
98 if (it == names.input_name_to_index.end())
99 CV_Error_(Error::StsObjectNotFound, ("DNN/ORT: input '%s' is not found", inpname.c_str()));
100 inputIdx = (size_t)it->second;
101 }
102 inputMat.copyTo(netInputLayer->blobs[inputIdx]);
103 }
104 ort_staged_inputs.clear();
105}

107static int cvTypeFromONNXElemType(const ONNXTensorElementDataType t)
108{
109 switch (t)
110 {
111 case ONNX_TENSOR_ELEMENT_DATA_TYPE_FLOAT: return CV_32F;
112 case ONNX_TENSOR_ELEMENT_DATA_TYPE_UINT8: return CV_8U;
113 case ONNX_TENSOR_ELEMENT_DATA_TYPE_INT8: return CV_8S;
114 case ONNX_TENSOR_ELEMENT_DATA_TYPE_UINT16: return CV_16U;
115 case ONNX_TENSOR_ELEMENT_DATA_TYPE_INT16: return CV_16S;
116 case ONNX_TENSOR_ELEMENT_DATA_TYPE_INT32: return CV_32S;
117 case ONNX_TENSOR_ELEMENT_DATA_TYPE_INT64: return CV_64S;
118 case ONNX_TENSOR_ELEMENT_DATA_TYPE_BOOL: return CV_8U;
119 case ONNX_TENSOR_ELEMENT_DATA_TYPE_DOUBLE: return CV_64F;
120 case ONNX_TENSOR_ELEMENT_DATA_TYPE_FLOAT16: return CV_16F;
121 case ONNX_TENSOR_ELEMENT_DATA_TYPE_BFLOAT16:return CV_16BF;
122 case ONNX_TENSOR_ELEMENT_DATA_TYPE_UINT32: return CV_32U;
123 case ONNX_TENSOR_ELEMENT_DATA_TYPE_UINT64: return CV_64U;
124 default:
125 return -1;

129static Ort::Value createOrtTensorFromMat(Ort::Session& session,
130 size_t inputIdx,
131 const Ort::MemoryInfo& memory_info,
132 Mat& inputBlob,
133 std::vector<int64_t>& inputDims,
134 ONNXTensorElementDataType& in_elem_type)
135{
136 Ort::TypeInfo in_type_info = session.GetInputTypeInfo(inputIdx);
137 Ort::ConstTensorTypeAndShapeInfo in_tensor_info = in_type_info.GetTensorTypeAndShapeInfo();
138 in_elem_type = in_tensor_info.GetElementType();
139
140 const int cvInType = cvTypeFromONNXElemType(in_elem_type);
141 if (cvInType < 0)
142 CV_Error_(Error::StsNotImplemented, ("DNN/ORT: unsupported ORT input element type: %d", (int)in_elem_type));
143
144 if (inputBlob.type() != cvInType)
145 inputBlob.convertTo(inputBlob, cvInType);
146
147 if (!inputBlob.isContinuous())
148 inputBlob = inputBlob.clone();
149
150 inputDims.clear();
151 inputDims.reserve((size_t)inputBlob.dims);
152 for (int i = 0; i < inputBlob.dims; i++)
153 inputDims.push_back((int64_t)inputBlob.size[i]);
154
155 const size_t nbytes = (size_t)inputBlob.total() * inputBlob.elemSize();
156 OrtValue* input_tensor_raw = nullptr;
157 Ort::ThrowOnError(Ort::GetApi().CreateTensorWithDataAsOrtValue(
158 memory_info,
159 inputBlob.data,
160 nbytes,
161 inputDims.data(),
162 inputDims.size(),
163 in_elem_type,
164 &input_tensor_raw));
165 return Ort::Value(input_tensor_raw);
166}

168std::vector<Mat> Net::Impl::runOrtSession(std::vector<Mat> inputBlobs, const std::vector<int>& outIdxs)
169{
170 CV_Assert(this->ort_session);
171 Ort::Session& session = *this->ort_session;
172
173 if (!this->ort_names_cache)
174 this->ort_names_cache = std::make_shared<OrtNamesCache>(session);
175
176 OrtNamesCache& names = *this->ort_names_cache;
177 if (names.input_names.empty())
178 CV_Error(Error::StsError, "DNN/ORT: ORT session has no inputs");
179 if (names.output_names.empty())
180 CV_Error(Error::StsError, "DNN/ORT: ORT session has no outputs");
181
182 const size_t ninputs = names.input_names.size();
183 if (inputBlobs.size() != ninputs)
184 CV_Error_(Error::StsBadArg, ("DNN/ORT: expected %zu inputs, but got %zu", ninputs, inputBlobs.size()));
185
186 std::vector<const char*> in_names;
187 in_names.reserve(ninputs);
188 for (size_t i = 0; i < ninputs; ++i)
189 in_names.push_back(names.input_names[i].c_str());
190
191 std::vector<const char*> out_names;
192 if (outIdxs.empty())
193 {
194 out_names.reserve(names.output_names.size());
195 for (const std::string& n : names.output_names)
196 out_names.push_back(n.c_str());
197 }
198 else
199 {
200 out_names.reserve(outIdxs.size());
201 for (int idx : outIdxs)
202 {
203 CV_CheckGE(idx, 0, "DNN/ORT: output index must be non-negative");
204 CV_CheckLT((size_t)idx, names.output_names.size(), "DNN/ORT: output index is out of range");
205 out_names.push_back(names.output_names[(size_t)idx].c_str());
206 }
207 }
208
209 static const Ort::MemoryInfo memory_info =
210 Ort::MemoryInfo::CreateCpu(OrtDeviceAllocator, OrtMemTypeCPU);
211
212 std::vector<Ort::Value> input_tensors;
213 input_tensors.reserve(ninputs);
214 for (size_t i = 0; i < ninputs; ++i)
215 {
216 if (inputBlobs[i].empty())
217 CV_Error_(Error::StsError, ("DNN/ORT: input '%s' is empty", names.input_names[i].c_str()));
218
219 std::vector<int64_t> inputDims;
220 ONNXTensorElementDataType in_elem_type = ONNX_TENSOR_ELEMENT_DATA_TYPE_UNDEFINED;
221 input_tensors.push_back(createOrtTensorFromMat(session, i, memory_info, inputBlobs[i], inputDims, in_elem_type));
222 }
223
224 std::vector<Ort::Value> output_tensors = session.Run(
225 Ort::RunOptions{nullptr},
226 in_names.data(), input_tensors.data(), input_tensors.size(),
227 out_names.data(), out_names.size());
228 if (profilingMode != DNN_PROFILE_NONE) ort_profile_runs++;
229
230 CV_CheckEQ(output_tensors.size(), out_names.size(), "DNN/ORT: ORT returned unexpected number of outputs");
231
232 std::vector<Mat> results;
233 results.reserve(output_tensors.size());
234
235 for (Ort::Value& outv : output_tensors)
236 {
237 Ort::TensorTypeAndShapeInfo shape_info = outv.GetTensorTypeAndShapeInfo();
238 std::vector<int64_t> out_shape = shape_info.GetShape();
239
240 std::vector<int> out_dims;
241 out_dims.reserve(out_shape.size());
242 for (int64_t d : out_shape)
243 {
244 if (d < 0)
245 CV_Error(Error::StsError, "DNN/ORT: dynamic output shapes are not supported at runtime");
246 if (d > (int64_t)std::numeric_limits<int>::max())
247 CV_Error(Error::StsError, "DNN/ORT: output shape dimension is too large");
248 out_dims.push_back((int)d);
249 }
250
251 const ONNXTensorElementDataType out_elem_type = shape_info.GetElementType();
252 const int cvOutType = cvTypeFromONNXElemType(out_elem_type);
253 if (cvOutType < 0)
254 CV_Error_(Error::StsNotImplemented, ("DNN/ORT: unsupported ORT output element type: %d", (int)out_elem_type));
255
256 uint8_t* out_bytes = outv.GetTensorMutableData<uint8_t>();
257 Mat view(out_dims, cvOutType, out_bytes);
258 results.push_back(view.clone()); // detach from ORT-owned memory
259 }
260
261 return results;
262}

582void Net::Impl::finalizeGraph(const Ptr<Graph>& graph, bool useCUDA)
583{
584 GraphImpl* g = static_cast<GraphImpl*>(graph.get());
585 const std::vector<Ptr<LayerInfo> >& prog = g->prog_;
586 size_t i, nops = prog.size();
587 g->exec_.assign(nops, Ptr<Layer>());
588 g->execBackend_.assign(nops, DNN_BACKEND_OPENCV);
589
590#ifdef HAVE_CUDA
591 // Whole-graph CUDA gate: run on CUDA only if *every* op has a CUDA executor; if any op is
592 // unsupported (or the graph has control-flow subgraphs) run the entire graph on CPU. This
593 // avoids partial CPU<->CUDA execution and its host/device coherence hazards for now.
594 std::vector<Ptr<Layer> > cudaExecs;
595 bool graphOnCuda = false;
596 if (useCUDA && cudaInfo) {
597 graphOnCuda = true;
598 cudaExecs.assign(nops, Ptr<Layer>());
599 for (i = 0; i < nops; i++) {
600 const Ptr<LayerInfo>& op = prog[i];
601 if (!op)
602 continue;
603 if (op->subgraphs()) { graphOnCuda = false; break; } // control-flow bodies stay on CPU
604 Ptr<Layer> e = LayerFactory::createExec(op->type, DNN_BACKEND_CUDA, op, &cudaInfo->context);
605 if (!e) { graphOnCuda = false; break; }
606 cudaExecs[i] = e;
607 }
608 if (!graphOnCuda) {
609 cudaExecs.clear();
610 CV_LOG_INFO(NULL, cv::format("DNN/NewEngine: graph '%s' has layer(s) without CUDA support; "
611 "running the whole graph on CPU", graph->name().c_str()));
612 }
613 }
614#endif
615
616 for (i = 0; i < nops; i++) {
617 const Ptr<LayerInfo>& op = prog[i];
618 if (!op)
619 continue;
620
621 // recurse into subgraphs (If/Loop bodies) first
622 const std::vector<Ptr<Graph> >* subs = op->subgraphs();
623 if (subs) {
624 for (const Ptr<Graph>& sub : *subs)
625 finalizeGraph(sub, useCUDA);
626 }
627
628 Ptr<Layer> exec;
629 int backend = DNN_BACKEND_OPENCV;
630#ifdef HAVE_CUDA
631 if (graphOnCuda && cudaExecs[i]) {
632 exec = cudaExecs[i];
633 exec->preferableTarget = preferableTarget;
634 backend = DNN_BACKEND_CUDA;
635 }
636#endif
637 if (!exec) {
638 exec = LayerFactory::createExec(op->type, DNN_BACKEND_OPENCV, op, nullptr);
639 if (!exec)
640 exec = op.dynamicCast<Layer>();
641 backend = DNN_BACKEND_OPENCV;
642 }
643 CV_Assert(exec);
644 // Re-finalize can hand back the same object, so reset state for the new backend.
645 exec->packedWeightEpoch = 0;
646 exec->finalizedOnce = false;
647 exec->lastInpShapes.clear();
648 exec->lastInpTypes.clear();
649 g->exec_[i] = exec;
650 g->execBackend_[i] = backend;
651 CV_LOG_INFO(NULL, cv::format("DNN/NewEngine: finalize op #%zu '%s' (%s) -> %s",
652 i, op->name.c_str(), op->type.c_str(),
653 backend == DNN_BACKEND_CUDA ? "CUDA" : "CPU"));
654 }

659 fusedSnapshot.clear();
660 for (const Ptr<Graph>& g : allgraphs) {
661 FusedGraphSnapshot snap;
662 snap.graph = g;
663 const std::vector<Ptr<LayerInfo> >& prog = g->prog();
664 snap.prog = prog;
665 snap.inputs.reserve(prog.size());
666 snap.outputs.reserve(prog.size());
667 for (const Ptr<LayerInfo>& op : prog) {
668 snap.inputs.push_back(op ? op->inputs : std::vector<Arg>());
669 snap.outputs.push_back(op ? op->outputs : std::vector<Arg>());
670 }
671 fusedSnapshot.push_back(std::move(snap));
672 }
673}

675void Net::Impl::restoreFusedSnapshot()
676{
677 // Roll the graph back to its post-fusion state: undo the layer-input rewiring
678 // and remove the TransformLayout ops inserted by a previous useBlockLayout().
679 for (const FusedGraphSnapshot& snap : fusedSnapshot) {
680 for (size_t i = 0; i < snap.prog.size(); i++) {
681 const Ptr<LayerInfo>& op = snap.prog[i];
682 if (!op)
683 continue;
684 op->inputs = snap.inputs[i];
685 op->outputs = snap.outputs[i];
686 }
687 snap.graph->setProg(snap.prog);
688 }
689 totalLayers = updateGraphOfs(mainGraph, 0, true);
690}

692void Net::Impl::finalize()
693{
694#ifdef HAVE_ONNXRUNTIME
695 if (ort_session)
696 return; // ONNX Runtime manages its own execution session
697#endif
698 if (!mainGraph)
699 return;
700 if (!prepared)
701 prepareForInference();
702 if (finalized)
703 return;
704
705 // Snapshot the fused graph once so finalize() can re-run cleanly on a
706 // backend/target change (block layout + buffer assignment are destructive).
707 if (!fusedSnapshotValid) {
708 saveFusedSnapshot();
709 fusedSnapshotValid = true;
710 } else {
711 restoreFusedSnapshot();
712 }
713
714 bool useCUDA = false;
715#ifdef HAVE_CUDA
716 cudaArgBuffers.clear();
717 cudaArgHostDirty.clear();
718 cudaArgDeviceDirty.clear();
719 if (preferableBackend == DNN_BACKEND_CUDA && haveCUDA()) {
720 useCUDA = true;
721 if (!cudaInfo) {
722 cuda4dnn::csl::CSLContext context;
723 context.stream = cuda4dnn::csl::Stream(true);
724 context.cublas_handle = cuda4dnn::csl::cublas::Handle(context.stream);
725 context.cudnn_handle = cuda4dnn::csl::cudnn::Handle(context.stream);
726 auto d2h_stream = cuda4dnn::csl::Stream(true);
727 cudaInfo = std::unique_ptr<CudaInfo_t>(new CudaInfo_t(std::move(context), std::move(d2h_stream)));
728 }
729 }
730#endif
731
732 CV_LOG_INFO(NULL, cv::format("DNN/NewEngine: finalize() backend=%d target=%d useCUDA=%d over %zu graph(s)",
733 preferableBackend, preferableTarget, (int)useCUDA, allgraphs.size()));
734
735 for (const Ptr<Graph>& g : allgraphs)
736 finalizeGraph(g, useCUDA);
737 useBlockLayout();
738 assignBuffers();
739 totalLayers = updateGraphOfs(mainGraph, 0, true);
740
741 for (const Ptr<Graph>& g : allgraphs)
742 finalizeGraph(g, useCUDA);
743
744 finalized = true;
745}

747void Net::Impl::allocateLayerOutputs(
748 const Ptr<LayerInfo>& layer,
749 const std::vector<int>& inpTypes,
750 const std::vector<MatShape>& inpShapes,
751 std::vector<int>& outTypes,
752 std::vector<MatShape>& outShapes,
753 std::vector<std::pair<uchar*, size_t> >& outOrigData,
754 std::vector<Mat>& outputs,
755 std::vector<int>& tempTypes,
756 std::vector<MatShape>& tempShapes,
757 std::vector<Mat>& temps,
758 std::vector<Mat>& globalTemps,
759 bool useBufferPool)
760{
761 // In theory, when
762 // 1) useBufferPool==true,
763 // 2) the buffers in the pool are already big enough (e.g. when we already run inference a few times to let them grow)
764 // 3) getMemoryShapes() and getTypes() are implemented efficiently without any memory allocations
765 // the method allocateLayerOutputs() should not allocate any memory either.
766 //
767 // Well, currently it still may do so, because Mat::fit() may create e.g. 4D tensor on top of 1D buffer and then
768 // MatSize and MatStep will require dynamic memory allocation (those are very small buffers though).
769 // But we plan to make MatSize and MatStep lighter so that they don't use dynamic memory.
770 size_t noutputs = layer->outputs.size();
771 outShapes.clear();
772 outTypes.clear();
773 tempShapes.clear();
774 tempTypes.clear();
775 layer->getMemoryShapes(inpShapes, (int)noutputs, outShapes, tempShapes);
776 layer->getTypes(inpTypes, (int)noutputs, (int)tempShapes.size(), outTypes, tempTypes);
777 CV_Assert(tempShapes.size() == tempTypes.size());
778 CV_Assert(outShapes.size() == outTypes.size());
779 CV_Assert(outShapes.size() == noutputs);
780
781 for (int i = 0; i < (int)tempShapes.size(); i++)
782 CV_CheckGT(total(tempShapes[i]), (size_t)0, "");
783 outputs.assign(noutputs, Mat());
784 outOrigData.resize(noutputs);
785 for (size_t i = 0; i < noutputs; i++) {
786 Arg out = layer->outputs[i];
787 if (useBufferPool) {
788 Mat& out_t = argTensor(out);
789 out_t.fit(outShapes[i], outTypes[i]);
790 outputs[i] = out_t;
791 } else {
792 outputs[i].fit(outShapes[i], outTypes[i]);
793 }
794 outOrigData[i].first = outputs[i].u ? outputs[i].u->data : nullptr;
795 outOrigData[i].second = outputs[i].u ? outputs[i].u->size : 0;
796 }
797 // [TODO] probably there should be a smarter algorithm that e.g. sorts
798 // temp buffers by size in decreasing order and assigns global temps accordingly
799 // in order to minimize the total size of temp buffers
800 size_t ntemps = tempShapes.size();
801 temps.resize(ntemps);
802 globalTemps.resize(std::max(ntemps, globalTemps.size()));
803 for (size_t i = 0; i < ntemps; i++) {
804 globalTemps[i].fit(tempShapes[i], tempTypes[i]);
805 temps[i] = globalTemps[i];
806 }
807}

886void Net::Impl::forwardWithSingleOutput(const std::string& outname, OutputArrayOfArrays outputBlobs)
887{
888#ifdef HAVE_ONNXRUNTIME
889 if (useOrtEngine && mainGraph && modelFormat == DNN_MODEL_ONNX && !modelFileName.empty())
890 finalizeOrt();
891 if (this->ort_session)
892 {
893 if (!netInputLayer || netInputLayer->blobs.empty())
894 CV_Error(Error::StsError, "DNN/ORT: No input data found");
895
896 if (!this->ort_names_cache)
897 this->ort_names_cache = std::make_shared<OrtNamesCache>(*this->ort_session);
898
899 int outIdx = 0;

901 {
902 OrtNamesCache& names = *this->ort_names_cache;
903 auto it = names.output_name_to_index.find(outname);
904 if (it == names.output_name_to_index.end())
905 CV_Error_(Error::StsObjectNotFound, ("DNN/ORT: output '%s' is not found", outname.c_str()));
906 outIdx = it->second;

908
909 std::vector<int> outIdxs(1, outIdx);

914 }

916 {

922 if (it == argnames.end()) {
923 size_t excl = outname.rfind('!');

925 it = argnames.find(outname.substr(excl + 1));
926 }
927 }
928 if (it == argnames.end())
929 CV_Error_(Error::StsObjectNotFound,
930 ("DNN: tensor '%s' is not found in the graph", outname.c_str()));
931
932 Arg targetArg((int)it->second);
933
934 std::vector<Mat> inps, outs;
935 forwardMainGraph(inps, outs);
936
937 const std::vector<Arg>& gr_outputs = mainGraph->outputs();
938 for (size_t i = 0; i < gr_outputs.size(); i++) {
939 if (gr_outputs[i].idx == targetArg.idx) {
940 assignSingleOutput(outputBlobs, outs[i]);
941 return;
942 }
943 }
944
945 const ArgData& adata = args.at(targetArg.idx);
946 Mat result;
947 if (adata.kind == DNN_ARG_TEMP) {
948 int bufidx = bufidxs.at(targetArg.idx);
949 CV_Assert(bufidx >= 0 && bufidx < (int)buffers.size());
950 result = buffers[bufidx];
951 } else {
952 result = __tensors__.at(targetArg.idx);
953 }
954 if (result.shape().layout == DATA_LAYOUT_BLOCK) {
955 Mat converted;
956 transformLayout(result, converted, originalLayout, originalLayout, defaultC0);
957 assignSingleOutput(outputBlobs, converted);
958 } else {
959 assignSingleOutput(outputBlobs, result.clone());
960 }
961 return;
962 }
963 }
964
965 std::vector<Mat> inps, outs;
966 forwardMainGraph(inps, outs);
967 CV_Assert(!outs.empty());
968 assignSingleOutput(outputBlobs, outs[0]);
969}

1248 return;
1249 }
1250 Arg inp = gr_inputs[idx];
1251 const ArgData& adata = args.at(inp.idx);
1252 /*
1253 [TODO] add more detailed shape check
1254 if (adata.shape.dims != mshape.dims) {
1255 CV_Error_(Error::StsBadArg, ("wrong dimensionality of argument '%s': %d given, %d expected",
1256 adata.name.c_str(), tsize.ndims, adata.size.ndims));

1260 checkAndUpdateDim(graph, Node(), inp, k, tsize.size[k]);

1266 if ((adata_type == CV_16F || adata_type == CV_16BF) && !enableFP16)
1267 adata_type = CV_32F;

1270 !((adata_type == CV_64F || adata_type == CV_32F || adata_type == CV_16F || adata_type == CV_16BF) &&
1271 (mtype == CV_64F || mtype == CV_32F || mtype == CV_16F || mtype == CV_16BF)) &&
1272 !((adata_type == CV_8U || adata_type == CV_8S || adata_type == CV_16U || adata_type == CV_16S || adata_type == CV_32S || adata_type == CV_32U || adata_type == CV_64S || adata_type == CV_64U) &&
1273 (mtype == CV_8U || mtype == CV_8S || mtype == CV_16U || mtype == CV_16S || mtype == CV_32S || mtype == CV_32U || mtype == CV_64S || mtype == CV_64U)) &&
1274 !(adata.type == CV_16BF && mtype == CV_16U) && !(adata.type == CV_16F && mtype == CV_16U) &&
1275 !m.empty())
1276 {
1277 CV_Error_(Error::StsBadArg, ("incompatible type of input tensor #%zu '%s': %s given, %s expected",
1278 idx, adata.name.c_str(), typeToString(mtype).c_str(),
1279 typeToString(adata.type).c_str()));
1280 }

1282 // The op loop detects signature changes per layer; no global flag needed.
1283 inp_t.fit(mshape, adata_type);
1284
1285 if (adata.type == CV_16BF && mtype == CV_16U)
1286 {
1287 Mat tmp(mshape, CV_16BF, (void*)m.data);
1288 tmp.convertTo(inp_t, adata_type);
1289 }
1290 else if (adata.type == CV_16F && mtype == CV_16U)
1291 {
1292 Mat tmp(mshape, CV_16F, (void*)m.data);
1293 tmp.convertTo(inp_t, adata_type);
1294 }
1295 else
1296 {
1297 m.convertTo(inp_t, adata_type);
1298 }
1299 } else if (adata.kind == DNN_ARG_TEMP) {
1300 int bufidx = bufidxs.at(inp.idx);
1301 Mat& buf = buffers.at(bufidx);
1302 buf.fit(mshape, mtype); // minimize reallocations
1303 m.copyTo(buf);

1306 graph->name().data(), adata.name.c_str(),
1307 argKindToString(adata.kind).c_str()));
1308 }

1365void Net::Impl::cudaUploadArg(Arg arg, const Mat& hostMat)
1366{

1372 if (g.type() == src.type()) {
1373 g.upload(src, s);
1374 } else {

1376 Mat tmp;
1377 src.convertTo(tmp, CV_MAT_DEPTH(g.type()));
1378 g.upload(tmp, s);
1379 }
1380 cudaArgHostDirty[idx] = 0;
1381 cudaArgDeviceDirty[idx] = 0;
1382 }

1385void Net::Impl::cudaDownloadArg(Arg arg, Mat& hostMat)
1386{
1387 int idx = arg.idx;
1388 if (idx < 0 || idx >= (int)cudaArgDeviceDirty.size())
1389 return;
1390 if (cudaArgDeviceDirty[idx]) {
1391 cuda::GpuMatND& g = cudaArgBuffers[idx];
1392 cuda::Stream s = wrapCudaStream(cudaInfo->context.stream);
1393 Mat dst = asAtLeast1D(hostMat); // fill the scalar's storage through a 1D header
1394 if (g.type() == dst.type()) {
1395 g.download(dst, s);
1396 cudaInfo->context.stream.synchronize(); // host read follows immediately
1397 } else {
1398 // device stores half: copy down, then convert up to the host FP32 tensor.
1399 Mat tmp;
1400 g.download(tmp, s);
1401 cudaInfo->context.stream.synchronize();
1402 tmp.convertTo(dst, dst.type());
1403 }
1404 cudaArgDeviceDirty[idx] = 0;
1405 cudaArgHostDirty[idx] = 0;
1406 }

1430// Slice a Scan input at index `idx` along `axis`, removing that axis (contiguous result).
1431static Mat sliceScanAxis(const Mat& m, int axis, int idx)
1432{
1433 std::vector<Range> r(m.dims, Range::all());
1434 r[axis] = Range(idx, idx + 1);
1435 Mat sub = m(r).clone();
1436 std::vector<int> ns;
1437 for (int d = 0; d < m.dims; d++)
1438 if (d != axis) ns.push_back(m.size[d]);
1439 if (ns.empty()) ns.push_back(1);
1440 return sub.reshape(0, (int)ns.size(), &ns[0]);
1441}
1442
1443// Stack per-iteration Scan outputs into one tensor with a new axis at `axis`.
1444static Mat stackScanAxis(const std::vector<Mat>& perIter, int axis, bool reverse)
1445{
1446 if (perIter.empty()) return Mat();
1447 const Mat& first = perIter[0];
1448 const int e = first.dims, T = (int)perIter.size();
1449 if (axis < 0) axis += e + 1;
1450 CV_Assert(axis >= 0 && axis <= e);
1451
1452 std::vector<int> os, ss;
1453 for (int d = 0; d < e; d++) {
1454 if (d == axis) { os.push_back(T); ss.push_back(1); }
1455 os.push_back(first.size[d]);
1456 ss.push_back(first.size[d]);
1457 }
1458 if (axis == e) { os.push_back(T); ss.push_back(1); }
1459
1460 Mat stacked;
1461 stacked.create((int)os.size(), &os[0], first.type());
1462 for (int k = 0; k < T; k++) {
1463 int idx = reverse ? (T - 1 - k) : k;
1464 std::vector<Range> r(os.size(), Range::all());
1465 r[axis] = Range(k, k + 1);
1466 Mat dst = stacked(r);
1467 Mat src = perIter[idx].reshape(0, (int)ss.size(), &ss[0]);
1468 src.copyTo(dst);
1469 }
1470 return stacked;
1471}

1473void Net::Impl::forwardGraph(Ptr<Graph>& graph, InputArrayOfArrays inputs_,
1474 OutputArrayOfArrays outputs_, bool isMainGraph)
1475{
1476 auto graphofs_it = graphofs.find(graph->name());
1477 if (graphofs_it == graphofs.end()) {
1478 CV_Error_(Error::StsObjectNotFound, ("graph '%s' does not belong to the model", graph->name().c_str()));
1479 }
1480 std::ostream& strm_ = dump_strm ? *dump_strm : std::cout;
1481 GraphImpl* gimpl = static_cast<GraphImpl*>(graph.get());
1482 const std::vector<Ptr<LayerInfo> >& prog = graph->prog();
1483 size_t i, nops = prog.size();
1484 const std::vector<Arg>& gr_inputs = graph->inputs();
1485 const std::vector<Arg>& gr_outputs = graph->outputs();
1486 size_t n_gr_inputs = gr_inputs.size(), n_gr_outputs = gr_outputs.size();
1487 std::vector<Mat> inpMats, outMats, tempMats;
1488 std::vector<int> inpTypes, outTypes, tempTypes;
1489 std::vector<std::pair<uchar*, size_t> > outOrigData;
1490 std::vector<MatShape> inpShapes, outShapes, tempShapes;
1491 double tickfreq = getTickFrequency();
1492 int64_t timestamp = 0;
1493
1494 size_t graph_ofs = (size_t)graphofs_it->second;
1495 CV_Assert(graph_ofs + nops <= totalLayers);
1496 if (!inputs_.empty()) {
1497 if (inputs_.total() != n_gr_inputs) {
1498 CV_Error_(Error::StsBadArg, ("wrong number of inputs in graph '%s': %zu given, %zu expected",
1499 graph->name().data(), inputs_.total(), n_gr_inputs));
1500 }
1501 for (i = 0; i < n_gr_inputs; i++) {
1502 Mat m = inputs_.getMat((int)i);
1503 setGraphInput(graph, i, m);
1504 }
1505 }
1506#ifdef HAVE_CUDA
1507 // Graph inputs are host-authoritative and may be updated between forward() calls
1508 // (setInput writes them outside this function, so the loop above is skipped when
1509 // forward() is called with no explicit inputs). Mark their device copies stale each
1510 // forward so the current input is re-uploaded; otherwise a second forward with a
1511 // changed input would read the previous forward's stale device data.
1512 if (cudaInfo) {
1513 for (i = 0; i < n_gr_inputs; i++)
1514 cudaSetHostDirty(gr_inputs[i]);
1515 }
1516#endif
1517
1518 for (size_t opidx = 0; opidx < nops; opidx++) {
1519 const Ptr<LayerInfo>& op = prog.at(opidx);
1520 if (!op) // in theory we shouldn't have any 'nops' at this stage, but just in case we skip them.
1521 continue;
1522 Ptr<Layer> layer = (opidx < gimpl->exec_.size()) ? gimpl->exec_[opidx] : Ptr<Layer>();
1523 if (!layer)
1524 layer = op.dynamicCast<Layer>();
1525 CV_Assert(layer);
1526 int opBackend = (opidx < gimpl->execBackend_.size()) ? gimpl->execBackend_[opidx]
1527 : DNN_BACKEND_OPENCV;
1528 CV_UNUSED(opBackend); // only consumed by the HAVE_CUDA dispatch below
1529 const std::vector<Arg>& inputs = op->inputs;
1530 const std::vector<Arg>& outputs = op->outputs;
1531 size_t ninputs = inputs.size(), noutputs = outputs.size();
1532
1533 inpMats.resize(ninputs);
1534 inpTypes.resize(ninputs);
1535 inpShapes.resize(ninputs);
1536 outMats.clear();
1537 outOrigData.clear();
1538
1539 for (i = 0; i < ninputs; i++) {
1540 Arg inp = inputs[i];
1541#ifdef HAVE_CUDA
1542 // CPU op: bring any device-resident input back to host before reading its
1543 // shape/data. This must happen before allocateLayerOutputs(), which may fit()
1544 // an in-place op's output onto this very buffer and rewrite its header.
1545 if (opBackend != DNN_BACKEND_CUDA)
1546 cudaDownloadArg(inp, argTensor(inp));
1547#endif
1548 const Mat& m = argTensor(inp);
1549 inpMats[i] = m;
1550 inpTypes[i] = m.type();
1551 inpShapes[i] = m.shape();
1552 }
1553
1554 if (tracingMode != DNN_TRACE_NONE) {
1555 strm_ << "-----------\n";
1556 strm_ << "'" << graph->name() << "' [" << opidx << "/" << nops << "]. " << op->type << " node: " << op->name << "\n";
1557 for (i = 0; i < ninputs; i++) {
1558 Arg inp = inputs[i];
1559 traceArg(strm_, "Input", i, inp, false);
1560 }
1561 }
1562 bool dynamicOutShapes = op->dynamicOutputShapes();
1563 if (!dynamicOutShapes) {
1564 allocateLayerOutputs(op, inpTypes, inpShapes, outTypes, outShapes, outOrigData, outMats,
1565 tempTypes, tempShapes, tempMats, scratchBufs, true);
1566 } else {
1567 outMats.resize(noutputs);
1568 for (i = 0; i < noutputs; i++) {
1569 Arg out = outputs[i];
1570 outMats[i] = argTensor(out);
1571 }
1572 tempMats = scratchBufs;
1573 }
1574
1575 timestamp = getTickCount();
1576
1577 std::vector<Ptr<Graph> >* subgraphs = op->subgraphs();
1578 if (!subgraphs) {
1579 // Blobs live on 'op', packed buffers on the executor 'layer'.
1580 if (layer->packedWeightEpoch != op->weightEpoch) {
1581 layer->prepackWeights();
1582 layer->packedWeightEpoch = op->weightEpoch;
1583 // New weights: re-finalize too, for layers that pack inside finalize().
1584 layer->finalizedOnce = false;
1585 }
1586 // Re-finalize only when this layer's own input signature changed.
1587 if (!layer->finalizedOnce || layer->lastInpShapes != inpShapes || layer->lastInpTypes != inpTypes) {
1588 layer->finalize((InputArrayOfArrays)inpMats, (OutputArrayOfArrays)outMats);
1589 layer->lastInpShapes = inpShapes;
1590 layer->lastInpTypes = inpTypes;
1591 layer->finalizedOnce = true;
1592 }
1593#ifdef HAVE_CUDA
1594 if (opBackend == DNN_BACKEND_CUDA) {
1595 forwardOpCUDA(this, gimpl, opidx, inputs, outputs, inpMats, outMats);
1596 } else
1597#endif
1598 {
1599 // Device-resident inputs were already synced to host in the capture loop above.
1600 layer->forward(inpMats, outMats, tempMats);
1601#ifdef HAVE_CUDA
1602 // CPU produced fresh host data; invalidate any stale device copy of its outputs.
1603 for (size_t k = 0; k < noutputs; k++)
1604 cudaSetHostDirty(outputs[k]);
1605#endif
1606 }
1607 }
1608 else {
1609 Ptr<IfLayer> iflayer = layer.dynamicCast<IfLayer>();
1610 Ptr<LoopLayer> loopLayer = layer.dynamicCast<LoopLayer>();
1611 Ptr<ScanLayer> scanLayer = layer.dynamicCast<ScanLayer>();
1612 if (iflayer) {
1613 int branch = iflayer->branch(inpMats[0]);
1614 Ptr<Graph> subgraph = subgraphs->at(branch);
1615 std::vector<Mat> branchInputs;
1616 if (inpMats.size() > 1)
1617 branchInputs.assign(inpMats.begin() + 1, inpMats.end());
1618 forwardGraph(subgraph, branchInputs, outMats, false);
1619 }
1620 else if (loopLayer) {
1621 CV_Assert(subgraphs->size() == 1);
1622 Ptr<Graph> body = subgraphs->at(0);
1623
1624 int n_in = (int)inpMats.size();
1625 int n_state = n_in - 2;
1626 int n_accum = (int)body->outputs().size() - n_state - 1;
1627 CV_Assert(n_in >= 2 && n_state >= 0 && n_accum >= 0);
1628
1629 Mat mIter, mCond, tmp;
1630 mIter.create(1, 1, CV_64S);
1631 if (inpMats[1].empty())
1632 {
1633 mCond.create(1, 1, CV_8U);
1634 mCond.at<uchar>(0) = 1;
1635 }
1636 else
1637 {
1638 mCond = inpMats[1];
1639 }
1640 int64 iter = 0;
1641 int64 max_iter = inpMats[0].empty() ? -1 :
1642 (inpMats[0].convertTo(tmp, CV_64S), *tmp.ptr<int64>());
1643 bool active = loopLayer->cond(mCond);
1644
1645 std::vector<Mat> state(inpMats.begin() + 2, inpMats.end());
1646 std::vector<std::vector<Mat> > history(n_accum);
1647 std::vector<Mat> inputs(n_in), outputs;
1648
1649 while (active && (max_iter < 0 || iter < max_iter)) {
1650 mIter.at<int64>(0) = iter++;
1651 inputs[0] = mIter;
1652 inputs[1] = mCond;
1653 std::copy(state.begin(), state.end(), inputs.begin() + 2);
1654
1655 forwardGraph(body, inputs, outputs, false);
1656
1657 mCond = outputs[0];
1658 active = loopLayer->cond(mCond);
1659
1660 // Deep-copy: body buffers (and their UMats via Mat::fit reuse) are recycled across iterations.
1661 for (int i = 0; i < n_state; i++)
1662 state[i] = outputs[1 + i].clone();
1663 for (int i = 0; i < n_accum; i++)
1664 history[i].push_back(outputs[1 + n_state + i].clone());
1665 }
1666
1667 outMats.assign(state.begin(), state.end());
1668 outMats.resize(n_state + n_accum);
1669
1670 for (int i = 0; i < n_accum; ++i) {
1671 const std::vector<Mat>& perIter = history[i];
1672 if (perIter.empty()) {
1673 outMats[n_state + i] = Mat();
1674 continue;
1675 }
1676
1677 const Mat& first = perIter[0];
1678 MatShape elemShape = first.shape();
1679 int ndims = elemShape.dims;
1680 CV_Assert(ndims >= 0);
1681
1682 MatShape outShape(ndims + 1);
1683 outShape[0] = (int)perIter.size();
1684 for (int d = 0; d < ndims; ++d)
1685 outShape[d + 1] = elemShape[d];
1686
1687 Mat stacked;
1688 stacked.create(outShape, first.type());
1689
1690 for (size_t k = 0; k < perIter.size(); ++k)
1691 {
1692 const Mat& src = perIter[k];
1693 CV_Assert(src.type() == first.type());
1694 CV_Assert(src.total() == first.total());
1695
1696 Mat dst = stacked.reshape(0, (int)perIter.size())
1697 .row((int)k)
1698 .reshape(first.channels(), elemShape.dims, &elemShape[0]);
1699 src.copyTo(dst);
1700 }
1701 outMats[n_state + i] = stacked;
1702 }
1703 }
1704 else if (scanLayer) {
1705 CV_Assert(subgraphs->size() == 1);
1706 Ptr<Graph> body = subgraphs->at(0);
1707
1708 const int M = scanLayer->numScanInputs();
1709 const int bodyNIn = (int)body->inputs().size();
1710 const int S = bodyNIn - M; // loop-carried state count
1711 const int K = (int)body->outputs().size() - S; // scan output count
1712 CV_Assert(M > 0 && S >= 0 && K >= 0);
1713
1714 // Reject opset-8 (batch dim + sequence_lens); those semantics are not implemented.
1715 CV_CheckEQ((int)inpMats.size(), S + M,
1716 "Scan: opset-8 form (batch dim + sequence_lens) is not supported; re-export with opset>=9");
1717
1718 const std::vector<int>& iax = scanLayer->scanInputAxes();
1719 const std::vector<int>& oax = scanLayer->scanOutputAxes();
1720 const std::vector<int>& idir = scanLayer->scanInputDirections();
1721 const std::vector<int>& odir = scanLayer->scanOutputDirections();
1722 const std::vector<int>& orank = scanLayer->scanOutputRanks();
1723
1724 std::vector<Mat> scanIn(M);
1725 std::vector<int> inAxis(M);
1726 std::vector<char> inRev(M);
1727 int T = -1;
1728 for (int j = 0; j < M; j++) {
1729 scanIn[j] = inpMats[S + j];
1730 int ax = iax.empty() ? 0 : iax[j];
1731 if (ax < 0) ax += scanIn[j].dims;
1732 inAxis[j] = ax;
1733 inRev[j] = (char)(!idir.empty() && idir[j] != 0);
1734 int len = scanIn[j].size[ax];
1735 if (T < 0) T = len; else CV_Assert(T == len);
1736 }
1737 CV_Assert(T >= 0);
1738
1739 std::vector<Mat> state(S);
1740 for (int i = 0; i < S; i++) state[i] = inpMats[i];
1741
1742 std::vector<std::vector<Mat> > history(K);
1743 std::vector<Mat> inputs(bodyNIn), outputs;
1744
1745 for (int t = 0; t < T; t++) {
1746 for (int i = 0; i < S; i++) inputs[i] = state[i];
1747 for (int j = 0; j < M; j++) {
1748 int idx = inRev[j] ? (T - 1 - t) : t;
1749 inputs[S + j] = sliceScanAxis(scanIn[j], inAxis[j], idx);
1750 }
1751 forwardGraph(body, inputs, outputs, false);
1752 // Deep-copy: body buffers are recycled across iterations.
1753 // TODO: alias state in/out buffers like Loop to drop this copy.
1754 for (int i = 0; i < S; i++) state[i] = outputs[i].clone();
1755 for (int k = 0; k < K; k++) history[k].push_back(outputs[S + k].clone());
1756 }
1757
1758 outMats.assign(state.begin(), state.end());
1759 outMats.resize(S + K);
1760 for (int k = 0; k < K; k++) {
1761 int ax = oax.empty() ? 0 : oax[k];
1762 bool rev = !odir.empty() && odir[k] != 0;
1763 // 0-D scalar output is stored as [1]; drop it so T scalars stack to [T], not [T,1].
1764 if (k < (int)orank.size() && orank[k] == 0) {
1765 for (Mat& e : history[k])
1766 e = e.reshape(0, 0, nullptr);
1767 }
1768 outMats[S + k] = stackScanAxis(history[k], ax, rev);
1769 }
1770 }
1771 else {
1772 CV_Error_(Error::StsNotImplemented,
1773 ("unknown layer type '%s' with subgraphs", op->type.c_str()));
1774 }
1775 }
1776 CV_Assert(outMats.size() == noutputs);
1777

1779 Arg out = outputs[i];
1780 ArgData& adata = args[out.idx];
1781 const Mat& m = outMats[i];
1782 //checkRange(m, false);
1783 adata.type = m.type();
1784 adata.shape = m.shape();
1785 if (adata.kind == DNN_ARG_TEMP) {
1786 int bufidx = bufidxs.at(out.idx);
1787 Mat& buf = buffers.at(bufidx);
1788
1789 if (!dynamicOutShapes) {
1790 // a sanity check: make sure that the data was not reallocated during Layer::forward()
1791 // if the layer claims it does not produce dynamic-shape outputs.
1792 CV_Assert_N(buf.u == m.u,
1793 buf.shape() == m.shape(),
1794 buf.type() == m.type(),
1795 (!m.u || m.u->data == outOrigData[i].first),
1796 (!m.u || m.u->size == outOrigData[i].second));
1797 } else if (!buf.u || (m.u && m.u->size > buf.u->size)) {
1798 buf = m;
1799 } else {
1800 // this branch means that the layer still calls
1801 // 'create()' rather than 'fit()'; that needs to be fixed, but
1802 // we provide workaround here at the expense of extra copy.
1803 buf.fit(m.shape(), m.type());
1804 m.copyTo(buf);
1805 }
1806 } else {
1807 if (!dynamicOutShapes) {
1808 // the same sanity check for non-temp (graph output/state) tensors: the layer must
1809 // write into the preallocated tensor of the inferred shape/type. A mismatch here
1810 // means some op inside Layer::forward() reallocated it (e.g. a broadcast produced
1811 // an unexpected shape) and the result would silently detach from the graph.
1812 if (m.shape() != outShapes[i] || m.type() != outTypes[i] ||
1813 (m.u && (m.u->data != outOrigData[i].first || m.u->size != outOrigData[i].second)))
1814 {

1816 oss << "layer '" << layer->name << "' (" << layer->type << "): output #" << i
1817 << " changed during forward(): inferred shape " << outShapes[i]
1818 << " / type " << typeToString(outTypes[i])
1819 << ", actual " << m.shape() << " / " << typeToString(m.type())
1820 << (m.u && m.u->data != outOrigData[i].first
1821 ? "; the tensor was reallocated (the layer must write in place)" : "");
1822 CV_Error(Error::StsInternal, oss.str());
1823 }
1824 }
1825 __tensors__.at(out.idx) = m;
1826 }
1827 }
1828
1829 size_t ntemps = tempMats.size();

1831 for (size_t i = 0; i < ntemps; i++) {
1832 size_t newtotal_i = tempMats[i].total()*tempMats[i].elemSize();
1833 size_t total_i = scratchBufs[i].total()*scratchBufs[i].elemSize();
1834 if (newtotal_i > total_i) {
1835 scratchBufs[i] = tempMats[i];
1836 }
1837 }
1838
1839 timestamp = getTickCount() - timestamp;
1840 layersTimings[opidx + graph_ofs + 1] += timestamp;
1841
1842 if (tracingMode != DNN_TRACE_NONE) {
1843 strm_ << "TIME (\"" << layer->name << "\", \"" << layer->type << "\"): " <<
1844 format("%.2fms", (double)timestamp*1000./tickfreq) << "\n";
1845 for (i = 0; i < noutputs; i++) {
1846 Arg out = outputs[i];
1847 traceArg(strm_, "Output", i, out, tracingMode == DNN_TRACE_ALL);
1848 }
1849 }
1850 }
1851
1852 std::vector<Mat>& outputsVec = outputs_.getMatVecRef();
1853 outputsVec.resize(n_gr_outputs);
1854 for (i = 0; i < n_gr_outputs; i++) {
1855 Arg out = gr_outputs[i];
1856#ifdef HAVE_CUDA
1857 // A graph output produced on the device must be brought back to host before it is read.
1858 if (out.idx >= 0) {
1859 Mat& t = argTensor(out);
1860 cudaDownloadArg(out, t);
1861 }
1862#endif
1863 const Mat& outm = argTensor(out);
1864 if (isMainGraph) {
1865 if (outm.size.layout == DATA_LAYOUT_BLOCK) {
1866 transformLayout(outm, outputsVec[i], originalLayout, originalLayout, outm.size.C);
1867 } else {
1868 outputsVec[i].fit(outm.shape(), outm.type());
1869 outm.copyTo(outputsVec[i]);
1870 }
1871 } else {
1872 outputsVec[i] = outm;
1873 }
1874 }
