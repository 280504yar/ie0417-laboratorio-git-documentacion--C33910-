13class TypeNode(abc.ABC):
14 """This class and its derivatives used for construction parts of AST that
15 otherwise can't be constructed from the information provided by header
16 parser, because this information is either not available at that moment of
17 time or not available at all:
18 - There is no possible way to derive correspondence between C++ type
19 and its Python equivalent if it is not exposed from library
20 e.g. `cv::Rect`.
21 - There is no information about types visibility (see `ASTNodeTypeNode`).
22 """
23 compatible_to_runtime_usage = False
24 """Class-wide property that switches exported type names for several nodes.
25 Example:
26 >>> node = OptionalTypeNode(ASTNodeTypeNode("Size"))
27 >>> node.typename # TypeNode.compatible_to_runtime_usage == False
28 "Size | None"
29 >>> TypeNode.compatible_to_runtime_usage = True
30 >>> node.typename
31 "typing.Optional[Size]"
32 """

547class ASTNodeTypeNode(TypeNode):
548 """Type node representing a lazy ASTNode corresponding to type of
549 function argument or its return type or type of class property.
550 Introduced laziness nature resolves the types visibility issue - all types
551 should be known during function declaration to select an appropriate node
552 from the AST. Such knowledge leads to evaluation of all preprocessor
553 directives (`#include` particularly) for each processed header and might be
554 too expensive and error prone.
555 """
556 def __init__(self, ctype_name: str, typename: Optional[str] = None,
557 module_name: Optional[str] = None,
558 required_modules: Tuple[str, ...] = ()) -> None:
559 super().__init__(ctype_name, required_modules)
560 self._typename = typename if typename is not None else ctype_name
561 self._module_name = module_name
562 self._ast_node: Optional[weakref.ProxyType[ASTNode]] = None
563
564 @property
565 def ast_node(self):
566 return self._ast_node
567
568 @property
569 def typename(self) -> str:
570 if self._ast_node is None:
571 return self._typename
572 typename = self._ast_node.export_name
573 if self._ast_node.node_type is not ASTNodeType.Enumeration:
574 return typename
575 # NOTE: Special handling for enums
576 parent = self._ast_node.parent
577 while parent.node_type is ASTNodeType.Class:
578 typename = parent.export_name + "_" + typename
579 parent = parent.parent
580 return typename
581
582 @property
583 def full_typename(self) -> str:
584 if self._ast_node is not None:
585 if self._ast_node.node_type is not ASTNodeType.Enumeration:
586 return self._ast_node.full_export_name
587 # NOTE: enumerations are exported to module scope
588 typename = self._ast_node.export_name
589 parent = self._ast_node.parent
590 while parent.node_type is ASTNodeType.Class:
591 typename = parent.export_name + "_" + typename
592 parent = parent.parent
593 return parent.full_export_name + "." + typename
594 if self._module_name is not None:
595 return self._module_name + "." + self._typename
596 return self._typename
597
598 @property
599 def required_usage_imports(self) -> Generator[str, None, None]:
600 if self._module_name is None:
601 assert self._ast_node is not None, \
602 "Can't find a module for class '{}' exported as '{}'".format(
603 self.ctype_name, self.typename,
604 )
605 module = self._ast_node.parent
606 while module.node_type is not ASTNodeType.Namespace:
607 module = module.parent
608 yield "import " + module.full_export_name
609 else:
610 yield "import " + self._module_name
611
612 @property
613 def is_resolved(self) -> bool:
614 return self._ast_node is not None or self._module_name is not None
615
616 def resolve(self, root: ASTNode):
617 if self.is_resolved:
618 return
619
620 node = _resolve_symbol(root, self.typename)
621 if node is None:
622 raise TypeResolutionError('Failed to resolve "{}" exposed as "{}"'.format(
623 self.ctype_name, self.typename
624 ))
625 self._ast_node = weakref.proxy(node)
626
627 def relative_typename(self, module: str) -> str:
628 assert self._ast_node is not None or self._module_name is not None, \
629 "'{}' exported as '{}' is not resolved yet".format(self.ctype_name,
630 self.typename)
631 if self._module_name is None:
632 type_module = self._ast_node.parent # type: ignore
633 while type_module.node_type is not ASTNodeType.Namespace:
634 type_module = type_module.parent
635 module_name = type_module.full_export_name
636 else:
637 module_name = self._module_name
638 if module_name != module:
639 return self.full_typename
640 return self.full_typename[len(module_name) + 1:]

643class AggregatedTypeNode(TypeNode):
644 """Base type node for type nodes representing an aggregation of another
645 type nodes e.g. tuple, sequence or callable."""
646 def __init__(self, ctype_name: str, items: Sequence[TypeNode],
647 required_modules: Tuple[str, ...] = ()) -> None:
648 super().__init__(ctype_name, required_modules)
649 self.items = list(items)
650
651 @property
652 def is_resolved(self) -> bool:
653 return all(item.is_resolved for item in self.items)
654
655 @property
656 def required_modules(self) -> Tuple[str, ...]:
657 return (*chain.from_iterable(item.required_modules for item in self.items),
658 *self._required_modules)
659
660 def resolve(self, root: ASTNode) -> None:
661 errors = []
662 for item in filter(lambda item: not item.is_resolved, self):
663 try:
664 item.resolve(root)
665 except TypeResolutionError as e:
666 errors.append(str(e))
667 if len(errors) > 0:
668 raise TypeResolutionError(
669 'Failed to resolve one of "{}" items. Errors: {}'.format(
670 self.full_typename, errors
671 )
672 )
673
674 def __iter__(self):
675 return iter(self.items)
676
677 def __len__(self) -> int:
678 return len(self.items)

682 for item in self:
683 yield from item.required_definition_imports

929def _resolve_symbol(root: Optional[ASTNode], full_symbol_name: str) -> Optional[ASTNode]:
930 """Searches for a symbol with the given full export name in the AST
931 starting from the `root`.
932
933 Args:
934 root (Optional[ASTNode]): Root of the examining AST.
935 full_symbol_name (str): Full export name of the symbol to find. Path
936 components can be divided by '.' or '_'.
937
938 Returns:
939 Optional[ASTNode]: ASTNode with full export name equal to
940 `full_symbol_name`, None otherwise.
941
942 >>> root = NamespaceNode('cv')
943 >>> cls = root.add_class('Algorithm').add_class('Params')
944 >>> _resolve_symbol(root, 'cv.Algorithm.Params') == cls
945 True
946
947 >>> root = NamespaceNode('cv')
948 >>> enum = root.add_namespace('detail').add_enumeration('AlgorithmType')
949 >>> _resolve_symbol(root, 'cv_detail_AlgorithmType') == enum
950 True
951
952 >>> root = NamespaceNode('cv')
953 >>> _resolve_symbol(root, 'cv.detail.Algorithm')
954 None
955
956 >>> root = NamespaceNode('cv')
957 >>> enum = root.add_namespace('detail').add_enumeration('AlgorithmType')
958 >>> _resolve_symbol(root, 'AlgorithmType')
959 None
960 """
961 def search_down_symbol(scope: Optional[ASTNode],
962 scope_sep: str) -> Optional[ASTNode]:
963 parts = full_symbol_name.split(scope_sep, maxsplit=1)
964 while len(parts) == 2:
965 # Try to find narrow scope
966 scope = _resolve_symbol(scope, parts[0])
967 if scope is None:
968 return None
969 # and resolve symbol in it
970 node = _resolve_symbol(scope, parts[1])
971 if node is not None:
972 return node
973 # symbol is not found, but narrowed scope is valid - diving further
974 parts = parts[1].split(scope_sep, maxsplit=1)
975 return None
976
977 assert root is not None, \
978 "Can't resolve symbol '{}' from NONE root".format(full_symbol_name)
979 # Looking for exact symbol match
980 for attr in filter(lambda attr: hasattr(root, attr),
981 ("namespaces", "classes", "enumerations")):
982 nodes_dict = getattr(root, attr)
983 node = nodes_dict.get(full_symbol_name, None)
984 if node is not None:
985 return node
986 # Symbol is not found, looking for more fine-grained scope if possible
987 for scope_sep in ("_", "."):
988 node = search_down_symbol(root, scope_sep)
989 if node is not None:
990 return node
991 return None
