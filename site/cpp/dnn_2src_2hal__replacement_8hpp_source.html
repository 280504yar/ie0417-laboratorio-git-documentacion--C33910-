141// Call an overridable DNN HAL hook; on CV_HAL_ERROR_OK return from the current
142// scope (skipping the built-in fallback), on NOT_IMPLEMENTED fall through to it.
143// Used inside a parallel_for_ worker lambda, so the early return leaves that one
144// task range to the HAL while the rest of the engine is unaffected.
145#define CALL_HAL(name, fun, ...) \
146{ \
147 int res = __CV_EXPAND(fun(__VA_ARGS__)); \
148 if (res == CV_HAL_ERROR_OK) \
149 return; \
150 else if (res != CV_HAL_ERROR_NOT_IMPLEMENTED) \
151 CV_Error_(cv::Error::StsInternal, \
152 ("HAL implementation " CVAUX_STR(name) " ==> " CVAUX_STR(fun) " returned %d (0x%08x)", res, res)); \
153}
