78inline float computeSrcGeneric(int dst, float scale, int limit, int len,
79 CoordTransMode coordTransMode, bool /*halfPixelCenters*/,
80 float start_coord = 0.0f, float end_coord = 1.0f)
81{
82 if (coordTransMode == CoordTransMode::TF_CROP_AND_RESIZE)
83 {
84 if (len > 1)
85 return start_coord * (limit - 1) + dst * (end_coord - start_coord) * (limit - 1) / float(len - 1);
86 else
87 return 0.5f * (start_coord + end_coord) * (limit - 1);
88 }
89 if (coordTransMode == CoordTransMode::PYTORCH_HALF_PIXEL)
90 return (len > 1) ? (dst + 0.5f)*scale - 0.5f : 0.f;
91 if (coordTransMode == CoordTransMode::HALF_PIXEL)
92 return (dst + 0.5f)*scale - 0.5f;
93 if (coordTransMode == CoordTransMode::TF_HALF_PIXEL_FOR_NN)
94 return (dst + 0.5f)*scale;
95 if (coordTransMode == CoordTransMode::HALF_PIXEL_SYMMETRIC)
96 {
97 // ONNX half_pixel_symmetric: offset = center*(1 - len_resized/(len_orig*x_scale)),
98 // with scale == 1/x_scale and limit == input length.
99 const float offset = limit*0.5f - len*scale*0.5f;
100 return offset + (dst + 0.5f)*scale - 0.5f;
101 }
102 return dst*scale;
103}

105static inline void buildNearestIndexMap(std::vector<int>& map,
106 int outLen,
107 int inLen,
108 float scale,
109 int len,
110 float start_coord,
111 float end_coord,
112 CoordTransMode coordTransMode,
113 NearestMode nearestMode,
114 bool halfPixelCenters)
115{
116 auto nearestIndex = [&](float src) {
117 const int f = cvFloor(src);
118 const float frac = src - f;
119 const float eps = 1e-6f;
120 int idx;
121 if (nearestMode == NearestMode::FLOOR) idx = cvFloor(src);
122 else if (nearestMode == NearestMode::CEIL) idx = cvCeil(src);
123 else if (nearestMode == NearestMode::ROUND_PREFER_CEIL) {
124 idx = (abs(frac - 0.5f) <= eps) ? (f + 1) : cvRound(src);
125 } else {
126 idx = (abs(frac - 0.5f) <= eps) ? f : cvRound(src);
127 }
128 return clamp(idx, 0, inLen - 1);
129 };
130
131 map.resize(outLen);
132 for (int i = 0; i < outLen; ++i)
133 {
134 float src = computeSrcGeneric(i, scale, inLen, len,
135 coordTransMode, halfPixelCenters, start_coord, end_coord);
136 if (coordTransMode == CoordTransMode::TF_CROP_AND_RESIZE) {
137 if (src < 0.f || src >= float(inLen)) {
138 map[i] = -1; // out of bounds
139 continue;
140 }
141 } else {
142 src = std::min(std::max(src, 0.f), float(inLen - 1));
143 }
144 map[i] = nearestIndex(src);
145 }
146}

148static inline void buildBilinearIndexAndLerp(std::vector<int>& i0,
149 std::vector<int>& i1,
150 std::vector<float>& frac,
151 std::vector<uint8_t>& outOfBounds,
152 int outLen,
153 int inLen,
154 float scale,
155 int len,
156 float start_coord,
157 float end_coord,
158 CoordTransMode coordTransMode,
159 bool halfPixelCenters,
160 bool tf_crop_and_resize_mode)
161{
162 i0.resize(outLen);
163 i1.resize(outLen);
164 frac.resize(outLen);
165 outOfBounds.assign(outLen, 0);
166
167 for (int o = 0; o < outLen; ++o)
168 {
169 float src = computeSrcGeneric(o, scale, inLen, len,
170 coordTransMode, halfPixelCenters, start_coord, end_coord);
171 if (tf_crop_and_resize_mode)
172 {
173 int base = int(std::floor(src));
174 if (base < 0 || base >= inLen - 1) {
175 outOfBounds[o] = 1;
176 i0[o] = 0;
177 i1[o] = 0;
178 frac[o] = 0.0f;
179 } else {
180 i0[o] = base;
181 i1[o] = base + 1;
182 frac[o] = src - float(base);
183 }
184 }
185 else
186 {
187 src = std::min(std::max(src, 0.f), std::max(0.f, float(inLen - 1) - 1e-6f));
188 int base = int(std::floor(src));
189 i0[o] = clamp(base, 0, inLen - 1);
190 i1[o] = clamp(base + 1, 0, inLen - 1);
191 frac[o] = src - float(base);
192 }
193 }
194}

196static inline void interpolateCubicResize(float x, float A, float* coeffs )
197{
198 coeffs[0] = ((A*(x + 1) - 5*A)*(x + 1) + 8*A)*(x + 1) - 4*A;
199 coeffs[1] = ((A + 2)*x - (A + 3))*x*x + 1;
200 coeffs[2] = ((A + 2)*(1 - x) - (A + 3))*(1 - x)*(1 - x) + 1;
201 coeffs[3] = 1.f - coeffs[0] - coeffs[1] - coeffs[2];
202}

204static inline void buildCubicIndexAndWeights(std::vector<std::array<int,4>>& ids,
205 std::vector<std::array<float,4>>& weights,
206 std::vector<uint8_t>& outOfBounds,
207 int outLen,
208 int inLen,
209 float scale,
210 int len,
211 float start_coord,
212 float end_coord,
213 CoordTransMode coordTransMode,
214 bool halfPixelCenters,
215 bool tf_crop_and_resize_mode,
216 bool excludeOutside,
217 float cubicA)
218{
219 ids.resize(outLen);
220 weights.resize(outLen);
221 outOfBounds.assign(outLen, 0);
222
223 for (int o = 0; o < outLen; ++o)
224 {
225 float src = computeSrcGeneric(o, scale, inLen, len,
226 coordTransMode, halfPixelCenters, start_coord, end_coord);
227 int i = int(std::floor(src));
228 float d = src - i;
229 float sw = 0.f;
230 bool hasOutOfBounds = false;
231 interpolateCubicResize(d, cubicA, weights[o].data());
232
233 if (!tf_crop_and_resize_mode && !excludeOutside)
234 {
235 for (int k = -1; k <= 2; ++k)
236 {
237 int idx = clamp(i + k, 0, inLen - 1);
238 ids[o][k+1] = idx;
239 sw += weights[o][k+1];
240 }
241 }
242 else if (tf_crop_and_resize_mode)
243 {
244 for (int k = -1; k <= 2; ++k)
245 {
246 int idx = i + k;
247 unsigned valid = (unsigned)idx < (unsigned)inLen;
248 ids[o][k+1] = valid ? idx : -1;
249 float w = weights[o][k+1];
250 float wv = valid ? w : 0.f;
251 weights[o][k+1] = wv;
252 sw += wv;
253 hasOutOfBounds |= !valid;
254 }
255 }
256 else
257 {
258 for (int k = -1; k <= 2; ++k)
259 {
260 int idx = i + k;
261 unsigned valid = (unsigned)idx < (unsigned)inLen;
262 ids[o][k+1] = valid ? idx : -1;
263 float w = weights[o][k+1];
264 float wv = valid ? w : 0.f;
265 weights[o][k+1] = wv;
266 sw += wv;
267 }
268 }
269
270 if (sw != 0.f)
271 for (int k = 0; k < 4; ++k)
272 weights[o][k] /= sw;
273
274 if (tf_crop_and_resize_mode && hasOutOfBounds)
275 outOfBounds[o] = 1;
276 }
277}

279template<typename T>
280void resizeNearest(const Mat &inp, Mat &out,
281 float scaleH, float scaleW,
282 int lenY, int lenX,
283 NearestMode nearestMode,
284 const String &coordTransMode,
285 bool halfPixelCenters,
286 float start_y = 0.0f, float end_y = 1.0f,
287 float start_x = 0.0f, float end_x = 1.0f,
288 float extrapolation_value = 0.0f)
289{
290 int inH = inp.size[2], inW = inp.size[3];
291 int outH = out.size[2], outW = out.size[3];
292 CV_Assert(inp.isContinuous() && out.isContinuous());
293
294 CoordTransMode coordMode = parseCoordTransMode(coordTransMode);
295 const bool tf_crop_and_resize_mode = (coordMode == CoordTransMode::TF_CROP_AND_RESIZE);
296
297 std::vector<int> mapY(outH);
298 buildNearestIndexMap(mapY, outH, inH, scaleH, lenY, start_y, end_y,
299 coordMode, nearestMode, halfPixelCenters);
300
301 std::vector<int> mapX(outW);
302 buildNearestIndexMap(mapX, outW, inW, scaleW, lenX, start_x, end_x,
303 coordMode, nearestMode, halfPixelCenters);
304
305 if (inp.shape().layout == DATA_LAYOUT_BLOCK) {
306 CV_Assert(inp.dims == 5 && out.dims == 5);
307 const int N = inp.size[0], C1 = inp.size[1], C0 = inp.size[4];
308
309 const size_t inStep0 = inp.step.p[0] / inp.elemSize();
310 const size_t inStep1 = inp.step.p[1] / inp.elemSize();
311 const size_t inStep2 = inp.step.p[2] / inp.elemSize();
312 const size_t inStep3 = inp.step.p[3] / inp.elemSize();
313 const size_t outStep0 = out.step.p[0] / out.elemSize();
314 const size_t outStep1 = out.step.p[1] / out.elemSize();
315 const size_t outStep2 = out.step.p[2] / out.elemSize();
316 const size_t outStep3 = out.step.p[3] / out.elemSize();
317 const size_t C0bytes = (size_t)C0 * sizeof(T);
318
319 const int nplanes = N * C1 * outH;
320 parallel_for_(Range(0, nplanes), [&](const Range& range) {
321 const T* inptr0 = reinterpret_cast<const T*>(inp.data);
322 T* outptr0 = reinterpret_cast<T*>(out.data);
323 T ext = saturate_cast<T>(extrapolation_value);
324
325 for (int plane = range.start; plane < range.end; ++plane) {
326 int t = plane;
327 int oy = t % outH; t /= outH;
328 int c1 = t % C1;
329 int n = t / C1;
330
331 int iy = mapY[oy];
332 T* outRow = outptr0 + n * outStep0 + c1 * outStep1 + oy * outStep2;
333
334 if (tf_crop_and_resize_mode && iy == -1) {
335 for (int ox = 0; ox < outW; ++ox) {
336 T* outPix = outRow + ox * outStep3;
337 for (int c0 = 0; c0 < C0; ++c0) outPix[c0] = ext;
338 }
339 continue;
340 }
341
342 const T* inRow = inptr0 + n * inStep0 + c1 * inStep1 + iy * inStep2;
343 for (int ox = 0; ox < outW; ++ox) {
344 int ix = mapX[ox];
345 if (tf_crop_and_resize_mode && ix == -1) {
346 T* outPix = outRow + ox * outStep3;
347 for (int c0 = 0; c0 < C0; ++c0) outPix[c0] = ext;
348 } else {
349 memcpy(outRow + ox * outStep3, inRow + ix * inStep3, C0bytes);
350 }
351 }
352 }
353 }, kResizeNumStripes);
354 return;
355 }
356
357 int numPlanes = inp.size[0] * inp.size[1];
358 Mat inpP = inp.reshape(1, numPlanes * inH);
359 Mat outP = out.reshape(1, numPlanes * outH);
360
361 const int nstripes = kResizeNumStripes;
362 parallel_for_(Range(0, nstripes), [&](const Range& range) {
363 int row0 = range.start * (outH * numPlanes) / nstripes;
364 float extrapolation_value_ = extrapolation_value;
365 int row1 = range.end * (outH * numPlanes) / nstripes - 1;
366 int plane0 = row0 / outH, plane1 = row1 / outH;
367 row0 %= outH;
368 row1 %= outH;
369
370 const int* mapYptr = mapY.data();
371 const int* mapXptr = mapX.data();
372
373 for (int p = plane0; p <= plane1; p++) {
374 int y0 = p == plane0 ? row0 : 0;
375 int y1 = p == plane1 ? row1 : outH - 1;
376 for (int y = y0; y <= y1; y++) {
377 int my = mapYptr[y];
378 if (tf_crop_and_resize_mode && my == -1) {
379 T* outRowFill = outP.ptr<T>(p * outH + y);
380 for (int x = 0; x < outW; ++x)
381 outRowFill[x] = T(extrapolation_value_);
382 continue;
383 }
384 const T* inpRow = inpP.ptr<T>(p * inH + my);
385 T* outRow = outP.ptr<T>(p * outH + y);
386 for (int x = 0; x < outW; ++x)
387 {
388 int mx = mapXptr[x];
389 if (tf_crop_and_resize_mode && mx == -1) {
390 outRow[x] = T(extrapolation_value_);
391 } else {
392 outRow[x] = inpRow[mx];
393 }
394 }
395 }
396 }
397 }, nstripes);
398}

400template<typename T>
401void resizeBilinear(const Mat &inp, Mat &out,
402 float scaleH, float scaleW,
403 int lenY, int lenX,
404 const String &coordTransMode,
405 bool halfPixelCenters,
406 float start_y = 0.0f, float end_y = 1.0f,
407 float start_x = 0.0f, float end_x = 1.0f,
408 float extrapolation_value = 0.0f)
409{
410 int inH = inp.size[2], inW = inp.size[3];
411 int outH = out.size[2], outW = out.size[3];
412 CV_Assert(inp.isContinuous() && out.isContinuous());
413
414 CoordTransMode coordMode = parseCoordTransMode(coordTransMode);
415 const bool tf_crop_and_resize_mode = (coordMode == CoordTransMode::TF_CROP_AND_RESIZE);
416
417 std::vector<int> x0(outW), x1(outW);
418 std::vector<float> lx(outW);
419 std::vector<uint8_t> outOfBoundsX(outW);
420 buildBilinearIndexAndLerp(x0, x1, lx, outOfBoundsX,
421 outW, inW, scaleW, lenX, start_x, end_x,
422 coordMode, halfPixelCenters, tf_crop_and_resize_mode);
423
424 std::vector<int> y0(outH), y1(outH);
425 std::vector<float> ly(outH);
426 std::vector<uint8_t> outOfBoundsY(outH);
427 buildBilinearIndexAndLerp(y0, y1, ly, outOfBoundsY,
428 outH, inH, scaleH, lenY, start_y, end_y,
429 coordMode, halfPixelCenters, tf_crop_and_resize_mode);
430
431 if (inp.shape().layout == DATA_LAYOUT_BLOCK) {
432 CV_Assert(inp.dims == 5 && out.dims == 5);
433 const int N = inp.size[0], C1 = inp.size[1], C0 = inp.size[4];
434
435 const size_t inStep0 = inp.step.p[0] / inp.elemSize();
436 const size_t inStep1 = inp.step.p[1] / inp.elemSize();
437 const size_t inStep2 = inp.step.p[2] / inp.elemSize();
438 const size_t inStep3 = inp.step.p[3] / inp.elemSize();
439 const size_t outStep0 = out.step.p[0] / out.elemSize();
440 const size_t outStep1 = out.step.p[1] / out.elemSize();
441 const size_t outStep2 = out.step.p[2] / out.elemSize();
442 const size_t outStep3 = out.step.p[3] / out.elemSize();
443
444 const int nplanes = N * C1 * outH;
445 parallel_for_(Range(0, nplanes), [&](const Range& range) {
446 const T* inptr0 = reinterpret_cast<const T*>(inp.data);
447 T* outptr0 = reinterpret_cast<T*>(out.data);
448 T ext = saturate_cast<T>(extrapolation_value);
449
450 for (int plane = range.start; plane < range.end; ++plane) {
451 int t = plane;
452 int oy = t % outH; t /= outH;
453 int c1 = t % C1;
454 int n = t / C1;
455
456 T* outRow = outptr0 + n * outStep0 + c1 * outStep1 + oy * outStep2;
457 if (tf_crop_and_resize_mode && outOfBoundsY[oy]) {
458 for (int ox = 0; ox < outW; ++ox) {
459 T* outPix = outRow + ox * outStep3;
460 for (int c0 = 0; c0 < C0; ++c0) outPix[c0] = ext;
461 }
462 continue;
463 }
464
465 const T* row00 = inptr0 + n * inStep0 + c1 * inStep1 + y0[oy] * inStep2;
466 const T* row01 = inptr0 + n * inStep0 + c1 * inStep1 + y1[oy] * inStep2;
467 float fy = ly[oy];
468
469 for (int ox = 0; ox < outW; ++ox) {
470 T* outPix = outRow + ox * outStep3;
471 if (tf_crop_and_resize_mode && outOfBoundsX[ox]) {
472 for (int c0 = 0; c0 < C0; ++c0) outPix[c0] = ext;
473 continue;
474 }
475 const T* p00 = row00 + x0[ox] * inStep3;
476 const T* p01 = row00 + x1[ox] * inStep3;
477 const T* p10 = row01 + x0[ox] * inStep3;
478 const T* p11 = row01 + x1[ox] * inStep3;
479 float fx = lx[ox];
480 for (int c0 = 0; c0 < C0; ++c0) {
481 float top = static_cast<float>(p00[c0]) + fx * (static_cast<float>(p01[c0]) - static_cast<float>(p00[c0]));
482 float bot = static_cast<float>(p10[c0]) + fx * (static_cast<float>(p11[c0]) - static_cast<float>(p10[c0]));
483 outPix[c0] = saturate_cast<T>(top + fy * (bot - top));
484 }
485 }
486 }
487 }, kResizeNumStripes);
488 return;
489 }
490
491 int numPlanes = inp.size[0] * inp.size[1];
492 Mat inpP = inp.reshape(1, numPlanes * inH);
493 Mat outP = out.reshape(1, numPlanes * outH);
494
495 const int nstripes = kResizeNumStripes;
496 parallel_for_(Range(0, nstripes), [&](const Range& range) {
497 int row0 = range.start * (outH * numPlanes) / nstripes;
498 int row1 = range.end * (outH * numPlanes) / nstripes - 1;
499 int plane0 = row0 / outH, plane1 = row1 / outH;
500 row0 %= outH;
501 row1 %= outH;
502
503 const int* y0ptr = y0.data();
504 const int* y1ptr = y1.data();
505 const float* lyptr = ly.data();
506 const int* x0ptr = x0.data();
507 const float* lxptr = lx.data();
508 const uint8_t* outOfBoundsYptr = outOfBoundsY.data();
509 const uint8_t* outOfBoundsXptr = outOfBoundsX.data();
510 float extrapolation_value_ = extrapolation_value;
511 const bool tf_crop_and_resize_mode_ = tf_crop_and_resize_mode;
512 std::vector<float> hbufbuf(inW + 3);
513 float* hbuf = hbufbuf.data() + 1;
514
515 for (int p = plane0; p <= plane1; ++p)
516 {
517 int oy0 = (p == plane0) ? row0 : 0;
518 int oy1 = (p == plane1) ? row1 : outH - 1;
519 for (int oy = oy0; oy <= oy1; ++oy)
520 {
521 if (tf_crop_and_resize_mode_ && outOfBoundsYptr[oy]) {
522 T* outRowFill = outP.ptr<T>(p * outH + oy);
523 for (int ox = 0; ox < outW; ++ox)
524 outRowFill[ox] = T(extrapolation_value_);
525 continue;
526 }
527
528 const T* row0ptr = inpP.ptr<T>( p * inH + y0ptr[oy] );
529 const T* row1ptr = inpP.ptr<T>( p * inH + y1ptr[oy] );
530 float fy = lyptr[oy];
531
532 T* outRowBase = outP.ptr<T>( p * outH + oy );
533
534 for (int ix = 0; ix < inW; ++ix)
535 {
536 float v0 = static_cast<float>(row0ptr[ix]);
537 float v1 = static_cast<float>(row1ptr[ix]);
538 hbuf[ix] = v0 + fy * (v1 - v0);
539 }
540 hbuf[-1] = hbuf[0];
541 hbuf[inW] = hbuf[inW - 1];
542 hbuf[inW + 1] = hbuf[inW - 1];
543
544 for (int ox = 0; ox < outW; ++ox)
545 {
546 if (tf_crop_and_resize_mode_ && outOfBoundsXptr[ox]) {
547 outRowBase[ox] = T(extrapolation_value_);
548 } else {
549 int xi = x0ptr[ox];
550 float fx = lxptr[ox];
551
552 float left = hbuf[xi];
553 float res = left + fx * (hbuf[xi + 1] - left);
554 outRowBase[ox] = T(res);
555 }
556 }
557 }
558 }
559 }, nstripes);
560}

575 CoordTransMode coordMode = parseCoordTransMode(coordTransMode);
576 const bool tf_crop_and_resize_mode = (coordMode == CoordTransMode::TF_CROP_AND_RESIZE);
577
578 std::vector<std::array<int,4>> x_id(outW);
579 std::vector<std::array<float,4>> x_w (outW);
580 std::vector<uint8_t> outOfBoundsX(outW);
581 buildCubicIndexAndWeights(x_id, x_w, outOfBoundsX,
582 outW, inW, scaleW, lenX, start_x, end_x,
583 coordMode, halfPixelCenters, tf_crop_and_resize_mode,
584 excludeOutside, cubicA);
585
586 std::vector<std::array<int,4>> y_id(outH);
587 std::vector<std::array<float,4>> y_w (outH);
588 std::vector<uint8_t> outOfBoundsY(outH);
589 buildCubicIndexAndWeights(y_id, y_w, outOfBoundsY,
590 outH, inH, scaleH, lenY, start_y, end_y,
591 coordMode, halfPixelCenters, tf_crop_and_resize_mode,
592 excludeOutside, cubicA);
593
594 if (inp.shape().layout == DATA_LAYOUT_BLOCK) {
595 CV_Assert(inp.dims == 5 && out.dims == 5);
596 CV_Assert(inp.isContinuous() && out.isContinuous());
597 const int N = inp.size[0], C1 = inp.size[1], C0 = inp.size[4];
598
599 const size_t inStep0 = inp.step.p[0] / inp.elemSize();
600 const size_t inStep1 = inp.step.p[1] / inp.elemSize();
601 const size_t inStep2 = inp.step.p[2] / inp.elemSize();
602 const size_t inStep3 = inp.step.p[3] / inp.elemSize();
603 const size_t outStep0 = out.step.p[0] / out.elemSize();
604 const size_t outStep1 = out.step.p[1] / out.elemSize();
605 const size_t outStep2 = out.step.p[2] / out.elemSize();
606 const size_t outStep3 = out.step.p[3] / out.elemSize();
607
608 const int nplanes = N * C1 * outH;
609 parallel_for_(Range(0, nplanes), [&](const Range& range) {
610 const T* inptr0 = reinterpret_cast<const T*>(inp.data);
611 T* outptr0 = reinterpret_cast<T*>(out.data);
612 T ext = saturate_cast<T>(extrapolation_value);
613
614 for (int plane = range.start; plane < range.end; ++plane) {
615 int t = plane;
616 int oy = t % outH; t /= outH;
617 int c1 = t % C1;
618 int n = t / C1;
619
620 T* outRow = outptr0 + n * outStep0 + c1 * outStep1 + oy * outStep2;
621 if (tf_crop_and_resize_mode && outOfBoundsY[oy]) {
622 for (int ox = 0; ox < outW; ++ox) {
623 T* outPix = outRow + ox * outStep3;
624 for (int c0 = 0; c0 < C0; ++c0) outPix[c0] = ext;
625 }
626 continue;
627 }
628
629 const int yy0 = y_id[oy][0], yy1 = y_id[oy][1], yy2 = y_id[oy][2], yy3 = y_id[oy][3];
630 const float wy0 = y_w[oy][0], wy1 = y_w[oy][1], wy2 = y_w[oy][2], wy3 = y_w[oy][3];
631
632 const T* row0 = yy0 >= 0 ? inptr0 + n * inStep0 + c1 * inStep1 + yy0 * inStep2 : nullptr;
633 const T* row1 = yy1 >= 0 ? inptr0 + n * inStep0 + c1 * inStep1 + yy1 * inStep2 : nullptr;
634 const T* row2 = yy2 >= 0 ? inptr0 + n * inStep0 + c1 * inStep1 + yy2 * inStep2 : nullptr;
635 const T* row3 = yy3 >= 0 ? inptr0 + n * inStep0 + c1 * inStep1 + yy3 * inStep2 : nullptr;
636
637 for (int ox = 0; ox < outW; ++ox) {
638 T* outPix = outRow + ox * outStep3;
639 if (tf_crop_and_resize_mode && outOfBoundsX[ox]) {
640 for (int c0 = 0; c0 < C0; ++c0) outPix[c0] = ext;
641 continue;
642 }
643 const int xx0 = x_id[ox][0], xx1 = x_id[ox][1], xx2 = x_id[ox][2], xx3 = x_id[ox][3];
644 const float wx0 = x_w[ox][0], wx1 = x_w[ox][1], wx2 = x_w[ox][2], wx3 = x_w[ox][3];
645
646 for (int c0 = 0; c0 < C0; ++c0) {
647 float val = 0.f;
648 if (row0) {
649 if (xx0 >= 0) val += wy0 * wx0 * static_cast<float>(row0[xx0 * inStep3 + c0]);
650 if (xx1 >= 0) val += wy0 * wx1 * static_cast<float>(row0[xx1 * inStep3 + c0]);
651 if (xx2 >= 0) val += wy0 * wx2 * static_cast<float>(row0[xx2 * inStep3 + c0]);
652 if (xx3 >= 0) val += wy0 * wx3 * static_cast<float>(row0[xx3 * inStep3 + c0]);
653 }
654 if (row1) {
655 if (xx0 >= 0) val += wy1 * wx0 * static_cast<float>(row1[xx0 * inStep3 + c0]);
656 if (xx1 >= 0) val += wy1 * wx1 * static_cast<float>(row1[xx1 * inStep3 + c0]);
657 if (xx2 >= 0) val += wy1 * wx2 * static_cast<float>(row1[xx2 * inStep3 + c0]);
658 if (xx3 >= 0) val += wy1 * wx3 * static_cast<float>(row1[xx3 * inStep3 + c0]);
659 }
660 if (row2) {
661 if (xx0 >= 0) val += wy2 * wx0 * static_cast<float>(row2[xx0 * inStep3 + c0]);
662 if (xx1 >= 0) val += wy2 * wx1 * static_cast<float>(row2[xx1 * inStep3 + c0]);
663 if (xx2 >= 0) val += wy2 * wx2 * static_cast<float>(row2[xx2 * inStep3 + c0]);
664 if (xx3 >= 0) val += wy2 * wx3 * static_cast<float>(row2[xx3 * inStep3 + c0]);
665 }
666 if (row3) {
667 if (xx0 >= 0) val += wy3 * wx0 * static_cast<float>(row3[xx0 * inStep3 + c0]);
668 if (xx1 >= 0) val += wy3 * wx1 * static_cast<float>(row3[xx1 * inStep3 + c0]);
669 if (xx2 >= 0) val += wy3 * wx2 * static_cast<float>(row3[xx2 * inStep3 + c0]);
670 if (xx3 >= 0) val += wy3 * wx3 * static_cast<float>(row3[xx3 * inStep3 + c0]);
671 }
672 outPix[c0] = saturate_cast<T>(val);
673 }
674 }
675 }
676 }, kResizeNumStripes);
677 return;
678 }
679
680 int numPlanes = inp.size[0] * inp.size[1];
681 Mat inpPlanes = inp.reshape(1, numPlanes * inH);
682 Mat outPlanes = out.reshape(1, numPlanes * outH);
683
684 const int nstripes = kResizeNumStripes;
685 parallel_for_(Range(0, nstripes), [&](const Range& range) {
686 int row0 = range.start * (outH * numPlanes) / nstripes;
687 int row1 = range.end * (outH * numPlanes) / nstripes - 1;
688 int plane0 = row0 / outH, plane1 = row1 / outH;
689 row0 %= outH;
690 row1 %= outH;
691
692 const bool tf_crop_and_resize_mode_ = tf_crop_and_resize_mode;
693 const uint8_t* outOfBoundsYptr = outOfBoundsY.data();
694 const uint8_t* outOfBoundsXptr = outOfBoundsX.data();
695 float extrapolation_value_ = extrapolation_value;
696 std::vector<float> hbuf(inW, 0.f);
697
698 for (int p = plane0; p <= plane1; ++p)
699 {
700 int oy0 = (p == plane0) ? row0 : 0;
701 int oy1 = (p == plane1) ? row1 : outH - 1;
702 const T* inpBase = inpPlanes.ptr<T>(p * inH);
703 for (int oy = oy0; oy <= oy1; ++oy)
704 {
705 T* outRow = outPlanes.ptr<T>(p * outH) + oy * outW;
706
707 if (tf_crop_and_resize_mode_ && outOfBoundsYptr[oy]) {
708 for (int ox = 0; ox < outW; ++ox)
709 outRow[ox] = cv::saturate_cast<T>(extrapolation_value_);
710 continue;
711 }
712
713 const float w0y = y_w[oy][0];
714 const float w1y = y_w[oy][1];
715 const float w2y = y_w[oy][2];
716 const float w3y = y_w[oy][3];
717
718 int yy0, yy1, yy2, yy3;
719 yy0 = y_id[oy][0];
720 yy1 = y_id[oy][1];
721 yy2 = y_id[oy][2];
722 yy3 = y_id[oy][3];
723
724 const T* ptr0 = (yy0 >= 0) ? (inpBase + (size_t)yy0 * inW) : nullptr;
725 const T* ptr1 = (yy1 >= 0) ? (inpBase + (size_t)yy1 * inW) : nullptr;
726 const T* ptr2 = (yy2 >= 0) ? (inpBase + (size_t)yy2 * inW) : nullptr;
727 const T* ptr3 = (yy3 >= 0) ? (inpBase + (size_t)yy3 * inW) : nullptr;
728
729 if (!ptr0 && !ptr1 && !ptr2 && !ptr3) {
730 for (int ix = 0; ix < inW; ++ix)
731 {
732 hbuf[ix] = 0.f;
733 }
734 } else {
735 const T* ptrNZ = ptr0 ? ptr0 : (ptr1 ? ptr1 : (ptr2 ? ptr2 : ptr3));
736 float w0 = ptr0 ? w0y : 0.f;
737 float w1 = ptr1 ? w1y : 0.f;
738 float w2 = ptr2 ? w2y : 0.f;
739 float w3 = ptr3 ? w3y : 0.f;
740 if (!ptr0) ptr0 = ptrNZ;
741 if (!ptr1) ptr1 = ptrNZ;
742 if (!ptr2) ptr2 = ptrNZ;
743 if (!ptr3) ptr3 = ptrNZ;
744
745 for (int ix = 0; ix < inW; ++ix)
746 {
747 hbuf[ix] = static_cast<float>(ptr0[ix]) * w0 +
748 static_cast<float>(ptr1[ix]) * w1 +
749 static_cast<float>(ptr2[ix]) * w2 +
750 static_cast<float>(ptr3[ix]) * w3;
751 }
752 }
753
754 for (int ox = 0; ox < outW; ++ox)
755 {
756 if (tf_crop_and_resize_mode_ && outOfBoundsXptr[ox]) {
757 outRow[ox] = cv::saturate_cast<T>(extrapolation_value_);
758 continue;
759 }
760 const int xx = x_id[ox][1];
761 const float w0x = x_w[ox][0];
762 const float w1x = x_w[ox][1];
763 const float w2x = x_w[ox][2];
764 const float w3x = x_w[ox][3];
765 float val;
766 if (1 <= xx && xx + 3 < inW) {
767 val = hbuf[xx - 1] * w0x + hbuf[xx] * w1x + hbuf[xx + 1] * w2x + hbuf[xx + 2] * w3x;
768 } else {
769 const int xx0 = x_id[ox][0];
770 const int xx1 = x_id[ox][1];
771 const int xx2 = x_id[ox][2];
772 const int xx3 = x_id[ox][3];
773 val = 0.f;
774 if (xx0 >= 0) val += hbuf[xx0] * w0x;
775 if (xx1 >= 0) val += hbuf[xx1] * w1x;
776 if (xx2 >= 0) val += hbuf[xx2] * w2x;
777 if (xx3 >= 0) val += hbuf[xx3] * w3x;
778 }
779 outRow[ox] = cv::saturate_cast<T>(val);
780 }
781 }
782 }
783 }, nstripes);
784}

786// ---- ONNX antialias (PIL-style) resampling ----------------------------------
787static inline float aaTriangle(float x)
788{
789 x = std::abs(x);
790 return x < 1.f ? 1.f - x : 0.f;
791}
792
793static inline float aaCubic(float x, float a)
794{
795 x = std::abs(x);
796 if (x < 1.f) return ((a + 2.f)*x - (a + 3.f))*x*x + 1.f;
797 if (x < 2.f) return a*(((x - 5.f)*x + 8.f)*x - 4.f);
798 return 0.f;
799}

801// Per-output filter taps for one axis. After clamping out-of-bound samples to
802// the edge (exclude_outside == false), the contributing indices are contiguous,
803// so each output stores a start index 'lo', a 'cnt' and an offset into 'w'.
804struct AAWeights
805{
806 std::vector<int> lo, cnt, ofs;
807 std::vector<float> w;
808};
809
810static void buildAAWeights(AAWeights& p, int inS, int outS, float xscale,
811 bool cubic, float cubicA, CoordTransMode coordMode)
812{
813 const float scaleC = 1.f / xscale; // input/output direction
814 const float radius = cubic ? 2.f : 1.f;
815 const float support = scaleC >= 1.f ? radius*scaleC : radius;
816 const float inv = scaleC >= 1.f ? 1.f/scaleC : 1.f;
817
818 p.lo.resize(outS); p.cnt.resize(outS); p.ofs.resize(outS);
819 p.w.clear();
820 std::vector<float> tmp;
821 for (int y = 0; y < outS; y++)
822 {
823 const float center = computeSrcGeneric(y, scaleC, inS, outS, coordMode, true);
824 const int xmin = (int)std::floor(center - support + 0.5f);
825 const int xmax = (int)std::floor(center + support + 0.5f); // inclusive
826 const int lo = std::min(std::max(xmin, 0), inS - 1);
827 const int hi = std::min(std::max(xmax, 0), inS - 1);
828 const int cnt = hi - lo + 1;
829 tmp.assign(cnt, 0.f);
830 float tot = 0.f;
831 for (int x = xmin; x <= xmax; x++)
832 {
833 const float wt = cubic ? aaCubic((x - center)*inv, cubicA)
834 : aaTriangle((x - center)*inv);
835 const int idx = std::min(std::max(x, 0), inS - 1);
836 tmp[idx - lo] += wt;
837 tot += wt;
838 }
839 p.lo[y] = lo; p.cnt[y] = cnt; p.ofs[y] = (int)p.w.size();
840 for (int k = 0; k < cnt; k++)
841 p.w.push_back(tot != 0.f ? tmp[k] / tot : 0.f);
842 }
843}

845template<typename T>
846void resizeAntialias(const Mat& inp, Mat& out,
847 float xscaleH, float xscaleW,
848 bool cubic, float cubicA, CoordTransMode coordMode)
849{
850 CV_Assert(inp.dims == 4 && out.dims == 4 && inp.isContinuous() && out.isContinuous());
851 const int N = inp.size[0], C = inp.size[1];
852 const int inH = inp.size[2], inW = inp.size[3];
853 const int outH = out.size[2], outW = out.size[3];
854
855 AAWeights px, py;
856 buildAAWeights(px, inW, outW, xscaleW, cubic, cubicA, coordMode);
857 buildAAWeights(py, inH, outH, xscaleH, cubic, cubicA, coordMode);
858
859 const int planes = N * C;
860 parallel_for_(Range(0, planes), [&](const Range& r) {
861 std::vector<float> buf((size_t)inH * outW);
862 for (int pl = r.start; pl < r.end; pl++)
863 {
864 const T* inPlane = inp.ptr<T>(0) + (size_t)pl * inH * inW;
865 T* outPlane = out.ptr<T>(0) + (size_t)pl * outH * outW;
866 // Horizontal pass: inp[inH x inW] -> buf[inH x outW].
867 for (int y = 0; y < inH; y++)
868 {
869 const T* inRow = inPlane + (size_t)y * inW;
870 float* bufRow = buf.data() + (size_t)y * outW;
871 for (int ox = 0; ox < outW; ox++)
872 {
873 const float* w = px.w.data() + px.ofs[ox];
874 const int lo = px.lo[ox], cnt = px.cnt[ox];
875 float acc = 0.f;
876 for (int k = 0; k < cnt; k++)
877 acc += w[k] * (float)inRow[lo + k];
878 bufRow[ox] = acc;
879 }
880 }
881 // Vertical pass: buf[inH x outW] -> out[outH x outW].
882 for (int oy = 0; oy < outH; oy++)
883 {
884 const float* w = py.w.data() + py.ofs[oy];
885 const int lo = py.lo[oy], cnt = py.cnt[oy];
886 T* outRow = outPlane + (size_t)oy * outW;
887 for (int ox = 0; ox < outW; ox++)
888 {
889 float acc = 0.f;
890 for (int k = 0; k < cnt; k++)
891 acc += w[k] * buf[(size_t)(lo + k) * outW + ox];
892 outRow[ox] = saturate_cast<T>(acc);
893 }
894 }
895 }
896 });
897}

1007 MatShape getOutShape(const MatShape& inpShape, const std::vector<int>& sizes,
1008 const std::vector<float>& scales) const
1009 {
1010 // ONNX Resize allows either "sizes" or "scales" input. These tensors may
1011 // describe all 4 dims (N,C,H,W) or only spatial dims (H,W) when accompanied
1012 // by an "axes" input equal to {2,3}. To stay backwards-compatible, we keep
1013 // the legacy 4-element handling but also accept 2-element vectors.
1014
1015 CV_Assert((sizes.empty() ^ scales.empty()) &&
1016 (sizes.empty() ? (scales.size() == 4 || scales.size() == 2)
1017 : (sizes.size() == 4 || sizes.size() == 2)));
1018
1019 MatShape outShape = inpShape;
1020 const int inH = inpShape[2], inW = inpShape[3];
1021 if (!sizes.empty()) {
1022 int hIdx, wIdx;
1023 spatialIndices(sizes.size(), hIdx, wIdx);
1024 int szH = sizes[hIdx], szW = sizes[wIdx];
1025 if (keepAspectPolicy == "not_larger" || keepAspectPolicy == "not_smaller") {
1026 float scH = float(szH) / inH, scW = float(szW) / inW;
1027 float sc = keepAspectPolicy == "not_larger" ? std::min(scH, scW)
1028 : std::max(scH, scW);
1029 outShape[2] = int(std::round(sc * inH));
1030 outShape[3] = int(std::round(sc * inW));
1031 } else {
1032 outShape[2] = szH;
1033 outShape[3] = szW;
1034 }
1035 } else {
1036 int hIdx, wIdx;
1037 spatialIndices(scales.size(), hIdx, wIdx);
1038 outShape[2] = cvFloor(inH * scales[hIdx]);
1039 outShape[3] = cvFloor(inW * scales[wIdx]);
1040 }
1041 return outShape;
1042 }

1061 // (blobs[0] = roi, blobs[1] = scales, blobs[2] = sizes, blobs[3] = axes).
1062 if (ninputs == 1 && !this->blobs.empty()) {
1063 std::vector<int> sizes;
1064 std::vector<float> scales;
1065 if (this->blobs.size() >= 3 && this->blobs[2].total() > 0)
1066 tensorToIntVec(this->blobs[2], sizes);
1067 if (this->blobs.size() >= 2 && this->blobs[1].total() > 0)
1068 tensorToFloatVec(this->blobs[1], scales);
1069
1070 if (!sizes.empty() || !scales.empty()) {
1071 outputs[0] = getOutShape(inputs[0], sizes, scales);
1072 // in-place if spatial dims unchanged
1073 return (outputs[0][2] == inputs[0][2]) && (outputs[0][3] == inputs[0][3]);
1074 }
1075 }
1076
1077 if (ninputs == 1) {
1078 outputs[0][2] = zoomFactorHeight > 0 ? cvFloor(inputs[0][2] * zoomFactorHeight) : outHeight0;
1079 outputs[0][3] = zoomFactorWidth > 0 ? cvFloor(inputs[0][3] * zoomFactorWidth) : outWidth0;
1080 } else if (ninputs == 2 && inputs[1].dims == 4) {
1081 outputs[0][2] = inputs[1][2];
1082 outputs[0][3] = inputs[1][3];
1083 } else {
1084 Net::Impl* netimpl_ = getNetImpl(this);
1085 std::vector<int> sizes;
1086 std::vector<float> scales;
1087 if (ninputs >= 4) {
1088 Mat sizesTensor = netimpl_->argTensor(this->inputs[3]);
1089 tensorToIntVec(sizesTensor, sizes);
1090 }
1091
1092 Mat scalesTensor = netimpl_->argTensor(this->inputs[(ninputs == 2) ? 1 : 2]);
1093 tensorToFloatVec(scalesTensor, scales);
1094 outputs[0] = getOutShape(inputs[0], sizes, scales);
1095 }
1096 // We can work in-place (do nothing) if input shape == output shape.
1097 return (outputs[0][2] == inputs[0][2]) && (outputs[0][3] == inputs[0][3]);
1098 }

1134 void updateOutSizeAndScale(const MatShape& inpShape, const MatShape& outShape)
1135 {
1136 CV_Assert(inpShape.dims >= 4 && outShape.dims >= 4);
1137 outHeight = outShape[2];
1138 outWidth = outShape[3];
1139 if (alignCorners && outHeight > 1)
1140 scaleHeight = float(inpShape[2] - 1) / (outHeight - 1);
1141 else
1142 scaleHeight = float(inpShape[2]) / outHeight;
1143
1144 if (alignCorners && outWidth > 1)
1145 scaleWidth = float(inpShape[3] - 1) / (outWidth - 1);
1146 else
1147 scaleWidth = float(inpShape[3]) / outWidth;

1150 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr,
1151 OutputArrayOfArrays internals_arr) CV_OVERRIDE
1152 {
1153 CV_TRACE_FUNCTION();
1154 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
1155
1156 std::vector<int> sizes;
1157 std::vector<float> scales;
1158 std::vector<Mat> inputs;
1159 inputs_arr.getMatVector(inputs);
1160 size_t ninputs = inputs.size();
1161 CV_Assert(ninputs > 0);
1162
1163 Mat& inp_ = inputs[0];
1164
1165 MatShape inpShape = inp_.shape();
1166 MatShape outShape;
1167
1168 // Rank-3 (N,C,W): fold a unit H axis so the rank-4 kernels run, unfold on output.
1169 const bool fold1d = (inp_.dims == 3);
1170 MatShape outShape1d;
1171 if (fold1d) {
1172 int outW = zoomFactorWidth > 0 ? cvFloor(inpShape[2] * zoomFactorWidth) : outWidth0;
1173 outShape1d = inpShape;
1174 outShape1d[2] = outW;
1175 inp_ = inp_.reshape(1, MatShape({inpShape[0], inpShape[1], 1, inpShape[2]}));
1176 inpShape = inp_.shape();
1177 }
1178
1179 if (ninputs == 1) {
1180 outShape = inpShape;
1181 outShape[2] = zoomFactorHeight > 0 ? cvFloor(inpShape[2] * zoomFactorHeight) : outHeight0;
1182 outShape[3] = zoomFactorWidth > 0 ? cvFloor(inpShape[3] * zoomFactorWidth) : outWidth0;
1183 } else if (ninputs == 2 && inputs[0].dims == 4 && inputs[1].dims == 4) {
1184 outShape = inpShape;
1185 outShape[2] = inputs[1].size[2];
1186 outShape[3] = inputs[1].size[3];
1187 } else {
1188 if (ninputs >= 4) {
1189 Mat sizesTensor = inputs[3];
1190 tensorToIntVec(sizesTensor, sizes);
1191 }
1192 Mat scalesTensor = inputs[(ninputs == 2) ? 1 : 2];
1193 tensorToFloatVec(scalesTensor, scales);
1194 outShape = getOutShape(inpShape, sizes, scales);
1195 }
1196
1197 int length_resized_y = outShape[2];
1198 int length_resized_x = outShape[3];
1199 updateOutSizeAndScale(inpShape, outShape);
1200
1201 // Read ROI if dynamicROI is enabled
1202 if (dynamicROI && coordTransModeE == CoordTransMode::TF_CROP_AND_RESIZE && ninputs >= 2)
1203 {
1204 Mat roiTensor = inputs[1];
1205 std::vector<float> roi;
1206 tensorToFloatVec(roiTensor, roi);
1207 if (axesAttr.size() == 2 && roi.size() == 4)
1208 {
1209 // ROI given per "axes": [start_axes[0], start_axes[1], end_axes[0], end_axes[1]]
1210 float start[4] = {0.f, 0.f, 0.f, 0.f};
1211 float end[4] = {1.f, 1.f, 1.f, 1.f};
1212 for (int k = 0; k < 2; k++) {
1213 int ax = axesAttr[k] < 0 ? axesAttr[k] + 4 : axesAttr[k];
1214 start[ax] = roi[k];
1215 end[ax] = roi[2 + k];
1216 }
1217 roi_start_y = start[2]; roi_start_x = start[3];
1218 roi_end_y = end[2]; roi_end_x = end[3];
1219 }
1220 else if (roi.size() >= 4)
1221 {
1222 if (roi.size() == 4) {
1223 roi_start_y = roi[0];
1224 roi_start_x = roi[1];
1225 roi_end_y = roi[2];
1226 roi_end_x = roi[3];
1227 } else if (roi.size() == 6) {
1228 roi_start_y = roi[1];
1229 roi_start_x = roi[2];
1230 roi_end_y = roi[4];
1231 roi_end_x = roi[5];
1232 } else if (roi.size() == 8) {
1233 roi_start_y = roi[2];
1234 roi_start_x = roi[3];
1235 roi_end_y = roi[6];
1236 roi_end_x = roi[7];
1237 }
1238 }
1239 }
1240
1241 if (sizes.empty() && !scales.empty() && halfPixelCenters)
1242 {
1243 int hIdx, wIdx;
1244 spatialIndices(scales.size(), hIdx, wIdx);
1245 scaleHeight = 1.f / scales[hIdx];
1246 scaleWidth = 1.f / scales[wIdx];
1247 }
1248 else if (sizes.empty() && !scales.empty() && alignCorners)
1249 {
1250 int hIdx, wIdx;
1251 spatialIndices(scales.size(), hIdx, wIdx);
1252 float lenH = inpShape[2] * scales[hIdx];
1253 float lenW = inpShape[3] * scales[wIdx];
1254 if (lenH > 1.f) scaleHeight = float(inpShape[2] - 1) / (lenH - 1.f);
1255 if (lenW > 1.f) scaleWidth = float(inpShape[3] - 1) / (lenW - 1.f);
1256 }
1257
1258 auto kind = outputs_arr.kind();
1259 Mat out_;
1260 UMat uout_;
1261 if (kind == _InputArray::STD_VECTOR_MAT) {
1262 std::vector<Mat>& outputs = outputs_arr.getMatVecRef();
1263 outputs[0].fit(fold1d ? outShape1d : outShape, inp_.type());
1264 out_ = fold1d ? outputs[0].reshape(1, outShape) : outputs[0];
1265
1266 if (outShape == inpShape)
1267 {
1268 inp_.copyTo(out_);
1269 return;
1270 }
1271 }
1272 else {
1273 CV_Assert(kind == _InputArray::STD_VECTOR_UMAT);
1274 std::vector<UMat>& u_outputs = outputs_arr.getUMatVecRef();
1275 u_outputs[0].fit(fold1d ? outShape1d : outShape, inp_.type());
1276 uout_ = fold1d ? u_outputs[0].reshape(1, outShape) : u_outputs[0];
1277 if (outShape == inpShape)
1278 {
1279 inp_.copyTo(uout_);
1280 return;
1281 }
1282 out_.create(outShape, inp_.type());
1283 }
1284
1285 int depth = inp_.type(), orig_depth = depth;
1286
1287 // Bilinear/cubic/antialias have no fixed-point CV_32S path yet.
1288 if (depth == CV_32S && interpolation != "nearest") {
1289 CV_Error(Error::StsNotImplemented,
1290 "Resize2: CV_32S is currently only supported with nearest-neighbor "
1291 "interpolation; bilinear/cubic/antialias would need a fixed-point "
1292 "implementation to preserve int32 precision");
1293 }
1294
1295 Mat inp, out;
1296 if (depth != CV_32F && depth != CV_8S && depth != CV_8U && depth != CV_16F && depth != CV_16BF &&
1297 depth != CV_32S) {
1298 inp_.convertTo(inp, CV_32F);
1299 out.fit(outShape, CV_32F);
1300 depth = CV_32F;
1301 } else {
1302 inp = inp_;
1303 out = out_;
1304 }
1305
1306 if (antialias && inp.dims == 4 &&
1307 (interpolation == "bilinear" || interpolation == "opencv_linear" || interpolation == "cubic"))
1308 {
1309 const bool cubic = (interpolation == "cubic");
1310 float xsH, xsW;
1311 if (!scales.empty()) {
1312 int hIdx, wIdx;
1313 spatialIndices(scales.size(), hIdx, wIdx);
1314 xsH = scales[hIdx]; xsW = scales[wIdx];
1315 } else {
1316 xsH = float(outShape[2]) / inpShape[2];
1317 xsW = float(outShape[3]) / inpShape[3];
1318 }
1319 switch (depth) {
1320 case CV_8S: resizeAntialias<int8_t>(inp, out, xsH, xsW, cubic, cubicCoeffA, coordTransModeE); break;
1321 case CV_8U: resizeAntialias<uint8_t>(inp, out, xsH, xsW, cubic, cubicCoeffA, coordTransModeE); break;
1322 case CV_16F: resizeAntialias<hfloat>(inp, out, xsH, xsW, cubic, cubicCoeffA, coordTransModeE); break;
1323 case CV_16BF: resizeAntialias<bfloat>(inp, out, xsH, xsW, cubic, cubicCoeffA, coordTransModeE); break;
1324 case CV_32F: resizeAntialias<float>(inp, out, xsH, xsW, cubic, cubicCoeffA, coordTransModeE); break;
1325 default: CV_Error(Error::StsUnsupportedFormat, "Unsupported depth");
1326 }
1327 }
1328 else if(interpolation=="nearest"){
1329 switch(depth){
1330 case CV_8S:
1331 resizeNearest<int8_t>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,nearestModeE,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1332 break;
1333 case CV_8U:
1334 resizeNearest<uint8_t>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,nearestModeE,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1335 break;
1336 case CV_16F:
1337 resizeNearest<hfloat>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,nearestModeE,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1338 break;
1339 case CV_16BF:
1340 resizeNearest<bfloat>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,nearestModeE,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1341 break;
1342 case CV_32F:
1343 resizeNearest<float>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,nearestModeE,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1344 break;
1345 case CV_32S:
1346 resizeNearest<int32_t>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,nearestModeE,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1347 break;
1348 default: CV_Error(Error::StsUnsupportedFormat,"Unsupported depth");
1349 }
1350 }
1351 else if(interpolation=="bilinear"||interpolation=="opencv_linear"){
1352 switch(depth){
1353 case CV_8S:
1354 resizeBilinear<int8_t>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1355 break;
1356 case CV_8U:
1357 resizeBilinear<uint8_t>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1358 break;
1359 case CV_16F:
1360 resizeBilinear<hfloat>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1361 break;
1362 case CV_16BF:
1363 resizeBilinear<bfloat>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1364 break;
1365 case CV_32F:
1366 resizeBilinear<float>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1367 break;
1368 default: CV_Error(Error::StsUnsupportedFormat,"Unsupported depth");
1369 }
1370 }
1371 else if(interpolation=="cubic"){
1372 switch (depth) {
1373 case CV_8S:
1374 resizeCubic<int8_t>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,cubicCoeffA,excludeOutside,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1375 break;
1376 case CV_8U:
1377 resizeCubic<uint8_t>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,cubicCoeffA,excludeOutside,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1378 break;
1379 case CV_16F:
1380 resizeCubic<hfloat>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,cubicCoeffA,excludeOutside,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1381 break;
1382 case CV_16BF:
1383 resizeCubic<bfloat>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,cubicCoeffA,excludeOutside,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1384 break;
1385 case CV_32F:
1386 resizeCubic<float>(inp,out,scaleHeight,scaleWidth,length_resized_y,length_resized_x,cubicCoeffA,excludeOutside,coordTransMode,halfPixelCenters,roi_start_y,roi_end_y,roi_start_x,roi_end_x,extrapolation_value);
1387 break;
1388 default:
1389 CV_Error(Error::StsUnsupportedFormat, "Unsupported depth");
1390 }
1391 }
1392 else
1393 CV_Error(Error::StsNotImplemented,"Unknown interpolation: "+interpolation);
1394
1395 if (orig_depth != depth) {
1396 if (!uout_.empty())
1397 out.convertTo(uout_, orig_depth);
1398 else
1399 out.convertTo(out_, orig_depth);
1400 }
1401 else if (!uout_.empty()) {
1402 out.copyTo(uout_);
1403 }
1404 }
