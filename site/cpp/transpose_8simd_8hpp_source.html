93void transpose_32bit_blocks_simd(const uchar* src, size_t sstep,
94 uchar* dst, size_t dstep, Size sz)
95{
96 const uint32_t* src32 = reinterpret_cast<const uint32_t*>(src);
97 uint32_t* dst32 = reinterpret_cast<uint32_t*>(dst);
98 const size_t sstep_e = sstep / sizeof(uint32_t);
99 const size_t dstep_e = dstep / sizeof(uint32_t);
100 const int m = sz.width, n = sz.height;
101
102 // Single-threaded for small inputs to avoid parallel_for_ overhead.
103 const int64_t kMinParallelElements = 64 * 1024;
104 if ((int64_t)m * n < kMinParallelElements)
105 {
106 transpose_32bit_blocks_tile(src32, sstep_e, dst32, dstep_e, 0, m, 0, n);
107 return;
108 }
109
110 const int TILE = 32;
111 const int mtiles = (m + TILE - 1) / TILE;
112 const int ntiles = (n + TILE - 1) / TILE;
113 const int64_t totalTiles = (int64_t)mtiles * ntiles;
114 parallel_for_(Range(0, (int)totalTiles), [&](const Range& r)
115 {
116 for (int idx = r.start; idx < r.end; idx++)
117 {
118 int ti = idx / ntiles;
119 int tj = idx - ti * ntiles;
120 int i_lo = ti * TILE, i_hi = std::min(i_lo + TILE, m);
121 int j_lo = tj * TILE, j_hi = std::min(j_lo + TILE, n);
122 transpose_32bit_blocks_tile(src32, sstep_e, dst32, dstep_e, i_lo, i_hi, j_lo, j_hi);
123 }
124 });
125}
