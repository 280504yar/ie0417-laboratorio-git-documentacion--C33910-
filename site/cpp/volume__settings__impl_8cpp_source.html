11static Vec4i calcVolumeStrides(Point3i volumeResolution, bool ZFirstMemOrder)
12{
13 // (xRes*yRes*zRes) array
14 // Depending on zFirstMemOrder arg:
15 // &elem(x, y, z) = data + x*zRes*yRes + y*zRes + z;
16 // &elem(x, y, z) = data + x + y*xRes + z*xRes*yRes;
17 int xdim, ydim, zdim;
18 if (ZFirstMemOrder)
19 {
20 xdim = volumeResolution.z * volumeResolution.y;
21 ydim = volumeResolution.z;
22 zdim = 1;
23 }
24 else
25 {
26 xdim = 1;
27 ydim = volumeResolution.x;
28 zdim = volumeResolution.x * volumeResolution.y;
29 }
30 return Vec4i(xdim, ydim, zdim);
31}
