16static void drawTriangle(Vec4f verts[3], Vec3f colors[3], Mat& depthBuf, Mat& colorBuf,
17 TriangleRasterizeSettings settings)
18{
19 // this will be useful during refactoring
20 // if there's gonna be more supported data types
21 CV_DbgAssert(depthBuf.empty() || depthBuf.type() == CV_32FC1);
22 CV_DbgAssert(colorBuf.empty() || colorBuf.type() == CV_32FC3);
23
24 // any of buffers can be empty
25 int width = std::max(colorBuf.cols, depthBuf.cols);
26 int height = std::max(colorBuf.rows, depthBuf.rows);
27
28 Point minPt(width, height), maxPt(0, 0);
29 for (int i = 0; i < 3; i++)
30 {
31 // round down to cover the whole pixel
32 int x = (int)(verts[i][0]), y = (int)(verts[i][1]);
33 minPt.x = std::min( x, minPt.x);
34 minPt.y = std::min( y, minPt.y);
35 maxPt.x = std::max(x + 1, maxPt.x);
36 maxPt.y = std::max(y + 1, maxPt.y);
37 }
38
39 minPt.x = std::max(minPt.x, 0); maxPt.x = std::min(maxPt.x, width);
40 minPt.y = std::max(minPt.y, 0); maxPt.y = std::min(maxPt.y, height);
41
42 Point2f a(verts[0][0], verts[0][1]), b(verts[1][0], verts[1][1]), c(verts[2][0], verts[2][1]);
43 Point2f bc = b - c, ac = a - c;
44 float d = ac.x*bc.y - ac.y*bc.x;
45
46 // culling and degenerated triangle removal
47 if ((settings.cullingMode == RASTERIZE_CULLING_CW && d <= 0) ||
48 (settings.cullingMode == RASTERIZE_CULLING_CCW && d >= 0) ||
49 (abs(d) < 1e-6))
50 {
51 return;
52 }
53
54 float invd = 1.f / d;
55 Vec3f zinv { verts[0][2], verts[1][2], verts[2][2] };
56 Vec3f w { verts[0][3], verts[1][3], verts[2][3] };
57
58 for (int y = minPt.y; y < maxPt.y; y++)
59 {
60 for (int x = minPt.x; x < maxPt.x; x++)
61 {
62 Point2f p(x + 0.5f, y + 0.5f), pc = p - c;
63 // barycentric coordinates
64 Vec3f f;
65 f[0] = ( pc.x * bc.y - pc.y * bc.x) * invd;
66 f[1] = ( pc.y * ac.x - pc.x * ac.y) * invd;
67 f[2] = 1.f - f[0] - f[1];
68 // if inside the triangle
69 if ((f[0] >= 0) && (f[1] >= 0) && (f[2] >= 0))
70 {
71 bool update = false;
72 if (!depthBuf.empty())
73 {
74 float zCurrent = depthBuf.at<float>(height - 1 - y, x);
75 float zNew = f[0] * zinv[0] + f[1] * zinv[1] + f[2] * zinv[2];
76 if (zNew < zCurrent)
77 {
78 update = true;
79 depthBuf.at<float>(height - 1 - y, x) = zNew;
80 }
81 }
82 else // RASTERIZE_SHADING_WHITE
83 {
84 update = true;
85 }
86
87 if (!colorBuf.empty() && update)
88 {
89 Vec3f color;
90 if (settings.shadingType == RASTERIZE_SHADING_WHITE)
91 {
92 color = { 1.f, 1.f, 1.f };
93 }
94 else if (settings.shadingType == RASTERIZE_SHADING_FLAT)
95 {
96 color = colors[0];
97 }
98 else // TriangleShadingType::Shaded
99 {
100 float zInter = 1.0f / (f[0] * w[0] + f[1] * w[1] + f[2] * w[2]);
101 color = { 0, 0, 0 };
102 for (int j = 0; j < 3; j++)
103 {
104 color += (f[j] * w[j]) * colors[j];
105 }
106 color *= zInter;
107 }
108 colorBuf.at<Vec3f>(height - 1 - y, x) = color;
109 }
110 }
111 }
112 }
113}

116// values outside of [zNear, zFar] have to be restored
117// [0, 1] -> [zNear, zFar]
118static void linearizeDepth(const Mat& inbuf, const Mat& validMask, Mat outbuf, double zFar, double zNear)
119{
120 CV_Assert(inbuf.type() == CV_32FC1);
121 CV_Assert(validMask.type() == CV_8UC1 || validMask.type() == CV_8SC1 || validMask.type() == CV_BoolC1);
122 CV_Assert(outbuf.type() == CV_32FC1);
123 CV_Assert(outbuf.size() == inbuf.size());
124
125 float scaleNear = (float)(1.0 / zNear);
126 float scaleFar = (float)(1.0 / zFar);
127 for (int y = 0; y < inbuf.rows; y++)
128 {
129 const float* inp = inbuf.ptr<float>(y);
130 const uchar * validPtr = validMask.ptr<uchar>(y);
131 float * outp = outbuf.ptr<float>(y);
132 for (int x = 0; x < inbuf.cols; x++)
133 {
134 if (validPtr[x])
135 {
136 float d = inp[x];
137 // precision-optimized version of this:
138 //float z = - zFar * zNear / (d * (zFar - zNear) - zFar);
139 float z = 1.f / ((1.f - d) * scaleNear + d * scaleFar );
140 outp[x] = z;
141 }
142 }
143 }
144}

146// [zNear, zFar] -> [0, 1]
147static void invertDepth(const Mat& inbuf, Mat& outbuf, Mat& validMask, double zNear, double zFar)
148{
149 CV_Assert(inbuf.type() == CV_32FC1);
150 outbuf.create(inbuf.size(), CV_32FC1);
151 validMask.create(inbuf.size(), CV_8UC1);
152
153 float fNear = (float)zNear, fFar = (float)zFar;
154 float zadd = (float)(zFar / (zFar - zNear));
155 float zmul = (float)(-zNear * zFar / (zFar - zNear));
156 for (int y = 0; y < inbuf.rows; y++)
157 {
158 const float * inp = inbuf.ptr<float>(y);
159 float * outp = outbuf.ptr<float>(y);
160 uchar * validPtr = validMask.ptr<uchar>(y);
161 for (int x = 0; x < inbuf.cols; x++)
162 {
163 float z = inp[x];
164 uchar m = (z >= fNear) && (z <= fFar);
165 z = std::max(std::min(z, fFar), fNear);
166 // precision-optimized version of this:
167 // outp[x] = (z - zNear) / z * zFar / (zFar - zNear);
168 outp[x] = zadd + zmul / z;
169 validPtr[x] = m;
170 }
171 }
172}

175static void triangleRasterizeInternal(InputArray _vertices, InputArray _indices, InputArray _colors,
176 Mat& colorBuf, Mat& depthBuf,
177 InputArray world2cam, double fovyRadians, double zNear, double zFar,
178 const TriangleRasterizeSettings& settings)
179{
180 CV_Assert(world2cam.type() == CV_32FC1 || world2cam.type() == CV_64FC1);
181 CV_Assert((world2cam.size() == Size {4, 3}) || (world2cam.size() == Size {4, 4}));
182
183 CV_Assert((fovyRadians > 0) && (fovyRadians < CV_PI));
184 CV_Assert(zNear > 0);
185 CV_Assert(zFar > zNear);
186
187 Mat cpMat;
188 world2cam.getMat().convertTo(cpMat, CV_64FC1);
189 Matx44d camPoseMat = Matx44d::eye();
190 for (int i = 0; i < 3; i++)
191 {
192 for (int j = 0; j < 4; j++)
193 {
194 camPoseMat(i, j) = cpMat.at<double>(i, j);
195 }
196 }
197
198 if(_indices.empty())
199 {
200 return;
201 }
202
203 CV_CheckFalse(_vertices.empty(), "No vertices provided along with indices array");
204
205 Mat vertices, colors, triangles;
206 int nVerts = 0, nColors = 0, nTriangles = 0;
207
208 int vertexType = _vertices.type();
209 CV_Assert(vertexType == CV_32FC1 || vertexType == CV_32FC3);
210 vertices = _vertices.getMat();
211 // transform 3xN matrix to Nx3, except 3x3
212 if ((_vertices.channels() == 1) && (_vertices.rows() == 3) && (_vertices.cols() != 3))
213 {
214 vertices = vertices.t();
215 }
216 // This transposition is performed on 1xN matrix so it's almost free in terms of performance
217 vertices = vertices.reshape(3, 1).t();
218 nVerts = (int)vertices.total();
219
220 int indexType = _indices.type();
221 CV_Assert(indexType == CV_32SC1 || indexType == CV_32SC3);
222 triangles = _indices.getMat();
223 // transform 3xN matrix to Nx3, except 3x3
224 if ((_indices.channels() == 1) && (_indices.rows() == 3) && (_indices.cols() != 3))
225 {
226 triangles = triangles.t();
227 }
228 // This transposition is performed on 1xN matrix so it's almost free in terms of performance
229 triangles = triangles.reshape(3, 1).t();
230 nTriangles = (int)triangles.total();
231
232 if (!_colors.empty())
233 {
234 int colorType = _colors.type();
235 CV_Assert(colorType == CV_32FC1 || colorType == CV_32FC3);
236 colors = _colors.getMat();
237 // transform 3xN matrix to Nx3, except 3x3
238 if ((_colors.channels() == 1) && (_colors.rows() == 3) && (_colors.cols() != 3))
239 {
240 colors = colors.t();
241 }
242 colors = colors.reshape(3, 1).t();
243 nColors = (int)colors.total();
244
245 CV_Assert(nColors == nVerts);
246 }
247
248 // any of buffers can be empty
249 Size imgSize {std::max(colorBuf.cols, depthBuf.cols), std::max(colorBuf.rows, depthBuf.rows)};
250
251 // world-to-camera coord system
252 Matx44d lookAtMatrix = camPoseMat;
253
254 double ys = 1.0 / std::tan(fovyRadians / 2);
255 double xs = ys / (double)imgSize.width * (double)imgSize.height;
256 double zz = (zNear + zFar) / (zNear - zFar);
257 double zw = 2.0 * zFar * zNear / (zNear - zFar);
258
259 // camera to NDC: [-1, 1]^3
260 Matx44d perspectMatrix (xs, 0, 0, 0,
261 0, ys, 0, 0,
262 0, 0, zz, zw,
263 0, 0, -1, 0);
264
265 Matx44f mvpMatrix = perspectMatrix * lookAtMatrix;
266
267 // vertex transform stage
268
269 Mat screenVertices(vertices.size(), CV_32FC4);
270 for (int i = 0; i < nVerts; i++)
271 {
272 Vec3f vglobal = vertices.at<Vec3f>(i);
273
274 float x_num = std::fma(mvpMatrix(0,0), vglobal[0],
275 std::fma(mvpMatrix(0,1), vglobal[1],
276 std::fma(mvpMatrix(0,2), vglobal[2], mvpMatrix(0,3))));
277 float y_num = std::fma(mvpMatrix(1,0), vglobal[0],
278 std::fma(mvpMatrix(1,1), vglobal[1],
279 std::fma(mvpMatrix(1,2), vglobal[2], mvpMatrix(1,3))));
280 float z_num = std::fma(mvpMatrix(2,0), vglobal[0],
281 std::fma(mvpMatrix(2,1), vglobal[1],
282 std::fma(mvpMatrix(2,2), vglobal[2], mvpMatrix(2,3))));
283 float w_num = std::fma(mvpMatrix(3,0), vglobal[0],
284 std::fma(mvpMatrix(3,1), vglobal[1],
285 std::fma(mvpMatrix(3,2), vglobal[2], mvpMatrix(3,3))));
286
287 float invw = 1.f / w_num;
288
289 // [-1, 1]^3 => [0, width] x [0, height] x [0, 1]
290 Vec4f vscreen = {
291 std::fma(x_num * invw, 0.5f * (float)imgSize.width, 0.5f * (float)imgSize.width),
292 std::fma(y_num * invw, 0.5f * (float)imgSize.height, 0.5f * (float)imgSize.height),
293 std::fma(z_num * invw, 0.5f, 0.5f),
294 invw
295 };
296
297 screenVertices.at<Vec4f>(i) = vscreen;
298 }
299
300 // draw stage
301
302 for (int t = 0; t < nTriangles; t++)
303 {
304 Vec3i tri = triangles.at<Vec3i>(t);
305
306 Vec3f col[3];
307 Vec4f ver[3];
308 for (int i = 0; i < 3; i++)
309 {
310 int idx = tri[i];
311 CV_DbgAssert(idx >= 0 && idx < nVerts);
312
313 col[i] = colors.empty() ? Vec3f::all(0) : colors.at<Vec3f>(idx);
314 ver[i] = screenVertices.at<Vec4f>(idx);
315 }
316
317 drawTriangle(ver, col, depthBuf, colorBuf, settings);
318 }
319}

322void triangleRasterizeDepth(InputArray _vertices, InputArray _indices, InputOutputArray _depthBuf,
323 InputArray world2cam, double fovY, double zNear, double zFar,
324 const TriangleRasterizeSettings& settings)
325{
326 CV_Assert(!_depthBuf.empty());
327 CV_Assert(_depthBuf.type() == CV_32FC1);
328
329 Mat emptyColorBuf;
330 // out-of-range values from user-provided depthBuf should not be altered, let's mark them
331 Mat_<uchar> validMask;
332 Mat depthBuf;
333 if (settings.glCompatibleMode == RASTERIZE_COMPAT_INVDEPTH)
334 {
335 depthBuf = _depthBuf.getMat();
336 }
337 else // RASTERIZE_COMPAT_DISABLED
338 {
339 invertDepth(_depthBuf.getMat(), depthBuf, validMask, zNear, zFar);
340 }
341
342 triangleRasterizeInternal(_vertices, _indices, noArray(), emptyColorBuf, depthBuf, world2cam, fovY, zNear, zFar, settings);
343
344 if (settings.glCompatibleMode == RASTERIZE_COMPAT_DISABLED)
345 {
346 linearizeDepth(depthBuf, validMask, _depthBuf.getMat(), zFar, zNear);
347 }
348}
