17/* Inserts layout conversion operations (if needed) into the model graph and subgraphs.
18
19 Some of the operations
20 (let's call them 'operations of category B' or B-operations, 'B' stands for 'Block'),
21 most notably Convolution (including depthwise convolution), ConvTranspose, MaxPool and AveragePool,
22 can be computed more efficiently if data is represented in so-called block layout,
23 i.e. when 4D tensor NxCxHxW is represented in memory as 5D tensor NxC1xHxWxC0,
24 where all C channels of the original tensor are split into C1 groups of C0 channels each.
25 For each spatial location (y=y0, x=x0) each group of C0 channels is stored sequentially.
26 C1 is thus computed as following: C1 = (C + C0-1)/C0, where division is performed with truncation.
27
28 Some other operations (let's call them A-operations, 'A' stands for 'Any'),
29 most notably unary element-wise operations or special cases of binary
30 element-wise operations can be efficiently computed in any layout, including block layout.
31
32 Finally, all other operations (denoted as C-operations, 'C' for 'Casual' or 'Channels')
33 do not support block layout at all. The inputs should come in the original model format
34 (e.g. NCHW in the case of Onnx).
35
36 We want to transform the graph so that:
37 1. B-operations always take the inputs in block layout. If not, the inputs must be converted to
38 block layout prior to B-operation. Note that only the first input should be converted
39 in the case of Convolution, convolution weights (if constant) are pre-processed separately.
40 2. C-operations always take the inputs in non-block layout, e.g. NCHW. If some of the inputs are
41 stored in block layout, they must be converted from block layout prior to C-operation.
42 3. the number of layout transformation operations is minimal,
43 i.e. we don't do transformations unless it's necessary.
44
45 Note that this graph transformation is applied after fusion, since inside a fused operation
46 (e.g. 'Convolution + Batch Norm + Activation + Adding Skip connection') we don't need to
47 transform layout. That is, we have to deal with less operations at this stage.
48*/
49
50struct BlockLayoutTransformer
51{
52 BlockLayoutTransformer(Net::Impl* netimpl_) : netimpl(netimpl_) {}

54 Net::Impl* netimpl;
55 vector<DataLayout> layouts; // layouts for each argument
56 vector<Arg> blockCache; // if an Arg needs to be converted to block layout and
57 // if it's used by several operations,
58 // then we reuse once transformed arg,
59 // don't transform it several times
60 vector<Arg> nonblockCache; // another cache of non-block args
61 DataLayout defaultLayout;
62 int trlayoutIdx;

64 std::pair<Arg, PLayer> getProperArg(const Arg& arg, bool block, int defaultC0)
65 {
66 if (arg.empty() || (layouts[arg.idx] == DATA_LAYOUT_BLOCK) == block)
67 return {arg, PLayer()};
68 std::vector<Arg> *cache, *altCache;
69 if (block) {
70 cache = &blockCache;
71 altCache = &nonblockCache;
72 } else {
73 cache = &nonblockCache;
74 altCache = &blockCache;
75 }
76 Arg cached = cache->at(arg.idx);
77 if (!cached.empty())
78 return {cached, PLayer()};
79 size_t nargs = netimpl->args.size();
80 CV_Assert(layouts.size() == nargs &&
81 cache->size() == nargs &&
82 altCache->size() == nargs);
83
84 const ArgData& adata = netimpl->args.at(arg.idx);
85 const char* suffix = block ? "block" : "nonblock";
86
87 std::string newname = format("%s.%s", adata.name.c_str(), suffix);
88 int idx = 1;
89 for ( ; netimpl->haveArg(newname); idx++) {
90 newname = format("%s.%s%d", adata.name.c_str(), suffix, idx);
91 }
92
93 cached = netimpl->newArg(newname, DNN_ARG_TEMP);
94 CV_Assert(size_t(cached.idx) == nargs);
95
96 cache->at(arg.idx) = cached;
97 cache->push_back(cached);
98 altCache->push_back(arg);
99
100 LayerParams params;
101 params.name = format("trlayout.%d", trlayoutIdx++);
102 params.type = "TransformLayout";
103 params.set("layout", int(block ? DATA_LAYOUT_BLOCK : defaultLayout));
104 params.set("C0", int(defaultC0));
105 PLayer trlayer = TransformLayoutLayer::create(params);
106
107 trlayer->netimpl = netimpl;
108 trlayer->inputs = {arg};
109 trlayer->outputs = {cached};
110 layouts.push_back(block ? DATA_LAYOUT_BLOCK : defaultLayout);
111
112 return {cached, trlayer};
113 }

115 void transformGraph(PGraph& g)
116 {
117 const vector<PLayer>& currProg = g->prog();
118 int defaultC0 = netimpl->defaultC0;
119 vector<PLayer> newProg;
120 std::vector<Arg> newInputs;
121 std::vector<DataLayout> inputLayoutsOrig, inputLayoutsNew, outputLayouts;
122 size_t nchanges = 0;
123
124 for (size_t opidx = 0; opidx < currProg.size(); opidx++) {
125 const PLayer& layer = currProg[opidx];
126 const vector<Arg>& inputs = layer->inputs;
127 const vector<Arg>& outputs = layer->outputs;
128 size_t ninputs = inputs.size(), noutputs = outputs.size();
129 std::string op_name = layer->type;
130 std::string name = layer->name;
131 vector<PGraph>* subgraphs = layer->subgraphs();
132 bool deviceOp = g->opBackend((int)opidx) != DNN_BACKEND_OPENCV;
133 //std::cout << "name: " << name << ", op_name: " << op_name << ", inp0 layout: " << layoutToString(layouts[inputs[0].idx]) << "\n";
134
135 if (subgraphs) {
136 for (PGraph& subgraph: *subgraphs) {
137 size_t nargsBefore = netimpl->args.size();
138 CV_Assert_N(nargsBefore == blockCache.size(), nargsBefore == nonblockCache.size());
139 transformGraph(subgraph);
140 size_t nargsAfter = netimpl->args.size();
141 CV_Assert_N(nargsAfter == blockCache.size(), nargsAfter == nonblockCache.size());
142 // after transforming a subgraph we 'forget' all cached transformations,
143 // because they need to be redone in the outer graph and/or other subgraphs.
144 // however, we cannot remove the newly added arguments, because we need to properly
145 // assign memory buffers for them etc.
146 for (size_t i = nargsBefore; i < nargsAfter; i++) {
147 Arg b = blockCache[i];
148 Arg nb = nonblockCache[i];
149 if (size_t(b.idx) < nargsBefore) {
150 nonblockCache[b.idx] = Arg();
151 }
152 if (size_t(nb.idx) < nargsBefore) {
153 blockCache[nb.idx] = Arg();
154 }
155 }
156 nchanges += nargsAfter > nargsBefore;
157 }
158 }
159
160 inputLayoutsOrig.clear();
161 for (size_t i = 0; i < ninputs; i++) {
162 inputLayoutsOrig.push_back(layouts.at(inputs[i].idx));
163 }
164
165 layer->getLayouts(inputLayoutsOrig, inputLayoutsNew, int(noutputs), outputLayouts);
166 CV_Assert(inputLayoutsNew.size() == ninputs);
167 CV_Assert(outputLayouts.size() == noutputs);
168
169 if (deviceOp) {
170 for (size_t i = 0; i < ninputs; i++)
171 inputLayoutsNew[i] = defaultLayout;
172 for (size_t i = 0; i < noutputs; i++)
173 outputLayouts[i] = defaultLayout;
174 }
175
176 newInputs.clear();
177 bool changedInputs = false;
178 for (size_t i = 0; i < ninputs; i++) {
179 Arg inp = inputs[i];
180 DataLayout prevLayout = inputLayoutsOrig[i];
181 DataLayout newLayout = inputLayoutsNew[i];
182 if (!inp.empty()) {
183 if (prevLayout == DATA_LAYOUT_BLOCK) {
184 blockCache.at(inp.idx) = inp;
185 } else {
186 nonblockCache.at(inp.idx) = inp;
187 }
188 }
189 if (newLayout != prevLayout &&
190 (newLayout == DATA_LAYOUT_BLOCK ||
191 prevLayout == DATA_LAYOUT_BLOCK)) {
192 auto p = getProperArg(inp, newLayout == DATA_LAYOUT_BLOCK, defaultC0);
193 newInputs.push_back(p.first);
194 if (p.second) {
195 newProg.push_back(p.second);
196 }
197 changedInputs = true;
198 nchanges++;
199 } else {
200 newInputs.push_back(inp);
201 }
202 }
203
204 for (size_t i = 0; i < noutputs; i++) {
205 layouts.at(outputs[i].idx) = outputLayouts[i];
206 }
207
208 if (changedInputs) {
209 layer->inputs = newInputs;
210 }
211 newProg.push_back(layer);
212 }
213
214 if (nchanges > 0) {
215 g->setProg(newProg);
216 }
217 }

219 void transform()
220 {
221 size_t nargs = netimpl->args.size();
222 defaultLayout = netimpl->originalLayout;
223 trlayoutIdx = 0;
224
225 layouts.assign(nargs, DATA_LAYOUT_UNKNOWN);
226 blockCache.assign(nargs, Arg());
227 nonblockCache.assign(nargs, Arg());
228
229 transformGraph(netimpl->mainGraph);
230 }
231};
