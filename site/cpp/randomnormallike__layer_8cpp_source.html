26 void fillRandomNormal(OutputArray out, float mean, float scale,
27 bool has_seed, float seed)
28 {
29 CV_Assert(out.isMat() || out.isUMat());
30 const Scalar mean_s = Scalar::all(mean);
31 const Scalar scale_s = Scalar::all(scale);
32
33 RNG local_rng;
34 if (has_seed)
35 {
36 uint64 seed_u64 = (uint64)std::llround((double)seed);
37 if (!seed_u64)
38 seed_u64 = 0x12345678ULL;
39 local_rng = RNG(seed_u64);
40 }
41
42 RNG& rng = has_seed ? local_rng : theRNG();
43
44 if (out.isMat())
45 {
46 Mat& m = out.getMatRef();
47 rng.fill(m, RNG::NORMAL, mean_s, scale_s);
48 }
49 else
50 {
51 UMat& u = out.getUMatRef();
52 rng.fill(u, RNG::NORMAL, mean_s, scale_s);
53 }
54 }

110 void forward(InputArrayOfArrays inputs_arr,
111 OutputArrayOfArrays outputs_arr,
112 OutputArrayOfArrays) CV_OVERRIDE
113 {
114 CV_TRACE_FUNCTION();
115 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
116
117 Size size = inputs_arr.size();
118 int ninputs = size.area();
119 CV_Assert(ninputs == 1);
120
121 Mat inp = inputs_arr.getMat(0);
122 MatShape outShape = inp.shape();
123
124 int outType = outputType >= 0 ? outputType : inp.type();
125
126 auto kind = outputs_arr.kind();
127 if (kind == _InputArray::STD_VECTOR_MAT) {
128 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
129 outs.resize(1);
130 outs[0].fit(outShape, outType);
131 fillRandomNormal(outs[0], mean, scale, has_seed, seed);
132 } else if (kind == _InputArray::STD_VECTOR_UMAT) {
133 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
134 outs.resize(1);
135 outs[0].fit(outShape, outType);
136 fillRandomNormal(outs[0], mean, scale, has_seed, seed);
137 } else {
138 CV_Error(Error::StsNotImplemented, "");
139 }
140 }
