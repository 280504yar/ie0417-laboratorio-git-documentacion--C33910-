938template<typename T>
939struct CV_EXPORTS Accumulator
940{
941 typedef T Type;
942};
943
944template<> struct Accumulator<unsigned char> { typedef float Type; };
945template<> struct Accumulator<unsigned short> { typedef float Type; };
946template<> struct Accumulator<char> { typedef float Type; };
947template<> struct Accumulator<short> { typedef float Type; };
948
949/*
950 * Squared Euclidean distance functor
951 */
952template<class T>
953struct CV_EXPORTS SL2
954{
955 static const NormTypes normType = NORM_L2SQR;
956 typedef T ValueType;
957 typedef typename Accumulator<T>::Type ResultType;
958
959 ResultType operator()( const T* a, const T* b, int size ) const
960 {
961 return normL2Sqr<ValueType, ResultType>(a, b, size);
962 }
963};
964
965/*
966 * Euclidean distance functor
967 */
968template<class T>
969struct L2
970{
971 static const NormTypes normType = NORM_L2;
972 typedef T ValueType;
973 typedef typename Accumulator<T>::Type ResultType;
974
975 ResultType operator()( const T* a, const T* b, int size ) const
976 {
977 return (ResultType)std::sqrt((double)normL2Sqr<ValueType, ResultType>(a, b, size));
978 }
979};
980
981/*
982 * Manhattan distance (city block distance) functor
983 */
984template<class T>
985struct L1
986{
987 static const NormTypes normType = NORM_L1;
988 typedef T ValueType;
989 typedef typename Accumulator<T>::Type ResultType;
990
991 ResultType operator()( const T* a, const T* b, int size ) const
992 {
993 return normL1<ValueType, ResultType>(a, b, size);
994 }
995};
