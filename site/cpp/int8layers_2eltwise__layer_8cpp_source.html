47 EltwiseLayerInt8Impl(const LayerParams& params)
48 : outputChannels(0)
49 {
50 setParamsFrom(params);
51 offset = params.get<float>("offset", 0.f);
52 hasVecInput = false;
53 op = SUM;
54 if (params.has("operation"))
55 {
56 String operation = toLowerCase(params.get<String>("operation"));
57 if (operation == "prod")
58 op = PROD;
59 else if (operation == "sum")
60 op = SUM;
61 else if (operation == "max")
62 op = MAX;
63 else
64 CV_Error(cv::Error::StsBadArg, "Unknown operation type \"" + operation + "\"");
65 }
66
67 if (params.has("coeff"))
68 {
69 DictValue paramCoeff = params.get("coeff");
70 int i, n = paramCoeff.size();
71 coeffs.resize(n);
72 for (i = 0; i < n; i++)
73 {
74 coeffs[i] = paramCoeff.get<float>(i);
75 }
76 }
77
78 if (params.has("input_zeropoints"))
79 {
80 DictValue zp = params.get("input_zeropoints");
81 int i, n = zp.size();
82 zeropoints.resize(n);
83 for (i = 0; i < n; i++)
84 {
85 zeropoints[i] = zp.get<int>(i);
86 }
87 }
88
89 if (params.has("input_scales"))
90 {
91 DictValue sc = params.get("input_scales");
92 int i, n = sc.size();
93 scales.resize(n);
94 for (i = 0; i < n; i++)
95 {
96 scales[i] = sc.get<float>(i);
97 }
98 }
99
100 output_zp = params.get<int>("zeropoints", 0);
101 output_sc = params.get<float>("scales", 1.0f);
102
103 channelsModeInput = ELTWISE_CHANNNELS_SAME;
104 if (params.has("output_channels_mode"))
105 {
106 String v = toLowerCase(params.get<String>("output_channels_mode"));
107 if (v == "same")
108 {
109 channelsModeInput = ELTWISE_CHANNNELS_SAME;
110 }
111 else if (v == "input_0")
112 {
113 channelsModeInput = ELTWISE_CHANNNELS_INPUT_0;
114 }
115 else if (v == "input_0_truncate")
116 {
117 channelsModeInput = ELTWISE_CHANNNELS_INPUT_0_TRUNCATE;
118 }
119 else if (v == "max_input_channels")
120 {
121 channelsModeInput = ELTWISE_CHANNNELS_USE_MAX;
122 if (op != SUM)
123 CV_Error(cv::Error::StsBadArg, "[" + type + "]:(" + name + ") 'max' channels mode is limited to SUM operation only");
124 }
125 else
126 CV_Error(cv::Error::StsBadArg, "[" + type + "]:(" + name + ") unknown channels mode: \"" + v + "\"");
127 }
128 channelsMode = channelsModeInput;
129
130 // TODO Must have checks for other unknown options

141 void ensureQuantizationParams()
142 {
143 if (!coeffs.empty())
144 return;
145
146 if (scales.empty())
147 return;
148
149 CV_CheckEQ(scales.size(), zeropoints.size(), "EltwiseInt8: scales and zeropoints sizes must match");
150 CV_Assert(output_sc > 0.0f);
151
152 if (op == SUM)
153 {
154 coeffs.resize(scales.size());
155 offset = (float)output_zp;
156 for (size_t i = 0; i < scales.size(); i++)
157 {
158 coeffs[i] = scales[i] / output_sc;
159 offset -= coeffs[i] * zeropoints[i];
160 }
161 }
162 else if (op == PROD)
163 {
164 coeffs.resize(scales.size());
165 coeffs[0] = scales[0] / output_sc;
166 for (size_t i = 1; i < scales.size(); i++)
167 coeffs[i] = scales[i];
168 offset = (float)output_zp;
169 }
170 else if (op == MAX)
171 {
172 for (size_t i = 0; i < scales.size(); i++)
173 {
174 const bool sameQuant =
175 std::abs(scales[i] - output_sc) < 1e-6f &&
176 zeropoints[i] == output_zp;
177 if (!sameQuant)
178 {
179 CV_Error(Error::StsBadArg,
180 "EltwiseInt8 'max' requires identical quantization "
181 "(same scale and zero-point) for all inputs and output.");
182 }
183 }
184 }
185 else
186 {
187 CV_Error(Error::StsNotImplemented, "Unsupported eltwise operation");
188 }
189 }

191 bool getMemoryShapes(const std::vector<MatShape> &inputs,
192 const int requiredOutputs,
193 std::vector<MatShape> &outputs,
194 std::vector<MatShape> &internals) const CV_OVERRIDE
195 {
196 CV_Assert(inputs.size() >= 2);
197 CV_Assert(inputs[0].size() >= 2);
198 CV_Assert(coeffs.size() == 0 || coeffs.size() == inputs.size());
199 CV_Assert(op == SUM || op == PROD || coeffs.size() == 0);
200
201 int dims = inputs[0].size();
202 // Number of channels in output shape is determined by the first input tensor.
203 bool variableChannels = false;
204 int numChannels = inputs[0][1];
205 for (size_t i = 1; i < inputs.size(); i++)
206 {
207 CV_Assert(inputs[0][0] == inputs[i][0]); // batch sizes are equal
208
209 int input_channels = inputs[i][1];
210 if (numChannels != input_channels)
211 variableChannels = true;
212
213 if (channelsModeInput == ELTWISE_CHANNNELS_SAME)
214 {
215 CV_Assert(numChannels == input_channels);
216 }
217 else if (channelsModeInput == ELTWISE_CHANNNELS_INPUT_0)
218 {
219 CV_Assert(numChannels >= input_channels);
220 }
221 else if (channelsModeInput == ELTWISE_CHANNNELS_INPUT_0_TRUNCATE)
222 {
223 // nothing to check
224 }
225 else if (channelsModeInput == ELTWISE_CHANNNELS_USE_MAX)
226 {
227 numChannels = std::max(numChannels, input_channels);
228 }
229 else
230 {
231 CV_Assert(0 && "Internal error");
232 }
233 }
234
235 channelsMode = variableChannels ? channelsModeInput : ELTWISE_CHANNNELS_SAME;
236 outputChannels = numChannels;
237
238 outputs.assign(1, inputs[0]);
239 outputs[0][1] = numChannels;
240
241 if (dims > 2)
242 {
243 size_t vecIdx = 0;
244 bool isVecFound = false;
245 for (size_t i = 0; i < inputs.size(); i++)
246 {
247 bool allOnes = isAllOnes(inputs[i], 2, dims);
248 if (!allOnes && !isVecFound)
249 {
250 vecIdx = i;
251 isVecFound = true;
252 }
253
254 if (!allOnes && i != vecIdx)
255 {
256 for (size_t j = 2; j < dims; j++)
257 {
258 CV_Assert(inputs[vecIdx][j] == inputs[i][j]);
259 }
260 }
261 }
262
263 if (channelsModeInput == ELTWISE_CHANNNELS_SAME && isVecFound)
264 {
265 for (size_t j = 2; j < dims; j++)
266 {
267 outputs[0][j] = inputs[vecIdx][j];
268 }
269 }
270 }
271
272 return false;
273 }

291 virtual Ptr<BackendNode> initTimVX(void* timVXInfo_,
292 const std::vector<Ptr<BackendWrapper> > &inputsWrapper,
293 const std::vector<Ptr<BackendWrapper> > &outputsWrapper,
294 bool isLast) CV_OVERRIDE
295 {
296#ifdef HAVE_TIMVX
297 ensureQuantizationParams();
298 // tvGraph Initialization.
299 if (inputsWrapper.size() != 2)
300 return Ptr<BackendNode>();
301
302 auto timVxInfo = reinterpret_cast<TimVXInfo *>(timVXInfo_);
303 CV_Assert(timVxInfo);
304 Ptr<TimVXGraph> tvGraph = timVxInfo->getGraph();
305 CV_Assert(tvGraph);
306 Ptr<tim::vx::Graph> graph = tvGraph->graph;
307
308 bool isSub = false;
309 // TODO: support variable coeffs.
310 if (op == SUM)
311 {
312 CV_Assert(coeffs.size() == scales.size());
313 std::vector<float> originalCoeffs;
314
315 for (int i = 0; i < coeffs.size(); i++)
316 {
317 originalCoeffs.push_back(coeffs[i] * output_sc / scales[i]);
318 }
319
320 float eps = std::numeric_limits<float>::epsilon();
321 if (std::fabs(originalCoeffs[0] - 1.0f) <= eps * std::fabs(originalCoeffs[0] + 1.0f) &&
322 std::fabs(originalCoeffs[1] + 1.0f) <= eps * std::fabs(originalCoeffs[1] - 1.0f))
323 {
324 // Sub, if coeffs = {1., -1.}, isSub = true.
325 isSub = true;
326 }
327 else if (std::fabs(originalCoeffs[0] - 1.0f) <= eps * std::fabs(originalCoeffs[0] + 1.0f) &&
328 std::abs(originalCoeffs[1] - 1.0f) <= eps * std::abs(originalCoeffs[1] + 1.0f))
329 {
330 // Sum, if coeff = {1., 1.}, isSub = false.
331 isSub = false;
332 }
333 else
334 {
335 return Ptr<BackendNode>();
336 }
337 }
338
339 std::vector<int> inputsIndex, outputsIndex;
340 int input_index = -1, output_index = -1;
341 CV_Assert(channelsModeInput == ELTWISE_CHANNNELS_SAME);
342
343 // Input
344 Ptr<TimVXBackendWrapper> inputWrapper;
345
346 CV_Assert(!scales.empty() && !zeropoints.empty());
347
348 for (int i = 0; i<inputsWrapper.size(); i++){
349 inputWrapper = inputsWrapper[i].dynamicCast<TimVXBackendWrapper>();
350
351 if (inputWrapper->isTensor())
352 {
353 input_index = tvGraph->getTensorIndex(inputWrapper->getTensor());
354 if (input_index == -1)
355 {
356 // Copy To New inputWrapper
357 Mat tmp = inputWrapper->getMat();
358 inputWrapper = Ptr<TimVXBackendWrapper>(new TimVXBackendWrapper(tmp));
359 }
360 }
361
362 if (!inputWrapper->isTensor())
363 {
364 Ptr<tim::vx::Quantization> tvInputQuant = Ptr<tim::vx::Quantization>(
365 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, scales[i], zeropoints[i]));
366 inputWrapper->createTensor(graph,tim::vx::TensorAttribute::INPUT, tvInputQuant);
367 input_index = tvGraph->addWrapper(inputWrapper);
368 }
369
370 inputsIndex.push_back(input_index);
371 }
372
373 // Output
374 CV_Assert(outputsWrapper.size() == 1);
375 Ptr<TimVXBackendWrapper> outputWrapper = outputsWrapper[0].dynamicCast<TimVXBackendWrapper>();
376 Ptr<tim::vx::Quantization> outputQuant = Ptr<tim::vx::Quantization>(
377 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, output_sc, output_zp));
378
379 if (isLast)
380 {
381 auto shapeType = getShapeTypeFromMat(outputWrapper->getMat());
382
383 // For Graph Output tensor, we need to set tensor shape before createTensor().
384 outputWrapper->setTensorShape(shapeType);
385 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::OUTPUT, outputQuant);
386 }
387 else
388 {
389 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::TRANSIENT, outputQuant);
390 }
391 output_index = tvGraph->addWrapper(outputWrapper);
392 outputsIndex.push_back(output_index);
393
394 std::shared_ptr<tim::vx::Operation> tvEltwise;
395
396 switch (op) {
397 case SUM:
398 if (isSub)
399 tvEltwise = graph->CreateOperation<tim::vx::ops::Sub>();
400 else
401 tvEltwise = graph->CreateOperation<tim::vx::ops::Add>();
402 break;
403 case PROD:
404 tvEltwise = graph->CreateOperation<tim::vx::ops::Multiply>();
405 break;
406 case MAX:
407 tvEltwise = graph->CreateOperation<tim::vx::ops::Maximum>();
408 break;
409 default:
410 CV_Error(Error::StsNotImplemented, "Unsupported eltwise operation");
411 }
412
413 Ptr<TimVXBackendNode> tvBackendNode = new TimVXBackendNode(tvGraph, tvEltwise, inputsIndex, outputsIndex);
414
415 return tvBackendNode;
416#endif // HAVE_TIMVX
417 return Ptr<BackendNode>();
418 }

420#ifdef HAVE_DNN_NGRAPH
421 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> > &inputs,
422 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
423 {
424 ensureQuantizationParams();
425 CV_Assert(nodes.size() >= 2);
426 std::vector<ov::Output<ov::Node>> ieInpNodes(nodes.size());
427 for (size_t i = 0; i < nodes.size(); i++)
428 {
429 ieInpNodes[i] = nodes[i].dynamicCast<InfEngineNgraphNode>()->node;
430
431 float input_sc = !coeffs.empty() ? coeffs[i] : 1.0f;
432 float input_zp = op == PROD ? zeropoints[i] : 0.0f;
433 ieInpNodes[i] = ngraphDequantize(ieInpNodes[i], input_sc, input_zp);
434 }
435
436 auto res = ieInpNodes[0];
437 for (size_t i = 1; i < ieInpNodes.size(); i++)
438 {
439 switch (op) {
440 case SUM: res = std::make_shared<ov::op::v1::Add>(res, ieInpNodes[i]); break;
441 case PROD: res = std::make_shared<ov::op::v1::Multiply>(res, ieInpNodes[i]); break;
442 case MAX: res = std::make_shared<ov::op::v1::Maximum>(res, ieInpNodes[i]); break;
443 default: CV_Error(Error::StsNotImplemented, "Unsupported eltwise operation");
444 }
445 }
446
447 res = ngraphQuantize(res, 1.0f, offset);
448
449 return new InfEngineNgraphNode(res);
450 }

453 class EltwiseInvoker : public ParallelLoopBody
454 {
455 EltwiseLayerInt8Impl& self;
456 std::vector<const Mat*> srcs;
457 std::vector<int> srcNumChannels;
458 int nsrcs;
459 Mat* dst;
460 Mat* buf;
461 std::vector<float> coeffs;
462 std::vector<int> zeropoints;
463 int nstripes;
464 const Mat* activLUT;
465 const ActivationLayerInt8* activ;
466 int channels;
467 size_t planeSize;
468 float offset;
469
470 EltwiseInvoker(EltwiseLayerInt8Impl& self_)
471 : self(self_)
472 , nsrcs(0), dst(0), buf(0), nstripes(0), activLUT(0), activ(0), channels(0)
473 , planeSize(0), offset(0)
474 {}
475
476 public:
477 static void run(EltwiseLayerInt8Impl& self,
478 const Mat* srcs, int nsrcs, Mat& buf, Mat& dst,
479 int nstripes, float offset)
480 {
481 const EltwiseOp op = self.op;
482 CV_Check(dst.dims, 1 < dst.dims && dst.dims <= 5, ""); CV_CheckTypeEQ(dst.type(), CV_8SC1, ""); CV_Assert(dst.isContinuous());
483 CV_Assert(self.coeffs.empty() || self.coeffs.size() == (size_t)nsrcs);
484 CV_CheckGE(nsrcs, 2, "");
485
486 CV_Assert(self.outputChannels == dst.size[1]);
487
488 EltwiseInvoker p(self);
489 p.srcs.resize(nsrcs);
490 p.srcNumChannels.resize(nsrcs);
491 p.coeffs = self.coeffs; // can be sorted
492 p.zeropoints = self.zeropoints;
493
494 bool sortInputs = false;
495 for( int i = 0; i < nsrcs; i++ )
496 {
497 p.srcs[i] = &srcs[i];
498 CV_CheckEQ(srcs[i].dims, dst.dims, "");
499 CV_Assert(srcs[i].isContinuous());
500 CV_Assert(srcs[i].type() == dst.type());
501 p.srcNumChannels[i] = (srcs[i].dims >= 4) ? srcs[i].size[1] : 1;
502
503 if (self.channelsMode == ELTWISE_CHANNNELS_SAME)
504 {
505 CV_Assert(srcs[i].size == dst.size);
506 }
507 else if (self.channelsMode == ELTWISE_CHANNNELS_INPUT_0)
508 {
509 if (i == 0)
510 CV_Assert(srcs[0].size == dst.size);
511 CV_Assert(self.outputChannels >= p.srcNumChannels[i]);
512 sortInputs = true;
513 }
514 else if (self.channelsMode == ELTWISE_CHANNNELS_INPUT_0_TRUNCATE)
515 {
516 if (i == 0)
517 CV_Assert(srcs[0].size == dst.size);
518 sortInputs = true;
519 }
520 else if (self.channelsMode == ELTWISE_CHANNNELS_USE_MAX)
521 {
522 CV_Assert(op == SUM);
523 CV_Assert(self.outputChannels >= p.srcNumChannels[i]);
524 sortInputs = true;
525 }
526 else
527 {
528 CV_Assert(0 && "Internal error");
529 }
530
531 if (sortInputs)
532 {
533 // Sort srcs and coefficients in the desc order by number of channels
534 for (int j = i; j >= 1; j--)
535 {
536 if (std::min(self.outputChannels, p.srcs[j - 1]->size[1]) < std::min(self.outputChannels, p.srcs[j]->size[1]))
537 {
538 std::swap(p.srcs[j - 1], p.srcs[j]);
539 std::swap(p.srcNumChannels[j - 1], p.srcNumChannels[j]);
540 if (!p.coeffs.empty())
541 std::swap(p.coeffs[j - 1], p.coeffs[j]);
542 if (!p.zeropoints.empty())
543 std::swap(p.zeropoints[j - 1], p.zeropoints[j]);
544 }
545 else
546 break;
547 }
548 }
549 }
550
551 p.nsrcs = nsrcs;
552 p.dst = &dst;
553 p.buf = &buf;
554 p.nstripes = nstripes;
555 p.offset = offset;
556 p.channels = (dst.dims >= 4 ? dst.size[1] : 1);
557
558 p.planeSize = dst.total(dst.dims >= 4 ? 2 : 1);
559 CV_CheckEQ(dst.total(), dst.size[0] * p.channels * p.planeSize, "");
560 p.activLUT = &self.activationLUT;
561 p.activ = !self.activationLUT.empty() ? self.activ.get() : 0;
562
563 parallel_for_(Range(0, nstripes), p, nstripes);
564 }
565
566 void operator()(const Range& r) const CV_OVERRIDE
567 {
568 const EltwiseOp op = self.op;
569 size_t total = dst->size[0]*planeSize;
570 size_t stripeSize = (total + nstripes - 1)/nstripes;
571 size_t stripeStart = r.start*stripeSize;
572 size_t stripeEnd = std::min(r.end*stripeSize, total);
573 const float* coeffsptr = !coeffs.empty() ? &coeffs[0] : 0;
574 const int* zeropointsptr = !zeropoints.empty() ? &zeropoints[0] : 0;
575 const int8_t* lutptr = !activLUT->empty() ? activLUT->ptr<int8_t>() : 0;
576 int8_t* dstptr0 = dst->ptr<int8_t>();
577 float* bufptr0 = buf->ptr<float>();
578 int blockSize0 = 1 << 12;
579 CV_Assert(op != PROD || zeropointsptr);
580 CV_Assert((op != PROD && op != SUM) || coeffsptr);
581 for (size_t ofs = stripeStart; ofs < stripeEnd; )
582 {
583 int sampleIdx = (int)(ofs / planeSize);
584 int delta = (int)ofs - sampleIdx * planeSize;
585 int blockSize = std::min(blockSize0, std::min((int)(stripeEnd - ofs), (int)planeSize - delta));
586 if( blockSize <= 0 )
587 break;
588 ofs += blockSize;
589
590 for (int c = 0; c < channels; c++)
591 {
592 size_t dstIdx = delta + (sampleIdx*channels + c)*planeSize;
593 int8_t* dstptr = dstptr0 + dstIdx;
594 float* bufptr = bufptr0 + dstIdx;
595 bool directOutput = false;
596
597 // process first two inputs
598 {
599 const int8_t* srcptr0 = srcs[0]->ptr<int8_t>() + dstIdx;
600
601 const int inputIdx = 1;
602 int src1_channels = srcNumChannels[inputIdx];
603 if (c >= src1_channels)
604 {
605 // no data from second input
606 if (!coeffsptr)
607 {
608 for (int j = 0; j < blockSize; j++)
609 {
610 dstptr[j] = srcptr0[j];
611 }
612 }
613 else
614 {
615 float c0 = coeffsptr[0];
616 int z0 = op == PROD ? zeropointsptr[0] : 0;
617 for (int j = 0; j < blockSize; j++)
618 {
619 bufptr[j] = c0 * (srcptr0[j] - z0);
620 }
621 }
622 }
623 else
624 {
625 size_t srcIdx = delta + (sampleIdx * src1_channels + c) * planeSize;
626 const int8_t* srcptrI = srcs[inputIdx]->ptr<int8_t>() + srcIdx;
627
628 if (op == PROD)
629 {
630 float c0 = coeffsptr[0];
631 float c1 = coeffsptr[1];
632 int z0 = zeropointsptr[0];
633 int z1 = zeropointsptr[1];
634 for (int j = 0; j < blockSize; j++)
635 {
636 bufptr[j] = (c0*(srcptr0[j] - z0)) * (c1*(srcptrI[j] - z1));
637 }
638 }
639 else if (op == MAX)
640 {
641 int j = 0;
642 #if CV_SIMD128
643 for (; j <= blockSize - 16; j += 16)
644 {
645 v_int8x16 va = v_load(srcptr0 + j);
646 v_int8x16 vb = v_load(srcptrI + j);
647 v_store(dstptr + j, v_max(va, vb));
648 }
649 #endif
650 for (; j < blockSize; j++)
651 {
652 dstptr[j] = std::max(srcptr0[j], srcptrI[j]);
653 }
654 }
655 else if (op == SUM)
656 {
657 float c0 = coeffsptr[0];
658 float c1 = coeffsptr[1];
659 if (nsrcs == 2)
660 {
661 directOutput = true;
662 int j = 0;
663 #if CV_SIMD128
664 {
665 v_float32x4 vc0 = v_setall_f32(c0);
666 v_float32x4 vc1 = v_setall_f32(c1);
667 v_float32x4 voffs = v_setall_f32(offset);
668 for (; j <= blockSize - 16; j += 16)
669 {
670 v_int8x16 va = v_load(srcptr0 + j);
671 v_int8x16 vb = v_load(srcptrI + j);
672
673 v_int16x8 va_lo, va_hi, vb_lo, vb_hi;
674 v_expand(va, va_lo, va_hi);
675 v_expand(vb, vb_lo, vb_hi);
676
677 v_int32x4 va0, va1, va2, va3, vb0, vb1, vb2, vb3;

679 v_expand(va_hi, va2, va3);
680 v_expand(vb_lo, vb0, vb1);
681 v_expand(vb_hi, vb2, vb3);

684 v_mul(v_cvt_f32(vb0), vc1)), voffs);

686 v_mul(v_cvt_f32(vb1), vc1)), voffs);

688 v_mul(v_cvt_f32(vb2), vc1)), voffs);

690 v_mul(v_cvt_f32(vb3), vc1)), voffs);

694 }
695 }
696 #endif
697 for (; j < blockSize; j++)
698 {

700 }

704 int j = 0;
705 #if CV_SIMD128

728 v_store(bufptr + j + 8, v_add(v_mul(v_cvt_f32(va2), vc0),

731 v_mul(v_cvt_f32(vb3), vc1)));
732 }
733 }
734 #endif

751 continue; // no data from second input

753 const int8_t* srcptrI = srcs[inputIdx]->ptr<int8_t>() + srcIdx;
754

756 {
757 float cI = coeffsptr[inputIdx];

762 }
763 }
764 else if (op == MAX)
765 {
766 int j = 0;
767 #if CV_SIMD128
768 for (; j <= blockSize - 16; j += 16)
769 {
770 v_store(dstptr + j, v_max(v_load(dstptr + j), v_load(srcptrI + j)));
771 }
772 #endif

781 int j = 0;
782 #if CV_SIMD128
783 {

785 for (; j <= blockSize - 16; j += 16)
786 {

801 v_mul(v_cvt_f32(vi3), vcI)));
802 }
803 }
804 #endif
805 for (; j < blockSize; j++)
806 {
807 bufptr[j] += cI * srcptrI[j];
808 }
809 }
810 else
811 CV_Error(Error::StsInternal, "");
812 }
813
814 // add offset and saturate cast to int8
815 if ((op == SUM || op == PROD) && !directOutput)
816 {
817 int j = 0;
818 #if CV_SIMD128
819 {
820 v_float32x4 voffs = v_setall_f32(offset);
821 for (; j <= blockSize - 16; j += 16)
822 {
823 v_float32x4 r0 = v_add(v_load(bufptr + j), voffs);
824 v_float32x4 r1 = v_add(v_load(bufptr + j + 4), voffs);
825 v_float32x4 r2 = v_add(v_load(bufptr + j + 8), voffs);
826 v_float32x4 r3 = v_add(v_load(bufptr + j + 12), voffs);
827 v_store(dstptr + j, v_pack(v_pack(v_round(r0), v_round(r1)),
828 v_pack(v_round(r2), v_round(r3))));
829 }
830 }
831 #endif
832 for (; j < blockSize; j++)
833 {
834 dstptr[j] = saturate_cast<int8_t>(std::round(bufptr[j] + offset));
835 }
836 }
837 }
838 if( activ )
839 {
840 int8_t* ptr = dstptr0 + delta + sampleIdx*channels*planeSize;
841 activ->forwardSlice(ptr, lutptr, ptr, blockSize, planeSize, 0, channels);
842 }
843 }
844 }

847 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
848 {
849 CV_TRACE_FUNCTION();
850 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
851 ensureQuantizationParams();
852
853 std::vector<Mat> inputs, outputs;
854 inputs_arr.getMatVector(inputs);
855 outputs_arr.getMatVector(outputs);
856
857 CV_Assert(outputs.size() == 1);
858 const int nstripes = getNumThreads();
859
860 if (channelsModeInput == ELTWISE_CHANNNELS_SAME && inputs[0].dims > 2)
861 {
862 for (size_t i = 0; i < inputs.size(); i++)
863 {
864 MatShape inpShape = inputs[i].shape();
865 bool allOnes = isAllOnes(inpShape, 2, inputs[i].dims);
866
867 if (allOnes)
868 {
869 Mat tmpInput = inputs[i];
870 MatShape outShape = outputs[0].shape();
871 size_t xSize = outShape[2];
872 for (int j = 3; j < outShape.dims; j++)
873 xSize *= outShape[j];
874
875 int dimVec[3] = {outShape[0], outShape[1], (int) xSize};
876 std::vector<int> matSizesVec(&dimVec[0], &dimVec[0] + 3);
877 inputs[i] = Mat(matSizesVec, tmpInput.type());
878
879 std::vector<int> idx(outShape.size(), 0);
880 std::vector<int> outIdx(inpShape.size(), 0);
881
882 for (size_t j = 0; j < outShape[0]; j++)
883 {
884 outIdx[0] = idx[0] = j;
885 for(size_t k = 0; k < outShape[1]; k++)
886 {
887 outIdx[1] = idx[1] = k;
888 for (size_t x = 0; x < xSize; x++)
889 {
890 outIdx[2] = x;
891 inputs[i].at<int8_t>(outIdx.data()) = tmpInput.at<int8_t>(idx.data());
892 }
893 }
894 }
895 inputs[i] = inputs[i].reshape(0, outShape);
896 }
897 }
898 }
899
900 Mat buf = Mat(shape(outputs[0]), CV_32F); // to store intermediate results
901 EltwiseInvoker::run(*this, &inputs[0], (int)inputs.size(), buf, outputs[0], nstripes, offset);
902 }
