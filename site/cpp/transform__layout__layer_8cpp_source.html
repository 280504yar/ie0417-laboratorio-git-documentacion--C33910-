41template<typename _Tp>
42static inline void transpose8x8(const _Tp* inp_, size_t istep,
43 _Tp* out_, size_t ostep)
44{
45#if defined(__AVX2__)
46 if (sizeof(_Tp) == 4u) {
47 // 8x8 32-bit transpose via 256-bit AVX2: 8 unpack + 8 shuffle + 8 perm.
48 // Roughly 2x faster than the four v_transpose4x4 path on AVX2 hosts.
49 const float* inp = (const float*)inp_;
50 float* out = (float*)out_;
51 __m256 r0 = _mm256_loadu_ps(inp + istep*0);
52 __m256 r1 = _mm256_loadu_ps(inp + istep*1);
53 __m256 r2 = _mm256_loadu_ps(inp + istep*2);
54 __m256 r3 = _mm256_loadu_ps(inp + istep*3);
55 __m256 r4 = _mm256_loadu_ps(inp + istep*4);
56 __m256 r5 = _mm256_loadu_ps(inp + istep*5);
57 __m256 r6 = _mm256_loadu_ps(inp + istep*6);
58 __m256 r7 = _mm256_loadu_ps(inp + istep*7);
59
60 __m256 t0 = _mm256_unpacklo_ps(r0, r1);
61 __m256 t1 = _mm256_unpackhi_ps(r0, r1);
62 __m256 t2 = _mm256_unpacklo_ps(r2, r3);
63 __m256 t3 = _mm256_unpackhi_ps(r2, r3);
64 __m256 t4 = _mm256_unpacklo_ps(r4, r5);
65 __m256 t5 = _mm256_unpackhi_ps(r4, r5);
66 __m256 t6 = _mm256_unpacklo_ps(r6, r7);
67 __m256 t7 = _mm256_unpackhi_ps(r6, r7);
68
69 __m256 v0 = _mm256_shuffle_ps(t0, t2, _MM_SHUFFLE(1,0,1,0));
70 __m256 v1 = _mm256_shuffle_ps(t0, t2, _MM_SHUFFLE(3,2,3,2));
71 __m256 v2 = _mm256_shuffle_ps(t1, t3, _MM_SHUFFLE(1,0,1,0));
72 __m256 v3 = _mm256_shuffle_ps(t1, t3, _MM_SHUFFLE(3,2,3,2));
73 __m256 v4 = _mm256_shuffle_ps(t4, t6, _MM_SHUFFLE(1,0,1,0));
74 __m256 v5 = _mm256_shuffle_ps(t4, t6, _MM_SHUFFLE(3,2,3,2));
75 __m256 v6 = _mm256_shuffle_ps(t5, t7, _MM_SHUFFLE(1,0,1,0));
76 __m256 v7 = _mm256_shuffle_ps(t5, t7, _MM_SHUFFLE(3,2,3,2));
77
78 // 0x20 -> {lo of A, lo of B}; 0x31 -> {hi of A, hi of B}
79 _mm256_storeu_ps(out + ostep*0, _mm256_permute2f128_ps(v0, v4, 0x20));
80 _mm256_storeu_ps(out + ostep*1, _mm256_permute2f128_ps(v1, v5, 0x20));
81 _mm256_storeu_ps(out + ostep*2, _mm256_permute2f128_ps(v2, v6, 0x20));
82 _mm256_storeu_ps(out + ostep*3, _mm256_permute2f128_ps(v3, v7, 0x20));
83 _mm256_storeu_ps(out + ostep*4, _mm256_permute2f128_ps(v0, v4, 0x31));
84 _mm256_storeu_ps(out + ostep*5, _mm256_permute2f128_ps(v1, v5, 0x31));
85 _mm256_storeu_ps(out + ostep*6, _mm256_permute2f128_ps(v2, v6, 0x31));
86 _mm256_storeu_ps(out + ostep*7, _mm256_permute2f128_ps(v3, v7, 0x31));
87 return;
88 }
89#endif
90#if CV_SIMD128
91 if (sizeof(_Tp) == 4u) {
92 const uint32_t* inp = (const uint32_t*)inp_;
93 uint32_t* out = (uint32_t*)out_;
94 v_uint32x4 a0, a1, a2, a3, b0, b1, b2, b3;
95
96 a0 = v_load(inp + istep*0);
97 a1 = v_load(inp + istep*1);
98 a2 = v_load(inp + istep*2);
99 a3 = v_load(inp + istep*3);
100 v_transpose4x4(a0, a1, a2, a3, b0, b1, b2, b3);
101 v_store(out + ostep*0, b0);
102 v_store(out + ostep*1, b1);
103 v_store(out + ostep*2, b2);
104 v_store(out + ostep*3, b3);
105
106 a0 = v_load(inp + istep*0 + 4);
107 a1 = v_load(inp + istep*1 + 4);
108 a2 = v_load(inp + istep*2 + 4);
109 a3 = v_load(inp + istep*3 + 4);
110 v_transpose4x4(a0, a1, a2, a3, b0, b1, b2, b3);
111 v_store(out + ostep*4, b0);
112 v_store(out + ostep*5, b1);
113 v_store(out + ostep*6, b2);
114 v_store(out + ostep*7, b3);
115
116 a0 = v_load(inp + istep*4);
117 a1 = v_load(inp + istep*5);
118 a2 = v_load(inp + istep*6);
119 a3 = v_load(inp + istep*7);
120 v_transpose4x4(a0, a1, a2, a3, b0, b1, b2, b3);
121 v_store(out + ostep*0 + 4, b0);
122 v_store(out + ostep*1 + 4, b1);
123 v_store(out + ostep*2 + 4, b2);
124 v_store(out + ostep*3 + 4, b3);
125
126 a0 = v_load(inp + istep*4 + 4);
127 a1 = v_load(inp + istep*5 + 4);
128 a2 = v_load(inp + istep*6 + 4);
129 a3 = v_load(inp + istep*7 + 4);
130 v_transpose4x4(a0, a1, a2, a3, b0, b1, b2, b3);
131 v_store(out + ostep*4 + 4, b0);
132 v_store(out + ostep*5 + 4, b1);
133 v_store(out + ostep*6 + 4, b2);
134 v_store(out + ostep*7 + 4, b3);
135 } else
136#endif
137 {
138 for (int i = 0; i < 8; i++)
139 for (int j = 0; j < 8; j++)
140 out_[i * ostep + j] = inp_[j * istep + i];
141 }
142}

144template <typename _Tp>
145void transformLayoutInterleave_(const _Tp* inp_base, _Tp* out_base, int C, size_t len,
146 int nc, int nzc, size_t dlen)
147{
148 size_t i = 0;
149 for (; i + 7u < dlen; i += 8u)
150 {
151 int c = 0;
152
153 for (; c + 7u < nzc; c += 8u) {
154 transpose8x8<_Tp>(inp_base + c * len + i, len, out_base + i * nc + c, nc);
155 }
156
157 for (; c < nzc; ++c) {
158 _Tp* outptr = out_base + i * nc + c;
159 const _Tp* inptr = inp_base + c * len + i;
160 outptr[0 * nc] = inptr[0];
161 outptr[1 * nc] = inptr[1];
162 outptr[2 * nc] = inptr[2];
163 outptr[3 * nc] = inptr[3];
164 outptr[4 * nc] = inptr[4];
165 outptr[5 * nc] = inptr[5];
166 outptr[6 * nc] = inptr[6];
167 outptr[7 * nc] = inptr[7];
168 }
169
170 for (; c < nc; ++c) {
171 _Tp* outptr = out_base + i * nc + c;
172 outptr[0 * nc] = (_Tp)0; outptr[1 * nc] = (_Tp)0; outptr[2 * nc] = (_Tp)0; outptr[3 * nc] = (_Tp)0;
173 outptr[4 * nc] = (_Tp)0; outptr[5 * nc] = (_Tp)0; outptr[6 * nc] = (_Tp)0; outptr[7 * nc] = (_Tp)0;
174 }
175 }
176 for (; i < dlen; ++i) {
177 _Tp* outptr = out_base + i * nc;
178 for (int c = 0; c < nc; ++c) {
179 outptr[c] = c < nzc ? inp_base[c*len + i] : (_Tp)0;
180 }
181 }
182}

184template <typename _Tp>
185void transformLayoutDeinterleave_(const _Tp* inp_base, _Tp* out_base, int C, size_t len,
186 int nc, int nzc, size_t dlen)
187{
188 size_t i = 0;
189 for (; i + 7u < dlen; i += 8u)
190 {
191 int c = 0;
192
193 for (; c + 7u < nzc; c += 8u)
194 {
195 transpose8x8<_Tp>(inp_base + i * nc + c, nc, out_base + c * len + i, len);
196 }
197
198 for (; c < nzc; ++c)
199 {
200 const _Tp* inptr = inp_base + i * nc + c;
201 _Tp* outptr = out_base + c * len + i;
202 outptr[0] = inptr[0 * nc];
203 outptr[1] = inptr[1 * nc];
204 outptr[2] = inptr[2 * nc];
205 outptr[3] = inptr[3 * nc];
206 outptr[4] = inptr[4 * nc];
207 outptr[5] = inptr[5 * nc];
208 outptr[6] = inptr[6 * nc];
209 outptr[7] = inptr[7 * nc];
210 }
211 }
212 for (; i < dlen; ++i)
213 {
214 const _Tp* inptr = inp_base + i * nc;
215 for (int c = 0; c < nzc; ++c) {
216 out_base[c*len + i] = inptr[c];
217 }
218 }

242TransformLayoutFunc getTransformLayoutFunc(DataLayout inplayout, DataLayout outlayout, size_t esz)
243{
244 if (inplayout == DATA_LAYOUT_NCHW &&
245 (outlayout == DATA_LAYOUT_BLOCK || outlayout == DATA_LAYOUT_NHWC)) {
246 return esz == 1u ? transformLayoutInterleave_8u :
247 esz == 2u ? transformLayoutInterleave_16u :
248 esz == 4u ? transformLayoutInterleave_32u :
249 esz == 8u ? transformLayoutInterleave_64u : nullptr;
250 }
251 if ((inplayout == DATA_LAYOUT_BLOCK || inplayout == DATA_LAYOUT_NHWC) &&
252 outlayout == DATA_LAYOUT_NCHW) {
253 return esz == 1u ? transformLayoutDeinterleave_8u :
254 esz == 2u ? transformLayoutDeinterleave_16u :
255 esz == 4u ? transformLayoutDeinterleave_32u :
256 esz == 8u ? transformLayoutDeinterleave_64u : nullptr;
257 }
258 return nullptr;
259}

261void transformLayout(const Mat& inp, Mat& out,
262 DataLayout outlayout,
263 DataLayout defaultLayout,
264 int C0)
265{
266 CV_Assert(defaultLayout == DATA_LAYOUT_NCHW || defaultLayout == DATA_LAYOUT_NHWC);
267 CV_Assert(outlayout == DATA_LAYOUT_BLOCK || outlayout == DATA_LAYOUT_NCHW || outlayout == DATA_LAYOUT_NHWC);
268
269 MatShape inpshape = inp.size;
270 /*if (inpshape.layout == DATA_LAYOUT_NCHW &&
271 inpshape.dims == 4 && inpshape[1] == 272 && inpshape[2] == 14 && inpshape[3] == 14) {
272 putchar('.');
273 }*/
274
275 if (inpshape.layout == DATA_LAYOUT_UNKNOWN) {
276 inpshape.layout = defaultLayout;
277 }
278 DataLayout inplayout = inpshape.layout;
279 MatShape outshape = inferTransformLayoutShape(inpshape, outlayout, defaultLayout, C0);
280 out.fit(outshape, inp.type());
281
282 if (inp.empty())
283 return;
284
285 if (inplayout == outlayout) {
286 inp.copyTo(out);
287 return;
288 }
289
290 CV_Assert_N(inp.isContinuous(), out.isContinuous());
291
292 size_t esz = inp.elemSize();
293 TransformLayoutFunc kernel = getTransformLayoutFunc(inplayout, outlayout, esz);
294 CV_Assert(kernel != nullptr);
295
296 int N = inpshape[0];
297 int C = inpshape.channels();
298 C0 = inplayout == DATA_LAYOUT_BLOCK ? inpshape.back() : C0;
299 int C1 = (C + C0 - 1) / C0;
300
301 size_t planesize = 1;
302 int inp_sp0 = inplayout == DATA_LAYOUT_NHWC ? 1 : 2;
303 int inp_sp1 = inplayout == DATA_LAYOUT_NCHW ? inpshape.dims : inpshape.dims-1;
304 for (int i = inp_sp0; i < inp_sp1; i++) {
305 planesize *= (size_t)inpshape[i];
306 }
307
308 size_t total = N*C1*planesize*C0;
309 // Tuned to keep small encoder/decoder transforms (e.g. 256ch * 14*14 ~ 50K elems)
310 // from running single-threaded. 16K elems ~ 64 KB ~ L1-resident chunk.
311 constexpr size_t min_elems_per_chunk = 1 << 14;
312 int nblocks = int((total + min_elems_per_chunk/2) / min_elems_per_chunk);
313 int nthreads = std::max(1, getNumThreads());
314 nblocks = clamp(nblocks, 1, std::max(N*C1, 1) * 16);
315 nblocks = (nblocks + N*C1 - 1)/(N*C1);
316 nblocks = std::min(nblocks, std::max(1, nthreads));
317
318 int total_chunks = N * C1 * nblocks;
319 double nstripes = std::min((double)total_chunks, (double)nthreads);
320 parallel_for_(Range(0, total_chunks), [&](const Range& range)
321 {
322 int dchunk = 1u;
323 bool interleave = inplayout == DATA_LAYOUT_NCHW;
324 const uint8_t* inptr = (const uint8_t*)inp.data;
325 uint8_t* outptr = (uint8_t*)out.data;
326
327 for (int chunk = range.start; chunk < range.end; chunk += dchunk)
328 {
329 int n = chunk/(C1*nblocks);
330 int c1 = (chunk % (C1*nblocks))/nblocks;
331 int block = chunk % nblocks;
332 int nc = C0;
333 int nzc = std::min(nc, C - c1*C0);
334 dchunk = std::min(nblocks - block, range.end - chunk);
335 size_t block_start = block * planesize / nblocks;
336 size_t block_end = (block + dchunk) * planesize / nblocks;
337 size_t dlen = block_end - block_start;
338 size_t inpofs = ((n * C1 + c1) * planesize + block_start) * nc * esz;
339 size_t outofs = ((n * C + c1 * C0) * planesize + block_start) * esz;
340 if (interleave) {
341 std::swap(inpofs, outofs);
342 }
343
344 kernel(inptr + inpofs, outptr + outofs, C, planesize, nc, nzc, dlen);
345 }
346 }, nstripes);
347}
