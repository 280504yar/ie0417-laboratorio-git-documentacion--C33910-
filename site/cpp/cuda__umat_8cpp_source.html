22class CudaUMatAllocator CV_FINAL : public MatAllocator
23{
24public:
25 UMatData* allocate(int dims, const int* sizes, int type, void* data0,
26 size_t* step, AccessFlag /*flags*/, UMatUsageFlags /*usageFlags*/) const CV_OVERRIDE
27 {
28 const size_t elemSize = CV_ELEM_SIZE(type);
29 size_t total = elemSize;
30 for (int i = 0; i < dims; i++)
31 total *= (size_t)sizes[i];
32
33 if (step)
34 {
35 step[dims - 1] = elemSize;
36 for (int i = dims - 2; i >= 0; i--)
37 step[i] = step[i + 1] * (size_t)sizes[i + 1];
38 }
39
40 UMatData* u = new UMatData(this);
41 u->size = total;
42
43 if (total > 0)
44 cudaSafeCall(cudaMalloc(&u->handle, total));
45
46 if (data0)
47 {
48 u->data = u->origdata = static_cast<uchar*>(data0);
49 u->flags |= UMatData::USER_ALLOCATED;
50 u->markDeviceCopyObsolete(true);
51 }
52 else
53 {
54 u->markHostCopyObsolete(true);
55 }
56 return u;
57 }
58
59 bool allocate(UMatData* u, AccessFlag /*accessFlags*/, UMatUsageFlags /*usageFlags*/) const CV_OVERRIDE
60 {
61 if (!u)
62 return false;
63 if (!u->handle && u->size > 0)
64 {
65 cudaSafeCall(cudaMalloc(&u->handle, u->size));
66 if (u->data)
67 u->markDeviceCopyObsolete(true);
68 else
69 u->markHostCopyObsolete(true);
70 }
71 return true;
72 }
73
74 void deallocate(UMatData* u) const CV_OVERRIDE
75 {
76 if (!u)
77 return;
78 if (u->handle)
79 cudaSafeCall(cudaFree(u->handle));
80 if (u->data && !(u->flags & UMatData::USER_ALLOCATED))
81 fastFree(u->data);
82 delete u;
83 }
84
85 void map(UMatData* u, AccessFlag accessFlags) const CV_OVERRIDE
86 {
87 if (!u)
88 return;
89 if (!u->data && u->size > 0)
90 u->data = u->origdata = static_cast<uchar*>(fastMalloc(u->size));
91 if (u->hostCopyObsolete() && u->handle && u->data)
92 {
93 cudaSafeCall(cudaMemcpy(u->data, u->handle, u->size, cudaMemcpyDeviceToHost));
94 u->markHostCopyObsolete(false);
95 }
96 if (!!(accessFlags & ACCESS_WRITE))
97 u->markDeviceCopyObsolete(true);
98 }
99
100 void unmap(UMatData* u) const CV_OVERRIDE
101 {
102 if (!u)
103 return;
104 if (u->deviceCopyObsolete() && u->handle && u->data)
105 {
106 cudaSafeCall(cudaMemcpy(u->handle, u->data, u->size, cudaMemcpyHostToDevice));
107 u->markDeviceCopyObsolete(false);
108 }
109 }
110
111 void download(UMatData* u, void* dstptr, int dims, const size_t sz[],
112 const size_t srcofs[], const size_t srcstep[],
113 const size_t dststep[]) const CV_OVERRIDE
114 {
115 if (!u || !u->handle || !dstptr)
116 return;
117 copyPlanes((uchar*)u->handle, srcofs, srcstep, (uchar*)dstptr, 0, dststep,
118 dims, sz, cudaMemcpyDeviceToHost);
119 }
120
121 void upload(UMatData* u, const void* srcptr, int dims, const size_t sz[],
122 const size_t dstofs[], const size_t dststep[],
123 const size_t srcstep[]) const CV_OVERRIDE
124 {
125 if (!u || !srcptr)
126 return;
127 if (!u->handle && u->size > 0)
128 cudaSafeCall(cudaMalloc(&u->handle, u->size));
129 copyPlanes((uchar*)srcptr, NULL, srcstep, (uchar*)u->handle, dstofs, dststep,
130 dims, sz, cudaMemcpyHostToDevice);
131 u->markHostCopyObsolete(true);
132 u->markDeviceCopyObsolete(false);
133 }

135 // Same-world D2D only; UMat::copyTo calls copy() solely when both share this allocator.
136 void copy(UMatData* src, UMatData* dst, int dims, const size_t sz[],
137 const size_t srcofs[], const size_t srcstep[],
138 const size_t dstofs[], const size_t dststep[], bool /*sync*/) const CV_OVERRIDE
139 {
140 if (!src || !dst || !src->handle || !dst->handle)
141 return;
142 copyPlanes((uchar*)src->handle, srcofs, srcstep, (uchar*)dst->handle, dstofs, dststep,
143 dims, sz, cudaMemcpyDeviceToDevice);
144 dst->markHostCopyObsolete(true);
145 dst->markDeviceCopyObsolete(false);
146 }
147
148private:
149 // sz/steps follow the MatAllocator convention: the last dim carries element bytes.
150 static void copyPlanes(uchar* srcbase, const size_t srcofs[], const size_t srcstep[],
151 uchar* dstbase, const size_t dstofs[], const size_t dststep[],
152 int dims, const size_t sz[], cudaMemcpyKind kind)
153 {
154 int isz[CV_MAX_DIM];
155 uchar* srcptr = srcbase;
156 uchar* dstptr = dstbase;
157 for (int i = 0; i < dims; i++)
158 {
159 CV_Assert(sz[i] <= (size_t)INT_MAX);
160 if (sz[i] == 0)
161 return;
162 if (srcofs)
163 srcptr += srcofs[i] * (i <= dims - 2 ? srcstep[i] : 1);
164 if (dstofs)
165 dstptr += dstofs[i] * (i <= dims - 2 ? dststep[i] : 1);
166 isz[i] = (int)sz[i];
167 }
168
169 Mat src(dims, isz, CV_8U, srcptr, srcstep);
170 Mat dst(dims, isz, CV_8U, dstptr, dststep);
171
172 const Mat* arrays[] = { &src, &dst };
173 uchar* ptrs[2];
174 NAryMatIterator it(arrays, ptrs, 2);
175 const size_t planesz = it.size;
176 for (size_t j = 0; j < it.nplanes; j++, ++it)
177 cudaSafeCall(cudaMemcpy(ptrs[1], ptrs[0], planesz, kind));
178 }
179};
