44 BatchNormLayerImpl(const LayerParams& params)
45 : dims(-1)
46 {
47 setParamsFrom(params);
48 CV_Assert(blobs.size() >= 2);
49
50 hasWeights = params.get<bool>("has_weight", false);
51 hasBias = params.get<bool>("has_bias", false);
52 useGlobalStats = params.get<bool>("use_global_stats", true);
53 if(params.get<bool>("scale_bias", false))
54 hasWeights = hasBias = true;
55 epsilon = params.get<float>("eps", 1E-5);
56
57 // std::cout << params.get<float>("momentum", 0.9) << std::endl;
58 momentum = params.get<float>("momentum", 0.9);
59
60 size_t n = blobs[0].total();
61 CV_Assert(blobs[1].total() == n &&
62 blobs[0].isContinuous() && blobs[1].isContinuous() &&
63 blobs[0].type() == CV_32F && blobs[1].type() == CV_32F);
64
65 float varMeanScale = 1.f;
66 if (!hasWeights && !hasBias && blobs.size() > 2 && useGlobalStats) {
67 CV_Assert(blobs.size() == 3); CV_CheckTypeEQ(blobs[2].type(), CV_32FC1, "");
68 varMeanScale = blobs[2].at<float>(0);
69 if (varMeanScale != 0)
70 varMeanScale = 1/varMeanScale;
71 }
72
73 const int biasBlobIndex = blobs.size() - 1;
74 const int weightsBlobIndex = biasBlobIndex - hasBias;
75
76 if( hasWeights )
77 {
78 CV_Assert((size_t)weightsBlobIndex < blobs.size());
79 const Mat& w = blobs[weightsBlobIndex];
80 CV_Assert(w.isContinuous() && w.type() == CV_32F && w.total() == (size_t)n);
81 }
82
83 if( hasBias )
84 {
85 CV_Assert((size_t)biasBlobIndex < blobs.size());
86 const Mat& b = blobs[biasBlobIndex];
87 CV_Assert(b.isContinuous() && b.type() == CV_32F && b.total() == (size_t)n);
88 }
89
90 const float* meanData = blobs[0].ptr<float>();
91 const float* stdData = blobs[1].ptr<float>();
92 const float* weightsData = hasWeights ? blobs[weightsBlobIndex].ptr<float>() : 0;
93 const float* biasData = hasBias ? blobs[biasBlobIndex].ptr<float>() : 0;
94
95 // 1-D [n], NOT 1xn: the fused scale/bias participate in element-wise ops against 0/1-D
96 // inputs, and under the broadcasting rules (4) op (1,4) yields (1,4) - a 2-D result that
97 // would not fit a preallocated 1-D output blob. 1-D weights keep every shape exact.
98 const int sz1d[] = { (int)n };
99 origin_weights.create(1, sz1d, CV_32F);
100 origin_bias.create(1, sz1d, CV_32F);
101
102 float* dstWeightsData = origin_weights.ptr<float>();
103 float* dstBiasData = origin_bias.ptr<float>();
104
105 for (size_t i = 0; i < n; ++i)
106 {
107 float w = (hasWeights ? weightsData[i] : 1.0f) / sqrt(stdData[i] * varMeanScale + epsilon);
108 dstWeightsData[i] = w;
109 dstBiasData[i] = (hasBias ? biasData[i] : 0.0f) - w * meanData[i] * varMeanScale;
110 }
111 }

113 virtual void finalize(InputArrayOfArrays, OutputArrayOfArrays) CV_OVERRIDE
114 {
115 origin_weights.copyTo(weights_);
116 origin_bias.copyTo(bias_);
117 }

122 shift = bias_;

125 virtual bool tryFuse(Ptr<Layer>& top) CV_OVERRIDE
126 {
127 Mat w, b;
128 top->getScaleShift(w, b);
129 if (w.empty() && b.empty())
130 return false;
131
132 const int numChannels = weights_.total();
133 const int numFusedWeights = w.total();
134 const int numFusedBias = b.total();
135
136 if ((numFusedWeights != numChannels && numFusedWeights != 1 && !w.empty()) ||
137 (numFusedBias != numChannels && numFusedBias != 1 && !b.empty()))
138 return false;
139
140 // reshape the fused factors to 1-D [numChannels], matching weights_/bias_ - a 1xn operand
141 // would broadcast the result up to 2-D and detach weights_ from its expected 1-D shape
142 const int fsz[] = { numChannels };
143 if (!w.empty())
144 {
145 if (numFusedWeights == 1)
146 {
147 multiply(weights_, w.at<float>(0), weights_);
148 multiply(bias_, w.at<float>(0), bias_);
149 }
150 else
151 {
152 w = w.reshape(1, 1, fsz);
153 multiply(weights_, w, weights_);
154 multiply(bias_, w, bias_);
155 }
156 }
157 if (!b.empty())
158 {
159 if (numFusedBias == 1)
160 add(bias_, b.at<float>(0), bias_);
161 else
162 add(bias_, b.reshape(1, 1, fsz), bias_);
163 }
164 return true;
165 }

196#ifdef HAVE_OPENCL
197 bool forward_ocl(InputArrayOfArrays inputs_, OutputArrayOfArrays outputs_, OutputArrayOfArrays internals_)
198 {
199 std::vector<UMat> inputs;
200 std::vector<UMat> outputs;
201
202 bool use_half = (inputs_.depth() == CV_16F);
203 inputs_.getUMatVector(inputs);
204 outputs_.getUMatVector(outputs);
205
206 CV_Assert(blobs.size() >= 2);
207 CV_Assert(inputs.size() == 1);
208
209 if (use_half && inputs[0].dims == 2)
210 return false;
211
212 if (umat_weight.empty())
213 {
214 weights_.copyTo(umat_weight);
215 bias_.copyTo(umat_bias);
216 }
217
218 UMat &inpBlob = inputs[0];
219 int groups = inpBlob.size[0];
220 int channels = inpBlob.size[1];
221 int planeSize = 1;
222 for (size_t i = 2; i < inpBlob.dims; i++) {
223 planeSize *= inpBlob.size[i];
224 }
225
226 String opts = (use_half) ? " -DDtype=half" : " -DDtype=float";
227 for (size_t ii = 0; ii < outputs.size(); ii++)
228 {
229 if (inpBlob.dims == 2)
230 {
231 UMat& src = inputs[ii];
232 UMat& dst = outputs[ii];
233 multiply(src, weights_, dst);
234 add(dst, bias_, dst);
235 }
236 else
237 {
238 MatShape s = shape(groups * channels, planeSize);
239 UMat src = inputs[ii].reshape(1, s.size(), &s[0]);
240 UMat dst = outputs[ii].reshape(1, s.size(), &s[0]);
241 int number = (s[1] % 8 == 0) ? 8 : ((s[1] % 4 == 0) ? 4 : 1);
242 String buildopt = format("-DNUM=%d", number) + opts;
243 String kname = format("batch_norm%d", number);
244 if (number == 1)
245 buildopt += format(" -Dconvert_T=convert_%s", use_half ? "half" : "float");
246 else
247 buildopt += format(" -Dconvert_T=convert_%s%d", use_half ? "half" : "float", number);
248 ocl::Kernel kernel(kname.c_str(), ocl::dnn::batchnorm_oclsrc, buildopt);
249 if (kernel.empty())
250 return false;
251 size_t global[] = { (size_t)s[0], (size_t)(s[1] / number) };
252 kernel.set(0, ocl::KernelArg::PtrReadOnly(src));
253 kernel.set(1, (int)s[0]);
254 kernel.set(2, (int)s[1]);
255 kernel.set(3, (int)channels);
256 kernel.set(4, ocl::KernelArg::PtrReadOnly(umat_weight));
257 kernel.set(5, ocl::KernelArg::PtrReadOnly(umat_bias));
258 kernel.set(6, ocl::KernelArg::PtrWriteOnly(dst));
259 bool ret = kernel.run_(2, global, NULL, false);
260 if (!ret)
261 return false;
262 }
263 }
264 return true;
265 }
266#endif

268 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
269 {
270 CV_TRACE_FUNCTION();
271 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
272
273 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
274 forward_ocl(inputs_arr, outputs_arr, internals_arr))
275
276 if (inputs_arr.depth() == CV_16F)
277 {
278 forward_fallback(inputs_arr, outputs_arr, internals_arr);
279 return;
280 }
281
282 std::vector<Mat> inputs, outputs;
283 inputs_arr.getMatVector(inputs);
284 outputs_arr.getMatVector(outputs);
285
286 if (inputs[0].dims <= 1) { // Handling for 0D and 1D
287 Mat &inpBlob = inputs[0];
288 Mat &outBlob = outputs[0];
289 CV_Assert(inpBlob.total() == weights_.total());
290 // run on 1-D [n] views (a 0-D blob views as [1]) so every operand shape matches exactly:
291 // the result lands in the preallocated output blob in place, no realloc/detach possible
292 const int n1[] = { (int)inpBlob.total() };
293 Mat inp1d = inpBlob.reshape(1, 1, n1);
294 Mat out1d = outBlob.reshape(1, 1, n1);
295 cv::multiply(inp1d, weights_, out1d);
296 cv::add(out1d, bias_, out1d);
297 return;
298 }
299
300 CV_Assert(blobs.size() >= 2);
301 CV_Assert(inputs.size() == 1);
302
303 Mat &inpBlob = inputs[0];
304 int planeSize = 1;
305 for (size_t i = 2; i < inpBlob.dims; i++) {
306 planeSize *= inpBlob.size[i];
307 }
308
309 for (size_t ii = 0; ii < outputs.size(); ii++)
310 {
311 Mat &outBlob = outputs[ii];
312 for(int num = 0; num < outBlob.size[0]; num++)
313 {
314 for (int n = 0; n < outBlob.size[1]; n++)
315 {
316 float w = weights_.at<float>(n);
317 float b = bias_.at<float>(n);
318 Mat inpBlobPlane(1, planeSize, CV_32F, inpBlob.ptr<float>(num, n));
319 Mat outBlobPlane(1, planeSize, CV_32F, outBlob.ptr<float>(num, n));
320 inpBlobPlane.convertTo(outBlobPlane, CV_32F, w, b);
321 }
322 }
323 }
324 }

326 void forwardSlice(const float* srcptr, float* dstptr, int len, size_t planeSize, int cn0, int cn1) const CV_OVERRIDE
327 {
328 for( int cn = cn0; cn < cn1; cn++, srcptr += planeSize, dstptr += planeSize )
329 {
330 int i = 0;
331 float w = weights_.at<float>(cn);
332 float b = bias_.at<float>(cn);
333#if CV_SIMD128
334 v_float32x4 wV = v_setall_f32(w), bV = v_setall_f32(b);
335 for( ; i <= len - 16; i += 16 )
336 {
337 v_float32x4 x0 = v_load(srcptr + i);
338 v_float32x4 x1 = v_load(srcptr + i + 4);
339 v_float32x4 x2 = v_load(srcptr + i + 8);
340 v_float32x4 x3 = v_load(srcptr + i + 12);
341 x0 = v_muladd(x0, wV, bV);
342 x1 = v_muladd(x1, wV, bV);
343 x2 = v_muladd(x2, wV, bV);
344 x3 = v_muladd(x3, wV, bV);
345 v_store(dstptr + i, x0);
346 v_store(dstptr + i + 4, x1);
347 v_store(dstptr + i + 8, x2);
348 v_store(dstptr + i + 12, x3);
349 }
350#endif
351 for( ; i < len; i++ )
352 dstptr[i] = w * srcptr[i] + b;
353 }
354 }

444#ifdef HAVE_WEBNN
445 virtual Ptr<BackendNode> initWebnn(const std::vector<Ptr<BackendWrapper> >& inputs, const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
446 {
447 Ptr<WebnnBackendNode> node = nodes[0].dynamicCast<WebnnBackendNode>();
448 auto& webnnInpOperand = node->operand;
449 auto& webnnGraphBuilder = node->net->builder;
450 std::vector<int32_t> weights_shape = webnn::getShape(weights_);
451 ml::Operand weights = webnn::BuildConstant(webnnGraphBuilder, weights_shape, weights_.data, weights_.total()*weights_.elemSize(), ml::OperandType::Float32);
452 std::vector<int32_t> shape(dims, 1);
453 shape[1] = weights_shape[1];
454 ml::Operand weights_reshaped = webnnGraphBuilder.Reshape(weights, shape.data(), shape.size());
455 ml::Operand mul_res = webnnGraphBuilder.Mul(webnnInpOperand, weights_reshaped);
456 std::vector<int32_t> bias_shape = webnn::getShape(bias_);
457 ml::Operand bias = webnn::BuildConstant(webnnGraphBuilder, bias_shape, bias_.data, bias_.total()*bias_.elemSize(), ml::OperandType::Float32);
458 shape[1] = bias_shape[1];
459 ml::Operand bias_reshaped = webnnGraphBuilder.Reshape(bias, shape.data(), shape.size());
460 ml::Operand add_res = webnnGraphBuilder.Add(mul_res, bias_reshaped);
461 return Ptr<BackendNode>(new WebnnBackendNode(add_res));
462 }
