49//The code is very fast and performs also shadow detection.

477// shadow detection performed per pixel
478// should work for rgb data, could be useful for gray scale and depth data as well
479// See: Prati,Mikic,Trivedi,Cucchiara,"Detecting Moving Shadows...",IEEE PAMI,2003.
480static inline bool
481detectShadowGMM(const float* data, int nchannels, int nmodes,
482 const GMM* gmm, const float* mean,
483 float Tb, float TB, float tau)
484{
485 float tWeight = 0;
486
487 // check all the components marked as background:
488 for( int mode = 0; mode < nmodes; mode++, mean += nchannels )
489 {
490 GMM g = gmm[mode];
491
492 float numerator = 0.0f;
493 float denominator = 0.0f;
494 for( int c = 0; c < nchannels; c++ )
495 {
496 numerator += data[c] * mean[c];
497 denominator += mean[c] * mean[c];
498 }
499
500 // no division by zero allowed
501 if( denominator == 0 )
502 return false;
503
504 // if tau < a < 1 then also check the color distortion
505 if( numerator <= denominator && numerator >= tau*denominator )
506 {
507 float a = numerator / denominator;
508 float dist2a = 0.0f;
509
510 for( int c = 0; c < nchannels; c++ )
511 {
512 float dD= a*mean[c] - data[c];
513 dist2a += dD*dD;
514 }
515
516 if (dist2a < Tb*g.variance*a*a)
517 return true;
518 };
519
520 tWeight += g.weight;
521 if( tWeight > TB )
522 return false;
523 };
524 return false;
525}

573 void operator()(const Range& range) const CV_OVERRIDE
574 {
575 int y0 = range.start, y1 = range.end;
576 int ncols = src->cols, nchannels = src->channels();
577 AutoBuffer<float> buf(src->cols*nchannels);
578 float alpha1 = 1.f - alphaT;
579 float dData[CV_CN_MAX];
580
581 for( int y = y0; y < y1; y++ )
582 {
583 const float* data = buf.data();
584 if( src->depth() != CV_32F )
585 src->row(y).convertTo(Mat(1, ncols, CV_32FC(nchannels), (void*)data), CV_32F);
586 else
587 data = src->ptr<float>(y);
588
589 float* mean = mean0 + ncols*nmixtures*nchannels*y;
590 GMM* gmm = gmm0 + ncols*nmixtures*y;
591 uchar* modesUsed = modesUsed0 + ncols*y;
592 uchar* mask = dst->ptr(y);
593
594 for( int x = 0; x < ncols; x++, data += nchannels, gmm += nmixtures, mean += nmixtures*nchannels )
595 {
596
597 // Check that foreground mask exists
598 if (!knownForegroundMask.empty())
599 {
600 // If input mask states pixel is foreground
601 if (knownForegroundMask.at<uchar>(y, x) > 0)
602 {
603 mask[x] = 255; // ensure output mask marks this pixel as FG
604 continue;
605 }
606 }
607
608 //calculate distances to the modes (+ sort)
609 //here we need to go in descending order!!!
610 bool background = false;//return value -> true - the pixel classified as background
611
612 //internal:
613 bool fitsPDF = false;//if it remains zero a new GMM mode will be added
614 int nmodes = modesUsed[x];//current number of modes in GMM
615 float totalWeight = 0.f;
616
617 float* mean_m = mean;
618
620 //go through all modes
621 for( int mode = 0; mode < nmodes; mode++, mean_m += nchannels )
622 {
623 float weight = alpha1*gmm[mode].weight + prune;//need only weight if fit is found
624 int swap_count = 0;
626 //fit not found yet
627 if( !fitsPDF )
628 {
629 //check if it belongs to some of the remaining modes
630 float var = gmm[mode].variance;
631
632 //calculate difference and distance
633 float dist2;
634
635 if( nchannels == 3 )
636 {
637 dData[0] = mean_m[0] - data[0];
638 dData[1] = mean_m[1] - data[1];
639 dData[2] = mean_m[2] - data[2];
640 dist2 = dData[0]*dData[0] + dData[1]*dData[1] + dData[2]*dData[2];
641 }
642 else
643 {
644 dist2 = 0.f;
645 for( int c = 0; c < nchannels; c++ )
646 {
647 dData[c] = mean_m[c] - data[c];
648 dist2 += dData[c]*dData[c];
649 }
650 }
651
652 //background? - Tb - usually larger than Tg
653 if( totalWeight < TB && dist2 < Tb*var )
654 background = true;
655
656 //check fit
657 if( dist2 < Tg*var )
658 {
660 //belongs to the mode
661 fitsPDF = true;
662
663 //update distribution
664
665 //update weight
666 weight += alphaT;
667 float k = alphaT/weight;
668
669 //update mean
670 for( int c = 0; c < nchannels; c++ )
671 mean_m[c] -= k*dData[c];
672
673 //update variance
674 float varnew = var + k*(dist2-var);
675 //limit the variance
676 varnew = MAX(varnew, varMin);
677 varnew = MIN(varnew, varMax);
678 gmm[mode].variance = varnew;
679
680 //sort
681 //all other weights are at the same place and
682 //only the matched (iModes) is higher -> just find the new place for it
683 for( int i = mode; i > 0; i-- )
684 {
685 //check one up
686 if( weight < gmm[i-1].weight )
687 break;
688
689 swap_count++;
690 //swap one up
691 std::swap(gmm[i], gmm[i-1]);
692 for( int c = 0; c < nchannels; c++ )
693 std::swap(mean[i*nchannels + c], mean[(i-1)*nchannels + c]);
694 }
695 //belongs to the mode - bFitsPDF becomes 1
697 }
698 }
699
700 //check prune
701 if( weight < -prune )
702 {
703 weight = 0.0;
704 nmodes--;
705 }
706
707 gmm[mode-swap_count].weight = weight;//update weight by the calculated value
708 totalWeight += weight;
709 }
710 //go through all modes
712
713 // Renormalize weights. In the special case that the pixel does
714 // not agree with any modes, set weights to zero (a new mode will be added below).
715 float invWeight = 0.f;
716 if (std::abs(totalWeight) > FLT_EPSILON) {
717 invWeight = 1.f/totalWeight;
718 }
719
720 for( int mode = 0; mode < nmodes; mode++ )
721 {
722 gmm[mode].weight *= invWeight;
723 }
724
725 //make new mode if needed and exit
726 if( !fitsPDF && alphaT > 0.f )
727 {
728 // replace the weakest or add a new one
729 int mode = nmodes == nmixtures ? nmixtures-1 : nmodes++;
730
731 if (nmodes==1)
732 gmm[mode].weight = 1.f;
733 else
734 {
735 gmm[mode].weight = alphaT;
736
737 // renormalize all other weights
738 for( int i = 0; i < nmodes-1; i++ )
739 gmm[i].weight *= alpha1;
740 }
741
742 // init
743 for( int c = 0; c < nchannels; c++ )
744 mean[mode*nchannels + c] = data[c];
745
746 gmm[mode].variance = varInit;
747
748 //sort
749 //find the new place for it
750 for( int i = nmodes - 1; i > 0; i-- )
751 {
752 // check one up
753 if( alphaT < gmm[i-1].weight )
754 break;
755
756 // swap one up
757 std::swap(gmm[i], gmm[i-1]);
758 for( int c = 0; c < nchannels; c++ )
759 std::swap(mean[i*nchannels + c], mean[(i-1)*nchannels + c]);
760 }
761 }
762
763 //set the number of modes
764 modesUsed[x] = uchar(nmodes);
765 mask[x] = background ? 0 :
766 detectShadows && detectShadowGMM(data, nchannels, nmodes, gmm, mean, Tb, TB, tau) ?
767 shadowVal : 255;
768 }
769 }
770 }
771
772 const Mat* src;
773 Mat* dst;
774 GMM* gmm0;
775 float* mean0;
776 uchar* modesUsed0;
777
778 int nmixtures;
779 float alphaT, Tb, TB, Tg;
780 float varInit, varMin, varMax, prune, tau;
781
782 bool detectShadows;
783 uchar shadowVal;
784 const Mat& knownForegroundMask;
785};

890

892
893 if(!knownForegroundMask.empty())
894 {
895 CV_Assert(knownForegroundMask.type() == CV_8UC1);
896 CV_Assert(knownForegroundMask.size() == image.size());
897 }
898
899 ++nframes;
900 learningRate = learningRate >= 0 && nframes > 1 ? learningRate : 1./std::min( 2*nframes, history );
901 CV_Assert(learningRate >= 0);
902
903 parallel_for_(Range(0, image.rows),
904 MOG2Invoker(image, fgmask,
905 bgmodel.ptr<GMM>(),
906 (float*)(bgmodel.ptr() + sizeof(GMM)*nmixtures*image.rows*image.cols),
907 bgmodelUsedModes.ptr(), nmixtures, (float)learningRate,
908 (float)varThreshold,
909 backgroundRatio, varThresholdGen,
910 fVarInit, fVarMin, fVarMax, float(-learningRate*fCT), fTau,
911 bShadowDetection, nShadowDetection, knownForegroundMask),
912 image.total()/(double)(1 << 16));
913}

915template <typename T, int CN>
916void BackgroundSubtractorMOG2Impl::getBackgroundImage_intern(OutputArray backgroundImage) const
917{
918 CV_INSTRUMENT_REGION();
919
920 Mat meanBackground(frameSize, frameType, Scalar::all(0));
921 int firstGaussianIdx = 0;
922 const GMM* gmm = bgmodel.ptr<GMM>();
923 const float* mean = reinterpret_cast<const float*>(gmm + frameSize.width*frameSize.height*nmixtures);
924 Vec<float,CN> meanVal(0.f);
925 for(int row=0; row<meanBackground.rows; row++)
926 {
927 for(int col=0; col<meanBackground.cols; col++)
928 {
929 int nmodes = bgmodelUsedModes.at<uchar>(row, col);
930 float totalWeight = 0.f;
931 for(int gaussianIdx = firstGaussianIdx; gaussianIdx < firstGaussianIdx + nmodes; gaussianIdx++)
932 {
933 GMM gaussian = gmm[gaussianIdx];
934 size_t meanPosition = gaussianIdx*CN;
935 for(int chn = 0; chn < CN; chn++)
936 {
937 meanVal(chn) += gaussian.weight * mean[meanPosition + chn];
938 }
939 totalWeight += gaussian.weight;
940
941 if(totalWeight > backgroundRatio)
942 break;
943 }
944 float invWeight = 0.f;
945 if (std::abs(totalWeight) > FLT_EPSILON) {
946 invWeight = 1.f/totalWeight;
947 }
948
949 meanBackground.at<Vec<T,CN> >(row, col) = Vec<T,CN>(meanVal * invWeight);
950 meanVal = 0.f;
951
952 firstGaussianIdx += nmixtures;
953 }
954 }
955 meanBackground.copyTo(backgroundImage);
956}

959{
960 CV_Assert(frameType == CV_8UC1 || frameType == CV_8UC3 || frameType == CV_32FC1 || frameType == CV_32FC3);
961
962#ifdef HAVE_OPENCL

966

981 break;
