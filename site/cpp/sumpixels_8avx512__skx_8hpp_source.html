32 void calculate_integral_avx512(const uchar *src, size_t _srcstep,
33 double *sum, size_t _sumstep,
34 double *sqsum, size_t _sqsumstep,
35 int width, int height)
36 {
37 const int srcstep = (int)(_srcstep/sizeof(uchar));
38 const int sumstep = (int)(_sumstep/sizeof(double));
39 const int sqsumstep = (int)(_sqsumstep/sizeof(double));
40 const int ops_per_line = width * num_channels;
41
42 // Clear the first line of the sum as per spec (see integral documentation)
43 // Also adjust the index of sum and sqsum to be at the real 0th element
44 // and not point to the border pixel so it stays in sync with the src pointer
45 memset( sum, 0, (ops_per_line+num_channels)*sizeof(double));
46 sum += num_channels;
47
48 if (sqsum) {
49 memset( sqsum, 0, (ops_per_line+num_channels)*sizeof(double));
50 sqsum += num_channels;
51 }
52
53 // Now calculate the integral over the whole image one line at a time
54 for(int y = 0; y < height; y++) {
55 const uchar * src_line = &src[y*srcstep];
56 double * sum_above = &sum[y*sumstep];
57 double * sum_line = &sum_above[sumstep];
58 double * sqsum_above = (sqsum) ? &sqsum[y*sqsumstep] : NULL;
59 double * sqsum_line = (sqsum) ? &sqsum_above[sqsumstep] : NULL;
60
61 calculate_integral_for_line(src_line, sum_line, sum_above, sqsum_line, sqsum_above, ops_per_line);
62
63 }
64 }

66 static CV_ALWAYS_INLINE
67 void calculate_integral_for_line(const uchar *srcs,
68 double *sums, double *sums_above,
69 double *sqsums, double *sqsums_above,
70 int num_ops_in_line)
71 {
72 __m512i sum_accumulator = _mm512_setzero_si512(); // holds rolling sums for the line
73 __m512i sqsum_accumulator = _mm512_setzero_si512(); // holds rolling sqsums for the line
74
75 // The first element on each line must be zeroes as per spec (see integral documentation)
76 zero_out_border_pixel(sums, sqsums);
77
78 // Do all 64 byte chunk operations then do the last bits that don't fit in a 64 byte chunk
79 aligned_integral( srcs, sums, sums_above, sqsums, sqsums_above, sum_accumulator, sqsum_accumulator, num_ops_in_line);
80 post_aligned_integral(srcs, sums, sums_above, sqsums, sqsums_above, sum_accumulator, sqsum_accumulator, num_ops_in_line);
81
82 }

85 static CV_ALWAYS_INLINE
86 void zero_out_border_pixel(double *sums, double *sqsums)
87 {
88 // Note the negative index is because the sums/sqsums pointers point to the first real pixel
89 // after the border pixel so we have to look backwards
90 _mm512_mask_storeu_epi64(&sums[-(ptrdiff_t)num_channels], (1<<num_channels)-1, _mm512_setzero_si512());
91 if (sqsums)
92 _mm512_mask_storeu_epi64(&sqsums[-(ptrdiff_t)num_channels], (1<<num_channels)-1, _mm512_setzero_si512());
93 }

96 static CV_ALWAYS_INLINE
97 void aligned_integral(const uchar *&srcs,
98 double *&sums, double *&sums_above,
99 double *&sqsum, double *&sqsum_above,
100 __m512i &sum_accumulator, __m512i &sqsum_accumulator,
101 int num_ops_in_line)
102 {
103 // This function handles full 64 byte chunks of the source data at a time until it gets to the part of
104 // the line that no longer contains a full 64 byte chunk. Other code will handle the last part.
105
106 const int num_chunks = num_ops_in_line >> 6; // quick int divide by 64
107
108 for (int index_64byte_chunk = 0; index_64byte_chunk < num_chunks; index_64byte_chunk++){
109 integral_64_operations_avx512((__m512i *) srcs,
110 (__m512i *) sums, (__m512i *) sums_above,
111 (__m512i *) sqsum, (__m512i *) sqsum_above,
112 0xFFFFFFFFFFFFFFFF, sum_accumulator, sqsum_accumulator);
113 srcs+=64; sums+=64; sums_above+=64;
114 if (sqsum){ sqsum+= 64; sqsum_above+=64; }
115 }
116 }

119 static CV_ALWAYS_INLINE
120 void post_aligned_integral(const uchar *srcs,
121 const double *sums, const double *sums_above,
122 const double *sqsum, const double *sqsum_above,
123 __m512i &sum_accumulator, __m512i &sqsum_accumulator,
124 int num_ops_in_line)
125 {
126 // This function handles the last few straggling operations that are not a full chunk of 64 operations
127 // We use the same algorithm, but we calculate a different operation mask using (num_ops % 64).
128
129 const unsigned int num_operations = (unsigned int) num_ops_in_line & 0x3F; // Quick int modulo 64
130
131 if (num_operations > 0) {
132 __mmask64 operation_mask = (1ULL << num_operations) - 1ULL;
133
134 integral_64_operations_avx512((__m512i *) srcs, (__m512i *) sums, (__m512i *) sums_above,
135 (__m512i *) sqsum, (__m512i *) sqsum_above,
136 operation_mask, sum_accumulator, sqsum_accumulator);
137 }
138 }

141 static CV_ALWAYS_INLINE
142 void integral_64_operations_avx512(const __m512i *srcs,
143 __m512i *sums, const __m512i *sums_above,
144 __m512i *sqsums, const __m512i *sqsums_above,
145 __mmask64 data_mask,
146 __m512i &sum_accumulator, __m512i &sqsum_accumulator)
147 {
148 __m512i src_64byte_chunk = read_64_bytes(srcs, data_mask);
149
150 while (data_mask) {
151 __m128i src_16bytes = extract_lower_16bytes(src_64byte_chunk);
152
153 __m512i src_longs_lo = convert_lower_8bytes_to_longs(src_16bytes);
154 __m512i src_longs_hi = convert_lower_8bytes_to_longs(shift_right_8_bytes(src_16bytes));
155
156 // Calculate integral for the sum on the 8 lanes at a time
157 integral_8_operations(src_longs_lo, sums_above, data_mask, sums, sum_accumulator);
158 integral_8_operations(src_longs_hi, sums_above+1, data_mask>>8, sums+1, sum_accumulator);
159
160 if (sqsums) {
161 __m512i squared_source_lo = square_m512(src_longs_lo);
162 __m512i squared_source_hi = square_m512(src_longs_hi);
163
164 integral_8_operations(squared_source_lo, sqsums_above, data_mask, sqsums, sqsum_accumulator);
165 integral_8_operations(squared_source_hi, sqsums_above+1, data_mask>>8, sqsums+1, sqsum_accumulator);
166 sqsums += 2;
167 sqsums_above+=2;
168 }
169
170 // Prepare for next iteration of loop
171 // shift source to align next 16 bytes to lane 0, shift the mask, and advance the pointers
172 sums += 2;
173 sums_above += 2;
174 data_mask = data_mask >> 16;
175 src_64byte_chunk = shift_right_16_bytes(src_64byte_chunk);
176
177 }
178
179 }

182 static CV_ALWAYS_INLINE
183 void integral_8_operations(const __m512i src_longs, const __m512i *above_values_ptr, __mmask64 data_mask,
184 __m512i *results_ptr, __m512i &accumulator)
185 {
186 // NOTE that the calculate_integral function referenced here must be implemented in the templated
187 // derivatives because the algorithm depends heavily on the number of channels in the image
188 //
189 _mm512_mask_storeu_pd(
190 results_ptr, // Store the result here
191 (__mmask8)data_mask, // Using the data mask to avoid overrunning the line
192 calculate_integral( // Writing the value of the integral derived from:
193 src_longs, // input data
194 _mm512_maskz_loadu_pd((__mmask8)data_mask, above_values_ptr), // and the results from line above
195 accumulator // keeping track of the accumulator
196 )
197 );
198 }

253//============================================================================================================
254// This the only section that needs to change with respect to algorithm based on the number of channels
255// It is responsible for returning the calculation of 8 lanes worth of the integral and returning in the
256// accumulated sums in the accumulator parameter (NOTE: accumulator is an input and output parameter)
257//
258// The function prototype that needs to be implemented is:
259//
260// __m512d calculate_integral(const __m512i src_longs, const __m512d above_values, __m512i &accumulator){ ... }
261//
262// Description of parameters:
263// INPUTS:
264// src_longs : 8 lanes worth of the source bytes converted to 64 bit integers
265// above_values: 8 lanes worth of the result values from the line above (See the integral spec)
266// accumulator : 8 lanes worth of sums from the previous iteration
267// IMPORTANT NOTE: This parameter is both an INPUT AND OUTPUT parameter to this function
268//
269// OUTPUTS:
270// return value: The actual integral value for all 8 lanes which is defined by the spec as
271// the sum of all channel values to the left of a given pixel plus the result
272// written to the line directly above the current line.
273// accumulator: This is an input and and output. This parameter should be left with the accumulated
274// sums for the current 8 lanes keeping all entries in the proper lane (do not shuffle it)
275//
276// Below here is the channel specific implementation
277//
278
279//========================================
280// 1 Channel Integral Implementation
281//========================================
282template<>
283CV_ALWAYS_INLINE
284__m512d IntegralCalculator < 1 > ::calculate_integral(const __m512i src_longs, const __m512d above_values, __m512i &accumulator)
285{
286 // One channel support is implemented differently than 2, 3, or 4 channel
287 // One channel support has more horizontal operations that cannot be made vertical without losing performance
288 // The logical operations needed look like:
289 // Vertical LANES : |7|6|5|4|3|2|1|0|
290 // src_longs : |H|G|F|E|D|C|B|A|
291 // shift_by_1 : + |G|F|E|D|C|B|A| |
292 // shift_by_2 : + |F|E|D|C|B|A| | |
293 // shift_by_3 : + |E|D|C|B|A| | | |
294 // shift_by_4 : + |D|C|B|A| | | | |
295 // shift_by_5 : + |C|B|A| | | | | |
296 // shift_by_6 : + |B|A| | | | | | |
297 // shift_by_7 : + |A| | | | | | | |
298 // carry_over_idxs : + |7|7|7|7|7|7|7|7| (index position of result from previous iteration)
299 // = integral
300 //
301 // If we do this vertically we end up losing performance because of the number of operations. We will instead
302 // do a horizontal add tree to create the vertical sections we need as a tree
303 // Vertical Lanes: | 7 | 6 | 5 | 4 | 3 | 2 | 1 | 0 |
304 // src_longs: | H | G | F | E | D | C | B | A |
305 // horiz_sum_1: | | | | | G+H | E+F | C+D | A+B |
306 // horiz_sum_2: | | | | | | | EFGH | ABCD |
307 //
308 const __m512i horiz_sum_1 = m512_hadd(src_longs); // indexes for the permutes below (3,2,1,0) = (GH, EF, CD, AB)
309 const __m512i horiz_sum_2 = m512_hadd(horiz_sum_1); // indexes for the permutes below (9, 8) = (EFGH, ABCD)
310
311 // Then we can use the partial sums by looking at the vertical stacks above and realize that, for example
312 // ABCD appears vertically in lanes 7, 6, 5, 4, and 3 so we will permute the values so that all partial products
313 // appear in the right lanes. and sum them up along with the carry over value from the accumulator. So we setup
314 // the lanes like:
315 // Vertical Lanes: | 7 | 6 | 5 | 4 | 3 | 2 | 1 | 0 |
316 // s1 : | 0 | G | 0 | E | 0 | C | 0 | A |
317 // s2 : | ABCD | ABCD | ABCD | ABCD | ABCD | AB | AB | 0 |
318 // s3 : | EFGH | EF | EF | 0 | 0 | 0 | 0 | 0 |
319 // +------+------+------+------+------+------+------+------+
320 // sum : | A..H | A..G | A..F | A..E | A..D | A..C | A..B | A | Integral :-)
321 //
322 const __m512i s1 = _mm512_maskz_mov_epi64(0x55, src_longs); // 0 G 0 E 0 D 0 C 0 A
323 const __m512i s2 = _mm512_permutex2var_epi64(horiz_sum_1, _mm512_set_epi64(8,8,8,8,8,0,0,4), horiz_sum_2);
324 const __m512i s3 = _mm512_permutex2var_epi64(horiz_sum_1, _mm512_set_epi64(9,2,2,4,4,4,4,4), horiz_sum_2);
325
326 // Now we use the rolling sum from the previous iteration from accumulator and replicate it into carry_over
327 // And sum everything up into the accumulator
328 //
329 const __m512i carry_over = _mm512_permutex2var_epi64(accumulator, _mm512_set_epi64(7,7,7,7,7,7,7,7), accumulator);
330 accumulator = _mm512_add_epi64(_mm512_add_epi64(s2, s3), _mm512_add_epi64(carry_over, s1));
331
332 // Convert to double precision and store
333 //
334 __m512d integral_pd = _mm512_add_pd(_mm512_cvtepu64_pd(accumulator), above_values);
335 return integral_pd;
336}

339//========================================
340// 2 Channel Integral Implementation
341//========================================
342template<>
343CV_ALWAYS_INLINE
344__m512d IntegralCalculator < 2 > ::calculate_integral(const __m512i src_longs, const __m512d above_values, __m512i &accumulator)
345{
346 __m512i carryover_idxs = _mm512_set_epi64(7, 6, 7, 6, 7, 6, 7, 6);
347
348 // Align data to prepare for the adds:
349 // shifts data left by 3 and 6 qwords(lanes) and gets rolling sum in all lanes
350 // Vertical LANES : 76543210
351 // src_longs : HGFEDCBA
352 // shift2lanes : + FEDCBA
353 // shift4lanes : + DCBA
354 // shift6lanes : + BA
355 // carry_over_idxs : + 76767676 (index position of result from previous iteration)
356 // = integral
357 __m512i shift2lanes = _mm512_maskz_expand_epi64(0xFC, src_longs);
358 __m512i shift4lanes = _mm512_maskz_expand_epi64(0xF0, src_longs);
359 __m512i shift6lanes = _mm512_maskz_expand_epi64(0xC0, src_longs);
360 __m512i carry_over = _mm512_permutex2var_epi64(accumulator, carryover_idxs, accumulator);
361
362 // Add all values in tree form for perf ((0+2) + (4+6))
363 __m512i sum_shift_02 = _mm512_add_epi64(src_longs, shift2lanes);
364 __m512i sum_shift_46 = _mm512_add_epi64(shift4lanes, shift6lanes);
365 __m512i sum_all = _mm512_add_epi64(sum_shift_02, sum_shift_46);
366 accumulator = _mm512_add_epi64(sum_all, carry_over);
367
368 // Convert to packed double and add to the line above to get the true integral value
369 __m512d accumulator_pd = _mm512_cvtepu64_pd(accumulator);
370 __m512d integral_pd = _mm512_add_pd(accumulator_pd, above_values);
371 return integral_pd;
372}

378template<>
379CV_ALWAYS_INLINE
380__m512d IntegralCalculator < 3 > ::calculate_integral(const __m512i src_longs, const __m512d above_values, __m512i &accumulator)
381{
382 __m512i carryover_idxs = _mm512_set_epi64(6, 5, 7, 6, 5, 7, 6, 5);
383
384 // Align data to prepare for the adds:
385 // shifts data left by 3 and 6 qwords(lanes) and gets rolling sum in all lanes
386 // Vertical LANES: 76543210
387 // src_longs : HGFEDCBA
388 // shift3lanes : + EDCBA
389 // shift6lanes : + BA
390 // carry_over_idxs : + 65765765 (index position of result from previous iteration)
391 // = integral
392 __m512i shift3lanes = _mm512_maskz_expand_epi64(0xF8, src_longs);
393 __m512i shift6lanes = _mm512_maskz_expand_epi64(0xC0, src_longs);
394 __m512i carry_over = _mm512_permutex2var_epi64(accumulator, carryover_idxs, accumulator);
395
396 // Do the adds in tree form
397 __m512i sum_shift_03 = _mm512_add_epi64(src_longs, shift3lanes);
398 __m512i sum_rest = _mm512_add_epi64(shift6lanes, carry_over);
399 accumulator = _mm512_add_epi64(sum_shift_03, sum_rest);
400
401 // Convert to packed double and add to the line above to get the true integral value
402 __m512d accumulator_pd = _mm512_cvtepu64_pd(accumulator);
403 __m512d integral_pd = _mm512_add_pd(accumulator_pd, above_values);
404 return integral_pd;
405}

412CV_ALWAYS_INLINE
413__m512d IntegralCalculator < 4 > ::calculate_integral(const __m512i src_longs, const __m512d above_values, __m512i &accumulator)
414{
415 __m512i carryover_idxs = _mm512_set_epi64(7, 6, 5, 4, 7, 6, 5, 4);
416
417 // Align data to prepare for the adds:
418 // shifts data left by 3 and 6 qwords(lanes) and gets rolling sum in all lanes
419 // Vertical LANES: 76543210
420 // src_longs : HGFEDCBA
421 // shift4lanes : + DCBA
422 // carry_over_idxs : + 76547654 (index position of result from previous iteration)
423 // = integral
424 __m512i shifted4lanes = _mm512_maskz_expand_epi64(0xF0, src_longs);
425 __m512i carry_over = _mm512_permutex2var_epi64(accumulator, carryover_idxs, accumulator);

428 __m512i sum_shift_04 = _mm512_add_epi64(src_longs, shifted4lanes);

432 __m512d accumulator_pd = _mm512_cvtepu64_pd(accumulator);
433 __m512d integral_pd = _mm512_add_pd(accumulator_pd, above_values);

441void calculate_integral_avx512(const uchar *src, size_t _srcstep,
442 double *sum, size_t _sumstep,
443 double *sqsum, size_t _sqsumstep,
444 int width, int height, int cn)
445{
446 CV_INSTRUMENT_REGION();
447
448 switch(cn){
449 case 1: {
450 IntegralCalculator< 1 > calculator;
451 calculator.calculate_integral_avx512(src, _srcstep, sum, _sumstep, sqsum, _sqsumstep, width, height);
452 break;
453 }
454 case 2: {
455 IntegralCalculator< 2 > calculator;
456 calculator.calculate_integral_avx512(src, _srcstep, sum, _sumstep, sqsum, _sqsumstep, width, height);
457 break;
458 }
459 case 3: {
460 IntegralCalculator< 3 > calculator;
461 calculator.calculate_integral_avx512(src, _srcstep, sum, _sumstep, sqsum, _sqsumstep, width, height);
462 break;
463 }
464 case 4: {
465 IntegralCalculator< 4 > calculator;
466 calculator.calculate_integral_avx512(src, _srcstep, sum, _sumstep, sqsum, _sqsumstep, width, height);
467 break;
468 }
469 }
