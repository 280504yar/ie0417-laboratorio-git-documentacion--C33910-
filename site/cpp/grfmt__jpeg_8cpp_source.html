128METHODDEF(void)
129skip_input_data(j_decompress_ptr cinfo, long num_bytes)
130{
131 JpegSource* source = (JpegSource*) cinfo->src;
132
133 if( num_bytes > (long)source->pub.bytes_in_buffer )
134 {
135 // We need to skip more data than we have in the buffer.
136 // This will force the JPEG library to suspend decoding.
137 source->skip = (int)(num_bytes - source->pub.bytes_in_buffer);
138 source->pub.next_input_byte += source->pub.bytes_in_buffer;
139 source->pub.bytes_in_buffer = 0;
140 }
141 else
142 {
143 // Skip portion of the buffer
144 source->pub.bytes_in_buffer -= num_bytes;
145 source->pub.next_input_byte += num_bytes;
146 source->skip = 0;
147 }
148}

151static void jpeg_buffer_src(j_decompress_ptr cinfo, JpegSource* source)
152{

220bool JpegDecoder::readHeader()
221{
222 volatile bool result = false;
223 close();
224
225 JpegState* state = new JpegState;
226 m_state = state;
227 state->cinfo.err = jpeg_std_error(&state->jerr.pub);
228 state->jerr.pub.error_exit = error_exit;
229
230 if( setjmp( state->jerr.setjmp_buffer ) == 0 )
231 {
232 jpeg_create_decompress( &state->cinfo );
233
234 if( !m_buf.empty() )
235 {
236 jpeg_buffer_src(&state->cinfo, &state->source);
237 state->source.pub.next_input_byte = m_buf.ptr();
238 state->source.pub.bytes_in_buffer = m_buf.cols*m_buf.rows*m_buf.elemSize();
239 }
240 else
241 {
242 m_f = fopen( m_filename.c_str(), "rb" );
243 if( m_f )
244 jpeg_stdio_src( &state->cinfo, m_f );
245 }
246
247 if (state->cinfo.src != 0)
248 {
249 jpeg_save_markers(&state->cinfo, APP1, 0xffff);
250 jpeg_save_markers(&state->cinfo, APP2, 0xffff);
251 jpeg_read_header( &state->cinfo, TRUE );
252
253 const std::streamsize EXIF_HEADER_SIZE = 6; // "Exif\0\0"
254 const std::streamsize XMP_HEADER_SIZE = 29; // "http://ns.adobe.com/xap/1.0/"
255 const std::streamsize ICC_HEADER_SIZE = 14; // "ICC_PROFILE\0" + seq/total
256
257 for (jpeg_saved_marker_ptr cmarker = state->cinfo.marker_list; cmarker != nullptr; cmarker = cmarker->next)
258 {
259 // Handle APP1 marker: could be Exif or XMP
260 if (cmarker->marker == APP1 && cmarker->data_length > EXIF_HEADER_SIZE)
261 {
262 unsigned char* data = cmarker->data;
263
264 // Check for Exif data
265 if (std::memcmp(data, "Exif\0\0", EXIF_HEADER_SIZE) == 0)
266 {
267 m_exif.parseExif(data + EXIF_HEADER_SIZE, cmarker->data_length - EXIF_HEADER_SIZE);
268 }
269 // Check for XMP metadata
270 else if (m_read_options && cmarker->data_length >= XMP_HEADER_SIZE &&
271 std::memcmp(data, "http://ns.adobe.com/xap/1.0/", XMP_HEADER_SIZE) == 0)
272 {
273 std::vector<uchar>& xmp = m_metadata[IMAGE_METADATA_XMP];
274 xmp.insert(xmp.end(), data, data + cmarker->data_length);
275 }
276 }
277
278 // Handle APP2 marker: typically contains ICC profile data
279 if (m_read_options && cmarker->marker == APP2 && cmarker->data_length > ICC_HEADER_SIZE)
280 {
281 const unsigned char* data = cmarker->data;
282 std::vector<uchar>& iccp = m_metadata[IMAGE_METADATA_ICCP];
283 iccp.insert(iccp.end(), data + ICC_HEADER_SIZE, data + cmarker->data_length);
284 }
285 }
286
287 state->cinfo.scale_num=1;
288 state->cinfo.scale_denom = m_scale_denom;
289 m_scale_denom=1; // trick! to know which decoder used scale_denom see imread_
290 jpeg_calc_output_dimensions(&state->cinfo);
291 m_width = state->cinfo.output_width;
292 m_height = state->cinfo.output_height;
293 m_type = state->cinfo.num_components > 1 ? CV_8UC3 : CV_8UC1;
294 result = true;
295 }
296 }
297
298 return result;
299}

369/*
370 * Parse the DHT table.
371 * This code comes from jpeg6b (jdmarker.c).
372 */
373static
374int my_jpeg_load_dht (struct jpeg_decompress_struct *info, unsigned char *dht,
375 JHUFF_TBL *ac_tables[], JHUFF_TBL *dc_tables[])
376{
377 unsigned int length = (dht[2] << 8) + dht[3] - 2;
378 unsigned int pos = 4;
379 unsigned int count, i;
380 int index;
381
382 JHUFF_TBL **hufftbl;
383 unsigned char bits[17];
384 unsigned char huffval[256] = {0};
385
386 while (length > 16)
387 {
388 bits[0] = 0;
389 index = dht[pos++];
390 count = 0;
391 for (i = 1; i <= 16; ++i)
392 {
393 bits[i] = dht[pos++];
394 count += bits[i];
395 }
396 length -= 17;
397
398 if (count > 256 || count > length)
399 return -1;
400
401 for (i = 0; i < count; ++i)
402 huffval[i] = dht[pos++];
403 length -= count;
404
405 if (index & 0x10)
406 {
407 index &= ~0x10;
408 hufftbl = &ac_tables[index];
409 }
410 else
411 hufftbl = &dc_tables[index];
412
413 if (index < 0 || index >= NUM_HUFF_TBLS)
414 return -1;
415
416 if (*hufftbl == NULL)
417 *hufftbl = jpeg_alloc_huff_table ((j_common_ptr)info);
418 if (*hufftbl == NULL)
419 return -1;
420
421 memcpy ((*hufftbl)->bits, bits, sizeof (*hufftbl)->bits);
422 memcpy ((*hufftbl)->huffval, huffval, sizeof (*hufftbl)->huffval);
423 }
424
425 if (length != 0)
426 return -1;
427
428 return 0;
429}

437bool JpegDecoder::readData( Mat& img )
438{
439 volatile bool result = false;
440 const bool color = img.channels() > 1;
441
442 if( m_state && m_width && m_height )
443 {
444 jpeg_decompress_struct* cinfo = &((JpegState*)m_state)->cinfo;
445 JpegErrorMgr* jerr = &((JpegState*)m_state)->jerr;
446
447 if( setjmp( jerr->setjmp_buffer ) == 0 )
448 {
449#ifdef CV_MANUAL_JPEG_STD_HUFF_TABLES
450 /* check if this is a mjpeg image format */
451 if ( cinfo->ac_huff_tbl_ptrs[0] == NULL &&
452 cinfo->ac_huff_tbl_ptrs[1] == NULL &&
453 cinfo->dc_huff_tbl_ptrs[0] == NULL &&
454 cinfo->dc_huff_tbl_ptrs[1] == NULL )
455 {
456 /* yes, this is a mjpeg image format, so load the correct
457 huffman table */
458 my_jpeg_load_dht( cinfo,
459 my_jpeg_odml_dht,
460 cinfo->ac_huff_tbl_ptrs,
461 cinfo->dc_huff_tbl_ptrs );
462 }
463#endif
464
465 // See https://github.com/opencv/opencv/issues/25274
466 // Conversion CMYK->BGR is not supported in libjpeg-turbo.
467 // So supporting both directly and indirectly is necessary.
468 bool doDirectRead = false;
469
470 if( color )
471 {
472 if( cinfo->num_components != 4 )
473 {
474#ifdef JCS_EXTENSIONS
475 cinfo->out_color_space = m_use_rgb ? JCS_EXT_RGB : JCS_EXT_BGR;
476 cinfo->out_color_components = 3;
477 doDirectRead = true; // BGR -> BGR
478#else
479 cinfo->out_color_space = JCS_RGB;
480 cinfo->out_color_components = 3;
481 doDirectRead = m_use_rgb ? true : false; // RGB -> BGR
482#endif
483 }
484 else
485 {
486 cinfo->out_color_space = JCS_CMYK;
487 cinfo->out_color_components = 4;
488 doDirectRead = false; // CMYK -> BGR
489 }
490 }
491 else
492 {
493 if( cinfo->num_components != 4 )
494 {
495 cinfo->out_color_space = JCS_GRAYSCALE;
496 cinfo->out_color_components = 1;
497 doDirectRead = true; // GRAY -> GRAY
498 }
499 else
500 {
501 cinfo->out_color_space = JCS_CMYK;
502 cinfo->out_color_components = 4;
503 doDirectRead = false; // CMYK -> GRAY
504 }
505 }
506
507 jpeg_start_decompress( cinfo );
508
509 if( doDirectRead)
510 {
511 for( int iy = 0 ; iy < m_height; iy ++ )
512 {
513 uchar* data = img.ptr<uchar>(iy);
514 if (jpeg_read_scanlines( cinfo, &data, 1 ) != 1) return false;
515 }
516 }
517 else
518 {
519 JSAMPARRAY buffer = (*cinfo->mem->alloc_sarray)((j_common_ptr)cinfo,
520 JPOOL_IMAGE, m_width*4, 1 );
521
522 for( int iy = 0 ; iy < m_height; iy ++ )
523 {
524 uchar* data = img.ptr<uchar>(iy);
525 if (jpeg_read_scanlines( cinfo, buffer, 1 ) != 1) return false;
526
527 if( color )
528 {
529 if (m_use_rgb)
530 {
531 if( cinfo->out_color_components == 3 )
532 icvCvt_BGR2RGB_8u_C3R( buffer[0], 0, data, 0, Size(m_width,1) );
533 else
534 icvCvt_CMYK2RGB_8u_C4C3R( buffer[0], 0, data, 0, Size(m_width,1) );
535 }
536 else
537 {
538 if( cinfo->out_color_components == 3 )
539 icvCvt_RGB2BGR_8u_C3R( buffer[0], 0, data, 0, Size(m_width,1) );
540 else
541 icvCvt_CMYK2BGR_8u_C4C3R( buffer[0], 0, data, 0, Size(m_width,1) );
542 }
543 }
544 else
545 {
546 if( cinfo->out_color_components == 1 )
547 memcpy( data, buffer[0], m_width );
548 else
549 icvCvt_CMYK2Gray_8u_C4C1R( buffer[0], 0, data, 0, Size(m_width,1) );
550 }
551 }
552 }
553
554 result = true;
555 jpeg_finish_decompress( cinfo );
556 }
557 }
558
559 return result;
560}

632bool JpegEncoder::write( const Mat& img, const std::vector<int>& params )
633{
634 m_last_error.clear();

636 struct fileWrapper
637 {
638 FILE* f;
639
640 fileWrapper() : f(0) {}
641 ~fileWrapper() { if(f) fclose(f); }
642 };
643 volatile bool result = false;
644 fileWrapper fw;
645 int width = img.cols, height = img.rows;
646
647 std::vector<uchar> out_buf(1 << 12);
648
649 struct jpeg_compress_struct cinfo;
650 JpegErrorMgr jerr;
651 JpegDestination dest;
652
653 cinfo.err = jpeg_std_error(&jerr.pub);
654 jerr.pub.error_exit = error_exit;
655 jpeg_create_compress(&cinfo);
656
657 if( !m_buf )
658 {
659 fw.f = fopen( m_filename.c_str(), "wb" );
660 if( !fw.f )
661 goto _exit_;
662 jpeg_stdio_dest( &cinfo, fw.f );
663 }
664 else
665 {
666 dest.dst = m_buf;
667 dest.buf = &out_buf;
668
669 jpeg_buffer_dest( &cinfo, &dest );
670
671 dest.pub.next_output_byte = &out_buf[0];
672 dest.pub.free_in_buffer = out_buf.size();
673 }
674
675 if( setjmp( jerr.setjmp_buffer ) == 0 )
676 {
677 cinfo.image_width = width;
678 cinfo.image_height = height;
679
680 int _channels = img.channels();
681 int channels = _channels > 1 ? 3 : 1;
682
683 bool doDirectWrite = false;
684 switch( _channels )
685 {
686 case 1:
687 cinfo.input_components = 1;
688 cinfo.in_color_space = JCS_GRAYSCALE;
689 doDirectWrite = true; // GRAY -> GRAY
690 break;
691 case 3:
692#ifdef JCS_EXTENSIONS
693 cinfo.input_components = 3;
694 cinfo.in_color_space = JCS_EXT_BGR;
695 doDirectWrite = true; // BGR -> BGR
696#else
697 cinfo.input_components = 3;
698 cinfo.in_color_space = JCS_RGB;
699 doDirectWrite = false; // BGR -> RGB
700#endif
701 break;
702 case 4:
703#ifdef JCS_EXTENSIONS
704 cinfo.input_components = 4;
705 cinfo.in_color_space = JCS_EXT_BGRX;
706 doDirectWrite = true; // BGRX -> BGRX
707#else
708 cinfo.input_components = 3;
709 cinfo.in_color_space = JCS_RGB;
710 doDirectWrite = false; // BGRA -> RGB
711#endif
712 break;
713 default:
714 CV_Error(cv::Error::StsError, cv::format("Unsupported number of _channels: %06d", _channels) );
715 break;
716 }
717
718 int quality = 95;
719 int progressive = 0;
720 int optimize = 0;
721 int rst_interval = 0;
722 int luma_quality = -1;
723 int chroma_quality = -1;
724 uint32_t sampling_factor = 0; // same as 0x221111
725
726 for( size_t i = 0; i < params.size(); i += 2 )
727 {
728 const int value = params[i+1];
729 if( params[i] == IMWRITE_JPEG_QUALITY )
730 {
731 quality = MIN(MAX(value, 0), 100);
732 if(value != quality) {
733 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_JPEG_QUALITY must be between 0 to 100. It is fallbacked to %d", value, quality));
734 }
735 }
736
737 if( params[i] == IMWRITE_JPEG_PROGRESSIVE )
738 {
739 progressive = MIN(MAX(value,0), 1);
740 if(value != progressive) {
741 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_JPEG_PROGRESSIVE must be 0 or 1. It is fallbacked to %d", value, progressive));
742 }
743 }
744
745 if( params[i] == IMWRITE_JPEG_OPTIMIZE )
746 {
747 optimize = MIN(MAX(value,0), 1);
748 if(value != optimize) {
749 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_JPEG_OPTIMIZE must be 0 or 1. It is fallbacked to %d", value, optimize));
750 }
751 }
752
753 if( params[i] == IMWRITE_JPEG_LUMA_QUALITY )
754 {
755 if (value >= 0)
756 {
757 luma_quality = MIN(MAX(value, 0), 100);
758 if(value != luma_quality) {
759 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_JPEG_LUMA_QUALITY must be between 0 to 100. It is fallbacked to %d.", value, luma_quality));
760 }
761
762 quality = luma_quality;
763
764 if (chroma_quality < 0)
765 {
766 chroma_quality = luma_quality;
767 }
768 } else {
769 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_JPEG_LUMA_QUALITY must be between 0 to 100. It is ignored.", value));
770 }
771 }
772
773 if( params[i] == IMWRITE_JPEG_CHROMA_QUALITY )
774 {
775 if (params[i+1] >= 0)
776 {
777 chroma_quality = MIN(MAX(value, 0), 100);
778 if(value != chroma_quality) {
779 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_JPEG_CHROMA_QUALITY must be between 0 to 100. It is fallbacked to %d.", value, chroma_quality));
780 }
781 } else {
782 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_JPEG_CHROMA_QUALITY must be between 0 to 100. It is ignored.", value));
783 }
784 }
785
786 if( params[i] == IMWRITE_JPEG_RST_INTERVAL )
787 {
788 rst_interval = MIN(MAX(value, 0), 65535L);
789 if(value != rst_interval) {
790 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_JPEG_RST_INTERVAL must be between 0 to 65535. It is fallbacked to %d.", value, rst_interval));
791 }
792 }
793
794 if( params[i] == IMWRITE_JPEG_SAMPLING_FACTOR )
795 {
796 sampling_factor = static_cast<uint32_t>(value);
797
798 switch ( sampling_factor )
799 {
800 case IMWRITE_JPEG_SAMPLING_FACTOR_411:
801 case IMWRITE_JPEG_SAMPLING_FACTOR_420:
802 case IMWRITE_JPEG_SAMPLING_FACTOR_422:
803 case IMWRITE_JPEG_SAMPLING_FACTOR_440:
804 case IMWRITE_JPEG_SAMPLING_FACTOR_444:
805 // OK.
806 break;
807
808 default:
809 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_JPEG_SAMPLING_FACTOR must be one of ImwriteJPEGSamplingFactorParams. It is fallbacked to IMWRITE_JPEG_SAMPLING_FACTOR_420.", value));
810 sampling_factor = 0;
811 break;
812 }
813 }
814 }
815
816 jpeg_set_defaults( &cinfo );
817 cinfo.restart_interval = rst_interval;
818
819 jpeg_set_quality( &cinfo, quality,
820 TRUE /* limit to baseline-JPEG values */ );
821 if( progressive )
822 jpeg_simple_progression( &cinfo );
823 if( optimize )
824 cinfo.optimize_coding = TRUE;
825
826 if( (channels > 1) && ( sampling_factor != 0 ) )
827 {
828 cinfo.comp_info[0].h_samp_factor = (sampling_factor >> 20 ) & 0xF;
829 cinfo.comp_info[0].v_samp_factor = (sampling_factor >> 16 ) & 0xF;
830 cinfo.comp_info[1].h_samp_factor = (sampling_factor >> 12 ) & 0xF;
831 cinfo.comp_info[1].v_samp_factor = (sampling_factor >> 8 ) & 0xF;
832 cinfo.comp_info[2].h_samp_factor = (sampling_factor >> 4 ) & 0xF;
833 cinfo.comp_info[2].v_samp_factor = (sampling_factor >> 0 ) & 0xF;
834 }
835
836 if (luma_quality >= 0 && chroma_quality >= 0)
837 {
838#if JPEG_LIB_VERSION >= 70
839 cinfo.q_scale_factor[0] = jpeg_quality_scaling(luma_quality);
840 cinfo.q_scale_factor[1] = jpeg_quality_scaling(chroma_quality);
841 if ( luma_quality != chroma_quality )
842 {
843 /* disable subsampling - ref. Libjpeg.txt */
844 cinfo.comp_info[0].v_samp_factor = 1;
845 cinfo.comp_info[0].h_samp_factor = 1;
846 cinfo.comp_info[1].v_samp_factor = 1;
847 cinfo.comp_info[1].h_samp_factor = 1;
848 cinfo.comp_info[2].v_samp_factor = 1;
849 cinfo.comp_info[2].h_samp_factor = 1;
850 }
851 jpeg_default_qtables( &cinfo, TRUE );
852#else
853 // See https://github.com/opencv/opencv/issues/25646
854 CV_LOG_ONCE_WARNING(NULL, cv::format("IMWRITE_JPEG_LUMA/CHROMA_QUALITY are not supported bacause JPEG_LIB_VERSION < 70."));
855#endif // #if JPEG_LIB_VERSION >= 70
856 }
857
858 jpeg_start_compress( &cinfo, TRUE );
859
860 if (!m_metadata.empty()) {
861 const std::vector<uchar>& metadata_exif = m_metadata[IMAGE_METADATA_EXIF];
862 size_t exif_size = metadata_exif.size();
863 if (exif_size > 0u) {
864 const char app1_exif_prefix[] = {'E', 'x', 'i', 'f', '\0', '\0'};
865 size_t app1_exif_prefix_size = sizeof(app1_exif_prefix);
866 size_t data_size = exif_size + app1_exif_prefix_size;
867
868 std::vector<uchar> metadata_app1(data_size);
869 uchar* data = metadata_app1.data();
870 memcpy(data, app1_exif_prefix, app1_exif_prefix_size);
871 memcpy(data + app1_exif_prefix_size, metadata_exif.data(), exif_size);
872 jpeg_write_marker(&cinfo, JPEG_APP0 + 1, data, (unsigned)data_size);
873 }
874
875 const std::vector<uchar>& metadata_xmp = m_metadata[IMAGE_METADATA_XMP];
876 size_t xmp_size = metadata_xmp.size();
877 if (xmp_size > 0u) {
878 jpeg_write_marker(&cinfo, JPEG_APP0 + 1, metadata_xmp.data(), (unsigned)xmp_size);
879 }
880
881 const std::vector<uchar>& metadata_iccp = m_metadata[IMAGE_METADATA_ICCP];
882 size_t iccp_size = metadata_iccp.size();
883 if (iccp_size > 0u) {
884 const char app1_iccp_prefix[] = {'I','C','C','_','P','R','O','F','I','L','E','\0','\1','\1'};
885 size_t app1_iccp_prefix_size = sizeof(app1_iccp_prefix);
886 size_t data_size = iccp_size + app1_iccp_prefix_size;
887
888 std::vector<uchar> metadata_app1(data_size);
889 uchar* data = metadata_app1.data();
890 memcpy(data, app1_iccp_prefix, app1_iccp_prefix_size);
891 memcpy(data + app1_iccp_prefix_size, metadata_iccp.data(), iccp_size);
892 jpeg_write_marker(&cinfo, JPEG_APP0 + 2, data, (unsigned)data_size);
893 }
894 }
895
896 if( doDirectWrite )
897 {
898 for( int y = 0; y < height; y++ )
899 {
900 uchar *data = const_cast<uchar*>(img.ptr<uchar>(y));
901 jpeg_write_scanlines( &cinfo, &data, 1 );
902 }
903 }
904 else
905 {
906 CV_Check(_channels, (_channels == 3) || (_channels == 4), "Unsupported number of channels(indirect write)");
907
908 AutoBuffer<uchar> _buffer;
909 _buffer.allocate(width*channels);
910 uchar *buffer = _buffer.data();
911
912 for( int y = 0; y < height; y++ )
913 {
914 uchar *data = const_cast<uchar*>(img.ptr<uchar>(y));
915 if( _channels == 3 )
916 {
917 icvCvt_BGR2RGB_8u_C3R( data, 0, buffer, 0, Size(width,1) );
918 }
919 else // if( _channels == 4 )
920 {
921 icvCvt_BGRA2BGR_8u_C4C3R( data, 0, buffer, 0, Size(width,1), 2 );
922 }
923 jpeg_write_scanlines( &cinfo, &buffer, 1 );
924 }
925 }
926
927 jpeg_finish_compress( &cinfo );
928 result = true;
929 }
930
931_exit_:
932
933 if(!result)
934 {
935 char jmsg_buf[JMSG_LENGTH_MAX];
936 jerr.pub.format_message((j_common_ptr)&cinfo, jmsg_buf);
937 m_last_error = jmsg_buf;
938 }
939
940 jpeg_destroy_compress( &cinfo );
941
942 return result;
943}
