55static const size_t OUT_OF_RANGE = (size_t)1 << (sizeof(size_t)*8 - 2);

57static void
58calcHistLookupTables_8u( const Mat& hist, const SparseMat& shist,
59 int dims, const float** ranges, const double* uniranges,
60 bool uniform, bool issparse, std::vector<size_t>& _tab )
61{
62 const int low = 0, high = 256;
63 int i, j;
64 _tab.resize((high-low)*dims);
65 size_t* tab = &_tab[0];
66
67 if( uniform )
68 {
69 for( i = 0; i < dims; i++ )
70 {
71 double a = uniranges[i*2];
72 double b = uniranges[i*2+1];
73 int sz = !issparse ? hist.size[i] : shist.size(i);
74 size_t step = !issparse ? hist.step[i] : 1;
75
76 double v_lo = ranges ? ranges[i][0] : 0;
77 double v_hi = ranges ? ranges[i][1] : 256;
78
79 for( j = low; j < high; j++ )
80 {
81 int idx = cvFloor(j*a + b);
82 size_t written_idx = OUT_OF_RANGE;
83 if (j >= v_lo && j < v_hi)
84 {
85 idx = CV_CLAMP_INT(idx, 0, sz - 1);
86 written_idx = idx*step;
87 }
88 tab[i*(high - low) + j - low] = written_idx;
89 }
90 }
91 }
92 else if (ranges)
93 {
94 for( i = 0; i < dims; i++ )
95 {
96 int limit = std::min(cvCeil(ranges[i][0]), high);
97 int idx = -1, sz = !issparse ? hist.size[i] : shist.size(i);
98 size_t written_idx = OUT_OF_RANGE;
99 size_t step = !issparse ? hist.step[i] : 1;
100
101 for(j = low;;)
102 {
103 for( ; j < limit; j++ )
104 tab[i*(high - low) + j - low] = written_idx;
105
106 if( (unsigned)(++idx) < (unsigned)sz )
107 {
108 limit = std::min(cvCeil(ranges[i][idx+1]), high);
109 written_idx = idx*step;
110 }
111 else
112 {
113 for( ; j < high; j++ )
114 tab[i*(high - low) + j - low] = OUT_OF_RANGE;
115 break;
116 }
117 }
118 }
119 }
120 else
121 {
122 CV_Error(Error::StsBadArg, "Either ranges, either uniform ranges should be provided");
123 }

127static void histPrepareImages( const Mat* images, int nimages, const int* channels,
128 const Mat& mask, int dims, const int* histSize,
129 const float** ranges, bool uniform,
130 std::vector<uchar*>& ptrs, std::vector<int>& deltas,
131 Size& imsize, std::vector<double>& uniranges )
132{
133 int i, j, c;
134 CV_Assert( channels != 0 || nimages == dims );
135
136 imsize = images[0].size();
137 int depth = images[0].depth(), esz1 = (int)images[0].elemSize1();
138 bool isContinuous = true;
139
140 ptrs.resize(dims + 1);
141 deltas.resize((dims + 1)*2);
142
143 for( i = 0; i < dims; i++ )
144 {
145 if(!channels)
146 {
147 j = i;
148 c = 0;
149 CV_Assert( images[j].channels() == 1 );
150 }
151 else
152 {
153 c = channels[i];
154 CV_Assert( c >= 0 );
155 for( j = 0; j < nimages; c -= images[j].channels(), j++ )
156 if( c < images[j].channels() )
157 break;
158 CV_Assert( j < nimages );
159 }
160
161 CV_Assert( images[j].size() == imsize && images[j].depth() == depth );
162 if( !images[j].isContinuous() )
163 isContinuous = false;
164 ptrs[i] = images[j].data + c*esz1;
165 deltas[i*2] = images[j].channels();
166 deltas[i*2+1] = (int)(images[j].step/esz1 - imsize.width*deltas[i*2]);
167 }
168
169 if( !mask.empty() )
170 {
171 CV_Assert( mask.size() == imsize && mask.channels() == 1 );
172 isContinuous = isContinuous && mask.isContinuous();
173 ptrs[dims] = mask.data;
174 deltas[dims*2] = 1;
175 deltas[dims*2 + 1] = (int)(mask.step/mask.elemSize1());
176 }
177
178 if( isContinuous )
179 {
180 imsize.width *= imsize.height;
181 imsize.height = 1;
182 }
183
184 if( !ranges ) // implicit uniform ranges for 8U
185 {
186 CV_Assert( depth == CV_8U );
187
188 uniranges.resize( dims*2 );
189 for( i = 0; i < dims; i++ )
190 {
191 uniranges[i*2] = histSize[i]/256.;
192 uniranges[i*2+1] = 0;
193 }
194 }
195 else if( uniform )
196 {
197 uniranges.resize( dims*2 );
198 for( i = 0; i < dims; i++ )
199 {
200 CV_Assert( ranges[i] && ranges[i][0] < ranges[i][1] );
201 double low = ranges[i][0], high = ranges[i][1];
202 double t = histSize[i]/(high - low);
203 uniranges[i*2] = t;
204 uniranges[i*2+1] = -t*low;
205#if 0 // This should be true by math, but it is not accurate numerically
206 CV_Assert(cvFloor(low * uniranges[i*2] + uniranges[i*2+1]) == 0);
207 CV_Assert((high * uniranges[i*2] + uniranges[i*2+1]) < histSize[i]);
208#endif
209 }
210 }
211 else
212 {
213 for( i = 0; i < dims; i++ )
214 {
215 size_t n = histSize[i];
216 for(size_t k = 0; k < n; k++ )
217 CV_Assert( ranges[i][k] < ranges[i][k+1] );
218 }
219 }
220}

225template<typename T> static void
226calcHist_( std::vector<uchar*>& _ptrs, const std::vector<int>& _deltas,
227 Size imsize, Mat& hist, int dims, const float** _ranges,
228 const double* _uniranges, bool uniform )
229{
230 T** ptrs = (T**)&_ptrs[0];
231 const int* deltas = &_deltas[0];
232 uchar* H = hist.ptr();
233 int i, x;
234 const uchar* mask = _ptrs[dims];
235 int mstep = _deltas[dims*2 + 1];
236 int size[CV_MAX_DIM];
237 size_t hstep[CV_MAX_DIM];
238
239 for( i = 0; i < dims; i++ )
240 {
241 size[i] = hist.size[i];
242 hstep[i] = hist.step[i];
243 }
244
245 if( uniform )
246 {
247 const double* uniranges = &_uniranges[0];
248
249 if( dims == 1 )
250 {
251 double a = uniranges[0], b = uniranges[1];
252 int sz = size[0], d0 = deltas[0], step0 = deltas[1];
253 const T* p0 = (const T*)ptrs[0];
254
255 double v0_lo = _ranges[0][0];
256 double v0_hi = _ranges[0][1];
257
258 for( ; imsize.height--; p0 += step0, mask += mstep )
259 {
260 if( !mask )
261 for( x = 0; x < imsize.width; x++, p0 += d0 )
262 {
263 double v0 = (double)*p0;
264 int idx = cvFloor(v0*a + b);
265 if (v0 < v0_lo || v0 >= v0_hi)
266 continue;
267 idx = CV_CLAMP_INT(idx, 0, sz - 1);
268 CV_DbgAssert((unsigned)idx < (unsigned)sz);
269 ((int*)H)[idx]++;
270 }
271 else
272 for( x = 0; x < imsize.width; x++, p0 += d0 )
273 if( mask[x] )
274 {
275 double v0 = (double)*p0;
276 int idx = cvFloor(v0*a + b);
277 if (v0 < v0_lo || v0 >= v0_hi)
278 continue;
279 idx = CV_CLAMP_INT(idx, 0, sz - 1);
280 CV_DbgAssert((unsigned)idx < (unsigned)sz);
281 ((int*)H)[idx]++;
282 }
283 }
284 return;
285 }
286 else if( dims == 2 )
287 {
288 double a0 = uniranges[0], b0 = uniranges[1], a1 = uniranges[2], b1 = uniranges[3];
289 int sz0 = size[0], sz1 = size[1];
290 int d0 = deltas[0], step0 = deltas[1],
291 d1 = deltas[2], step1 = deltas[3];
292 size_t hstep0 = hstep[0];
293 const T* p0 = (const T*)ptrs[0];
294 const T* p1 = (const T*)ptrs[1];
295
296 double v0_lo = _ranges[0][0];
297 double v0_hi = _ranges[0][1];
298 double v1_lo = _ranges[1][0];
299 double v1_hi = _ranges[1][1];
300
301 for( ; imsize.height--; p0 += step0, p1 += step1, mask += mstep )
302 {
303 if( !mask )
304 for( x = 0; x < imsize.width; x++, p0 += d0, p1 += d1 )
305 {
306 double v0 = (double)*p0;
307 double v1 = (double)*p1;
308 int idx0 = cvFloor(v0*a0 + b0);
309 int idx1 = cvFloor(v1*a1 + b1);
310 if (v0 < v0_lo || v0 >= v0_hi)
311 continue;
312 if (v1 < v1_lo || v1 >= v1_hi)
313 continue;
314 idx0 = CV_CLAMP_INT(idx0, 0, sz0 - 1);
315 idx1 = CV_CLAMP_INT(idx1, 0, sz1 - 1);
316 CV_DbgAssert((unsigned)idx0 < (unsigned)sz0 && (unsigned)idx1 < (unsigned)sz1);
317 ((int*)(H + hstep0*idx0))[idx1]++;
318 }
319 else
320 for( x = 0; x < imsize.width; x++, p0 += d0, p1 += d1 )
321 if( mask[x] )
322 {
323 double v0 = (double)*p0;
324 double v1 = (double)*p1;
325 int idx0 = cvFloor(v0*a0 + b0);
326 int idx1 = cvFloor(v1*a1 + b1);
327 if (v0 < v0_lo || v0 >= v0_hi)
328 continue;
329 if (v1 < v1_lo || v1 >= v1_hi)
330 continue;
331 idx0 = CV_CLAMP_INT(idx0, 0, sz0 - 1);
332 idx1 = CV_CLAMP_INT(idx1, 0, sz1 - 1);
333 CV_DbgAssert((unsigned)idx0 < (unsigned)sz0 && (unsigned)idx1 < (unsigned)sz1);
334 ((int*)(H + hstep0*idx0))[idx1]++;
335 }
336 }
337 return;
338 }
339 else if( dims == 3 )
340 {
341 double a0 = uniranges[0], b0 = uniranges[1],
342 a1 = uniranges[2], b1 = uniranges[3],
343 a2 = uniranges[4], b2 = uniranges[5];
344 int sz0 = size[0], sz1 = size[1], sz2 = size[2];
345 int d0 = deltas[0], step0 = deltas[1],
346 d1 = deltas[2], step1 = deltas[3],
347 d2 = deltas[4], step2 = deltas[5];
348 size_t hstep0 = hstep[0], hstep1 = hstep[1];
349 const T* p0 = (const T*)ptrs[0];
350 const T* p1 = (const T*)ptrs[1];
351 const T* p2 = (const T*)ptrs[2];
352
353 double v0_lo = _ranges[0][0];
354 double v0_hi = _ranges[0][1];
355 double v1_lo = _ranges[1][0];
356 double v1_hi = _ranges[1][1];
357 double v2_lo = _ranges[2][0];
358 double v2_hi = _ranges[2][1];
359
360 for( ; imsize.height--; p0 += step0, p1 += step1, p2 += step2, mask += mstep )
361 {
362 if( !mask )
363 for( x = 0; x < imsize.width; x++, p0 += d0, p1 += d1, p2 += d2 )
364 {
365 double v0 = (double)*p0;
366 double v1 = (double)*p1;
367 double v2 = (double)*p2;
368 int idx0 = cvFloor(v0*a0 + b0);
369 int idx1 = cvFloor(v1*a1 + b1);
370 int idx2 = cvFloor(v2*a2 + b2);
371 if (v0 < v0_lo || v0 >= v0_hi)
372 continue;
373 if (v1 < v1_lo || v1 >= v1_hi)
374 continue;
375 if (v2 < v2_lo || v2 >= v2_hi)
376 continue;
377 idx0 = CV_CLAMP_INT(idx0, 0, sz0 - 1);
378 idx1 = CV_CLAMP_INT(idx1, 0, sz1 - 1);
379 idx2 = CV_CLAMP_INT(idx2, 0, sz2 - 1);
380 CV_DbgAssert(
381 (unsigned)idx0 < (unsigned)sz0 &&
382 (unsigned)idx1 < (unsigned)sz1 &&
383 (unsigned)idx2 < (unsigned)sz2);
384 ((int*)(H + hstep0*idx0 + hstep1*idx1))[idx2]++;
385 }
386 else
387 for( x = 0; x < imsize.width; x++, p0 += d0, p1 += d1, p2 += d2 )
388 if( mask[x] )
389 {
390 double v0 = (double)*p0;
391 double v1 = (double)*p1;
392 double v2 = (double)*p2;
393 int idx0 = cvFloor(v0*a0 + b0);
394 int idx1 = cvFloor(v1*a1 + b1);
395 int idx2 = cvFloor(v2*a2 + b2);
396 if (v0 < v0_lo || v0 >= v0_hi)
397 continue;
398 if (v1 < v1_lo || v1 >= v1_hi)
399 continue;
400 if (v2 < v2_lo || v2 >= v2_hi)
401 continue;
402 idx0 = CV_CLAMP_INT(idx0, 0, sz0 - 1);
403 idx1 = CV_CLAMP_INT(idx1, 0, sz1 - 1);
404 idx2 = CV_CLAMP_INT(idx2, 0, sz2 - 1);
405 CV_DbgAssert(
406 (unsigned)idx0 < (unsigned)sz0 &&
407 (unsigned)idx1 < (unsigned)sz1 &&
408 (unsigned)idx2 < (unsigned)sz2);
409 ((int*)(H + hstep0*idx0 + hstep1*idx1))[idx2]++;
410 }
411 }
412 }
413 else
414 {
415 for( ; imsize.height--; mask += mstep )
416 {
417 if( !mask )
418 for( x = 0; x < imsize.width; x++ )
419 {
420 uchar* Hptr = H;
421 for( i = 0; i < dims; i++ )
422 {
423 double v_lo = _ranges[i][0];
424 double v_hi = _ranges[i][1];
425 double v = *ptrs[i];
426 if (v < v_lo || v >= v_hi)
427 break;
428 int idx = cvFloor(v*uniranges[i*2] + uniranges[i*2+1]);
429 idx = CV_CLAMP_INT(idx, 0, size[i] - 1);
430 CV_DbgAssert((unsigned)idx < (unsigned)size[i]);
431 ptrs[i] += deltas[i*2];
432 Hptr += idx*hstep[i];
433 }
434
435 if( i == dims )
436 ++*((int*)Hptr);
437 else
438 for( ; i < dims; i++ )
439 ptrs[i] += deltas[i*2];
440 }
441 else
442 for( x = 0; x < imsize.width; x++ )
443 {
444 uchar* Hptr = H;
445 i = 0;
446 if( mask[x] )
447 for( ; i < dims; i++ )
448 {
449 double v_lo = _ranges[i][0];
450 double v_hi = _ranges[i][1];
451 double v = *ptrs[i];
452 if (v < v_lo || v >= v_hi)
453 break;
454 int idx = cvFloor(v*uniranges[i*2] + uniranges[i*2+1]);
455 idx = CV_CLAMP_INT(idx, 0, size[i] - 1);
456 CV_DbgAssert((unsigned)idx < (unsigned)size[i]);
457 ptrs[i] += deltas[i*2];
458 Hptr += idx*hstep[i];
459 }
460
461 if( i == dims )
462 ++*((int*)Hptr);
463 else
464 for( ; i < dims; i++ )
465 ptrs[i] += deltas[i*2];
466 }
467 for( i = 0; i < dims; i++ )
468 ptrs[i] += deltas[i*2 + 1];
469 }
470 }
471 }
472 else if (_ranges)
473 {
474 // non-uniform histogram
475 const float* ranges[CV_MAX_DIM];
476 for( i = 0; i < dims; i++ )
477 ranges[i] = &_ranges[i][0];
478
479 for( ; imsize.height--; mask += mstep )
480 {
481 for( x = 0; x < imsize.width; x++ )
482 {
483 uchar* Hptr = H;
484 i = 0;
485
486 if( !mask || mask[x] )
487 for( ; i < dims; i++ )
488 {
489 float v = (float)*ptrs[i];
490 const float* R = ranges[i];
491 int idx = -1, sz = size[i];
492
493 while( v >= R[idx+1] && ++idx < sz )
494 ; // nop
495
496 if( (unsigned)idx >= (unsigned)sz )
497 break;
498
499 ptrs[i] += deltas[i*2];
500 Hptr += idx*hstep[i];
501 }
502
503 if( i == dims )
504 ++*((int*)Hptr);
505 else
506 for( ; i < dims; i++ )
507 ptrs[i] += deltas[i*2];
508 }
509
510 for( i = 0; i < dims; i++ )
511 ptrs[i] += deltas[i*2 + 1];
512 }
513 }
514 else
515 {
516 CV_Error(Error::StsBadArg, "Either ranges, either uniform ranges should be provided");
517 }
518}

521static void
522calcHist_8u( std::vector<uchar*>& _ptrs, const std::vector<int>& _deltas,
523 Size imsize, Mat& hist, int dims, const float** _ranges,
524 const double* _uniranges, bool uniform )
525{
526 uchar** ptrs = &_ptrs[0];
527 const int* deltas = &_deltas[0];
528 uchar* H = hist.ptr();
529 int x;
530 const uchar* mask = _ptrs[dims];
531 int mstep = _deltas[dims*2 + 1];
532 std::vector<size_t> _tab;
533
534 calcHistLookupTables_8u( hist, SparseMat(), dims, _ranges, _uniranges, uniform, false, _tab );
535 const size_t* tab = &_tab[0];
536
537 if( dims == 1 )
538 {
539 int d0 = deltas[0], step0 = deltas[1];
540 int matH[256] = { 0, };
541 const uchar* p0 = (const uchar*)ptrs[0];
542
543 for( ; imsize.height--; p0 += step0, mask += mstep )
544 {
545 if( !mask )
546 {
547 if( d0 == 1 )
548 {
549 for( x = 0; x <= imsize.width - 4; x += 4 )
550 {
551 int t0 = p0[x], t1 = p0[x+1];
552 matH[t0]++; matH[t1]++;
553 t0 = p0[x+2]; t1 = p0[x+3];
554 matH[t0]++; matH[t1]++;
555 }
556 p0 += x;
557 }
558 else
559 for( x = 0; x <= imsize.width - 4; x += 4 )
560 {
561 int t0 = p0[0], t1 = p0[d0];
562 matH[t0]++; matH[t1]++;
563 p0 += d0*2;
564 t0 = p0[0]; t1 = p0[d0];
565 matH[t0]++; matH[t1]++;
566 p0 += d0*2;
567 }
568
569 for( ; x < imsize.width; x++, p0 += d0 )
570 matH[*p0]++;
571 }
572 else
573 for( x = 0; x < imsize.width; x++, p0 += d0 )
574 if( mask[x] )
575 matH[*p0]++;
576 }
577
578 for(int i = 0; i < 256; i++ )
579 {
580 size_t hidx = tab[i];
581 if( hidx < OUT_OF_RANGE )
582 *(int*)(H + hidx) += matH[i];
583 }
584 }
585 else if( dims == 2 )
586 {
587 int d0 = deltas[0], step0 = deltas[1],
588 d1 = deltas[2], step1 = deltas[3];
589 const uchar* p0 = (const uchar*)ptrs[0];
590 const uchar* p1 = (const uchar*)ptrs[1];
591
592 for( ; imsize.height--; p0 += step0, p1 += step1, mask += mstep )
593 {
594 if( !mask )
595 for( x = 0; x < imsize.width; x++, p0 += d0, p1 += d1 )
596 {
597 size_t idx = tab[*p0] + tab[*p1 + 256];
598 if( idx < OUT_OF_RANGE )
599 ++*(int*)(H + idx);
600 }
601 else
602 for( x = 0; x < imsize.width; x++, p0 += d0, p1 += d1 )
603 {
604 size_t idx;
605 if( mask[x] && (idx = tab[*p0] + tab[*p1 + 256]) < OUT_OF_RANGE )
606 ++*(int*)(H + idx);
607 }
608 }
609 }
610 else if( dims == 3 )
611 {
612 int d0 = deltas[0], step0 = deltas[1],
613 d1 = deltas[2], step1 = deltas[3],
614 d2 = deltas[4], step2 = deltas[5];
615
616 const uchar* p0 = (const uchar*)ptrs[0];
617 const uchar* p1 = (const uchar*)ptrs[1];
618 const uchar* p2 = (const uchar*)ptrs[2];
619
620 for( ; imsize.height--; p0 += step0, p1 += step1, p2 += step2, mask += mstep )
621 {
622 if( !mask )
623 for( x = 0; x < imsize.width; x++, p0 += d0, p1 += d1, p2 += d2 )
624 {
625 size_t idx = tab[*p0] + tab[*p1 + 256] + tab[*p2 + 512];
626 if( idx < OUT_OF_RANGE )
627 ++*(int*)(H + idx);
628 }
629 else
630 for( x = 0; x < imsize.width; x++, p0 += d0, p1 += d1, p2 += d2 )
631 {
632 size_t idx;
633 if( mask[x] && (idx = tab[*p0] + tab[*p1 + 256] + tab[*p2 + 512]) < OUT_OF_RANGE )
634 ++*(int*)(H + idx);
635 }
636 }
637 }
638 else
639 {
640 for( ; imsize.height--; mask += mstep )
641 {
642 if( !mask )
643 for( x = 0; x < imsize.width; x++ )
644 {
645 uchar* Hptr = H;
646 int i = 0;
647 for( ; i < dims; i++ )

954template<typename T> static void
955calcSparseHist_( std::vector<uchar*>& _ptrs, const std::vector<int>& _deltas,
956 Size imsize, SparseMat& hist, int dims, const float** _ranges,
957 const double* _uniranges, bool uniform )
958{
959 T** ptrs = (T**)&_ptrs[0];
960 const int* deltas = &_deltas[0];
961 int i, x;
962 const uchar* mask = _ptrs[dims];
963 int mstep = _deltas[dims*2 + 1];
964 const int* size = hist.hdr->size;
965 int idx[CV_MAX_DIM];
966
967 if( uniform )
968 {
969 const double* uniranges = &_uniranges[0];
970
971 for( ; imsize.height--; mask += mstep )
972 {
973 for( x = 0; x < imsize.width; x++ )
974 {
975 i = 0;
976 if( !mask || mask[x] )
977 for( ; i < dims; i++ )
978 {
979 idx[i] = cvFloor(*ptrs[i]*uniranges[i*2] + uniranges[i*2+1]);
980 if( (unsigned)idx[i] >= (unsigned)size[i] )
981 break;
982 ptrs[i] += deltas[i*2];
983 }
984
985 if( i == dims )
986 ++*(int*)hist.ptr(idx, true);
987 else
988 for( ; i < dims; i++ )
989 ptrs[i] += deltas[i*2];
990 }
991 for( i = 0; i < dims; i++ )
992 ptrs[i] += deltas[i*2 + 1];
993 }
994 }
995 else if (_ranges)
996 {
997 // non-uniform histogram
998 const float* ranges[CV_MAX_DIM];
999 for( i = 0; i < dims; i++ )
1000 ranges[i] = &_ranges[i][0];
1001
1002 for( ; imsize.height--; mask += mstep )
1003 {
1004 for( x = 0; x < imsize.width; x++ )
1005 {
1006 i = 0;
1007
1008 if( !mask || mask[x] )
1009 for( ; i < dims; i++ )
1010 {
1011 float v = (float)*ptrs[i];
1012 const float* R = ranges[i];
1013 int j = -1, sz = size[i];
1014
1015 while( v >= R[j+1] && ++j < sz )
1016 ; // nop
1017
1018 if( (unsigned)j >= (unsigned)sz )
1019 break;
1020 ptrs[i] += deltas[i*2];
1021 idx[i] = j;
1022 }
1023
1024 if( i == dims )
1025 ++*(int*)hist.ptr(idx, true);
1026 else
1027 for( ; i < dims; i++ )
1028 ptrs[i] += deltas[i*2];
1029 }
1030
1031 for( i = 0; i < dims; i++ )
1032 ptrs[i] += deltas[i*2 + 1];
1033 }
1034 }
1035 else
1036 {
1037 CV_Error(Error::StsBadArg, "Either ranges, either uniform ranges should be provided");
1038 }

1285 for( ; imsize.height--; p0 += step0, bproj += bpstep )
1286 {
1287 for( x = 0; x < imsize.width; x++, p0 += d0 )
1288 {
1289 int idx = cvFloor(*p0*a + b);
1290 bproj[x] = (unsigned)idx < (unsigned)sz ? saturate_cast<BT>(((const float*)H)[idx]*scale) : 0;
1291 }
1292 }
1293 }
1294 else if( dims == 2 )
1295 {
1296 double a0 = uniranges[0], b0 = uniranges[1],
1297 a1 = uniranges[2], b1 = uniranges[3];
1298 int sz0 = size[0], sz1 = size[1];
1299 int d0 = deltas[0], step0 = deltas[1],
1300 d1 = deltas[2], step1 = deltas[3];
1301 size_t hstep0 = hstep[0];
1302 const T* p0 = (const T*)ptrs[0];
1303 const T* p1 = (const T*)ptrs[1];
1304
1305 for( ; imsize.height--; p0 += step0, p1 += step1, bproj += bpstep )
1306 {
1307 for( x = 0; x < imsize.width; x++, p0 += d0, p1 += d1 )
1308 {
1309 int idx0 = cvFloor(*p0*a0 + b0);
1310 int idx1 = cvFloor(*p1*a1 + b1);
1311 bproj[x] = (unsigned)idx0 < (unsigned)sz0 &&
1312 (unsigned)idx1 < (unsigned)sz1 ?
1313 saturate_cast<BT>(((const float*)(H + hstep0*idx0))[idx1]*scale) : 0;
1314 }
1315 }
1316 }
1317 else if( dims == 3 )
1318 {
1319 double a0 = uniranges[0], b0 = uniranges[1],
1320 a1 = uniranges[2], b1 = uniranges[3],
1321 a2 = uniranges[4], b2 = uniranges[5];
1322 int sz0 = size[0], sz1 = size[1], sz2 = size[2];
1323 int d0 = deltas[0], step0 = deltas[1],
1324 d1 = deltas[2], step1 = deltas[3],
1325 d2 = deltas[4], step2 = deltas[5];
1326 size_t hstep0 = hstep[0], hstep1 = hstep[1];
1327 const T* p0 = (const T*)ptrs[0];
1328 const T* p1 = (const T*)ptrs[1];
1329 const T* p2 = (const T*)ptrs[2];
1330
1331 for( ; imsize.height--; p0 += step0, p1 += step1, p2 += step2, bproj += bpstep )
1332 {
1333 for( x = 0; x < imsize.width; x++, p0 += d0, p1 += d1, p2 += d2 )
1334 {
1335 int idx0 = cvFloor(*p0*a0 + b0);
1336 int idx1 = cvFloor(*p1*a1 + b1);
1337 int idx2 = cvFloor(*p2*a2 + b2);
1338 bproj[x] = (unsigned)idx0 < (unsigned)sz0 &&
1339 (unsigned)idx1 < (unsigned)sz1 &&
1340 (unsigned)idx2 < (unsigned)sz2 ?
1341 saturate_cast<BT>(((const float*)(H + hstep0*idx0 + hstep1*idx1))[idx2]*scale) : 0;
1342 }
1343 }
1344 }
1345 else
1346 {
1347 for( ; imsize.height--; bproj += bpstep )
1348 {
1349 for( x = 0; x < imsize.width; x++ )
1350 {
1351 const uchar* Hptr = H;
1352 for( i = 0; i < dims; i++ )
1353 {
1354 int idx = cvFloor(*ptrs[i]*uniranges[i*2] + uniranges[i*2+1]);
1355 if( (unsigned)idx >= (unsigned)size[i] || (_ranges && *ptrs[i] >= _ranges[i][1]))
1356 break;
1357 ptrs[i] += deltas[i*2];
1358 Hptr += idx*hstep[i];
1359 }
1360
1361 if( i == dims )
1362 bproj[x] = saturate_cast<BT>(*(const float*)Hptr*scale);
1363 else
1364 {
1365 bproj[x] = 0;
1366 for( ; i < dims; i++ )
1367 ptrs[i] += deltas[i*2];
1368 }
1369 }
1370 for( i = 0; i < dims; i++ )
1371 ptrs[i] += deltas[i*2 + 1];
1372 }
1373 }
1374 }
1375 else if (_ranges)
1376 {
1377 // non-uniform histogram
1378 const float* ranges[CV_MAX_DIM];
1379 for( i = 0; i < dims; i++ )
1380 ranges[i] = &_ranges[i][0];
1381
1382 for( ; imsize.height--; bproj += bpstep )
1383 {
1384 for( x = 0; x < imsize.width; x++ )
1385 {
1386 const uchar* Hptr = H;
1387 for( i = 0; i < dims; i++ )
1388 {
1389 float v = (float)*ptrs[i];
1390 const float* R = ranges[i];
1391 int idx = -1, sz = size[i];
1392
1393 while( v >= R[idx+1] && ++idx < sz )
1394 ; // nop
1395
1396 if( (unsigned)idx >= (unsigned)sz )
1397 break;
1398
1399 ptrs[i] += deltas[i*2];
1400 Hptr += idx*hstep[i];
1401 }
1402
1403 if( i == dims )
1404 bproj[x] = saturate_cast<BT>(*(const float*)Hptr*scale);
1405 else
1406 {
1407 bproj[x] = 0;
1408 for( ; i < dims; i++ )
1409 ptrs[i] += deltas[i*2];
1410 }
1411 }
1412
1413 for( i = 0; i < dims; i++ )
1414 ptrs[i] += deltas[i*2 + 1];
1415 }
1416 }
1417 else
1418 {
1419 CV_Error(Error::StsBadArg, "Either ranges, either uniform ranges should be provided");
1420 }

1424static void
1425calcBackProj_8u( std::vector<uchar*>& _ptrs, const std::vector<int>& _deltas,
1426 Size imsize, const Mat& hist, int dims, const float** _ranges,
1427 const double* _uniranges, float scale, bool uniform )
1428{
1429 uchar** ptrs = &_ptrs[0];
1430 const int* deltas = &_deltas[0];
1431 const uchar* H = hist.ptr();
1432 int i, x;
1433 uchar* bproj = _ptrs[dims];
1434 int bpstep = _deltas[dims*2 + 1];
1435 std::vector<size_t> _tab;
1436
1437 calcHistLookupTables_8u( hist, SparseMat(), dims, _ranges, _uniranges, uniform, false, _tab );
1438 const size_t* tab = &_tab[0];
1439
1440 if( dims == 1 )
1441 {
1442 int d0 = deltas[0], step0 = deltas[1];
1443 uchar matH[256] = {0};
1444 const uchar* p0 = (const uchar*)ptrs[0];
1445
1446 for( i = 0; i < 256; i++ )
1447 {
1448 size_t hidx = tab[i];
1449 if( hidx < OUT_OF_RANGE )
1450 matH[i] = saturate_cast<uchar>(*(float*)(H + hidx)*scale);
1451 }
1452
1453 for( ; imsize.height--; p0 += step0, bproj += bpstep )
1454 {
1455 if( d0 == 1 )
1456 {
1457 for( x = 0; x <= imsize.width - 4; x += 4 )
1458 {
1459 uchar t0 = matH[p0[x]], t1 = matH[p0[x+1]];
1460 bproj[x] = t0; bproj[x+1] = t1;
1461 t0 = matH[p0[x+2]]; t1 = matH[p0[x+3]];
1462 bproj[x+2] = t0; bproj[x+3] = t1;
1463 }
1464 p0 += x;
1465 }
1466 else
1467 for( x = 0; x <= imsize.width - 4; x += 4 )
1468 {
1469 uchar t0 = matH[p0[0]], t1 = matH[p0[d0]];
1470 bproj[x] = t0; bproj[x+1] = t1;
1471 p0 += d0*2;
1472 t0 = matH[p0[0]]; t1 = matH[p0[d0]];
1473 bproj[x+2] = t0; bproj[x+3] = t1;
1474 p0 += d0*2;
1475 }
1476
1477 for( ; x < imsize.width; x++, p0 += d0 )
1478 bproj[x] = matH[*p0];
1479 }
1480 }
1481 else if( dims == 2 )
1482 {
1483 int d0 = deltas[0], step0 = deltas[1],
1484 d1 = deltas[2], step1 = deltas[3];
1485 const uchar* p0 = (const uchar*)ptrs[0];
1486 const uchar* p1 = (const uchar*)ptrs[1];
1487
1488 for( ; imsize.height--; p0 += step0, p1 += step1, bproj += bpstep )
1489 {
1490 for( x = 0; x < imsize.width; x++, p0 += d0, p1 += d1 )
1491 {
1492 size_t idx = tab[*p0] + tab[*p1 + 256];
1493 bproj[x] = idx < OUT_OF_RANGE ? saturate_cast<uchar>(*(const float*)(H + idx)*scale) : 0;
1494 }
1495 }
1496 }
1497 else if( dims == 3 )
1498 {
1499 int d0 = deltas[0], step0 = deltas[1],
1500 d1 = deltas[2], step1 = deltas[3],
1501 d2 = deltas[4], step2 = deltas[5];
1502 const uchar* p0 = (const uchar*)ptrs[0];
1503 const uchar* p1 = (const uchar*)ptrs[1];
1504 const uchar* p2 = (const uchar*)ptrs[2];
1505
1506 for( ; imsize.height--; p0 += step0, p1 += step1, p2 += step2, bproj += bpstep )
1507 {
1508 for( x = 0; x < imsize.width; x++, p0 += d0, p1 += d1, p2 += d2 )
1509 {
1510 size_t idx = tab[*p0] + tab[*p1 + 256] + tab[*p2 + 512];
1511 bproj[x] = idx < OUT_OF_RANGE ? saturate_cast<uchar>(*(const float*)(H + idx)*scale) : 0;
1512 }
1513 }
1514 }
1515 else
1516 {
1517 for( ; imsize.height--; bproj += bpstep )
1518 {
1519 for( x = 0; x < imsize.width; x++ )
1520 {
1521 const uchar* Hptr = H;
1522 for( i = 0; i < dims; i++ )
1523 {
1524 size_t idx = tab[*ptrs[i] + i*256];
1525 if( idx >= OUT_OF_RANGE )
1526 break;
1527 ptrs[i] += deltas[i*2];
1528 Hptr += idx;
1529 }
1530
1531 if( i == dims )
1532 bproj[x] = saturate_cast<uchar>(*(const float*)Hptr*scale);
1533 else
1534 {
1535 bproj[x] = 0;
1536 for( ; i < dims; i++ )
1537 ptrs[i] += deltas[i*2];
1538 }
1539 }
1540 for( i = 0; i < dims; i++ )
1541 ptrs[i] += deltas[i*2 + 1];
1542 }
1543 }
1544}

1988 const float* h2 = it.planes[1].ptr<float>();
1989 const int len = it.planes[0].rows*it.planes[0].cols*H1.channels();
1990 j = 0;
1991
1992 if( (method == cv::HISTCMP_CHISQR) || (method == cv::HISTCMP_CHISQR_ALT))
1993 {
1994#if CV_SIMD_64F || CV_SIMD_SCALABLE_64F
1995 v_float64 v_eps = vx_setall_f64(DBL_EPSILON);
1996 v_float64 v_one = vx_setall_f64(1.f);
1997 v_float64 v_zero = vx_setzero_f64();
1998 v_float64 v_res = vx_setzero_f64();

2000 {
2001 v_float32 v_h1 = vx_load(h1 + j), v_h2 = vx_load(h2 + j);

2003 v_float64 v_h2_l = v_cvt_f64(v_h2), v_h2_h = v_cvt_f64_high(v_h2);

2012 v_b_l = v_h1_l;
2013 v_b_h = v_h1_h;

2037 double b = (method == cv::HISTCMP_CHISQR) ? h1[j] : h1[j] + h2[j];
2038 if( fabs(b) > DBL_EPSILON )
2039 result += a*a/b;
2040 }
2041 }
2042 else if( method == cv::HISTCMP_CORREL )
2043 {
2044#if (CV_SIMD_64F || CV_SIMD_SCALABLE_64F)
2045 v_float64 v_s1 = vx_setzero_f64();
2046 v_float64 v_s2 = vx_setzero_f64();
2047 v_float64 v_s11 = vx_setzero_f64();
2048 v_float64 v_s12 = vx_setzero_f64();

2053 v_float32 v_b = vx_load(h2 + j);

2057 v_float64 v_bd = v_cvt_f64(v_b);

2059 v_s11 = v_muladd(v_ad, v_ad, v_s11);
2060 v_s22 = v_muladd(v_bd, v_bd, v_s22);
2061 v_s1 = v_add(v_s1, v_ad);
2062 v_s2 = v_add(v_s2, v_bd);
2063
2064 // 2-3
2065 v_ad = v_cvt_f64_high(v_a);
2066 v_bd = v_cvt_f64_high(v_b);

2068 v_s11 = v_muladd(v_ad, v_ad, v_s11);
2069 v_s22 = v_muladd(v_bd, v_bd, v_s22);
2070 v_s1 = v_add(v_s1, v_ad);
2071 v_s2 = v_add(v_s2, v_bd);
2072 }
2073 s12 += v_reduce_sum(v_s12);
2074 s11 += v_reduce_sum(v_s11);
2075 s22 += v_reduce_sum(v_s22);
2076 s1 += v_reduce_sum(v_s1);
2077 s2 += v_reduce_sum(v_s2);
2078#elif CV_SIMD && 0 //Disable vectorization for cv::HISTCMP_CORREL if f64 is unsupported due to low precision
2079 v_float32 v_s1 = vx_setzero_f32();
2080 v_float32 v_s2 = vx_setzero_f32();
2081 v_float32 v_s11 = vx_setzero_f32();
2082 v_float32 v_s12 = vx_setzero_f32();
2083 v_float32 v_s22 = vx_setzero_f32();
2084 for (; j <= len - VTraits<v_float32>::vlanes(); j += VTraits<v_float32>::vlanes())
2085 {
2086 v_float32 v_a = vx_load(h1 + j);
2087 v_float32 v_b = vx_load(h2 + j);
2088
2089 v_s12 = v_muladd(v_a, v_b, v_s12);
2090 v_s11 = v_muladd(v_a, v_a, v_s11);
2091 v_s22 = v_muladd(v_b, v_b, v_s22);
2092 v_s1 += v_a;

2094 }
2095 s12 += v_reduce_sum(v_s12);
2096 s11 += v_reduce_sum(v_s11);
2097 s22 += v_reduce_sum(v_s22);
2098 s1 += v_reduce_sum(v_s1);
2099 s2 += v_reduce_sum(v_s2);
2100#endif

2102 {

2104 double b = h2[j];
2105
2106 s12 += a*b;
2107 s1 += a;
2108 s11 += a*a;
2109 s2 += b;
2110 s22 += b*b;
2111 }
2112 }

2114 {
2115#if (CV_SIMD_64F || CV_SIMD_SCALABLE_64F)

2123#elif CV_SIMD && 0 // Disable vectorization for CV_COMP_INTERSECT if f64 is unsupported due to low precision
2124 // See https://github.com/opencv/opencv/issues/24757
2125 v_float32 v_result = vx_setzero_f32();
2126 for (; j <= len - VTraits<v_float32>::vlanes(); j += VTraits<v_float32>::vlanes())
2127 {
2128 v_float32 v_src = v_min(vx_load(h1 + j), vx_load(h2 + j));
2129 v_result = v_add(v_result, v_src);
2130 }
2131 result += v_reduce_sum(v_result);
2132#endif
2133 for( ; j < len; j++ )
2134 result += std::min(h1[j], h2[j]);
2135 }
2136 else if( method == cv::HISTCMP_BHATTACHARYYA )
2137 {
2138#if (CV_SIMD_64F || CV_SIMD_SCALABLE_64F)
2139 v_float64 v_s1 = vx_setzero_f64();
2140 v_float64 v_s2 = vx_setzero_f64();
2141 v_float64 v_result = vx_setzero_f64();
2142 for ( ; j <= len - VTraits<v_float32>::vlanes(); j += VTraits<v_float32>::vlanes())
2143 {
2144 v_float32 v_a = vx_load(h1 + j);
2145 v_float32 v_b = vx_load(h2 + j);
2146
2147 v_float64 v_ad = v_cvt_f64(v_a);
2148 v_float64 v_bd = v_cvt_f64(v_b);
2149 v_s1 = v_add(v_s1, v_ad);
2150 v_s2 = v_add(v_s2, v_bd);
2151 v_result = v_add(v_result, v_sqrt(v_mul(v_ad, v_bd)));
2152
2153 v_ad = v_cvt_f64_high(v_a);
2154 v_bd = v_cvt_f64_high(v_b);
2155 v_s1 = v_add(v_s1, v_ad);
2156 v_s2 = v_add(v_s2, v_bd);
2157 v_result = v_add(v_result, v_sqrt(v_mul(v_ad, v_bd)));
2158 }
2159 s1 += v_reduce_sum(v_s1);
2160 s2 += v_reduce_sum(v_s2);
2161 result += v_reduce_sum(v_result);
2162#elif CV_SIMD && 0 //Disable vectorization for cv::HISTCMP_BHATTACHARYYA if f64 is unsupported due to low precision
2163 v_float32 v_s1 = vx_setzero_f32();
2164 v_float32 v_s2 = vx_setzero_f32();
2165 v_float32 v_result = vx_setzero_f32();
2166 for (; j <= len - VTraits<v_float32>::vlanes(); j += VTraits<v_float32>::vlanes())
2167 {
2168 v_float32 v_a = vx_load(h1 + j);
2169 v_float32 v_b = vx_load(h2 + j);
2170 v_s1 += v_a;
2171 v_s2 += v_b;
2172 v_result += v_sqrt(v_a * v_b);
2173 }
2174 s1 += v_reduce_sum(v_s1);
2175 s2 += v_reduce_sum(v_s2);
2176 result += v_reduce_sum(v_result);
2177#endif
2178 for( ; j < len; j++ )
2179 {
2180 double a = h1[j];
2181 double b = h2[j];
2182 result += std::sqrt(a*b);
2183 s1 += a;
2184 s2 += b;
2185 }
2186 }
2187 else if( method == cv::HISTCMP_KL_DIV )
2188 {
2189 for( ; j < len; j++ )
2190 {
2191 double p = h1[j];
2192 double q = h2[j];
2193 if( fabs(p) <= DBL_EPSILON ) {
2194 continue;
2195 }
2196 if( fabs(q) <= DBL_EPSILON ) {
2197 q = 1e-10;
2198 }
2199 result += p * std::log( p / q );
2200 }
2201 }
2202 else
2203 CV_Error( cv::Error::StsBadArg, "Unknown comparison method" );
2204 }
2205
2206 if( method == cv::HISTCMP_CHISQR_ALT )
2207 result *= 2;
2208 else if( method == cv::HISTCMP_CORREL )
2209 {
2210 size_t total = H1.total();
2211 double scale = 1./total;
2212 double num = s12 - s1*s2*scale;
2213 double denom2 = (s11 - s1*s1*scale)*(s22 - s2*s2*scale);
2214 result = std::abs(denom2) > DBL_EPSILON ? num/std::sqrt(denom2) : 1.;
2215 }
2216 else if( method == cv::HISTCMP_BHATTACHARYYA )
2217 {
2218 s1 *= s2;
2219 s1 = fabs(s1) > FLT_EPSILON ? 1./std::sqrt(s1) : 1.;
2220 result = std::sqrt(std::max(1. - result*s1, 0.));
2221 }
2222
2223 return result;
2224}
