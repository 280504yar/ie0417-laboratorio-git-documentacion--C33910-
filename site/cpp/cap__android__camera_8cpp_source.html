43template <typename T> struct RangeValue {
44 T min, max;
45
46 RangeValue(T minv = 0, T maxv = 0) : min(minv), max(maxv) {}
47 bool isValid() const { return (min != max); }
48 T percentage(int percent) const {
49 return static_cast<T>(min + ((max - min) * percent) / 100);
50 }
51 T clamp( T value ) const {
52 return (value > max) ? max : ((value < min) ? min : value);
53 }
54};

179 bool grabFrame() CV_OVERRIDE
180 {
181 AImage* img;
182 {
183 std::unique_lock<std::mutex> lock(mtx);
184 media_status_t mStatus = AImageReader_acquireLatestImage(imageReader.get(), &img);
185 if (mStatus != AMEDIA_OK) {
186 if (mStatus == AMEDIA_IMGREADER_NO_BUFFER_AVAILABLE) {
187 // this error is not fatal - we just need to wait for a buffer to become available
188 LOGW("No Buffer Available error occurred - waiting for callback");
189 waitingCapture = true;
190 captureSuccess = false;
191 auto start = std::chrono::system_clock::now();
192 bool captured = condition.wait_for(lock, std::chrono::seconds(
193 CAPTURE_TIMEOUT_SECONDS), [this]{ return captureSuccess; });
194 waitingCapture = false;
195 if (captured) {
196 mStatus = AImageReader_acquireLatestImage(imageReader.get(), &img);
197 // even though an image has been captured we may not be able to acquire it
198 // straight away so we poll every 10ms
199 while (mStatus == AMEDIA_IMGREADER_NO_BUFFER_AVAILABLE &&
200 elapsedTimeFrom(start) < CAPTURE_TIMEOUT_SECONDS) {
201 std::this_thread::sleep_for(std::chrono::milliseconds(CAPTURE_POLL_INTERVAL_MS));
202 mStatus = AImageReader_acquireLatestImage(imageReader.get(), &img);
203 }
204 if (mStatus != AMEDIA_OK) {
205 LOGE("Acquire image failed with error code: %d", mStatus);
206 if (elapsedTimeFrom(start) >= CAPTURE_TIMEOUT_SECONDS) {
207 LOGE("Image acquisition timed out");
208 }
209 return false;
210 }
211 } else {
212 LOGE("Capture failed or callback timed out");
213 return false;
214 }
215 } else {
216 LOGE("Acquire image failed with error code: %d", mStatus);
217 return false;
218 }
219 }
220 }
221 AObjPtr<AImage> image(img, AImage_delete);
222 int32_t srcFormat = -1;
223 AImage_getFormat(image.get(), &srcFormat);
224 if (srcFormat != AIMAGE_FORMAT_YUV_420_888) {
225 LOGE("Incorrect image format");
226 return false;
227 }
228 int32_t srcPlanes = 0;
229 AImage_getNumberOfPlanes(image.get(), &srcPlanes);
230 if (srcPlanes != 3) {
231 LOGE("Incorrect number of planes in image data");
232 return false;
233 }
234 int32_t yStride, uvStride;
235 uint8_t *yPixel, *uPixel, *vPixel;
236 int32_t yLen, uLen, vLen;
237 int32_t uvPixelStride;
238 AImage_getPlaneRowStride(image.get(), 0, &yStride);
239 AImage_getPlaneRowStride(image.get(), 1, &uvStride);
240 AImage_getPlaneData(image.get(), 0, &yPixel, &yLen);
241 AImage_getPlaneData(image.get(), 1, &uPixel, &uLen);
242 AImage_getPlaneData(image.get(), 2, &vPixel, &vLen);
243 AImage_getPlanePixelStride(image.get(), 1, &uvPixelStride);
244 int32_t yBufferLen = yLen;
245
246 if ( (uvPixelStride == 2) && (uPixel == vPixel + 1) &&
247 (yLen == (yStride * (frameHeight - 1)) + frameWidth) &&
248 (uLen == (uvStride * ((frameHeight / 2) - 1)) + frameWidth - 1) &&
249 (uvStride == yStride) && (vLen == uLen) ) {
250 frameStride = yStride;
251 yBufferLen = frameStride * frameHeight;
252 colorFormat = COLOR_FormatYUV420SemiPlanar;
253 if (fourCC == FOURCC_UNKNOWN) {
254 fourCC = FOURCC_NV21;
255 }
256 } else if ( (uvPixelStride == 1) && (uPixel == vPixel + vLen) &&
257 (yLen == frameWidth * frameHeight) &&
258 (uLen == yLen / 4) && (vLen == uLen) ) {
259 colorFormat = COLOR_FormatYUV420Planar;
260 if (fourCC == FOURCC_UNKNOWN) {
261 fourCC = FOURCC_YV12;
262 }
263 } else {
264 colorFormat = COLOR_FormatUnknown;
265 fourCC = FOURCC_UNKNOWN;
266 LOGE("Unsupported format");
267 return false;
268 }
269
270 buffer.clear();
271 buffer.insert(buffer.end(), yPixel, yPixel + yBufferLen);
272 buffer.insert(buffer.end(), vPixel, vPixel + yBufferLen / 2);
273 return true;
274 }

276 bool retrieveFrame(int, OutputArray out) CV_OVERRIDE
277 {
278 if (buffer.empty()) {
279 return false;
280 }
281 if (colorFormat == COLOR_FormatYUV420Planar) {
282 const Mat yuv(frameHeight + frameHeight/2, frameWidth, CV_8UC1, buffer.data());
283 switch (fourCC) {
284 case FOURCC_BGRA:
285 cvtColor(yuv, out, COLOR_YUV2BGRA_YV12);
286 break;
287 case FOURCC_RGBA:
288 cvtColor(yuv, out, COLOR_YUV2RGBA_YV12);
289 break;
290 case FOURCC_BGR:
291 cvtColor(yuv, out, COLOR_YUV2BGR_YV12);
292 break;
293 case FOURCC_RGB:
294 cvtColor(yuv, out, COLOR_YUV2RGB_YV12);
295 break;
296 case FOURCC_GRAY:
297 cvtColor(yuv, out, COLOR_YUV2GRAY_YV12);
298 break;
299 case FOURCC_YV12:
300 yuv.copyTo(out);
301 break;
302 default:
303 LOGE("Unexpected FOURCC value: %d", fourCC);
304 return false;
305 }
306 } else if (colorFormat == COLOR_FormatYUV420SemiPlanar) {
307 const Mat yuv(frameHeight + frameHeight/2, frameWidth, CV_8UC1, buffer.data(), frameStride);
308 switch (fourCC) {
309 case FOURCC_BGRA:
310 cvtColor(yuv, out, COLOR_YUV2BGRA_NV21);
311 break;
312 case FOURCC_RGBA:
313 cvtColor(yuv, out, COLOR_YUV2RGBA_NV21);
314 break;
315 case FOURCC_BGR:
316 cvtColor(yuv, out, COLOR_YUV2BGR_NV21);
317 break;
318 case FOURCC_RGB:
319 cvtColor(yuv, out, COLOR_YUV2RGB_NV21);
320 break;
321 case FOURCC_GRAY:
322 cvtColor(yuv, out, COLOR_YUV2GRAY_NV21);
323 break;
324 case FOURCC_NV21:
325 yuv.copyTo(out);
326 break;
327 default:
328 LOGE("Unexpected FOURCC value: %d", fourCC);
329 return false;
330 }
331 } else {
332 LOGE("Unsupported video format: %d", colorFormat);
333 return false;
334 }
335 return true;
336 }

338 double getProperty(int property_id) const CV_OVERRIDE
339 {
340 switch (property_id) {
341 case CAP_PROP_FRAME_WIDTH:
342 return isOpened() ? frameWidth : desiredWidth;
343 case CAP_PROP_FRAME_HEIGHT:
344 return isOpened() ? frameHeight : desiredHeight;
345 case CAP_PROP_AUTO_EXPOSURE:
346 return (aeMode == ACAMERA_CONTROL_AE_MODE_ON) ? 1 : 0;
347 case CAP_PROP_EXPOSURE:
348 return exposureTime;
349 case CAP_PROP_ISO_SPEED:
350 return sensitivity;
351 case CAP_PROP_FOURCC:
352 return fourCC;
353 case CAP_PROP_ANDROID_DEVICE_TORCH:
354 return (flashMode == ACAMERA_FLASH_MODE_TORCH) ? 1 : 0;
355 case CAP_PROP_ZOOM:
356 return zoomRange.isValid() ? zoomRatio : -1;
357 default:
358 break;
359 }
360 // unknown parameter or value not available
361 return CAP_PROP_UNKNOWN;
362 }

364 bool setProperty(int property_id, double value) CV_OVERRIDE
365 {
366 switch (property_id) {
367 case CAP_PROP_FRAME_WIDTH:
368 desiredWidth = static_cast<int32_t>(value);
369 setWidthHeight(setupWidth);
370 return true;
371 case CAP_PROP_FRAME_HEIGHT:
372 desiredHeight = static_cast<int32_t>(value);
373 setWidthHeight(setupHeight);
374 return true;
375 case CAP_PROP_FOURCC:
376 {
377 uint32_t newFourCC = cvRound(value);
378 if (fourCC == newFourCC) {
379 return true;
380 } else {
381 switch (newFourCC) {
382 case FOURCC_BGR:
383 case FOURCC_RGB:
384 case FOURCC_BGRA:
385 case FOURCC_RGBA:
386 case FOURCC_GRAY:
387 fourCC = newFourCC;
388 return true;
389 case FOURCC_YV12:
390 if (colorFormat == COLOR_FormatYUV420Planar) {
391 fourCC = newFourCC;
392 return true;
393 } else {
394 LOGE("Unsupported FOURCC conversion COLOR_FormatYUV420SemiPlanar"
395 " -> COLOR_FormatYUV420Planar");
396 return false;
397 }
398 case FOURCC_NV21:
399 if (colorFormat == COLOR_FormatYUV420SemiPlanar) {
400 fourCC = newFourCC;
401 return true;
402 } else {
403 LOGE("Unsupported FOURCC conversion COLOR_FormatYUV420Planar"
404 " -> COLOR_FormatYUV420SemiPlanar");
405 return false;
406 }
407 default:
408 LOGE("Unsupported FOURCC value: %d\n", fourCC);
409 return false;
410 }
411 }
412 }
413 case CAP_PROP_AUTO_EXPOSURE:
414 aeMode = (value != 0) ? ACAMERA_CONTROL_AE_MODE_ON : ACAMERA_CONTROL_AE_MODE_OFF;
415 if (isOpened()) {
416 return submitRequest(ACaptureRequest_setEntry_u8, ACAMERA_CONTROL_AE_MODE, aeMode);
417 }
418 return true;
419 case CAP_PROP_EXPOSURE:
420 if (isOpened() && exposureRange.isValid()) {
421 exposureTime = exposureRange.clamp(static_cast<int64_t>(value));
422 LOGI("Setting CAP_PROP_EXPOSURE will have no effect unless CAP_PROP_AUTO_EXPOSURE is off");
423 return submitRequest(ACaptureRequest_setEntry_i64, ACAMERA_SENSOR_EXPOSURE_TIME, exposureTime);
424 }
425 return false;
426 case CAP_PROP_ISO_SPEED:
427 if (isOpened() && sensitivityRange.isValid()) {
428 sensitivity = sensitivityRange.clamp(static_cast<int32_t>(value));
429 LOGI("Setting CAP_PROP_ISO_SPEED will have no effect unless CAP_PROP_AUTO_EXPOSURE is off");
430 return submitRequest(ACaptureRequest_setEntry_i32, ACAMERA_SENSOR_SENSITIVITY, sensitivity);
431 }
432 return false;
433 case CAP_PROP_ZOOM:
434 if (isOpened() && zoomRange.isValid()) {
435 zoomRatio = zoomRange.clamp(static_cast<float>(value));
436 return submitRequest(ACaptureRequest_setEntry_float, ACAMERA_CONTROL_ZOOM_RATIO, zoomRatio);
437 }
438 return true;
439 case CAP_PROP_ANDROID_DEVICE_TORCH:
440 flashMode = (value != 0) ? ACAMERA_FLASH_MODE_TORCH : ACAMERA_FLASH_MODE_OFF;
441 if (isOpened()) {
442 return submitRequest(ACaptureRequest_setEntry_u8, ACAMERA_FLASH_MODE, flashMode);
443 }
444 return true;
445 default:
446 break;
447 }
448 return false;
449 }

451 bool initCapture()
452 {
453 cameraManager.reset(ACameraManager_create());
454 if (!cameraManager) {
455 LOGE("Cannot create camera manager!");
456 return false;
457 }
458 ACameraIdList* cameraIds;
459 camera_status_t cStatus = ACameraManager_getCameraIdList(cameraManager.get(), &cameraIds);
460 if (cStatus != ACAMERA_OK) {
461 LOGE("Get camera list failed with error code: %d", cStatus);
462 return false;
463 }
464 AObjPtr<ACameraIdList> cameraIdList(cameraIds, ACameraManager_deleteCameraIdList);
465 if (deviceIndex < 0 || deviceIndex >= cameraIds->numCameras) {
466 LOGE("Camera index out of range %d (Number of cameras: %d)", deviceIndex, cameraIds->numCameras);
467 return false;
468 }
469 const char *cameraId = cameraIdList.get()->cameraIds[deviceIndex];
470
471 ACameraDevice* camera;
472 cStatus = ACameraManager_openCamera(cameraManager.get(), cameraId, &deviceCallbacks, &camera);
473 if (cStatus != ACAMERA_OK) {
474 LOGE("Open camera failed with error code: %d", cStatus);
475 return false;
476 }
477 cameraDevice.reset(camera);
478
479 ACameraMetadata* metadata;
480 cStatus = ACameraManager_getCameraCharacteristics(cameraManager.get(), cameraId, &metadata);
481 if (cStatus != ACAMERA_OK) {
482 LOGE("Get camera characteristics failed with error code: %d", cStatus);
483 return false;
484 }
485 AObjPtr<ACameraMetadata> cameraMetadata(metadata, ACameraMetadata_free);
486
487 getPropertyRanges(cameraMetadata.get());
488
489 int32_t bestMatchWidth = 0, bestMatchHeight = 0;
490 findResolutionMatch(cameraMetadata.get(), bestMatchWidth, bestMatchHeight);
491 LOGI("Best resolution match: %dx%d", bestMatchWidth, bestMatchHeight);
492
493 AImageReader* reader;
494 media_status_t mStatus = AImageReader_new(bestMatchWidth, bestMatchHeight,
495 AIMAGE_FORMAT_YUV_420_888, MAX_BUF_COUNT, &reader);
496 if (mStatus != AMEDIA_OK) {
497 LOGE("ImageReader creation failed with error code: %d", mStatus);
498 return false;
499 }
500 frameWidth = bestMatchWidth;
501 frameHeight = bestMatchHeight;
502 imageReader.reset(reader);
503
504 ANativeWindow* window;
505 mStatus = AImageReader_getWindow(imageReader.get(), &window);
506 if (mStatus != AMEDIA_OK) {
507 LOGE("Could not get ANativeWindow: %d", mStatus);
508 return false;
509 }
510 nativeWindow.reset(window);
511 ANativeWindow_acquire(nativeWindow.get());
512
513 ACaptureSessionOutputContainer* container;
514 cStatus = ACaptureSessionOutputContainer_create(&container);
515 if (cStatus != ACAMERA_OK) {
516 LOGE("CaptureSessionOutputContainer creation failed with error code: %d", cStatus);
517 return false;
518 }
519 outputContainer.reset(container);
520
521 ACaptureSessionOutput* output;
522 cStatus = ACaptureSessionOutput_create(nativeWindow.get(), &output);
523 if (cStatus != ACAMERA_OK) {
524 LOGE("CaptureSessionOutput creation failed with error code: %d", cStatus);
525 return false;
526 }
527 sessionOutput.reset(output);
528
529 cStatus = ACaptureSessionOutputContainer_add(outputContainer.get(), sessionOutput.get());
530 if (cStatus != ACAMERA_OK) {
531 LOGE("CaptureSessionOutput Container add failed with error code: %d", cStatus);
532 return false;
533 }
534 sessionOutputAdded = true;
535
536 ACameraOutputTarget* target;
537 cStatus = ACameraOutputTarget_create(nativeWindow.get(), &target);
538 if (cStatus != ACAMERA_OK) {
539 LOGE("CameraOutputTarget creation failed with error code: %d", cStatus);
540 return false;
541 }
542 outputTarget.reset(target);
543
544 ACaptureRequest* request;
545 cStatus = ACameraDevice_createCaptureRequest(cameraDevice.get(), TEMPLATE_PREVIEW, &request);
546 if (cStatus != ACAMERA_OK) {
547 LOGE("CaptureRequest creation failed with error code: %d", cStatus);
548 return false;
549 }
550 captureRequest.reset(request);
551
552 cStatus = ACaptureRequest_addTarget(captureRequest.get(), outputTarget.get());
553 if (cStatus != ACAMERA_OK) {
554 LOGE("Add target to CaptureRequest failed with error code: %d", cStatus);
555 return false;
556 }
557 targetAdded = true;
558
559 ACameraCaptureSession* session;
560 cStatus = ACameraDevice_createCaptureSession(cameraDevice.get(),
561 outputContainer.get(), &sessionCallbacks, &session);
562 if (cStatus != ACAMERA_OK) {
563 LOGE("CaptureSession creation failed with error code: %d", cStatus);
564 return false;
565 }
566 captureSession.reset(session);
567
568 ACaptureRequest_setEntry_u8(captureRequest.get(), ACAMERA_CONTROL_AE_MODE, 1, &aeMode);
569 if (aeMode != ACAMERA_CONTROL_AE_MODE_ON) {
570 ACaptureRequest_setEntry_i32(captureRequest.get(), ACAMERA_SENSOR_SENSITIVITY, 1, &sensitivity);
571 ACaptureRequest_setEntry_i64(captureRequest.get(), ACAMERA_SENSOR_EXPOSURE_TIME, 1, &exposureTime);
572 }
573 if (zoomRange.isValid()) {
574 ACaptureRequest_setEntry_float(captureRequest.get(), ACAMERA_CONTROL_ZOOM_RATIO, 1, &zoomRatio);
575 }
576 ACaptureRequest_setEntry_u8(captureRequest.get(), ACAMERA_FLASH_MODE, 1, &flashMode);
577
578 cStatus = ACameraCaptureSession_setRepeatingRequest(captureSession.get(),
579 &captureCallbacks, 1, &request, nullptr);
580 if (cStatus != ACAMERA_OK) {
581 LOGE("CameraCaptureSession set repeating request failed with error code: %d", cStatus);
582 return false;
583 }
584 return true;
585 }

587private:
588 void getPropertyRanges(const ACameraMetadata* metadata)
589 {
590 camera_status_t cStatus;
591 ACameraMetadata_const_entry val;
592
593 cStatus = ACameraMetadata_getConstEntry(metadata, ACAMERA_SENSOR_INFO_EXPOSURE_TIME_RANGE, &val);
594 if (cStatus == ACAMERA_OK) {
595 exposureRange.min = exposureTimeLimits.clamp(val.data.i64[0]);
596 exposureRange.max = exposureTimeLimits.clamp(val.data.i64[1]);
597 exposureTime = exposureRange.percentage(2);
598 } else {
599 LOGW("Unsupported ACAMERA_SENSOR_INFO_EXPOSURE_TIME_RANGE");
600 exposureRange.min = exposureRange.max = 0;
601 exposureTime = 0;
602 }
603
604 cStatus = ACameraMetadata_getConstEntry(metadata, ACAMERA_SENSOR_INFO_SENSITIVITY_RANGE, &val);
605 if (cStatus == ACAMERA_OK){
606 sensitivityRange.min = val.data.i32[0];
607 sensitivityRange.max = val.data.i32[1];
608 sensitivity = sensitivityRange.percentage(2);
609 } else {
610 LOGW("Unsupported ACAMERA_SENSOR_INFO_SENSITIVITY_RANGE");
611 sensitivityRange.min = sensitivityRange.max = 0;
612 sensitivity = 0;
613 }
614
615 cStatus = ACameraMetadata_getConstEntry(metadata, ACAMERA_CONTROL_ZOOM_RATIO_RANGE, &val);
616 if (cStatus == ACAMERA_OK){
617 zoomRange.min = val.data.f[0];
618 zoomRange.max = val.data.f[1];
619 zoomRatio = zoomRange.clamp(zoomRatio);
620 } else {
621 LOGW("Unsupported ACAMERA_CONTROL_ZOOM_RATIO_RANGE");

627 // calculate a score based on how well the width and height match the desired width and height
628 // basically draw the 2 rectangle on top of each other and take the ratio of the non-overlapping
629 // area to the overlapping area
630 double getScore(int32_t width, int32_t height) const {
631 double area1 = width * height;
632 double area2 = desiredWidth * desiredHeight;
633 if ((width < desiredWidth) == (height < desiredHeight)) {
634 return (width < desiredWidth) ? (area2 - area1)/area1 : (area1 - area2)/area2;
635 } else {
636 int32_t overlappedWidth = std::min(width, desiredWidth);
637 int32_t overlappedHeight = std::min(height, desiredHeight);
638 double overlappedArea = overlappedWidth * overlappedHeight;
639 return (area1 + area2 - overlappedArea)/overlappedArea;
640 }
641 }

643 void findResolutionMatch(const ACameraMetadata* metadata,
644 int32_t &bestMatchWidth, int32_t &bestMatchHeight) const {
645 ACameraMetadata_const_entry entry = {};
646 ACameraMetadata_getConstEntry(metadata, ACAMERA_SCALER_AVAILABLE_STREAM_CONFIGURATIONS, &entry);
647
648 double bestScore = std::numeric_limits<double>::max();
649 for (uint32_t i = 0; i < entry.count; i += 4) {
650 int32_t input = entry.data.i32[i + 3];
651 int32_t format = entry.data.i32[i + 0];
652
653 if (!input && format == AIMAGE_FORMAT_YUV_420_888) {
654 int32_t width = entry.data.i32[i + 1];
655 int32_t height = entry.data.i32[i + 2];
656
657 if (width == desiredWidth && height == desiredHeight) {
658 bestMatchWidth = width;
659 bestMatchHeight = height;
660 return;
661 }
662
663 double score = getScore(width, height);
664 if (score < bestScore) {
665 bestMatchWidth = width;
666 bestMatchHeight = height;
667 bestScore = score;
668 }
669 }
670 }
