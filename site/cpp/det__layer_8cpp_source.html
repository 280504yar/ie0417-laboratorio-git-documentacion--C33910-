31 bool getMemoryShapes(const std::vector<MatShape>& inputs,
32 const int /*requiredOutputs*/,
33 std::vector<MatShape>& outputs,
34 std::vector<MatShape>& /*internals*/) const CV_OVERRIDE
35 {
36 CV_Assert(inputs.size() == 1);
37 const MatShape& in = inputs[0];
38 CV_Assert(in.size() >= 2);
39
40 int n0 = in[in.size() - 2];
41 int n1 = in[in.size() - 1];
42 CV_Assert(n0 == -1 || n1 == -1 || n0 == n1);
43
44 MatShape out;
45 if (in.size() > 2)
46 out.assign(in.begin(), in.end() - 2);
47 else
48 out = MatShape::scalar();
49
50 outputs.assign(1, out);
51 return false;
52 }

68 void forward(InputArrayOfArrays inputs_arr,
69 OutputArrayOfArrays outputs_arr,
70 OutputArrayOfArrays /*internals_arr*/) CV_OVERRIDE
71 {
72 std::vector<Mat> inputs, outputs;
73 inputs_arr.getMatVector(inputs);
74 outputs_arr.getMatVector(outputs);
75
76 CV_Assert(inputs.size() == 1);
77 const Mat& X = inputs[0];
78
79 CV_Assert(X.dims >= 2);
80 int n = X.size[X.dims - 1];
81 int m = X.size[X.dims - 2];
82 CV_Assert(n == m);
83
84 size_t batch = X.total() / (X.size[X.dims - 2] * X.size[X.dims - 1]);
85
86 if (X.dims > 2)
87 {
88 int outDims = X.dims - 2;
89 std::vector<int> outSizes(X.size.p, X.size.p + outDims);
90 outputs[0].create(outDims, outSizes.data(), X.type());
91 }
92 // else: scalar output is pre-allocated by the engine; create() would detach it.
93
94 const int type = X.type();
95 const size_t elemSz = X.elemSize();
96 const size_t matStrideBytes = (size_t)n * (size_t)m * elemSz;
97
98 const uchar* base = X.data;
99 uchar* outp = outputs[0].ptr();
100
101 parallel_for_(Range(0, static_cast<int>(batch)), [&](const Range& r){
102 Mat temp;
103 for (int bi = r.start; bi < r.end; ++bi)
104 {
105 size_t b = static_cast<size_t>(bi);
106 const uchar* p = base + b * matStrideBytes;
107 Mat A(m, n, type, const_cast<uchar*>(p));
108
109 double det;
110 if (type == CV_32F || type == CV_64F) {
111 det = determinant(A);
112 } else {
113 A.convertTo(temp, CV_32F);
114 det = determinant(temp);
115 }
116
117 if (type == CV_32F)
118 reinterpret_cast<float*>(outp)[b] = static_cast<float>(det);
119 else if (type == CV_64F)
120 reinterpret_cast<double*>(outp)[b] = det;
121 else if (type == CV_16F)
122 reinterpret_cast<hfloat*>(outp)[b] = saturate_cast<hfloat>(det);
123 else if (type == CV_16BF)
124 reinterpret_cast<bfloat*>(outp)[b] = saturate_cast<bfloat>(det);
125 else
126 CV_Error(Error::BadDepth, "Unsupported input/output depth for DetLayer");
127 }
128 });
129 }
130};
