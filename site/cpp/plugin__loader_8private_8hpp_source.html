74LibHandle_t libraryLoad_(const FileSystemPath_t& filename)
75{
76#if defined(_WIN32)
77# ifdef WINRT

79# else
80 return LoadLibraryW(filename.c_str());

82#elif defined(__linux__) || defined(__APPLE__) || defined(__OpenBSD__) || defined(__FreeBSD__) || defined(__HAIKU__) || defined(__GLIBC__) || defined(__EMSCRIPTEN__) || defined(__CYGWIN__)

84 CV_LOG_IF_DEBUG(NULL, !handle, "dlopen() error: " << dlerror());
