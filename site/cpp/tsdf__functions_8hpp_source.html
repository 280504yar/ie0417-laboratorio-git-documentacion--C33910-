52inline TsdfType floatToTsdf(float num)
53{
54 //CV_Assert(-1 < num <= 1);
55 int8_t res = int8_t(num * (-128.f));

84inline depthType bilinearDepth(const Depth& m, cv::Point2f pt)
85{
86 const bool fixMissingData = false;
87 const depthType defaultValue = qnan;
88 // NaN passes both bounds checks below, so reject it explicitly
89 if (cvIsNaN(pt.x) || cvIsNaN(pt.y) ||
90 pt.x < 0 || pt.x >= m.cols - 1 ||
91 pt.y < 0 || pt.y >= m.rows - 1)
92 return defaultValue;
93
94 int xi = cvFloor(pt.x), yi = cvFloor(pt.y);
95
96 const depthType* row0 = m[yi + 0];
97 const depthType* row1 = m[yi + 1];
98
99 depthType v00 = row0[xi + 0];
100 depthType v01 = row0[xi + 1];
101 depthType v10 = row1[xi + 0];
102 depthType v11 = row1[xi + 1];
103
104 // assume correct depth is positive
105 bool b00 = v00 > 0;
106 bool b01 = v01 > 0;
107 bool b10 = v10 > 0;
108 bool b11 = v11 > 0;
109
110 if (!fixMissingData)
111 {
112 if (!(b00 && b01 && b10 && b11))
113 return defaultValue;
114 else
115 {
116 float tx = pt.x - xi, ty = pt.y - yi;
117 depthType v0 = v00 + tx * (v01 - v00);
118 depthType v1 = v10 + tx * (v11 - v10);
119 return v0 + ty * (v1 - v0);
120 }
121 }
122 else
123 {
124 int nz = b00 + b01 + b10 + b11;
125 if (nz == 0)
126 {
127 return defaultValue;
128 }
129 if (nz == 1)
130 {
131 if (b00) return v00;
132 if (b01) return v01;
133 if (b10) return v10;
134 if (b11) return v11;
135 }
136 if (nz == 2)
137 {
138 if (b00 && b10) v01 = v00, v11 = v10;
139 if (b01 && b11) v00 = v01, v10 = v11;
140 if (b00 && b01) v10 = v00, v11 = v01;
141 if (b10 && b11) v00 = v10, v01 = v11;
142 if (b00 && b11) v01 = v10 = (v00 + v11) * 0.5f;
143 if (b01 && b10) v00 = v11 = (v01 + v10) * 0.5f;
144 }
145 if (nz == 3)
146 {
147 if (!b00) v00 = v10 + v01 - v11;
148 if (!b01) v01 = v00 + v11 - v10;
149 if (!b10) v10 = v00 + v11 - v01;
150 if (!b11) v11 = v01 + v10 - v00;
151 }
152
153 float tx = pt.x - xi, ty = pt.y - yi;
154 depthType v0 = v00 + tx * (v01 - v00);
155 depthType v1 = v10 + tx * (v11 - v10);
156 return v0 + ty * (v1 - v0);
157 }
158}
