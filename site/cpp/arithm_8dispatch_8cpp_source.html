37// ---- tier 3: op-level dispatcher used by the compiler --------------------------------------------
38TKernel getElemwiseFunc(TOp op, int depth0, int depth1, int depth2, int rdepth)
39{
40 (void)depth2;
41
42 if (op == OP_CAST) return getCastFunc(depth0, rdepth, false);
43 if (op == OP_CONVERT_SCALE) return getCastFunc(depth0, rdepth, true);
44
45 if (op == OP_ADD || op == OP_SUB)
46 {
47 if (depth0 != depth1) return {}; // operands must be the same type
48 return op == OP_ADD ? getAddFunc(depth0, rdepth) : getSubFunc(depth0, rdepth);
49 }
50
51 // OP_ADDW (addWeighted): a*alpha+b*beta+gamma; operands same type T, result R (T/f32 for small ints
52 // + f16/bf16/f32, f64 otherwise). alpha/beta/gamma travel in the instruction's params.
53 if (op == OP_ADDW)
54 {
55 if (depth0 != depth1) return {};
56 return getAddWeightedFunc(depth0, rdepth);
57 }
58
59 // OP_MUL / OP_DIV / OP_POW: operands same type T; compute in the float work type (rdepth).
60 if (op == OP_MUL || op == OP_DIV || op == OP_POW)
61 {
62 if (depth0 != depth1) return {};
63 if (op == OP_MUL) return getMulFunc(depth0, rdepth);
64 if (op == OP_POW) return getPowFunc(depth0, rdepth);
65 // integer inputs guard divide-by-zero (-> 0); floats do not (a/0 -> inf, matching cv::divide).
66 const bool isflt = depth0==CV_16F || depth0==CV_16BF || depth0==CV_32F || depth0==CV_64F;
67 return getDivFunc(depth0, rdepth, !isflt);
68 }
69
70 // OP_MIN / OP_MAX: T x T -> T.
71 if (op == OP_MIN || op == OP_MAX)
72 {
73 if (depth0 != depth1 || rdepth != depth0) return {};
74 return op == OP_MIN ? getMinFunc(depth0, rdepth) : getMaxFunc(depth0, rdepth);
75 }
76
77 // OP_ABSDIFF: result is absdiffResultDepth(T) (unsigned same width for signed).
78 if (op == OP_ABSDIFF)
79 {
80 if (depth0 != depth1) return {};
81 return getAbsdiffFunc(depth0, rdepth);
82 }
83
84 // OP_CMP_*: T x T -> u8 mask (0 / 1 / 255, value via TKernel::flags).
85 if (opCategory(op) == CAT_COMPARE)
86 {
87 if (depth0 != depth1 || rdepth != CV_8U) return {};
88 return getCmpFunc(op, depth0);
89 }
90
91 // OP_AND / OP_OR / OP_XOR: bit-pattern op, T x T -> T (same depth), dispatched by element size.
92 if (op == OP_AND || op == OP_OR || op == OP_XOR)
93 {
94 if (depth0 != depth1 || rdepth != depth0) return {};
95 return getBitwiseFunc(op, CV_ELEM_SIZE1(depth0));
96 }
97
98 // OP_NOT: ~x, one input, T -> T (same depth), dispatched by element size.
99 if (op == OP_NOT)
100 {
101 if (rdepth != depth0) return {};
102 return getNotFunc(CV_ELEM_SIZE1(depth0));
103 }
104
105 // OP_HYPOT / OP_ATAN2: T x T -> T over the float depths (integer inputs are the compiler's job).
106 if (op == OP_HYPOT || op == OP_ATAN2)
107 {
108 if (depth0 != depth1) return {};
109 return op == OP_HYPOT ? getHypotFunc(depth0, rdepth) : getAtan2Func(depth0, rdepth);
110 }
111
112 // unary math: T -> T over the float depths (math.simd.hpp). Anything else - integer input,
113 // widening/narrowing result - is the compiler's job (cast to a float working type first).
114 if (op == OP_SQRT || op == OP_EXP || op == OP_LOG || op == OP_SIN || op == OP_COS ||
115 op == OP_TANH || op == OP_ERF || op == OP_RELU)
116 {
117 if (rdepth != depth0) return {};
118 return getMathFunc(op, depth0);
119 }
120
121 // OP_SELECT: depth0 is the (1-byte, never cast) mask; both branches and dst share one depth.
122 if (op == OP_SELECT)
123 {
124 if (depth1 != depth2 || rdepth != depth1) return {};
125 return getSelectFunc(depth0, rdepth);
126 }
127
128 // OP_CLAMP: x, lo, hi and dst all share one depth (emitTernary unifies them).
129 if (op == OP_CLAMP)
130 {
131 if (depth1 != depth0 || depth2 != depth0 || rdepth != depth0) return {};
132 return getClampFunc(rdepth);
133 }
134
135 return {};
136}

142// Legacy cv::hal entry point, still declared in core/hal/hal.hpp and used by external code (the
143// G-API fluid backend in opencv_contrib calls it directly): forward to the element-wise engine's
144// u8 multiply kernel. `scale` is a pointer to a double, as in the old contract.
145void mul8u(const uchar* src1, size_t step1, const uchar* src2, size_t step2,
146 uchar* dst, size_t step, int width, int height, void* scale)
147{
148 const double params[] = { scale ? *(const double*)scale : 1.0 };
149 ew::TKernel k = ew::getMulFunc(CV_8U, CV_8U);
150 k.fptr(src1, step1, 1, src2, step2, 1, nullptr, 0, 0, dst, step, width, height,
151 params, k.flags, k.userdata);
152}

154// Legacy cv::hal bitwise entry points, still declared in core/hal/hal.hpp and used by other modules
155// (e.g. opencv_objdetect's aruco) and external code: forward to the element-wise engine's byte-wise
156// bitwise kernels. No scalar params - bitwise ops ignore them.
157static void bitwise8u(ew::TOp op, const uchar* src1, size_t step1, const uchar* src2, size_t step2,
158 uchar* dst, size_t step, int width, int height)
159{
160 const double noparams[4] = {};
161 ew::TKernel k = ew::getBitwiseFunc(op, 1);
162 CV_Assert(k.fptr != nullptr);
163 k.fptr(src1, step1, 1, src2, step2, 1, nullptr, 0, 0, dst, step, width, height,
164 noparams, k.flags, k.userdata);
165}
