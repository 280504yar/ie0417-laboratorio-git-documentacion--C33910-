117 void getAspectRatios(const LayerParams &params)
118 {
119 DictValue aspectRatioParameter;
120 bool aspectRatioRetieved = getParameterDict(params, "aspect_ratio", aspectRatioParameter);
121 if (!aspectRatioRetieved)
122 return;
123
124 for (int i = 0; i < aspectRatioParameter.size(); ++i)
125 {
126 float aspectRatio = aspectRatioParameter.get<float>(i);
127 bool alreadyExists = fabs(aspectRatio - 1.f) < 1e-6f;
128
129 for (size_t j = 0; j < _aspectRatios.size() && !alreadyExists; ++j)
130 {
131 alreadyExists = fabs(aspectRatio - _aspectRatios[j]) < 1e-6;
132 }
133 if (!alreadyExists)
134 {
135 _aspectRatios.push_back(aspectRatio);
136 if (_flip)
137 {
138 _aspectRatios.push_back(1./aspectRatio);
139 }
140 }
141 }
142 }

160 void getVariance(const LayerParams &params)
161 {
162 DictValue varianceParameter;
163 bool varianceParameterRetrieved = getParameterDict(params, "variance", varianceParameter);
164 CV_Assert(varianceParameterRetrieved);
165
166 int varianceSize = varianceParameter.size();
167 if (varianceSize > 1)
168 {
169 // Must and only provide 4 variance.
170 CV_Assert(varianceSize == 4);
171
172 for (int i = 0; i < varianceSize; ++i)
173 {
174 float variance = varianceParameter.get<float>(i);
175 CV_Assert(variance > 0);
176 _variance.push_back(variance);
177 }
178 }
179 else
180 {
181 if (varianceSize == 1)
182 {
183 float variance = varianceParameter.get<float>(0);
184 CV_Assert(variance > 0);
185 _variance.push_back(variance);
186 }
187 else
188 {
189 // Set default to 0.1.
190 _variance.push_back(0.1f);
191 }
192 }
193 }

195 static LayerParams normalizeONNXParams(const LayerParams& params)
196 {
197 LayerParams p = params;
198 auto remap = [&](const std::string& from, const std::string& to) {
199 if (p.has(from) && !p.has(to))
200 p.set(to, p.get(from));
201 };
202 remap("variances", "variance");
203 remap("min_sizes", "min_size");
204 remap("max_sizes", "max_size");
205 remap("aspect_ratios", "aspect_ratio");
206
207 if (p.has("steps") && !p.has("step_h") && !p.has("step_w")) {
208 DictValue steps = p.get("steps");
209 if (steps.size() == 2) {
210 p.set("step_h", steps.get<float>(0));
211 p.set("step_w", steps.get<float>(1));
212 }
213 }
214 return p;
215 }

217 PriorBoxLayerImpl(const LayerParams &params_)
218 {
219 const LayerParams params = normalizeONNXParams(params_);
220 setParamsFrom(params);
221 _flip = getParameter<bool>(params, "flip", 0, false, true);
222 _clip = getParameter<bool>(params, "clip", 0, false, true);
223 _bboxesNormalized = getParameter<bool>(params, "normalized_bbox", 0, false, true);
224
225 getParams("min_size", params, &_minSize);
226 getAspectRatios(params);
227 getVariance(params);
228
229 if (params.has("max_size"))
230 {
231 getParams("max_size", params, &_maxSize);
232 CV_Assert(_minSize.size() == _maxSize.size());
233 for (int i = 0; i < _maxSize.size(); i++)
234 CV_Assert(_minSize[i] < _maxSize[i]);
235 }
236
237 std::vector<float> widths, heights;
238 getParams("width", params, &widths);
239 getParams("height", params, &heights);
240 _explicitSizes = !widths.empty();
241 CV_Assert(widths.size() == heights.size());
242
243 if (_explicitSizes)
244 {
245 CV_Assert(_aspectRatios.empty());
246 CV_Assert(!params.has("min_size"));
247 CV_Assert(!params.has("max_size"));
248 _boxWidths = widths;
249 _boxHeights = heights;
250 }
251 else
252 {
253 CV_Assert(!_minSize.empty());
254 for (int i = 0; i < _minSize.size(); ++i)
255 {
256 float minSize = _minSize[i];
257 CV_Assert(minSize > 0);
258 _boxWidths.push_back(minSize);
259 _boxHeights.push_back(minSize);
260
261 if (_maxSize.size() > 0)
262 {
263 float size = sqrt(minSize * _maxSize[i]);
264 _boxWidths.push_back(size);
265 _boxHeights.push_back(size);
266 }
267
268 // rest of priors
269 for (size_t r = 0; r < _aspectRatios.size(); ++r)
270 {
271 float arSqrt = sqrt(_aspectRatios[r]);
272 _boxWidths.push_back(minSize * arSqrt);
273 _boxHeights.push_back(minSize / arSqrt);
274 }
275 }
276 }
277 CV_Assert(_boxWidths.size() == _boxHeights.size());
278 _numPriors = _boxWidths.size();
279
280 if (params.has("step_h") || params.has("step_w")) {
281 CV_Assert(!params.has("step"));
282 _stepY = getParameter<float>(params, "step_h");
283 CV_Assert(_stepY > 0.);
284 _stepX = getParameter<float>(params, "step_w");
285 CV_Assert(_stepX > 0.);
286 } else if (params.has("step")) {
287 const float step = getParameter<float>(params, "step");
288 CV_Assert(step > 0);
289 _stepY = step;
290 _stepX = step;
291 } else {
292 _stepY = 0;
293 _stepX = 0;
294 }
295 if (params.has("offset_h") || params.has("offset_w"))
296 {
297 CV_Assert_N(!params.has("offset"), params.has("offset_h"), params.has("offset_w"));
298 getParams("offset_h", params, &_offsetsY);
299 getParams("offset_w", params, &_offsetsX);
300 CV_Assert(_offsetsX.size() == _offsetsY.size());
301 _numPriors *= std::max((size_t)1, 2 * (_offsetsX.size() - 1));
302 }
303 else
304 {
305 float offset = getParameter<float>(params, "offset", 0, false, 0.5);
306 _offsetsX.assign(1, offset);
307 _offsetsY.assign(1, offset);
308 }
309 }

321 bool getMemoryShapes(const std::vector<MatShape> &inputs,
322 const int requiredOutputs,
323 std::vector<MatShape> &outputs,
324 std::vector<MatShape> &internals) const CV_OVERRIDE
325 {
326 CV_Assert(!inputs.empty());
327
328 int layerHeight = inputs[0][2];
329 int layerWidth = inputs[0][3];
330
331 // Since all images in a batch has same height and width, we only need to
332 // generate one set of priors which can be shared across all images.
333 size_t outNum = 1;
334 // 2 channels. First channel stores the mean of each prior coordinate.
335 // Second channel stores the variance of each prior coordinate.
336 size_t outChannels = 2;
337
338 outputs.resize(1, shape(outNum, outChannels,
339 layerHeight * layerWidth * _numPriors * 4));
340
341 return false;
342 }

344 void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays) CV_OVERRIDE
345 {
346 std::vector<Mat> inputs;
347 inputs_arr.getMatVector(inputs);
348
349 CV_CheckGT(inputs.size(), (size_t)1, "");
350 CV_CheckEQ(inputs[0].dims, 4, ""); CV_CheckEQ(inputs[1].dims, 4, "");
351 int layerWidth = inputs[0].size[3];
352 int layerHeight = inputs[0].size[2];
353
354 int imageWidth = inputs[1].size[3];
355 int imageHeight = inputs[1].size[2];
356
357 _stepY = _stepY == 0 ? (static_cast<float>(imageHeight) / layerHeight) : _stepY;
358 _stepX = _stepX == 0 ? (static_cast<float>(imageWidth) / layerWidth) : _stepX;
359 }

361#ifdef HAVE_OPENCL
362 bool forward_ocl(InputArrayOfArrays inps, OutputArrayOfArrays outs, OutputArrayOfArrays internals)
363 {
364 std::vector<UMat> inputs;
365 std::vector<UMat> outputs;
366
367 bool use_half = (inps.depth() == CV_16F);
368 inps.getUMatVector(inputs);
369 outs.getUMatVector(outputs);
370
371 int _layerWidth = inputs[0].size[3];
372 int _layerHeight = inputs[0].size[2];
373
374 int _imageWidth = inputs[1].size[3];
375 int _imageHeight = inputs[1].size[2];
376
377 if (umat_offsetsX.empty())
378 {
379 Mat offsetsX(1, _offsetsX.size(), CV_32FC1, &_offsetsX[0]);
380 Mat offsetsY(1, _offsetsY.size(), CV_32FC1, &_offsetsY[0]);
381 Mat variance(1, _variance.size(), CV_32FC1, &_variance[0]);
382 Mat widths(1, _boxWidths.size(), CV_32FC1, &_boxWidths[0]);
383 Mat heights(1, _boxHeights.size(), CV_32FC1, &_boxHeights[0]);
384
385 offsetsX.copyTo(umat_offsetsX);
386 offsetsY.copyTo(umat_offsetsY);
387 variance.copyTo(umat_variance);
388 widths.copyTo(umat_widths);
389 heights.copyTo(umat_heights);
390 }
391
392 String opts;
393 if (use_half)
394 opts = "-DDtype=half -DDtype4=half4 -Dconvert_T=convert_half4";
395 else
396 opts = "-DDtype=float -DDtype4=float4 -Dconvert_T=convert_float4";
397
398 size_t nthreads = _layerHeight * _layerWidth;
399 ocl::Kernel kernel("prior_box", ocl::dnn::prior_box_oclsrc, opts);
400
401 kernel.set(0, (int)nthreads);
402 kernel.set(1, (float)_stepX);
403 kernel.set(2, (float)_stepY);
404 kernel.set(3, ocl::KernelArg::PtrReadOnly(umat_offsetsX));
405 kernel.set(4, ocl::KernelArg::PtrReadOnly(umat_offsetsY));
406 kernel.set(5, (int)_offsetsX.size());
407 kernel.set(6, ocl::KernelArg::PtrReadOnly(umat_widths));
408 kernel.set(7, ocl::KernelArg::PtrReadOnly(umat_heights));
409 kernel.set(8, (int)_boxWidths.size());
410 kernel.set(9, ocl::KernelArg::PtrWriteOnly(outputs[0]));
411 kernel.set(10, (int)_layerHeight);
412 kernel.set(11, (int)_layerWidth);
413 kernel.set(12, (int)_imageHeight);
414 kernel.set(13, (int)_imageWidth);
415 kernel.run(1, &nthreads, NULL, false);
416
417 // clip the prior's coordinate such that it is within [0, 1]
418 if (_clip)
419 {
420 ocl::Kernel kernel("clip", ocl::dnn::prior_box_oclsrc, opts);
421 size_t nthreads = _layerHeight * _layerWidth * _numPriors * 4;
422 if (!kernel.args((int)nthreads, ocl::KernelArg::PtrReadWrite(outputs[0]))
423 .run(1, &nthreads, NULL, false))
424 return false;
425 }
426
427 // set the variance.
428 {
429 ocl::Kernel kernel("set_variance", ocl::dnn::prior_box_oclsrc, opts);
430 int offset = (int)total(shape(outputs[0]), 2);
431 size_t nthreads = _layerHeight * _layerWidth * _numPriors;
432 kernel.set(0, (int)nthreads);
433 kernel.set(1, (int)offset);
434 kernel.set(2, (int)_variance.size());
435 kernel.set(3, ocl::KernelArg::PtrReadOnly(umat_variance));
436 kernel.set(4, ocl::KernelArg::PtrWriteOnly(outputs[0]));
437 if (!kernel.run(1, &nthreads, NULL, false))
438 return false;
439 }
440 return true;
441 }
442#endif

444 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
445 {
446 CV_TRACE_FUNCTION();
447 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
448
449 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
450 forward_ocl(inputs_arr, outputs_arr, internals_arr))
451
452 if (inputs_arr.depth() == CV_16F)
453 {
454 forward_fallback(inputs_arr, outputs_arr, internals_arr);
455 return;
456 }
457
458 std::vector<Mat> inputs, outputs;
459 inputs_arr.getMatVector(inputs);
460 outputs_arr.getMatVector(outputs);
461
462 // PriorBox uses layer parameters and inputs[0]/inputs[1] shapes; tensor data is never read.
463 // Extra inputs (e.g., in ssd_vgg16.onnx) are ignored, so require >= 2 inputs instead of exactly 2.
464 CV_Assert(inputs.size() >= 2);
465
466 int _layerWidth = inputs[0].size[3];
467 int _layerHeight = inputs[0].size[2];
468
469 int _imageWidth = inputs[1].size[3];
470 int _imageHeight = inputs[1].size[2];
471
472 float* outputPtr = outputs[0].ptr<float>();
473 float _boxWidth, _boxHeight;
474 for (size_t h = 0; h < _layerHeight; ++h)
475 {
476 for (size_t w = 0; w < _layerWidth; ++w)
477 {
478 for (size_t i = 0; i < _boxWidths.size(); ++i)
479 {
480 _boxWidth = _boxWidths[i];
481 _boxHeight = _boxHeights[i];
482 for (int j = 0; j < _offsetsX.size(); ++j)
483 {
484 float center_x = (w + _offsetsX[j]) * _stepX;
485 float center_y = (h + _offsetsY[j]) * _stepY;
486 outputPtr = addPrior(center_x, center_y, _boxWidth, _boxHeight, _imageWidth,
487 _imageHeight, _bboxesNormalized, outputPtr);
488 }
489 }
490 }
491 }
492 // clip the prior's coordinate such that it is within [0, 1]
493 if (_clip)
494 {
495 int _outChannelSize = _layerHeight * _layerWidth * _numPriors * 4;
496 outputPtr = outputs[0].ptr<float>();
497 for (size_t d = 0; d < _outChannelSize; ++d)
498 {
499 outputPtr[d] = std::min<float>(std::max<float>(outputPtr[d], 0.), 1.);
500 }
501 }
502 // set the variance.
503 outputPtr = outputs[0].ptr<float>(0, 1);
504 if(_variance.size() == 1)
505 {
506 Mat secondChannel(1, outputs[0].size[2], CV_32F, outputPtr);
507 secondChannel.setTo(Scalar::all(_variance[0]));
508 }
509 else
510 {
511 int count = 0;
512 for (size_t h = 0; h < _layerHeight; ++h)
513 {
514 for (size_t w = 0; w < _layerWidth; ++w)
515 {
516 for (size_t i = 0; i < _numPriors; ++i)
517 {
518 for (int j = 0; j < 4; ++j)
519 {
520 outputPtr[count] = _variance[j];
521 ++count;
522 }
523 }
524 }
525 }
526 }
527 }

531 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> >& inputs, const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
532 {
533 // PriorBox uses layer parameters and inputs[0]/inputs[1] shapes; tensor data is never read.
534 // Extra inputs (e.g., in ssd_vgg16.onnx) are ignored, so require >= 2 inputs instead of exactly 2.
535 CV_Assert(nodes.size() >= 2);
536 auto layer = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
537 auto image = nodes[1].dynamicCast<InfEngineNgraphNode>()->node;
538 auto layer_shape = std::make_shared<ov::op::v3::ShapeOf>(layer);
539 auto image_shape = std::make_shared<ov::op::v3::ShapeOf>(image);
540
541 auto lower_bounds = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{1}, std::vector<int64_t>{2});
542 auto upper_bounds = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{1}, std::vector<int64_t>{4});
543 auto strides = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{1}, std::vector<int64_t>{1});
544
545 auto slice_layer = std::make_shared<ov::op::v1::StridedSlice>(layer_shape,
546 lower_bounds, upper_bounds, strides, std::vector<int64_t>{}, std::vector<int64_t>{});
547 auto slice_image = std::make_shared<ov::op::v1::StridedSlice>(image_shape,
548 lower_bounds, upper_bounds, strides, std::vector<int64_t>{}, std::vector<int64_t>{});
549
550 if (_explicitSizes)
551 {
552 CV_Assert_N(!_boxWidths.empty(), !_boxHeights.empty(), !_variance.empty());
553 CV_Assert(_boxWidths.size() == _boxHeights.size());
554 ov::op::v0::PriorBoxClustered::Attributes attrs;
555 attrs.widths = _boxWidths;
556 attrs.heights = _boxHeights;
557 attrs.clip = _clip;
558 CV_CheckEQ(_offsetsX.size(), (size_t)1, ""); CV_CheckEQ(_offsetsY.size(), (size_t)1, ""); CV_CheckEQ(_offsetsX[0], _offsetsY[0], "");
559 attrs.offset = _offsetsX[0];
560 attrs.step_heights = _stepY;
561 attrs.step_widths = _stepX;
562 attrs.variances = _variance;
563
564 auto priorBox = std::make_shared<ov::op::v0::PriorBoxClustered>(slice_layer, slice_image, attrs);
565 auto axis = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{1}, std::vector<int64_t>{0});
566 auto unsqueeze = std::make_shared<ov::op::v0::Unsqueeze>(priorBox, axis);
567 return Ptr<BackendNode>(new InfEngineNgraphNode(unsqueeze));
568 }
569 else
570 {
571 ov::op::v0::PriorBox::Attributes attrs;
572 attrs.min_size = _minSize;
573 attrs.max_size = _maxSize;
574 // doesn't work with empty aspectRatio
575 attrs.aspect_ratio = !_aspectRatios.empty()? _aspectRatios : std::vector<float>{1.0f};
576 attrs.clip = _clip;
577 attrs.flip = false;
578 attrs.variance = _variance;
579 CV_CheckEQ(_offsetsX.size(), (size_t)1, ""); CV_CheckEQ(_offsetsY.size(), (size_t)1, ""); CV_CheckEQ(_offsetsX[0], _offsetsY[0], "");
580 attrs.offset = _offsetsX[0];
581
582 attrs.step = _stepX;
583 attrs.scale_all_sizes = !_aspectRatios.empty();
584
585 auto priorBox = std::make_shared<ov::op::v0::PriorBox>(slice_layer, slice_image, attrs);
586 auto axis = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{1}, std::vector<int64_t>{0});
587 auto unsqueeze = std::make_shared<ov::op::v0::Unsqueeze>(priorBox, axis);
588 return Ptr<BackendNode>(new InfEngineNgraphNode(unsqueeze));
589 }
590 }

634 virtual int64 getFLOPS(const std::vector<MatShape> &inputs,
635 const std::vector<MatShape> &outputs) const CV_OVERRIDE
636 {
637 CV_UNUSED(outputs); // suppress unused variable warning
638 long flops = 0;
639
640 for (int i = 0; i < inputs.size(); i++)
