16static uint32_t findMemoryType(uint32_t memoryTypeBits, VkMemoryPropertyFlags properties)
17{
18 for (uint32_t i = 0; i < physicalDeviceMemoryProperties.memoryTypeCount; ++i)
19 {
20 if ((memoryTypeBits & (1 << i)) &&
21 ((physicalDeviceMemoryProperties.memoryTypes[i].propertyFlags & properties) == properties))
22 return i;
23 }
24 return uint32_t(-1);
25}

31bool Buffer::init(size_t size_in_bytes, const char* data)
32{
33 if (buffer_ != VK_NULL_HANDLE)
34 {
35 CV_LOG_WARNING(NULL, "Warn: Buffer object already inited!");
36 return false;
37 }
38
39 VkBufferCreateInfo bufferCreateInfo = {};
40 bufferCreateInfo.sType = VK_STRUCTURE_TYPE_BUFFER_CREATE_INFO;
41 bufferCreateInfo.size = (VkDeviceSize)size_in_bytes;
42 bufferCreateInfo.usage = usageFlag_;
43 bufferCreateInfo.sharingMode = VK_SHARING_MODE_EXCLUSIVE;
44 VK_CHECK_RESULT(vkCreateBuffer(kDevice, &bufferCreateInfo, NULL, &buffer_));
45
46 VkMemoryRequirements memoryRequirements;
47 vkGetBufferMemoryRequirements(kDevice, buffer_, &memoryRequirements);
48
49 VkMemoryAllocateInfo allocateInfo = {};
50 allocateInfo.sType = VK_STRUCTURE_TYPE_MEMORY_ALLOCATE_INFO;
51 allocateInfo.allocationSize = memoryRequirements.size;
52
53 // TODO: Try to optimize the memory at discrete graphics card. For AMD and GPU discrete graphics card,
54 // we should use VK_MEMORY_PROPERTY_DEVICE_LOCAL_BIT.
55
56 allocateInfo.memoryTypeIndex = findMemoryType(memoryRequirements.memoryTypeBits,
57 VK_MEMORY_PROPERTY_HOST_COHERENT_BIT |
58 VK_MEMORY_PROPERTY_HOST_VISIBLE_BIT
59 );
60 VK_CHECK_RESULT(vkAllocateMemory(kDevice, &allocateInfo, NULL, &memory_));
61
62 if (data)
63 {
64 char* dst;
65 VK_CHECK_RESULT(vkMapMemory(kDevice, memory_, 0, size_in_bytes, 0, (void **)&dst));
66 memcpy(dst, data, size_in_bytes);
67 vkUnmapMemory(kDevice, memory_);
68 }
69
70 VK_CHECK_RESULT(vkBindBufferMemory(kDevice, buffer_, memory_, 0));
71 return true;
72}
