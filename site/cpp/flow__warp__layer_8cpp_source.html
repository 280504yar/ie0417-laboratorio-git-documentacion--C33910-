39 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
40 {
41 CV_TRACE_FUNCTION();
42 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
43
44 std::vector<Mat> inputs, outputs;
45 inputs_arr.getMatVector(inputs);
46 outputs_arr.getMatVector(outputs);
47
48 const int out_n = outputs[0].size[0];
49 const int out_c = outputs[0].size[1];
50 const int out_h = outputs[0].size[2];
51 const int out_w = outputs[0].size[3];
52
53 const int area = out_w * out_h;
54 const int total = area * out_c;
55
56 const float* image_data = inputs[0].ptr<float>();
57 const float* flow_data = inputs[1].ptr<float>();
58 float* out_data = outputs[0].ptr<float>();
59
60 for (int n = 0; n < out_n; n++)
61 {
62 int off = total * n;
63 for (int x = 0; x < out_w; x++)
64 {
65 for (int y = 0; y < out_h; y++)
66 {
67 int idx = 2 * area * n + y * out_w + x;
68 float fx = flow_data[idx];
69 float fy = flow_data[idx + area];
70
71 float x2 = x + fx;
72 float y2 = y + fy;
73
74 if (x2 >= 0 && y2 >= 0 && x2 < out_w && y2 < out_h)
75 {
76 int ix2_L = x2;
77 float alpha = x2 - ix2_L;
78
79 int iy2_T = y2;
80 float beta = y2 - iy2_T;
81
82 int ix2_R = std::min(ix2_L + 1, out_w - 1);
83 int iy2_B = std::min(iy2_T + 1, out_h - 1);
84
85 for (int c = 0; c < out_c; c++)
86 {
87 float TL = image_data[off + c * area + iy2_T * out_w + ix2_L];
88 float TR = image_data[off + c * area + iy2_T * out_w + ix2_R];
89 float BL = image_data[off + c * area + iy2_B * out_w + ix2_L];
90 float BR = image_data[off + c * area + iy2_B * out_w + ix2_R];
91
92 out_data[off + c * area + y * out_w + x] = (1 - alpha) * (1 - beta) * TL +
93 (1 - alpha) * beta * BL +
94 alpha * (1 - beta) * TR +
95 alpha * beta * BR;
96 }
97 }
98 else
99 {
100 for (int c = 0; c < out_c; c++)
101 out_data[off + c * area + y * out_w + x] = fill_value;
102 }
103 }
104 }
105 }
106 }
