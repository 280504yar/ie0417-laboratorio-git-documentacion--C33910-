14// Utility function for safe integer conversions
15template <typename D, typename S>
16inline D safe_int_cast(S val, const char * msg = 0)
17{
18 typedef std::numeric_limits<S> st;
19 typedef std::numeric_limits<D> dt;
20 CV_StaticAssert(st::is_integer && dt::is_integer, "Integer type is expected");
21 const bool in_range_r = (double)val <= (double)dt::max();
22 const bool in_range_l = (double)val >= (double)dt::min();
23 if (!in_range_r || !in_range_l)
24 {
25 if (!msg)
26 CV_Error(Error::StsOutOfRange,
27 cv::format("Can not convert integer values (%s -> %s), value 0x%jx is out of range", typeid(S).name(), typeid(D).name(), (uintmax_t)val));
28 else
29 CV_Error(Error::StsOutOfRange, msg);
30 }
31 return static_cast<D>(val);
32}

56struct AviMainHeader
57{
58 uint32_t dwMicroSecPerFrame; // The period between video frames
59 uint32_t dwMaxBytesPerSec; // Maximum data rate of the file
60 uint32_t dwReserved1; // 0
61 uint32_t dwFlags; // 0x10 AVIF_HASINDEX: The AVI file has an idx1 chunk containing an index at the end of the file.
62 uint32_t dwTotalFrames; // Field of the main header specifies the total number of frames of data in file.
63 uint32_t dwInitialFrames; // Is used for interleaved files
64 uint32_t dwStreams; // Specifies the number of streams in the file.
65 uint32_t dwSuggestedBufferSize; // Field specifies the suggested buffer size forreading the file
66 uint32_t dwWidth; // Fields specify the width of the AVIfile in pixels.
67 uint32_t dwHeight; // Fields specify the height of the AVIfile in pixels.
68 uint32_t dwReserved[4]; // 0, 0, 0, 0
69};

291bool AVIReadContainer::parseIndex(uint32_t index_size, frame_list& in_frame_list)
292{
293 uint64_t index_end = m_file_stream->tellg();
294 index_end += index_size;
295 bool result = false;
296
297 while(m_file_stream && (m_file_stream->tellg() < index_end))
298 {
299 AviIndex idx1;
300 *m_file_stream >> idx1;
301
302 if(idx1.ckid == m_stream_id)
303 {
304 uint64_t absolute_pos = m_movi_start + idx1.dwChunkOffset;
305
306 if(absolute_pos < m_movi_end)
307 {
308 in_frame_list.push_back(std::make_pair(absolute_pos, idx1.dwChunkLength));
309 }
310 else
311 {
312 //unsupported case
313 fprintf(stderr, "Frame offset points outside movi section.\n");
314 }
315 }
316
317 result = true;
318 }
319
320 return result;
321}

323bool AVIReadContainer::parseStrl(char stream_id, Codecs codec_)
324{
325 RiffChunk strh;
326 *m_file_stream >> strh;
327
328 if(m_file_stream && strh.m_four_cc == STRH_CC)
329 {
330 AviStreamHeader strm_hdr;
331 *m_file_stream >> strm_hdr;
332
333 if (codec_ == MJPEG)
334 {
335 if(strm_hdr.fccType == VIDS_CC && strm_hdr.fccHandler == MJPG_CC)
336 {
337 uint8_t first_digit = (stream_id/10) + '0';
338 uint8_t second_digit = (stream_id%10) + '0';
339
340 if(m_stream_id == 0)
341 {
342 m_stream_id = CV_FOURCC(first_digit, second_digit, 'd', 'c');
343 m_fps = double(strm_hdr.dwRate)/strm_hdr.dwScale;
344 }
345 else
346 {
347 //second mjpeg video stream found which is not supported
348 fprintf(stderr, "More than one video stream found within AVI/AVIX list. Stream %c%cdc would be ignored\n", first_digit, second_digit);
349 }
350
351 return true;
352 }
353 }
354 }
355
356 return false;
357}

378bool AVIReadContainer::parseHdrlList(Codecs codec_)
379{
380 bool result = false;
381
382 RiffChunk avih;
383 *m_file_stream >> avih;
384
385 if(m_file_stream && avih.m_four_cc == AVIH_CC)
386 {
387 uint64_t next_strl_list = m_file_stream->tellg();
388 next_strl_list += avih.m_size;
389
390 AviMainHeader avi_hdr;
391 *m_file_stream >> avi_hdr;
392
393 if(m_file_stream)
394 {
395 m_is_indx_present = ((avi_hdr.dwFlags & 0x10) != 0);
396 uint32_t number_of_streams = avi_hdr.dwStreams;
397 CV_Assert(number_of_streams < 0xFF);
398 m_width = avi_hdr.dwWidth;
399 m_height = avi_hdr.dwHeight;
400
401 //the number of strl lists must be equal to number of streams specified in main avi header
402 for(uint32_t i = 0; i < number_of_streams; ++i)
403 {
404 m_file_stream->seekg(next_strl_list);
405 RiffList strl_list;
406 *m_file_stream >> strl_list;
407
408 if( m_file_stream && strl_list.m_riff_or_list_cc == LIST_CC && strl_list.m_list_type_cc == STRL_CC )
409 {
410 next_strl_list = m_file_stream->tellg();
411 //RiffList::m_size includes fourCC field which we have already read
412 next_strl_list += (strl_list.m_size - 4);
413
414 result = parseStrl((char)i, codec_);
415 }
416 else
417 {
418 printError(strl_list, STRL_CC);
419 }
420 }
421 }
422 }
423 else
424 {
425 printError(avih, AVIH_CC);
426 }
427
428 return result;
429}

431bool AVIReadContainer::parseAviWithFrameList(frame_list& in_frame_list, Codecs codec_)
432{
433 RiffList hdrl_list;
434 *m_file_stream >> hdrl_list;
435
436 if( m_file_stream && hdrl_list.m_riff_or_list_cc == LIST_CC && hdrl_list.m_list_type_cc == HDRL_CC )
437 {
438 uint64_t next_list = m_file_stream->tellg();
439 //RiffList::m_size includes fourCC field which we have already read
440 next_list += (hdrl_list.m_size - 4);
441 //parseHdrlList sets m_is_indx_present flag which would be used later
442 if(parseHdrlList(codec_))
443 {
444 m_file_stream->seekg(next_list);
445
446 RiffList some_list;
447 *m_file_stream >> some_list;
448
449 //an optional section INFO
450 if(m_file_stream && some_list.m_riff_or_list_cc == LIST_CC && some_list.m_list_type_cc == INFO_CC)
451 {
452 next_list = m_file_stream->tellg();
453 //RiffList::m_size includes fourCC field which we have already read
454 next_list += (some_list.m_size - 4);
455 parseInfo();
456
457 m_file_stream->seekg(next_list);
458 *m_file_stream >> some_list;
459 }
460
461 //an optional section JUNK
462 skipJunk(some_list);
463
464 //we are expecting to find here movi list. Must present in avi
465 if(m_file_stream && some_list.m_riff_or_list_cc == LIST_CC && some_list.m_list_type_cc == MOVI_CC)
466 {
467 bool is_index_found = false;
468
469 m_movi_start = m_file_stream->tellg();
470 m_movi_start -= 4;
471
472 m_movi_end = m_movi_start + some_list.m_size;
473 //if m_is_indx_present is set to true we should find index
474 if(m_is_indx_present)
475 {
476 //we are expecting to find index section after movi list
477 uint32_t indx_pos = (uint32_t)m_movi_start + 4;
478 indx_pos += (some_list.m_size - 4);
479 m_file_stream->seekg(indx_pos);
480
481 RiffChunk index_chunk;
482 *m_file_stream >> index_chunk;
483
484 if(m_file_stream && index_chunk.m_four_cc == IDX1_CC)
485 {
486 is_index_found = parseIndex(index_chunk.m_size, in_frame_list);
487 //we are not going anywhere else
488 }
489 else
490 {
491 printError(index_chunk, IDX1_CC);
492 }
493 }
494 //index not present or we were not able to find it
495 //parsing movi list
496 if(!is_index_found)
497 {
498 //not implemented
499 parseMovi(in_frame_list);
500
501 fprintf(stderr, "Failed to parse avi: index was not found\n");
502 //we are not going anywhere else
503 }
504 }
505 else
506 {
507 printError(some_list, MOVI_CC);
508 }
509 }
510 }
511 else
512 {
513 printError(hdrl_list, HDRL_CC);
514 }
515
516 return in_frame_list.size() > 0;
517}

519std::vector<char> AVIReadContainer::readFrame(frame_iterator it)
520{
521 m_file_stream->seekg(it->first);
522
523 RiffChunk chunk;
524 *(m_file_stream) >> chunk;
525
526 // Assertion added to prevent complaints from static analysis tools
527 // as the chunk size is read from a file then used to allocate
528 // memory. 64MB was chosen arbitrarily as an upper bound but it may
529 // be useful to make it configurable.
530 CV_Assert(chunk.m_size <= 67108864);
531
532 std::vector<char> result;
533
534 result.reserve(chunk.m_size);
535 result.resize(chunk.m_size);
536
537 m_file_stream->read(&(result[0]), chunk.m_size); // result.data() failed with MSVS2008
538
539 return result;
540}

542bool AVIReadContainer::parseRiff(frame_list &m_mjpeg_frames_)
543{
544 bool result = false;
545 while(*m_file_stream)
546 {
547 RiffList riff_list;
548
549 *m_file_stream >> riff_list;
550
551 if( *m_file_stream && riff_list.m_riff_or_list_cc == RIFF_CC &&
552 ((riff_list.m_list_type_cc == AVI_CC) | (riff_list.m_list_type_cc == AVIX_CC)) )
553 {
554 uint64_t next_riff = m_file_stream->tellg();
555 //RiffList::m_size includes fourCC field which we have already read
556 next_riff += (riff_list.m_size - 4);
557
558 bool is_parsed = parseAvi(m_mjpeg_frames_, MJPEG);
559 result = result || is_parsed;
560 m_file_stream->seekg(next_riff);
561 }
562 else
563 {
564 break;
565 }
566 }
567 return result;
568}

667{

688void BitStream::putBytes(const uchar* buf, int count)
689{
690 uchar* data = (uchar*)buf;
691 CV_Assert(data && m_current && count >= 0);
692 if( m_current >= m_end )
693 writeBlock();
694
695 while( count )
696 {
697 int l = (int)(m_end - m_current);
698
699 if (l > count)
700 l = count;
701
702 if( l > 0 )
703 {
704 memcpy(m_current, data, l);
705 m_current += l;
706 data += l;
707 count -= l;
708 }
709 if( m_current >= m_end )
710 writeBlock();
711 }
712}

743void BitStream::patchInt(uint32_t val, size_t pos)
744{
745 if( pos >= m_pos )
746 {
747 ptrdiff_t delta = safe_int_cast<ptrdiff_t>(pos - m_pos, "Failed to seek in AVI buffer: value is out of range");
748 CV_Assert( delta < m_current - m_start );
749 m_start[delta] = (uchar)val;
750 m_start[delta+1] = (uchar)(val >> 8);
751 m_start[delta+2] = (uchar)(val >> 16);
752 m_start[delta+3] = (uchar)(val >> 24);
753 }
754 else
755 {
756 std::streamoff fpos = output.tellp();
757 output.seekp(safe_int_cast<std::streamoff>(pos, "Failed to seek in AVI file: value is out of range"));
758 uchar buf[] = { (uchar)val, (uchar)(val >> 8), (uchar)(val >> 16), (uchar)(val >> 24) };
759 output.write((char *)buf, 4);
760 output.seekp(fpos);
761 }
762}

764void BitStream::jput(unsigned currval)
765{
766 uchar v;
767 uchar* ptr = m_current;
768 v = (uchar)(currval >> 24);
769 *ptr++ = v;
770 if( v == 255 )
771 *ptr++ = 0;
772 v = (uchar)(currval >> 16);
773 *ptr++ = v;
774 if( v == 255 )
775 *ptr++ = 0;
776 v = (uchar)(currval >> 8);
777 *ptr++ = v;
778 if( v == 255 )
779 *ptr++ = 0;
780 v = (uchar)currval;
781 *ptr++ = v;
782 if( v == 255 )
783 *ptr++ = 0;
784 m_current = ptr;
785 if( m_current >= m_end )
786 writeBlock();
787}

789void BitStream::jflush(unsigned currval, int bitIdx)
790{
791 uchar v;
792 uchar* ptr = m_current;
793 currval |= (1 << bitIdx)-1;
794 while( bitIdx < 32 )
795 {
796 v = (uchar)(currval >> 24);
797 *ptr++ = v;
798 if( v == 255 )
799 *ptr++ = 0;
800 currval <<= 8;
801 bitIdx += 8;
802 }
803 m_current = ptr;
804 if( m_current >= m_end )
805 writeBlock();
806}

903 strm->putInt(width);
904 strm->putInt(height);
905 strm->putShort(1); // planes (1 means interleaved data (after decompression))
906
907 strm->putShort(8 * channels); // bits per pixel
908 switch (codec_) {
909 case MJPEG:

911 break;
912 }
913 strm->putInt(width * height * channels);
914 strm->putInt(0);
915 strm->putInt(0);
916 strm->putInt(0);
917 strm->putInt(0);
918 // Must be indx chunk
919 endWriteChunk(); // end strf
920
921 endWriteChunk(); // end strl
922
923 // odml
924 startWriteChunk(LIST_CC);
925 strm->putInt(ODML_CC);
926 startWriteChunk(DMLH_CC);
927
928 frameNumIndexes.push_back(strm->getPos());
929
930 strm->putInt(0);
931 strm->putInt(0);
932
933 endWriteChunk(); // end dmlh
934 endWriteChunk(); // end odml
935
936 endWriteChunk(); // end hdrl
937
938 // JUNK
939 startWriteChunk(JUNK_CC);
940 size_t pos = strm->getPos();
941 for( ; pos < (size_t)JUNK_SEEK; pos += 4 )
942 strm->putInt(0);
943 endWriteChunk(); // end JUNK
