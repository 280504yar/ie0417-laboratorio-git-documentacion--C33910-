264 CV_CheckType(hostMatType, hostMatType == CV_32F || hostMatType == CV_16F || hostMatType == CV_8S || hostMatType == CV_8U || hostMatType == CV_32S || hostMatType == CV_64S, "");
265

272 else if (hostMatType == CV_64S)
273 return Ptr<BackendNode>(new NodeType<int64_t>(std::forward<Args>(args)...));
274 else if (hostMatType == CV_16F) // device tensor already stored as half (FP16 target)
275 return Ptr<BackendNode>(new NodeType<half>(std::forward<Args>(args)...));
276 else if (hostMatType == CV_32F)
277 {
278 if (targetId == DNN_TARGET_CUDA_FP16)
279 return Ptr<BackendNode>(new NodeType<half>(std::forward<Args>(args)...));
280 else if (targetId == DNN_TARGET_CUDA)
281 return Ptr<BackendNode>(new NodeType<float>(std::forward<Args>(args)...));
282 }

394 void convert_D2H_background(const cv::Mat& mat, cuda4dnn::csl::View<DEVICE_T> view, cuda4dnn::csl::ManagedPtr<HOST_T>& device_temp, const cuda4dnn::csl::Stream& stream, const cuda4dnn::csl::Stream& d2h_stream, cuda4dnn::csl::Event& d2h_event);

396 template <> inline
397 void convert_D2H_background<half, float>(const cv::Mat& mat, cuda4dnn::csl::View<half> view, cuda4dnn::csl::ManagedPtr<float>& device_temp, const cuda4dnn::csl::Stream& stream, const cuda4dnn::csl::Stream& d2h_stream, cuda4dnn::csl::Event& d2h_event) {
398 if (device_temp.size() < view.size())
399 device_temp.reset(view.size());
400 auto temp_span = cuda4dnn::csl::Span<float>(device_temp.get(), view.size());
401
402 /* The conversion kernel should can be executed in the background stream for better
403 * performance. We do it in the inference stream to prevent an unexplained performance
404 * regression on RTX 2080 Ti. Executing conversion kernel in the background stream causes
405 * everything to slow down (even operations that appear before the background transfer).
406 *
407 * TODO: identify the cause and move conversion kernel to the background stream
408 */
409 cuda4dnn::kernels::fp16_to_fp32(stream, temp_span, view);
410
411 d2h_event.record(stream); // mark position in inference stream
412 cuda4dnn::csl::StreamWaitOnEvent(d2h_stream, d2h_event); // don't start transfer until data is available
413 cuda4dnn::csl::memcpy<float>(reinterpret_cast<float*>(mat.data), temp_span.data(), view.size(), d2h_stream);
414 }

509 /* Pre-conditions:
510 * - there must be no other instance of `GenericCUDABackendWrapper` which wraps the host memory used by `m`
511 * - the host memory must remain allocated throughout the lifetime of this object

513 * Post-conditions:

516 GenericCUDABackendWrapper(Mat& m)
517 : CUDABackendWrapper(TargetID)
518 {
519 shape = cv::dnn::shape(m);
520 hostMatDepth = m.depth();
521 offset = 0;
522
523 shared_block = std::make_shared<shared_block_type>();
524 shared_block->host_dirty = true;
525 shared_block->device_dirty = false;
526
527 shared_block->host = m;
528
529 try {
530 shared_block->memGuard = cuda4dnn::csl::MemoryLockGuard(m.data, m.total() * m.elemSize());
531 } catch (...) {
532 /* a common reason for failure is that the host system (for example, a Jetson device) does not support it */
533 /* we ignore the failure as this is just an optimization and not a requirement */
534 }
535
536 shared_block->device = cuda4dnn::csl::ManagedPtr<DEVICE_T>(m.total());
537 }

539 GenericCUDABackendWrapper(const Ptr<BackendWrapper>& base_, const MatShape& shape_)
540 : CUDABackendWrapper(TargetID)
541 {
542 const Ptr<GenericCUDABackendWrapper> base = base_.dynamicCast<GenericCUDABackendWrapper>();
543 CV_Assert(base);
544
545 shape = shape_;
546 hostMatDepth = base_->getHostMatDepth();
547 offset = 0;
548 shared_block = base->shared_block;
549
550 auto numel = total(shape_);
551 if (numel > shared_block->device.size())
552 {
553 /* if the host memory was already page-locked, release it and register again with the new size */
554 shared_block->memGuard = cuda4dnn::csl::MemoryLockGuard();
555 try {
556 CV_Assert(shared_block->host.type() == CV_32F);
557 shared_block->memGuard = cuda4dnn::csl::MemoryLockGuard(shared_block->host.data, numel * sizeof(float));
558 } catch (...) {
559 /* a common reason for failure is that the host system (for example, a Jetson device) does not support it */
560 /* we ignore the failure as this is just an optimization and not a requirement */
561 }
562 shared_block->device.reset(numel);
563 }

574 void copyToHost() override {
575 if (shared_block->device_dirty) {
576 CV_Assert(offset == 0); /* we cannot track each piece of the memory separately */
577
578 shared_block->host_dirty = false;
579 shared_block->device_dirty = false;
580
581 /* If the wrapper is being reused, the device tensor might be larger in size than the wrapper.
582 * Using the device tensor does not give incorrect code but leads to unused region of memory being copied.
583 *
584 * We use a view to ensure that only the required region of memory is copied.
585 */
586 auto view = tensor_view_type(shared_block->device.get(), std::begin(shape), std::end(shape));
587
588 auto& mat = shared_block->host;
589 CV_Assert(mat.isContinuous());
590
591 cuda4dnn::detail::convert_D2H<DEVICE_T, HOST_T>(mat, view, shared_block->device_temp, shared_block->stream);
592 shared_block->stream.synchronize();
593 } else if(shared_block->d2h_event && shared_block->d2h_event.busy()) {
594 /* wait for the background copy to finish */
595 shared_block->d2h_event.synchronize();
596 }
597 }

599 void copyToHostInBackground() override {
600 CV_Assert(shared_block->d2h_stream);
601 if (shared_block->device_dirty) {
602 shared_block->host_dirty = false;
603 shared_block->device_dirty = false;
604
605 auto view = tensor_view_type(shared_block->device.get(), std::begin(shape), std::end(shape));
606
607 auto& mat = shared_block->host;
608 CV_Assert(mat.isContinuous());
609
610 if (!shared_block->d2h_event)
611 shared_block->d2h_event = cuda4dnn::csl::Event(true);
612 cuda4dnn::detail::convert_D2H_background<DEVICE_T, HOST_T>(mat, view, shared_block->device_temp, shared_block->stream, shared_block->d2h_stream, shared_block->d2h_event);
613 shared_block->d2h_event.record(shared_block->d2h_stream); // record position so that we can check status later
614 }

674 /* Optimization Note: use getSpan() and getView() judiciously
675 *
676 * getSpan() is meant to be used when the memory is going to be modified
677 * getView() is meant to be used when the memory is only going to be read
678 *
679 * getSpan() marks the device memory as dirty but getView() does not
680 *
681 * getView() implicitly performs host to device memory transfer if required
682 * getSpan() does not perform any synchronization (use copyToDevice if sync. is required)
683 */
684 tensor_span_type getSpan() noexcept {
685 setDeviceDirty();

690 copyToDevice();
