5// Implementation of cv::BroadcastOp (declared in opencv2/core/mat.hpp): an op-agnostic driver for
6// broadcasting element-wise traversal. It computes the numpy-broadcast iteration space over a flat
7// list of operand Mats, collapses contiguous dims, partitions into tiles for parallel_for_, and hands
8// each 2D tile's per-operand slices to a `body` callback (which owns all semantics).

18// Per-operand steps along the (collapsed) iteration axes are kept in a MatStep (a value-type holding
19// MAX_DIMS size_t entries). Here the entries are steps in ELEMENTS (one scalar / channel value), not
20// the byte steps a Mat stores - the container is reused for its fixed-size storage and [] access.

26// How a Mat's channels are mapped into the logical (shape, step, esz1) handed to the geometry:
27// CH_FOLD : channels stay scalar-wise, folded into the innermost dim (back() *= cn, step 1).
28// Used (expandChannels=true) when no channel broadcast is needed (all single-channel,
29// or all same-cn with equal back()) - the body then sees single-channel data.
30// CH_DIM : channels become an explicit innermost iteration dim (cn, step 1); single-channel
31// operands get a size-1 channel that broadcasts 1->N. Used when channel broadcast is
32// needed (mixed channel counts, or multichannel with differing back()).
33// CH_ELEM : channels stay inside the element (esz = full elemSize, no channel dim). Used with
34// expandChannels=false; the body handles channels itself (deinterleave fast path).
35enum ChMode { CH_FOLD, CH_DIM, CH_ELEM };
36
37static void matLayout(const Mat& m, ChMode mode, MatShape& shp, MatStep& step, int& esz1)
38{
39 const int nd = m.dims, cn = m.channels();
40 if (mode == CH_ELEM)
41 {
42 esz1 = (int)m.elemSize(); // one full (cn-channel) pixel
43 shp.resize(nd);
44 for (int i = 0; i < nd; i++) { shp[i] = m.size[i]; step[i] = m.step[i] / esz1; }
45 return;
46 }
47 esz1 = (int)m.elemSize1(); // one scalar (channel value)
48 if (mode == CH_DIM)
49 {
50 shp.resize(nd + 1);
51 for (int i = 0; i < nd; i++) { shp[i] = m.size[i]; step[i] = m.step[i] / esz1; }
52 shp[nd] = cn; step[nd] = 1; // channels = explicit innermost dim
53 }
54 else // CH_FOLD
55 {
56 if (nd == 0) // 0-dim scalar: channels are the only dim
57 {
58 shp.assign(1, cn);
59 step[0] = 1;
60 }
61 else
62 {
63 shp.resize(nd);
64 for (int i = 0; i < nd; i++) { shp[i] = m.size[i]; step[i] = m.step[i] / esz1; }
65 shp[nd - 1] *= cn; // fold channels into the innermost dim
66 step[nd - 1] = 1; // scalars are contiguous there
67 }
68 }
69}

71// A single-channel scalar (one value, cn==1, total()==1) broadcasts into everything trivially
72// (step 0 on every axis incl. channels), so it must NOT force CH_DIM - it is excluded from the
73// channel-mode decision entirely.
74static bool isSingleChannelScalar(const Mat& m)
75{
76 return m.channels() == 1 && m.total() == 1;
77}
78
79// Decide (globally, across all operands) how channels are presented for expandChannels=true:
80// CH_FOLD when no channel broadcast is needed, CH_DIM when it is. Single-channel scalars are
81// excluded first; among the rest, multichannel operands must all share the same cn (an (n,m) mix
82// with both > 1 is an error).
83static ChMode decideChannelMode(const Mat* const* arrays, int K)
84{
85 int N = 1; // the single multichannel count, if any
86 for (int k = 0; k < K; k++)
87 {
88 if (isSingleChannelScalar(*arrays[k])) continue;
89 int c = arrays[k]->channels();
90 if (c > 1) { if (N == 1) N = c; else CV_Assert(N == c && "ew: (n,m) channel mix unsupported"); }
91 }
92 if (N == 1) return CH_FOLD; // all single-channel -> fold (a no-op)
93
94 bool allMulti = true, sameBack = true;
95 int back = -1;
96 for (int k = 0; k < K; k++)
97 {
98 const Mat& a = *arrays[k];
99 if (isSingleChannelScalar(a)) continue;
100 if (a.channels() != N) allMulti = false;
101 int b = a.dims > 0 ? a.size[a.dims - 1] : 1; // 0-dim scalar has no spatial back (=1)
102 if (back < 0) back = b; else if (b != back) sameBack = false;
103 }
104 return (allMulti && sameBack) ? CH_FOLD : CH_DIM; // fold only if no channel broadcast
105}

107// numpy-style broadcast of several right-aligned shapes.
108static bool broadcastShape(const MatShape* shps, int K, MatShape& out)
109{
110 size_t nd = 0;
111 for (int k = 0; k < K; k++) nd = std::max(nd, shps[k].size());
112 out.assign(nd, 1);
113 for (int k = 0; k < K; k++)
114 {
115 const MatShape& s = shps[k];
116 size_t off = nd - s.size();
117 for (size_t i = 0; i < s.size(); i++)
118 {
119 int d = s[i], &o = out[off + i];
120 if (o == 1) o = d;
121 else if (d != 1 && d != o) return false;
122 }
123 }
124 return true;
125}

127// Right-align an arg's own (shp,step) to nd dims; broadcast dims get step 0.
128static void alignArg(const MatShape& shp, const MatStep& step, int nd,
129 MatStep& as, MatShape& ash)
130{
131 as.clear();
132 ash.assign(nd, 1);
133 int off = nd - (int)shp.size();
134 for (int i = 0; i < (int)shp.size(); i++)
135 {
136 int d = shp[i];
137 ash[off + i] = d;
138 as[off + i] = (d == 1) ? 0 : step[i];
139 }
140}

142// Collapse adjacent dims that are contiguous (and broadcast-consistent) across all args.
143static int collapseDims(MatStep* S, MatShape* H, int K, MatShape& D)
144{
145 int nd = (int)D.size();
146 if (nd <= 1) return nd;
147
148 int j = nd - 1;
149 for (int i = j - 1; i >= 0; i--)
150 {
151 bool contig = true, scalar = true, consist = true;
152 for (int k = 0; k < K; k++)
153 {
154 size_t st = S[k][j] * (size_t)H[k][j];
155 bool prevScalar = H[k][j] == 1;
156 bool curScalar = H[k][i] == 1;
157 contig = contig && (st == S[k][i]);
158 scalar = scalar && curScalar;
159 consist = consist && (curScalar == prevScalar);
160 }
161 if (contig && (consist || scalar))
162 {
163 for (int k = 0; k < K; k++) H[k][j] *= H[k][i];
164 D[j] *= D[i];
165 }
166 else
167 {
168 j--;
169 if (i < j)
170 {
171 for (int k = 0; k < K; k++) { H[k][j] = H[k][i]; S[k][j] = S[k][i]; }
172 D[j] = D[i];
173 }
174 }
175 }
176
177 int m = nd - j;
178 for (int d = 0; d < m; d++)
179 {
180 D[d] = D[j + d];
181 for (int k = 0; k < K; k++) { S[k][d] = S[k][j + d]; H[k][d] = H[k][j + d]; }
182 }
183 D.resize(m);
184 for (int k = 0; k < K; k++) H[k].resize(m);
185
186 // Zero out steps of broadcast (size-1) dims (numpy step==0 trick).
187 for (int d = 0; d < m; d++)
188 for (int k = 0; k < K; k++)
189 if (H[k][d] == 1) S[k][d] = 0;
190
191 return m;
192}

194// Fast geometry for the dominant case: every operand is either (a) an array sharing ONE common
195// shape - same dims, sizes and channel count - and contiguous, or (b) a single-channel scalar
196// (cn==1, total()==1). Then the whole traversal is a single contiguous 1D run of `total` scalars
197// (channels folded in): arrays get stepx 1, scalars stepx 0. This skips decideChannelMode /
198// broadcastShape / alignArg / collapseDims and all their per-operand buffers entirely. Returns
199// false (leaving outputs untouched) when the operands don't fit, so the caller runs general
200// geometry. expandChannels=false (CH_ELEM) keeps channels in the element and is left to general.
201static bool fastSameShape(const Mat* const* arrays, int K, bool expandChannels,
202 uchar** base, int* esz1, MatStep* S, MatShape& D, int& m)
203{
204 if (!expandChannels) return false;
205 int ref = -1;
206 for (int k = 0; k < K; k++)
207 if (!isSingleChannelScalar(*arrays[k])) { ref = k; break; }
208 if (ref < 0) return false; // all single-channel scalars: let general handle
209 const Mat& R = *arrays[ref];
210 const int rdims = R.dims, rcn = R.channels();
211 for (int k = 0; k < K; k++)
212 {
213 const Mat& a = *arrays[k];
214 if (isSingleChannelScalar(a)) continue;
215 if (a.channels() != rcn || a.dims != rdims || !a.isContinuous()) return false;
216 for (int i = 0; i < rdims; i++) if (a.size[i] != R.size[i]) return false;
217 }
218 const long long total = (long long)R.total() * rcn; // channels folded into the 1D run
219 CV_Assert(total <= (long long)INT_MAX);
220 for (int k = 0; k < K; k++)
221 {
222 const Mat& a = *arrays[k];
223 base[k] = (uchar*)a.data;
224 esz1[k] = (int)a.elemSize1();
225 S[k][0] = isSingleChannelScalar(a) ? 0 : 1;
226 }
227 D.assign(1, (int)total);
228 m = 1;
229 return true;
230}

236// constants used as template arguments inside a lambda (C2975).
237static constexpr int MAX_DIMS = MatShape::MAX_DIMS;
238static constexpr int LOCAL_OPS = 8;
239
240void BroadcastOp::run(const Mat* const* arrays, int narrays,
241 const std::function<void(const Tile&)>& body,
242 bool expandChannels,
243 double nstripes)
244{
245 const int K = narrays;
246 CV_Assert(K >= 1 && arrays != nullptr);
247
248 // ---- 1-3. geometry: per-operand collapsed steps S[k], element sizes esz1[k], base
249 // pointers, and the collapsed iteration shape D (m dims). The fast path handles the
250 // dominant "all same-shape arrays (+ single-channel scalars)" case in one shot; the
251 // general path does decideChannelMode + broadcastShape + align + collapse. ----
252 AutoBuffer<MatStep, LOCAL_OPS> S(K);
253 AutoBuffer<int, LOCAL_OPS> esz1(K);
254 AutoBuffer<uchar*, LOCAL_OPS> base(K);
255 MatShape D;
256 int m;
257
258 if (!fastSameShape(arrays, K, expandChannels, base.data(), esz1.data(), S.data(), D, m))
259 {
260 const ChMode mode = expandChannels ? decideChannelMode(arrays, K) : CH_ELEM;
261 AutoBuffer<MatShape, LOCAL_OPS> shp(K);
262 AutoBuffer<MatStep, LOCAL_OPS> stp(K);
263 for (int k = 0; k < K; k++)
264 {
265 matLayout(*arrays[k], mode, shp[k], stp[k], esz1[k]);
266 base[k] = (uchar*)arrays[k]->data;
267 }
268 MatShape full;
269 CV_Assert(broadcastShape(shp.data(), K, full) && "ew: operands are not broadcast-compatible");
270 const int nd = (int)full.size();
271 AutoBuffer<MatShape, LOCAL_OPS> H(K);
272 for (int k = 0; k < K; k++) alignArg(shp[k], stp[k], nd, S[k], H[k]);
273 D = full;
274 m = collapseDims(S.data(), H.data(), K, D);
275
276 // For a cv::Mat the innermost (channel/last) axis is contiguous, so after collapse the
277 // innermost stride is always in {0,1}. No gather, no materialization.
278 for (int k = 0; k < K; k++)
279 CV_Assert(S[k][m - 1] <= 1 && "ew: unexpected innermost stride > 1");
280 }
281
282 // ---- 4. inner 2D tile axes: width = D[m-1], height = D[m-2] (if any) ----
283 const int wAxis = m - 1;
284 const int hAxis = (m >= 2) ? m - 2 : -1;
285 const int W = D[wAxis];
286 const int Hgt = (hAxis >= 0) ? D[hAxis] : 1;
287 const int nOuter = (hAxis >= 0) ? m - 2 : m - 1; // outer ("plane") axes = [0 .. nOuter)
288 long long nplanes = 1;
289 for (int d = 0; d < nOuter; d++) nplanes *= D[d];
290
291 // ---- 5. desired parallel stripe count (work hint) ----
292 const long long total = nplanes * (long long)Hgt * (long long)W;
293 double stripes = nstripes;
294 if (stripes <= 0) // broadcastOp can't see the body's cost;
295 stripes = (double)total * 100.0 / (double)(1 << 18); // assume ~100 cycles/element
296 const int wantTiles = std::max(1, (int)std::lround(stripes));
297
298 // ---- 6. tile only for PARALLELISM. broadcastOp is op-agnostic: it does not know the body's
299 // temp-buffer footprint, so it does NOT tile for L1 - that is the body's job (it
300 // re-fragments a tile's width into L1-sized chunks for the fused intermediates).
301 // Start with the largest tile (one 2D block per plane) and split (height first, then
302 // width) only until there are at least `wantTiles` tiles. Bigger tiles => fewer
303 // body/decode calls. Width is G-aligned only in the fully-contiguous (1D) case. ----
304 const int G = 16; // SIMD/cacheline granule
305 int tw = W, th = Hgt;
306
307 auto ntilesOf = [&](int tw_, int th_) {
308 long long nw = (W + tw_ - 1) / tw_, nh = (Hgt + th_ - 1) / th_;
309 return nplanes * nh * nw;
310 };
311 long long ntiles = ntilesOf(tw, th);
312 while (ntiles < wantTiles && th > 1) // split height for parallelism
313 {
314 th = (th + 1) / 2;
315 ntiles = ntilesOf(tw, th);
316 }
317 while (ntiles < wantTiles && tw > G) // then split width
318 {
319 tw = std::max(G, tw / 2);
320 if (hAxis < 0 && tw > G) tw -= tw % G; // keep width aligned in the 1D case
321 ntiles = ntilesOf(tw, th);
322 }
323 CV_Assert(ntiles <= (long long)INT_MAX);
324
325 const int ntilesW = (W + tw - 1) / tw;
326 const int ntilesH = (Hgt + th - 1) / th;
327
328 // ---- 7. execution; decode tile index -> per-operand slices. stepx/stepy are the same for
329 // every tile, so they are set ONCE; only the per-tile base pointer is recomputed. ----
330 auto runRange = [&](const Range& r)
331 {
332 AutoBuffer<Slice, LOCAL_OPS> slices(K);
333
334 // Fast 1D path (m==1: one contiguous axis after collapse, no outer planes, height 1).
335 // ntilesH==1 and nplanes==1, so the tile index IS the width-tile index - no div/mod, no
336 // plane multi-index decode, no inner step loop. This is the same-shape / fully-contiguous
337 // common case.
338 if (m == 1)
339 {
340 for (int k = 0; k < K; k++) { slices[k].stepy = 0; slices[k].stepx = S[k][0]; }
341 Tile tile;
342 tile.height = 1; tile.narrays = K; tile.slices = slices.data();
343 for (int t = r.start; t < r.end; t++)
344 {
345 const int wofs = t * tw, ww = std::min(tw, W - wofs);
346 for (int k = 0; k < K; k++)
347 slices[k].ptr = base[k] + (size_t)wofs * S[k][0] * (size_t)esz1[k];
348 tile.width = ww;
349 body(tile);
350 }
351 return;
352 }
353
354 std::array<int, MAX_DIMS> idx;
355 for (int k = 0; k < K; k++) // steps are tile-independent: set once
356 {
357 slices[k].stepy = (hAxis >= 0) ? S[k][hAxis] : 0;
358 slices[k].stepx = S[k][wAxis];
359 }
360 for (int t = r.start; t < r.end; t++)
361 {
362 int wt = t % ntilesW;
363 int rest = t / ntilesW;
364 int ht = rest % ntilesH;
365 int plane = rest / ntilesH;
366
367 const int wofs = wt * tw, ww = std::min(tw, W - wofs);
368 const int hofs = ht * th, hh = std::min(th, Hgt - hofs);
369
370 int p = plane; // decode plane -> outer multi-index
371 for (int d = nOuter - 1; d >= 0; d--) {
372 int dd = D[d];
373 int np = p / dd;
374 idx[d] = p - np * dd;
375 p = np;
376 }
377
378 for (int k = 0; k < K; k++)
379 {
380 size_t off = (size_t)wofs * S[k][wAxis];
381 for (int d = 0; d < nOuter; d++) off += (size_t)idx[d] * S[k][d];
382 if (hAxis >= 0) off += (size_t)hofs * S[k][hAxis];
383 slices[k].ptr = base[k] + off * (size_t)esz1[k];
384 }
385
386 Tile tile;
387 tile.width = ww; tile.height = hh; tile.narrays = K; tile.slices = slices.data();
388 body(tile);
389 }
390 };
391
392 // Single tile (small work, wantTiles==1) => run inline and skip the parallel framework
393 // entirely: its dispatch (std::function wrap + Range machinery + backend hop) is pure
394 // overhead when there is nothing to parallelize, and dominates small-array latency.
395 if (ntiles == 1)
396 runRange(Range(0, 1));
397 else
398 parallel_for_(Range(0, (int)ntiles), runRange, stripes);
399}
