98 GpuMat createMat(Size size, int type, Size& size0, Point& ofs, bool useRoi)
99 {
100 size0 = size;
101
102 if (useRoi)
103 {
104 size0.width += randomInt(5, 15);
105 size0.height += randomInt(5, 15);
106 }
107
108 GpuMat d_m(size0, type);
109 if (size0 != size) {
110 ofs = Point((size0.width - size.width) / 2, (size0.height - size.height) / 2);
111 d_m = d_m(Rect(ofs, size));
112 }
113
114 return d_m;
115 }

159 void DeviceManager::load(int i)

162 devices_.reserve(1);
163
164 std::ostringstream msg;
165
166 if (i < 0 || i >= getCudaEnabledDeviceCount())

172 DeviceInfo info(i);

175 {
176 msg << "Device " << i << " [" << info.name() << "] is NOT compatible with current CUDA module build";

179
180 devices_.push_back(info);

232 template <typename T, typename OutT> std::string printMatValImpl(const Mat& m, Point p)
233 {
234 const int cn = m.channels();
235
236 std::ostringstream ostr;
237 ostr << "(";
238
239 p.x /= cn;
240
241 ostr << static_cast<OutT>(m.at<T>(p.y, p.x * cn));
242 for (int c = 1; c < m.channels(); ++c)
243 {
244 ostr << ", " << static_cast<OutT>(m.at<T>(p.y, p.x * cn + c));
245 }
246 ostr << ")";
247
248 return ostr.str();
249 }

251 std::string printMatVal(const Mat& m, Point p)
252 {
253 typedef std::string (*func_t)(const Mat& m, Point p);

265 void minMaxLocGold(const Mat& src, double* minVal_, double* maxVal_, Point* minLoc_, Point* maxLoc_, const Mat& mask)
266 {
267 if (src.depth() != CV_8S)
268 {
269 minMaxLoc(src, minVal_, maxVal_, minLoc_, maxLoc_, mask);
270 return;
271 }
272
273 // OpenCV's minMaxLoc doesn't support CV_8S type
274 double minVal = std::numeric_limits<double>::max();
275 Point minLoc(-1, -1);
276
277 double maxVal = -std::numeric_limits<double>::max();
278 Point maxLoc(-1, -1);
279
280 for (int y = 0; y < src.rows; ++y)
281 {
282 const schar* src_row = src.ptr<schar>(y);
283 const uchar* mask_row = mask.empty() ? 0 : mask.ptr<uchar>(y);
284
285 for (int x = 0; x < src.cols; ++x)
286 {
287 if (!mask_row || mask_row[x])
288 {
289 schar val = src_row[x];
290
291 if (val < minVal)
292 {
293 minVal = val;
294 minLoc = cv::Point(x, y);
295 }
296
297 if (val > maxVal)
298 {
299 maxVal = val;
300 maxLoc = cv::Point(x, y);
301 }
302 }
303 }
304 }
305
306 if (minVal_) *minVal_ = minVal;
307 if (maxVal_) *maxVal_ = maxVal;
308
309 if (minLoc_) *minLoc_ = minLoc;
310 if (maxLoc_) *maxLoc_ = maxLoc;
311 }

325 AssertionResult assertMatNear(const char* expr1, const char* expr2, const char* eps_expr, InputArray m1_, InputArray m2_, double eps)
326 {
327 Mat m1 = getMat(m1_);
328 Mat m2 = getMat(m2_);
329
330 if (m1.size() != m2.size())
331 {
332 std::stringstream msg;
333 msg << "Matrices \"" << expr1 << "\" and \"" << expr2 << "\" have different sizes : \""
334 << expr1 << "\" [" << PrintToString(m1.size()) << "] vs \""
335 << expr2 << "\" [" << PrintToString(m2.size()) << "]";
336 return AssertionFailure() << msg.str();
337 }
338
339 if (m1.type() != m2.type())
340 {
341 std::stringstream msg;
342 msg << "Matrices \"" << expr1 << "\" and \"" << expr2 << "\" have different types : \""
343 << expr1 << "\" [" << PrintToString(MatType(m1.type())) << "] vs \""
344 << expr2 << "\" [" << PrintToString(MatType(m2.type())) << "]";
345 return AssertionFailure() << msg.str();
346 }
347
348 Mat diff;
349 absdiff(m1.reshape(1), m2.reshape(1), diff);
350
351 double maxVal = 0.0;
352 Point maxLoc;
353 minMaxLocGold(diff, 0, &maxVal, 0, &maxLoc);
354
355 if (maxVal > eps)
356 {
357 std::stringstream msg;
358 msg << "The max difference between matrices \"" << expr1 << "\" and \"" << expr2
359 << "\" is " << maxVal << " at (" << maxLoc.y << ", " << maxLoc.x / m1.channels() << ")"
360 << ", which exceeds \"" << eps_expr << "\", where \""
361 << expr1 << "\" at (" << maxLoc.y << ", " << maxLoc.x / m1.channels() << ") evaluates to " << printMatVal(m1, maxLoc) << ", \""
362 << expr2 << "\" at (" << maxLoc.y << ", " << maxLoc.x / m1.channels() << ") evaluates to " << printMatVal(m2, maxLoc) << ", \""
363 << eps_expr << "\" evaluates to " << eps;
364 return AssertionFailure() << msg.str();
365 }
366
367 return AssertionSuccess();
368 }

450 bool keyPointsEquals(const cv::KeyPoint& p1, const cv::KeyPoint& p2)
451 {
452 const double maxPtDif = 1.0;
453 const double maxSizeDif = 1.0;
454 const double maxAngleDif = 2.0;
455 const double maxResponseDif = 0.1;
456
457 double dist = cv::norm(p1.pt - p2.pt);
458
459 if (dist < maxPtDif &&
460 fabs(p1.size - p2.size) < maxSizeDif &&
461 abs(p1.angle - p2.angle) < maxAngleDif &&
462 abs(p1.response - p2.response) < maxResponseDif &&
463 p1.octave == p2.octave &&
464 p1.class_id == p2.class_id)
465 {
466 return true;
467 }
468
469 return false;
470 }

472 struct KeyPointLess
473 {
474 bool operator()(const cv::KeyPoint& kp1, const cv::KeyPoint& kp2) const
475 {
476 return kp1.pt.y < kp2.pt.y || (kp1.pt.y == kp2.pt.y && kp1.pt.x < kp2.pt.x);
477 }
478 };

481 testing::AssertionResult assertKeyPointsEquals(const char* gold_expr, const char* actual_expr, std::vector<cv::KeyPoint>& gold, std::vector<cv::KeyPoint>& actual)
482 {
483 if (gold.size() != actual.size())
484 {
485 std::stringstream msg;
486 msg << "KeyPoints size mistmach\n"
487 << "\"" << gold_expr << "\" : " << gold.size() << "\n"
488 << "\"" << actual_expr << "\" : " << actual.size();
489 return AssertionFailure() << msg.str();
490 }
491
492 std::sort(actual.begin(), actual.end(), KeyPointLess());
493 std::sort(gold.begin(), gold.end(), KeyPointLess());
494
495 for (size_t i = 0; i < gold.size(); ++i)
496 {
497 const cv::KeyPoint& p1 = gold[i];
498 const cv::KeyPoint& p2 = actual[i];
499
500 if (!keyPointsEquals(p1, p2))
501 {
502 std::stringstream msg;
503 msg << "KeyPoints differ at " << i << "\n"
504 << "\"" << gold_expr << "\" vs \"" << actual_expr << "\" : \n"
505 << "pt : " << testing::PrintToString(p1.pt) << " vs " << testing::PrintToString(p2.pt) << "\n"
506 << "size : " << p1.size << " vs " << p2.size << "\n"
507 << "angle : " << p1.angle << " vs " << p2.angle << "\n"
508 << "response : " << p1.response << " vs " << p2.response << "\n"
509 << "octave : " << p1.octave << " vs " << p2.octave << "\n"
510 << "class_id : " << p1.class_id << " vs " << p2.class_id;
511 return AssertionFailure() << msg.str();
512 }
513 }
514
515 return ::testing::AssertionSuccess();
516 }

518 int getMatchedPointsCount(std::vector<cv::KeyPoint>& gold, std::vector<cv::KeyPoint>& actual)
519 {
520 std::sort(actual.begin(), actual.end(), KeyPointLess());
521 std::sort(gold.begin(), gold.end(), KeyPointLess());
522
523 int validCount = 0;
524
525 if (actual.size() == gold.size())
526 {
527 for (size_t i = 0; i < gold.size(); ++i)
528 {
529 const cv::KeyPoint& p1 = gold[i];
530 const cv::KeyPoint& p2 = actual[i];
531
532 if (keyPointsEquals(p1, p2))
533 ++validCount;
534 }
535 }
536 else
537 {
538 std::vector<cv::KeyPoint>& shorter = gold;
539 std::vector<cv::KeyPoint>& longer = actual;
540 if (actual.size() < gold.size())
541 {
542 shorter = actual;
543 longer = gold;
544 }
545 for (size_t i = 0; i < shorter.size(); ++i)
546 {
547 const cv::KeyPoint& p1 = shorter[i];
548 const cv::KeyPoint& p2 = longer[i];
549 const cv::KeyPoint& p3 = longer[i+1];
550
551 if (keyPointsEquals(p1, p2) || keyPointsEquals(p1, p3))
552 ++validCount;
553 }
554 }
555
556 return validCount;
557 }
