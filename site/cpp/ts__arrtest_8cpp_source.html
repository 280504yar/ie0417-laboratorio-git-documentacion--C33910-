75int ArrayTest::read_params( const cv::FileStorage& fs )
76{
77 int code = BaseTest::read_params( fs );
78 if( code < 0 )
79 return code;
80
81 read( find_param( fs, "min_log_array_size" ), min_log_array_size, min_log_array_size );
82 read( find_param( fs, "max_log_array_size" ), max_log_array_size, max_log_array_size );
83 read( find_param( fs, "test_case_count" ), test_case_count, test_case_count );
84 test_case_count = cvRound( test_case_count*ts->get_test_case_count_scale() );
85
86 min_log_array_size = clipInt( min_log_array_size, 0, 20 );
87 max_log_array_size = clipInt( max_log_array_size, min_log_array_size, 20 );
88 test_case_count = clipInt( test_case_count, 0, 100000 );
89
90 return code;
91}

117int ArrayTest::prepare_test_case( int test_case_idx )
118{
119 int code = 1;
120 size_t max_arr = test_array.size();
121 vector<vector<Size> > sizes(max_arr);
122 vector<vector<Size> > whole_sizes(max_arr);
123 vector<vector<int> > types(max_arr);
124 size_t i, j;
125 RNG& rng = ts->get_rng();
126
127 for( i = 0; i < max_arr; i++ )
128 {
129 size_t sizei = std::max(test_array[i].size(), (size_t)1);
130 sizes[i].resize(sizei);
131 types[i].resize(sizei);
132 whole_sizes[i].resize(sizei);
133 }
134
135 get_test_array_types_and_sizes( test_case_idx, sizes, types );
136
137 for( i = 0; i < max_arr; i++ )
138 {
139 size_t sizei = test_array[i].size();
140 for( j = 0; j < sizei; j++ )
141 {
142 unsigned t = randInt(rng);
143 bool create_mask = true, use_roi = false;
144 Size size(sizes[i][j]), whole_size = size;
145 Rect roi(0, 0, 0, 0);
146
147 create_mask = (t & 6) == 0; // ~ each of 3 tests will use mask
148 use_roi = (t & 8) != 0;
149 if( use_roi )
150 {
151 whole_size.width += randInt(rng) % 10;
152 whole_size.height += randInt(rng) % 10;
153 }
154
155 if( size.width > 0 && size.height > 0 &&
156 types[i][j] >= 0 && (i != MASK || create_mask) )
157 {
158 if( use_roi )
159 {
160 roi.width = size.width;
161 roi.height = size.height;
162 if( whole_size.width > size.width )
163 roi.x = randInt(rng) % (whole_size.width - size.width);
164
165 if( whole_size.height > size.height )
166 roi.y = randInt(rng) % (whole_size.height - size.height);
167 }
168 if( use_roi )
169 {
170 // TODO: re-enable actual ROI selection
171 // test_mat[i][j] = test_mat[i][j](roi);
172 // Size sz;
173 // Point ofs;
174 // test_mat[i][j].locateROI(sz, ofs);
175 test_mat[i][j].create(roi.size(), types[i][j]);
176 }
177 else
178 {
179 test_mat[i][j].create(whole_size, types[i][j]);
180 }
181 }
182 else
183 {
184 test_mat[i][j].create(Size(), CV_8U);
185 }
186 if( !test_mat[i][j].empty() )
187 fill_array( test_case_idx, (int)i, (int)j, test_mat[i][j] );
188 }
189 }
190 return code;
191}

194void ArrayTest::get_minmax_bounds( int i, int /*j*/, int type, Scalar& low, Scalar& high )
195{
196 double l, u;
197 int depth = CV_MAT_DEPTH(type);
198
199 if( i == MASK )
200 {
201 l = -2;
202 u = 2;
203 }
204 else if( depth < CV_32S )
205 {
206 l = getMinVal(type);
207 u = getMaxVal(type);
208 }
209 else
210 {
211 u = depth == CV_32S ? 1000000 : 1000.;
212 l = -u;
213 }
214
215 low = Scalar::all(l);
216 high = Scalar::all(u);
217}

248int ArrayTest::validate_test_results( int test_case_idx )
249{
250 static const char* arr_names[] = { "input", "input/output", "output",
251 "ref input/output", "ref output",
252 "temporary", "mask" };
253 size_t i, j;
254 prepare_to_validation( test_case_idx );
255
256 for( i = 0; i < 2; i++ )
257 {
258 int i0 = i == 0 ? OUTPUT : INPUT_OUTPUT;
259 int i1 = i == 0 ? REF_OUTPUT : REF_INPUT_OUTPUT;
260 size_t sizei = test_mat[i0].size();
261
262 CV_Assert( sizei == test_mat[i1].size() );
263 for( j = 0; j < sizei; j++ )
264 {
265 double err_level;
266 int code;
267
268 if( test_mat[i1][j].empty() )
269 continue;
270
271 err_level = get_success_error_level( test_case_idx, i0, (int)j );
272 code = cmpEps2(ts, test_mat[i0][j], test_mat[i1][j], err_level, element_wise_relative_error, arr_names[i0]);
273
274 if (code == 0) continue;
275
276
277 for( i0 = 0; i0 < (int)test_mat.size(); i0++ )
278 {
279 size_t sizei0 = test_mat[i0].size();
280 if( i0 == REF_INPUT_OUTPUT || i0 == OUTPUT || i0 == TEMP )
281 continue;
282 for( i1 = 0; i1 < (int)sizei0; i1++ )
283 {
284 const Mat& arr = test_mat[i0][i1];
285 if( !arr.empty() )
286 {
287 string sizestr = vec2str(", ", &arr.size[0], arr.dims);
288 ts->printf( TS::LOG, "%s array %d type=%sC%d, size=(%s)\n",
289 arr_names[i0], i1, getTypeName(arr.depth()),
290 arr.channels(), sizestr.c_str() );
291 }
292 }
293 }
294 ts->set_failed_test_info( code );
295 return code;
296 }
297 }
298
299 return 0;
300}
