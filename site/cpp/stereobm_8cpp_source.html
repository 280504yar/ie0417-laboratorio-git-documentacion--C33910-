127static void prefilterNorm( const Mat& src, Mat& dst, int winsize, int ftzero, int *buf )
128{
129 int x, y, wsz2 = winsize/2;
130 int* vsum = buf + (wsz2 + 1);
131 int scale_g = winsize*winsize/8, scale_s = (1024 + scale_g)/(scale_g*2);
132 const int OFS = 256*5, TABSZ = OFS*2 + 256;
133 uchar tab[TABSZ];
134 const uchar* sptr = src.ptr();
135 int srcstep = (int)src.step;
136 Size size = src.size();
137
138 scale_g *= scale_s;
139
140 for( x = 0; x < TABSZ; x++ )
141 tab[x] = (uchar)(x - OFS < -ftzero ? 0 : x - OFS > ftzero ? ftzero*2 : x - OFS + ftzero);
142
143 for( x = 0; x < size.width; x++ )
144 vsum[x] = (ushort)(sptr[x]*(wsz2 + 2));
145
146 for( y = 1; y < wsz2; y++ )
147 {
148 for( x = 0; x < size.width; x++ )
149 vsum[x] = (ushort)(vsum[x] + sptr[srcstep*y + x]);
150 }
151
152 for( y = 0; y < size.height; y++ )
153 {
154 const uchar* top = sptr + srcstep*MAX(y-wsz2-1,0);
155 const uchar* bottom = sptr + srcstep*MIN(y+wsz2,size.height-1);
156 const uchar* prev = sptr + srcstep*MAX(y-1,0);
157 const uchar* curr = sptr + srcstep*y;
158 const uchar* next = sptr + srcstep*MIN(y+1,size.height-1);
159 uchar* dptr = dst.ptr<uchar>(y);
160
161 for( x = 0; x < size.width; x++ )
162 vsum[x] = (ushort)(vsum[x] + bottom[x] - top[x]);
163
164 for( x = 0; x <= wsz2; x++ )
165 {
166 vsum[-x-1] = vsum[0];
167 vsum[size.width+x] = vsum[size.width-1];
168 }
169
170 int sum = vsum[0]*(wsz2 + 1);
171 for( x = 1; x <= wsz2; x++ )
172 sum += vsum[x];
173
174 int val = ((curr[0]*5 + curr[1] + prev[0] + next[0])*scale_g - sum*scale_s) >> 10;
175 dptr[0] = tab[val + OFS];
176
177 for( x = 1; x < size.width-1; x++ )
178 {
179 sum += vsum[x+wsz2] - vsum[x-wsz2-1];
180 val = ((curr[x]*4 + curr[x-1] + curr[x+1] + prev[x] + next[x])*scale_g - sum*scale_s) >> 10;
181 dptr[x] = tab[val + OFS];
182 }
183
184 sum += vsum[x+wsz2] - vsum[x-wsz2-1];
185 val = ((curr[x]*5 + curr[x-1] + prev[x] + next[x])*scale_g - sum*scale_s) >> 10;
186 dptr[x] = tab[val + OFS];
187 }
188}

209static void
210prefilterXSobel( const Mat& src, Mat& dst, int ftzero )
211{
212 int x, y;
213 const int OFS = 256*4, TABSZ = OFS*2 + 256;
214 uchar tab[TABSZ] = { 0 };
215 Size size = src.size();
216
217 for( x = 0; x < TABSZ; x++ )
218 tab[x] = (uchar)(x - OFS < -ftzero ? 0 : x - OFS > ftzero ? ftzero*2 : x - OFS + ftzero);
219 uchar val0 = tab[0 + OFS];
220
221 for( y = 0; y < size.height-1; y += 2 )
222 {
223 const uchar* srow1 = src.ptr<uchar>(y);
224 const uchar* srow0 = y > 0 ? srow1 - src.step : size.height > 1 ? srow1 + src.step : srow1;
225 const uchar* srow2 = y < size.height-1 ? srow1 + src.step : size.height > 1 ? srow1 - src.step : srow1;
226 const uchar* srow3 = y < size.height-2 ? srow1 + src.step*2 : srow1;
227 uchar* dptr0 = dst.ptr<uchar>(y);
228 uchar* dptr1 = dptr0 + dst.step;
229
230 dptr0[0] = dptr0[size.width-1] = dptr1[0] = dptr1[size.width-1] = val0;
231 x = 1;
232
233#if (CV_SIMD || CV_SIMD_SCALABLE)
234 {
235 v_int16 ftz = vx_setall_s16((short) ftzero);
236 v_int16 ftz2 = vx_setall_s16((short)(ftzero*2));
237 v_int16 z = vx_setzero_s16();
238
239 for(; x <= (size.width - 1) - VTraits<v_int16>::vlanes(); x += VTraits<v_int16>::vlanes())
240 {
241 v_int16 s00 = v_reinterpret_as_s16(vx_load_expand(srow0 + x + 1));
242 v_int16 s01 = v_reinterpret_as_s16(vx_load_expand(srow0 + x - 1));
243 v_int16 s10 = v_reinterpret_as_s16(vx_load_expand(srow1 + x + 1));
244 v_int16 s11 = v_reinterpret_as_s16(vx_load_expand(srow1 + x - 1));
245 v_int16 s20 = v_reinterpret_as_s16(vx_load_expand(srow2 + x + 1));
246 v_int16 s21 = v_reinterpret_as_s16(vx_load_expand(srow2 + x - 1));
247 v_int16 s30 = v_reinterpret_as_s16(vx_load_expand(srow3 + x + 1));
248 v_int16 s31 = v_reinterpret_as_s16(vx_load_expand(srow3 + x - 1));
249
250 v_int16 d0 = v_sub(s00, s01);
251 v_int16 d1 = v_sub(s10, s11);
252 v_int16 d2 = v_sub(s20, s21);
253 v_int16 d3 = v_sub(s30, s31);
254
255 v_uint16 v0 = v_reinterpret_as_u16(v_max(v_min(v_add(v_add(v_add(v_add(d0, d1), d1), d2), ftz), ftz2), z));
256 v_uint16 v1 = v_reinterpret_as_u16(v_max(v_min(v_add(v_add(v_add(v_add(d1, d2), d2), d3), ftz), ftz2), z));
257
258 v_pack_store(dptr0 + x, v0);
259 v_pack_store(dptr1 + x, v1);
260 }
261 }
262#endif
263
264 for( ; x < size.width-1; x++ )
265 {
266 int d0 = srow0[x+1] - srow0[x-1], d1 = srow1[x+1] - srow1[x-1],
267 d2 = srow2[x+1] - srow2[x-1], d3 = srow3[x+1] - srow3[x-1];
268 int v0 = tab[d0 + d1*2 + d2 + OFS];
269 int v1 = tab[d1 + d2*2 + d3 + OFS];
270 dptr0[x] = (uchar)v0;
271 dptr1[x] = (uchar)v1;
272 }
273 }
274
275 for( ; y < size.height; y++ )
276 {
277 uchar* dptr = dst.ptr<uchar>(y);
278 x = 0;
279#if (CV_SIMD || CV_SIMD_SCALABLE)
280 {
281 v_uint8 val0_16 = vx_setall_u8(val0);
282 for(; x <= size.width-VTraits<v_uint8>::vlanes(); x+=VTraits<v_uint8>::vlanes())
283 v_store(dptr + x, val0_16);
284 }
285#endif
286 for(; x < size.width; x++ )
287 dptr[x] = val0;
288 }
289}

295template <typename T>
296struct dispShiftTemplate
297{ };

302 enum { value = DISPARITY_SHIFT_16S };

314template<>
315inline short dispDescale(int v1, int v2, int d)
316{
317 return (short)((v1*256 + (d != 0 ? v2*256/d : 0) + 15) >> 4);
318}

327class BufferBM
328{
329 static const int TABSZ = 256;
330public:
331 std::vector<int*> sad;
332 std::vector<int*> hsad;
333 std::vector<int*> htext;
334 std::vector<uchar*> cbuf0;
335 std::vector<ushort*> sad_short;
336 std::vector<ushort*> hsad_short;
337 int *prefilter[2];
338 uchar tab[TABSZ];
339private:
340 utils::BufferArea area;
341
342public:
343 BufferBM(size_t nstripes, size_t width, size_t height, const StereoBMParams& params)
344 : sad(nstripes, NULL),
345 hsad(nstripes, NULL),
346 htext(nstripes, NULL),
347 cbuf0(nstripes, NULL),
348 sad_short(nstripes, NULL),
349 hsad_short(nstripes, NULL),
350 prefilter()
351 {
352 const int wsz = params.SADWindowSize;
353 const int ndisp = params.numDisparities;
354 const int ftzero = params.preFilterCap;
355 for (size_t i = 0; i < nstripes; ++i)
356 {
357 // 1D: [1][ ndisp ][1]
358#if (CV_SIMD || CV_SIMD_SCALABLE)
359 if (params.useShorts())
360 area.allocate(sad_short[i], ndisp + 2);
361 else
362#endif
363 area.allocate(sad[i], ndisp + 2);
364
365 // 2D: [ wsz/2 + 1 ][ height ][ wsz/2 + 1 ] * [ ndisp ]
366#if (CV_SIMD || CV_SIMD_SCALABLE)
367 if (params.useShorts())
368 area.allocate(hsad_short[i], (height + wsz + 2) * ndisp);
369 else
370#endif
371 area.allocate(hsad[i], (height + wsz + 2) * ndisp);
372
373 // 1D: [ wsz/2 + 1 ][ height ][ wsz/2 + 1 ]
374 area.allocate(htext[i], (height + wsz + 2));
375
376 // 3D: [ wsz/2 + 1 ][ height ][ wsz/2 + 1 ] * [ ndisp ] * [ wsz/2 + 1 ][ wsz/2 + 1 ]
377 area.allocate(cbuf0[i], ((height + wsz + 2) * ndisp * (wsz + 2) + 256));
378 }
379 if (params.useNormPrefilter())
380 {
381 for (size_t i = 0; i < 2; ++i)
382 area.allocate(prefilter[i], width + params.preFilterSize + 2);
383 }
384 area.commit();
385
386 // static table
387 for (int x = 0; x < TABSZ; x++)
388 tab[x] = (uchar)std::abs(x - ftzero);
389 }

392#if (CV_SIMD || CV_SIMD_SCALABLE)
393template <typename dType>
394static void findStereoCorrespondenceBM_SIMD( const Mat& left, const Mat& right,
395 Mat& disp, Mat& cost, const StereoBMParams& state,
396 int _dy0, int _dy1, const BufferBM & bufX, size_t bufNum )
397{
398 int x, y, d;
399 int wsz = state.SADWindowSize, wsz2 = wsz/2;
400 int dy0 = MIN(_dy0, wsz2+1), dy1 = MIN(_dy1, wsz2+1);
401 int ndisp = state.numDisparities;
402 int mindisp = state.minDisparity;
403 int lofs = MAX(ndisp - 1 + mindisp, 0);
404 int rofs = -MIN(ndisp - 1 + mindisp, 0);
405 int width = left.cols, height = left.rows;
406 int width1 = width - rofs - ndisp + 1;
407 int textureThreshold = state.textureThreshold;
408 int uniquenessRatio = state.uniquenessRatio;
409 const int disp_shift = dispShiftTemplate<dType>::value;
410 dType FILTERED = (dType)((mindisp - 1) << disp_shift);
411
412 ushort *hsad, *hsad_sub;
413 uchar *cbuf;
414 const uchar* lptr0 = left.ptr() + lofs;
415 const uchar* rptr0 = right.ptr() + rofs;
416 const uchar *lptr, *lptr_sub, *rptr;
417 dType* dptr = disp.ptr<dType>();
418 int sstep = (int)left.step;
419 int dstep = (int)(disp.step/sizeof(dptr[0]));
420 int cstep = (height + dy0 + dy1)*ndisp;
421 short costbuf = 0;
422 int coststep = cost.data ? (int)(cost.step/sizeof(costbuf)) : 0;
423 const uchar * tab = bufX.tab;
424 short v_seq[VTraits<v_int16>::max_nlanes];
425 for (short i = 0; i < VTraits<v_int16>::vlanes(); ++i)
426 v_seq[i] = i;
427
428 ushort *sad = bufX.sad_short[bufNum] + 1;
429 ushort *hsad0 = bufX.hsad_short[bufNum] + (wsz2 + 1) * ndisp;
430 int *htext = bufX.htext[bufNum] + (wsz2 + 1);
431 uchar *cbuf0 = bufX.cbuf0[bufNum] + (wsz2 + 1) * ndisp;
432
433 // initialize buffers
434 memset(sad - 1, 0, (ndisp + 2) * sizeof(sad[0]));
435 memset(hsad0 - dy0 * ndisp, 0, (height + wsz + 2) * ndisp * sizeof(hsad[0]));
436 memset(htext - dy0, 0, (height + wsz + 2) * sizeof(htext[0]));
437
438 for( x = -wsz2-1; x < wsz2; x++ )
439 {
440 hsad = hsad0 - dy0*ndisp; cbuf = cbuf0 + (x + wsz2 + 1)*cstep - dy0*ndisp;
441 lptr = lptr0 + MIN(MAX(x, -lofs), width-lofs-1) - dy0*sstep;
442 rptr = rptr0 + MIN(MAX(x, -rofs), width-rofs-ndisp) - dy0*sstep;
443
444 for( y = -dy0; y < height + dy1; y++, hsad += ndisp, cbuf += ndisp, lptr += sstep, rptr += sstep )
445 {
446 int lval = lptr[0];
447 v_uint8 lv = vx_setall_u8((uchar)lval);
448 for( d = 0; d <= ndisp - VTraits<v_uint8>::vlanes(); d += VTraits<v_uint8>::vlanes() )
449 {
450 v_uint8 diff = v_absdiff(lv, vx_load(rptr + d));
451 v_store(cbuf + d, diff);
452 v_store(hsad + d, v_add(vx_load(hsad + d), v_expand_low(diff)));
453 v_store(hsad + d + VTraits<v_uint16>::vlanes(), v_add(vx_load(hsad + d + VTraits<v_uint16>::vlanes()), v_expand_high(diff)));
454 }
455 if( d <= ndisp - VTraits<v_uint16>::vlanes() )
456 {
457 v_uint8 diff = v_absdiff(lv, vx_load_low(rptr + d));
458 v_store_low(cbuf + d, diff);
459 v_store(hsad + d, v_add(vx_load(hsad + d), v_expand_low(diff)));
460 d += VTraits<v_uint16>::vlanes();
461 }
462 for( ; d < ndisp; d++ )
463 {
464 int diff = abs(lval - rptr[d]);
465 cbuf[d] = (uchar)diff;
466 hsad[d] += (ushort)diff;
467 }
468 htext[y] += tab[lval];
469 }
470 }
471
472 // initialize the left and right borders of the disparity map
473 for( y = 0; y < height; y++ )
474 {
475 for( x = 0; x < lofs; x++ )
476 dptr[y*dstep + x] = FILTERED;
477 for( x = lofs + width1; x < width; x++ )
478 dptr[y*dstep + x] = FILTERED;
479 }
480 dptr += lofs;
481
482 for( x = 0; x < width1; x++, dptr++ )
483 {
484 short* costptr = cost.data ? cost.ptr<short>() + lofs + x : &costbuf;
485 int x0 = x - wsz2 - 1, x1 = x + wsz2;
486 const uchar* cbuf_sub = cbuf0 + ((x0 + wsz2 + 1) % (wsz + 1))*cstep - dy0*ndisp;
487 cbuf = cbuf0 + ((x1 + wsz2 + 1) % (wsz + 1))*cstep - dy0*ndisp;
488 hsad = hsad0 - dy0*ndisp;
489 lptr_sub = lptr0 + MIN(MAX(x0, -lofs), width-1-lofs) - dy0*sstep;
490 lptr = lptr0 + MIN(MAX(x1, -lofs), width-1-lofs) - dy0*sstep;
491 rptr = rptr0 + MIN(MAX(x1, -rofs), width-ndisp-rofs) - dy0*sstep;
492
493 for( y = -dy0; y < height + dy1; y++, cbuf += ndisp, cbuf_sub += ndisp,
494 hsad += ndisp, lptr += sstep, lptr_sub += sstep, rptr += sstep )
495 {
496 int lval = lptr[0];
497 v_uint8 lv = vx_setall_u8((uchar)lval);
498 for( d = 0; d <= ndisp - VTraits<v_uint8>::vlanes(); d += VTraits<v_uint8>::vlanes() )
499 {
500 v_uint8 diff = v_absdiff(lv, vx_load(rptr + d));
501 v_int8 cbs = v_reinterpret_as_s8(vx_load(cbuf_sub + d));
502 v_store(cbuf + d, diff);
503 v_store(hsad + d, v_reinterpret_as_u16(v_sub(v_reinterpret_as_s16(v_add(vx_load(hsad + d), v_expand_low(diff))), v_expand_low(cbs))));
504 v_store(hsad + d + VTraits<v_uint16>::vlanes(), v_reinterpret_as_u16(v_sub(v_reinterpret_as_s16(v_add(vx_load(hsad + d + VTraits<v_uint16>::vlanes()), v_expand_high(diff))), v_expand_high(cbs))));
505 }
506 if( d <= ndisp - VTraits<v_uint16>::vlanes())
507 {
508 v_uint8 diff = v_absdiff(lv, vx_load_low(rptr + d));
509 v_store_low(cbuf + d, diff);
510 v_store(hsad + d, v_reinterpret_as_u16(v_sub(v_reinterpret_as_s16(v_add(vx_load(hsad + d), v_expand_low(diff))), vx_load_expand((schar *)cbuf_sub + d))));
511 d += VTraits<v_uint16>::vlanes();
512 }
513 for( ; d < ndisp; d++ )
514 {
515 int diff = abs(lval - rptr[d]);
516 cbuf[d] = (uchar)diff;
517 hsad[d] = hsad[d] + (ushort)diff - cbuf_sub[d];
518 }
519 htext[y] += tab[lval] - tab[lptr_sub[0]];
520 }
521
522 // fill borders
523 for( y = dy1; y <= wsz2; y++ )
524 htext[height+y] = htext[height+dy1-1];
525 for( y = -wsz2-1; y < -dy0; y++ )
526 htext[y] = htext[-dy0];
527
528 // initialize sums
529 for( d = 0; d < ndisp; d++ )
530 sad[d] = (ushort)(hsad0[d-ndisp*dy0]*(wsz2 + 2 - dy0));
531
532 hsad = hsad0 + (1 - dy0)*ndisp;
533 for( y = 1 - dy0; y < wsz2; y++, hsad += ndisp )
534 {
535 for( d = 0; d <= ndisp-2*VTraits<v_uint16>::vlanes(); d += 2*VTraits<v_uint16>::vlanes() )
536 {
537 v_store(sad + d, v_add(vx_load(sad + d), vx_load(hsad + d)));
538 v_store(sad + d + VTraits<v_uint16>::vlanes(), v_add(vx_load(sad + d + VTraits<v_uint16>::vlanes()), vx_load(hsad + d + VTraits<v_uint16>::vlanes())));
539 }
540 if( d <= ndisp-VTraits<v_uint16>::vlanes() )
541 {
542 v_store(sad + d, v_add(vx_load(sad + d), vx_load(hsad + d)));
543 d += VTraits<v_uint16>::vlanes();
544 }
545 if( d <= ndisp-VTraits<v_uint16>::vlanes()/2 )
546 {
547 v_store_low(sad + d, v_add(vx_load_low(sad + d), vx_load_low(hsad + d)));
548 d += VTraits<v_uint16>::vlanes()/2;
549 }
550 for( ; d < ndisp; d++ )
551 sad[d] = sad[d] + hsad[d];
552 }
553 int tsum = 0;
554 for( y = -wsz2-1; y < wsz2; y++ )
555 tsum += htext[y];
556
557 // finally, start the real processing
558 for( y = 0; y < height; y++ )
559 {
560 int minsad = INT_MAX, mind = -1;
561 hsad = hsad0 + MIN(y + wsz2, height+dy1-1)*ndisp;
562 hsad_sub = hsad0 + MAX(y - wsz2 - 1, -dy0)*ndisp;
563 v_int16 minsad8 = vx_setall_s16(SHRT_MAX);
564 v_int16 mind8 = vx_setall_s16(0);
565
566 for( d = 0; d <= ndisp - 2*VTraits<v_int16>::vlanes(); d += 2*VTraits<v_int16>::vlanes() )
567 {
568 v_int16 sad8 = v_add(v_sub(v_reinterpret_as_s16(vx_load(hsad + d)), v_reinterpret_as_s16(vx_load(hsad_sub + d))), v_reinterpret_as_s16(vx_load(sad + d)));
569 v_store(sad + d, v_reinterpret_as_u16(sad8));
570 mind8 = v_max(mind8, v_and(v_gt(minsad8, sad8), vx_setall_s16((short)d)));
571 minsad8 = v_min(minsad8, sad8);
572
573 sad8 = v_add(v_sub(v_reinterpret_as_s16(vx_load(hsad + d + VTraits<v_int16>::vlanes())), v_reinterpret_as_s16(vx_load(hsad_sub + d + VTraits<v_int16>::vlanes()))), v_reinterpret_as_s16(vx_load(sad + d + VTraits<v_int16>::vlanes())));
574 v_store(sad + d + VTraits<v_int16>::vlanes(), v_reinterpret_as_u16(sad8));
575 mind8 = v_max(mind8, v_and(v_gt(minsad8, sad8), vx_setall_s16((short)(d + VTraits<v_int16>::vlanes()))));
576 minsad8 = v_min(minsad8, sad8);
577 }
578 if( d <= ndisp - VTraits<v_int16>::vlanes() )
579 {
580 v_int16 sad8 = v_add(v_sub(v_reinterpret_as_s16(vx_load(hsad + d)), v_reinterpret_as_s16(vx_load(hsad_sub + d))), v_reinterpret_as_s16(vx_load(sad + d)));
581 v_store(sad + d, v_reinterpret_as_u16(sad8));
582 mind8 = v_max(mind8, v_and(v_gt(minsad8, sad8), vx_setall_s16((short)d)));
583 minsad8 = v_min(minsad8, sad8);
584 d += VTraits<v_int16>::vlanes();
585 }
586 minsad = v_reduce_min(minsad8);
587 v_int16 v_mask = (v_eq(vx_setall_s16((short)minsad), minsad8));
588 mind = v_reduce_min(v_or(v_and(v_add(mind8, vx_load(v_seq)), v_mask), v_and(vx_setall_s16(32767), v_not(v_mask))));
589 for( ; d < ndisp; d++ )
590 {
591 int sad8 = (int)(hsad[d]) - hsad_sub[d] + sad[d];
592 sad[d] = (ushort)sad8;
593 if(minsad > sad8)
594 {
595 mind = d;
596 minsad = sad8;
597 }
598 }
599
600 tsum += htext[y + wsz2] - htext[y - wsz2 - 1];
601 if( tsum < textureThreshold )
602 {
603 dptr[y*dstep] = FILTERED;
604 continue;
605 }
606
607 if( uniquenessRatio > 0 )
608 {
609 int thresh = minsad + (minsad * uniquenessRatio/100);
610 v_int32 thresh4 = vx_setall_s32(thresh + 1);
611 v_int32 d1 = vx_setall_s32(mind-1), d2 = vx_setall_s32(mind+1);
612 v_int32 dd_4 = vx_setall_s32(VTraits<v_int32>::vlanes());
613 v_int32 d4 = vx_load_expand(v_seq);
614
615 for( d = 0; d <= ndisp - VTraits<v_int16>::vlanes(); d += VTraits<v_int16>::vlanes() )
616 {
617 v_int32 sad4_l, sad4_h;
618 v_expand(v_reinterpret_as_s16(vx_load(sad + d)), sad4_l, sad4_h);
619 if( v_check_any(v_and(v_gt(thresh4, sad4_l), v_or(v_gt(d1, d4), v_gt(d4, d2)))) )
620 break;
621 d4 = v_add(d4, dd_4);
622 if( v_check_any(v_and(v_gt(thresh4, sad4_h), v_or(v_gt(d1, d4), v_gt(d4, d2)))) )
623 break;
624 d4 = v_add(d4, dd_4);
625 }
626 if( d <= ndisp - VTraits<v_int16>::vlanes() )
627 {
628 dptr[y*dstep] = FILTERED;
629 continue;
630 }
631 if( d <= ndisp - VTraits<v_int32>::vlanes() )
632 {
633 v_int32 sad4_l = vx_load_expand((short*)sad + d);
634 if (v_check_any(v_and(v_gt(thresh4, sad4_l), v_or(v_gt(d1, d4), v_gt(d4, d2)))))
635 {
636 dptr[y*dstep] = FILTERED;
637 continue;
638 }
639 d += VTraits<v_int16>::vlanes();
640 }
641 for( ; d < ndisp; d++ )
642 {
643 if( (thresh + 1) > sad[d] && ((mind - 1) > d || d > (mind + 1)) )
644 break;
645 }
646 if( d < ndisp )
647 {
648 dptr[y*dstep] = FILTERED;
649 continue;
650 }
651 }
652
653 if( 0 < mind && mind < ndisp - 1 )
654 {
655 int p = sad[mind+1], n = sad[mind-1];
656 d = p + n - 2*sad[mind] + std::abs(p - n);
657 dptr[y*dstep] = dispDescale<dType>(ndisp - mind - 1 + mindisp, p-n, d);
658 }
659 else
660 dptr[y*dstep] = dispDescale<dType>(ndisp - mind - 1 + mindisp, 0, 0);
661 costptr[y*coststep] = sad[mind];
662 }
663 }
664}

667template <typename mType>
668static void
669findStereoCorrespondenceBM( const Mat& left, const Mat& right,
670 Mat& disp, Mat& cost, const StereoBMParams& state,
671 int _dy0, int _dy1, const BufferBM & bufX, size_t bufNum )
672{
673
674 int x, y, d;
675 int wsz = state.SADWindowSize, wsz2 = wsz/2;
676 int dy0 = MIN(_dy0, wsz2+1), dy1 = MIN(_dy1, wsz2+1);
677 int ndisp = state.numDisparities;
678 int mindisp = state.minDisparity;
679 int lofs = MAX(ndisp - 1 + mindisp, 0);
680 int rofs = -MIN(ndisp - 1 + mindisp, 0);
681 int width = left.cols, height = left.rows;
682 int width1 = width - rofs - ndisp + 1;
683 int textureThreshold = state.textureThreshold;
684 int uniquenessRatio = state.uniquenessRatio;
685 const int disp_shift = dispShiftTemplate<mType>::value;
686 mType FILTERED = (mType)((mindisp - 1) << disp_shift);
687
688 int *hsad, *hsad_sub;
689 uchar *cbuf;
690 const uchar* lptr0 = left.ptr() + lofs;
691 const uchar* rptr0 = right.ptr() + rofs;
692 const uchar *lptr, *lptr_sub, *rptr;
693 mType* dptr = disp.ptr<mType>();
694 int sstep = (int)left.step;
695 int dstep = (int)(disp.step/sizeof(dptr[0]));
696 int cstep = (height+dy0+dy1)*ndisp;
697 int costbuf = 0;
698 int coststep = cost.data ? (int)(cost.step/sizeof(costbuf)) : 0;
699 const uchar * tab = bufX.tab;
700
701#if (CV_SIMD || CV_SIMD_SCALABLE)
702 int v_seq[VTraits<v_int32>::max_nlanes];
703 for (int i = 0; i < VTraits<v_int32>::vlanes(); ++i)
704 v_seq[i] = i;
705 v_int32 d0_4 = vx_load(v_seq), dd_4 = vx_setall_s32(VTraits<v_int32>::vlanes());
706#endif
707
708 int *sad = bufX.sad[bufNum] + 1;
709 int *hsad0 = bufX.hsad[bufNum] + (wsz2 + 1) * ndisp;
710 int *htext = bufX.htext[bufNum] + (wsz2 + 1);
711 uchar *cbuf0 = bufX.cbuf0[bufNum] + (wsz2 + 1) * ndisp;
712
713 // initialize buffers
714 memset(sad - 1, 0, (ndisp + 2) * sizeof(sad[0]));
715 memset(hsad0 - dy0 * ndisp, 0, (height + wsz + 2) * ndisp * sizeof(hsad[0]));
716 memset(htext - dy0, 0, (height + wsz + 2) * sizeof(htext[0]));
717
718 for( x = -wsz2-1; x < wsz2; x++ )
719 {
720 hsad = hsad0 - dy0*ndisp; cbuf = cbuf0 + (x + wsz2 + 1)*cstep - dy0*ndisp;
721 lptr = lptr0 + std::min(std::max(x, -lofs), width-lofs-1) - dy0*sstep;
722 rptr = rptr0 + std::min(std::max(x, -rofs), width-rofs-ndisp) - dy0*sstep;
723 for( y = -dy0; y < height + dy1; y++, hsad += ndisp, cbuf += ndisp, lptr += sstep, rptr += sstep )
724 {
725 int lval = lptr[0];
726 d = 0;
727#if (CV_SIMD || CV_SIMD_SCALABLE)
728 {
729 v_uint8 lv = vx_setall_u8((uchar)lval);
730
731 for( ; d <= ndisp - VTraits<v_uint8>::vlanes(); d += VTraits<v_uint8>::vlanes() )
732 {
733 v_uint8 rv = vx_load(rptr + d);
734 v_int32 hsad_0 = vx_load(hsad + d);
735 v_int32 hsad_1 = vx_load(hsad + d + VTraits<v_int32>::vlanes());
736 v_int32 hsad_2 = vx_load(hsad + d + 2*VTraits<v_int32>::vlanes());
737 v_int32 hsad_3 = vx_load(hsad + d + 3*VTraits<v_int32>::vlanes());
738 v_uint8 diff = v_absdiff(lv, rv);
739 v_store(cbuf + d, diff);
740
741 v_uint16 diff0, diff1;
742 v_uint32 diff00, diff01, diff10, diff11;
743 v_expand(diff, diff0, diff1);
744 v_expand(diff0, diff00, diff01);
745 v_expand(diff1, diff10, diff11);
746
747 hsad_0 = v_add(hsad_0, v_reinterpret_as_s32(diff00));
748 hsad_1 = v_add(hsad_1, v_reinterpret_as_s32(diff01));
749 hsad_2 = v_add(hsad_2, v_reinterpret_as_s32(diff10));
750 hsad_3 = v_add(hsad_3, v_reinterpret_as_s32(diff11));
751
752 v_store(hsad + d, hsad_0);
753 v_store(hsad + d + VTraits<v_int32>::vlanes(), hsad_1);
754 v_store(hsad + d + 2*VTraits<v_int32>::vlanes(), hsad_2);
755 v_store(hsad + d + 3*VTraits<v_int32>::vlanes(), hsad_3);
756 }
757 }
758#endif
759 for( ; d < ndisp; d++ )
760 {
761 int diff = std::abs(lval - rptr[d]);
762 cbuf[d] = (uchar)diff;
763 hsad[d] = (int)(hsad[d] + diff);
764 }
765 htext[y] += tab[lval];
766 }
767 }
768
769 // initialize the left and right borders of the disparity map
770 for( y = 0; y < height; y++ )
771 {
772 for( x = 0; x < lofs; x++ )
773 dptr[y*dstep + x] = FILTERED;
774 for( x = lofs + width1; x < width; x++ )
775 dptr[y*dstep + x] = FILTERED;
776 }
777 dptr += lofs;
778
779 for( x = 0; x < width1; x++, dptr++ )
780 {
781 int* costptr = cost.data ? cost.ptr<int>() + lofs + x : &costbuf;
782 int x0 = x - wsz2 - 1, x1 = x + wsz2;
783 const uchar* cbuf_sub = cbuf0 + ((x0 + wsz2 + 1) % (wsz + 1))*cstep - dy0*ndisp;
784 cbuf = cbuf0 + ((x1 + wsz2 + 1) % (wsz + 1))*cstep - dy0*ndisp;
785 hsad = hsad0 - dy0*ndisp;
786 lptr_sub = lptr0 + MIN(MAX(x0, -lofs), width-1-lofs) - dy0*sstep;
787 lptr = lptr0 + MIN(MAX(x1, -lofs), width-1-lofs) - dy0*sstep;
788 rptr = rptr0 + MIN(MAX(x1, -rofs), width-ndisp-rofs) - dy0*sstep;
789
790 for( y = -dy0; y < height + dy1; y++, cbuf += ndisp, cbuf_sub += ndisp,
791 hsad += ndisp, lptr += sstep, lptr_sub += sstep, rptr += sstep )
792 {
793 int lval = lptr[0];
794 d = 0;
795#if (CV_SIMD || CV_SIMD_SCALABLE)
796 {
797 v_uint8 lv = vx_setall_u8((uchar)lval);
798 for( ; d <= ndisp - VTraits<v_uint8>::vlanes(); d += VTraits<v_uint8>::vlanes() )
799 {
800 v_uint8 rv = vx_load(rptr + d);
801 v_int32 hsad_0 = vx_load(hsad + d);
802 v_int32 hsad_1 = vx_load(hsad + d + VTraits<v_int32>::vlanes());
803 v_int32 hsad_2 = vx_load(hsad + d + 2*VTraits<v_int32>::vlanes());
804 v_int32 hsad_3 = vx_load(hsad + d + 3*VTraits<v_int32>::vlanes());
805 v_uint8 cbs = vx_load(cbuf_sub + d);
806 v_uint8 diff = v_absdiff(lv, rv);
807 v_store(cbuf + d, diff);
808
809 v_uint16 diff0, diff1, cbs0, cbs1;
810 v_int32 diff00, diff01, diff10, diff11, cbs00, cbs01, cbs10, cbs11;
811 v_expand(diff, diff0, diff1);
812 v_expand(cbs, cbs0, cbs1);
813 v_expand(v_reinterpret_as_s16(diff0), diff00, diff01);
814 v_expand(v_reinterpret_as_s16(diff1), diff10, diff11);
815 v_expand(v_reinterpret_as_s16(cbs0), cbs00, cbs01);
816 v_expand(v_reinterpret_as_s16(cbs1), cbs10, cbs11);
817
818 v_int32 diff_0 = v_sub(diff00, cbs00);
819 v_int32 diff_1 = v_sub(diff01, cbs01);
820 v_int32 diff_2 = v_sub(diff10, cbs10);
821 v_int32 diff_3 = v_sub(diff11, cbs11);
822 hsad_0 = v_add(hsad_0, diff_0);
823 hsad_1 = v_add(hsad_1, diff_1);
824 hsad_2 = v_add(hsad_2, diff_2);
825 hsad_3 = v_add(hsad_3, diff_3);
826
827 v_store(hsad + d, hsad_0);
828 v_store(hsad + d + VTraits<v_int32>::vlanes(), hsad_1);
829 v_store(hsad + d + 2*VTraits<v_int32>::vlanes(), hsad_2);
830 v_store(hsad + d + 3*VTraits<v_int32>::vlanes(), hsad_3);
831 }
832 }
833#endif
834 for( ; d < ndisp; d++ )
835 {
836 int diff = std::abs(lval - rptr[d]);
837 cbuf[d] = (uchar)diff;
838 hsad[d] = hsad[d] + diff - cbuf_sub[d];
839 }
840 htext[y] += tab[lval] - tab[lptr_sub[0]];
841 }
842
843 // fill borders
844 for( y = dy1; y <= wsz2; y++ )
845 htext[height+y] = htext[height+dy1-1];
846 for( y = -wsz2-1; y < -dy0; y++ )
847 htext[y] = htext[-dy0];
848
849 // initialize sums
850 for( d = 0; d < ndisp; d++ )
851 sad[d] = (int)(hsad0[d-ndisp*dy0]*(wsz2 + 2 - dy0));
852
853 hsad = hsad0 + (1 - dy0)*ndisp;
854 for( y = 1 - dy0; y < wsz2; y++, hsad += ndisp )
855 {
856 d = 0;
857#if (CV_SIMD || CV_SIMD_SCALABLE)
858 {
859 for( d = 0; d <= ndisp-2*VTraits<v_int32>::vlanes(); d += 2*VTraits<v_int32>::vlanes() )
860 {
861 v_int32 s0 = vx_load(sad + d);
862 v_int32 s1 = vx_load(sad + d + VTraits<v_int32>::vlanes());
863 v_int32 t0 = vx_load(hsad + d);
864 v_int32 t1 = vx_load(hsad + d + VTraits<v_int32>::vlanes());
865 s0 = v_add(s0, t0);
866 s1 = v_add(s1, t1);
867 v_store(sad + d, s0);
868 v_store(sad + d + VTraits<v_int32>::vlanes(), s1);
869 }
870 }
871#endif
872 for( ; d < ndisp; d++ )
873 sad[d] = (int)(sad[d] + hsad[d]);
874 }
875 int tsum = 0;
876 for( y = -wsz2-1; y < wsz2; y++ )
877 tsum += htext[y];
878
879 // finally, start the real processing
880 for( y = 0; y < height; y++ )
881 {
882 int minsad = INT_MAX, mind = -1;
883 hsad = hsad0 + MIN(y + wsz2, height+dy1-1)*ndisp;
884 hsad_sub = hsad0 + MAX(y - wsz2 - 1, -dy0)*ndisp;
885 d = 0;
886#if (CV_SIMD || CV_SIMD_SCALABLE)
887 {
888 v_int32 minsad4 = vx_setall_s32(INT_MAX);
889 v_int32 mind4 = vx_setall_s32(0), d4 = d0_4;
890
891 for( ; d <= ndisp - 2*VTraits<v_int32>::vlanes(); d += 2*VTraits<v_int32>::vlanes() )
892 {
893 v_int32 sad4 = v_sub(v_add(vx_load(sad + d), vx_load(hsad + d)), vx_load(hsad_sub + d));
894 v_store(sad + d, sad4);
895 mind4 = v_select(v_gt(minsad4, sad4), d4, mind4);
896 minsad4 = v_min(minsad4, sad4);
897 d4 = v_add(d4, dd_4);
898
899 sad4 = v_sub(v_add(vx_load(sad + d + VTraits<v_int32>::vlanes()), vx_load(hsad + d + VTraits<v_int32>::vlanes())), vx_load(hsad_sub + d + VTraits<v_int32>::vlanes()));
900 v_store(sad + d + VTraits<v_int32>::vlanes(), sad4);
901 mind4 = v_select(v_gt(minsad4, sad4), d4, mind4);
902 minsad4 = v_min(minsad4, sad4);
903 d4 = v_add(d4, dd_4);
904 }
905
906 int CV_DECL_ALIGNED(CV_SIMD_WIDTH) minsad_buf[VTraits<v_int32>::max_nlanes], mind_buf[VTraits<v_int32>::max_nlanes];
907 v_store(minsad_buf, minsad4);
908 v_store(mind_buf, mind4);
909 for (int i = 0; i < VTraits<v_int32>::vlanes(); ++i)
910 if(minsad_buf[i] < minsad || (minsad == minsad_buf[i] && mind_buf[i] < mind)) { minsad = minsad_buf[i]; mind = mind_buf[i]; }
911 }
912#endif
913 for( ; d < ndisp; d++ )
914 {
915 int currsad = sad[d] + hsad[d] - hsad_sub[d];
916 sad[d] = currsad;
917 if( currsad < minsad )
918 {
919 minsad = currsad;
920 mind = d;
921 }
922 }
923
924 tsum += htext[y + wsz2] - htext[y - wsz2 - 1];
925 if( tsum < textureThreshold )
926 {
927 dptr[y*dstep] = FILTERED;
928 continue;
929 }
930
931 if( uniquenessRatio > 0 )
932 {
933 int thresh = minsad + (minsad * uniquenessRatio/100);
934 for( d = 0; d < ndisp; d++ )
935 {
936 if( (d < mind-1 || d > mind+1) && sad[d] <= thresh)
937 break;
938 }
939 if( d < ndisp )
940 {
941 dptr[y*dstep] = FILTERED;
942 continue;
943 }
944 }
945
946 {
947 sad[-1] = sad[1];
948 sad[ndisp] = sad[ndisp-2];
949 int p = sad[mind+1], n = sad[mind-1];
950 d = p + n - 2*sad[mind] + std::abs(p - n);
951 dptr[y*dstep] = dispDescale<mType>(ndisp - mind - 1 + mindisp, p-n, d);
952
953 costptr[y*coststep] = sad[mind];
954 }
955 }
956 }
