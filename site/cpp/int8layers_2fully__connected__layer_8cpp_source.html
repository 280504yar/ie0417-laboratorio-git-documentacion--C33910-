161 virtual Ptr<BackendNode> initTimVX(void* timVXInfo_,
162 const std::vector<Ptr<BackendWrapper> > &inputsWrapper,
163 const std::vector<Ptr<BackendWrapper> > &outputsWrapper,
164 bool isLast) CV_OVERRIDE
165 {
166#ifdef HAVE_TIMVX
167 // tvGraph Initialization.
168 auto timVxInfo = reinterpret_cast<TimVXInfo *>(timVXInfo_);
169 CV_Assert(timVxInfo);
170 Ptr<TimVXGraph> tvGraph = timVxInfo->getGraph();
171 CV_Assert(tvGraph);
172 Ptr<tim::vx::Graph> graph = tvGraph->graph;
173
174 int numOutput = blobs[0].size[0];
175 Mat weightMat = blobs[0];
176
177 std::vector<int> inputsIndex;
178 std::vector<int> outputsIndex;
179
180 std::vector<float> weight_scs, bias_scs;
181 std::vector<int32_t> weight_zps;
182
183 bias_scs.resize(numOutput);
184 weight_scs.resize(numOutput);
185
186 for (int i = 0; i < numOutput; i++)
187 {
188 bias_scs[i] = outputMultiplier.at<float>(i) * output_sc;
189 weight_scs[i] = bias_scs[i] / input_sc;
190 }
191
192 weight_zps.assign(numOutput, 0);
193
194 // input Tensor
195 auto inputWrapper = inputsWrapper[0].dynamicCast<TimVXBackendWrapper>();
196 int input_index = -1, weight_index = -1, output_index = -1;
197
198 if (inputWrapper->isTensor())
199 {
200 input_index = tvGraph->getTensorIndex(inputWrapper->getTensor());
201 if (input_index == -1)
202 {
203 // Copy To New inputWrapper
204 Mat tmp = inputWrapper->getMat();
205 inputWrapper = Ptr<TimVXBackendWrapper>(new TimVXBackendWrapper(tmp));
206 }
207 }
208
209 if (!inputWrapper->isTensor() || input_index == -1)
210 {
211 Ptr<tim::vx::Quantization> tvInputQuant = Ptr<tim::vx::Quantization>(
212 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, input_sc, input_zp));
213 inputWrapper->createTensor(graph,tim::vx::TensorAttribute::INPUT, tvInputQuant);
214 input_index = tvGraph->addWrapper(inputWrapper);
215 }
216 inputsIndex.push_back(input_index);
217
218 // weight tensor
219 Ptr<TimVXBackendWrapper> weightWrapper = Ptr<TimVXBackendWrapper>(new TimVXBackendWrapper(weightMat));
220 Ptr<tim::vx::Quantization> weightQuant;
221
222 bool tvSymmetric;
223 tvSymmetric = getQuantType(weight_scs, numOutput);
224
225 if (tvSymmetric)
226 {
227 // TODO! fix the following issue.
228 // TimVX does not support the SYMMETRIC PER CHANNEL MatMul.
229 return Ptr<BackendNode>();
230 }
231 else
232 {
233 weightQuant = Ptr<tim::vx::Quantization>(
234 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, weight_scs[0], 0));
235 }
236 weightWrapper->createTensor(graph,tim::vx::TensorAttribute::CONSTANT, weightQuant);
237
238 weight_index = tvGraph->addWrapper(weightWrapper);
239 inputsIndex.push_back(weight_index);
240
241 // Output tensor
242 CV_Assert(outputsWrapper.size() == 1);
243 Ptr<TimVXBackendWrapper> outputWrapper = outputsWrapper[0].dynamicCast<TimVXBackendWrapper>();
244 Ptr<tim::vx::Quantization> outputQuant = Ptr<tim::vx::Quantization>(
245 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, output_sc, output_zp));
246
247 if (isLast)
248 {
249 auto shapeType = getShapeTypeFromMat(outputWrapper->getMat());
250
251 // For Graph Output tensor, we need to set tensor shape before createTensor().
252 outputWrapper->setTensorShape(shapeType);
253 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::OUTPUT, outputQuant);
254 }
255 else
256 {
257 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::TRANSIENT, outputQuant);
258 }
259
260 output_index = tvGraph->addWrapper(outputWrapper);
261 outputsIndex.push_back(output_index);
262
263 std::shared_ptr<tim::vx::Operation> tvMatmul;
264
265 tvMatmul = graph->CreateOperation<tim::vx::ops::Matmul>(false, true);
266
267 Ptr<TimVXBackendNode> tvBackendNode = new TimVXBackendNode(tvGraph, tvMatmul, inputsIndex, outputsIndex);
268
269 return tvBackendNode;
270#endif // HAVE_TIMVX
271 return Ptr<BackendNode>();

280 static void run(const Mat& srcMat, const Mat& weights, const Mat& biasMat, const Mat& outputMultiplier,
281 const Mat& activationLUT, Mat& dstMat, const ActivationLayerInt8* activ, int nstripes, int outZp)
282 {
283 CV_Assert( srcMat.dims == 2 && srcMat.cols == weights.cols &&
284 dstMat.rows == srcMat.rows && dstMat.cols == weights.rows &&
285 srcMat.type() == weights.type() && srcMat.type() == CV_8S &&
286 dstMat.type() == CV_32S && biasMat.type() == CV_32S &&
287 biasMat.isContinuous() && (int)biasMat.total() == dstMat.cols );
288
289 FullyConnected p;
290
291 p.srcMat = &srcMat;
292 p.weights = &weights;
293 p.biasMat = &biasMat;
294 p.outputMultiplier = &outputMultiplier;
295 p.activationLUT = &activationLUT;
296 p.dstMat = &dstMat;
297 p.nstripes = nstripes;
298 p.outZp = outZp;
299 p.activ = !activationLUT.empty() ? activ : 0;
300 p.useAVX2 = checkHardwareSupport(CPU_AVX2);
301 p.useAVX512 = CV_CPU_HAS_SUPPORT_AVX512_SKX;
302 p.useLASX = checkHardwareSupport(CPU_LASX);
303 p.useRVV = checkHardwareSupport(CPU_RVV);
304
305 parallel_for_(Range(0, nstripes), p, nstripes);
306 }

308 void operator()(const Range& r) const CV_OVERRIDE
309 {
310 int valign = FullyConnectedLayerInt8Impl::VEC_ALIGN;
311 int nsamples = srcMat->rows;
312 int nw0 = weights->rows;
313 int k, vecsize = srcMat->cols;
314 int vecsize_aligned = (int)alignSize(vecsize, VEC_ALIGN);
315 size_t total = (size_t)nsamples*nw0;
316 size_t stripeSize = (total + nstripes - 1)/nstripes;
317 size_t stripeStart = r.start*stripeSize;
318 size_t stripeEnd = r.end == nstripes ? total : std::min(r.end*stripeSize, total);
319 size_t wstep = weights->step1();
320 AutoBuffer<int8_t> srcbuf(vecsize_aligned + valign);
321 int8_t* sptr = alignPtr(srcbuf.data(), (int)(valign*sizeof(int8_t)));
322 const int* lutptr = !activationLUT->empty() ? activationLUT->ptr<int>() : 0;
323
324 for( k = vecsize; k < vecsize_aligned; k++ )
325 sptr[k] = 0;
326
327 for( size_t ofs = stripeStart; ofs < stripeEnd; )
328 {
329 int sampleIdx = (int)(ofs / nw0);
330 int delta = (int)(ofs - (size_t)sampleIdx*nw0);
331 const int8_t* sptr_ = srcMat->ptr<int8_t>(sampleIdx);
332 const int8_t* wptr = weights->ptr<int8_t>(delta);
333 int* dptr = dstMat->ptr<int>(sampleIdx) + delta;
334 const int* biasptr = biasMat->ptr<int>() + delta;
335 const float* multptr = outputMultiplier->ptr<float>() + delta;
336 int nw = std::min(nw0 - delta, (int)(stripeEnd - ofs));
337
338 memcpy(sptr, sptr_, vecsize*sizeof(sptr[0]));
339 #if CV_TRY_AVX512_SKX
340 if( useAVX512 )
341 opt_AVX512_SKX::fastGEMM1T( sptr, wptr, wstep, biasptr, multptr, dptr, nw, vecsize, outZp );
342 else
343 #endif
344 #if CV_TRY_AVX2
345 if( useAVX2 )
346 opt_AVX2::fastGEMM1T( sptr, wptr, wstep, biasptr, multptr, dptr, nw, vecsize, outZp );
347 else
348 #endif
349 #if CV_TRY_LASX
350 if( useLASX )
351 opt_LASX::fastGEMM1T( sptr, wptr, wstep, biasptr, multptr, dptr, nw, vecsize, outZp );
352 else
353 #endif
354 #if CV_TRY_RVV && CV_RVV
355 if( useRVV )
356 opt_RVV::fastGEMM1T( sptr, wptr, wstep, biasptr, multptr, dptr, nw, vecsize, outZp );
357 else
358 #endif
359 #if CV_RVP052
360 if( 1 )
361 opt_RVP052::fastGEMM1T( sptr, wptr, wstep, biasptr, multptr, dptr, nw, vecsize, outZp );
362 else
363 #endif
364 {
365 int i = 0;
366 #if CV_SIMD128
367 for( ; i <= nw - 4; i += 4, wptr += 4*wstep )
368 {
369 v_int32x4 vs0 = v_setzero_s32(), vs1 = v_setzero_s32(),
370 vs2 = v_setzero_s32(), vs3 = v_setzero_s32();
371 v_int32x4 outzp = v_setall_s32(outZp), outmin = v_setall_s32(-128), outmax = v_setall_s32(127);
372 v_int32x4 s = v_load(biasptr + i);
373 v_float32x4 mult = v_load(multptr + i);
374
375 for( k = 0; k < vecsize; k += 16 )
376 {
377 v_int8x16 v = v_load_aligned(sptr + k);
378 vs0 = v_dotprod_expand_fast(v, v_load_aligned(wptr + k), vs0);
379 vs1 = v_dotprod_expand_fast(v, v_load_aligned(wptr + wstep + k), vs1);
380 vs2 = v_dotprod_expand_fast(v, v_load_aligned(wptr + wstep*2 + k), vs2);
381 vs3 = v_dotprod_expand_fast(v, v_load_aligned(wptr + wstep*3 + k), vs3);
382 }
383
384 s = v_add(s, v_int32x4(v_reduce_sum(vs0), v_reduce_sum(vs1), v_reduce_sum(vs2), v_reduce_sum(vs3)));
385 v_int32x4 out = v_add(outzp, v_round(v_mul(v_cvt_f32(s), mult)));
386 v_store(dptr + i, v_min(v_max(out, outmin), outmax));
387 }
388 #endif
389
390 for( ; i < nw; i++, wptr += wstep )
391 {
392 int s0 = biasptr[i];
393 float mult0 = multptr[i];
394
395 for( k = 0; k < vecsize; k++ )
396 {
397 int8_t v = sptr[k];
398 s0 += (int)v*wptr[k];
399 }
400 int out0 = outZp + (int)std::round(s0*mult0);
401 dptr[i] = std::min(std::max(out0, -128), 127);
402 }
403 }
404
405 if(activ)
406 activ->forwardSlice(dptr, lutptr, dptr, 1, 1, delta, delta + nw);
407
408 ofs += nw;
409 }
410 }

422 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
423 {
424 CV_TRACE_FUNCTION();
425 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
426
427 std::vector<Mat> input, output;
428 inputs_arr.getMatVector(input);
429 outputs_arr.getMatVector(output);
430
431 int axisCan = normalize_axis(axis, input[0].dims);
432 int outerSize = input[0].total(0, axisCan);
433 Mat srcMat0 = input[0].reshape(1, outerSize);
434 Mat srcMat;
435 if (srcMat0.type() == CV_8U) {
436 // Convert uint8 to int8 by subtracting 128.
437 // Zero-point was adjusted at fusion time to compensate.
438 srcMat0.convertTo(srcMat, CV_8S, 1, -128);
439 } else {
440 srcMat = srcMat0;
441 }
442
443 Mat dstMat = output[0].reshape(1, outerSize);
444 Mat dstMatInt32= Mat(shape(dstMat), CV_32S);
445
446 const int nstripes = outerSize <= 4 ? 1 : getNumThreads();
447 FullyConnected::run(srcMat, weightsMat, biasMat, outputMultiplier, activationLUT, dstMatInt32, activ.get(), nstripes, output_zp);
448 dstMatInt32.convertTo(dstMat, output_type);

468 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> > &inputs,
469 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
470 {
471 CV_CheckTypeEQ(blobs[0].type(), CV_8S, ""); // weights
472 CV_CheckTypeEQ(blobs[1].type(), CV_32S, ""); // bias
473 CV_CheckTypeEQ(outputMultiplier.type(), CV_32F, "");
474
475 ov::Output<ov::Node> input = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
476 ov::Output<ov::Node> ieWeights, ieBias, matmul;
477 bool transA = false, transB = true;
478 size_t numOutput = blobs[0].size[0];
479
480 if (nodes.size() == 2)
481 {
482 CV_Error(Error::StsNotImplemented, "");
483 // auto inp2 = nodes[1].dynamicCast<InfEngineNgraphNode>()->node;
484 // matmul = std::make_shared<ov::op::v0::MatMul>(ieInpNode, inp2, transA, transB);
485 }
486 else
487 {
488 std::vector<int> shape(1 + normalize_axis(axis, input.get_shape().size()), 0);
489 shape[shape.size() - 1] = -1;
490 input = std::make_shared<ov::op::v1::Reshape>(
491 input,
492 std::make_shared<ov::op::v0::Constant>(ov::element::i32, ov::Shape{shape.size()}, shape.data()),
493 true
494 );
495
496 input = ngraphDequantize(input, input_sc, input_zp);
497
498 const float low = -128, high = 127;
499 std::vector<float> inpLows(numOutput, low);
500 std::vector<float> inpHighs(numOutput, high);
501 std::vector<float> outLows(numOutput);
502 std::vector<float> outHighs(numOutput);
503 for (int i = 0; i < numOutput; ++i) {
504 outLows[i] = low * outputMultiplier.ptr<float>()[i] * output_sc / input_sc;
505 outHighs[i] = high * outputMultiplier.ptr<float>()[i] * output_sc / input_sc;
506 }
507
508 std::vector<size_t> weight_shape{(size_t)blobs[0].size[0], (size_t)blobs[0].size[1]};
509 ieWeights = std::make_shared<ov::op::v0::Constant>(ov::element::i8, weight_shape, blobs[0].data);
510 ieWeights = std::make_shared<ov::op::v0::Convert>(ieWeights, ov::element::f32);
511 ieWeights = std::make_shared<ov::op::v0::FakeQuantize>(ieWeights,
512 std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{numOutput, 1}, inpLows.data()),
513 std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{numOutput, 1}, inpHighs.data()),
514 std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{numOutput, 1}, outLows.data()),
515 std::make_shared<ov::op::v0::Constant>(ov::element::f32, ov::Shape{numOutput, 1}, outHighs.data()),
516 256 // levels
517 );
518 matmul = std::make_shared<ov::op::v0::MatMul>(input, ieWeights, transA, transB);
519 }
520
521 if (blobs.size() > 1) {
522 int32_t* bias = blobs[1].ptr<int32_t>();
523 std::vector<float> ovBias(blobs[1].total());
524 for (int i = 0; i < ovBias.size(); ++i) {
525 ovBias[i] = (bias[i] + input_zp * cv::sum(blobs[0].row(i))[0]) * outputMultiplier.ptr<float>()[i] * output_sc;
526 }
527 auto bias_node = std::make_shared<ov::op::v0::Constant>(ov::element::f32,
528 ov::Shape{blobs[1].total()}, ovBias.data());
529 matmul = std::make_shared<ov::op::v1::Add>(matmul, bias_node);
530 }
531
532 matmul = ngraphQuantize(matmul, output_sc, output_zp);
533
534 return new InfEngineNgraphNode(matmul);
535 }
