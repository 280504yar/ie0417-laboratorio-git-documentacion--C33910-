547 break;

603void cv::ogl::Buffer::copyFrom(InputArray arr, Target target, bool autoRelease)
604{
605#ifndef HAVE_OPENGL
606 CV_UNUSED(arr);
607 CV_UNUSED(target);
608 CV_UNUSED(autoRelease);
609 throw_no_ogl();
610#else
611 const int kind = arr.kind();
612
613 const Size asize = arr.size();
614 const int atype = arr.type();
615 create(asize, atype, target, autoRelease);
616
617 switch (kind)
618 {
619 case _InputArray::OPENGL_BUFFER:
620 {
621 ogl::Buffer buf = arr.getOGlBuffer();
622 impl_->copyFrom(buf.bufId(), asize.area() * CV_ELEM_SIZE(atype));
623 break;
624 }
625
626 case _InputArray::CUDA_GPU_MAT:
627 {
628 #ifndef HAVE_CUDA
629 throw_no_cuda();
630 #else
631 GpuMat dmat = arr.getGpuMat();
632 impl_->copyFrom(dmat.data, dmat.step, dmat.cols * dmat.elemSize(), dmat.rows);
633 #endif
634
635 break;
636 }
637
638 default:
639 {
640 Mat mat = arr.getMat();
641 CV_Assert( mat.isContinuous() );
642 impl_->copyFrom(asize.area() * CV_ELEM_SIZE(atype), mat.data);
643 }
644 }
645#endif
