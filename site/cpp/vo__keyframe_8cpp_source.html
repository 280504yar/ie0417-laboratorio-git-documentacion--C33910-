15double rotationAngleDeg(const Matx44d& A, const Matx44d& B)
16{
17 Matx33d Ra(A(0,0),A(0,1),A(0,2), A(1,0),A(1,1),A(1,2), A(2,0),A(2,1),A(2,2));
18 Matx33d Rb(B(0,0),B(0,1),B(0,2), B(1,0),B(1,1),B(1,2), B(2,0),B(2,1),B(2,2));
19 Matx33d D = Rb * Ra.t();
20 double tr = D(0,0) + D(1,1) + D(2,2);
21 double c = std::max(-1.0, std::min(1.0, (tr - 1.0) * 0.5));
22 return std::acos(c) * 180.0 / CV_PI;
23}

25Point3d cameraCenterWorld(const Matx44d& T_cw)
26{
27 Matx33d R(T_cw(0,0),T_cw(0,1),T_cw(0,2),
28 T_cw(1,0),T_cw(1,1),T_cw(1,2),
29 T_cw(2,0),T_cw(2,1),T_cw(2,2));
30 Matx31d t(T_cw(0,3),T_cw(1,3),T_cw(2,3));
31 Matx31d C = -R.t() * t;
32 return Point3d(C(0),C(1),C(2));
33}

35double parallaxDeg(const Point3d& Xw, const Matx44d& A_cw, const Matx44d& B_cw)
36{
37 Point3d CA = cameraCenterWorld(A_cw), CB = cameraCenterWorld(B_cw);
38 Point3d vA = Xw - CA, vB = Xw - CB;
39 double nA = std::sqrt(vA.dot(vA)), nB = std::sqrt(vB.dot(vB));
40 if (nA < 1e-9 || nB < 1e-9) return 0.0;
41 double c = std::max(-1.0, std::min(1.0, vA.dot(vB) / (nA * nB)));
42 return std::acos(c) * 180.0 / CV_PI;
43}

52Matx44d makePose(const Mat& R, const Mat& t)
53{
54 CV_Assert(R.rows == 3 && R.cols == 3 && t.total() == 3);
55 Matx44d T = Matx44d::eye();
56 Mat Rd, td;
57 R.convertTo(Rd, CV_64F);
58 t.reshape(1,3).convertTo(td, CV_64F);
59 for (int i = 0; i < 3; ++i)
60 {
61 for (int j = 0; j < 3; ++j) T(i,j) = Rd.at<double>(i,j);
62 T(i,3) = td.at<double>(i,0);
63 }
64 return T;
65}

77void updateCovisibility(KeyFrame* kf)
78{
79 if (!kf) return;
80
81 kf->covisibility.clear();
82
83 for (MapPoint* mp : kf->mapPoints)
84 {
85 if (!mp || mp->bad) continue;
86 for (auto& [obsKf, kpIdx] : mp->observations)
87 {
88 if (obsKf == kf) continue;
89 kf->covisibility[obsKf]++;
90 obsKf->covisibility[kf]++;
91 }
92 }
93
94 rebuildOrderedCovisibility(kf);
95 for (auto& [nb, cnt] : kf->covisibility)
96 rebuildOrderedCovisibility(nb);
97}

101bool VisualOdometryImpl::shouldPromoteKeyframe(int nInliers, const Matx44d& T_cw,
102 String& reason) const
103{
104 if (framesSinceKf <= params.kfMinFrames) return false;
105
106 if (framesSinceKf > params.kfMaxFrames)
107 {
108 reason = format("timeout(%d)", framesSinceKf);
109 return true;
110 }
111
112 if (!lastKf) return false;
113
114 // Motion-based triggers: rotation then translation (geometric, frame-count independent)
115 double rot = detail::rotationAngleDeg(lastKf->poseCw, T_cw);
116 if (rot > params.kfRotThreshDeg)
117 {
118 reason = format("rot=%.1fdeg", rot);
119 return true;
120 }
121
122 // Tracking quality drop
123 int minInliers = (lastKfInliers > 0)
124 ? std::max(params.kfMinInliers, (int)(params.kfInlierRatio * lastKfInliers))
125 : params.kfMinInliers;
126 if (nInliers < minInliers)
127 {
128 reason = format("inliers=%d<%d", nInliers, minInliers);
129 return true;
130 }
131
132 Point3d cCur = detail::cameraCenterWorld(T_cw);
133 Point3d cLast = detail::cameraCenterWorld(lastKf->poseCw);
134 Point3d d = cCur - cLast;
135 double transDist = std::sqrt(d.dot(d));
136 if (transDist > params.kfTransThresh)
137 {
138 reason = format("trans=%.3f", transDist);
139 return true;
140 }
141
142 return false;
143}
