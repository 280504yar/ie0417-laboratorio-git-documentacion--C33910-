24 SoftMaxLayerInt8Impl(const LayerParams& params)
25 {
26 setParamsFrom(params);
27
28 axis = params.get<int>("axis", 1);
29 logSoftMax = params.get<bool>("log_softmax", false);
30 coerced_2d = params.get<bool>("coerced_2d", false);
31
32 input_sc = params.get<float>("input_scale");
33 input_zp = params.get<int>("input_zeropoint");
34
35 output_sc = params.get<float>("scales");
36 output_zp = params.get<int>("zeropoints");
37
38 if (blobs.empty()) // if no lookUpTable is found
39 {
40 Mat lookUpTable(1, 256, CV_32F);
41 float* table = lookUpTable.ptr<float>();
42 for (int i = -128; i < 128; i++)
43 {
44 float x = input_sc * (i - 127); // ensures exp(x) is always between (0, 1)
45 table[i + 128] = std::exp(x);
46 }
47 blobs.push_back(lookUpTable);
48 }

62 virtual void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr) CV_OVERRIDE {
63 std::vector<Mat> inputs;
64 inputs_arr.getMatVector(inputs);
65 auto src = inputs[0];
66 auto dims_src = src.dims;
67 auto shape_src = shape(src);
68 axis = normalize_axis(axis, dims_src);
69
70 if (!coerced_2d) {
71 is_transpose_needed = (axis == dims_src - 1) ? false : true;
72 if (is_transpose_needed) {
73 permutation.resize(dims_src);
74 std::iota(permutation.begin(), permutation.end(), 0);
75 permutation[axis] = dims_src - 1;
76 permutation[dims_src - 1] = axis;
77
78 transposed_shape.resize(dims_src);
79 std::transform(permutation.begin(), permutation.end(), transposed_shape.begin(), [&shape_src](int axis) { return shape_src[axis]; });
80 N = std::accumulate(transposed_shape.begin(), transposed_shape.end() - 1, 1, std::multiplies<int>());
81 D = transposed_shape.back();
82
83 return;
84 }
85 }
86
87 N = src.total(0, axis);
88 D = src.total(axis);
89 }

104 virtual Ptr<BackendNode> initTimVX(void* timVXInfo_,
105 const std::vector<Ptr<BackendWrapper> > &inputsWrapper,
106 const std::vector<Ptr<BackendWrapper> > &outputsWrapper,
107 bool isLast) CV_OVERRIDE
108 {
109#ifdef HAVE_TIMVX
110 // tvGraph Initialization.
111 auto timVxInfo = reinterpret_cast<TimVXInfo *>(timVXInfo_);
112 CV_Assert(timVxInfo);
113 Ptr<TimVXGraph> tvGraph = timVxInfo->getGraph();
114 CV_Assert(tvGraph);
115 Ptr<tim::vx::Graph> graph = tvGraph->graph;
116
117 std::vector<int> inputsIndex, outputsIndex;
118 int input_index, output_index;
119
120 // input Tensor
121 CV_Assert(inputsWrapper.size() == 1);
122 Ptr<TimVXBackendWrapper> inputWrapper = inputsWrapper[0].dynamicCast<TimVXBackendWrapper>();
123 const Mat &src = inputWrapper->getMat();
124
125 // convert axis from OpenCV NCHW toTimVX WHCN.
126 int tvAxis = src.dims - 1 - normalize_axis(axis, src.dims);
127 if(tvAxis < 0)
128 tvAxis = 0; // default value is 0.
129
130 if (inputWrapper->isTensor())
131 {
132 input_index = tvGraph->getTensorIndex(inputWrapper->getTensor());
133 if (input_index == -1)
134 {
135 // Copy To New inputWrapper
136 Mat tmp = inputWrapper->getMat();
137 inputWrapper = Ptr<TimVXBackendWrapper>(new TimVXBackendWrapper(tmp));
138 }
139 }
140
141 if (!inputWrapper->isTensor())
142 {
143 Ptr<tim::vx::Quantization> tvInputQuant = Ptr<tim::vx::Quantization>(
144 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, input_sc, input_zp));
145 inputWrapper->createTensor(graph,tim::vx::TensorAttribute::INPUT, tvInputQuant);
146 input_index = tvGraph->addWrapper(inputWrapper);
147 }
148 inputsIndex.push_back(input_index);
149
150 // output tensor
151 CV_Assert(outputsWrapper.size() == 1);
152 Ptr<TimVXBackendWrapper> outputWrapper = outputsWrapper[0].dynamicCast<TimVXBackendWrapper>();
153 Mat dstMat = outputWrapper->getMat();
154 Ptr<tim::vx::Quantization> outputQuant = Ptr<tim::vx::Quantization>(
155 new tim::vx::Quantization(tim::vx::QuantType::ASYMMETRIC, output_sc, output_zp));
156
157 Ptr<tim::vx::Tensor> outputTensor;
158
159 if (isLast)
160 {
161 auto shapeType = getShapeTypeFromMat(outputWrapper->getMat());
162
163 // For Graph Output tensor, we need to set tensor shape before createTensor().
164 outputWrapper->setTensorShape(shapeType);
165 if (dstMat.type() == CV_32F)
166 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::OUTPUT);
167 else
168 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::OUTPUT, outputQuant);
169 }
170 else
171 {
172 if (dstMat.type() == CV_32F)
173 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::TRANSIENT);
174 else
175 outputWrapper->createTensor(graph, tim::vx::TensorAttribute::TRANSIENT, outputQuant);
176 }
177 output_index = tvGraph->addWrapper(outputWrapper);
178 outputsIndex.push_back(output_index);
179
180 std::shared_ptr<tim::vx::Operation> tvSoftmax;
181
182 if (logSoftMax)
183 {
184 tvSoftmax = graph->CreateOperation<tim::vx::ops::LogSoftmax>(tvAxis);
185
186 }
187 else
188 {
189 tvSoftmax = graph->CreateOperation<tim::vx::ops::Softmax>(1.0f, tvAxis);
190 }
191
192 Ptr<TimVXBackendNode> tvBackendNode = new TimVXBackendNode(tvGraph, tvSoftmax, inputsIndex, outputsIndex);
193
194 return tvBackendNode;
195#endif // HAVE_TIMVX
196 return Ptr<BackendNode>();
197 }

220 class SoftmaxInt8Invoker : public ParallelLoopBody {
221 public:
222 const Mat& src_;
223 Mat& dst_;

236 SoftmaxInt8Invoker(const Mat& src, Mat& dst, const Mat& lookup_table, int N, int D, float y_scale, int y_zero_point)
237 : src_(src), dst_(dst), lookup_table_(lookup_table), N_(N), D_(D), y_scale_(1.f / y_scale), y_zero_point_(y_zero_point) {
238 threads = N_;
239 cost_per_thread = D_;

242 static void run(const Mat& src, Mat& dst, const Mat& lookup_table, int N, int D, float y_scale, int y_zero_point) {
243 CV_Assert(src.isContinuous());
244 CV_Assert(dst.isContinuous());
245 CV_CheckTypeEQ(src.type(), CV_8S, "DNN/SoftmaxInt8: type of input must be int8");
246 CV_CheckTypeEQ(dst.type(), CV_8S, "DNN/SoftmaxInt8: type of output must be int8");
247
248 SoftmaxInt8Invoker p(src, dst, lookup_table, N, D, y_scale, y_zero_point);
249
250 double nstripes = ((size_t)p.threads * p.cost_per_thread) * (1 / 1024.0);
251 parallel_for_(Range(0, p.threads), p, nstripes);
252 }
253
254 void operator()(const Range& r) const CV_OVERRIDE {
255 int start = r.start;
256 int end = r.end;
257
258 const int8_t* p_src = src_.ptr<int8_t>();
259 int8_t* p_dst = dst_.ptr<int8_t>();
260 const float* table = lookup_table_.ptr<float>();
261
262 for (int i = start; i < end; ++i) {
263 const int8_t* x = p_src + i * D_;
264 int8_t* y = p_dst + i * D_;
265
266 float vsum = 0;
267 for (int j = 0; j < D_; ++j) {
268 const uint8_t idx = uint8_t((*x++) + 128);
269 vsum += table[idx];
270 }
271
272 // FIXME: avoid divide by vsum==0
273
274 x = p_src + i * D_;
275 if (with_log) {
276 for (int j = 0; j < D_; ++j) {
277 const uint8_t idx = uint8_t((*x++) + 128);
278 const float v = table[idx];
279 *y++ = saturate_cast<int8_t>(std::nearbyintf(y_scale_ * std::log(v / vsum)) + y_zero_point_);
280 }
281 } else {
282 for (int j = 0; j < D_; ++j) {
283 const uint8_t idx = uint8_t((*x++) + 128);
284 const float v = table[idx];
285 *y++ = saturate_cast<int8_t>(std::nearbyintf(y_scale_ * v / vsum) + y_zero_point_);
286 }
287 }
288 }
289 }
290 };

362 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
363 {
364 CV_TRACE_FUNCTION();
365 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
366
367 std::vector<Mat> inputs, outputs;
368 inputs_arr.getMatVector(inputs);
369 outputs_arr.getMatVector(outputs);
370
371 Mat src, dst;
372
373 if (!coerced_2d && is_transpose_needed) {
374 transposeND(inputs[0], permutation, src);
375 dst = Mat::zeros(transposed_shape.size(), transposed_shape.data(), outputs[0].type());
376 } else {
377 src = inputs[0];
378 dst = outputs[0];
379 }
380
381 switch (dst.type()) {
382 case CV_8S: {
383 if (logSoftMax) {
384 SoftmaxInt8Invoker<true>::run(src, dst, blobs[0], N, D, output_sc, output_zp);
385 } else {
386 SoftmaxInt8Invoker<false>::run(src, dst, blobs[0], N, D, output_sc, output_zp);
387 }
388 } break;
389 case CV_32F: {
390 if (logSoftMax) {
391 SoftmaxInt8OutputFloatInvoker<true>::run(src, dst, blobs[0], N, D);
392 } else {
393 SoftmaxInt8OutputFloatInvoker<false>::run(src, dst, blobs[0], N, D);
394 }
395 } break;
396 default: CV_Error(cv::Error::BadDepth, "DNN/SoftmaxInt8: Unsupported output type");
397 }
398
399 if (!coerced_2d && is_transpose_needed) {
400 transposeND(dst, permutation, outputs[0]);
401 }
402 }

404 int64 getFLOPS(const std::vector<MatShape> &inputs,
405 const std::vector<MatShape> &outputs) const CV_OVERRIDE
406 {
407 CV_UNUSED(outputs); // suppress unused variable warning
408 int64 flops = 0;
409
410 for (int i = 0; i < inputs.size(); i++)
411 {
412 flops += 4*total(inputs[i]);
413 }
414
415 return flops;
416 }
