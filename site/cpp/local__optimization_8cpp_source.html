31 // In lo_sampler_ the sample size should be set and be equal gc_sample_size_
32 GraphCutImpl (const Ptr<Estimator> &estimator_, const Ptr<Quality> &quality_,
33 const Ptr<NeighborhoodGraph> &neighborhood_graph_, const Ptr<RandomGenerator> &lo_sampler_,
34 double threshold_, double spatial_coherence_term, int gc_inner_iteration_number_, Ptr<Termination> termination_) :
35 neighborhood_graph (neighborhood_graph_), estimator (estimator_), quality (quality_),
36 lo_sampler (lo_sampler_), error (quality_->getErrorFnc()), termination(termination_) {
37
38 points_size = quality_->getPointsSize();
39 spatial_coherence = spatial_coherence_term;
40 sqr_trunc_thr = threshold_ * 2.25; // threshold is already squared
41 gc_sample_size = lo_sampler_->getSubsetSize();
42 lo_inner_iterations = gc_inner_iteration_number_;
43 one_minus_lambda = 1.0 - spatial_coherence;
44
45 energies = std::vector<double>(points_size);
46 labeling_inliers = std::vector<int>(points_size);
47 used_edges = std::vector<bool>(points_size*points_size);
48 gc_models = std::vector<Mat> (estimator->getMaxNumSolutionsNonMinimal());
49 }

51 bool refineModel (const Mat &best_model, const Score &best_model_score,
52 Mat &new_model, Score &new_model_score) override {
53 if (best_model_score.inlier_number < estimator->getNonMinimalSampleSize())
54 return false;
55
56 // improve best model by non minimal estimation
57 new_model_score = Score(); // set score to inf (worst case)
58 best_model.copyTo(new_model);
59
60 bool is_best_model_updated = true;
61 while (is_best_model_updated) {
62 is_best_model_updated = false;
63
64 // Build graph problem. Apply graph cut to G
65 int labeling_inliers_size = labeling(new_model);
66 for (int iter = 0; iter < lo_inner_iterations; iter++) {
67 // sample to generate min (|I_7m|, |I|)
68 int num_of_estimated_models;
69 if (labeling_inliers_size > gc_sample_size) {
70 // generate random subset in range <0; |I|>
71 num_of_estimated_models = estimator->estimateModelNonMinimalSample
72 (lo_sampler->generateUniqueRandomSubset(labeling_inliers,
73 labeling_inliers_size), gc_sample_size, gc_models, weights);
74 } else {
75 if (iter > 0) break; // break inliers are not updated
76 num_of_estimated_models = estimator->estimateModelNonMinimalSample
77 (labeling_inliers, labeling_inliers_size, gc_models, weights);
78 }
79 for (int model_idx = 0; model_idx < num_of_estimated_models; model_idx++) {
80 const Score gc_temp_score = quality->getScore(gc_models[model_idx]);
81 // store the best model from estimated models
82 if (gc_temp_score.isBetter(new_model_score)) {
83 is_best_model_updated = true;
84 new_model_score = gc_temp_score;
85 gc_models[model_idx].copyTo(new_model);
86 }
87 }
88
89 if (termination != nullptr && is_best_model_updated && current_ransac_iter > termination->update(best_model, best_model_score.inlier_number)) {
90 is_best_model_updated = false; // to break outer loop
91 }
92
93 } // end of inner GC local optimization
94 } // end of while loop
95 return true;
96 }
97
98private:
99 // find inliers using graph cut algorithm.
100 int labeling (const Mat& model) {
101 const auto &errors = error->getErrors(model);
102 detail::GCGraph<double> graph;
103
104 for (int pt = 0; pt < points_size; pt++)
105 graph.addVtx();
106
107 // The distance and energy for each point
108 double tmp_squared_distance, energy;
109
110 // Estimate the vertex capacities
111 for (int pt = 0; pt < points_size; pt++) {
112 tmp_squared_distance = errors[pt];
113 if (std::isnan(tmp_squared_distance))
114 tmp_squared_distance = std::numeric_limits<float>::max();
115 energy = tmp_squared_distance / sqr_trunc_thr; // Truncated quadratic cost
116
117 if (tmp_squared_distance <= sqr_trunc_thr)
118 graph.addTermWeights(pt, 0, one_minus_lambda * (1 - energy));
119 else
120 graph.addTermWeights(pt, one_minus_lambda * energy, 0);
121
122 energies[pt] = energy > 1 ? 1 : energy;
123 }
124
125 std::fill(used_edges.begin(), used_edges.end(), false);
126
127 bool has_edges = false;
128 // Iterate through all points and set their edges
129 for (int point_idx = 0; point_idx < points_size; ++point_idx) {
130 energy = energies[point_idx];
131
132 // Iterate through all neighbors
133 for (int actual_neighbor_idx : neighborhood_graph->getNeighbors(point_idx)) {
134 if (actual_neighbor_idx == point_idx ||
135 used_edges[actual_neighbor_idx*points_size + point_idx] ||
136 used_edges[point_idx*points_size + actual_neighbor_idx])
137 continue;
138
139 used_edges[actual_neighbor_idx*points_size + point_idx] = true;
140 used_edges[point_idx*points_size + actual_neighbor_idx] = true;
141
142 double a = (0.5 * (energy + energies[actual_neighbor_idx])) * spatial_coherence,
143 b = spatial_coherence, c = spatial_coherence, d = 0;
144 graph.addTermWeights(point_idx, d, a);
145 b -= a;
146 if (b + c < 0)
147 continue; // invalid regularity
148 if (b < 0) {
149 graph.addTermWeights(point_idx, 0, b);
150 graph.addTermWeights(actual_neighbor_idx, 0, -b);
151 graph.addEdges(point_idx, actual_neighbor_idx, 0, b + c);
152 } else if (c < 0) {
153 graph.addTermWeights(point_idx, 0, -c);
154 graph.addTermWeights(actual_neighbor_idx, 0, c);
155 graph.addEdges(point_idx, actual_neighbor_idx, b + c, 0);
156 } else
157 graph.addEdges(point_idx, actual_neighbor_idx, b, c);
158 has_edges = true;
159 }
160 }
161 if (! has_edges)
162 return quality->getInliers(model, labeling_inliers);
163 graph.maxFlow();
164
165 int inlier_number = 0;
166 for (int pt = 0; pt < points_size; pt++)
167 if (! graph.inSourceSegment(pt)) // check for sink
168 labeling_inliers[inlier_number++] = pt;
169 return inlier_number;
170 }
171 int getNumLOoptimizations () const override { return num_lo_optimizations; }

190

193

195
196 double threshold, new_threshold, threshold_step;
197 std::vector<double> weights;
198public:
199 InnerIterativeLocalOptimizationImpl (const Ptr<Estimator> &estimator_, const Ptr<Quality> &quality_,
200 const Ptr<RandomGenerator> &lo_sampler_, int pts_size,
201 double threshold_, bool is_iterative_, int lo_iter_sample_size_,
202 int lo_inner_iterations_=10, int lo_iter_max_iterations_=5,
203 double threshold_multiplier_=4)
204 : estimator (estimator_), quality (quality_), lo_sampler (lo_sampler_)
205 , lo_iter_sample_size(0), new_threshold(0), threshold_step(0) {
206 lo_inner_max_iterations = lo_inner_iterations_;
207 lo_iter_max_iterations = lo_iter_max_iterations_;
208
209 threshold = threshold_;
210 lo_sample_size = lo_sampler->getSubsetSize();
211 is_iterative = is_iterative_;
212 if (is_iterative) {
213 lo_iter_sample_size = lo_iter_sample_size_;
214 lo_iter_sampler = UniformRandomGenerator::create(0/*state*/, pts_size, lo_iter_sample_size_);
215 lo_iter_models = std::vector<Mat>(estimator->getMaxNumSolutionsNonMinimal());
216 virtual_inliers = std::vector<int>(pts_size);
217 new_threshold = threshold_multiplier_ * threshold;
218 // reduce multiplier threshold K·θ by this number in each iteration.
219 // In the last iteration there be original threshold θ.
220 threshold_step = (new_threshold - threshold) / lo_iter_max_iterations_;
221 }
222 lo_models = std::vector<Mat>(estimator->getMaxNumSolutionsNonMinimal());
223
224 // Allocate max memory to avoid reallocation
225 inliers_of_best_model = std::vector<int>(pts_size);

228 /*
229 * Implementation of Locally Optimized Ransac
230 * Inner + Iterative
231 */
232 bool refineModel (const Mat &so_far_the_best_model, const Score &best_model_score,
233 Mat &new_model, Score &new_model_score) override {
234 if (best_model_score.inlier_number < estimator->getNonMinimalSampleSize())
235 return false;
236
237 so_far_the_best_model.copyTo(new_model);
238 new_model_score = best_model_score;
239 // get inliers from so far the best model.
240 int num_inliers_of_best_model = quality->getInliers(so_far_the_best_model,
241 inliers_of_best_model);
242
243 // Inner Local Optimization Ransac.
244 for (int iters = 0; iters < lo_inner_max_iterations; iters++) {
245 int num_estimated_models;
246 // Generate sample of lo_sample_size from inliers from the best model.
247 if (num_inliers_of_best_model > lo_sample_size) {
248 // if there are many inliers take limited number at random.
249 num_estimated_models = estimator->estimateModelNonMinimalSample
250 (lo_sampler->generateUniqueRandomSubset(inliers_of_best_model,
251 num_inliers_of_best_model), lo_sample_size, lo_models, weights);
252 } else {
253 // if model was not updated in first iteration, so break.
254 if (iters > 0) break;
255 // if inliers are less than limited number of sample then take all for estimation
256 // if it fails -> end Lo.
257 num_estimated_models = estimator->estimateModelNonMinimalSample
258 (inliers_of_best_model, num_inliers_of_best_model, lo_models, weights);
259 }
260
262 for (int model_idx = 0; model_idx < num_estimated_models; model_idx++) {
263 const Score temp_score = quality->getScore(lo_models[model_idx]);
264 if (temp_score.isBetter(new_model_score)) {
265 new_model_score = temp_score;
266 lo_models[model_idx].copyTo(new_model);
267 }
268 }
269
270 if (is_iterative) {
271 double lo_threshold = new_threshold;
272 // get max virtual inliers. Note that they are nor real inliers,
273 // because we got them with bigger threshold.
274 int virtual_inliers_size = quality->getInliers
275 (new_model, virtual_inliers, lo_threshold);
276
277 Mat lo_iter_model;
278 Score lo_iter_score = Score(); // set worst case
279 for (int iterations = 0; iterations < lo_iter_max_iterations; iterations++) {
280 lo_threshold -= threshold_step;
281
282 if (virtual_inliers_size > lo_iter_sample_size) {
283 // if there are more inliers than limit for sample size then generate at random
284 // sample from LO model.
285 num_estimated_models = estimator->estimateModelNonMinimalSample
286 (lo_iter_sampler->generateUniqueRandomSubset (virtual_inliers,
287 virtual_inliers_size), lo_iter_sample_size, lo_iter_models, weights);
288 } else {
289 // break if failed, very low probability that it will not fail in next iterations
290 // estimate model with all virtual inliers
291 num_estimated_models = estimator->estimateModelNonMinimalSample
292 (virtual_inliers, virtual_inliers_size, lo_iter_models, weights);
293 }
294 if (num_estimated_models == 0) break;
295
296 // Get score and update virtual inliers with current threshold
298 lo_iter_models[0].copyTo(lo_iter_model);
299 lo_iter_score = quality->getScore(lo_iter_model);
300 for (int model_idx = 1; model_idx < num_estimated_models; model_idx++) {
301 const Score temp_score = quality->getScore(lo_iter_models[model_idx]);
302 if (temp_score.isBetter(lo_iter_score)) {
303 lo_iter_score = temp_score;
304 lo_iter_models[model_idx].copyTo(lo_iter_model);
305 }
306 }
307
308 if (iterations != lo_iter_max_iterations-1)
309 virtual_inliers_size = quality->getInliers(lo_iter_model, virtual_inliers, lo_threshold);
310 }
311
312 if (lo_iter_score.isBetter(new_model_score)) {
313 new_model_score = lo_iter_score;
314 lo_iter_model.copyTo(new_model);
315 }
316 }
317
318 if (num_inliers_of_best_model < new_model_score.inlier_number && iters != lo_inner_max_iterations-1)
319 num_inliers_of_best_model = quality->getInliers (new_model, inliers_of_best_model);
320 }
321 return true;
322 }
323};

335class SimpleLocalOptimizationImpl : public SimpleLocalOptimization {
336private:
337 const Ptr<Quality> quality;
338 const Ptr<Error> error;
339 const Ptr<NonMinimalSolver> estimator;
340 const Ptr<Termination> termination;
341 const Ptr<RandomGenerator> random_generator;
342 const Ptr<WeightFunction> weight_fnc;
343 // unlike to @random_generator which has fixed subset size
344 // @random_generator_smaller_subset is used to draw smaller
345 // amount of points which depends on current number of inliers
346 Ptr<RandomGenerator> random_generator_smaller_subset;
347 int points_size, max_lo_iters, non_min_sample_size, current_ransac_iter;
348 std::vector<double> weights;
349 std::vector<int> inliers;
350 std::vector<cv::Mat> models;
351 double inlier_threshold_sqr;
352 int num_lo_optimizations = 0;
353 bool updated_lo = false;
354public:
355 SimpleLocalOptimizationImpl (const Ptr<Quality> &quality_, const Ptr<NonMinimalSolver> &estimator_,
356 const Ptr<Termination> termination_, const Ptr<RandomGenerator> &random_gen, Ptr<WeightFunction> weight_fnc_,
357 int max_lo_iters_, double inlier_threshold_sqr_, bool update_lo_) :
358 quality(quality_), error(quality_->getErrorFnc()), estimator(estimator_), termination(termination_),
359 random_generator(random_gen), weight_fnc(weight_fnc_) {
360 max_lo_iters = max_lo_iters_;
361 non_min_sample_size = random_generator->getSubsetSize();
362 current_ransac_iter = 0;
363 inliers = std::vector<int>(quality_->getPointsSize());
364 models = std::vector<cv::Mat>(estimator_->getMaxNumberOfSolutions());
365 points_size = quality_->getPointsSize();
366 inlier_threshold_sqr = inlier_threshold_sqr_;
367 if (weight_fnc != nullptr) weights = std::vector<double>(points_size);
368 random_generator_smaller_subset = nullptr;
369 updated_lo = update_lo_;
370 }
371 void setCurrentRANSACiter (int ransac_iter) override { current_ransac_iter = ransac_iter; }
372 int getNumLOoptimizations () const override { return num_lo_optimizations; }
373 bool refineModel (const Mat &best_model, const Score &best_model_score, Mat &new_model, Score &new_model_score) override {
374 new_model_score = best_model_score;
375 best_model.copyTo(new_model);
376
377 int num_inliers;
378 if (weights.empty())
379 num_inliers = Quality::getInliers(error, best_model, inliers, inlier_threshold_sqr);
380 else num_inliers = weight_fnc->getInliersWeights(error->getErrors(best_model), inliers, weights);
381 auto update_generator = [&] (int num_inls) {
382 if (num_inls <= non_min_sample_size) {
383 // add new random generator if number of inliers is fewer than non-minimal sample size
384 const int new_sample_size = (int)(0.6*num_inls);
385 if (new_sample_size <= estimator->getMinimumRequiredSampleSize())
386 return false;
387 if (random_generator_smaller_subset == nullptr)
388 random_generator_smaller_subset = UniformRandomGenerator::create(num_inls/*state*/, quality->getPointsSize(), new_sample_size);
389 else random_generator_smaller_subset->setSubsetSize(new_sample_size);
390 }
391 return true;
392 };
393 if (!update_generator(num_inliers))
394 return false;
395
396 int max_lo_iters_ = max_lo_iters, last_update = 0, last_inliers_update = 0;
397 for (int iter = 0; iter < max_lo_iters_; iter++) {
398 int num_models;
399 if (num_inliers <= non_min_sample_size)
400 num_models = estimator->estimate(new_model, random_generator_smaller_subset->generateUniqueRandomSubset(inliers, num_inliers),
401 random_generator_smaller_subset->getSubsetSize(), models, weights);
402 else num_models = estimator->estimate(new_model, random_generator->generateUniqueRandomSubset(inliers, num_inliers), non_min_sample_size, models, weights);
403 for (int m = 0; m < num_models; m++) {
404 const auto score = quality->getScore(models[m]);
405 if (score.isBetter(new_model_score)) {
406 last_update = iter; last_inliers_update = new_model_score.inlier_number - score.inlier_number;
407 if (updated_lo) {
408 if (max_lo_iters_ < iter + 5 && last_inliers_update >= 1) {
409 max_lo_iters_ = iter + 5;
410 }
411 }
412 models[m].copyTo(new_model);
413 new_model_score = score;
414 if (termination != nullptr && current_ransac_iter > termination->update(new_model, new_model_score.inlier_number))
415 return true; // terminate LO if max number of iterations reached
416 if (score.inlier_number >= best_model_score.inlier_number ||
417 score.inlier_number > non_min_sample_size) {
418 // update inliers if new model has more than previous best model
419 if (weights.empty())
420 num_inliers = Quality::getInliers(error, best_model, inliers, inlier_threshold_sqr);
421 else num_inliers = weight_fnc->getInliersWeights(error->getErrors(best_model), inliers, weights);
422 if (!update_generator(num_inliers))
423 return true;
424 }
425
426 }
427 }
428 if (updated_lo && iter - last_update >= 10) {
429 break;
430 }
431 }
432 return true;
433 }
434};

441class MagsacWeightFunctionImpl : public MagsacWeightFunction {
442private:
443 const std::vector<double> &stored_gamma_values;
444 double C, max_sigma, max_sigma_sqr, scale_of_stored_gammas, one_over_sigma, gamma_k, squared_sigma_max_2, rescale_err;
445 int DoF;
446 unsigned int stored_gamma_number_min1;
447public:
448 MagsacWeightFunctionImpl (const Ptr<GammaValues> &gamma_generator,
449 int DoF_, double upper_incomplete_of_sigma_quantile, double C_, double max_sigma_) :
450 stored_gamma_values (gamma_generator->getGammaValues()) {
451 gamma_k = upper_incomplete_of_sigma_quantile;
452 stored_gamma_number_min1 = static_cast<unsigned int>(gamma_generator->getTableSize()-1);
453 scale_of_stored_gammas = gamma_generator->getScaleOfGammaValues();
454 DoF = DoF_; C = C_;
455 max_sigma = max_sigma_;
456 squared_sigma_max_2 = max_sigma * max_sigma * 2.0;
457 one_over_sigma = C * std::pow(2, (DoF - 1.0) * 0.5) / max_sigma;
458 max_sigma_sqr = squared_sigma_max_2 * 0.5;
459 rescale_err = scale_of_stored_gammas / squared_sigma_max_2;
460 }
461 int getInliersWeights (const std::vector<float> &errors, std::vector<int> &inliers, std::vector<double> &weights) const override {
462 return getInliersWeights(errors, inliers, weights, one_over_sigma, rescale_err, max_sigma_sqr);
463 }
464 int getInliersWeights (const std::vector<float> &errors, std::vector<int> &inliers, std::vector<double> &weights, double thr_sqr) const override {
465 const auto _max_sigma = thr_sqr;
466 const auto _squared_sigma_max_2 = _max_sigma * _max_sigma * 2.0;
467 const auto _one_over_sigma = C * std::pow(2, (DoF - 1.0) * 0.5) / _max_sigma;
468 const auto _max_sigma_sqr = _squared_sigma_max_2 * 0.5;
469 const auto _rescale_err = scale_of_stored_gammas / _squared_sigma_max_2;
470 return getInliersWeights(errors, inliers, weights, _one_over_sigma, _rescale_err, _max_sigma_sqr);
471 }
472 double getThreshold () const override {
473 return max_sigma_sqr;
474 }
475private:
476 int getInliersWeights (const std::vector<float> &errors, std::vector<int> &inliers, std::vector<double> &weights,
477 double _one_over_sigma, double _rescale_err, double _max_sigma_sqr) const {
478 int num_inliers = 0, p = 0;
479 for (const auto &e : errors) {
480 // Calculate the residual of the current point
481 if (e < _max_sigma_sqr) {
482 // Get the position of the gamma value in the lookup table
483 auto x = static_cast<unsigned int>(_rescale_err * e);
484 if (x > stored_gamma_number_min1)
485 x = stored_gamma_number_min1;
486 inliers[num_inliers] = p; // store index of point for LSQ
487 weights[num_inliers++] = _one_over_sigma * (stored_gamma_values[x] - gamma_k);
488 }
489 p++;
490 }
491 return num_inliers;
492 }
493};

496 return makePtr<MagsacWeightFunctionImpl>(gamma_generator_, DoF_, upper_incomplete_of_sigma_quantile, C_, max_sigma_);
497}

518 max_iters = max_iters_;
519 points_size = quality_->getPointsSize();
520 threshold = quality_->getThreshold();
521 iou_thr = iou_thr_;
522 is_covariance = dynamic_cast<const cv::usac::CovarianceSolver*>(solver_.get()) != nullptr;
523 mask = std::vector<bool>(points_size);
524 mask_best = std::vector<bool>(points_size);
525 inliers = std::vector<int>(points_size);
526 if (weight_fnc != nullptr) {
527 weights = std::vector<double>(points_size);
528 max_thr = weight_fnc->getThreshold();

531 }

533
534 bool polishSoFarTheBestModel (const Mat &model, const Score &best_model_score,
535 Mat &new_model, Score &new_model_score) override {
536 int num_inliers = 0;
537 if (weights.empty()) {
538 quality->getInliers(model, mask_best);
539 if (!is_covariance)
540 for (int p = 0; p < points_size; p++)
541 if (mask_best[p]) inliers[num_inliers++] = p;
542 } else {
543 errors_best = error_fnc->getErrors(model);
544 num_inliers = weight_fnc->getInliersWeights(errors_best, inliers, weights, max_thr);
545 }
546 new_model_score = best_model_score;
547 model.copyTo(new_model);
548 int last_update = -1;
549 for (int iter = 0; iter < max_iters; iter++) {
550 int num_sols;
551 if (is_covariance) num_sols = solver->estimate(mask_best, models, weights);
552 else num_sols = solver->estimate(new_model, inliers, num_inliers, models, weights);
553 Score prev_score;
554 for (int i = 0; i < num_sols; i++) {
555 const auto &errors = error_fnc->getErrors(models[i]);
556 const auto score = quality->getScore(errors);
557 if (score.isBetter(new_model_score)) {
558 last_update = iter;
559 models[i].copyTo(new_model);
560 errors_best = errors;
561 prev_score = new_model_score;
562 new_model_score = score;
563 }
564 }
565 if (weights.empty()) {
566 if (iter > last_update)
567 break;
568 else {
569 Quality::getInliers(errors_best, mask, threshold);
570 if (Utils::intersectionOverUnion(mask, mask_best) >= iou_thr)
571 return true;
572 mask_best = mask;
573 num_inliers = 0;
574 if (!is_covariance)
575 for (int p = 0; p < points_size; p++)
576 if (mask_best[p]) inliers[num_inliers++] = p;
577 }
578 } else {
579 if (iter > last_update) {
580 // new model is worse
581 if (CHANGE_WEIGHTS) {
582 // if failed more than 5 times then break
583 if (iter - std::max(0, last_update) >= 5)
584 break;
585 // try to change weights by changing threshold
586 if (fabs(new_model_score.score - prev_score.score) < FLT_EPSILON) {
587 // increase threshold if new model score is the same as the same as previous
588 max_thr *= 1.05; // increase by 5%
589 } else if (iter > last_update) {
590 // decrease max threshold if model is worse
591 max_thr *= 0.9; // decrease by 10%
592 }
593 } else break; // break if not changing weights
594 }
595 // generate new weights and inliers
596 num_inliers = weight_fnc->getInliersWeights(errors_best, inliers, weights, max_thr);
597 }
598 }
599 return last_update >= 0;
600 }
