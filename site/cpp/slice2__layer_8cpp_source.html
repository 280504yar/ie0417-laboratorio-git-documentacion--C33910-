64 MatShape getOutShape(const MatShape& inpShape,
65 const std::vector<int>& starts_,
66 const std::vector<int>& ends_,
67 const std::vector<int>& axes_,
68 const std::vector<int>& steps_,
69 int* allStarts = nullptr,
70 int* allEnds = nullptr,
71 int* allSteps = nullptr) const
72 {
73 bool sliceMask[MatShape::MAX_DIMS];
74
75 int ndims = inpShape.dims;
76 int nstarts = (int)starts_.size(), nends = (int)ends_.size();
77 int naxes = (int)axes_.size(), nsteps = (int)steps_.size();
78
79 CV_Assert_N(nstarts > 0, nstarts <= ndims, nstarts == nends);
80 CV_Assert(naxes == 0 || naxes == nstarts);
81 CV_Assert(nsteps == 0 || nsteps == nstarts);
82
83 MatShape outShape = inpShape;
84
85 for (int i = 0; i < ndims; i++) {
86 sliceMask[i] = false;
87 if (allStarts)
88 allStarts[i] = 0;
89 if (allEnds)
90 allEnds[i] = inpShape[i];
91 if (allSteps)
92 allSteps[i] = 1;
93 }
94
95 for (int i = 0; i < nstarts; i++) {
96 int axis = i;
97 if (!axes_.empty()) {
98 axis = axes_[i];
99 axis = normalize_axis(axis, ndims);
100 if (sliceMask[axis]) {
101 CV_Error(Error::StsBadArg, "duplicate axis occurs in Slice");
102 }
103 }
104 sliceMask[axis] = true;
105 int inpsz = inpShape[axis];
106 int start = starts_[i];
107 int end = ends_[i];
108 int step = 1;
109 if (!steps_.empty())
110 step = steps_[i];
111 CV_Assert(step != 0);
112 start = start < 0 ? std::max(start + inpsz, 0) :
113 std::min(start, inpsz - (step < 0));
114 end = end < 0 ? std::max(end + inpsz, -(step < 0)) :
115 std::min(end, inpsz);
116 if (allStarts)
117 allStarts[axis] = start;
118 if (allSteps)
119 allSteps[axis] = step;
120 int outsz = step > 0 ? (end - start + step-1)/step :
121 (start - end - step-1)/(-step);
122 if (outsz < 0) {
123 outsz = 0;
124 end = start;
125 }
126 if (allEnds)
127 allEnds[axis] = end;
128 outShape[axis] = outsz;
129 }
130
131 return outShape;
132 }

136 int getLayouts(const std::vector<DataLayout>& actualInputs,
137 std::vector<DataLayout>& desiredInputs,
138 const int requiredOutputs,
139 std::vector<DataLayout>& outputs) const CV_OVERRIDE
140 {
141 auto* netimpl_ = getNetImpl(this);
142 DataLayout defaultLayout = netimpl_->originalLayout;
143 const size_t ninputs = actualInputs.size();
144 desiredInputs = actualInputs;
145 outputs.assign(requiredOutputs, DATA_LAYOUT_UNKNOWN);
146
147 const bool inputIsBlock = ninputs >= 1 && actualInputs[0] == DATA_LAYOUT_BLOCK;
148
149 std::vector<int> resolvedAxes = axes;
150 if (resolvedAxes.empty() && this->inputs.size() > 3 &&
151 netimpl_->isConstArg(this->inputs[3])) {
152 Mat axesT = netimpl_->argTensor(this->inputs[3]);
153 tensorToIntVec(axesT, resolvedAxes);
154 }
155 bool axesOK = !resolvedAxes.empty();
156 if (axesOK) {
157 const int channelAxis = (defaultLayout == DATA_LAYOUT_NCHW) ? 1 :
158 (defaultLayout == DATA_LAYOUT_NHWC) ? 3 : -1;
159 for (int a : resolvedAxes) {
160 if (a < 0 || a == channelAxis) { axesOK = false; break; }
161 }
162 }
163
164 if (inputIsBlock && axesOK) {
165 outputs.assign(requiredOutputs, DATA_LAYOUT_BLOCK);
166 } else if (inputIsBlock) {
167 desiredInputs[0] = defaultLayout;
168 }
169 return outputs[0] == DATA_LAYOUT_BLOCK ? netimpl_->defaultC0 : 0;
170 }

225 template <typename T>
226 class ParallelSlice : public cv::ParallelLoopBody
227 {
228 public:
229 // parallel_axis: dim to split across threads. Leading dims (0..parallel_axis-1) are
230 // collapsed into a constant offset since they have size 1 in the output.
231 ParallelSlice(const Mat& inp, Mat& out,
232 const std::vector<Range>& ranges,
233 const std::vector<int>& steps,
234 int parallel_axis)
235 : inp_(inp), out_(out), ranges_(ranges), steps_(steps),
236 parallel_axis_(parallel_axis)
237 {
238 dims_ = inp.dims;
239 es_ = inp.elemSize();
240
241 inp_strides_.resize(dims_);
242 out_strides_.resize(dims_);
243 for(int i=0; i<dims_; ++i) {
244 inp_strides_[i] = inp.step.p[i];
245 out_strides_[i] = out.step.p[i];
246 }
247
248 preamble_src_ = 0;
249 for (int d = 0; d < parallel_axis_; d++) {
250 preamble_src_ += (size_t)ranges_[d].start * inp_strides_[d];
251 }
252 }
253
254 void operator()(const Range& range) const CV_OVERRIDE
255 {
256 const int axis = parallel_axis_;
257 const int b = ranges_[axis].start;
258 const int s = steps_[axis];
259
260 const uchar* src_base = inp_.ptr() + preamble_src_;
261 uchar* dst_base = out_.ptr();
262
263 if (s == 1 && axis < dims_ - 1 && is_fully_contiguous(axis)) {
264 const size_t unit_bytes = inp_strides_[axis];
265 const size_t count = (size_t)(range.end - range.start);
266 const size_t src_off = (size_t)(b + range.start) * unit_bytes;
267 const size_t dst_off = (size_t)range.start * out_strides_[axis];
268 std::memcpy(dst_base + dst_off, src_base + src_off, count * unit_bytes);
269 return;
270 }
271
272 for (int i = range.start; i < range.end; ++i)
273 {
274 int k = b + i * s;
275 size_t src_offset = (size_t)k * inp_strides_[axis];
276 size_t dst_offset = (size_t)i * out_strides_[axis];
277 if (axis == dims_ - 1)
278 std::memcpy(dst_base + dst_offset, src_base + src_offset, es_);
279 else
280 recursive_copy(axis + 1, src_base + src_offset, dst_base + dst_offset);
281 }
282 }
283
284 void recursive_copy(int dim, const uchar* src_ptr, uchar* dst_ptr) const
285 {
286 if (dim >= dims_) return;
287
288 int begin = ranges_[dim].start;
289 int end = ranges_[dim].end;
290 int step = steps_[dim];
291
292 if (dim == dims_ - 1)
293 {
294 if (step == 1)
295 {
296 size_t count = end - begin;
297 std::memcpy(dst_ptr, src_ptr + begin * inp_strides_[dim], count * es_);
298 }
299 else
300 {
301 const uchar* s_ptr = src_ptr + begin * inp_strides_[dim];
302 uchar* d_ptr = dst_ptr;
303 size_t s_stride = step * inp_strides_[dim];
304 size_t d_stride = out_strides_[dim];
305
306 if (step > 0)
307 {
308 for (int k = begin; k < end; k += step)
309 {
310 *(T*)d_ptr = *(const T*)s_ptr;
311 s_ptr += s_stride;
312 d_ptr += d_stride;
313 }
314 }
315 else
316 {
317 for (int k = begin; k > end; k += step)
318 {
319 *(T*)d_ptr = *(const T*)s_ptr;
320 s_ptr += s_stride;
321 d_ptr += d_stride;
322 }
323 }
324 }
325 return;
326 }
327
328 if (step == 1 && is_fully_contiguous(dim))
329 {
330 size_t count = end - begin;
331 size_t bytes = count * inp_strides_[dim];
332 std::memcpy(dst_ptr, src_ptr + begin * inp_strides_[dim], bytes);
333 return;
334 }
335
336 size_t src_stride = step * inp_strides_[dim];
337 size_t dst_stride = out_strides_[dim];
338
339 const uchar* s_ptr = src_ptr + begin * inp_strides_[dim];
340 uchar* d_ptr = dst_ptr;
341
342 if (step > 0)
343 {
344 for (int k = begin; k < end; k += step)
345 {
346 recursive_copy(dim + 1, s_ptr, d_ptr);
347 s_ptr += src_stride;
348 d_ptr += dst_stride;
349 }
350 }
351 else
352 {
353 for (int k = begin; k > end; k += step)
354 {
355 recursive_copy(dim + 1, s_ptr, d_ptr);
356 s_ptr += src_stride;
357 d_ptr += dst_stride;
358 }
359 }
360 }
361
362 bool is_fully_contiguous(int dim) const
363 {
364 size_t expected_step = es_;
365 for (int d = dims_ - 1; d >= dim; --d)
366 {
367 if (inp_.step[d] != expected_step) return false;
368 if (d > dim) {
369 if (steps_[d] != 1) return false;
370 if (ranges_[d].start != 0 || ranges_[d].end != inp_.size[d]) return false;
371 expected_step *= inp_.size[d];
372 }
373 }
374 return true;
375 }

390 template <typename T>
391 void run_parallel(const Mat& inp, Mat& out, const std::vector<Range>& ranges, const std::vector<int>& steps)
392 {
393 int dims = inp.dims;
394 // Skip leading size-1 dims; otherwise the whole slice is one stripe.
395 int parallel_axis = 0;
396 while (parallel_axis < dims - 1 && out.size[parallel_axis] == 1) {
397 parallel_axis++;
398 }
399 int parallel_size = out.size[parallel_axis];
400 ParallelSlice<T> body(inp, out, ranges, steps, parallel_axis);
401 // One stripe per thread; per-element body is tiny so dynamic chunking
402 // would burn cycles on task sync.
403 int nthreads = std::max(1, getNumThreads());
404 double nstripes = std::min((double)parallel_size, (double)nthreads);
405 parallel_for_(Range(0, parallel_size), body, nstripes);
406 }

408 // Slicing copies elements without interpreting them, so only the width matters.
409 void run_parallel_by_width(const Mat& inp, Mat& out,
410 const std::vector<Range>& ranges, const std::vector<int>& steps)
411 {
412 if (inp.elemSize() == 1) {
413 run_parallel<uint8_t>(inp, out, ranges, steps);
414 } else if (inp.elemSize() == 2) {
415 run_parallel<uint16_t>(inp, out, ranges, steps);
416 } else if (inp.elemSize() == 4) {
417 run_parallel<uint32_t>(inp, out, ranges, steps);
418 } else {
419 CV_Assert(inp.elemSize() == 8);
420 run_parallel<uint64_t>(inp, out, ranges, steps);
421 }
422 }

424 void forward(InputArrayOfArrays inputs_arr,
425 OutputArrayOfArrays outputs_arr,
426 OutputArrayOfArrays) CV_OVERRIDE
427 {
428 CV_TRACE_FUNCTION();
429 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
430
431 Size size = inputs_arr.size();
432 int ninputs = size.area();
433 checkNumInputs(ninputs);
434
435 int inpType = inputs_arr.type(0);
436 MatShape inpShape = inputs_arr.shape(0);
437 std::vector<int> tempStarts, tempEnds, tempAxes, steps;
438 const std::vector<int> *starts_ = &starts, *ends_ = &ends, *axes_ = &axes;
439
440 if (ninputs > 1) {
441 Mat startsTensor = inputs_arr.getMat(1);
442 tensorToIntVec(startsTensor, tempStarts);
443 starts_ = &tempStarts;
444 Mat endsTensor = inputs_arr.getMat(2);
445 tensorToIntVec(endsTensor, tempEnds);
446 ends_ = &tempEnds;
447 if (ninputs > 3) {
448 Mat axesTensor = inputs_arr.getMat(3);
449 tensorToIntVec(axesTensor, tempAxes);
450 axes_ = &tempAxes;
451 }
452 if (ninputs > 4) {
453 Mat stepsTensor = inputs_arr.getMat(4);
454 tensorToIntVec(stepsTensor, steps);
455 }
456 }
457 int allStarts[MatShape::MAX_DIMS];
458 int allEnds[MatShape::MAX_DIMS];
459 int allSteps[MatShape::MAX_DIMS];
460 MatShape outShape = getOutShape(inpShape, *starts_, *ends_, *axes_, steps,
461 allStarts, allEnds, allSteps);
462
463 std::vector<Range> ranges;
464 std::vector<int> steps_vec;
465 for (int i = 0; i < inpShape.dims; ++i) {
466 ranges.push_back(Range(allStarts[i], allEnds[i]));
467 steps_vec.push_back(allSteps[i]);
468 }
469
470 int outKind = outputs_arr.kind();
471 CV_Assert(outKind == _InputArray::STD_VECTOR_MAT ||
472 outKind == _InputArray::STD_VECTOR_UMAT);
473
474 if (outKind == _InputArray::STD_VECTOR_MAT) {
475 Mat inp = inputs_arr.getMat(0);
476 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
477 outs.resize(1);
478 outs[0].fit(outShape, inpType);
479
480 run_parallel_by_width(inp, outs[0], ranges, steps_vec);
481 } else {
482 Mat inp = inputs_arr.getMat(0);
483 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
484 outs.resize(1);
485 outs[0].fit(outShape, inpType);
486 Mat temp(outShape, inpType);
487
488 run_parallel_by_width(inp, temp, ranges, steps_vec);
489
490 temp.copyTo(outs[0]);
491 }
492 }
