14template<typename T>
15T inline norm_vec(const Vec<T, 3>& vec)
16{
17 return std::sqrt(vec[0] * vec[0] + vec[1] * vec[1] + vec[2] * vec[2]);
18}

29template<typename T>
30Mat_<T> computeRadius(const Mat& points)
31{
32 typedef Vec<T, 4> PointT;
33
34 // Compute the
35 Size size(points.cols, points.rows);
36 Mat_<T> r(size);
37 if (points.isContinuous())
38 size = Size(points.cols * points.rows, 1);
39 for (int y = 0; y < size.height; ++y)
40 {
41 const PointT* point = points.ptr < PointT >(y), * point_end = points.ptr < PointT >(y) + size.width;
42 T* row = r[y];
43 for (; point != point_end; ++point, ++row)
44 *row = norm_vec(*point);
45 }
46
47 return r;
48}

50// Compute theta and phi according to equation 3 of
51// ``Fast and Accurate Computation of Surface Normals from Range Images``
52// by H. Badino, D. Huber, Y. Park and T. Kanade
53template<typename T>
54void computeThetaPhi(int rows, int cols, const Matx<T, 3, 3>& K, Mat& cos_theta, Mat& sin_theta,
55 Mat& cos_phi, Mat& sin_phi)
56{
57 // Create some bogus coordinates
58 Mat depth_image = K(0, 0) * Mat_<T> ::ones(rows, cols);
59 Mat points3d;
60 depthTo3d(depth_image, Mat(K), points3d);
61
62 //typedef Vec<T, 3> Vec3T;
63 typedef Vec<T, 4> Vec4T;
64
65 cos_theta = Mat_<T>(rows, cols);
66 sin_theta = Mat_<T>(rows, cols);
67 cos_phi = Mat_<T>(rows, cols);
68 sin_phi = Mat_<T>(rows, cols);
69 Mat r = computeRadius<T>(points3d);
70 for (int y = 0; y < rows; ++y)
71 {
72 T* row_cos_theta = cos_theta.ptr <T>(y), * row_sin_theta = sin_theta.ptr <T>(y);
73 T* row_cos_phi = cos_phi.ptr <T>(y), * row_sin_phi = sin_phi.ptr <T>(y);
74 const Vec4T* row_points = points3d.ptr <Vec4T>(y),
75 * row_points_end = points3d.ptr <Vec4T>(y) + points3d.cols;
76 const T* row_r = r.ptr < T >(y);
77 for (; row_points < row_points_end;
78 ++row_cos_theta, ++row_sin_theta, ++row_cos_phi, ++row_sin_phi, ++row_points, ++row_r)
79 {
80 // In the paper, z goes away from the camera, y goes down, x goes right
81 // OpenCV has the same conventions
82 // Theta goes from z to x (and actually goes from -pi/2 to pi/2, phi goes from z to y
83 float theta = (float)std::atan2(row_points->val[0], row_points->val[2]);
84 *row_cos_theta = std::cos(theta);
85 *row_sin_theta = std::sin(theta);
86 float phi = (float)std::asin(row_points->val[1] / (*row_r));
87 *row_cos_phi = std::cos(phi);
88 *row_sin_phi = std::sin(phi);
89 }
90 }
91}

96template<typename T>
97inline void signNormal(const Vec<T, 3>& normal_in, Vec<T, 3>& normal_out)
98{
99 Vec<T, 3> res;
100 if (normal_in[2] > 0)
101 res = -normal_in / norm_vec(normal_in);
102 else
103 res = normal_in / norm_vec(normal_in);
104
105 normal_out[0] = res[0];
106 normal_out[1] = res[1];
107 normal_out[2] = res[2];
108}

119 normal_out[1] = res[1];

122}

127template<typename T>
128inline void signNormal(T a, T b, T c, Vec<T, 3>& normal)
129{
130 T norm = 1 / std::sqrt(a * a + b * b + c * c);
131 if (c > 0)
132 {
133 normal[0] = -a * norm;
134 normal[1] = -b * norm;
135 normal[2] = -c * norm;
136 }
137 else
138 {
139 normal[0] = a * norm;
140 normal[1] = b * norm;
141 normal[2] = c * norm;
142 }
143}
144template<typename T>
145inline void signNormal(T a, T b, T c, Vec<T, 4>& normal)
146{
147 T norm = 1 / std::sqrt(a * a + b * b + c * c);
148 if (c > 0)
149 {
150 normal[0] = -a * norm;
151 normal[1] = -b * norm;
152 normal[2] = -c * norm;
153 normal[3] = 0;
154 }
155 else
156 {
157 normal[0] = a * norm;
158 normal[1] = b * norm;
159 normal[2] = c * norm;
160 normal[3] = 0;
161 }

235 virtual void apply(InputArray points3d_in, OutputArray normals_out) const CV_OVERRIDE
236 {
237 Mat points3d_ori = points3d_in.getMat();
238
239 CV_Assert(points3d_ori.dims == 2);
240
241 // Either we have 3d points or a depth image
242
243 bool ptsAre4F = (points3d_ori.channels() == 4) && (points3d_ori.depth() == CV_32F || points3d_ori.depth() == CV_64F);
244 bool ptsAreDepth = (points3d_ori.channels() == 1) && (points3d_ori.depth() == CV_16U || points3d_ori.depth() == CV_32F || points3d_ori.depth() == CV_64F);
245 if (method == RGBD_NORMALS_METHOD_FALS || method == RGBD_NORMALS_METHOD_SRI || method == RGBD_NORMALS_METHOD_CROSS_PRODUCT)
246 {
247 if (!ptsAre4F)
248 {
249 CV_Error(Error::StsBadArg, "Input image should have 4 float-point channels");
250 }
251 }
252 else if (method == RGBD_NORMALS_METHOD_LINEMOD)
253 {
254 if (!ptsAre4F && !ptsAreDepth)
255 {
256 CV_Error(Error::StsBadArg, "Input image should have 4 float-point channels or have 1 ushort or float-point channel");
257 }
258 }
259 else
260 {
261 CV_Error(Error::StsInternal, "Unknown normal computer algorithm");
262 }
263
264 // Initialize the pimpl
265 cache();
266
267 // Precompute something for RGBD_NORMALS_METHOD_SRI and RGBD_NORMALS_METHOD_FALS
268 Mat points3d;
269 if (method != RGBD_NORMALS_METHOD_LINEMOD)
270 {
271 // Make the points have the right depth
272 if (points3d_ori.depth() == dtype)
273 points3d = points3d_ori;
274 else
275 points3d_ori.convertTo(points3d, dtype);
276 }
277
278 // Get the normals
279 normals_out.create(points3d_ori.size(), CV_MAKETYPE(dtype, 4));
280 if (points3d_ori.empty())
281 return;
282
283 Mat normals = normals_out.getMat();
284 if ((method == RGBD_NORMALS_METHOD_FALS) || (method == RGBD_NORMALS_METHOD_SRI))
285 {
286 // Compute the distance to the points
287 Mat radius = computeRadius<T>(points3d);
288 compute(radius, normals);
289 }
290 else if (method == RGBD_NORMALS_METHOD_LINEMOD)
291 {
292 compute(points3d_ori, normals);
293 }
294 else if (method == RGBD_NORMALS_METHOD_CROSS_PRODUCT)
295 {
296 compute(points3d, normals);
297 }
298 else
299 {
300 CV_Error(Error::StsInternal, "Unknown normal computer algorithm");
301 }
302 }
303
304 int rows, cols;
305 Mat K, K_ori;
306 int windowSize;
307 RgbdNormalsMethod method;
308 mutable bool cacheIsDirty;
309};

318template<typename T>
319class FALS : public RgbdNormalsImpl<T>
320{
321public:
322 typedef Matx<T, 3, 3> Mat33T;
323 typedef Vec<T, 9> Vec9T;
324 typedef Vec<T, 4> Vec4T;
325 typedef Vec<T, 3> Vec3T;
326
327 FALS(int _rows, int _cols, int _windowSize, const Mat& _K) :
328 RgbdNormalsImpl<T>(_rows, _cols, _windowSize, _K, RgbdNormals::RGBD_NORMALS_METHOD_FALS)
329 { }
330 virtual ~FALS() CV_OVERRIDE
331 { }
332
335 virtual void cache() const CV_OVERRIDE
336 {
337 if (!this->cacheIsDirty)
338 return;
339
340 // Compute theta and phi according to equation 3
341 Mat cos_theta, sin_theta, cos_phi, sin_phi;
342 computeThetaPhi<T>(this->rows, this->cols, this->K, cos_theta, sin_theta, cos_phi, sin_phi);
343
344 // Compute all the v_i for every points
345 std::vector<Mat> channels(3);
346 channels[0] = sin_theta.mul(cos_phi);
347 channels[1] = sin_phi;
348 channels[2] = cos_theta.mul(cos_phi);
349 merge(channels, V_);
350
351 // Compute M
352 Mat_<Vec9T> M(this->rows, this->cols);
353 Mat33T VVt;
354 const Vec3T* vec = V_[0];
355 Vec9T* M_ptr = M[0], * M_ptr_end = M_ptr + this->rows * this->cols;
356 for (; M_ptr != M_ptr_end; ++vec, ++M_ptr)
357 {
358 VVt = (*vec) * vec->t();
359 *M_ptr = Vec9T(VVt.val);
360 }
361
362 boxFilter(M, M, M.depth(), Size(this->windowSize, this->windowSize), Point(-1, -1), false);
363
364 // Compute M's inverse
365 Mat33T M_inv;
366 M_inv_.create(this->rows, this->cols);
367 Vec9T* M_inv_ptr = M_inv_[0];
368 for (M_ptr = &M(0); M_ptr != M_ptr_end; ++M_inv_ptr, ++M_ptr)
369 {
370 // We have a semi-definite matrix
371 invert(Mat33T(M_ptr->val), M_inv, DECOMP_CHOLESKY);
372 *M_inv_ptr = Vec9T(M_inv.val);
373 }
374
375 this->cacheIsDirty = false;
376 }
377
382 virtual void compute(const Mat& r, Mat& normals) const CV_OVERRIDE
383 {
384 // Compute B
385 Mat_<Vec3T> B(this->rows, this->cols);
386
387 const T* row_r = r.ptr < T >(0), * row_r_end = row_r + this->rows * this->cols;
388 const Vec3T* row_V = V_[0];
389 Vec3T* row_B = B[0];
390 for (; row_r != row_r_end; ++row_r, ++row_B, ++row_V)
391 {
392 Vec3T val = (*row_V) / (*row_r);
393 if (cvIsInf(val[0]) || cvIsNaN(val[0]) ||
394 cvIsInf(val[1]) || cvIsNaN(val[1]) ||
395 cvIsInf(val[2]) || cvIsNaN(val[2]))
396 *row_B = Vec3T();
397 else
398 *row_B = val;
399 }
400
401 // Apply a box filter to B
402 boxFilter(B, B, B.depth(), Size(this->windowSize, this->windowSize), Point(-1, -1), false);
403
404 // compute the Minv*B products
405 row_r = r.ptr < T >(0);
406 const Vec3T* B_vec = B[0];
407 const Mat33T* M_inv = reinterpret_cast<const Mat33T*>(M_inv_.ptr(0));
408 //Vec3T* normal = normals.ptr<Vec3T>(0);
409 Vec4T* normal = normals.ptr<Vec4T>(0);
410 for (; row_r != row_r_end; ++row_r, ++B_vec, ++normal, ++M_inv)
411 if (cvIsNaN(*row_r))
412 {
413 (*normal)[0] = *row_r;
414 (*normal)[1] = *row_r;
415 (*normal)[2] = *row_r;
416 (*normal)[3] = 0;
417 }
418 else
419 {
420 Mat33T Mr = *M_inv;
421 Vec3T Br = *B_vec;
422 Vec3T MBr(Mr(0, 0) * Br[0] + Mr(0, 1) * Br[1] + Mr(0, 2) * Br[2],
423 Mr(1, 0) * Br[0] + Mr(1, 1) * Br[1] + Mr(1, 2) * Br[2],
424 Mr(2, 0) * Br[0] + Mr(2, 1) * Br[1] + Mr(2, 2) * Br[2]);
425 signNormal(MBr, *normal);
426 }
427 }
428
429 // Cached data
430 mutable Mat_<Vec3T> V_;
431 mutable Mat_<Vec9T> M_inv_;
432};

444template<typename T, typename U>
445void multiply_by_K_inv(const Matx<T, 3, 3>& K_inv, U a, U b, U c, Vec<T, 3>& res)
446{
447 res[0] = (T)(K_inv(0, 0) * a + K_inv(0, 1) * b + K_inv(0, 2) * c);
448 res[1] = (T)(K_inv(1, 1) * b + K_inv(1, 2) * c);
449 res[2] = (T)c;
450}

521 template<typename DepthDepth, typename ContainerDepth>
522 Mat computeImpl(const Mat_<DepthDepth>& depthIn, Mat& normals) const
523 {
524 const int r = 5; // used to be 7
525 const int sample_step = r;
526 const int square_size = ((2 * r / sample_step) + 1);
527 long offsets[square_size * square_size];
528 long offsets_x[square_size * square_size];
529 long offsets_y[square_size * square_size];
530 long offsets_x_x[square_size * square_size];
531 long offsets_x_y[square_size * square_size];
532 long offsets_y_y[square_size * square_size];
533 for (int j = -r, index = 0; j <= r; j += sample_step)
534 for (int i = -r; i <= r; i += sample_step, ++index)
535 {
536 offsets_x[index] = i;
537 offsets_y[index] = j;
538 offsets_x_x[index] = i * i;
539 offsets_x_y[index] = i * j;
540 offsets_y_y[index] = j * j;
541 offsets[index] = j * this->cols + i;
542 }
543
544 // Define K_inv by hand, just for higher accuracy
545 Mat33T K_inv = Matx<T, 3, 3>::eye(), kmat;
546 this->K.copyTo(kmat);
547 K_inv(0, 0) = 1.0f / kmat(0, 0);
548 K_inv(0, 1) = -kmat(0, 1) / (kmat(0, 0) * kmat(1, 1));
549 K_inv(0, 2) = (kmat(0, 1) * kmat(1, 2) - kmat(0, 2) * kmat(1, 1)) / (kmat(0, 0) * kmat(1, 1));
550 K_inv(1, 1) = 1 / kmat(1, 1);
551 K_inv(1, 2) = -kmat(1, 2) / kmat(1, 1);
552
553 Vec3T X1_minus_X, X2_minus_X;
554
555 ContainerDepth difference_threshold((ContainerDepth)differenceThreshold);
556 //TODO: fixit, difference threshold should not depend on input type
557 difference_threshold *= (ContainerDepth)(std::is_same<DepthDepth, ushort>::value ? 1000.0 : 1.0);
558 normals.setTo(std::numeric_limits<DepthDepth>::quiet_NaN());
559 for (int y = r; y < this->rows - r - 1; ++y)
560 {
561 const DepthDepth* p_line = reinterpret_cast<const DepthDepth*>(depthIn.ptr(y, r));
562 Vec4T* normal = normals.ptr<Vec4T>(y, r);
563
564 for (int x = r; x < this->cols - r - 1; ++x)
565 {
566 DepthDepth d = p_line[0];
567
568 // accum
569 long A[4];
570 A[0] = A[1] = A[2] = A[3] = 0;
571 ContainerDepth b[2];
572 b[0] = b[1] = 0;
573 for (unsigned int i = 0; i < square_size * square_size; ++i) {
574 // We need to cast to ContainerDepth in case we have unsigned DepthDepth
575 ContainerDepth delta = ContainerDepth(p_line[offsets[i]]) - ContainerDepth(d);
576 if (std::abs(delta) > difference_threshold)
577 continue;
578
579 A[0] += offsets_x_x[i];
580 A[1] += offsets_x_y[i];
581 A[3] += offsets_y_y[i];
582 b[0] += offsets_x[i] * delta;
583 b[1] += offsets_y[i] * delta;
584 }
585
586 // solve for the optimal gradient D of equation (8)
587 long det = A[0] * A[3] - A[1] * A[1];
588 // We should divide the following two by det, but instead, we multiply
589 // X1_minus_X and X2_minus_X by det (which does not matter as we normalize the normals)
590 // Therefore, no division is done: this is only for speedup
591 ContainerDepth dx = (A[3] * b[0] - A[1] * b[1]);
592 ContainerDepth dy = (-A[1] * b[0] + A[0] * b[1]);
593
594 // Compute the dot product
595 //Vec3T X = K_inv * Vec3T(x, y, 1) * depth(y, x);
596 //Vec3T X1 = K_inv * Vec3T(x + 1, y, 1) * (depth(y, x) + dx);
597 //Vec3T X2 = K_inv * Vec3T(x, y + 1, 1) * (depth(y, x) + dy);
598 //Vec3T nor = (X1 - X).cross(X2 - X);
599 multiply_by_K_inv(K_inv, d * det + (x + 1) * dx, y * dx, dx, X1_minus_X);
600 multiply_by_K_inv(K_inv, x * dy, d * det + (y + 1) * dy, dy, X2_minus_X);
601 Vec3T nor = X1_minus_X.cross(X2_minus_X);
602 signNormal(nor, *normal);
603
604 ++p_line;
605 ++normal;
606 }
607 }
608
609 return normals;
610 }
611
612 double differenceThreshold;

622template<typename T>
623class SRI : public RgbdNormalsImpl<T>
624{
625public:
626 typedef Matx<T, 3, 3> Mat33T;
627 typedef Vec<T, 9> Vec9T;
628 typedef Vec<T, 4> Vec4T;
629 typedef Vec<T, 3> Vec3T;
630
631 SRI(int _rows, int _cols, int _windowSize, const Mat& _K) :
632 RgbdNormalsImpl<T>(_rows, _cols, _windowSize, _K, RgbdNormals::RGBD_NORMALS_METHOD_SRI),
633 phi_step_(0),
634 theta_step_(0)
635 { }
636
639 virtual void cache() const CV_OVERRIDE
640 {
641 if (!this->cacheIsDirty)
642 return;
643
644 Mat_<T> cos_theta, sin_theta, cos_phi, sin_phi;
645 computeThetaPhi<T>(this->rows, this->cols, this->K, cos_theta, sin_theta, cos_phi, sin_phi);
646
647 // Create the derivative kernels
648 getDerivKernels(kx_dx_, ky_dx_, 1, 0, this->windowSize, true, this->dtype);
649 getDerivKernels(kx_dy_, ky_dy_, 0, 1, this->windowSize, true, this->dtype);
650
651 // Get the mapping function for SRI
652 float min_theta = (float)std::asin(sin_theta(0, 0)), max_theta = (float)std::asin(sin_theta(0, this->cols - 1));
653 float min_phi = (float)std::asin(sin_phi(0, this->cols / 2 - 1)), max_phi = (float)std::asin(sin_phi(this->rows - 1, this->cols / 2 - 1));
654
655 std::vector<Point3f> points3d(this->cols * this->rows);
656 R_hat_.create(this->rows, this->cols);
657 phi_step_ = float(max_phi - min_phi) / (this->rows - 1);
658 theta_step_ = float(max_theta - min_theta) / (this->cols - 1);
659 for (int phi_int = 0, k = 0; phi_int < this->rows; ++phi_int)
660 {
661 float phi = min_phi + phi_int * phi_step_;
662 float phi_sin = std::sin(phi), phi_cos = std::cos(phi);
663 for (int theta_int = 0; theta_int < this->cols; ++theta_int, ++k)
664 {
665 float theta = min_theta + theta_int * theta_step_;
666 float theta_sin = std::sin(theta), theta_cos = std::cos(theta);
667 // Store the 3d point to project it later
668 Point3f pp(theta_sin * phi_cos, phi_sin, theta_cos * phi_cos);
669 points3d[k] = pp;
670
671 // Cache the rotation matrix and negate it
672 Matx<T, 3, 3> mat = Matx<T, 3, 3> (0, 1, 0, 0, 0, 1, 1, 0, 0) *
673 Matx<T, 3, 3> (theta_cos, -theta_sin, 0, theta_sin, theta_cos, 0, 0, 0, 1) *
674 Matx<T, 3, 3> (phi_cos, 0, -phi_sin, 0, 1, 0, phi_sin, 0, phi_cos);
675
676 for (unsigned char i = 0; i < 3; ++i)
677 mat(i, 1) = mat(i, 1) / phi_cos;
678 // The second part of the matrix is never explained in the paper ... but look at the wikipedia normal article
679 mat(0, 0) = mat(0, 0) - 2 * pp.x;
680 mat(1, 0) = mat(1, 0) - 2 * pp.y;
681 mat(2, 0) = mat(2, 0) - 2 * pp.z;
682
683 R_hat_(phi_int, theta_int) = Vec9T(mat.val);
684 }
685 }
686
687 map_.create(this->rows, this->cols);
688 projectPoints(points3d, Mat(3, 1, CV_32FC1, Scalar::all(0.0f)), Mat(3, 1, CV_32FC1, Scalar::all(0.0f)), this->K, Mat(), map_);
689 map_ = map_.reshape(2, this->rows);
690 convertMaps(map_, Mat(), xy_, fxy_, CV_16SC2);
691
692 //map for converting from Spherical coordinate space to Euclidean space
693 euclideanMap_.create(this->rows, this->cols);
694 Matx<T, 3, 3> km(this->K);
695 float invFx = (float)(1.0f / km(0, 0)), cx = (float)(km(0, 2));
696 double invFy = 1.0f / (km(1, 1)), cy = km(1, 2);
697 for (int i = 0; i < this->rows; i++)
698 {
699 float y = (float)((i - cy) * invFy);
700 for (int j = 0; j < this->cols; j++)
701 {
702 float x = (j - cx) * invFx;
703 float theta = std::atan(x);
704 float phi = std::asin(y / std::sqrt(x * x + y * y + 1.0f));
705
706 euclideanMap_(i, j) = Vec2f((theta - min_theta) / theta_step_, (phi - min_phi) / phi_step_);
707 }
708 }
709 //convert map to 2 maps in short format for increasing speed in remap function
710 convertMaps(euclideanMap_, Mat(), invxy_, invfxy_, CV_16SC2);
711
712 // Update the kernels: the steps are due to the fact that derivatives will be computed on a grid where
713 // the step is not 1. Only need to do it on one dimension as it computes derivatives in only one direction
714 kx_dx_ /= theta_step_;
715 ky_dy_ /= phi_step_;
716
717 this->cacheIsDirty = false;
718 }
719
724 virtual void compute(const Mat& in, Mat& normals_out) const CV_OVERRIDE
725 {
726 const Mat_<T>& r_non_interp = in;
727
728 // Interpolate the radial image to make derivatives meaningful
729 Mat_<T> r;
730 // higher quality remapping does not help here
731 remap(r_non_interp, r, xy_, fxy_, INTER_LINEAR);
732
733 // Compute the derivatives with respect to theta and phi
734 // TODO add bilateral filtering (as done in kinfu)
735 Mat_<T> r_theta, r_phi;
736 sepFilter2D(r, r_theta, r.depth(), kx_dx_, ky_dx_);
737 //current OpenCV version sometimes corrupts r matrix after second call of sepFilter2D
738 //it depends on resolution, be careful
739 sepFilter2D(r, r_phi, r.depth(), kx_dy_, ky_dy_);
740
741 // Fill the result matrix
742 Mat_<Vec4T> normals(this->rows, this->cols);
743
744 const T* r_theta_ptr = r_theta[0], * r_theta_ptr_end = r_theta_ptr + this->rows * this->cols;
745 const T* r_phi_ptr = r_phi[0];
746 const Mat33T* R = reinterpret_cast<const Mat33T*>(R_hat_[0]);
747 const T* r_ptr = r[0];
748 Vec4T* normal = normals[0];
749 for (; r_theta_ptr != r_theta_ptr_end; ++r_theta_ptr, ++r_phi_ptr, ++R, ++r_ptr, ++normal)
750 {
751 if (cvIsNaN(*r_ptr))
752 {
753 (*normal)[0] = *r_ptr;
754 (*normal)[1] = *r_ptr;
755 (*normal)[2] = *r_ptr;
756 (*normal)[3] = 0;
757 }
758 else
759 {
760 T r_theta_over_r = (*r_theta_ptr) / (*r_ptr);
761 T r_phi_over_r = (*r_phi_ptr) / (*r_ptr);
762 // R(1,1) is 0
763 signNormal((*R)(0, 0) + (*R)(0, 1) * r_theta_over_r + (*R)(0, 2) * r_phi_over_r,
764 (*R)(1, 0) + (*R)(1, 2) * r_phi_over_r,
765 (*R)(2, 0) + (*R)(2, 1) * r_theta_over_r + (*R)(2, 2) * r_phi_over_r, *normal);
766 }
767 }
768
769 remap(normals, normals_out, invxy_, invfxy_, INTER_LINEAR);
770 normal = normals_out.ptr<Vec4T>(0);
771 Vec4T* normal_end = normal + this->rows * this->cols;
772 for (; normal != normal_end; ++normal)
773 signNormal((*normal)[0], (*normal)[1], (*normal)[2], *normal);
774 }
775
776 // Cached data
778 mutable Mat_<Vec9T> R_hat_;
779 mutable float phi_step_, theta_step_;
780
782 mutable Mat kx_dx_, ky_dx_, kx_dy_, ky_dy_;
784 mutable Mat_<Vec2f> map_;
785 mutable Mat xy_, fxy_;
786
787 mutable Mat_<Vec2f> euclideanMap_;
788 mutable Mat invxy_, invfxy_;
789};

793/* Uses the simpliest possible method for normals calculation: calculates cross product between two vectors
794(pointAt(x+1, y) - pointAt(x, y)) and (pointAt(x, y+1) - pointAt(x, y)) */
795
796template<typename DataType>
797class CrossProduct : public RgbdNormalsImpl<DataType>
798{
799public:
800 typedef Vec<DataType, 3> Vec3T;
801 typedef Vec<DataType, 4> Vec4T;
802 typedef Point3_<DataType> Point3T;
803
804 CrossProduct(int _rows, int _cols, int _windowSize, const Mat& _K) :
805 RgbdNormalsImpl<DataType>(_rows, _cols, _windowSize, _K, RgbdNormals::RGBD_NORMALS_METHOD_CROSS_PRODUCT)
806 { }
807
810 virtual void cache() const CV_OVERRIDE
811 {
812 this->cacheIsDirty = false;
813 }
814
815 static inline Point3T fromVec(Vec4T v)
816 {
817 return {v[0], v[1], v[2]};
818 }
819
820 static inline Vec4T toVec4(Point3T p)
821 {
822 return {p.x, p.y, p.z, 0};
823 }
824
825 static inline bool haveNaNs(Point3T p)
826 {
827 return cvIsNaN(p.x) || cvIsNaN(p.y) || cvIsNaN(p.z);
828 }
829
835 virtual void compute(const Mat& points, Mat& normals) const CV_OVERRIDE
836 {
837 for(int y = 0; y < this->rows; y++)
838 {
839 const Vec4T* ptsRow0 = points.ptr<Vec4T>(y);
840 const Vec4T* ptsRow1 = (y < this->rows - 1) ? points.ptr<Vec4T>(y + 1) : nullptr;
841 Vec4T *normRow = normals.ptr<Vec4T>(y);
842
843 for (int x = 0; x < this->cols; x++)
844 {
845 Point3T v00 = fromVec(ptsRow0[x]);
846 const float qnan = std::numeric_limits<float>::quiet_NaN();
847 Point3T n(qnan, qnan, qnan);
848
849 if ((x < this->cols - 1) && (y < this->rows - 1) && !haveNaNs(v00))
850 {
851 Point3T v01 = fromVec(ptsRow0[x + 1]);
852 Point3T v10 = fromVec(ptsRow1[x]);
853
854 if (!haveNaNs(v01) && !haveNaNs(v10))
855 {
856 Vec3T vec = (v10 - v00).cross(v01 - v00);
857 n = normalize(vec);
858 }
859 }
860
861 normRow[x] = toVec4(n);
862 }
863 }
864 }
865};

871Ptr<RgbdNormals> RgbdNormals::create(int rows, int cols, int depth, InputArray K, int windowSize, float diffThreshold, RgbdNormalsMethod method)
872{
873 CV_Assert(rows > 0 && cols > 0 && (depth == CV_32F || depth == CV_64F));
874 CV_Assert(windowSize == 1 || windowSize == 3 || windowSize == 5 || windowSize == 7);
875 CV_Assert(K.cols() == 3 && K.rows() == 3 && (K.depth() == CV_32F || K.depth() == CV_64F));
876
877 Mat mK = K.getMat();
878 Ptr<RgbdNormals> ptr;
879 switch (method)
880 {
881 case (RGBD_NORMALS_METHOD_FALS):
882 {
883 if (depth == CV_32F)
884 ptr = makePtr<FALS<float> >(rows, cols, windowSize, mK);
885 else
886 ptr = makePtr<FALS<double>>(rows, cols, windowSize, mK);
887 break;
888 }
889 case (RGBD_NORMALS_METHOD_LINEMOD):
890 {
891 CV_Assert(diffThreshold > 0);
892 if (depth == CV_32F)
893 ptr = makePtr<LINEMOD<float> >(rows, cols, windowSize, mK, diffThreshold);
894 else
895 ptr = makePtr<LINEMOD<double>>(rows, cols, windowSize, mK, diffThreshold);
896 break;
897 }
898 case RGBD_NORMALS_METHOD_SRI:
899 {
900 if (depth == CV_32F)
901 ptr = makePtr<SRI<float> >(rows, cols, windowSize, mK);
902 else
903 ptr = makePtr<SRI<double>>(rows, cols, windowSize, mK);
904 break;
905 }
906 case RGBD_NORMALS_METHOD_CROSS_PRODUCT:
907 {
908 if (depth == CV_32F)
909 ptr = makePtr<CrossProduct<float> >(rows, cols, windowSize, mK);
910 else
911 ptr = makePtr<CrossProduct<double>>(rows, cols, windowSize, mK);
912 break;
913 }
914 default:
915 CV_Error(Error::StsBadArg, "Unknown normals compute algorithm");
916 }
917
918 return ptr;
919}
