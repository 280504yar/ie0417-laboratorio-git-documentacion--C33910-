146 bool checkCompatibility(const OpenCV_API_Header& api_header, unsigned int abi_version, unsigned int api_version, bool checkMinorOpenCVVersion)
147 {
148 if (api_header.opencv_version_major != CV_VERSION_MAJOR)
149 {
150 CV_LOG_ERROR(NULL, "Video I/O: wrong OpenCV major version used by plugin '" << api_header.api_description << "': " <<
151 cv::format("%d.%d, OpenCV version is '" CV_VERSION "'", api_header.opencv_version_major, api_header.opencv_version_minor))
152 return false;
153 }
154 if (!checkMinorOpenCVVersion)
155 {
156 // no checks for OpenCV minor version
157 }
158 else if (api_header.opencv_version_minor != CV_VERSION_MINOR)
159 {
160 CV_LOG_ERROR(NULL, "Video I/O: wrong OpenCV minor version used by plugin '" << api_header.api_description << "': " <<
161 cv::format("%d.%d, OpenCV version is '" CV_VERSION "'", api_header.opencv_version_major, api_header.opencv_version_minor))
162 return false;
163 }
164 CV_LOG_INFO(NULL, "Video I/O: initialized '" << api_header.api_description << "': built with "
165 << cv::format("OpenCV %d.%d (ABI/API = %d/%d)",
166 api_header.opencv_version_major, api_header.opencv_version_minor,
167 api_header.min_api_version, api_header.api_version)
168 << ", current OpenCV version is '" CV_VERSION "' (ABI/API = " << abi_version << "/" << api_version << ")"
169 );
170 if (api_header.min_api_version != abi_version) // future: range can be here
171 {
172 // actually this should never happen due to checks in plugin's init() function
173 CV_LOG_ERROR(NULL, "Video I/O: plugin is not supported due to incompatible ABI = " << api_header.min_api_version);
174 return false;
175 }
176 if (api_header.api_version != api_version)
177 {
178 CV_LOG_INFO(NULL, "Video I/O: NOTE: plugin is supported, but there is API version mismath: "
179 << cv::format("plugin API level (%d) != OpenCV API level (%d)", api_header.api_version, api_version));
180 if (api_header.api_version < api_version)
181 {
182 CV_LOG_INFO(NULL, "Video I/O: NOTE: some functionality may be unavailable due to lack of support by plugin implementation");
183 }
184 }
185 return true;
186 }

301static
302std::vector<FileSystemPath_t> getPluginCandidates(const std::string& baseName)
303{
304 using namespace cv::utils;
305 using namespace cv::utils::fs;
306 const std::string baseName_l = toLowerCase(baseName);
307 const std::string baseName_u = toUpperCase(baseName);
308 const FileSystemPath_t baseName_l_fs = toFileSystemPath(baseName_l);
309 std::vector<FileSystemPath_t> paths;
310 const std::vector<std::string> paths_ = getConfigurationParameterPaths("OPENCV_VIDEOIO_PLUGIN_PATH", std::vector<std::string>());
311 if (paths_.size() != 0)
312 {
313 for (size_t i = 0; i < paths_.size(); i++)
314 {
315 paths.push_back(toFileSystemPath(paths_[i]));
316 }
317 }
318 else
319 {
320 FileSystemPath_t binaryLocation;
321 if (getBinLocation(binaryLocation))
322 {
323 binaryLocation = getParent(binaryLocation);
324#ifndef CV_VIDEOIO_PLUGIN_SUBDIRECTORY
325 paths.push_back(binaryLocation);
326#else
327 paths.push_back(binaryLocation + toFileSystemPath("/") + toFileSystemPath(CV_VIDEOIO_PLUGIN_SUBDIRECTORY_STR));
328#endif
329 }
330 }
331 const std::string default_expr = libraryPrefix() + "opencv_videoio_" + baseName_l + "*" + librarySuffix();
332 const std::string plugin_expr = getConfigurationParameterString((std::string("OPENCV_VIDEOIO_PLUGIN_") + baseName_u).c_str(), default_expr.c_str());
333 std::vector<FileSystemPath_t> results;
334#ifdef _WIN32
335 FileSystemPath_t moduleName = toFileSystemPath(libraryPrefix() + "opencv_videoio_" + baseName_l + librarySuffix());
336 if (baseName_u == "FFMPEG") // backward compatibility
337 {
338 const std::string ffmpeg_env_path = cv::utils::getConfigurationParameterString("OPENCV_FFMPEG_DLL_DIR");
339 if (!ffmpeg_env_path.empty())
340 {
341 results.push_back(toFileSystemPath(ffmpeg_env_path + "\\" + toPrintablePath(moduleName)));
342 }
343 }
344 if (plugin_expr != default_expr)
345 {
346 moduleName = toFileSystemPath(plugin_expr);
347 results.push_back(moduleName);
348 }
349 for (const FileSystemPath_t& path : paths)
350 {
351 results.push_back(path + L"\\" + moduleName);
352 }
353 results.push_back(moduleName);
354#if defined(_DEBUG) && defined(DEBUG_POSTFIX)
355 if (baseName_u == "FFMPEG") // backward compatibility
356 {
357 const FileSystemPath_t templ = toFileSystemPath(CVAUX_STR(DEBUG_POSTFIX) ".dll");
358 FileSystemPath_t nonDebugName(moduleName);
359 size_t suf = nonDebugName.rfind(templ);
360 if (suf != FileSystemPath_t::npos)
361 {
362 nonDebugName.replace(suf, suf + templ.size(), L".dll");
363 results.push_back(nonDebugName);
364 }
365 }
366#endif // _DEBUG && DEBUG_POSTFIX
367#else
368 CV_LOG_INFO(NULL, "VideoIO plugin (" << baseName << "): glob is '" << plugin_expr << "', " << paths.size() << " location(s)");
369 for (const std::string& path : paths)
370 {
371 if (path.empty())
372 continue;
373 std::vector<std::string> candidates;
374 cv::glob(utils::fs::join(path, plugin_expr), candidates);
375 // Prefer candisates with higher versions
376 // TODO: implemented accurate versions-based comparator
377 std::sort(candidates.begin(), candidates.end(), std::greater<std::string>());
378 CV_LOG_INFO(NULL, " - " << path << ": " << candidates.size());
379 copy(candidates.begin(), candidates.end(), back_inserter(results));
380 }
381#endif
382 CV_LOG_INFO(NULL, "Found " << results.size() << " plugin(s) for " << baseName);
383 return results;
384}

386void PluginBackendFactory::loadPlugin()
387{
388 for (const FileSystemPath_t& plugin : getPluginCandidates(baseName_))
389 {
390 auto lib = makePtr<cv::plugin::impl::DynamicLib>(plugin);
391 if (!lib->isLoaded())
392 continue;
393 try
394 {
395 Ptr<PluginBackend> pluginBackend = makePtr<PluginBackend>(lib);
396 if (!pluginBackend)
397 return;
398 if (pluginBackend->capture_api_)
399 {
400 if (pluginBackend->capture_api_->v0.id != id_)
401 {
402 CV_LOG_ERROR(NULL, "Video I/O: plugin '" << pluginBackend->capture_api_->api_header.api_description <<
403 "': unexpected backend ID: " <<
404 pluginBackend->capture_api_->v0.id << " vs " << (int)id_ << " (expected)");
405 return;
406 }
407 }
408 if (pluginBackend->writer_api_)
409 {
410 if (pluginBackend->writer_api_->v0.id != id_)
411 {
412 CV_LOG_ERROR(NULL, "Video I/O: plugin '" << pluginBackend->writer_api_->api_header.api_description <<
413 "': unexpected backend ID: " <<
414 pluginBackend->writer_api_->v0.id << " vs " << (int)id_ << " (expected)");
415 return;
416 }
417 }
418 if (pluginBackend->plugin_api_)
419 {
420 if (pluginBackend->plugin_api_->v0.captureAPI != id_)
421 {
422 CV_LOG_ERROR(NULL, "Video I/O: plugin '" << pluginBackend->plugin_api_->api_header.api_description <<
423 "': unexpected backend ID: " <<
424 pluginBackend->plugin_api_->v0.captureAPI << " vs " << (int)id_ << " (expected)");
425 return;
426 }
427 }
428 if (pluginBackend->capture_api_ == NULL && pluginBackend->writer_api_ == NULL
429 && pluginBackend->plugin_api_ == NULL)
430 {
431 CV_LOG_ERROR(NULL, "Video I/O: no compatible plugin API for backend ID: " << (int)id_);
432 return;
433 }
434 backend = pluginBackend;
435 return;
436 }
437 catch (...)
438 {
439 CV_LOG_WARNING(NULL, "Video I/O: exception during plugin initialization: " << toPrintablePath(plugin) << ". SKIP");
440 }
441 }

447class PluginCapture : public cv::IVideoCapture
448{
449 const OpenCV_VideoIO_Capture_Plugin_API* plugin_api_;
450 CvPluginCapture capture_;
451 Ptr<IStreamReader> readStream_;
452
453public:
454 static
455 Ptr<PluginCapture> create(const OpenCV_VideoIO_Capture_Plugin_API* plugin_api,
456 const std::string &filename, const Ptr<IStreamReader>& stream, int camera, const VideoCaptureParameters& params)
457 {
458 CV_Assert(plugin_api);
459 CV_Assert(plugin_api->v0.Capture_release);
460
461 CvPluginCapture capture = NULL;
462 if (stream && plugin_api->api_header.api_version >= 2 && plugin_api->v2.Capture_open_stream)
463 {
464 std::vector<int> vint_params = params.getIntVector();
465 int* c_params = vint_params.data();
466 unsigned n_params = (unsigned)(vint_params.size() / 2);
467
468 if (CV_ERROR_OK == plugin_api->v2.Capture_open_stream(
469 stream.get(),
470 [](void* opaque, char* buffer, long long size) -> long long {
471 CV_LOG_VERBOSE(NULL, 0, "IStreamReader::read(" << size << ")...");
472 auto is = reinterpret_cast<IStreamReader*>(opaque);
473 try {
474 return is->read(buffer, size);
475 } catch (...) {
476 CV_LOG_WARNING(NULL, "IStreamReader::read(" << size << ") failed");
477 return 0;
478 }
479 },
480 [](void* opaque, long long offset, int way) -> long long {
481 CV_LOG_VERBOSE(NULL, 0, "IStreamReader::seek(" << offset << ", way=" << way << ")...");
482 auto is = reinterpret_cast<IStreamReader*>(opaque);
483 try {
484 return is->seek(offset, way);
485 } catch (...) {
486 CV_LOG_WARNING(NULL, "IStreamReader::seek(" << offset << ", way=" << way << ") failed");
487 return -1;
488 }
489 }, c_params, n_params, &capture))
490 {
491 CV_Assert(capture);
492 return makePtr<PluginCapture>(plugin_api, capture, stream);
493 }
494 }
495 else if (stream)
496 return Ptr<PluginCapture>();
497
498 if (plugin_api->api_header.api_version >= 1 && plugin_api->v1.Capture_open_with_params)
499 {
500 std::vector<int> vint_params = params.getIntVector();
501 int* c_params = vint_params.data();
502 unsigned n_params = (unsigned)(vint_params.size() / 2);
503
504 if (CV_ERROR_OK == plugin_api->v1.Capture_open_with_params(
505 filename.empty() ? 0 : filename.c_str(), camera, c_params, n_params, &capture))
506 {
507 CV_Assert(capture);
508 return makePtr<PluginCapture>(plugin_api, capture);
509 }
510 }
511 else if (plugin_api->v0.Capture_open)
512 {
513 if (CV_ERROR_OK == plugin_api->v0.Capture_open(filename.empty() ? 0 : filename.c_str(), camera, &capture))
514 {
515 CV_Assert(capture);
516 Ptr<PluginCapture> cap = makePtr<PluginCapture>(plugin_api, capture);
517 if (cap && !params.empty())
518 {
519 applyParametersFallback(cap, params);
520 }
521 return cap;
522 }
523 }
524
525 return Ptr<PluginCapture>();
526 }

562 }
563 static CvResult CV_API_CALL retrieve_callback(int stream_idx, const unsigned char* data, int step, int width, int height, int type, void* userdata)
564 {
565 CV_UNUSED(stream_idx);
566 cv::_OutputArray* dst = static_cast<cv::_OutputArray*>(userdata);
567 if (!dst)
568 return CV_ERROR_FAIL;
569 int depth = CV_MAT_DEPTH(type);
570 // [TODO] Remove this condition after rebuilding plugins or add a new
571 // version of plugins. Convert type from the old one to the new one (5 bits)
572 if (depth > 7) {
573 type = CV_MAKETYPE((type & 7), (type >> 3) + 1);
574 }
575 cv::Mat(cv::Size(width, height), type, (void*)data, step).copyTo(*dst);
576 return CV_ERROR_OK;
577 }
578 bool retrieveFrame(int idx, cv::OutputArray img) CV_OVERRIDE
579 {
580 bool res = false;
581 if (plugin_api_->v0.Capture_retreive)
582 if (CV_ERROR_OK == plugin_api_->v0.Capture_retreive(capture_, idx, retrieve_callback, (cv::_OutputArray*)&img))
583 res = true;
584 return res;
585 }
586 bool isOpened() const CV_OVERRIDE
587 {
588 return capture_ != NULL; // TODO always true
589 }
590 int getCaptureDomain() CV_OVERRIDE
591 {
592 return plugin_api_->v0.id;
593 }

599class PluginWriter : public cv::IVideoWriter
600{
601 const OpenCV_VideoIO_Writer_Plugin_API* plugin_api_;
602 CvPluginWriter writer_;
603
604public:
605 static
606 Ptr<PluginWriter> create(const OpenCV_VideoIO_Writer_Plugin_API* plugin_api,
607 const std::string& filename, int fourcc, double fps, const cv::Size& sz,
608 const VideoWriterParameters& params)
609 {
610 CV_Assert(plugin_api);
611 CV_Assert(plugin_api->v0.Writer_release);
612 CV_Assert(!filename.empty());
613
614 CvPluginWriter writer = NULL;
615
616 if (plugin_api->api_header.api_version >= 1 && plugin_api->v1.Writer_open_with_params)
617 {
618 std::vector<int> vint_params = params.getIntVector();
619 int* c_params = &vint_params[0];
620 unsigned n_params = (unsigned)(vint_params.size() / 2);
621
622 if (CV_ERROR_OK == plugin_api->v1.Writer_open_with_params(filename.c_str(), fourcc, fps, sz.width, sz.height, c_params, n_params, &writer))
623 {
624 CV_Assert(writer);
625 return makePtr<PluginWriter>(plugin_api, writer);
626 }
627 }
628 else if (plugin_api->v0.Writer_open)
629 {
630 const bool isColor = params.get(VIDEOWRITER_PROP_IS_COLOR, true);
631 const int depth = params.get(VIDEOWRITER_PROP_DEPTH, CV_8U);
632 if (depth != CV_8U)
633 {
634 CV_LOG_WARNING(NULL, "Video I/O plugin doesn't support (due to lower API level) creation of VideoWriter with depth != CV_8U");
635 return Ptr<PluginWriter>();
636 }
637 if (params.warnUnusedParameters())
638 {
639 CV_LOG_ERROR(NULL, "VIDEOIO: unsupported parameters in VideoWriter, see logger INFO channel for details");
640 return Ptr<PluginWriter>();
