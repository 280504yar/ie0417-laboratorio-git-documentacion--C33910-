20// 1 2
21// 3 4 5 6
22// 7 8 9 10 11 12 13 14

24// Children of node i: 2*i+1, 2*i+2
25// Parent of node i: (i-1) / 2

27// Heap property: a parent is greater than (or equal to) its children.

45static inline void _swap_default(zmaxheap_t *heap, int a, int b)
46{
47 float t = heap->values[a];
48 heap->values[a] = heap->values[b];
49 heap->values[b] = t;
50
51 cv::AutoBuffer<char> tmp(heap->el_sz);
52 memcpy(tmp.data(), &heap->data[a*heap->el_sz], heap->el_sz);
53 memcpy(&heap->data[a*heap->el_sz], &heap->data[b*heap->el_sz], heap->el_sz);
54 memcpy(&heap->data[b*heap->el_sz], tmp.data(), heap->el_sz);
55}

57static inline void _swap_pointer(zmaxheap_t *heap, int a, int b)
58{
59 float t = heap->values[a];
60 heap->values[a] = heap->values[b];
61 heap->values[b] = t;
62
63 void **pp = (void**) heap->data;
64 void *tmp = pp[a];
65 pp[a] = pp[b];
66 pp[b] = tmp;
67}

90static void _zmaxheap_ensure_capacity(zmaxheap_t *heap, int capacity)
91{
92 if (heap->alloc >= capacity)
93 return;
94
95 int newcap = heap->alloc;
96
97 while (newcap < capacity) {
98 if (newcap < MIN_CAPACITY) {
99 newcap = MIN_CAPACITY;
100 continue;
101 }
102
103 newcap *= 2;
104 }
105
106 heap->values = (float*)realloc(heap->values, newcap * sizeof(float));
107 heap->data = (char*)realloc(heap->data, newcap * heap->el_sz);
108 heap->alloc = newcap;
109}

111void zmaxheap_add(zmaxheap_t *heap, void *p, float v)
112{
113 _zmaxheap_ensure_capacity(heap, heap->size + 1);
114
115 int idx = heap->size;
116
117 heap->values[idx] = v;
118 memcpy(&heap->data[idx*heap->el_sz], p, heap->el_sz);
119
120 heap->size++;
121
122 while (idx > 0) {
123
124 int parent = (idx - 1) / 2;
125
126 // we're done!
127 if (heap->values[parent] >= v)
128 break;
129
130 // else, swap and recurse upwards.
131 heap->swap(heap, idx, parent);
132 idx = parent;
133 }
134}

136// Removes the item in the heap at the given index. Returns 1 if the
137// item existed. 0 Indicates an invalid idx (heap is smaller than
138// idx). This is mostly intended to be used by zmaxheap_remove_max.
139static int zmaxheap_remove_index(zmaxheap_t *heap, int idx, void *p, float *v)
140{
141 if (idx >= heap->size)
142 return 0;
143
144 // copy out the requested element from the heap.
145 if (v != NULL)
146 *v = heap->values[idx];
147 if (p != NULL)
148 memcpy(p, &heap->data[idx*heap->el_sz], heap->el_sz);
149
150 heap->size--;
151
152 // If this element is already the last one, then there's nothing
153 // for us to do.
154 if (idx == heap->size)
155 return 1;
156
157 // copy last element to first element. (which probably upsets
158 // the heap property).
159 heap->values[idx] = heap->values[heap->size];
160 memcpy(&heap->data[idx*heap->el_sz], &heap->data[heap->el_sz * heap->size], heap->el_sz);
161
162 // now fix the heap. Note, as we descend, we're "pushing down"
163 // the same node the entire time. Thus, while the index of the
164 // parent might change, the parent_score doesn't.
165 int parent = idx;
166 float parent_score = heap->values[idx];
167
168 // descend, fixing the heap.
169 while (parent < heap->size) {
170
171 int left = 2*parent + 1;
172 int right = left + 1;
173
174// assert(parent_score == heap->values[parent]);
175
176 float left_score = (left < heap->size) ? heap->values[left] : -INFINITY;
177 float right_score = (right < heap->size) ? heap->values[right] : -INFINITY;
178
179 // put the biggest of (parent, left, right) as the parent.
180
181 // already okay?
182 if (parent_score >= left_score && parent_score >= right_score)
183 break;
184
185 // if we got here, then one of the children is bigger than the parent.
186 if (left_score >= right_score) {
187 CV_Assert(left < heap->size);
188 heap->swap(heap, parent, left);
189 parent = left;
190 } else {
191 // right_score can't be less than left_score if right_score is -INFINITY.
192 CV_Assert(right < heap->size);
193 heap->swap(heap, parent, right);
194 parent = right;
195 }
196 }
197
198 return 1;
199}
