17template <typename T>
18static std::string pycv_dumpArray(const T* arr, int n)
19{
20 std::ostringstream out;
21 out << "[";
22 for (int i = 0; i < n; ++i)
23 out << " " << arr[i];
24 out << " ]";
25 return out.str();

28static inline std::string getArrayTypeName(PyArrayObject* arr)
29{
30 PyArray_Descr* dtype = PyArray_DESCR(arr);
31 PySafeObject dtype_str(PyObject_Str(reinterpret_cast<PyObject*>(dtype)));
32 if (!dtype_str)
33 {
34 // Fallback to typenum value
35 return cv::format("%d", PyArray_TYPE(arr));
36 }
37 std::string type_name;
38 if (!getUnicodeString(dtype_str, type_name))
39 {
40 // Failed to get string from bytes object - clear set TypeError and
41 // fallback to typenum value
42 PyErr_Clear();
43 return cv::format("%d", PyArray_TYPE(arr));
44 }
45 return type_name;
46}

52// special case, when the converter needs full ArgInfo structure
53template<>
54bool pyopencv_to(PyObject* o, Mat& m, const ArgInfo& info)
55{
56 if(!o || o == Py_None)
57 {
58 if( !m.data )
59 m.allocator = &GetNumpyAllocator();
60 return true;
61 }
62
63 if( PyInt_Check(o) )
64 {
65 double v[] = {static_cast<double>(PyInt_AsLong((PyObject*)o)), 0., 0., 0.};
66 if ( info.arithm_op_src )
67 {
68 // Normally cv.XXX(x) means cv.XXX( (x, 0., 0., 0.) );
69 // However cv.add(mat,x) means cv::add(mat, (x,x,x,x) ).
70 v[1] = v[0];
71 v[2] = v[0];
72 v[3] = v[0];
73 }
74 m = Mat(4, 1, CV_64F, v).clone();
75 return true;
76 }
77 if( PyFloat_Check(o) )
78 {
79 double v[] = {PyFloat_AsDouble((PyObject*)o), 0., 0., 0.};
80
81 if ( info.arithm_op_src )
82 {
83 // Normally cv.XXX(x) means cv.XXX( (x, 0., 0., 0.) );
84 // However cv.add(mat,x) means cv::add(mat, (x,x,x,x) ).
85 v[1] = v[0];
86 v[2] = v[0];
87 v[3] = v[0];
88 }
89 m = Mat(4, 1, CV_64F, v).clone();
90 return true;
91 }
92 if( PyTuple_Check(o) )
93 {
94 // see https://github.com/opencv/opencv/issues/24057
95 const int sz = (int)PyTuple_Size((PyObject*)o);
96 const int sz2 = info.arithm_op_src ? std::max(4, sz) : sz; // Scalar has 4 elements.
97 m = Mat::zeros(sz2, 1, CV_64F);
98 for( int i = 0; i < sz; i++ )
99 {
100 PyObject* oi = PyTuple_GetItem(o, i);
101 if( PyInt_Check(oi) )
102 m.at<double>(i) = (double)PyInt_AsLong(oi);
103 else if( PyFloat_Check(oi) )
104 m.at<double>(i) = (double)PyFloat_AsDouble(oi);
105 else
106 {
107 failmsg("%s is not a numerical tuple", info.name);
108 m.release();
109 return false;
110 }
111 }
112 return true;
113 }
114
115 if( !PyArray_Check(o) )
116 {
117 failmsg("%s is not a numpy array, neither a scalar", info.name);
118 return false;
119 }
120
121 PyArrayObject* oarr = (PyArrayObject*) o;
122
123 if (info.outputarg && !PyArray_ISWRITEABLE(oarr))
124 {
125 failmsg("%s marked as output argument, but provided NumPy array "
126 "marked as readonly", info.name);
127 return false;
128 }
129
130 bool needcopy = false, needcast = false;
131 int typenum = PyArray_TYPE(oarr), new_typenum = typenum;
132 int type = numpyTypeToCvDepth(typenum);
133
134 if( type < 0 )
135 {
136 if( typenum == NPY_INT64 || typenum == NPY_LONG || typenum == NPY_LONGLONG )
137 {
138 needcopy = needcast = true;
139 new_typenum = NPY_INT;
140 type = CV_32S;
141 }
142 else
143 {
144 const std::string dtype_name = getArrayTypeName(oarr);
145 failmsg("%s data type = %s is not supported", info.name,
146 dtype_name.c_str());
147 return false;
148 }
149 }
150
151#ifndef CV_MAX_DIM
152 const int CV_MAX_DIM = 32;
153#endif
154
155 int ndims = PyArray_NDIM(oarr);
156 if(ndims >= CV_MAX_DIM)
157 {
158 failmsg("%s dimensionality (=%d) is too high", info.name, ndims);
159 return false;
160 }
161
162 size_t elemsize = CV_ELEM_SIZE1(type);
163 const npy_intp* _sizes = PyArray_DIMS(oarr);
164 const npy_intp* _strides = PyArray_STRIDES(oarr);
165
166 CV_LOG_DEBUG(NULL, "Incoming ndarray '" << info.name << "': ndims=" << ndims << " _sizes=" << pycv_dumpArray(_sizes, ndims) << " _strides=" << pycv_dumpArray(_strides, ndims));
167
168 bool ismultichannel = ndims == 3 && !info.nd_mat;
169 if (pyopencv_Mat_TypePtr && PyObject_TypeCheck(o, pyopencv_Mat_TypePtr))
170 {
171 bool wrapChannels = false;
172 PyObject* pyobj_wrap_channels = PyObject_GetAttrString(o, "wrap_channels");
173 if (pyobj_wrap_channels)
174 {
175 if (!pyopencv_to_safe(pyobj_wrap_channels, wrapChannels, ArgInfo("cv.Mat.wrap_channels", 0)))
176 {
177 // TODO extra message
178 Py_DECREF(pyobj_wrap_channels);
179 return false;
180 }
181 Py_DECREF(pyobj_wrap_channels);
182 }
183 ismultichannel = wrapChannels && ndims >= 1;
184 }
185
186 for( int i = ndims-1; i >= 0 && !needcopy; i-- )
187 {
188 // these checks handle cases of
189 // a) multi-dimensional (ndims > 2) arrays, as well as simpler 1- and 2-dimensional cases
190 // b) transposed arrays, where _strides[] elements go in non-descending order
191 // c) flipped arrays, where some of _strides[] elements are negative
192 // the _sizes[i] > 1 is needed to avoid spurious copies when NPY_RELAXED_STRIDES is set
193 if( (i == ndims-1 && _sizes[i] > 1 && (size_t)_strides[i] != elemsize) ||
194 (i < ndims-1 && _sizes[i] > 1 && _strides[i] < _strides[i+1]) )
195 needcopy = true;
196 }
197
198 if (ismultichannel)
199 {
200 int channels = ndims >= 1 ? (int)_sizes[ndims - 1] : 1;
201 if (channels < 1 || channels > CV_CN_MAX)
202 {
203 failmsg("%s unable to wrap channels, invalid count (%d, must be in [1, %d])", info.name, (int)channels, (int)CV_CN_MAX);
204 return false;
205 }
206 ndims--;
207 type |= CV_MAKETYPE(0, channels);
208
209 if (ndims >= 1 && _strides[ndims - 1] != (npy_intp)elemsize*_sizes[ndims])
210 needcopy = true;
211
212 elemsize = CV_ELEM_SIZE(type);
213 }
214
215 if (needcopy)
216 {
217 if (info.outputarg)
218 {
219 failmsg("Layout of the output array %s is incompatible with cv::Mat", info.name);
220 return false;
221 }
222
223 if( needcast ) {
224 o = PyArray_Cast(oarr, new_typenum);
225 oarr = (PyArrayObject*) o;
226 }
227 else {
228 oarr = PyArray_GETCONTIGUOUS(oarr);
229 o = (PyObject*) oarr;
230 }
231
232 _strides = PyArray_STRIDES(oarr);
233 }
234
235 int size[CV_MAX_DIM+1] = {};
236 size_t step[CV_MAX_DIM+1] = {};
237
238 // Normalize strides in case NPY_RELAXED_STRIDES is set
239 size_t default_step = elemsize;
240 for ( int i = ndims - 1; i >= 0; --i )
241 {
242 size[i] = (int)_sizes[i];
243 if ( size[i] > 1 )
244 {
245 step[i] = (size_t)_strides[i];
246 default_step = step[i] * size[i];
247 }
248 else
249 {
250 step[i] = default_step;
251 default_step *= size[i];
252 }
253 }
254
255 // see https://github.com/opencv/opencv/issues/24057
256 if ( ( info.arithm_op_src ) && ( ndims == 1 ) && ( size[0] <= 4 ) )
257 {
258 const int sz = size[0]; // Real Data Length(1, 2, 3 or 4)
259 const int sz2 = 4; // Scalar has 4 elements.
260 m = Mat::zeros(sz2, 1, CV_64F);
261
262 // Fill the Mat with array elements
263 bool filled = true;
264 const char *base_ptr = PyArray_BYTES(oarr);
265 for(int i = 0; i < sz && filled; i++ )
266 {
267 PyObject* oi = PyArray_GETITEM(oarr, base_ptr + step[0] * i); // new object
268 if( PyInt_Check(oi) )
269 m.at<double>(i) = (double)PyInt_AsLong(oi);
270 else if( PyFloat_Check(oi) )
271 m.at<double>(i) = (double)PyFloat_AsDouble(oi);
272 else
273 {
274 failmsg("%s has some non-numerical elements", info.name);
275 m.release();
276 filled = false;
277 }
278
279 Py_DECREF(oi);
280 }
281
282 if(needcopy)
283 Py_DECREF(o);
284
285 return filled;
286 }
287
288 // 0D (scalar) tensor: allocate a proper scalar Mat and copy the value.
289 if (ndims == 0)
290 {
291 m.fit(0, nullptr, type);
292 memcpy(m.data, PyArray_DATA(oarr), CV_ELEM_SIZE(type));
293 if (needcopy)
294 Py_DECREF(o);
295 return true;
296 }
297
298#if 1
299 CV_LOG_DEBUG(NULL, "Construct Mat: ndims=" << ndims << " size=" << pycv_dumpArray(size, ndims) << " step=" << pycv_dumpArray(step, ndims) << " type=" << cv::typeToString(type));
300#endif
301
302 m = Mat(ndims, size, type, PyArray_DATA(oarr), step);
303 m.u = GetNumpyAllocator().allocate(o, ndims, size, type, step);
304 m.addref();
305
306 if( !needcopy )
307 {
308 Py_INCREF(o);
309 }
310 m.allocator = &GetNumpyAllocator();
311
312 return true;
313}

315template<>
316PyObject* pyopencv_from(const cv::Mat& m)
317{
318 // NumPy has no bfloat16 dtype: widen CV_16BF to float32 (lossless).
319 if( m.depth() == CV_16BF )
320 {
321 cv::Mat m32f;
322 ERRWRAP2(m.convertTo(m32f, CV_32F));
323 return pyopencv_from(m32f);
324 }
325 if( m.empty() )
326 {
327 // empty() also catches a live buffer with a zero-length dim: return an empty array, not None.
328 if( m.dims >= 1 )
329 {
330 int cn = m.channels();
331 int typenum = cvDepthToNumpyType(m.depth());
332 int dims = m.dims;
333 cv::AutoBuffer<npy_intp> _sizes(dims + 1);
334 for( int i = 0; i < dims; i++ )
335 _sizes[i] = (npy_intp)m.size[i];
336 if( cn > 1 )
337 _sizes[dims++] = cn;
338 return PyArray_SimpleNew(dims, _sizes.data(), typenum);
339 }
340 Py_RETURN_NONE;
341 }
342 // 0D (scalar) Mat: return a true 0D numpy array.
343 if( m.dims == 0 )
344 {
345 int typenum = cvDepthToNumpyType(CV_MAT_DEPTH(m.type()));
346 PyObject* o = PyArray_SimpleNew(0, nullptr, typenum);
347 if( o )
348 memcpy(PyArray_DATA((PyArrayObject*)o), m.data, CV_ELEM_SIZE(m.type()));
349 return o;
350 }
351 cv::Mat temp, *p = (cv::Mat*)&m;
352 if(!p->u || p->allocator != &GetNumpyAllocator())
353 {
354 temp.allocator = &GetNumpyAllocator();
355 ERRWRAP2(m.copyTo(temp));
356 p = &temp;
357 }
358 PyObject* o = (PyObject*)p->u->userdata;
359 Py_INCREF(o);
360 return o;
361}

425 for (Py_ssize_t i = 0; i < PySequence_Size(o); i++) {
426 SafeSeqItem item_wrap(o, i);
427 PyObject *item = item_wrap.item;
428 if (PyFloat_Check(item) || PyInt_Check(item)) {
429 s[(int)i] = PyFloat_AsDouble(item);
430 } else {
431 failmsg("Scalar value for argument '%s' is not numeric", info.name);
432 return false;
433 }
434 }
435 } else {
436 if (PyFloat_Check(o) || PyInt_Check(o)) {
437 s = PyFloat_AsDouble(o);
438 } else {
439 failmsg("Scalar value for argument '%s' is not numeric", info.name);
440 return false;
441 }
442 }
443 return true;
444}

729template<>
730bool pyopencv_to(PyObject* obj, String &value, const ArgInfo& info)
731{
732 if(!obj || obj == Py_None)
733 {
734 return true;
735 }
736 std::string str;
737
738#if ((PY_VERSION_HEX >= 0x03060000) && !defined(Py_LIMITED_API)) || (Py_LIMITED_API >= 0x03060000)
739 PyObject* path_obj = NULL;
740 if (info.pathlike)
741 {
742 path_obj = PyOS_FSPath(obj);
743 if (PyErr_Occurred())
744 {
745 failmsg("Expected '%s' to be a str or path-like object", info.name);
746 return false;
747 }
748 obj = path_obj;
749 }
750#endif
751
752 bool result = false;
753 if (getUnicodeString(obj, str))
754 {
755 value = str;
756 result = true;
757 }
758 else
759 {
760 if (!PyErr_Occurred())
761 {
762#ifdef Py_LIMITED_API
763 failmsg("Can't convert object to 'str' for '%s'", info.name);
764#else
765 failmsg("Can't convert object of type '%s' to 'str' for '%s'",
766 obj->ob_type->tp_name, info.name);
767#endif
768 }
769 }
770
771#if ((PY_VERSION_HEX >= 0x03060000) && !defined(Py_LIMITED_API)) || (Py_LIMITED_API >= 0x03060000)
772 Py_XDECREF(path_obj);
773#endif
774
775 return result;
776}

869// --- RotatedRect
870
871static inline bool convertToRotatedRect(PyObject* obj, RotatedRect& dst)
872{
873 PyObject* type = PyObject_Type(obj);
874 if (getPyObjectAttr(type, "__module__") == MODULESTR &&
875 getPyObjectNameAttr(type) == "RotatedRect")
876 {
877 struct pyopencv_RotatedRect_t
878 {
879 PyObject_HEAD
880 cv::RotatedRect v;
881 };
882 dst = reinterpret_cast<pyopencv_RotatedRect_t*>(obj)->v;
883
884 Py_DECREF(type);
885 return true;
886 }
887 Py_DECREF(type);
888 return false;
889}
890
891template<>
892bool pyopencv_to(PyObject* obj, RotatedRect& dst, const ArgInfo& info)
893{
894 if (!obj || obj == Py_None)
895 {
896 return true;
897 }
898 // This is a workaround for compatibility with an initialization from tuple.
899 // Allows import RotatedRect as an object.
900 if (convertToRotatedRect(obj, dst))
901 {
902 return true;
903 }
904 if (!PySequence_Check(obj))
905 {
906 failmsg("Can't parse '%s' as RotatedRect."
907 "Input argument doesn't provide sequence protocol",
908 info.name);
909 return false;
910 }
911 const std::size_t sequenceSize = PySequence_Size(obj);
912 if (sequenceSize != 3)
913 {
914 failmsg("Can't parse '%s' as RotatedRect. Expected sequence length 3, got %lu",
915 info.name, sequenceSize);
916 return false;
917 }
918 {
919 const String centerItemName = format("'%s' center point", info.name);
920 const ArgInfo centerItemInfo(centerItemName.c_str(), 0);
921 SafeSeqItem centerItem(obj, 0);
922 if (!pyopencv_to(centerItem.item, dst.center, centerItemInfo))
923 {
924 return false;
925 }
926 }
927 {
928 const String sizeItemName = format("'%s' size", info.name);
929 const ArgInfo sizeItemInfo(sizeItemName.c_str(), 0);
930 SafeSeqItem sizeItem(obj, 1);
931 if (!pyopencv_to(sizeItem.item, dst.size, sizeItemInfo))
932 {
933 return false;
934 }
935 }
936 {
937 const String angleItemName = format("'%s' angle", info.name);
938 const ArgInfo angleItemInfo(angleItemName.c_str(), 0);
939 SafeSeqItem angleItem(obj, 2);
940 if (!pyopencv_to(angleItem.item, dst.angle, angleItemInfo))
941 {
942 return false;
943 }
944 }
945 return true;
946}
