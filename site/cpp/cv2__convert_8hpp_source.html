322template<typename K, typename V>
323bool pyopencv_to(PyObject *obj, std::map<K,V> &map, const ArgInfo& info)
324{
325 if (!obj || obj == Py_None)
326 {
327 return true;
328 }
329
330 PyObject* py_key = nullptr;
331 PyObject* py_value = nullptr;
332 Py_ssize_t pos = 0;
333
334 if (!PyDict_Check(obj)) {
335 failmsg("Can't parse '%s'. Input argument isn't dict or"
336 " an instance of subtype of the dict type", info.name);
337 return false;
338 }
339
340 while(PyDict_Next(obj, &pos, &py_key, &py_value))
341 {
342 K cpp_key;
343 if (!pyopencv_to(py_key, cpp_key, ArgInfo("key", 0))) {
344 failmsg("Can't parse dict key. Key on position %lu has a wrong type", pos);
345 return false;
346 }
347
348 V cpp_value;
349 if (!pyopencv_to(py_value, cpp_value, ArgInfo("value", 0))) {
350 failmsg("Can't parse dict value. Value on position %lu has a wrong type", pos);
351 return false;
352 }
353
354 map.emplace(cpp_key, cpp_value);
355 }
356 return true;
357}

359template <typename Tp>
360static bool pyopencv_to_generic_vec(PyObject* obj, std::vector<Tp>& value, const ArgInfo& info)
361{
362 if (!obj || obj == Py_None)
363 {
364 return true;
365 }
366 if (info.nd_mat && PyArray_Check(obj))
367 {
368 /*
369 If obj is marked as nd mat and of array type, it is parsed to a single
370 mat in the target vector to avoid being split into multiple mats
371 */
372 value.resize(1);
373 if (!pyopencv_to(obj, value.front(), info))
374 {
375 failmsg("Can't parse '%s'. Array item has a wrong type", info.name);
376 return false;
377 }
378 }
379 else // parse as sequence
380 {
381 if (!PySequence_Check(obj))
382 {
383 failmsg("Can't parse '%s'. Input argument doesn't provide sequence protocol", info.name);
384 return false;
385 }
386 const size_t n = static_cast<size_t>(PySequence_Size(obj));
387 value.resize(n);
388 for (size_t i = 0; i < n; i++)
389 {
390 SafeSeqItem item_wrap(obj, i);
391 if (!pyopencv_to(item_wrap.item, value[i], info))
392 {
393 failmsg("Can't parse '%s'. Sequence item with index %lu has a wrong type", info.name, i);
394 return false;
395 }
396 }
397 }
398 return true;
399}

428template <typename Tp>
429static PyObject* pyopencv_from_generic_vec(const std::vector<Tp>& value)
430{
431 Py_ssize_t n = static_cast<Py_ssize_t>(value.size());
432 PySafeObject seq(PyTuple_New(n));
433 for (Py_ssize_t i = 0; i < n; i++)
434 {
435 PyObject* item = pyopencv_from(value[i]);
436 // If item can't be assigned - PyTuple_SetItem raises exception and returns -1.
437 if (!item || PyTuple_SetItem(seq, i, item) == -1)
438 {
439 return NULL;
440 }
441 }
442 return seq.release();
443}

494template <typename Tp>
495struct pyopencvVecConverter
496{
497 typedef typename std::vector<Tp>::iterator VecIt;
498
499 static bool to(PyObject* obj, std::vector<Tp>& value, const ArgInfo& info)
500 {
501 if (!PyArray_Check(obj))
502 {
503 return pyopencv_to_generic_vec(obj, value, info);
504 }
505 // If user passed an array it is possible to make faster conversions in several cases
506 PyArrayObject* array_obj = reinterpret_cast<PyArrayObject*>(obj);
507 const NPY_TYPES target_type = asNumpyType<Tp>();
508 const NPY_TYPES source_type = static_cast<NPY_TYPES>(PyArray_TYPE(array_obj));
509 if (target_type == NPY_OBJECT)
510 {
511 // Non-planar arrays representing objects (e.g. array of N Rect is an array of shape Nx4) have NPY_OBJECT
512 // as their target type.
513 return pyopencv_to_generic_vec(obj, value, info);
514 }
515 if (PyArray_NDIM(array_obj) > 1)
516 {
517 failmsg("Can't parse %dD array as '%s' vector argument", PyArray_NDIM(array_obj), info.name);
518 return false;
519 }
520 if (target_type != source_type)
521 {
522 // Source type requires conversion
523 // Allowed conversions for target type is handled in the corresponding pyopencv_to function
524 return pyopencv_to_generic_vec(obj, value, info);
525 }
526 // For all other cases, all array data can be directly copied to std::vector data
527 // Simple `memcpy` is not possible because NumPy array can reference a slice of the bigger array:
528 // ```
529 // arr = np.ones((8, 4, 5), dtype=np.int32)
530 // convertible_to_vector_of_int = arr[:, 0, 1]
531 // ```
532 value.resize(static_cast<size_t>(PyArray_SIZE(array_obj)));
533 const npy_intp item_step = PyArray_STRIDE(array_obj, 0) / PyArray_ITEMSIZE(array_obj);
534 const Tp* data_ptr = static_cast<Tp*>(PyArray_DATA(array_obj));
535 for (VecIt it = value.begin(); it != value.end(); ++it, data_ptr += item_step) {
536 *it = *data_ptr;
537 }
538 return true;
539 }

541 static PyObject* from(const std::vector<Tp>& value)
542 {
543 if (value.empty())
544 {
545 return PyTuple_New(0);
546 }
547 return from(value, ::traits::IsRepresentableAsMatDataType<Tp>());
548 }
549
550private:
551 static PyObject* from(const std::vector<Tp>& value, ::traits::FalseType)
552 {
553 // Underlying type is not representable as Mat Data Type
554 return pyopencv_from_generic_vec(value);
555 }
556
557 static PyObject* from(const std::vector<Tp>& value, ::traits::TrueType)
558 {
559 // Underlying type is representable as Mat Data Type, so faster return type is available
560 typedef cv::DataType<Tp> DType;
561 typedef typename DType::channel_type UnderlyingArrayType;
562
563 // If Mat is always exposed as NumPy array this code path can be reduced to the following snipped:
564 // Mat src(value);
565 // PyObject* array = pyopencv_from(src);
566 // return PyArray_Squeeze(reinterpret_cast<PyArrayObject*>(array));
567 // This puts unnecessary restrictions on Mat object those might be avoided without losing the performance.
568 // Moreover, this version is a bit faster, because it doesn't create temporary objects with reference counting.
569
570 const NPY_TYPES target_type = asNumpyType<UnderlyingArrayType>();
571 const int cols = DType::channels;
572 PyObject* array = NULL;
573 if (cols == 1)
574 {
575 npy_intp dims = static_cast<npy_intp>(value.size());
576 array = PyArray_SimpleNew(1, &dims, target_type);
577 }
578 else
579 {
580 npy_intp dims[2] = {static_cast<npy_intp>(value.size()), cols};
581 array = PyArray_SimpleNew(2, dims, target_type);
582 }
583 if(!array)
584 {
585 // NumPy arrays with shape (N, 1) and (N) are not equal, so correct error message should distinguish
586 // them too.
587 cv::String shape;
588 if (cols > 1)
589 {
590 shape = cv::format("(%d x %d)", static_cast<int>(value.size()), cols);
591 }
592 else
593 {
594 shape = cv::format("(%d)", static_cast<int>(value.size()));
595 }
596 const cv::String error_message = cv::format("Can't allocate NumPy array for vector with dtype=%d and shape=%s",
597 static_cast<int>(target_type), shape.c_str());
598 emit_failmsg(PyExc_MemoryError, error_message.c_str());
599 return array;
600 }
601 // Fill the array
602 PyArrayObject* array_obj = reinterpret_cast<PyArrayObject*>(array);
603 UnderlyingArrayType* array_data = static_cast<UnderlyingArrayType*>(PyArray_DATA(array_obj));
604 // if Tp is representable as Mat DataType, so the following cast is pretty safe...
605 const UnderlyingArrayType* value_data = reinterpret_cast<const UnderlyingArrayType*>(value.data());
606 memcpy(array_data, value_data, sizeof(UnderlyingArrayType) * value.size() * static_cast<size_t>(cols));
607 return array;
608 }
609};

612template<std::size_t I = 0, typename... Tp>
613inline typename std::enable_if<I == sizeof...(Tp), void>::type
614convert_to_python_tuple(const std::tuple<Tp...>&, PyObject*) { }

616template<std::size_t I = 0, typename... Tp>
617inline typename std::enable_if<I < sizeof...(Tp), void>::type
618convert_to_python_tuple(const std::tuple<Tp...>& cpp_tuple, PyObject* py_tuple)
619{
620 PyObject* item = pyopencv_from(std::get<I>(cpp_tuple));
621
622 if (!item)
623 return;
624
625 PyTuple_SetItem(py_tuple, I, item);
626 convert_to_python_tuple<I + 1, Tp...>(cpp_tuple, py_tuple);
627}

629template<typename... Ts>
630PyObject* pyopencv_from(const std::tuple<Ts...>& cpp_tuple)
631{
632 size_t size = sizeof...(Ts);
633 PyObject* py_tuple = PyTuple_New(size);
634 convert_to_python_tuple(cpp_tuple, py_tuple);
635 size_t actual_size = PyTuple_Size(py_tuple);
636
637 if (actual_size < size)
638 {
639 Py_DECREF(py_tuple);
640 return NULL;
641 }
642
643 return py_tuple;
644}
