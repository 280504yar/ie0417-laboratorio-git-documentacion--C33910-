311double Regression::getElem(cv::Mat& m, int y, int x, int cn)
312{
313 switch (m.depth())
314 {
315 case CV_8U: return *(m.ptr<unsigned char>(y, x) + cn);
316 case CV_8S: return *(m.ptr<signed char>(y, x) + cn);
317 case CV_16U: return *(m.ptr<unsigned short>(y, x) + cn);
318 case CV_16S: return *(m.ptr<signed short>(y, x) + cn);
319 case CV_32S: return *(m.ptr<signed int>(y, x) + cn);
320 case CV_32F: return *(m.ptr<float>(y, x) + cn);
321 case CV_64F: return *(m.ptr<double>(y, x) + cn);
322 default: return 0;

353void Regression::verify(cv::FileNode node, cv::Mat actual, double eps, std::string argname, ERROR_TYPE err)
354{
355 if (!actual.empty() && actual.dims < 2) return;
356
357 double expect_min = (double)node["min"];
358 double expect_max = (double)node["max"];
359
360 if (err == ERROR_RELATIVE)
361 eps *= std::max(std::abs(expect_min), std::abs(expect_max));
362
363 double actual_min, actual_max;
364 cv::minMaxIdx(actual, &actual_min, &actual_max);
365
366 ASSERT_NEAR(expect_min, actual_min, eps)
367 << argname << " has unexpected minimal value" << std::endl;
368 ASSERT_NEAR(expect_max, actual_max, eps)
369 << argname << " has unexpected maximal value" << std::endl;
370
371 cv::FileNode last = node["last"];
372 double actual_last = getElem(actual, actual.size.p[0] - 1, actual.size.p[1] - 1, actual.channels() - 1);
373 int expect_cols = (int)last["x"] + 1;
374 int expect_rows = (int)last["y"] + 1;
375 ASSERT_EQ(expect_cols, actual.size.p[1])
376 << argname << " has unexpected number of columns" << std::endl;
377 ASSERT_EQ(expect_rows, actual.size.p[0])
378 << argname << " has unexpected number of rows" << std::endl;
379
380 double expect_last = (double)last["val"];
381 ASSERT_NEAR(expect_last, actual_last, eps)
382 << argname << " has unexpected value of the last element" << std::endl;
383
384 cv::FileNode rng1 = node["rng1"];
385 int x1 = rng1["x"];
386 int y1 = rng1["y"];
387 int cn1 = rng1["cn"];
388
389 double expect_rng1 = (double)rng1["val"];
390 // it is safe to use x1 and y1 without checks here because we have already
391 // verified that mat size is the same as recorded
392 double actual_rng1 = getElem(actual, y1, x1, cn1);
393
394 ASSERT_NEAR(expect_rng1, actual_rng1, eps)
395 << argname << " has unexpected value of the ["<< x1 << ":" << y1 << ":" << cn1 <<"] element" << std::endl;
396
397 cv::FileNode rng2 = node["rng2"];
398 int x2 = rng2["x"];
399 int y2 = rng2["y"];
400 int cn2 = rng2["cn"];
401
402 double expect_rng2 = (double)rng2["val"];

436static int countViolations(const cv::Mat& expected, const cv::Mat& actual, const cv::Mat& diff, double eps, double* max_violation = 0, double* max_allowed = 0)
437{
438 cv::Mat diff64f;
439 diff.reshape(1).convertTo(diff64f, CV_64F);
440
441 cv::Mat expected_abs = cv::abs(expected.reshape(1));
442 cv::Mat actual_abs = cv::abs(actual.reshape(1));
443 cv::Mat maximum, mask;
444 cv::max(expected_abs, actual_abs, maximum);
445 cv::multiply(maximum, cv::Vec<double, 1>(eps), maximum, CV_64F);
446 cv::compare(diff64f, maximum, mask, cv::CMP_GT);
447
448 int v = cv::countNonZero(mask);
449
450 if (v > 0 && max_violation != 0 && max_allowed != 0)
451 {
452 int loc[10] = {0};
453 cv::minMaxIdx(maximum, 0, max_allowed, 0, loc, mask);
454 *max_violation = diff64f.at<double>(loc[0], loc[1]);
455 }
456
457 return v;
458}

460void Regression::verify(cv::FileNode node, cv::InputArray array, double eps, ERROR_TYPE err)
461{
462 int expected_kind = (int)node["kind"];
463 int expected_type = (int)node["type"];
464 int array_type = array.type();
465 if (array_type != expected_type) {
466 // temporary hack; we optimistically assume that type in the computed and expected array should be the same.
467 // if they are different, it must be because of the change in type representation between OpenCV 5.x and OpenCV 2.x,3.x,4.x.
468 // need to add "type5" or something like that and use it in the newer files. Then type will always mean 'earlier than 5.x type'.
469 int depth = expected_type & 7;
470 int channels = ((expected_type >> 3) & 127) + 1;
471 expected_type = CV_MAKETYPE(depth, channels);
472 }
473 ASSERT_EQ(expected_kind, array.kind()) << " Argument \"" << node.name() << "\" has unexpected kind";
474 ASSERT_EQ(expected_type, array.type()) << " Argument \"" << node.name() << "\" has unexpected type";
475
476 cv::FileNode valnode = node["val"];
477 if (isVector(array))
478 {
479 int expected_length = (int)node["len"];
480 ASSERT_EQ(expected_length, (int)array.total()) << " Vector \"" << node.name() << "\" has unexpected length";
481 int idx = node["idx"];
482
483 cv::Mat actual = array.getMat(idx);
484
485 if (valnode.isNone())
486 {
487 ASSERT_LE((size_t)26, actual.total() * (size_t)actual.channels())
488 << " \"" << node.name() << "[" << idx << "]\" has unexpected number of elements";
489 verify(node, actual, eps, cv::format("%s[%d]", node.name().c_str(), idx), err);
490 }
491 else
492 {
493 cv::Mat expected;
494 valnode >> expected;
495
496 if(expected.empty())
497 {
498 ASSERT_TRUE(actual.empty())
499 << " expected empty " << node.name() << "[" << idx<< "]";
500 }
501 else
502 {
503 ASSERT_EQ(expected.size(), actual.size())
504 << " " << node.name() << "[" << idx<< "] has unexpected size";
505
506 cv::Mat diff;
507 cv::absdiff(expected, actual, diff);
508
509 if (err == ERROR_ABSOLUTE)
510 {
511 if (!cv::checkRange(diff, true, 0, 0, eps))
512 {
513 if(expected.total() * expected.channels() < 12)
514 std::cout << " Expected: " << std::endl << expected << std::endl << " Actual:" << std::endl << actual << std::endl;
515
516 double max;
517 cv::minMaxIdx(diff.reshape(1), 0, &max);
518
519 FAIL() << " Absolute difference (=" << max << ") between argument \""
520 << node.name() << "[" << idx << "]\" and expected value is greater than " << eps;
521 }
522 }
523 else if (err == ERROR_RELATIVE)
524 {
525 double maxv, maxa;
526 int violations = countViolations(expected, actual, diff, eps, &maxv, &maxa);
527 if (violations > 0)
528 {
529 if(expected.total() * expected.channels() < 12)
530 std::cout << " Expected: " << std::endl << expected << std::endl << " Actual:" << std::endl << actual << std::endl;
531
532 FAIL() << " Relative difference (" << maxv << " of " << maxa << " allowed) between argument \""
533 << node.name() << "[" << idx << "]\" and expected value is greater than " << eps << " in " << violations << " points";
534 }
535 }
536 }
537 }
538 }
539 else
540 {
541 if (valnode.isNone())
542 {
543 ASSERT_LE((size_t)26, array.total() * (size_t)array.channels())
544 << " Argument \"" << node.name() << "\" has unexpected number of elements";
545 verify(node, array.getMat(), eps, "Argument \"" + node.name() + "\"", err);
546 }
547 else
548 {
549 cv::Mat expected;
550 valnode >> expected;
551 cv::Mat actual = array.getMat();
552
553 if(expected.empty())
554 {
555 ASSERT_TRUE(actual.empty())
556 << " expected empty " << node.name();
557 }
558 else
559 {
560 ASSERT_EQ(expected.size(), actual.size())
561 << " Argument \"" << node.name() << "\" has unexpected size";
562
563 cv::Mat diff;
564 cv::absdiff(expected, actual, diff);
565
566 if (err == ERROR_ABSOLUTE)
567 {
568 if (!cv::checkRange(diff, true, 0, 0, eps))
569 {
570 if(expected.total() * expected.channels() < 12)
571 std::cout << " Expected: " << std::endl << expected << std::endl << " Actual:" << std::endl << actual << std::endl;
572
573 double max;
574 cv::minMaxIdx(diff.reshape(1), 0, &max);
575
576 FAIL() << " Difference (=" << max << ") between argument1 \"" << node.name()
577 << "\" and expected value is greater than " << eps;
578 }
579 }
580 else if (err == ERROR_RELATIVE)
581 {
582 double maxv, maxa;
583 int violations = countViolations(expected, actual, diff, eps, &maxv, &maxa);
584 if (violations > 0)
585 {
586 if(expected.total() * expected.channels() < 12)
587 std::cout << " Expected: " << std::endl << expected << std::endl << " Actual:" << std::endl << actual << std::endl;
588
589 FAIL() << " Relative difference (" << maxv << " of " << maxa << " allowed) between argument \"" << node.name()
590 << "\" and expected value is greater than " << eps << " in " << violations << " points";
591 }
592 }

798static void printNodeRec(cv::instr::InstrNode *pNode, cv::instr::InstrNode *pRoot)
799{
800 printf("%s", (pNode->m_payload.m_funName.substr(0, 40) + ((pNode->m_payload.m_funName.size()>40)?"...":"")).c_str());
801
802 // Write instrumentation flags
803 if(pNode->m_payload.m_instrType != cv::instr::TYPE_GENERAL || pNode->m_payload.m_implType != cv::instr::IMPL_PLAIN)
804 {
805 printf("<");
806 if(pNode->m_payload.m_instrType == cv::instr::TYPE_WRAPPER)
807 printf("W");
808 else if(pNode->m_payload.m_instrType == cv::instr::TYPE_FUN)
809 printf("F");
810 else if(pNode->m_payload.m_instrType == cv::instr::TYPE_MARKER)
811 printf("MARK");
812
813 if(pNode->m_payload.m_instrType != cv::instr::TYPE_GENERAL && pNode->m_payload.m_implType != cv::instr::IMPL_PLAIN)
814 printf("_");
815
816 if(pNode->m_payload.m_implType == cv::instr::IMPL_IPP)
817 printf("IPP");
818 else if(pNode->m_payload.m_implType == cv::instr::IMPL_OPENCL)
819 printf("OCL");
820
821 printf(">");
822 }
823
824 if(pNode->m_pParent)
825 {
826 printf(" - TC:%d C:%d", pNode->m_payload.m_threads, pNode->m_payload.m_counter);
827 printf(" T:%.2fms", pNode->m_payload.getTotalMs());
828 if(pNode->m_pParent->m_pParent)
829 printf(" L:%.0f%% G:%.0f%%", calcLocalWeight(pNode), calcGlobalWeight(pNode));
830 }
831 printf("\n");
832
833 {
834 // Group childes by name
835 for(size_t i = 1; i < pNode->m_childs.size(); i++)
836 {
837 if(pNode->m_childs[i-1]->m_payload.m_funName == pNode->m_childs[i]->m_payload.m_funName )
838 continue;
839 for(size_t j = i+1; j < pNode->m_childs.size(); j++)
840 {
841 if(pNode->m_childs[i-1]->m_payload.m_funName == pNode->m_childs[j]->m_payload.m_funName )
842 {
843 cv::swap(pNode->m_childs[i], pNode->m_childs[j]);
844 i++;
845 }
846 }
847 }
848 }
849
850 for(size_t i = 0; i < pNode->m_childs.size(); i++)
851 {
852 printShift(pNode->m_childs[i], pRoot);
853
854 if(i == pNode->m_childs.size()-1)
855 printf("\\---");
856 else
857 printf("|---");
858 printNodeRec(pNode->m_childs[i], pRoot);
859 }

1227{

1230 return;

1232 {
1233 if (wtype == WARMUP_RNG || wtype == WARMUP_WRITE)
1234 {

1236 if (depth == CV_8U)
1237 cv::randu(a, 0, 256);
1238 else if (depth == CV_8S)
1239 cv::randu(a, -128, 128);
1240 else if (depth == CV_16U)
1241 cv::randu(a, 0, 1024);
1242 else if (depth == CV_32F || depth == CV_64F || depth == CV_16F)
1243 cv::randu(a, -1.0, 1.0);
1244 else if (depth == CV_16S || depth == CV_32S)
1245 cv::randu(a, -4096, 4096);
1246 else
1247 CV_Error(cv::Error::StsUnsupportedFormat, "Unsupported format");
1248 }
1249 return;
1250 }
1251 else if (a.kind() != cv::_InputArray::STD_VECTOR_MAT && a.kind() != cv::_InputArray::STD_VECTOR_VECTOR)
1252 warmup_impl(a.getMat(), wtype);
1253 else
1254 {
1255 size_t total = a.total();
1256 for (size_t i = 0; i < total; ++i)
1257 warmup_impl(a.getMat((int)i), wtype);
1258 }

1290bool TestBase::next()
1291{
1292 static int64 lastActivityPrintTime = 0;
1293
1294 if (currentIter != (unsigned int)-1)
1295 {
1296 if (currentIter + 1 != times.size())
1297 ADD_FAILURE() << " next() is called before stopTimer()";
1298 }
1299 else
1300 {
1301 lastActivityPrintTime = 0;
1302 metrics.clear();

1304
1305 cv::theRNG().state = param_seed; //this rng should generate same numbers for each run
1306 ++currentIter;
1307
1308 bool has_next = false;
1309
1310 do {
1311 CV_Assert(currentIter == times.size());
1312 if (currentIter == 0)
1313 {
1314 has_next = true;
1315 break;
1316 }
1317
1318 if (getCurrentPerformanceStrategy() == PERF_STRATEGY_BASE)
1319 {
1320 has_next = currentIter < nIters && totalTime < timeLimit;
1321 }
1322 else
1323 {
1324 CV_Assert(getCurrentPerformanceStrategy() == PERF_STRATEGY_SIMPLE);
1325 if (totalTime - lastActivityPrintTime >= cv::getTickFrequency() * 10)
1326 {
1327 std::cout << '.' << std::endl;
1328 lastActivityPrintTime = totalTime;
1329 }
1330 if (currentIter >= nIters)
1331 {
1332 has_next = false;
1333 break;
1334 }
1335 if (currentIter < minIters)
1336 {
1337 has_next = true;
1338 break;
1339 }
1340
1341 calcMetrics();
1342
1343 if (fabs(metrics.mean) > 1e-6)
1344 has_next = metrics.stddev > perf_stability_criteria * fabs(metrics.mean);
1345 else
1346 has_next = true;
1347 }
1348 } while (false);
1349
1350 if (perf_validation_enabled && !has_next)
1351 {
1352 calcMetrics();
1353 double median_ms = metrics.median * 1000.0f / metrics.frequency;
1354
1355 const ::testing::TestInfo* const test_info = ::testing::UnitTest::GetInstance()->current_test_info();
1356 std::string name = (test_info == 0) ? "" :
1357 std::string(test_info->test_case_name()) + "--" + test_info->name();
1358
1359 if (!perf_validation_results.empty() && !name.empty())
1360 {
1361 std::map<std::string, float>::iterator i = perf_validation_results.find(name);
1362 bool isSame = false;
1363 bool found = false;
1364 bool grow = false;
1365 if (i != perf_validation_results.end())
1366 {
1367 found = true;
1368 double prev_result = i->second;
1369 grow = median_ms > prev_result;
1370 isSame = fabs(median_ms - prev_result) <= perf_validation_criteria * fabs(median_ms);
1371 if (!isSame)
1372 {
1373 if (perfValidationStage == 0)
1374 {
1375 printf("Performance is changed (samples = %d, median):\n %.2f ms (current)\n %.2f ms (previous)\n", (int)times.size(), median_ms, prev_result);
1376 }
1377 }
1378 }
1379 else
1380 {
1381 if (perfValidationStage == 0)
1382 printf("New performance result is detected\n");
1383 }
1384 if (!isSame)
1385 {
1386 if (perfValidationStage < 2)
1387 {
1388 if (perfValidationStage == 0 && currentIter <= minIters * 3 && currentIter < nIters)
1389 {
1390 unsigned int new_minIters = std::max(minIters * 5, currentIter * 3);
1391 printf("Increase minIters from %u to %u\n", minIters, new_minIters);
1392 minIters = new_minIters;
1393 has_next = true;
1394 perfValidationStage++;
1395 }
1396 else if (found && currentIter >= nIters &&
1397 median_ms > perf_validation_time_threshold_ms &&
1398 (grow || metrics.stddev > perf_stability_criteria * fabs(metrics.mean)))
1399 {
1400 CV_TRACE_REGION("idle_delay");
1401 printf("Performance is unstable, it may be a result of overheat problems\n");
1402 printf("Idle delay for %d ms... \n", perf_validation_idle_delay_ms);
1403#if defined _WIN32
1404#ifndef WINRT_8_0
1405 Sleep(perf_validation_idle_delay_ms);
1406#else
1407 WaitForSingleObjectEx(GetCurrentThread(), perf_validation_idle_delay_ms, FALSE);
1408#endif
1409#else
1410 usleep(perf_validation_idle_delay_ms * 1000);
1411#endif
1412 has_next = true;
1413 minIters = std::min(minIters * 5, nIters);
1414 // reset collected samples
1415 currentIter = 0;
1416 times.clear();
1417 metrics.clear();
1418 perfValidationStage += 2;
1419 }
1420 if (!has_next)
1421 {
1422 printf("Assume that current result is valid\n");
1423 }
1424 }
1425 else
1426 {
1427 printf("Re-measured performance result: %.2f ms\n", median_ms);
1428 }
1429 }
1430 }
1431
1432 if (!has_next && !name.empty())
1433 {
1434 savePerfValidationResult(name, (float)median_ms);
1435 }
1436 }
1437
1438#ifdef __ANDROID__
1439 if (log_power_checkpoints)
1440 {
1441 timeval tim;
1442 gettimeofday(&tim, NULL);
1443 unsigned long long t1 = tim.tv_sec * 1000LLU + (unsigned long long)(tim.tv_usec / 1000.f);
1444
1445 if (currentIter == 1) RecordProperty("test_start", cv::format("%llu",t1).c_str());
1446 if (!has_next) RecordProperty("test_complete", cv::format("%llu",t1).c_str());
1447 }
1448#endif
1449
1450 return has_next;
1451}

1516performance_metrics& TestBase::calcMetrics()
1517{
1518 CV_Assert(metrics.samples <= (unsigned int)currentIter);
1519 if ((metrics.samples == (unsigned int)currentIter) || times.size() == 0)
1520 return metrics;
1521
1522 metrics.bytesIn = getTotalInputSize();
1523 metrics.bytesOut = getTotalOutputSize();
1524 metrics.frequency = cv::getTickFrequency();
1525 metrics.samples = (unsigned int)times.size();
1526 metrics.outliers = 0;
1527
1528 if (metrics.terminationReason != performance_metrics::TERM_INTERRUPT && metrics.terminationReason != performance_metrics::TERM_EXCEPTION)
1529 {
1530 if (currentIter == nIters)
1531 metrics.terminationReason = performance_metrics::TERM_ITERATIONS;
1532 else if (totalTime >= timeLimit)
1533 metrics.terminationReason = performance_metrics::TERM_TIME;
1534 else
1535 metrics.terminationReason = performance_metrics::TERM_UNKNOWN;
1536 }
1537
1538 std::sort(times.begin(), times.end());
1539
1540 TimeVector::const_iterator start = times.begin();
1541 TimeVector::const_iterator end = times.end();
1542
1543 if (getCurrentPerformanceStrategy() == PERF_STRATEGY_BASE)
1544 {
1545 //estimate mean and stddev for log(time)
1546 double gmean = 0;
1547 double gstddev = 0;
1548 int n = 0;
1549 for(TimeVector::const_iterator i = times.begin(); i != times.end(); ++i)
1550 {
1551 double x = static_cast<double>(*i)/runsPerIteration;
1552 if (x < DBL_EPSILON) continue;
1553 double lx = log(x);
1554
1555 ++n;
1556 double delta = lx - gmean;
1557 gmean += delta / n;
1558 gstddev += delta * (lx - gmean);
1559 }
1560
1561 gstddev = n > 1 ? sqrt(gstddev / (n - 1)) : 0;
1562
1563 //filter outliers assuming log-normal distribution
1564 //http://stackoverflow.com/questions/1867426/modeling-distribution-of-performance-measurements
1565 if (gstddev > DBL_EPSILON)
1566 {
1567 double minout = exp(gmean - 3 * gstddev) * runsPerIteration;
1568 double maxout = exp(gmean + 3 * gstddev) * runsPerIteration;
1569 while(*start < minout) ++start, ++metrics.outliers;
1570 do --end, ++metrics.outliers; while(*end > maxout);
1571 ++end, --metrics.outliers;
1572 }
1573 }
1574 else if (getCurrentPerformanceStrategy() == PERF_STRATEGY_SIMPLE)
1575 {
1576 metrics.outliers = static_cast<int>(times.size() * param_max_outliers / 100);
1577 for (unsigned int i = 0; i < metrics.outliers; i++)
1578 --end;
1579 }
1580 else
1581 {
1582 CV_Assert(false);
1583 }
1584
1585 int offset = static_cast<int>(start - times.begin());
1586
1587 metrics.min = static_cast<double>(*start)/runsPerIteration;
1588 //calc final metrics
1589 unsigned int n = 0;
1590 double gmean = 0;
1591 double gstddev = 0;
1592 double mean = 0;
1593 double stddev = 0;
1594 unsigned int m = 0;
1595 for(; start != end; ++start)
1596 {
1597 double x = static_cast<double>(*start)/runsPerIteration;
1598 if (x > DBL_EPSILON)
1599 {
1600 double lx = log(x);
1601 ++m;
1602 double gdelta = lx - gmean;
1603 gmean += gdelta / m;
1604 gstddev += gdelta * (lx - gmean);
1605 }
1606 ++n;
1607 double delta = x - mean;
1608 mean += delta / n;
1609 stddev += delta * (x - mean);
1610 }
1611
1612 metrics.mean = mean;
1613 metrics.gmean = exp(gmean);
1614 metrics.gstddev = m > 1 ? sqrt(gstddev / (m - 1)) : 0;
1615 metrics.stddev = n > 1 ? sqrt(stddev / (n - 1)) : 0;
1616 metrics.median = (n % 2
1617 ? (double)times[offset + n / 2]
1618 : 0.5 * (times[offset + n / 2] + times[offset + n / 2 - 1])
1619 ) / runsPerIteration;
1620
1621 return metrics;
1622}

1627
1628 if (HasFailure()) return;
