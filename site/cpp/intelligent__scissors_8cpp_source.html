42#define ACOS_TABLE_SIZE 64
43// acos_table[x + ACOS_TABLE_SIZE] = acos(x / ACOS_TABLE_SIZE) / CV_PI (see add_local_cost)
44// x = [ -ACOS_TABLE_SIZE .. ACOS_TABLE_SIZE ]
45float* getAcosTable()
46{
47 constexpr int N = ACOS_TABLE_SIZE;
48 static bool initialized = false;
49 static float acos_table[2*N + 1] = { 0 };
50 if (!initialized)
51 {
52 const float CV_PI_inv = static_cast<float>(1.0 / CV_PI);
53 for (int i = -N; i <= N; i++)
54 {
55 acos_table[i + N] = acosf(i / (float)N) * CV_PI_inv;
56 }
57 initialized = true;
58 }
59 return acos_table;
60}

330 void extractEdgeFeature_(InputArray image)
331 {
332 CV_TRACE_FUNCTION();
333
334 if (edge_mode == FEATURE_CANNY)
335 {
336 CV_LOG_DEBUG(NULL, "Canny(" << edge_canny_threshold1 << ", " << edge_canny_threshold2 << ")");
337 Mat img_canny;
338 Canny(image, img_canny, edge_canny_threshold1, edge_canny_threshold2, edge_canny_apertureSize, edge_canny_L2gradient);
339#if 0
340 threshold(img_canny, non_edge_feature, 254, 1, THRESH_BINARY_INV);
341#else
342 // Canny result values are 0 or 255
343 bitwise_not(img_canny, non_edge_feature);
344 weight_non_edge_compute = weight_non_edge * (1.0f / 255.0f);
345#endif
346 }
347 else // if (edge_mode == FEATURE_ZERO_CROSSING)
348 {
349 initGrayscale_(image);
350 Mat_<short> laplacian;
351 Laplacian(grayscale_, laplacian, CV_16S, laplacianKernelSize);
352 Mat_<uchar> zero_crossing(src_size, 1);
353
354 const size_t zstep = zero_crossing.step[0];
355 for (int y = 0; y < src_size.height - 1; y++)
356 {
357 const short* row0 = laplacian.ptr<short>(y);
358 const short* row1 = laplacian.ptr<short>(y + 1);
359 uchar* zrow0 = zero_crossing.ptr<uchar>(y);
360 //uchar* zrow1 = zero_crossing.ptr<uchar>(y + 1);
361 for (int x = 0; x < src_size.width - 1; x++)
362 {
363 const int v = row0[x];
364 const int neg_v = -v;
365 // - * 1
366 // 2 3 4
367 const int v1 = row0[x + 1];
368 const int v2 = (x > 0) ? row1[x - 1] : v;
369 const int v3 = row1[x + 0];
370 const int v4 = row1[x + 1];
371 if (v < 0)
372 {
373 if (v1 > 0)
374 {
375 zrow0[x + ((v1 < neg_v) ? 1 : 0)] = 0;
376 }
377 if (v2 > 0)
378 {
379 zrow0[x + ((v2 < neg_v) ? (zstep - 1) : 0)] = 0;
380 }
381 if (v3 > 0)
382 {
383 zrow0[x + ((v3 < neg_v) ? (zstep + 0) : 0)] = 0;
384 }
385 if (v4 > 0)
386 {
387 zrow0[x + ((v4 < neg_v) ? (zstep + 1) : 0)] = 0;
388 }
389 }
390 else
391 {
392 if (v1 < 0)
393 {
394 zrow0[x + ((v1 > neg_v) ? 1 : 0)] = 0;
395 }
396 if (v2 < 0)
397 {
398 zrow0[x + ((v2 > neg_v) ? (zstep - 1) : 0)] = 0;
399 }
400 if (v3 < 0)
401 {
402 zrow0[x + ((v3 > neg_v) ? (zstep + 0) : 0)] = 0;
403 }
404 if (v4 < 0)
405 {
406 zrow0[x + ((v4 > neg_v) ? (zstep + 1) : 0)] = 0;
407 }
408 }
409 }
410 }
411
412 if (edge_gradient_magnitude_min_value > 0)
413 {
414 initImageMagnitude_(image);
415 Mat mask = image_magnitude_ < edge_gradient_magnitude_min_value;
416 zero_crossing.setTo(1, mask); // reset low-amplitude noise
417 }
418
419 non_edge_feature = zero_crossing;
420 }

424 void extractGradientDirection_(InputArray image)
425 {
426 CV_TRACE_FUNCTION();
427
428 initImageMagnitude_(image); // calls internally: initImageDerives_(image);
429 gradient_direction.create(src_size);
430 for (int y = 0; y < src_size.height; y++)
431 {
432 const float* magnitude_row = image_magnitude_.ptr<float>(y);
433 const float* Ix_row = Ix_.ptr<float>(y);
434 const float* Iy_row = Iy_.ptr<float>(y);
435 Point2f* gradient_direction_row = gradient_direction.ptr<Point2f>(y);
436 for (int x = 0; x < src_size.width; x++)
437 {
438 const float m = magnitude_row[x];
439 if (m > FLT_EPSILON)
440 {
441 float m_inv = 1.0f / m;
442 gradient_direction_row[x] = Point2f(Ix_row[x] * m_inv, Iy_row[x] * m_inv);
443 }
444 else
445 {
446 gradient_direction_row[x] = Point2f(0, 0);
447 }
448 }
449 }

452 void extractGradientMagnitude_(InputArray image)
453 {
454 CV_TRACE_FUNCTION();
455
456 initImageMagnitude_(image); // calls internally: initImageDerives_(image);
457 Mat m;
458 double max_m = 0;
459 if (gradient_magnitude_threshold_max > 0)
460 {
461 threshold(image_magnitude_, m, gradient_magnitude_threshold_max, 0, THRESH_TRUNC);
462 max_m = gradient_magnitude_threshold_max;
463 }
464 else
465 {
466 m = image_magnitude_;
467 minMaxLoc(m, 0, &max_m);
468 }
469 if (max_m <= FLT_EPSILON)
470 {
471 CV_LOG_INFO(NULL, "IntelligentScissorsMB: input image gradient is almost zero")
472 gradient_magnitude.create(src_size);
473 gradient_magnitude.setTo(0);
474 }
475 else
476 {
477 m.convertTo(gradient_magnitude, CV_32F, -1.0 / max_m, 1.0); // normalize and inverse to range 0..1
478 }

516 bool add_local_cost(const Point& p, const Point& q, float& cost_p, const float cost_q) const
517 {
518 if ((cost_p += weight_non_edge_compute * non_edge_feature.at<uchar>(q)) < cost_q)
519 {
520 const bool isDiag = (p.x != q.x) && (p.y != q.y);
521
522 float fG = gradient_magnitude.at<float>(q);
523 if (!isDiag)
524 {
525 fG *= sqrt2_inv;
526 }
527
528 if ((cost_p += weight_gradient_magnitude * fG) < cost_q)
529 {
530
531 const Point2f diff((float)(q.x - p.x), (float)(q.y - p.y));
532
533 const Point2f Ip = gradient_direction(p);
534 const Point2f Iq = gradient_direction(q);
535
536 const Point2f Dp(Ip.y, -Ip.x); // D(p) - 90 degrees clockwise
537 const Point2f Dq(Iq.y, -Iq.x); // D(q) - 90 degrees clockwise
538
539 float dp = Dp.dot(diff); // dp(p, q)
540 float dq = Dq.dot(diff); // dq(p, q)
541 if (dp < 0)
542 {
543 dp = -dp; // ensure dp >= 0
544 dq = -dq;
545 }
546
547 if (isDiag)
548 {
549 dp *= sqrt2_inv; // normalize length of (q - p)
550 dq *= sqrt2_inv; // normalize length of (q - p)
551 }
552
553#if 1
554 int dp_i = cvFloor(dp * ACOS_TABLE_SIZE); // dp is in range 0..1
555 dp_i = std::min(ACOS_TABLE_SIZE, std::max(0, dp_i));
556 int dq_i = cvFloor(dq * ACOS_TABLE_SIZE); // dq is in range -1..1
557 dq_i = std::min(ACOS_TABLE_SIZE, std::max(-ACOS_TABLE_SIZE, dq_i));
558 const float fD = acos_table[dp_i + ACOS_TABLE_SIZE] + acos_table[dq_i + ACOS_TABLE_SIZE];
559#else
560 const float CV_PI_inv = static_cast<float>(1.0 / CV_PI);
561 const float fD = (acosf(dp) + acosf(dq)) * CV_PI_inv; // TODO optimize acos calls (through tables)
562#endif
563
564 cost_p += weight_gradient_direction * fD;
565 }
566 }
567 return cost_p < cost_q;
568 }

581 void buildMap(const Point& start_point)
582 {
583 CV_TRACE_FUNCTION();
584
585 CV_Assert(!src_size.empty());
586 CV_Assert(!gradient_magnitude.empty() && "Features are missing. applyImage() must be called first");
587
588 CV_CheckGE(weight_non_edge + weight_gradient_direction + weight_gradient_magnitude, FLT_EPSILON, "");
589
590#if 0 // debug
591 Rect wholeImage(0, 0, src_size.width, src_size.height);
592 Rect roi = Rect(start_point.x - 5, start_point.y - 5, 11, 11) & wholeImage;
593 std::cout << roi << std::endl;
594 std::cout << gradient_magnitude(roi) << std::endl;
595 std::cout << gradient_direction(roi) << std::endl;
596 std::cout << non_edge_feature(roi) << std::endl;
597#endif
598
599 optimalPathsMap.release();
600 optimalPathsMap.create(src_size);
601 optimalPathsMap.setTo(0); // optimalPathsMap(start_point) = 0;
602
603 //
604 // Section 3.2
605 // Live-Wire 2-D DP graph search.
606 //
607
608 Mat_<float> cost_map(src_size, FLT_MAX); // g(q)
609 Mat_<uchar> processed(src_size, (uchar)0); // e(q)
610
611 // Note: std::vector is faster than std::deque
612 // TODO check std::set
613 std::priority_queue< Pix, std::vector<Pix>, std::greater<Pix> > L;
614
615 cost_map(start_point) = 0;
616 L.emplace(Pix{ start_point, 0/*cost*/ });
617
618 while (!L.empty())
619 {
620 Pix pix = L.top(); L.pop();
621 Point q = pix.pt; // 'q' from the article
622 if (processed(q))
623 continue; // already processed (with lower cost, see note below)
624 processed(q) = 1;
625#if 1
626 const float cost_q = pix.cost;
627#else
628 const float cost_q = cost_map(q);
629 CV_Assert(cost_q == pix.cost);
630#endif
631 for (int n = 0; n < 8; n++) // scan neighbours
632 {
633 Point r(q.x + neighbors[n][0], q.y + neighbors[n][1]); // 'r' from the article
634 if (r.x < 0 || r.x >= src_size.width || r.y < 0 || r.y >= src_size.height)
635 continue; // out of range
636
637#if !defined(__EMSCRIPTEN__) // slower in JS
638 float& cost_r = cost_map(r);
639 if (cost_r < cost_q)
640 continue; // already processed
641#else
642 if (processed(r))
643 continue; // already processed
644
645 float& cost_r = cost_map(r);
646 CV_DbgCheckLE(cost_q, cost_r, "INTERNAL ERROR: sorted queue is corrupted");
647#endif
648
649 float cost = cost_q;
650 if (add_local_cost(q, r, cost, cost_r))
651 {
652#if 0 // avoid compiler warning
653 if (cost_r != FLT_MAX)
654 {
655 // In article the point 'r' is removed from the queue L
656 // to be re-inserted again with sorting against new optimized cost.
657 // We can do nothing, because "new point" will be placed before in the sorted queue.
658 // Old point will be skipped through "if (processed(q))" check above after processing of new optimal candidate.
659 //
660 // This approach leads to some performance impact, however it is much smaller than element removal from the sorted queue.
661 // So, do nothing.
662 }
663#endif
664 cost_r = cost;
665 L.emplace(Pix{ r, cost });
666 optimalPathsMap(r) = (uchar)neighbors_encode[n];
667 }
668 }
669 }

672 void getContour(const Point& target, OutputArray contour_, bool backward)
673 {
674 CV_TRACE_FUNCTION();
675
676 CV_Assert(!optimalPathsMap.empty() && "buildMap() must be called before getContour()");
677
678 const int cols = optimalPathsMap.cols;
679 const int rows = optimalPathsMap.rows;
680
681 std::vector<Point> result; result.reserve(512);
682
683 size_t loop_check = 4096;
684 Point pt = target;
685 for (size_t i = 0; i < (size_t)rows * cols; i++) // don't hang on invalid maps
686 {
687 CV_CheckLT(pt.x, cols, "");
688 CV_CheckLT(pt.y, rows, "");
689 result.push_back(pt);
690 int direction = (int)optimalPathsMap(pt);
691 if (direction == 0)
692 break; // stop, start point is reached
693 CV_CheckLT(direction, 9, "Map is invalid");
694 Point next(pt.x + neighbors[direction - 1][0], pt.y + neighbors[direction - 1][1]);
695 pt = next;
696
697 if (result.size() == loop_check) // optional sanity check of invalid maps with loops (don't eat huge amount of memory)
698 {
699 loop_check *= 4; // next limit for loop check
700 for (const auto& pt_check : result)
701 {
702 CV_CheckNE(pt_check, pt, "Map is invalid. Contour loop is detected");
703 }
704 }
705 }
706
707 if (backward)
708 {
709 _InputArray(result).copyTo(contour_);
710 }
711 else
712 {
713 const int N = (int)result.size();
714 const int sz[1] = { N };
715 contour_.create(1, sz, CV_32SC2);
716 Mat_<Point> contour = contour_.getMat();
717 for (int i = 0; i < N; i++)
718 {
719 contour.at<Point>(i) = result[N - (i + 1)];
720 }
721 }
722 }
