12// Scaled-Dot-Product Attention on pre-projected Q, K^T, V. Fuses
13// QK^T -> [scale] -> softmax -> ·V -> Transpose -> Reshape into one
14// MlasFlashAttention call. Used by the SR-attention fusion pass.

26 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
27 const int requiredOutputs,
28 std::vector<MatShape> &outputs,
29 std::vector<MatShape> &internals) const CV_OVERRIDE {
30 CV_CheckEQ(inputs.size(), 3u, "DNN/SDPA: expects Q, K^T, V");
31 const auto &q = inputs[0]; // (B, H, S_q, D)
32 const auto &kT = inputs[1]; // (B, H, D, S_kv)
33 const auto &v = inputs[2]; // (B, H, S_kv, D_v)
34 CV_CheckEQ(q.dims, 4, "DNN/SDPA: Q must be 4D");
35 CV_CheckEQ(kT.dims, 4, "DNN/SDPA: K must be 4D");
36 CV_CheckEQ(v.dims, 4, "DNN/SDPA: V must be 4D");
37 const int B = q[0], H = q[1], S_q = q[2], D = q[3];
38 const int S_kv = v[2], D_v = v[3];
39 CV_CheckEQ(kT[0], B, "DNN/SDPA: K batch");
40 CV_CheckEQ(kT[1], H, "DNN/SDPA: K heads");
41 CV_CheckEQ(kT[2], D, "DNN/SDPA: K^T inner dim must equal Q's head_dim");
42 CV_CheckEQ(kT[3], S_kv, "DNN/SDPA: K^T outer dim must equal V's seq len");
43 CV_CheckEQ(v[0], B, "DNN/SDPA: V batch");
44 CV_CheckEQ(v[1], H, "DNN/SDPA: V heads");
45
46 outputs.assign(1, MatShape{B, S_q, H * D_v});
47 // Scratch for K^T re-transposed to (B, H, S_kv, D).
48 internals.assign(1, MatShape{B * H * S_kv * D});
49 return false;
50 }

52 virtual int64 getFLOPS(const std::vector<MatShape> &inputs,
53 const std::vector<MatShape> &outputs) const CV_OVERRIDE {
54 const auto &q = inputs[0];
55 const auto &v = inputs[2];
56 const int64 B = q[0], H = q[1], S_q = q[2], D = q[3];
57 const int64 S_kv = v[2], D_v = v[3];
58 return B * H * (CV_BIG_INT(2) * S_q * S_kv * D + 4 * S_q * S_kv
59 + CV_BIG_INT(2) * S_q * S_kv * D_v);
60 }

62 void forward(InputArrayOfArrays inputs_arr,
63 OutputArrayOfArrays outputs_arr,
64 OutputArrayOfArrays internals_arr) CV_OVERRIDE
65 {
66 CV_TRACE_FUNCTION();
67 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
68
69 if (inputs_arr.depth() == CV_16F) {
70 forward_fallback(inputs_arr, outputs_arr, internals_arr);
71 return;
72 }
73
74 std::vector<Mat> inputs, outputs, internals;
75 inputs_arr.getMatVector(inputs);
76 outputs_arr.getMatVector(outputs);
77 internals_arr.getMatVector(internals);
78
79 const Mat& Q = inputs[0]; // (B, H, S_q, D)
80 const Mat& KT = inputs[1]; // (B, H, D, S_kv)
81 const Mat& V = inputs[2]; // (B, H, S_kv, D_v)
82 Mat& Y = outputs[0]; // (B, S_q, H*D_v)
83
84 const int B = Q.size[0];
85 const int H = Q.size[1];
86 const int S_q = Q.size[2];
87 const int D = Q.size[3];
88 const int S_kv = V.size[2];
89 const int D_v = V.size[3];
90
91 // Re-transpose K^T (B,H,D,S_kv) -> K (B,H,S_kv,D) for MlasFlashAttention.
92 Mat &K = internals[0];
93 const float* kT_data = KT.ptr<const float>();
94 float* k_data = K.ptr<float>();
95 const size_t kT_bhd_stride = (size_t)D * S_kv;
96 const size_t k_bhd_stride = (size_t)S_kv * D;
97 cv::parallel_for_(cv::Range(0, B * H), [&](const cv::Range& r) {
98 for (int bh = r.start; bh < r.end; bh++) {
99 const float* src = kT_data + (size_t)bh * kT_bhd_stride;
100 float* dst = k_data + (size_t)bh * k_bhd_stride;
101 for (int s = 0; s < S_kv; s++) {
102 for (int d = 0; d < D; d++) {
103 dst[(size_t)s * D + d] = src[(size_t)d * S_kv + s];
104 }
105 }
106 }
107 });
108
109 const int q_block = std::min(256, S_q);
110 const int kv_block = std::min(256, S_kv);
111 const int threads = std::max(1, cv::getNumThreads());
112
113 const size_t per_thread =
114 mlasFlashAttentionBufferBytesPerThread(q_block, kv_block, D_v);
115 if (mlasAvailable() && per_thread > 0) {
116 flash_scratch.resize((size_t)threads * per_thread);
117 const float effective_scale = (scale > 0.f) ? (1.f / scale) : 1.f;
118 if (mlasFlashAttention(Q.ptr<float>(), K.ptr<float>(), V.ptr<float>(),
119 Y.ptr<float>(),
120 B, H, S_q, S_kv, D, D_v, effective_scale,
121 q_block, kv_block,
122 flash_scratch.data(), threads))
123 {
124 return;
125 }
126 }
127
128 forward_fallback_impl(Q, K, V, Y, B, H, S_q, S_kv, D, D_v);
129 }

131private:
132 // Unfused SDPA, used when MLAS isn't available. Not perf-critical.
133 void forward_fallback_impl(const Mat& Q, const Mat& K, const Mat& V, Mat& Y,
134 int B, int H, int S_q, int S_kv, int D, int D_v) const
135 {
136 const float* q_p = Q.ptr<const float>();
137 const float* k_p = K.ptr<const float>();
138 const float* v_p = V.ptr<const float>();
139 float* y_p = Y.ptr<float>();
140 const float inv_scale = (scale > 0.f) ? (1.f / scale) : 1.f;
141
142 cv::parallel_for_(cv::Range(0, B * H), [&](const cv::Range& r) {
143 std::vector<float> attn((size_t)S_q * S_kv);
144 for (int bh = r.start; bh < r.end; bh++) {
145 const int b = bh / H, h = bh % H;
146 const float* Q_bh = q_p + (size_t)bh * S_q * D;
147 const float* K_bh = k_p + (size_t)bh * S_kv * D;
148 const float* V_bh = v_p + (size_t)bh * S_kv * D_v;
149 float* Y_b = y_p + (size_t)b * S_q * H * D_v;
150
151 for (int i = 0; i < S_q; i++) {
152 for (int j = 0; j < S_kv; j++) {
153 float s = 0.f;
154 for (int d = 0; d < D; d++)
155 s += Q_bh[(size_t)i * D + d] * K_bh[(size_t)j * D + d];
156 attn[(size_t)i * S_kv + j] = s * inv_scale;
157 }
158 float mx = attn[(size_t)i * S_kv];
159 for (int j = 1; j < S_kv; j++)
160 if (attn[(size_t)i * S_kv + j] > mx) mx = attn[(size_t)i * S_kv + j];
161 float sum = 0.f;
162 for (int j = 0; j < S_kv; j++) {
163 float e = std::exp(attn[(size_t)i * S_kv + j] - mx);
164 attn[(size_t)i * S_kv + j] = e;
165 sum += e;
166 }
167 float inv_sum = 1.f / sum;
168 for (int j = 0; j < S_kv; j++)
169 attn[(size_t)i * S_kv + j] *= inv_sum;
170
171 // out[b, i, h, :] = attn[i, :] @ V_bh
172 float* out_row = Y_b + ((size_t)i * H + h) * D_v;
173 for (int d = 0; d < D_v; d++) out_row[d] = 0.f;
174 for (int j = 0; j < S_kv; j++) {
175 float a = attn[(size_t)i * S_kv + j];
176 const float* V_row = V_bh + (size_t)j * D_v;
177 for (int d = 0; d < D_v; d++)
178 out_row[d] += a * V_row[d];
179 }
180 }
181 }
182 });
183 }
