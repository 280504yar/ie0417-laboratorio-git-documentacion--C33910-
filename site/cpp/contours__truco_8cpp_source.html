9// Atomic operations to avoid data race between findStartContourPoint and
10// traceExternalContourMock. The race is benign functionally because threads
11// only modify FOREGROUND (255) to VISITED_OUTER_RIGHT (100) or VISITED_ (200),
12// all of which are NON-ZERO. findStartContourPoint only cares if a pixel is
13// zero or non-zero, so its termination conditions are unchanged. However,
14// concurrent read/write of different values is legally UB in C++ and triggers
15// TSAN.
16inline uchar atomicLoad(const uchar* ptr) {
17#ifdef CV_THREAD_SANITIZER
18 if constexpr (sizeof(std::atomic<uchar>) == sizeof(uchar)) {
19 return reinterpret_cast<const std::atomic<uchar>*>(ptr)->load(std::memory_order_relaxed);
20 }
21#endif
22 return *ptr;
23}
24
25inline void atomicStore(uchar* ptr, uchar val) {
26#ifdef CV_THREAD_SANITIZER
27 if constexpr (sizeof(std::atomic<uchar>) == sizeof(uchar)) {
28 reinterpret_cast<std::atomic<uchar>*>(ptr)->store(val, std::memory_order_relaxed);
29 return;
30 }
31#endif
32 *ptr = val;
33}

35// Tunable block size. 1024 points = 8KB (Fits easily in L1 Cache)
36template <size_t BLOCK_SIZE = 2048>
37class TRUCOPagedContour {
38public:
39 struct Block {
40 cv::Point data[BLOCK_SIZE];
41 };
42
43 TRUCOPagedContour() {
44 allocateBlock();
45 // Initialize pointers to the start of the first block
46 curr_ptr_ = all_blocks_[0]->data;
47 end_ptr_ = curr_ptr_ + BLOCK_SIZE;
48 }
49
50 ~TRUCOPagedContour() {
51 for (Block* b : all_blocks_) cv::fastFree(b);
52 }
53
54 // --- HOT PATH: Minimal instructions ---
55 // No counter updates, just raw pointer arithmetic.
56 inline void push_back(const cv::Point& pt) {
57 if (curr_ptr_ == end_ptr_) {
58 current_block_idx_++;
59 if (current_block_idx_ == all_blocks_.size()) {
60 allocateBlock();
61 }
62 curr_ptr_ = all_blocks_[current_block_idx_]->data;
63 end_ptr_ = curr_ptr_ + BLOCK_SIZE;
64 }
65 *curr_ptr_++ = pt;
66 }
67
68 inline void pop_back() {
69 // Safety check: do nothing if completely empty
70 if (current_block_idx_ == 0 && curr_ptr_ == all_blocks_[0]->data) return;
71
72 // Check if we are at the start of the current block
73 if (curr_ptr_ == all_blocks_[current_block_idx_]->data) {
74 // Move to the previous block
75 current_block_idx_--;
76 // Point to the end of the previous block
77 curr_ptr_ = all_blocks_[current_block_idx_]->data + BLOCK_SIZE;
78 end_ptr_ = curr_ptr_;
79 }
80 curr_ptr_--;
81 }
82
83 inline const cv::Point& back() const {
84 // Handle case where back() crosses block boundary
85 if (curr_ptr_ == all_blocks_[current_block_idx_]->data) {
86 return all_blocks_[current_block_idx_ - 1]->data[BLOCK_SIZE - 1];
87 }
88 return *(curr_ptr_ - 1);
89 }
90
91 inline const cv::Point& front() const {
92 return all_blocks_[0]->data[0];
93 }
94
95 // Calculated on demand (O(1) arithmetic, but slightly more math than reading a variable)
96 size_t size() const {
97 size_t elements_in_last = curr_ptr_ - all_blocks_[current_block_idx_]->data;
98 return (current_block_idx_ * BLOCK_SIZE) + elements_in_last;
99 }
100
101 void clear() {
102 current_block_idx_ = 0;
103 if (!all_blocks_.empty()) {
104 curr_ptr_ = all_blocks_[0]->data;
105 end_ptr_ = curr_ptr_ + BLOCK_SIZE;
106 }
107 }
108
109 // Optimized Copy: Uses block-wise memcpy
110 void copyTo(std::vector<cv::Point>& out) const {
111 size_t total = size();
112 out.resize(total);
113 if (total == 0) return;
114
115 cv::Point* dst = out.data();
116
117 // 1. Copy full blocks
118 for (size_t i = 0; i < current_block_idx_; ++i) {
119 std::memcpy(dst, all_blocks_[i]->data, BLOCK_SIZE * sizeof(cv::Point));
120 dst += BLOCK_SIZE;
121 }
122
123 // 2. Copy partial last block
124 size_t last_block_count = curr_ptr_ - all_blocks_[current_block_idx_]->data;
125 if (last_block_count > 0) {
126 std::memcpy(dst, all_blocks_[current_block_idx_]->data, last_block_count * sizeof(cv::Point));
127 }
128 }
129
130private:
131 void grow() {
132 current_block_idx_++;
133 if (current_block_idx_ == all_blocks_.size()) {
134 allocateBlock();
135 }
136 curr_ptr_ = all_blocks_[current_block_idx_]->data;
137 end_ptr_ = curr_ptr_ + BLOCK_SIZE;
138 }
139
140 void allocateBlock() {
141 Block* b = (Block*)cv::fastMalloc(sizeof(Block));
142 all_blocks_.push_back(b);
143 }
144
145 std::vector<Block*> all_blocks_;
146 size_t current_block_idx_ = 0;
147
148 // Fast pointers for the hot loop
149 cv::Point* curr_ptr_ = nullptr;
150 cv::Point* end_ptr_ = nullptr;
151};

166 // Accumulator: Vector of (Vector of Contours), where Contour is Vector of Points

169 TRUCOntourTracer(const cv::Mat& img,
170 const std::vector<cv::Range>& stripRanges,
171 AccumulatorType& accumulator,
172 size_t minSize)
173 : padded_(img), ranges_(stripRanges), accumulator_(accumulator), minSize_(minSize)
174 {
175 step_ = padded_.step;
176 int istep = (int)step_;
177 // 0: East (Right)
178 offsets_[0] = 1;
179 // 1: NE (Up-Right)
180 offsets_[1] = -istep + 1;
181 // 2: North (Up)
182 offsets_[2] = -istep;
183 // 3: NW (Up-Left)
184 offsets_[3] = -istep - 1;
185 // 4: West (Left)
186 offsets_[4] = -1;
187 // 5: SW (Down-Left)
188 offsets_[5] = istep - 1;
189 // 6: South (Down)
190 offsets_[6] = istep;
191 // 7: SE (Down-Right)
192 offsets_[7] = istep + 1;
193
194 memcpy(offsets_ + 8, offsets_, 8 * sizeof(int));
195

202 int search_idx = 5;
203 uchar* curr_ptr = row_ptr + c , * start_ptr = curr_ptr;
204 int dir=-1;
205 bool is_first_move = true;

207 while(true)
208 {
209 // Check neighbors
210 for (int n = 0; n < 8; ++n)

212 int idx = search_idx + n;

215 if (atomicLoad(neighbor) == BACKGROUND) continue;

247 bool traceContour( TRUCOPagedContour<4096>* buffer, int r,int c,uchar *row_ptr, const cv::Range& rowRange,bool isExternal)const{
248
249 buffer->clear();
250
251 int curr_x = c , curr_y = r;
252 int start_dir = -1 ;
253 int search_idx = isExternal ? 5 :1;
254 uchar* curr_ptr = row_ptr + c , * start_ptr = curr_ptr;
255 int dir=-1;
256 // int sign=isExternal?1:-1;
257
258 bool is_first_move = true;
259
260
261 // QUick test to find the first element of an internal contour. We know it should be NE
262 if(!isExternal){
263 int n=0;
264 for ( n = 0; n < 8; ++n)
265 {
266 int idx = search_idx + n;
267 if (atomicLoad(curr_ptr + offsets_[idx]) == BACKGROUND) continue;
268 if(curr_x+ dx_[ idx & 7]!=c+1 || curr_y+ dy_[ idx & 7]!=r-1) return false;
269 break;
270 }
271 if(n==8) return false;//isolated pixels must not be considered as internal
272 }
273 // 3. TRACING LOOP
274 while(true)
275 {
276 buffer->push_back({ (curr_x - 1), (curr_y - 1)});
277 // Check neighbors
278 for (int n = 0; n < 8; ++n)
279 {
280 int idx = search_idx + n;
281 // Use offset cache
282 uchar* neighbor = curr_ptr + offsets_[idx];
283 if (atomicLoad(neighbor) == BACKGROUND) continue;
284
285 dir = idx & 7;
286 // --- EXECUTE MOVE ---
287 curr_y += dy_[dir];
288 curr_x += dx_[dir];
289
290
291 // Check bounds //we need to move out of the range //if first line, and internal contour, we let it go, but no further from this line
292 if( curr_y < rowRange.start ){
293 return false;
294 }
295 if ((search_idx <= 1) || (dir <= search_idx - 2))
296 {
297 atomicStore(curr_ptr, VISITED_OUTER_RIGHT);
298 }
299 else if (atomicLoad(curr_ptr) == FOREGROUND)
300 {
301 atomicStore(curr_ptr, VISITED_);
302 }
303
304 // Short-circuit Jacob's Check
305 if (curr_ptr == start_ptr) {
306 if (!is_first_move && dir == start_dir) {
307 return true;//done
308 }
309 }
310
311 curr_ptr = neighbor;//move ptr
312
313 // Reset search index for Moore neighbor
314 search_idx = (dir +6) & 7;
315 break;
316
317 }
318 if (is_first_move) {
319 if(dir==-1){//single pixel
320 atomicStore(curr_ptr, VISITED_OUTER_RIGHT);
321 break;//not moved
322 }
323 start_dir = dir;
324 is_first_move = false;
325 }
326
327 }
328 return true;

331 void operator()(const cv::Range& range) const CV_OVERRIDE
332 {
333 // Pre-allocate buffer to avoid re-allocation during moves
334 TRUCOPagedContour<4096> buffer;
335
336 int cols = padded_.cols;
337
338 for (int i = range.start; i < range.end; ++i)
339 {
340 const cv::Range& rowRange = ranges_[i];
341 auto& local_contours = accumulator_[i];
342
343 // Hint for result vector size
344 local_contours.reserve(2048);
345
346 for (int r = rowRange.start; r <= rowRange.end; ++r)
347 {
348 uchar* row_ptr = padded_.data + r * step_;
349
350 // "c" is updated by the find* functions
351 for (int c = 1; c < cols - 1; )
352 {
353 // 1. FAST SCAN: Skip background pixels
354 if ((c = findStartContourPoint(row_ptr, cols, c)) == cols) break;
355
356 // 2. CHECK: Only process if actually FOREGROUND (redundancy check)
357 if (atomicLoad(&row_ptr[c]) == FOREGROUND && r<rowRange.end)
358 {
359 if( traceContour(&buffer,r,c,row_ptr,rowRange,true)){
360 // Post-processing
361 if (buffer.size() > 1 && buffer.back() == buffer.front()) {
362 buffer.pop_back();
363 }
364 if (buffer.size() >= minSize_) {
365 // Instead of copying the vector, we move it.
366 local_contours.emplace_back();
367 buffer.copyTo(local_contours.back());
368 if( r==rowRange.start){//register this so we can zip them as Suzuki&Abe method
369 local_contours.idx_external_firstLine.push_back((int)(local_contours.size()-1) );
370 }
371 }
372 }
373 }
374 else if (atomicLoad(&row_ptr[c]) == FOREGROUND && r==rowRange.end){//extra step to mark east pixels only so that internal contours can be correctly extracted in this line
375 traceExternalContourMock(r,c,row_ptr,rowRange);
376 }
377
378 // 3. FAST SCAN: Find end of current component to skip processing it again
379 c = findEndContourPoint(row_ptr, cols, c + 1);
380 if(c>=cols)break;//end of row
381 //internal contour
382 if (atomicLoad(&row_ptr[c - 1]) > VISITED_OUTER_RIGHT && r>rowRange.start){//inner contours of first line are handled by the thread above
383
384 if(traceContour(&buffer,r,c-1,row_ptr,rowRange,false)){
385 // Post-processing
386 if (buffer.size() > 1 && buffer.back() == buffer.front()) {
387 buffer.pop_back();
388 }
389 if (buffer.size() >= minSize_) {
390 local_contours.emplace_back();
391 buffer.copyTo(local_contours.back());
392 if( r==rowRange.end){//register this so we can zip them as Suzuki&Abe method
393 local_contours.idx_internal_lastLine.push_back((int)(local_contours.size() -1));
394 }
395 }
396 }
397 }
398 }

403 static inline int findStartContourPoint(uchar* src_data, int width, int j)
404 {
405#if (CV_SIMD || CV_SIMD_SCALABLE) && !defined(CV_THREAD_SANITIZER)
406 cv::v_uint8 v_zero = cv::vx_setzero_u8();
407 for (; j <= width - cv::VTraits<cv::v_uint8>::vlanes(); j += cv::VTraits<cv::v_uint8>::vlanes())
408 {
409 cv::v_uint8 vmask = (cv::v_ne(cv::vx_load((uchar*)(src_data + j)), v_zero));
410 if (cv::v_check_any(vmask))
411 {
412 j += cv::v_scan_forward(vmask);
413 return j;
414 }
415 }
416#endif
417 for (; j < width; ++j) {
418 if (atomicLoad(&src_data[j])) break;
419 }
420
421 return j;
422 }

424 inline static int findEndContourPoint(uchar* src_data,int width, int j)
425 {
426#if (CV_SIMD || CV_SIMD_SCALABLE) && !defined(CV_THREAD_SANITIZER)
427 if (j < width && !src_data[j])
428 {
429 return j;
430 }
431 else
432 {
433 cv::v_uint8 v_zero = cv::vx_setzero_u8();
434 for (; j <= width - cv::VTraits<cv::v_uint8>::vlanes(); j += cv::VTraits<cv::v_uint8>::vlanes())
435 {
436 cv::v_uint8 vmask = (cv::v_eq(cv::vx_load((uchar*)(src_data + j)), v_zero));
437 if (cv::v_check_any(vmask))
438 {
439 j += cv::v_scan_forward(vmask);
440 return j;
441 }
442 }
443 }
444#endif
445 for (; j < width; ++j) {
446 if (atomicLoad(&src_data[j]) == 0) break;
447 }

449 return j;

472void approxContour(std::vector<cv::Point> &inout,cv::ContourApproximationModes contApprox_) {
473 size_t n = inout.size();
474 if (n <= 1) return;
475
476 if (contApprox_ == cv::CHAIN_APPROX_SIMPLE) {
477 std::vector<cv::Point> result;
478 result.reserve(n);
479
480 for (size_t i = 0; i < n; ++i) {
481 size_t prev_i = (i == 0) ? n - 1 : i - 1;
482 size_t next_i = (i == n - 1) ? 0 : i + 1;
483
484 cv::Point v1 = inout[i] - inout[prev_i];
485 cv::Point v2 = inout[next_i] - inout[i];
486
487 if (v1 != v2) {
488 result.push_back(inout[i]);
489 }
490 }
491 if (result.empty() && n > 0) result.push_back(inout[0]);
492 inout = std::move(result);
493 } else if (contApprox_ == cv::CHAIN_APPROX_TC89_L1 || contApprox_ == cv::CHAIN_APPROX_TC89_KCOS) {
494 auto getCode = [](cv::Point d) -> schar {
495 if (d.x == 1) {
496 if (d.y == 0) return 0;
497 if (d.y == -1) return 1;
498 if (d.y == 1) return 7;
499 } else if (d.x == 0) {
500 if (d.y == -1) return 2;
501 if (d.y == 1) return 6;
502 } else if (d.x == -1) {
503 if (d.y == -1) return 3;
504 if (d.y == 0) return 4;
505 if (d.y == 1) return 5;
506 }
507 return 0;
508 };
509
510 cv::ContourCodesStorage::storage_t codesStorage;
511 cv::ContourCodesStorage codes(&codesStorage);
512 for (size_t i = 0; i < n; ++i) {
513 cv::Point delta = inout[(i + 1) % n] - inout[i];
514 codes.push_back(getCode(delta));
515 }
516
517 cv::ContourPointsStorage::storage_t pointsStorage;
518 cv::ContourPointsStorage points(&pointsStorage);
519 cv::approximateChainTC89(codes, inout[0], contApprox_, points);
520
521 inout.clear();
522 inout.reserve(points.size());
523 for (size_t i = 0; i < points.size(); ++i) {
524 inout.push_back(points.at(i));
525 }
526 }

532void findTRUContoursImpl(cv::Mat& padded,
533 std::vector<std::vector<cv::Point>>& outContours,
534 int minSize,int contApprox)
535{
536 // Load Balancing Logic
537 const int nstripes = cv::getNumThreads();
538 std::vector<cv::Range> balancedRanges;
539 if (nstripes > 1) {
540 int rowsPerStripe = (padded.rows - 2) / nstripes;
541 int remainingRows = (padded.rows - 2) % nstripes;
542 int currentRow = 1;
543 for (int t = 0; t < nstripes; ++t) {
544 int startRow = currentRow;
545 int endRow = startRow + rowsPerStripe + (t < remainingRows ? 1 : 0);
546 balancedRanges.emplace_back(startRow, endRow);
547 currentRow = endRow;
548 }
549 }
550 else {
551 balancedRanges.emplace_back(1, padded.rows - 1);
552 }
553 // Parallel Execution
554 std::vector<AccumulatorT> threadAccumulators(balancedRanges.size());
555 TRUCOntourTracer worker(padded, balancedRanges, threadAccumulators, minSize);
556 cv::parallel_for_(cv::Range(0, (int)balancedRanges.size()), worker);
557
558
559 // REORG To Match Suzuki & Abe's Contour Ordering.
560 // Every adjacent strip pair (t, t+1) shares a boundary row. On that row:
561 // * thread t recorded INTERNAL fragments -> idx_internal_lastLine (tail of accT)
562 // * thread t+1 recorded EXTERNAL fragments -> idx_external_firstLine (head of accT1)
563 // Both runs are already in X-ascending order (left-to-right raster scan),
564 // so a two-pointer merge restores the sequential Suzuki & Abe ordering.
565 for (size_t t = 0; t + 1 < threadAccumulators.size(); ++t) {
566 auto& accT = threadAccumulators[t];
567 auto& accT1 = threadAccumulators[t + 1];
568
569 const size_t kI = accT.idx_internal_lastLine.size();
570 const size_t kE = accT1.idx_external_firstLine.size();
571 if (kI == 0 && kE == 0) continue;
572
573 const size_t tailStart = accT.size() - kI;
574
575 // Merge by ascending X of each contour's first point. Contours are moved,
576 // not copied — only std::vector handles (three pointers) change hands.
577 std::vector<std::vector<cv::Point>> merged;
578 merged.reserve(kI + kE);
579 size_t i = tailStart, iEnd = accT.size();
580 size_t j = 0, jEnd = kE;
581 while (i < iEnd && j < jEnd) {
582 if (accT[i].front().x <= accT1[j].front().x)
583 merged.emplace_back(std::move(accT[i++]));
584 else
585 merged.emplace_back(std::move(accT1[j++]));
586 }
587 while (i < iEnd) merged.emplace_back(std::move(accT[i++]));
588 while (j < jEnd) merged.emplace_back(std::move(accT1[j++]));
589
590 // Replace accT's tail with the merged run.
591 accT.resize(tailStart);
592 for (auto& c : merged) accT.emplace_back(std::move(c));
593
594 // Drop the consumed externals from the head of accT1.
595 accT1.erase(accT1.begin(), accT1.begin() + kE);
596
597 // Any remaining bookkeeping in accT1 indexed absolute positions — fix them.
598 for (auto& idx : accT1.idx_internal_lastLine)
599 idx -= static_cast<int>(kE);
600
601 // This boundary's bookkeeping is now consumed.
602 accT.idx_internal_lastLine.clear();
603 accT1.idx_external_firstLine.clear();
604 }
605
606 // ZERO-COPY MERGE
607 size_t totalContours = 0;
608 for (auto& tVec : threadAccumulators)
609 totalContours += tVec.size();
610
611
612 outContours.clear();
613 outContours.reserve(totalContours);
614 // move the contours from thread accumulators to output without copying pixel data
615 for (auto& tVec : threadAccumulators) {
616 // move_iterator moves the vector internals (pointers) without copying pixel data
617 outContours.insert(outContours.end(),
618 std::make_move_iterator(tVec.begin()),
619 std::make_move_iterator(tVec.end()));
620 }
621 // reverse the order to match original findContours Suzuki&Abe
622 std::reverse(outContours.begin(), outContours.end());
623
624
625 //7. now, lets do the contour approximation if needed in parallel
626 if(contApprox!=cv::CHAIN_APPROX_NONE){
627 cv::parallel_for_(cv::Range(0, (int)outContours.size()), [&](const cv::Range& range){
628 for(int i=range.start;i<range.end;i++){
629 approxContour(outContours[i],(cv::ContourApproximationModes)contApprox);
630 }
631 });
632 }
633
634}

640//
641// Public API: Handles OutputArray and dispatches to the core implementation
642// This is a modified version of the original TRUCO parallel algorithm to produce the exact same output
643// as original findContours with RETR_LIST mode (no hierarchy, all contours are external). It also supports contour approximation.
644//
645// ==========================================================
646void findTRUContours(InputArray _src, OutputArrayOfArrays _contours, int minSize, bool binarize,int method)
647{
648 CV_INSTRUMENT_REGION();
649 Mat src = _src.getMat();
650 CV_Assert(src.type() == CV_8UC1);
651
652 // Buffer handling
653 cv::Mat padded;
654 cv::copyMakeBorder(src, padded, 1, 1, 1, 1, cv::BORDER_CONSTANT, 0);
655 if (binarize)
656 cv::threshold(padded, padded, 0, 255, cv::THRESH_BINARY);
657
658
659 // Fast path: caller passed std::vector<std::vector<cv::Point>> directly.
