23template<typename _Tp, int m> struct Matx_DetOp
24{
25 double operator ()(const Matx<_Tp, m, m>& a) const
26 {
27 Matx<_Tp, m, m> temp = a;
28 double p = LU(temp.val, m*sizeof(_Tp), m, 0, 0, 0);
29 if( p == 0 )
30 return p;
31 for( int i = 0; i < m; i++ )
32 p *= temp(i, i);
33 return p;
34 }
35};

54{
