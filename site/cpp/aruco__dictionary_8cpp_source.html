18struct CellBitMasks {
19 CellBitMasks(const Mat &onlyCellPixelRatio, int markerSize, float validBitIdThreshold)
20 : s((markerSize * markerSize + 8 - 1) / 8),
21 totalCells(markerSize * markerSize),
22 temp(4 * s),
23 not0(temp.data()), not1(not0 + s), notXor(not1 + s), temp0(temp.data() + 3 * s) {
24 uint8_t* not0Writable = temp.data();
25 uint8_t* not1Writable = not0Writable + s;
26 uint8_t* notXorWritable = not1Writable + s;
27
28 // Fill bit masks of cells that are not black (not0) and not white (not1).
29 unsigned char not0Byte = 0, not1Byte = 0;
30 int currentByte = 0, currentBit = 0;
31 for(int j = 0; j < markerSize; j++) {
32 const float* cellPixelRatioRow = onlyCellPixelRatio.ptr<float>(j);
33 for(int i = 0; i < markerSize; i++) {
34 not0Byte <<= 1; not1Byte <<= 1;
35 if(cellPixelRatioRow[i] > validBitIdThreshold) not0Byte |= 1;
36 if(cellPixelRatioRow[i] < 1 - validBitIdThreshold) not1Byte |= 1;
37 ++currentBit;
38 if(currentBit == 8) {
39 not0Writable[currentByte] = not0Byte;
40 not1Writable[currentByte] = not1Byte;
41 not0Byte = not1Byte = 0;
42 ++currentByte;
43 currentBit = 0;
44 }
45 }
46 }
47 if(currentBit != 0) {
48 not0Writable[currentByte] = not0Byte;
49 not1Writable[currentByte] = not1Byte;
50 }
51
52 // Computing: notXor = not0 ^ not1
53 hal::xor8u(not0, s, not1, s, notXorWritable, s, s, 1, nullptr);
54 }

56 CellBitMasks(const CellBitMasks&) = delete;
57 CellBitMasks& operator=(const CellBitMasks&) = delete;

59 // Smallest Hamming distance between these cell masks and dictionary marker `id`,
60 // searching the tested rotations; `rotation` returns the best one.
61 // Mutates the internal buffer (temp0).
62 int hammingDistanceToId(const Mat& bytesList, int id, bool allRotations, int& rotation) {
63 CV_Assert(id >= 0 && id < bytesList.rows);
64
65 const unsigned int nRotations = allRotations ? 4u : 1u;
66 int currentMinDistance = totalCells + 1;
67 rotation = -1;
68
69 const uchar* bytesRot = bytesList.ptr(id);
70 for(unsigned int r = 0; r < nRotations; r++, bytesRot += s) {
71 // Error if (marker is 0 and input is not 0) or (marker is 1 and input is not 1)
72 // i.e.: (!bytesRot && not0) || (bytesRot && not1)
73 // This is equivalent to: not0 ^ ((not0 ^ not1) & bytesRot)
74 // Computing: temp0 = (not0 ^ not1) & bytesRot
75 hal::and8u(notXor, s, bytesRot, s, temp0, s, s, 1, nullptr);
76 // Computing the final result (xor is performed internally).
77 int currentHamming = cv::hal::normHamming(not0, temp0, s);
78
79 if(currentHamming < currentMinDistance) {
80 currentMinDistance = currentHamming;
81 rotation = static_cast<int>(r);
82 // Break for perfect distance.
83 if(currentMinDistance == 0) break;
84 }
85 }
86
87 return currentMinDistance;
88 }
89
90 const int s; // bytes per rotation
91 const int totalCells;
92 std::vector<uint8_t> temp;
93 const uint8_t *not0, *not1, *notXor;
94 uint8_t *temp0; // internal scratch workspace
95};

108bool Dictionary::readDictionary(const cv::FileNode& fn) {
109 int nMarkers = 0, _markerSize = 0;
110 if (fn.empty() || !readParameter("nmarkers", nMarkers, fn) || !readParameter("markersize", _markerSize, fn))
111 return false;
112 Mat bytes(0, 0, CV_8UC1), marker(_markerSize, _markerSize, CV_8UC1);
113 std::string markerString;
114 for (int i = 0; i < nMarkers; i++) {
115 std::ostringstream ostr;
116 ostr << i;
117 if (!readParameter("marker_" + ostr.str(), markerString, fn))
118 return false;
119 for (int j = 0; j < (int) markerString.size(); j++)
120 marker.at<unsigned char>(j) = (markerString[j] == '0') ? 0 : 1;
121 bytes.push_back(Dictionary::getByteListFromBits(marker));
122 }
123 int _maxCorrectionBits = 0;
124 readParameter("maxCorrectionBits", _maxCorrectionBits, fn);
125 *this = Dictionary(bytes, _markerSize, _maxCorrectionBits);
126 return true;
127}

157bool Dictionary::identify(const Mat &onlyCellPixelRatio, CV_OUT int &idx, CV_OUT int &rotation, double maxCorrectionRate, float validBitIdThreshold) const {
158 CV_Assert(onlyCellPixelRatio.rows == markerSize && onlyCellPixelRatio.cols == markerSize);
159 CV_Assert(onlyCellPixelRatio.type() == CV_32FC1);
160
161 CellBitMasks cellBitMasks(onlyCellPixelRatio, markerSize, validBitIdThreshold);
162
163 int maxCorrectionRecalculed = int(double(maxCorrectionBits) * maxCorrectionRate);
164
165 idx = -1; // by default, not found
166
167 // search closest marker in dict
168 for(int m = 0; m < bytesList.rows; m++) {
169 int currentRotation = -1;
170 int currentMinDistance = cellBitMasks.hammingDistanceToId(bytesList, m, true, currentRotation);
171
172 // if maxCorrection is fulfilled, return this one
173 if(currentMinDistance <= maxCorrectionRecalculed) {
174 idx = m;
175 rotation = currentRotation;
176 break;
177 }
178 }
179
180 return idx != -1;
181}

193int Dictionary::getDistanceToId(InputArray bits, int id, bool allRotations) const {
194
195 CV_Assert(id >= 0 && id < bytesList.rows);
196
197 unsigned int nRotations = 4;
198 if(!allRotations) nRotations = 1;
199
200 Mat candidateBytes = getByteListFromBits(bits.getMat());
201 int currentMinDistance = int(bits.total() * bits.total());
202 for(unsigned int r = 0; r < nRotations; r++) {
203 int currentHamming = cv::hal::normHamming(
204 bytesList.ptr(id) + r*candidateBytes.cols,
205 candidateBytes.ptr(),
206 candidateBytes.cols);
207
208 if(currentHamming < currentMinDistance) {
209 currentMinDistance = currentHamming;
210 }
211 }
212 return currentMinDistance;
213}

229void Dictionary::generateImageMarker(int id, int sidePixels, OutputArray _img, int borderBits) const {
230 CV_Assert(sidePixels >= (markerSize + 2*borderBits));
231 CV_Assert(id < bytesList.rows);
232 CV_Assert(borderBits > 0);
233
234 _img.create(sidePixels, sidePixels, CV_8UC1);
235
236 // create small marker with 1 pixel per bin
237 Mat tinyMarker(markerSize + 2 * borderBits, markerSize + 2 * borderBits, CV_8UC1,
238 Scalar::all(0));
239 Mat innerRegion = tinyMarker.rowRange(borderBits, tinyMarker.rows - borderBits)
240 .colRange(borderBits, tinyMarker.cols - borderBits);
241 // put inner bits
242 Mat bits = getMarkerBits(id);
243 bits.convertTo(bits, CV_8U, 255.0);
244 CV_Assert(innerRegion.total() == bits.total());
245 bits.copyTo(innerRegion);
246
247 // resize tiny marker to output size
248 cv::resize(tinyMarker, _img.getMat(), _img.getMat().size(), 0, 0, INTER_NEAREST);
249}

252Mat Dictionary::getByteListFromBits(const Mat &bits) {
253 // integer ceil
254 int nbytes = (bits.cols * bits.rows + 8 - 1) / 8;
255
256 Mat candidateByteList(1, nbytes, CV_8UC4, Scalar::all(0));
257 unsigned char currentBit = 0;
258 int currentByte = 0;
259
260 // the 4 rotations
261 uchar* rot0 = candidateByteList.ptr();
262 uchar* rot1 = candidateByteList.ptr() + 1*nbytes;
263 uchar* rot2 = candidateByteList.ptr() + 2*nbytes;
264 uchar* rot3 = candidateByteList.ptr() + 3*nbytes;
265
266 for(int row = 0; row < bits.rows; row++) {
267 for(int col = 0; col < bits.cols; col++) {
268 // circular shift
269 rot0[currentByte] <<= 1;
270 rot1[currentByte] <<= 1;
271 rot2[currentByte] <<= 1;
272 rot3[currentByte] <<= 1;
273 // set bit
274 rot0[currentByte] |= bits.at<uchar>(row, col);
275 rot1[currentByte] |= bits.at<uchar>(col, bits.cols - 1 - row);
276 rot2[currentByte] |= bits.at<uchar>(bits.rows - 1 - row, bits.cols - 1 - col);
277 rot3[currentByte] |= bits.at<uchar>(bits.rows - 1 - col, row);
278 currentBit++;
279 if(currentBit == 8) {
280 // next byte
281 currentBit = 0;
282 currentByte++;
283 }
284 }
285 }
286 return candidateByteList;
287}

291
292 const int nbRotations = 4;
293 CV_Assert(markerId < bytesList.rows);
294 CV_Assert(rotationId < nbRotations);
295

297 Mat bitsUints = getBitsFromByteList(bytesList.rowRange(markerId, markerId + 1), markerSize, rotationId);

301 return bits;

305Mat Dictionary::getBitsFromByteList(const Mat &byteList, int markerSize, int rotationId) {
306 CV_Assert(byteList.total() > 0 &&
307 byteList.total() >= (unsigned int)markerSize * markerSize / 8 &&
308 byteList.total() <= (unsigned int)markerSize * markerSize / 8 + 1);
309
310 CV_Assert(byteList.channels() >= 4);
311 CV_Assert(rotationId >= 0 && rotationId < 4);
312
313 Mat bits = Mat::zeros(markerSize, markerSize, CV_8UC1);
314 unsigned char *bitsPtr = bits.ptr();
315
316 // Use a base offset for the selected rotation
317 int nbytes = (bits.cols * bits.rows + 8 - 1) / 8; // integer ceil
318 int base = rotationId * nbytes;
319 const unsigned char *currentBytePtr = byteList.ptr() + base;
320 const unsigned char *currentBytePtrEnd = currentBytePtr + bits.total() / 8;
321
322 for(;currentBytePtr < currentBytePtrEnd; ++currentBytePtr) {
323 unsigned char currentByte = *currentBytePtr;
324 for(int mask = 1 << 7; mask != 0; mask >>= 1) {
325 if (currentByte & mask) *bitsPtr = 1;
326 ++bitsPtr;
327 }
328 }
329 // if not enough bits for one more byte, we are in the end
330 // update bit position accordingly
331 if (bits.total() % 8 != 0) {
332 unsigned char currentByte = *currentBytePtrEnd;
333 int mask = 1 << ((bits.total() % 8) - 1);
334 for(; mask != 0; mask >>= 1) {
335 if (currentByte & mask) *bitsPtr = 1;
336 ++bitsPtr;
337 }
338 }
339 return bits;
340}

446static Mat _generateRandomMarker(int markerSize, RNG &rng) {
447 Mat marker(markerSize, markerSize, CV_8UC1, Scalar::all(0));
448 for(int i = 0; i < markerSize; i++) {
449 for(int j = 0; j < markerSize; j++) {

451 marker.at<unsigned char>(i, j) = bit;

465static int _getSelfDistance(const Mat &marker) {
466 Mat bytes = Dictionary::getByteListFromBits(marker);
467 int minHamming = (int)marker.total() + 1;
468 for(int r = 1; r < 4; r++) {
469 int currentHamming = cv::hal::normHamming(bytes.ptr(), bytes.ptr() + bytes.cols*r, bytes.cols);
470 if(currentHamming < minHamming) minHamming = currentHamming;
471 }
472 return minHamming;
473}

476Dictionary extendDictionary(int nMarkers, int markerSize, const Dictionary &baseDictionary, int randomSeed) {
477 CV_Assert(nMarkers > 0);
478 RNG rng((uint64)(randomSeed));
479
480 Dictionary out = Dictionary(Mat(), markerSize);
481 out.markerSize = markerSize;
482
483 // theoretical maximum intermarker distance
484 // See S. Garrido-Jurado, R. Muñoz-Salinas, F. J. Madrid-Cuevas, and M. J. Marín-Jiménez. 2014.
485 // "Automatic generation and detection of highly reliable fiducial markers under occlusion".
486 // Pattern Recogn. 47, 6 (June 2014), 2280-2292. DOI=10.1016/j.patcog.2014.01.005
487 int C = (int)std::floor(float(markerSize * markerSize) / 4.f);
488 int tau = 2 * (int)std::floor(float(C) * 4.f / 3.f);
489
490 // if baseDictionary is provided, calculate its intermarker distance
491 if(baseDictionary.bytesList.rows > 0) {
492 CV_Assert(baseDictionary.markerSize == markerSize);
493 out.bytesList = baseDictionary.bytesList.rowRange(0, min(nMarkers, baseDictionary.bytesList.rows)).clone();
494
495 int minDistance = markerSize * markerSize + 1;
496 for(int i = 0; i < out.bytesList.rows; i++) {
497 Mat markerBytes = out.bytesList.rowRange(i, i + 1);
498 Mat markerBits = Dictionary::getBitsFromByteList(markerBytes, markerSize);
499 minDistance = min(minDistance, _getSelfDistance(markerBits));
500 for(int j = i + 1; j < out.bytesList.rows; j++) {
501 minDistance = min(minDistance, out.getDistanceToId(markerBits, j));
502 }
503 }
504 tau = minDistance;
505 }
506
507 // current best option
508 int bestTau = 0;
509 Mat bestMarker;
510
511 // after these number of unproductive iterations, the best option is accepted
512 const int maxUnproductiveIterations = 5000;
513 int unproductiveIterations = 0;
514
515 while(out.bytesList.rows < nMarkers) {
516 Mat currentMarker = _generateRandomMarker(markerSize, rng);
517
518 int selfDistance = _getSelfDistance(currentMarker);
519 int minDistance = selfDistance;
520
521 // if self distance is better or equal than current best option, calculate distance
522 // to previous accepted markers
523 if(selfDistance >= bestTau) {
524 for(int i = 0; i < out.bytesList.rows; i++) {
525 int currentDistance = out.getDistanceToId(currentMarker, i);
526 minDistance = min(currentDistance, minDistance);
527 if(minDistance <= bestTau) {
528 break;
529 }
530 }
531 }
532
533 // if distance is high enough, accept the marker
534 if(minDistance >= tau) {
535 unproductiveIterations = 0;
536 bestTau = 0;
537 Mat bytes = Dictionary::getByteListFromBits(currentMarker);
538 out.bytesList.push_back(bytes);
539 } else {
540 unproductiveIterations++;
541
542 // if distance is not enough, but is better than the current best option
543 if(minDistance > bestTau) {
544 bestTau = minDistance;
545 bestMarker = currentMarker;
546 }
547
548 // if number of unproductive iterarions has been reached, accept the current best option
549 if(unproductiveIterations == maxUnproductiveIterations) {
550 unproductiveIterations = 0;
551 tau = bestTau;
552 bestTau = 0;
553 Mat bytes = Dictionary::getByteListFromBits(bestMarker);
554 out.bytesList.push_back(bytes);
555 }
556 }
557 }
558
559 // update the maximum number of correction bits for the generated dictionary
560 out.maxCorrectionBits = (tau - 1) / 2;
561
