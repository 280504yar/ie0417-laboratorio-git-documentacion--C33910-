70Mat BaseImageDecoder::getMetadata(ImageMetadataType type) const
71{
72 auto makeMat = [](const std::vector<unsigned char>& data) -> Mat {
73 return data.empty() ? Mat() : Mat(1, (int)data.size(), CV_8U, (void*)data.data());
74 };
75
76 switch (type) {
77 case IMAGE_METADATA_EXIF:
78 return makeMat(m_exif.getData());
79
80 case IMAGE_METADATA_XMP:
81 case IMAGE_METADATA_ICCP:
82 case IMAGE_METADATA_CICP:
83 return makeMat(m_metadata[type]);
84
85 default:
86 CV_LOG_WARNING(NULL, "Unknown metadata type requested: " << static_cast<int>(type));
87 break;
88 }
89 return Mat();
90}

189bool BaseImageEncoder::addMetadata(ImageMetadataType type, const Mat& metadata)
190{
191 CV_Assert_N(type >= IMAGE_METADATA_EXIF, type <= IMAGE_METADATA_MAX);
192 if (metadata.empty())
193 return true;
194 size_t itype = (size_t)type;
195 if (itype >= m_support_metadata.size() || !m_support_metadata[itype])
196 return false;
197 if (m_metadata.empty())
198 m_metadata.resize((size_t)IMAGE_METADATA_MAX+1);
199 CV_Assert(metadata.elemSize() == 1);
200 CV_Assert(metadata.isContinuous());
201 const unsigned char* data = metadata.ptr<unsigned char>();
202 m_metadata[itype].assign(data, data + metadata.total());
203 return true;
204}

211bool BaseImageEncoder::writemulti(const std::vector<Mat>& img_vec, const std::vector<int>& params)
212{
213 if(img_vec.size() > 1)
214 CV_LOG_INFO(NULL, "Multi page image will be written as animation with 1 second frame duration.");
215
216 Animation animation;
217 animation.frames = img_vec;
218
219 for (size_t i = 0; i < animation.frames.size(); i++)
220 {
221 animation.durations.push_back(1000);
222 }
223 return writeanimation(animation, params);
224}
