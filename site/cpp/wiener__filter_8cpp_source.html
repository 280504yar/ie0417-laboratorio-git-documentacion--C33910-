43void CWienerFilter::
44 wiener2(InputArray _src, OutputArray _dst, int szWindowX, int szWindowY)
45{
46
47 CV_Assert(szWindowX > 0 && szWindowY > 0);
48
49 Mat src = _src.getMat();
50
51 int nRows;
52 int nCols;
53 Scalar v = 0;
54 Mat p_kernel;
55 Mat srcStub;
56 //Now create a temporary holding matrix
57 Mat p_tmpMat1, p_tmpMat2, p_tmpMat3, p_tmpMat4;
58 double noise_power;
59
60 nRows = szWindowY;
61 nCols = szWindowX;
62
63 p_kernel = Mat(nRows, nCols, CV_32F, Scalar(1.0 / (double)(nRows * nCols)));
64
65 //Local mean of input
66 filter2D(src, p_tmpMat1, -1, p_kernel, Point(nCols / 2, nRows / 2)); //localMean
67
68 //Local variance of input
69 p_tmpMat2 = src.mul(src);
70 filter2D(p_tmpMat2, p_tmpMat3, -1, p_kernel, Point(nCols / 2, nRows / 2));
71
72 //Subtract off local_mean^2 from local variance
73 p_tmpMat4 = p_tmpMat1.mul(p_tmpMat1); //localMean^2
74 p_tmpMat3 = p_tmpMat3 - p_tmpMat4;
75 // Sub(p_tmpMat3, p_tmpMat4, p_tmpMat3); //filter(in^2) - localMean^2 ==> localVariance
76
77 //Estimate noise power
78 v = mean(p_tmpMat3);
79 noise_power = v.val[0];
80 // result = local_mean + ( max(0, localVar - noise) ./ max(localVar, noise)) .* (in - local_mean)
81
82 p_tmpMat4 = src - p_tmpMat1; //in - local_mean
83
84 p_tmpMat2 = max(p_tmpMat3, noise_power); //max(localVar, noise)
85
86 add(p_tmpMat3, Scalar(-noise_power), p_tmpMat3); //localVar - noise
87 p_tmpMat3 = max(p_tmpMat3, 0); // max(0, localVar - noise)
88
89 p_tmpMat3 = (p_tmpMat3 / p_tmpMat2); //max(0, localVar-noise) / max(localVar, noise)
90
91 Mat dst = p_tmpMat3.mul(p_tmpMat4);
92 dst = dst + p_tmpMat1;
93
94 _dst.assign(dst);
95}
