104static UMat init_sdiv_table()
105{
106 cv::Mat sdiv_mat(1, 256, CV_32SC1);
107 int* sdiv = sdiv_mat.ptr<int>();
108
109 const int hsv_shift = 12;
110 const int v = 255 << hsv_shift;
111
112 sdiv[0] = 0;
113 for(int i = 1; i < 256; i++ )
114 sdiv[i] = saturate_cast<int>(v/(1.*i));
115
116 cv::UMat result;
117 sdiv_mat.copyTo(result);
118 return result;
119}

121static UMat init_hdiv_table(int hrange)
122{
123 cv::Mat hdiv_mat(1, 256, CV_32SC1);
124 int* hdiv = hdiv_mat.ptr<int>();
125
126 const int hsv_shift = 12;
127 const int v = hrange << hsv_shift;
128
129 hdiv[0] = 0;
130 for (int i = 1; i < 256; i++ )
131 hdiv[i] = saturate_cast<int>(v/(6.*i));
132
133 cv::UMat result;

139bool oclCvtColorBGR2HSV( InputArray _src, OutputArray _dst, int bidx, bool full )
140{
141 OclHelper< Set<3, 4>, Set<3>, Set<CV_8U, CV_32F> > h(_src, _dst, 3);
142
143 int hrange = _src.depth() == CV_32F ? 360 : (!full ? 180 : 256);
144
145 cv::String options = (_src.depth() == CV_8U ?
146 format("-D HRANGE=%d -D BIDX=%d -D DCN=3", hrange, bidx) :
147 format("-D HSCALE=%ff -D BIDX=%d -D DCN=3", hrange*(1.f/360.f), bidx));
148
149 if(!h.createKernel("RGB2HSV", ocl::imgproc::color_hsv_oclsrc, options))
150 {
151 return false;
152 }
153
154 if(_src.depth() == CV_8U)
155 {
156 static UMat sdiv_data = init_sdiv_table();
157 UMat hdiv_data;
158
159 if (hrange == 180)
160 {
161 static UMat hdiv_data180 = init_hdiv_table(180);
162 hdiv_data = hdiv_data180;
163 }
164 else
165 {
166 static UMat hdiv_data256 = init_hdiv_table(256);
167 hdiv_data = hdiv_data256;
168 }
169
170 h.setArg(ocl::KernelArg::PtrReadOnly(sdiv_data));
171 h.setArg(ocl::KernelArg::PtrReadOnly(hdiv_data));
172 }
173
174 return h.run();
175}
