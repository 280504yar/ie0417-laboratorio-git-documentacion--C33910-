33void equalizeHistLut_( const uchar* src, uchar* dst, int len, const uchar* lut )
34{
35 int x = 0;
36#if CV_EQUALIZE_HIST_SIMD
37 const int nlanes = VTraits<v_uint8>::vlanes();
38 for( ; x <= len - nlanes; x += nlanes )
39 {
40 v_uint8 idx = vx_load(src + x);
41 v_uint8 result = v_lut(lut, idx);
42 v_store(dst + x, result);
43 }
44#endif
45 for( ; x <= len - 4; x += 4 )
46 {
47 uchar v0 = src[x], v1 = src[x+1];
48 dst[x] = lut[v0];
49 dst[x+1] = lut[v1];
50 v0 = src[x+2]; v1 = src[x+3];
51 dst[x+2] = lut[v0];
52 dst[x+3] = lut[v1];
53 }
54 for( ; x < len; x++ )
55 dst[x] = lut[src[x]];
56}
