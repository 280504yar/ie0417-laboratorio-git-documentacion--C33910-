43+ (NSString*)makeLabel:(BOOL)isIntType val:(NSNumber*)num {
44 if (isIntType) {
45 return [NSString stringWithFormat:@"%d", num.intValue];
46 } else {
47 int exponent = 1 + (int)log10(abs(num.doubleValue));
48 if (num.doubleValue == (double)num.intValue && num.doubleValue < 10000 && num.doubleValue > -10000) {
49 return [NSString stringWithFormat:@"%d", num.intValue];;
50 } else if (exponent <= 5 && exponent >= -1) {
51 return [NSString stringWithFormat:[NSString stringWithFormat:@"%%%d.%df", 6, MIN(5 - exponent, 4)], num.doubleValue];
52 } else {
53 return [[[NSString stringWithFormat:@"%.2e", num.doubleValue] stringByReplacingOccurrencesOfString:@"e+0" withString:@"e"] stringByReplacingOccurrencesOfString:@"e-0" withString:@"e-"];
54 }
55 }
