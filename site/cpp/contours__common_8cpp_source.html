15void cv::contourTreeToResults(CTree& tree,
16 int res_type,
17 OutputArrayOfArrays& _contours,
18 OutputArray& _hierarchy)
19{
20 // check if there are no results
21 if (tree.isEmpty() || (tree.elem(0).body.isEmpty() && (tree.elem(0).first_child == -1)))
22 {
23 _contours.clear();
24 return;
25 }
26
27 CV_Assert(tree.size() < (size_t)numeric_limits<int>::max());
28 // mapping for indexes (original -> resulting)
29 // -1 - based indexing
30 vector<int> index_mapping(tree.size() + 1, -1);
31
32 const int total = (int)tree.size() - 1;
33 _contours.create(total, 1, 0, -1, true);
34 {
35 int i = 0;
36 CIterator it(tree);
37 while (!it.isDone())
38 {
39 const CNode& elem = it.getNext_s();
40 CV_Assert(elem.self() != -1);
41 if (elem.self() == 0)
42 continue;
43 index_mapping.at(elem.self() + 1) = i;
44 CV_Assert(elem.body.size() < (size_t)numeric_limits<int>::max());
45 const int sz = (int)elem.body.size();
46 _contours.create(sz, 1, res_type, i, true);
47 if (sz > 0)
48 {
49 Mat cmat = _contours.getMat(i);
50 CV_Assert(cmat.isContinuous());
51 elem.body.copyTo(cmat.data);
52 }
53 ++i;
54 }
55 }
56
57 if (_hierarchy.needed())
58 {
59 _hierarchy.create(1, total, CV_32SC4, -1, true);
60 Mat h_mat = _hierarchy.getMat();
61 int i = 0;
62 CIterator it(tree);
63 while (!it.isDone())
64 {
65 const CNode& elem = it.getNext_s();
66 if (elem.self() == 0)
67 continue;
68 Vec4i& h_vec = h_mat.at<Vec4i>(i);
69 h_vec = Vec4i(index_mapping.at(elem.next + 1),
70 index_mapping.at(elem.prev + 1),
71 index_mapping.at(elem.first_child + 1),
72 index_mapping.at(elem.parent + 1));
73 ++i;
74 }
75 }
76}
