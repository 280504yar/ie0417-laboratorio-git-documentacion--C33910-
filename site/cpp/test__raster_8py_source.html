13def lookAtMatrixCal(position, lookat, upVector):
14 tmp = position - lookat
15 norm = numpy.linalg.norm(tmp)
16 w = tmp / norm
17 tmp = numpy.cross(upVector, w)
18 norm = numpy.linalg.norm(tmp)
19 u = tmp / norm
20 v = numpy.cross(w, u)
21 res = numpy.array([
22 [u[0], u[1], u[2], 0],
23 [v[0], v[1], v[2], 0],
24 [w[0], w[1], w[2], 0],
25 [0, 0, 0, 1.0]
26 ], dtype=numpy.float32)
27 translate = numpy.array([
28 [1.0, 0, 0, -position[0]],
29 [0, 1.0, 0, -position[1]],
30 [0, 0, 1.0, -position[2]],
31 [0, 0, 0, 1.0]
32 ], dtype=numpy.float32)
33 res = numpy.matmul(res, translate)
34 return res
