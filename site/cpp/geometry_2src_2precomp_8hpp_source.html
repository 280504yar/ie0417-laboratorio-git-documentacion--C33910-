130template<typename T> inline int compressElems( T* ptr, const uchar* mask, int mstep, int count )
131{
132 int i, j;
133 for( i = j = 0; i < count; i++ )
134 if( mask[i*mstep] )

136 if( i > j )
137 ptr[j] = ptr[i];
138 j++;
139 }
140 return j;

143static inline bool haveCollinearPoints( const Mat& m, int count )
144{
145 int j, k, i = count-1;
146 const Point2f* ptr = m.ptr<Point2f>();
147
148 // check that the i-th selected point does not belong
149 // to a line connecting some previously selected points
150 // also checks that points are not too close to each other
151 for( j = 0; j < i; j++ )
152 {
153 double dx1 = ptr[j].x - ptr[i].x;
154 double dy1 = ptr[j].y - ptr[i].y;
155 for( k = 0; k < j; k++ )
156 {
157 double dx2 = ptr[k].x - ptr[i].x;
158 double dy2 = ptr[k].y - ptr[i].y;
159 if( fabs(dx2*dy1 - dy2*dx1) <= FLT_EPSILON*(fabs(dx1) + fabs(dy1) + fabs(dx2) + fabs(dy2)))
160 return true;
161 }
162 }
163 return false;
164}
