63Tensor Tensor::reshape(const char* data, const std::vector<int>& shape, bool alloc, Format fmt)
64{
65 if (kDevice == VK_NULL_HANDLE)
66 {
67 CV_Error(Error::StsError, "device is NULL!");
68 return *this;
69 }
70
71 CV_Assert(/*shape.size() > 0 &&*/ shape.size() <= 6);
72
73 if (shape_ != shape) shape_ = shape;
74 if (checkFormat(fmt) && fmt != format_) format_ = fmt;
75
76 size_t new_size = shapeCount(shape_) * elementSize(format_);
77 if (alloc || new_size > size_in_byte_)
78 alloc = true;
79 size_in_byte_ = new_size;
80
81 if (alloc)
82 {
83 buffer_.reset(new Buffer(size_in_byte_, data, usageFlag_));
84 }
85 else if (data)
86 {
87 void* p = map();
88 memcpy(p, data, size_in_byte_);
89 unMap();
90 }
91
92 return *this;
93}
