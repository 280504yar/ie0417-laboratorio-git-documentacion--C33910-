39 int estimate (const std::vector<int> &sample, std::vector<Mat> &models) const override {
40 const float * pts = points_mat.ptr<float>();
41 // (1) Extract 4 null vectors from linear equations of epipolar constraint
42 std::vector<double> coefficients(45); // 5 pts=rows, 9 columns
43 auto *coefficients_ = &coefficients[0];
44 for (int i = 0; i < 5; i++) {
45 const int smpl = 4 * sample[i];
46 const auto x1 = pts[smpl], y1 = pts[smpl+1], x2 = pts[smpl+2], y2 = pts[smpl+3];
47 (*coefficients_++) = x2 * x1;
48 (*coefficients_++) = x2 * y1;
49 (*coefficients_++) = x2;
50 (*coefficients_++) = y2 * x1;
51 (*coefficients_++) = y2 * y1;
52 (*coefficients_++) = y2;
53 (*coefficients_++) = x1;
54 (*coefficients_++) = y1;
55 (*coefficients_++) = 1;
56 }
57
58 const int num_cols = 9, num_e_mat = 4;
59 double ee[36]; // 9*4
60 if (use_svd) {
61 Matx<double, 5, 9> coeffs (&coefficients[0]);
62 Mat D, U, Vt;
63 SVDecomp(coeffs, D, U, Vt, SVD::FULL_UV + SVD::MODIFY_A);
64 const auto * const vt = (double *) Vt.data;
65 for (int i = 0; i < num_e_mat; i++)
66 for (int j = 0; j < num_cols; j++)
67 ee[i * num_cols + j] = vt[(8-i)*num_cols+j];
68 } else {
69 // eliminate linear equations
70 if (!Math::eliminateUpperTriangular(coefficients, 5, num_cols))
71 return 0;
72 for (int i = 0; i < num_e_mat; i++)
73 for (int j = 5; j < num_cols; j++)
74 ee[num_cols * i + j] = (i + 5 == j) ? 1 : 0;
75 // use back-substitution
76 for (int e = 0; e < num_e_mat; e++) {
77 const int curr_e = num_cols * e;
78 // start from the last row
79 for (int i = 4; i >= 0; i--) {
80 const int row_i = i * num_cols;
81 double acc = 0;
82 for (int j = i + 1; j < num_cols; j++)
83 acc -= coefficients[row_i + j] * ee[curr_e + j];
84 ee[curr_e + i] = acc / coefficients[row_i + i];
85 // due to numerical errors return 0 solutions
86 if (std::isnan(ee[curr_e + i]))
87 return 0;
88 }
89 }
90 }

92 const Matx<double, 4, 9> null_space(ee);
93 const Matx<double, 4, 1> null_space_mat[3][3] = {
94 {null_space.col(0), null_space.col(3), null_space.col(6)},
95 {null_space.col(1), null_space.col(4), null_space.col(7)},
96 {null_space.col(2), null_space.col(5), null_space.col(8)}};

98 Matx<double, 1, 10> eet[3][3];

100 // (2) Use the rank constraint and the trace constraint to build ten third-order polynomial
101 // equations in the three unknowns. The monomials are ordered in GrLex order and
102 // represented in a 10×20 matrix, where each row corresponds to an equation and each column
103 // corresponds to a monomial
104 if (is_nister) {
105 for (int i = 0; i < 3; i++)
106 for (int j = 0; j < 3; j++)
107 // compute EE Transpose
108 // Shorthand for multiplying the Essential matrix with its transpose.
109 eet[i][j] = multPolysDegOne(null_space_mat[i][0].val, null_space_mat[j][0].val) +
110 multPolysDegOne(null_space_mat[i][1].val, null_space_mat[j][1].val) +
111 multPolysDegOne(null_space_mat[i][2].val, null_space_mat[j][2].val);
112
113 const Matx<double, 1, 10> trace = 0.5*(eet[0][0] + eet[1][1] + eet[2][2]);
114 // Trace constraint
115 for (int i = 0; i < 3; i++)
116 for (int j = 0; j < 3; j++)
117 Mat(multPolysDegOneAndTwoNister(i == 0 ? (eet[i][0] - trace).val : eet[i][0].val, null_space_mat[0][j].val) +
118 multPolysDegOneAndTwoNister(i == 1 ? (eet[i][1] - trace).val : eet[i][1].val, null_space_mat[1][j].val) +
119 multPolysDegOneAndTwoNister(i == 2 ? (eet[i][2] - trace).val : eet[i][2].val, null_space_mat[2][j].val)).copyTo(constraint_mat.row(1+3 * j + i));
120
121 // Rank = zero determinant constraint
122 Mat(multPolysDegOneAndTwoNister(
123 (multPolysDegOne(null_space_mat[0][1].val, null_space_mat[1][2].val) -
124 multPolysDegOne(null_space_mat[0][2].val, null_space_mat[1][1].val)).val,
125 null_space_mat[2][0].val) +
126 multPolysDegOneAndTwoNister(
127 (multPolysDegOne(null_space_mat[0][2].val, null_space_mat[1][0].val) -
128 multPolysDegOne(null_space_mat[0][0].val, null_space_mat[1][2].val)).val,
129 null_space_mat[2][1].val) +
130 multPolysDegOneAndTwoNister(
131 (multPolysDegOne(null_space_mat[0][0].val, null_space_mat[1][1].val) -
132 multPolysDegOne(null_space_mat[0][1].val, null_space_mat[1][0].val)).val,
133 null_space_mat[2][2].val)).copyTo(constraint_mat.row(0));
134
135 Matx<double, 10, 10> Acoef = constraint_mat.colRange(0, 10),
136 Bcoef = constraint_mat.colRange(10, 20), A_;
137
138 if (!solve(Acoef, Bcoef, A_, DECOMP_LU)) return 0;

140 double b[3 * 13];
141 const auto * const a = A_.val;
142 for (int i = 0; i < 3; i++) {
143 const int r1_idx = i * 2 + 4, r2_idx = i * 2 + 5; // process from 4th row
144 for (int j = 0, r1_j = 0, r2_j = 0; j < 13; j++)
145 b[i*13+j] = ((j == 0 || j == 4 || j == 8) ? 0 : a[r1_idx*A_.cols+r1_j++]) - ((j == 3 || j == 7 || j == 12) ? 0 : a[r2_idx*A_.cols+r2_j++]);
