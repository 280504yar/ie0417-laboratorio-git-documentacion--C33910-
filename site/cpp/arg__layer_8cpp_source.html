11class ArgLayerImpl CV_FINAL : public ArgLayer
12{
13public:
14 enum class ArgOp
15 {
16 MIN = 0,
17 MAX = 1,
18 };
19
20 ArgLayerImpl(const LayerParams& params)
21 {
22 setParamsFrom(params);
23
24 axis = params.get<int>("axis", 0);
25 keepdims = (params.get<int>("keepdims", 1) == 1);
26 select_last_index = (params.get<int>("select_last_index", 0) == 1);
27
28 const std::string& argOp = params.get<std::string>("op");
29
30 if (argOp == "max")
31 {
32 op = ArgOp::MAX;
33 }
34 else if (argOp == "min")
35 {
36 op = ArgOp::MIN;
37 }
38 else
39 {
40 CV_Error(Error::StsBadArg, "Unsupported operation");
41 }
42 }
43
44 virtual bool supportBackend(int backendId) CV_OVERRIDE
45 {
46 return backendId == DNN_BACKEND_OPENCV;
47 }
48
49 void handleKeepDims(MatShape& shape, const int axis_) const
50 {
51 if (keepdims)
52 {
53 shape[axis_] = 1;
54 }
55 else
56 {
57 shape.erase(shape.begin() + axis_);
58 }
59 }
60
61 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
62 const int requiredOutputs,
63 std::vector<MatShape> &outputs,
64 std::vector<MatShape> &internals) const CV_OVERRIDE
65 {
66 MatShape inpShape = inputs[0];
67 // no axis for scalar
68 if (inpShape.empty()){
69 CV_Assert(axis == 0);
70 }
71
72 const int axis_ = normalize_axis(axis, inpShape);
73 // handle dims = 0 situation
74 if (inpShape.dims > 0)
75 handleKeepDims(inpShape, axis_);
76 outputs.assign(1, inpShape);
77
78 return false;
79 }
80
81 virtual void getTypes(const std::vector<MatType>& inputs,
82 const int requiredOutputs,
83 const int requiredInternals,
84 std::vector<MatType>& outputs,
85 std::vector<MatType>& internals) const CV_OVERRIDE
86 {
87 outputs.assign(1, CV_64S);
88 }
89
90 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
91 {
92 CV_TRACE_FUNCTION();
93 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
94
95 std::vector<Mat> inputs, outputs;
96 inputs_arr.getMatVector(inputs);
97 outputs_arr.getMatVector(outputs);
98
99 CV_Assert_N(inputs.size() == 1, outputs.size() == 1);
100 MatShape outShape = shape(outputs[0]);
101 Mat output(outShape, CV_32SC1);
102
103 switch (op)
104 {
105 case ArgOp::MIN:
106 cv::reduceArgMin(inputs[0], output, axis, select_last_index);
107 break;
108 case ArgOp::MAX:
109 cv::reduceArgMax(inputs[0], output, axis, select_last_index);
110 break;
111 default:
112 CV_Error(Error::StsBadArg, "Unsupported operation.");
113 }
114
115 output = output.reshape(1, outShape);
116 output.convertTo(outputs[0], CV_64SC1);
117 }
118
119private:
120 // The axis in which to compute the arg indices. Accepted range is [-r, r-1] where r = rank(data).
121 int axis;
122 // Keep the reduced dimension or not
123 bool keepdims;
124 // Whether to select the first or the last index or Max/Min.
125 bool select_last_index;
126 // Operation to be performed
127 ArgOp op;
128};
