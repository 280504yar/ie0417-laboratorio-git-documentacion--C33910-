20template<int p>
21inline float specPow(float x)
22{
23 if(p % 2 == 0)
24 {
25 float v = specPow<p/2>(x);
26 return v*v;
27 }
28 else
29 {
30 float v = specPow<(p-1)/2>(x);
31 return v*v*x;
32 }
33}
34
35template<>
36inline float specPow<0>(float /*x*/)
37{
38 return 1.f;
39}
40
41template<>
42inline float specPow<1>(float x)
43{
44 return x;
45}

48struct RenderInvoker : ParallelLoopBody
49{
50 RenderInvoker(const Points& _points, const Normals& _normals, Mat_<Vec4b>& _img, Vec3f _lightPt, Size _sz) :
51 ParallelLoopBody(),
52 points(_points),
53 normals(_normals),
54 img(_img),
55 lightPt(_lightPt),
56 sz(_sz)
57 { }
58
59 virtual void operator ()(const Range& range) const override
60 {
61 for(int y = range.start; y < range.end; y++)
62 {
63 Vec4b* imgRow = img[y];
64 const ptype* ptsRow = points[y];
65 const ptype* nrmRow = normals[y];
66
67 for(int x = 0; x < sz.width; x++)
68 {
69 Point3f p = fromPtype(ptsRow[x]);
70 Point3f n = fromPtype(nrmRow[x]);
71
72 Vec4b color;
73
74 if(isNaN(p))
75 {
76 color = Vec4b(0, 32, 0, 0);
77 }
78 else
79 {
80 const float Ka = 0.3f; //ambient coeff
81 const float Kd = 0.5f; //diffuse coeff
82 const float Ks = 0.2f; //specular coeff
83 const int sp = 20; //specular power
84
85 const float Ax = 1.f; //ambient color, can be RGB
86 const float Dx = 1.f; //diffuse color, can be RGB
87 const float Sx = 1.f; //specular color, can be RGB
88 const float Lx = 1.f; //light color
89
90 Point3f l = normalize(lightPt - Vec3f(p));
91 Point3f v = normalize(-Vec3f(p));
92 Point3f r = normalize(Vec3f(2.f*n*n.dot(l) - l));
93
94 uchar ix = (uchar)((Ax*Ka*Dx + Lx*Kd*Dx*max(0.f, n.dot(l)) +
95 Lx*Ks*Sx*specPow<sp>(max(0.f, r.dot(v))))*255.f);
96 color = Vec4b(ix, ix, ix, 0);
97 }
98
99 imgRow[x] = color;
100 }
101 }
102 }
103
104 const Points& points;
105 const Normals& normals;
106 Mat_<Vec4b>& img;
107 Vec3f lightPt;
108 Size sz;
109};

156void pyrDownPointsNormals(const Points p, const Normals n, Points &pdown, Normals &ndown)
157{
158 CV_TRACE_FUNCTION();
159
160 for(int y = 0; y < pdown.rows; y++)
161 {
162 ptype* ptsRow = pdown[y];
163 ptype* nrmRow = ndown[y];
164 const ptype* pUpRow0 = p[2*y];
165 const ptype* pUpRow1 = p[2*y+1];
166 const ptype* nUpRow0 = n[2*y];
167 const ptype* nUpRow1 = n[2*y+1];
168 for(int x = 0; x < pdown.cols; x++)
169 {
170 Point3f point = nan3, normal = nan3;
171
172 Point3f d00 = fromPtype(pUpRow0[2*x]);
173 Point3f d01 = fromPtype(pUpRow0[2*x+1]);
174 Point3f d10 = fromPtype(pUpRow1[2*x]);
175 Point3f d11 = fromPtype(pUpRow1[2*x+1]);
176
177 if(!(isNaN(d00) || isNaN(d01) || isNaN(d10) || isNaN(d11)))
178 {
179 point = (d00 + d01 + d10 + d11)*0.25f;
180
181 Point3f n00 = fromPtype(nUpRow0[2*x]);
182 Point3f n01 = fromPtype(nUpRow0[2*x+1]);
183 Point3f n10 = fromPtype(nUpRow1[2*x]);
184 Point3f n11 = fromPtype(nUpRow1[2*x+1]);
185
186 normal = (n00 + n01 + n10 + n11)*0.25f;
187 }
188
189 ptsRow[x] = toPtype(point);
190 nrmRow[x] = toPtype(normal);
191 }
192 }

195struct PyrDownBilateralInvoker : ParallelLoopBody
196{
197 PyrDownBilateralInvoker(const Depth& _depth, Depth& _depthDown, float _sigma) :
198 ParallelLoopBody(),
199 depth(_depth),
200 depthDown(_depthDown),
201 sigma(_sigma)
202 { }
203
204 virtual void operator ()(const Range& range) const override
205 {
206 float sigma3 = sigma*3;
207 const int D = 5;
208
209 for(int y = range.start; y < range.end; y++)
210 {
211 depthType* downRow = depthDown[y];
212 const depthType* srcCenterRow = depth[2*y];
213
214 for(int x = 0; x < depthDown.cols; x++)
215 {
216 depthType center = srcCenterRow[2*x];
217
218 int sx = max(0, 2*x - D/2), ex = min(2*x - D/2 + D, depth.cols-1);
219 int sy = max(0, 2*y - D/2), ey = min(2*y - D/2 + D, depth.rows-1);
220
221 depthType sum = 0;
222 int count = 0;
223
224 for(int iy = sy; iy < ey; iy++)
225 {
226 const depthType* srcRow = depth[iy];
227 for(int ix = sx; ix < ex; ix++)
228 {
229 depthType val = srcRow[ix];
230 if(abs(val - center) < sigma3)
231 {
232 sum += val; count ++;
233 }
234 }
235 }
236
237 downRow[x] = (count == 0) ? 0 : sum / count;
238 }
239 }
240 }
241
242 const Depth& depth;
243 Depth& depthDown;
244 float sigma;
245};

262struct ComputePointsNormalsInvoker : ParallelLoopBody
263{
264 ComputePointsNormalsInvoker(const Depth& _depth, Points& _points, Normals& _normals,
265 const Intr::Reprojector& _reproj, float _dfac) :
266 ParallelLoopBody(),
267 depth(_depth),
268 points(_points),
269 normals(_normals),
270 reproj(_reproj),
271 dfac(_dfac)
272 { }
273
274 virtual void operator ()(const Range& range) const override
275 {
276 for(int y = range.start; y < range.end; y++)
277 {
278 const depthType* depthRow0 = depth[y];
279 const depthType* depthRow1 = (y < depth.rows - 1) ? depth[y + 1] : 0;
280 ptype *ptsRow = points[y];
281 ptype *normRow = normals[y];
282
283 for(int x = 0; x < depth.cols; x++)
284 {
285 depthType d00 = depthRow0[x];
286 depthType z00 = d00*dfac;
287 Point3f v00 = reproj(Point3f((float)x, (float)y, z00));
288
289 Point3f p = nan3, n = nan3;
290
291 if(x < depth.cols - 1 && y < depth.rows - 1)
292 {
293 depthType d01 = depthRow0[x+1];
294 depthType d10 = depthRow1[x];
295
296 depthType z01 = d01*dfac;
297 depthType z10 = d10*dfac;
298
299 // before it was
300 //if(z00*z01*z10 != 0)
301 if(z00 != 0 && z01 != 0 && z10 != 0)
302 {
303 Point3f v01 = reproj(Point3f((float)(x+1), (float)(y+0), z01));
304 Point3f v10 = reproj(Point3f((float)(x+0), (float)(y+1), z10));
305
306 cv::Vec3f vec = (v01-v00).cross(v10-v00);
307 n = -normalize(vec);
308 p = v00;
309 }
310 }
311
312 ptsRow[x] = toPtype(p);
313 normRow[x] = toPtype(n);
314 }
315 }
316 }
317
318 const Depth& depth;
319 Points& points;
320 Normals& normals;
321 const Intr::Reprojector& reproj;
322 float dfac;
323};

368bool computePointsNormalsGpu(const Intr intr, float depthFactor, const UMat& depth,
369 UMat& points, UMat& normals)
370{
371 CV_TRACE_FUNCTION();
372
373 CV_Assert(!points.empty() && !normals.empty());
374 CV_Assert(depth.size() == points.size());
375 CV_Assert(depth.size() == normals.size());
376 CV_Assert(depth.type() == DEPTH_TYPE);
377 CV_Assert(points.type() == POINT_TYPE);
378 CV_Assert(normals.type() == POINT_TYPE);
379
380 // conversion to meters
381 float dfac = 1.f/depthFactor;
382
383 Intr::Reprojector reproj = intr.makeReprojector();
384
385 cv::String errorStr;
386 cv::String name = "computePointsNormals";
387 ocl::ProgramSource source = ocl::ptcloud::kinfu_frame_oclsrc;
388 cv::String options = "-cl-mad-enable";
389 ocl::Kernel k;
390 k.create(name.c_str(), source, options, &errorStr);
391
392 if(k.empty())
393 return false;
394
395 Vec2f fxyinv(reproj.fxinv, reproj.fyinv), cxy(reproj.cx, reproj.cy);
396
397 k.args(ocl::KernelArg::WriteOnlyNoSize(points),
398 ocl::KernelArg::WriteOnlyNoSize(normals),
399 ocl::KernelArg::ReadOnly(depth),
400 fxyinv.val,
401 cxy.val,
402 dfac);
403
404 size_t globalSize[2];
405 globalSize[0] = (size_t)depth.cols;
406 globalSize[1] = (size_t)depth.rows;
407
408 return k.run(2, globalSize, NULL, true);
409}

539static bool ocl_makeFrameFromDepth(const UMat depth, OutputArrayOfArrays points, OutputArrayOfArrays normals,
540 const Intr intr, int levels, float depthFactor,
541 float sigmaDepth, float sigmaSpatial, int kernelSize,
542 float truncateThreshold)
543{
544 CV_TRACE_FUNCTION();
545
546 // looks like OpenCV's bilateral filter works the same as KinFu's
547 UMat smooth;
548 //TODO: fix that
549 // until 32f isn't implemented in OpenCV in OpenCL, we should use our workarounds
550 //bilateralFilter(udepth, smooth, kernelSize, sigmaDepth*depthFactor, sigmaSpatial);
551 if(!customBilateralFilterGpu(depth, smooth, kernelSize, sigmaDepth*depthFactor, sigmaSpatial))
552 return false;
553
554 // depth truncation can be used in some scenes
555 UMat depthThreshold;
556 if(truncateThreshold > 0.f)
557 threshold(smooth, depthThreshold, truncateThreshold * depthFactor, 0.0, THRESH_TOZERO_INV);
558 else
559 depthThreshold = smooth;
560
561 UMat scaled = depthThreshold;
562 Size sz = smooth.size();
563 points.create(levels, 1, POINT_TYPE);
564 normals.create(levels, 1, POINT_TYPE);
565 for(int i = 0; i < levels; i++)
566 {
567 UMat& p = points.getUMatRef(i);
568 UMat& n = normals.getUMatRef(i);
569 p.create(sz, POINT_TYPE);
570 n.create(sz, POINT_TYPE);
571
572 if(!computePointsNormalsGpu(intr.scale(i), depthFactor, scaled, p, n))
573 return false;
574
575 if(i < levels - 1)
576 {
577 sz.width /= 2, sz.height /= 2;
578 UMat halfDepth(sz, DEPTH_TYPE);
579 pyrDownBilateralGpu(scaled, halfDepth, sigmaDepth*depthFactor);
580 scaled = halfDepth;
581 }
582 }
583
584 return true;
585}

588static bool ocl_buildPyramidPointsNormals(const UMat points, const UMat normals,
589 OutputArrayOfArrays pyrPoints, OutputArrayOfArrays pyrNormals,
590 int levels)
591{
592 CV_TRACE_FUNCTION();

596
597 pyrPoints .getUMatRef(0) = points;
598 pyrNormals.getUMatRef(0) = normals;
599

601 for(int i = 1; i < levels; i++)
602 {
603 UMat p1 = pyrPoints .getUMat(i-1);
604 UMat n1 = pyrNormals.getUMat(i-1);
605
606 sz.width /= 2; sz.height /= 2;
607 UMat& p0 = pyrPoints .getUMatRef(i);
608 UMat& n0 = pyrNormals.getUMatRef(i);
609 p0.create(sz, POINT_TYPE);
610 n0.create(sz, POINT_TYPE);
611
612 if(!pyrDownPointsNormalsGpu(p1, n1, p0, n0))
613 return false;
614 }
615
616 return true;
617}

673void makeFrameFromDepth(InputArray _depth,
674 OutputArray pyrPoints, OutputArray pyrNormals,
675 const Matx33f _intr, int levels, float depthFactor,
676 float sigmaDepth, float sigmaSpatial, int kernelSize,
677 float truncateThreshold)
678{
679 CV_TRACE_FUNCTION();
680
681 CV_Assert(_depth.type() == DEPTH_TYPE);
682
683 Intr intr(_intr);
684 CV_OCL_RUN(_depth.isUMat() && pyrPoints.isUMatVector() && pyrNormals.isUMatVector(),
685 ocl_makeFrameFromDepth(_depth.getUMat(), pyrPoints, pyrNormals,
686 intr, levels, depthFactor,
687 sigmaDepth, sigmaSpatial, kernelSize,
688 truncateThreshold));
689
690 int kp = pyrPoints.kind(), kn = pyrNormals.kind();
691 // There can be UMats in the container (when OpenCL is off)
692 CV_Assert(kp == _InputArray::STD_ARRAY_MAT || kp == _InputArray::STD_VECTOR_MAT || kp == _InputArray::STD_VECTOR_UMAT);
693 CV_Assert(kn == _InputArray::STD_ARRAY_MAT || kn == _InputArray::STD_VECTOR_MAT || kp == _InputArray::STD_VECTOR_UMAT);
694
695 Depth depth = _depth.getMat();
696
697 // looks like OpenCV's bilateral filter works the same as KinFu's
698 Depth smooth;
699 Depth depthNoNans = depth.clone();
700 patchNaNs(depthNoNans);
701 bilateralFilter(depthNoNans, smooth, kernelSize, sigmaDepth * depthFactor, sigmaSpatial);
702
703 // depth truncation can be used in some scenes
704 Depth depthThreshold;
705 if(truncateThreshold > 0.f)
706 threshold(smooth, depthThreshold, truncateThreshold * depthFactor, 0.0, THRESH_TOZERO_INV);
707 else
708 depthThreshold = smooth;
709
710 // we don't need depth pyramid outside this method
711 // if we do, the code is to be refactored
712
713 Depth scaled = depthThreshold;
714 Size sz = smooth.size();
