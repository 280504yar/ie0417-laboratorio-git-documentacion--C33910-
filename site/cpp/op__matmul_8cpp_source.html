48bool OpMatMul::forward(std::vector<Tensor>& ins, std::vector<Tensor>& outs)
49{
50 CV_Assert((ins.size() == 1 || ins.size() == 2) && outs.size() == 1);
51 Shape inputShape = ins[0].getShape();
52 Shape outputShape = outs[0].getShape();
53 CV_Assert(inputShape.size() == outputShape.size());
54
55 CV_Assert(inputShape.size() == 2 || inputShape.size() == 4);
56
57 if (inputShape.size() == 2)
58 {
59 batch = 0;
60 Hi = inputShape[0];
61 Wi = inputShape[1];
62
63 H0 = outputShape[0];
64 W0 = outputShape[1];
65 }
66 else if (inputShape.size() == 4)
67 {
68 batch = inputShape[kShapeIdxBatch];
69 Hi = inputShape[kShapeIdxHeight];
70 Wi = inputShape[kShapeIdxWidth];
71
72 H0 = outputShape[kShapeIdxHeight];
73 W0 = outputShape[kShapeIdxWidth];
74 }
75
76 firstForward();
77
78 int KStrip = K/KSTRIP_LEN;
79 int KStripRemain = K - KStrip * KSTRIP_LEN;
80 std::vector<int> param = {M, K, N, KStrip, KStripRemain};
81
82 std::vector<int> shape = {(int)param.size()};
83 Tensor paramTensor = Tensor(reinterpret_cast<const char *>(param.data()), shape, kFormatInt32, VK_BUFFER_USAGE_UNIFORM_BUFFER_BIT);
84
85 std::string key = "gemm_spv";
86 destTypes = {
87 VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, // input
88 VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, // weight
89 VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, // out
90 VK_DESCRIPTOR_TYPE_UNIFORM_BUFFER // param
91 };
92
93 Ptr<Pipeline> pipeline = pipelineFactoryPtr->getPipeline(key, destTypes);
94 Ptr<Descriptor> desSet = pipeline->createSet();
95 Ptr<CommandBuffer> cmdBuffer = cmdPoolPtr->allocBuffer();
96
97 VkCommandBuffer cmdBufferReal = cmdBuffer->get();
98 desSet->writeTensor(ins[0], 0);
99
100 if (weightTensorPtr)
101 desSet->writeTensor(*weightTensorPtr, 1);
102 else
103 {
104 CV_Assert(ins.size() == 2);
105 desSet->writeTensor(ins[1], 1);
106 }
107
108 desSet->writeTensor(outs[0], 2);
109 desSet->writeTensor(paramTensor, 3); // TODO change the parameter from pushconstant to buffer.
110
111 cmdBuffer->beginRecord();
112 pipeline->bind(cmdBufferReal, desSet->get());
113 vkCmdDispatch(cmdBufferReal, group_x_, group_y_, group_z_);
114 cmdBuffer->endRecord();
115
116 cmdPoolPtr->submitAndWait(cmdBufferReal);
117
118 return true;
119}
