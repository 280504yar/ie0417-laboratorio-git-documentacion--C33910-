107 void knnSearch(InputArray _query, OutputArray _indices, OutputArray _dists, int knn, int search_k) CV_OVERRIDE
108 {
109 CV_Assert(!_query.empty() && _query.isContinuous());
110 Mat query = _query.getMat(), indices, dists;
111 CV_Assert(query.type() == cv::DataType<DataType>::type);
112 CV_Assert(knn > 0 && knn <= index->get_n_items());
113
114 int numQuery = query.rows;
115 if (_indices.needed())
116 {
117 indices = _indices.getMat();
118 if (!indices.isContinuous() || indices.type() != CV_32S ||
119 indices.rows != numQuery || indices.cols != knn)
120 {
121 if (!indices.isContinuous())
122 _indices.release();
123 _indices.create(numQuery, knn, CV_32S);
124 indices = _indices.getMat();
125 }
126 }
127 else
128 indices.create(numQuery, knn, CV_32S);
129
130 if (_dists.needed())
131 {
132 dists = _dists.getMat();
133 if (!dists.isContinuous() || dists.type() != cv::DataType<DataType>::type ||
134 dists.rows != numQuery || dists.cols != knn)
135 {
136 if (!_dists.isContinuous())
137 _dists.release();
138 _dists.create(numQuery, knn, cv::DataType<DataType>::type);
139 dists = _dists.getMat();
140 }
141 }
142 else
143 dists.create(numQuery, knn, cv::DataType<DataType>::type);
144
145 auto processBatch = [&](const Range& range)
146 {
147 std::vector<int> nns;
148 std::vector<DataType> distances;
149
150 for (int i = range.start; i < range.end; ++i)
151 {
152 index->get_nns_by_vector(query.ptr<DataType>(i), knn, search_k, &nns, &distances);
153
154 std::copy(nns.begin(), nns.end(), indices.ptr<int>(i));
155 std::copy(distances.begin(), distances.end(), dists.ptr<DataType>(i));
156
157 nns.clear();
158 distances.clear();
159 }
160 };
161
162 parallel_for_(Range(0, numQuery), processBatch);
163 }

242Ptr<ANNIndex> ANNIndex::create(int dim, ANNIndex::Distance distType)
243{
244 switch (distType)
245 {
246 case ANNIndex::DIST_EUCLIDEAN:
247 return makePtr<ANNIndexImpl<float, ::cvannoy::Euclidean>>(dim);
248 break;
249 case ANNIndex::DIST_MANHATTAN:
250 return makePtr<ANNIndexImpl<float, ::cvannoy::Manhattan>>(dim);
251 break;
252 case ANNIndex::DIST_ANGULAR:
253 return makePtr<ANNIndexImpl<float, ::cvannoy::Angular>>(dim);
254 break;
255 case ANNIndex::DIST_HAMMING:
256 return makePtr<ANNIndexImpl<uchar, ::cvannoy::Hamming>>(dim);
257 break;
258 case ANNIndex::DIST_DOTPRODUCT:
259 return makePtr<ANNIndexImpl<float, ::cvannoy::DotProduct>>(dim);
260 break;
261 default:
262 CV_Error(Error::StsBadArg, "Unknown/unsupported distance type");
263 }
264};
