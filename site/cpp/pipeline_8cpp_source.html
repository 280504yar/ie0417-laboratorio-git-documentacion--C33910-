43void Descriptor::writeBuffer(VkBuffer buffer, int bindIndex, size_t size, VkDeviceSize offset)
44{
45 CV_Assert(pipeline);
46 VkWriteDescriptorSet writeSet = {};
47 VkDescriptorBufferInfo sourceInfo;
48 sourceInfo.buffer = buffer;
49 sourceInfo.offset = offset;
50 sourceInfo.range = size;
51
52 writeSet.sType = VK_STRUCTURE_TYPE_WRITE_DESCRIPTOR_SET;
53 writeSet.descriptorCount = 1;
54
55 writeSet.descriptorType = pipeline->argType(bindIndex);
56 writeSet.dstBinding = bindIndex;
57 writeSet.pBufferInfo = &sourceInfo;
58 writeSet.dstSet = desSet;
59
60 vkUpdateDescriptorSets(kDevice, 1, &writeSet, 0, nullptr);
61}

80Pipeline::Pipeline(const uint32_t* spv, size_t length,
81 const std::vector<VkDescriptorType>& _bufferTypes, VkPipelineCache& cache,
82 const std::vector<uint32_t>& localSize) : bufferTypes(_bufferTypes)
83{
84 // Step1: create Module from spv file.
85 // TODO, add the local_size_x, local_size_y, and z at here.
86 VkShaderModule shaderModule;
87 VkShaderModuleCreateInfo shaderModuleCreateInfo
88 {
89 /* .sType = */ VK_STRUCTURE_TYPE_SHADER_MODULE_CREATE_INFO,
90 /* .pNext = */ nullptr,
91 /* .flags = */ 0,
92 /* .codeSize = */ length * sizeof(uint32_t),
93 /* .pCode = */ spv,
94 };
95 VK_CHECK_RESULT(vkCreateShaderModule(kDevice, &shaderModuleCreateInfo, nullptr, &shaderModule));
96
97 // Step2: according the bufferType info set the binding.
98 std::vector<VkDescriptorSetLayoutBinding> bindings;
99 std::map<VkDescriptorType, int> typeCount;
100
101 for (int i = 0; i < bufferTypes.size(); i++)
102 {
103 auto type = bufferTypes[i];
104 if (typeCount.find(type) == typeCount.end())
105 typeCount[type] = 1;
106 else
107 typeCount[type] += 1;
108
109 VkDescriptorSetLayoutBinding binding{(uint32_t)i, type, 1, VK_SHADER_STAGE_COMPUTE_BIT, nullptr};
110 bindings.emplace_back(binding);
111 }
112
113 // Step3 : Create DescriptorSetLayout and PipelineLayout
114 {
115 // Create DescriptorSetLayout
116 VkDescriptorSetLayoutCreateInfo setLayoutCreateInfo = {};
117 setLayoutCreateInfo.sType = VK_STRUCTURE_TYPE_DESCRIPTOR_SET_LAYOUT_CREATE_INFO;
118 setLayoutCreateInfo.bindingCount = bindings.size();
119 setLayoutCreateInfo.pBindings = &bindings[0];
120 VK_CHECK_RESULT(vkCreateDescriptorSetLayout(kDevice, &setLayoutCreateInfo, NULL, &setLayout));
121
122 // Create PipelineLayout
123 VkPipelineLayoutCreateInfo pipeline_layout_create_info = {};
124 pipeline_layout_create_info.sType = VK_STRUCTURE_TYPE_PIPELINE_LAYOUT_CREATE_INFO;
125 pipeline_layout_create_info.setLayoutCount = 1;
126 pipeline_layout_create_info.pSetLayouts = &setLayout;
127 VK_CHECK_RESULT(vkCreatePipelineLayout(kDevice, &pipeline_layout_create_info, NULL, &pipelineLayout));
128 }
129
130 //Step: 4 create pipelineVk instance.
131 VkPipelineShaderStageCreateInfo stageCreateInfo = {};
132 stageCreateInfo.sType = VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO;
133 stageCreateInfo.stage = VK_SHADER_STAGE_COMPUTE_BIT;
134 stageCreateInfo.module = shaderModule;
135 stageCreateInfo.pName = "main";
136
137 VkComputePipelineCreateInfo pipelineCreateInfo = {};
138 pipelineCreateInfo.sType = VK_STRUCTURE_TYPE_COMPUTE_PIPELINE_CREATE_INFO;
139 pipelineCreateInfo.stage = stageCreateInfo;
140 pipelineCreateInfo.layout = pipelineLayout;
141
142 cv::AutoLock lock(kContextMtx);
143 VK_CHECK_RESULT(vkCreateComputePipelines(kDevice, cache, 1, &pipelineCreateInfo, 0, &pipelineVK));
144
145 // Step5: destroy shaderModule
146 vkDestroyShaderModule(kDevice, shaderModule, nullptr);
147
148 // Step6: add typeCount to desPoolSize
149 for (auto& iter : typeCount)
150 {
151 VkDescriptorPoolSize s;
152 s.descriptorCount = iter.second;
153 s.type = iter.first;
154 desPoolSize.emplace_back(s);
155 }
156}

170Ptr<Descriptor> Pipeline::createSet()
171{
172 // find unused DescriptorSet at descriptorSetList, if not, create new one and save it at that list.
173 if (!descriptorPairQueue.empty())
174 {
175 auto iter = descriptorPairQueue.front();
176 descriptorPairQueue.pop();
177
178 Ptr<Descriptor> des = Descriptor::create(iter.first, iter.second, this);
179 des->needRelease = false; // Don't release and try to reuse it.
180 return des;
181 }
182
183 // create DescriptorPool
184 VkDescriptorPool descriptorPool;
185 VkDescriptorPoolCreateInfo info = {};
186 info.sType = VK_STRUCTURE_TYPE_DESCRIPTOR_POOL_CREATE_INFO;
187 info.maxSets = 1;
188 info.poolSizeCount = desPoolSize.size();
189 info.pPoolSizes = desPoolSize.data();
190 info.flags = VK_DESCRIPTOR_POOL_CREATE_FREE_DESCRIPTOR_SET_BIT;
191 VK_CHECK_RESULT(vkCreateDescriptorPool(kDevice, &info, NULL, &descriptorPool));
192
193 // Create DescriptorSet
194 VkDescriptorSet descriptorSet;
195 VkDescriptorSetAllocateInfo allocate_info = {};
196 allocate_info.sType = VK_STRUCTURE_TYPE_DESCRIPTOR_SET_ALLOCATE_INFO;
197 allocate_info.descriptorPool = descriptorPool;
198 allocate_info.descriptorSetCount = 1;
199 allocate_info.pSetLayouts = &setLayout;
200 VK_CHECK_RESULT(vkAllocateDescriptorSets(kDevice, &allocate_info, &descriptorSet));
201
202 Ptr<Descriptor> descriptor = Descriptor::create(descriptorPool, descriptorSet, this);
203 descriptor->needRelease = false; // Don't release and try to reuse it.
204 return descriptor;
205}

207Pipeline::~Pipeline()
208{
209 // Step1: destroy all descriptors in descriptorSetList.
210 while (!descriptorPairQueue.empty())
211 {
212 auto iter = descriptorPairQueue.front();
213 descriptorPairQueue.pop();
214
215 CV_Assert(iter.first && iter.second);
216 vkFreeDescriptorSets(kDevice, iter.first, 1, &iter.second);
217 vkDestroyDescriptorPool(kDevice, iter.first, nullptr);
218 }
219
220 // Step2: destroy other resources.
221 vkDestroyPipelineLayout(kDevice, pipelineLayout, nullptr);
222 vkDestroyDescriptorSetLayout(kDevice, setLayout, nullptr);
223 vkDestroyPipeline(kDevice, pipelineVK, nullptr);
224}

253void PipelineFactory::reset()
254{
255 // Step1: destroy old pipelineCache.
256 vkDestroyPipelineCache(kDevice, pipelineCache, nullptr);
257
258 // Step2: create new PipelineCache
259 VK_CHECK_RESULT(createPipelineCache(pipelineCache));
260
261 auto iter = pipelineCreated.begin();
262 for (int i = 0; i < pipelineCreated.size(); i++, iter++)
263 {
264 iter->second.release();
265 }
266
267 pipelineCreated.clear();
268}

270Ptr<Pipeline> PipelineFactory::getPipeline(const std::string& key, const std::vector<VkDescriptorType>& types,
271 const std::vector<uint32_t>& localSize)
272{
273 auto iter = pipelineCreated.find(key);
274 if (iter != pipelineCreated.end())
275 {
276 return iter->second;
277 }
278
279 // retrieve spv from SPVMaps with given key
280 auto iterSPV = SPVMaps.find(key);
281 if (iterSPV == SPVMaps.end())
282 CV_Error(cv::Error::StsError, "Can not create SPV with the given name:"+key+"!");
283
284 const uint32_t* spv = iterSPV->second.first;
285 size_t length = iterSPV->second.second;
286
287 Ptr<Pipeline> pipeline = Pipeline::create(spv, length, types, pipelineCache, localSize);
288
289 if (pipeline)
290 {
291 pipelineCreated.insert(std::make_pair(key, pipeline));
292 }
293 else
294 {
295 CV_Error(cv::Error::StsError, "Can not Created the VkPipeline "+key);
296 }
297
298 return pipeline;
299}
