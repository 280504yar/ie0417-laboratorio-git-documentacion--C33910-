92 void hardmaxApply(const cv::Mat& src, cv::Mat& dst, const int axis)
93 {
94 const auto *src_ptr = src.ptr<const T>();
95 auto *dst_ptr = dst.ptr<T>();
96
97 const size_t outer_size = src.total(0, axis);
98 const auto mid_size = static_cast<size_t>(src.size[axis]);
99 const size_t inner_size = src.total(axis + 1);
100 const size_t outer_step = src.total(axis);
101 double nstripes = (double) outer_size * inner_size / 1024.0;
102
103 parallel_for_(Range(0, outer_size), [&](const Range& range) {
104 for (size_t outer = range.start; outer < range.end; ++outer)
105 {
106 const size_t outer_offset = outer * outer_step;
107
108 for (size_t inner = 0; inner < inner_size; ++inner)
109 {
110 T max_val = std::numeric_limits<T>::lowest();
111 size_t max_idx = 0;
112
113 // Find max along the reduction axis
114 for (size_t mid = 0; mid < mid_size; ++mid)
115 {
116 const size_t src_idx = outer_offset + mid * inner_size + inner;
117 if (src_ptr[src_idx] > max_val)
118 {
119 max_val = src_ptr[src_idx];
120 max_idx = src_idx;

122 }

124 // Set 1 for max, 0 for others
