123bool parseDLPackTensor(DLManagedTensor* tensor, cv::cuda::GpuMatND& obj, bool copy)
124{
125 if (tensor->dl_tensor.byte_offset != 0)
126 {
127 PyErr_SetString(PyExc_BufferError, "Unimplemented from_dlpack for GpuMat with memory offset");
128 return false;
129 }
130 if (tensor->dl_tensor.device.device_type != kDLCUDA)
131 {
132 PyErr_SetString(PyExc_BufferError, "cuda_GpuMat.from_dlpack expects a tensor on CUDA device");
133 return false;
134 }
135 int type = DLPackTypeToCVType(tensor->dl_tensor.dtype, (int)tensor->dl_tensor.shape[2]);
136 if (type == -1)
137 return false;
138
139 std::vector<size_t> steps(tensor->dl_tensor.ndim - 1);
140 std::vector<int> sizes(tensor->dl_tensor.ndim);
141 for (int i = 0; i < tensor->dl_tensor.ndim - 1; ++i)
142 {
143 steps[i] = static_cast<size_t>(tensor->dl_tensor.strides[i] * tensor->dl_tensor.dtype.bits / 8);
144 sizes[i] = static_cast<int>(tensor->dl_tensor.shape[i]);
145 }
146 sizes.back() = static_cast<int>(tensor->dl_tensor.shape[tensor->dl_tensor.ndim - 1]);
147 obj = cv::cuda::GpuMatND(cv::MatShape(sizes.begin(), sizes.end()), type, tensor->dl_tensor.data, steps);
148 if (copy)
149 obj = obj.clone();
150 return true;
151}
