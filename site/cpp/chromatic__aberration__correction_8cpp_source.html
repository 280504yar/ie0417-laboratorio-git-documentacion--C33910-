9CV_EXPORTS_W void loadChromaticAberrationParams(const FileNode& fs,
10 OutputArray coeffMat,
11 Size& calib_size,
12 int& degree)
13{
14 int imgW = 0, imgH = 0;
15 fs["image_width"] >> imgW;
16 fs["image_height"] >> imgH;
17
18 CV_Assert(imgW > 0 && imgH > 0);
19
20 auto readChannel = [&](const char* key,
21 std::vector<double>& coeffs_x,
22 std::vector<double>& coeffs_y,
23 int& deg_out)
24 {
25 FileNode ch = fs[key];
26 if (ch.empty())
27 CV_Error_(Error::StsParseError,
28 ("Missing channel \"%s\"", key));
29
30 ch["coeffs_x"] >> coeffs_x;
31 ch["coeffs_y"] >> coeffs_y;
32
33 if (coeffs_x.empty() || coeffs_y.empty())
34 CV_Error_(Error::StsParseError,
35 ("%s: coeffs_x/coeffs_y missing", key));
36
37 if (coeffs_x.size() != coeffs_y.size())
38 CV_Error_(Error::StsBadSize,
39 ("%s: coeffs_x (%zu) vs coeffs_y (%zu)",
40 key, coeffs_x.size(), coeffs_y.size()));
41
42 if (!checkRange(coeffs_x, true) || !checkRange(coeffs_y, true))
43 CV_Error_(Error::StsBadArg,
44 ("%s: coefficient array contains NaN/Inf", key));
45 size_t m = coeffs_x.size();
46 double n_float = (std::sqrt(1.0 + 8.0 * m) - 3.0) / 2.0;
47 int deg = static_cast<int>(std::round(n_float));
48 size_t expected_m = static_cast<size_t>((deg + 1) * (deg + 2) / 2);
49 if (m != expected_m){
50 CV_Error_(Error::StsBadArg,
51 ("Coefficient count %zu is not triangular for degree %d "
52 "(expected %zu)", m, deg, expected_m));
53 }
54 deg_out = deg;
55 };
56
57 std::vector<double> red_x, red_y, blue_x, blue_y;
58 int deg_red = 0, deg_blue = 0;
59 readChannel("red_channel", red_x, red_y, deg_red);
60 readChannel("blue_channel", blue_x, blue_y, deg_blue);
61
62
63 if (red_x.size() != blue_x.size()){
64 CV_Error_(Error::StsBadSize,
65 ("Red (%zu) and blue (%zu) coefficient counts differ",
66 red_x.size(), blue_x.size()));
67 }
68 if (deg_red != deg_blue){
69 CV_Error_(Error::StsBadArg,
70 ("Red (%d) and blue (%d) degrees differ",
71 deg_red, deg_blue));
72 }
73
74 const int mterms = (int)red_x.size();
75
76 Mat tmp(4, mterms, CV_32F);
77
78 float* Bx = tmp.ptr<float>(0);
79 float* By = tmp.ptr<float>(1);
80 float* Rx = tmp.ptr<float>(2);
81 float* Ry = tmp.ptr<float>(3);
82
83 for (int i = 0; i < mterms; ++i) {
84 Bx[i] = static_cast<float>(blue_x[i]);
85 By[i] = static_cast<float>(blue_y[i]);
86 Rx[i] = static_cast<float>(red_x[i]);
87 Ry[i] = static_cast<float>(red_y[i]);
88 }
89
90 calib_size = Size(imgW, imgH);
91 degree = deg_red;
92 tmp.copyTo(coeffMat);
93}

95static void buildRemapsFromCoeffMat(int height, int width,
96 const Mat& coeffs,
97 int degree,
98 int rowX, int rowY,
99 Mat& map_x, Mat& map_y)
100{
101 if (coeffs.type() != CV_32F) {
102 CV_Error_(Error::StsUnsupportedFormat,
103 ("coeffs Mat must be CV_32F (got type=%d)", coeffs.type()));
104 }
105
106 if (coeffs.rows != 4) {
107 CV_Error_(Error::StsBadSize,
108 ("coeffs.rows must be 4 (Bx,By,Rx,Ry); got %d", coeffs.rows));
109 }
110
111 if (rowX < 0 || rowX >= coeffs.rows) {
112 CV_Error_(Error::StsOutOfRange,
113 ("rowX index %d out of range [0,%d)", rowX, coeffs.rows));
114 }
115 if (rowY < 0 || rowY >= coeffs.rows) {
116 CV_Error_(Error::StsOutOfRange,
117 ("rowY index %d out of range [0,%d)", rowY, coeffs.rows));
118 }
119
120 if (degree < 0) {
121 CV_Error_(Error::StsBadArg,
122 ("degree must be non-negative; got %d", degree));
123 }
124
125 const int expected_terms = (degree + 1) * (degree + 2) / 2;
126 if (coeffs.cols != expected_terms) {
127 CV_Error_(Error::StsBadSize,
128 ("coeffs.cols (%d) != expected polynomial term count (%d) for degree=%d",
129 coeffs.cols, expected_terms, degree));
130 }
131
132 if (width <= 0 || height <= 0) {
133 CV_Error_(Error::StsBadArg,
134 ("width (%d) and height (%d) must be positive", width, height));
135 }
136
137 Mat X(1, width, CV_32F), Y(height, 1, CV_32F);
138 for (int i = 0; i < width; ++i) X.at<float>(0,i) = (float)i;
139 for (int j = 0; j < height; ++j) Y.at<float>(j,0) = (float)j;
140
141 Mat Xgrid, Ygrid;
142 repeat(X, height, 1, Xgrid);
143 repeat(Y, 1, width, Ygrid);
144
145 Mat dx(height, width, CV_32F);
146 Mat dy(height, width, CV_32F);
147
148 const double mean_x = width * 0.5;
149 const double mean_y = height * 0.5;
150 const double inv_std_x = 1.0 / mean_x;
151 const double inv_std_y = 1.0 / mean_y;
152
153 const float* Cx = coeffs.ptr<float>(rowX);
154 const float* Cy = coeffs.ptr<float>(rowY);
155
156
157 parallel_for_(Range(0, height), [&](const Range& rows){
158 std::vector<double> x_pow(degree + 1);
159 std::vector<double> y_pow(degree + 1);
160 for (int y = rows.start; y < rows.end; ++y) {
161 const float* XR = Xgrid.ptr<float>(y);
162 const float* YR = Ygrid.ptr<float>(y);
163 float* DX = dx.ptr<float>(y);
164 float* DY = dy.ptr<float>(y);
165 for (int x = 0; x < width; ++x) {
166 const double xn = (XR[x] - mean_x) * inv_std_x;
167 const double yn = (YR[x] - mean_y) * inv_std_y;
168
169 x_pow[0] = y_pow[0] = 1.0;
170 for (int k = 1; k <= degree; ++k) {
171 x_pow[k] = x_pow[k-1] * xn;
172 y_pow[k] = y_pow[k-1] * yn;
173 }
174
175 double dxv = 0.0, dyv = 0.0;
176 int idx = 0;
177 for (int t = 0; t <= degree; ++t){
178 for (int i = 0; i <= t; ++i){
179 const int j = t - i;
180 const double term = x_pow[i] * y_pow[j];
181 dxv += Cx[idx] * term;
182 dyv += Cy[idx] * term;
183 ++idx;
184 }
185 }
186
187 DX[x] = (float)dxv;
188 DY[x] = (float)dyv;
189 }
190 }
191 });
192
193 map_x = Xgrid - dx;
194 map_y = Ygrid - dy;
195}

197void correctChromaticAberration(InputArray input_image,
198 InputArray coefficients,
199 OutputArray output_image,
200 const Size& calib_size,
201 int calib_degree,
202 int bayer_pattern)
203{
204 Mat image = input_image.getMat();
205 const Mat coeffMat = coefficients.getMat();
206
207 if (image.channels() == 1) {
208 if (bayer_pattern < 0) {
209 CV_Error_(Error::StsBadArg,
210 ("Single-channel input detected: must pass a valid bayer_pattern"));
211 }
212 Mat dem;
213 demosaicing(image, dem, bayer_pattern);
214 image = dem;
215 }
216
217 const int height = image.rows;
218 const int width = image.cols;
219
220 if (height != calib_size.height || width != calib_size.width) {
221 CV_Error_(Error::StsBadArg,
222 ("Image size %dx%d does not match calibration %dx%d",
223 width, height, calib_size.width, calib_size.height));
224 }
225
226 std::vector<Mat> channels;
227 split(image, channels);
228 Mat b = channels[0], g = channels[1], r = channels[2];
229
230 Mat map_x_r, map_y_r, map_x_b, map_y_b;
231 buildRemapsFromCoeffMat(height, width, coeffMat, calib_degree, 2, 3, map_x_r, map_y_r);
232 buildRemapsFromCoeffMat(height, width, coeffMat, calib_degree, 0, 1, map_x_b, map_y_b);
233
234 Mat r_corr, b_corr;
235 remap(r, r_corr, map_x_r, map_y_r, INTER_LINEAR, BORDER_REPLICATE);
236 remap(b, b_corr, map_x_b, map_y_b, INTER_LINEAR, BORDER_REPLICATE);
237
238 std::vector<Mat> corrected_channels = {b_corr, g, r_corr};
239 Mat corrected_image;
240 merge(corrected_channels, output_image);
241}
