90public:
91 PoolingLayerImpl(const LayerParams& params)
92 {
93 computeMaxIdx = true;
94 globalPooling = false;
95 isGlobalPooling = std::vector<bool>(3, false);
96
97 hasDynamicShapes = params.get<bool>("has_dynamic_shapes", false);
98 shapesInitialized = !hasDynamicShapes;
99
100 if (params.has("pool") || params.has("kernel_size") ||
101 params.has("kernel_w") || params.has("kernel_h"))
102 {
103 String pool = toLowerCase(params.get<String>("pool", "max"));
104 if (pool == "max")
105 type = MAX;
106 else if (pool == "ave")
107 type = AVE;
108 else if (pool == "stochastic")
109 type = STOCHASTIC;
110 else if (pool == "sum")
111 type = SUM;
112 else
113 CV_Error(Error::StsBadArg, "Unknown pooling type \"" + pool + "\"");
114
115 getPoolingKernelParams(params, kernel_size, isGlobalPooling, pads_begin, pads_end, strides, padMode);
116 globalPooling = isGlobalPooling[0] || isGlobalPooling[1] || isGlobalPooling[2];
117 }
118 else if (params.has("pooled_w") || params.has("pooled_h"))
119 {
120 type = ROI;
121 pooledSize.width = params.get<uint32_t>("pooled_w", 1);
122 pooledSize.height = params.get<uint32_t>("pooled_h", 1);
123 }
124 else if (params.has("output_dim") && params.has("group_size"))
125 {
126 type = PSROI;
127 pooledSize.width = params.get<int>("group_size");
128 pooledSize.height = pooledSize.width;
129 psRoiOutChannels = params.get<int>("output_dim");
130 }
131 else
132 CV_Error(Error::StsBadArg, "Cannot determine pooling type");
133 setParamsFrom(params);
134 ceilMode = params.get<bool>("ceil_mode", true);
135 spatialScale = params.get<float>("spatial_scale", 1);
136 avePoolPaddedArea = params.get<bool>("ave_pool_padded_area", true);
137 }

143 void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr) CV_OVERRIDE
144 {
145 std::vector<Mat> inputs, outputs;
146 inputs_arr.getMatVector(inputs);
147 outputs_arr.getMatVector(outputs);
148
149 CV_Assert(!inputs.empty());
150
151 std::vector<int> inp;
152 std::vector<int> out;
153 for (int i = 2; i < inputs[0].dims; i++) {
154 inp.push_back(inputs[0].size[i]);
155 out.push_back(outputs[0].size[i]);
156 }
157 if (globalPooling) {
158 std::vector<size_t> finalKernel;
159 for (int i = 0; i < inp.size(); i++) {
160 int idx = isGlobalPooling.size() - inp.size() + i;
161 finalKernel.push_back(isGlobalPooling[idx] ? inp[i] : kernel_size[idx]);
162 }
163 kernel_size = finalKernel;
164 }
165
166 getConvPoolPaddings(inp, kernel_size, strides, padMode, pads_begin, pads_end);
167
168 if (inputs[0].dims == 3)
169 {
170 // Pool1D
171 kernel_size.resize(1, kernel_size[0]);
172 strides.resize(1, strides[0]);
173 pads_begin.resize(1, pads_begin[0]);
174 pads_end.resize(1, pads_end[0]);
175 }
176
177#ifdef HAVE_OPENCL
178 poolOp.release();
179#endif
180 computeMaxIdx = type == MAX && outputs.size() == 2;
181 }

183 virtual bool supportBackend(int backendId) CV_OVERRIDE
184 {
185 if (backendId == DNN_BACKEND_CUDA)
186 {
187 return type == MAX || type == AVE || type == ROI;
188 }
189#ifdef HAVE_CANN
190 if (backendId == DNN_BACKEND_CANN)
191 {
192 return type == MAX || type == AVE;
193 }
194#endif
195#ifdef HAVE_INF_ENGINE
196 if (backendId == DNN_BACKEND_INFERENCE_ENGINE_NGRAPH)
197 {
198 return type != STOCHASTIC && kernel_size.size() > 1 && (kernel_size.size() != 3 || !isArmComputePlugin()) &&
199 (!computeMaxIdx || INF_ENGINE_VER_MAJOR_GE(INF_ENGINE_RELEASE_2022_1));
200 }
201#endif
202 if (backendId == DNN_BACKEND_OPENCV)
203 {
204 if (kernel_size.size() == 3)
205 return IS_DNN_CPU_TARGET(preferableTarget);
206 if (kernel_size.size() <= 2)
207 return true;
208 else
209 return false;
210 }
211 else if (backendId == DNN_BACKEND_WEBNN)
212 {
213 if (kernel_size.empty() || kernel_size.size() == 2)
214 {
215 if (!haveWebnn())
216 {
217 return false;
218 }
219 else
220 {
221 if (!ceilMode)
222 {
223 CV_LOG_WARNING(NULL, "ceilMode is not supported by WebNN backend.");
224 return false;
225 }
226 if (computeMaxIdx)
227 {
228 CV_LOG_WARNING(NULL, "Mask is not supported by WebNN backend.");
229 return false;
230 }
231 if (type != MAX && type != AVE)
232 {
233 if (type == STOCHASTIC)
234 {
235 CV_LOG_WARNING(NULL, "Stochastic Pooling is not supported by WebNN backend.");
236 }
237 if (type == SUM)
238 {
239 CV_LOG_WARNING(NULL, "Sum Pooling is not supported by WebNN backend.");
240 }
241 if (type == ROI)
242 {
243 CV_LOG_WARNING(NULL, "ROI Pooling is not supported by WebNN backend.");
244 }
245 if (type == PSROI)
246 {
247 CV_LOG_WARNING(NULL, "Position-sensitive ROI Pooling is not supported by WebNN backend.");
248 }
249 CV_LOG_WARNING(NULL, "WebNN backend only supports MaxPooling and AveragePooling currently.");
250 return false;
251 }
252 }
253 return true;
254 }
255 }
256 else if (backendId == DNN_BACKEND_TIMVX)
257 {
258#ifdef HAVE_TIMVX
259 if (kernel_size.size() == 3)
260 {
261 // fallback to CPU implementation.
262 preferableTarget = DNN_TARGET_CPU;
263 }
264#endif
265 return false;
266 }
267 return false;
268 }

270#ifdef HAVE_OPENCL
271 bool forward_ocl(InputArrayOfArrays inps, OutputArrayOfArrays outs, InputArrayOfArrays internals)
272 {
273 std::vector<UMat> inputs;
274 std::vector<UMat> outputs;
275
276 bool use_half = (inps.depth() == CV_16F);
277 inps.getUMatVector(inputs);
278 outs.getUMatVector(outputs);
279
280 if (poolOp.empty())
281 {
282 OCL4DNNPoolConfig config;
283
284 config.in_shape = shape(inputs[0]);
285 config.out_shape = shape(outputs[0]);
286 if (inputs[0].dims == 3)
287 {
288 //Pool1D
289 config.kernel = Size(kernel_size[0], 1);
290 config.stride = Size(strides[0], 1);
291 config.pad_l = pads_begin[0];
292 config.pad_t = 0;
293 config.pad_r = pads_end[0];
294 config.pad_b = 0;
295 }
296 else
297 {
298 config.kernel = Size(kernel_size[1], kernel_size[0]);
299 config.stride = Size(strides[1], strides[0]);
300 config.pad_l = pads_begin[1];
301 config.pad_t = pads_begin[0];
302 config.pad_r = pads_end[1];
303 config.pad_b = pads_end[0];
304 }
305 config.channels = inputs[0].size[1];
306 config.pool_method = type == MAX ? LIBDNN_POOLING_METHOD_MAX :
307 (type == AVE ? LIBDNN_POOLING_METHOD_AVE :
308 LIBDNN_POOLING_METHOD_STO);
309 config.avePoolPaddedArea = avePoolPaddedArea;
310 config.computeMaxIdx = computeMaxIdx;
311 config.use_half = use_half;
312 poolOp = Ptr<OCL4DNNPool<float> >(new OCL4DNNPool<float>(config));
313 }
314
315 CV_Assert_N(inputs.size() == 1, !outputs.empty(), !computeMaxIdx || outputs.size() == 2);
316 UMat& inpMat = inputs[0];
317 UMat& outMat = outputs[0];
318 UMat maskMat = computeMaxIdx ? outputs[1] : UMat();
319
320 CV_Assert(inpMat.offset == 0 && outMat.offset == 0);
321
322 return poolOp->Forward(inpMat, outMat, maskMat);
323 }

326 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
327 {
328 CV_TRACE_FUNCTION();
329 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
330
331 if (type == MAX || type == AVE || type == STOCHASTIC)
332 {
333 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
334 forward_ocl(inputs_arr, outputs_arr, internals_arr))
335 }
336 if (inputs_arr.depth() == CV_16F)
337 {
338 forward_fallback(inputs_arr, outputs_arr, internals_arr);
339 return;
340 }
341
342 std::vector<Mat> inputs, outputs;
343 inputs_arr.getMatVector(inputs);
344 outputs_arr.getMatVector(outputs);
345
346 switch (type)
347 {
348 case MAX:
349 {
350 CV_Assert_N(inputs.size() == 1, !computeMaxIdx || outputs.size() == 2);
351 Mat mask;
352 if (computeMaxIdx)
353 mask.create(shape(outputs[1]), CV_32F);
354 maxPooling(inputs[0], outputs[0], mask);
355 if (computeMaxIdx)
356 mask.convertTo(outputs[1], CV_64S);
357 break;
358 }
359 case AVE: case SUM:
360 CV_Assert_N(inputs.size() == 1, outputs.size() == 1);
361 avePooling(inputs[0], outputs[0]);
362 break;
363 case ROI: case PSROI:
364 CV_Assert_N(inputs.size() == 2, outputs.size() == 1);
365 roiPooling(inputs[0], inputs[1], outputs[0]);
366 break;
367 default:
368 CV_Error(Error::StsNotImplemented, "Not implemented");
369 break;
370 }
371 }

373#ifdef HAVE_CUDA
374 Ptr<BackendNode> initCUDA(
375 void *context_,
376 InputArrayOfArrays inputs_,
377 InputArrayOfArrays outputs_
378 ) override
379 {
380 auto context = reinterpret_cast<csl::CSLContext*>(context_);
381 if (type == ROI)
382 return make_cuda_node<cuda4dnn::ROIPoolingOp>(preferableTarget, std::move(context->stream), spatialScale);
383
384 std::vector<cuda::GpuMatND> inputs, outputs;
385 inputs_.getGpuMatNDVector(inputs);
386 outputs_.getGpuMatNDVector(outputs);
387 MatShape input_shape = inputs[0].size;
388
389 /* storing max indices is a special case and we deal with it separately */
390 if (computeMaxIdx) {
391 CV_Assert(type == MAX);
392
393 cuda4dnn::MaxPoolingConfiguration config;
394 config.window_size.assign(std::begin(kernel_size), std::end(kernel_size));
395 config.strides.assign(std::begin(strides), std::end(strides));
396
397 if (padMode.empty())
398 {
399 config.padMode = MaxPoolingConfiguration::PaddingMode::MANUAL;
400 config.pads_begin.assign(std::begin(pads_begin), std::end(pads_begin));
401 }
402 else if (padMode == "VALID")
403 {
404 config.padMode = MaxPoolingConfiguration::PaddingMode::VALID;
405 }
406 else if (padMode == "SAME")
407 {
408 config.padMode = MaxPoolingConfiguration::PaddingMode::SAME;
409 }
410 else
411 {
412 CV_Error(Error::StsNotImplemented, padMode + " padding mode not supported by PoolingLayer");
413 }
414
415 config.input_shape.assign(std::begin(input_shape), std::end(input_shape));
416
417 int indicesType = CV_MAT_DEPTH(outputs[1].type());
418 CV_CheckType(indicesType, indicesType == CV_32S || indicesType == CV_64S, "Unsupported indices type");
419
420 if (indicesType == CV_32S)
421 return make_cuda_node_with_indices<cuda4dnn::MaxPoolingOp, int32_t>(preferableTarget, CV_MAT_DEPTH(inputs[0].type()), std::move(context->stream), config);
422 else if (indicesType == CV_64S)
423 return make_cuda_node_with_indices<cuda4dnn::MaxPoolingOp, int64_t>(preferableTarget, CV_MAT_DEPTH(inputs[0].type()), std::move(context->stream), config);
424
425 CV_Error(Error::BadDepth, "Unsupported indices type");
426 return Ptr<BackendNode>();
427 }
428
429 if (input_shape.size() == 3)
430 {
431 // Pool1D
432 // We add an extra dim for input tensor, because CuDNN support pooling only with 2 and 3 spatial dimensions
433 input_shape.insert(std::end(input_shape) - 1, 1);
434
435 // Do the similar thing for the other parameters
436 pads_begin.insert(std::begin(pads_begin), 0);
437 pads_end.insert(std::begin(pads_end), 0);
438 strides.insert(std::begin(strides), 1);
439 kernel_size.insert(std::begin(kernel_size), 1);
440 }
441
442 PoolingConfiguration config;
443 if (type == MAX)
444 {
445 config.poolMode = PoolingConfiguration::PoolingMode::MAX;
446 }
447 else if (type == AVE && !avePoolPaddedArea)
448 {
449 config.poolMode = PoolingConfiguration::PoolingMode::AVERAGE_EXCLUDE_PADDING;
450 }
451 else if (type == AVE && avePoolPaddedArea)
452 {
453 config.poolMode = PoolingConfiguration::PoolingMode::AVERAGE_INCLUDE_PADDING;
454 }
455 else
456 {
457 CV_Error(Error::StsNotImplemented, "Unsupported pooling mode");
458 }
459
460 config.window_size.assign(std::begin(kernel_size), std::end(kernel_size));
461 config.strides.assign(std::begin(strides), std::end(strides));
462
463 if (padMode.empty())
464 {
465 config.padMode = PoolingConfiguration::PaddingMode::MANUAL;
466 config.pads_begin.assign(std::begin(pads_begin), std::end(pads_begin));
467 config.pads_end.assign(std::begin(pads_end), std::end(pads_end));
468 }
469 else if (padMode == "VALID")
470 {
471 config.padMode = PoolingConfiguration::PaddingMode::VALID;
472 }
473 else if (padMode == "SAME")
474 {
475 config.padMode = PoolingConfiguration::PaddingMode::SAME;
476 }
477 else
478 {
479 CV_Error(Error::StsNotImplemented, padMode + " padding mode not supported by PoolingLayer");
480 }
481
482 if (ceilMode)
483 config.roundMode = PoolingConfiguration::RoundingMode::CEIL;
484 else
485 config.roundMode = PoolingConfiguration::RoundingMode::FLOOR;
486
487 config.input_shape.assign(std::begin(input_shape), std::end(input_shape));
488
489 return make_cuda_node<cuda4dnn::PoolingOp>(preferableTarget, std::move(context->cudnn_handle), config);
490 }

569#ifdef HAVE_DNN_NGRAPH
570 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> >& inputs,
571 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
572 {
573 CV_Assert_N((inputs.size() == 1 && (type == MAX || type == AVE || type == SUM)) || inputs.size() == 2, nodes.size() == inputs.size());
574 auto& ieInpNode = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
575
576 ov::op::PadType pad_type = ov::op::PadType::EXPLICIT;
577 if (!padMode.empty())
578 pad_type = padMode == "VALID" ? ov::op::PadType::VALID : ov::op::PadType::SAME_UPPER;
579
580 auto rounding_type = ceilMode ? ov::op::RoundingType::CEIL : ov::op::RoundingType::FLOOR;
581 if (type == AVE) {
582 auto exclude_pad = !avePoolPaddedArea;
583 auto ave_pool = std::make_shared<ov::op::v1::AvgPool>(ieInpNode, ov::Strides(strides),
584 ov::Shape(pads_begin), ov::Shape(pads_end), ov::Shape(kernel_size),
585 exclude_pad, rounding_type, pad_type);
586 return Ptr<BackendNode>(new InfEngineNgraphNode(ave_pool));
587 }
588 else if (type == SUM) {
589 ov::Shape inpShape = ieInpNode.get_shape();
590 CV_Assert(inpShape.size() == 2 + kernel_size.size());
591 std::vector<int64_t> axes;
592 for (size_t i = 0; i < kernel_size.size(); i++)
593 {
594 if (inpShape[2 + i] == kernel_size[i])
595 axes.push_back(2 + i);
596 }
597 auto reduction_axes = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{axes.size()}, axes);
598 auto reduce_sum = std::make_shared<ov::op::v1::ReduceSum>(ieInpNode, reduction_axes, true);
599 return Ptr<BackendNode>(new InfEngineNgraphNode(reduce_sum));
600 }
601 else if (type == MAX) {
602 std::shared_ptr<ov::Node> max_pool;
603 if (computeMaxIdx) {
604 std::vector<size_t> dilations(kernel_size.size(), 1);
605 max_pool = std::make_shared<ov::op::v8::MaxPool>(ieInpNode, ov::Strides(strides), ov::Strides(dilations),
606 ov::Shape(pads_begin), ov::Shape(pads_end), ov::Shape(kernel_size),
607 rounding_type, pad_type);
608 } else {
609 max_pool = std::make_shared<ov::op::v1::MaxPool>(ieInpNode, ov::Strides(strides),
610 ov::Shape(pads_begin), ov::Shape(pads_end), ov::Shape(kernel_size),
611 rounding_type, pad_type);
612 }
613 return Ptr<BackendNode>(new InfEngineNgraphNode(max_pool));
614 }
615 else if (type == ROI) {
616 auto& coords = nodes[1].dynamicCast<InfEngineNgraphNode>()->node;
617 auto roi = std::make_shared<ov::op::v0::ROIPooling>(ieInpNode, coords,
618 ov::Shape{(size_t)pooledSize.height, (size_t)pooledSize.width}, spatialScale, "max");
619 return Ptr<BackendNode>(new InfEngineNgraphNode(roi));
620 }
621 else if (type == PSROI) {
622 auto& coords = nodes[1].dynamicCast<InfEngineNgraphNode>()->node;
623 auto psroi = std::make_shared<ov::op::v0::PSROIPooling>(ieInpNode, coords,
624 (size_t)psRoiOutChannels, (size_t)pooledSize.width, spatialScale, 1, 1, "average");
625 return Ptr<BackendNode>(new InfEngineNgraphNode(psroi));
626 }
627 else

672 class PoolingInvoker : public ParallelLoopBody
673 {
674 public:
675 const Mat* src, *rois;
676 Mat *dst, *mask;
677 int pad_l, pad_t, pad_r, pad_b;
678 bool avePoolPaddedArea;
679 int nstripes;
680 bool computeMaxIdx;
681 std::vector<int> ofsbuf;
682 int poolingType;
683 float spatialScale;
684
685 std::vector<size_t> pads_begin, pads_end;
686 std::vector<size_t> kernel_size;
687 std::vector<size_t> strides;
688
689 PoolingInvoker() : src(0), rois(0), dst(0), mask(0), pad_l(0), pad_t(0), pad_r(0), pad_b(0),
690 avePoolPaddedArea(false), nstripes(0),
691 computeMaxIdx(0), poolingType(MAX), spatialScale(0) {}
692
693 static void run(const Mat& src, const Mat& rois, Mat& dst, Mat& mask,
694 std::vector<size_t> kernel_size, std::vector<size_t> strides,
695 std::vector<size_t> pads_begin, std::vector<size_t> pads_end,
696 bool avePoolPaddedArea, int poolingType, float spatialScale,
697 bool computeMaxIdx, int nstripes)
698 {
699 CV_Assert_N(
700 src.isContinuous(), dst.isContinuous(),
701 src.type() == CV_32F, src.type() == dst.type(),
702 src.dims == 3 || src.dims == 4 || src.dims == 5, dst.dims == 3 || dst.dims == 4 || dst.dims == 5,
703 (((poolingType == ROI || poolingType == PSROI) &&
704 dst.size[0] == rois.size[0]) || src.size[0] == dst.size[0]),
705 poolingType == PSROI || src.size[1] == dst.size[1],
706 (mask.empty() || (mask.type() == src.type() && mask.size == dst.size)));
707
708 PoolingInvoker p;
709
710 bool isPool1D = src.dims == 3;
711 bool isPool3D = src.dims == 5;
712
713 p.src = &src;
714 p.rois = &rois;
715 p.dst = &dst;
716
717 p.kernel_size = kernel_size;
718 p.strides = strides;
719 p.pads_begin = pads_begin;
720 p.pads_end = pads_end;
721
722 p.mask = &mask;
723 p.pad_l = pads_begin.back();
724 p.pad_t = isPool1D ? 0 : pads_begin[pads_begin.size() - 2];
725 p.pad_r = pads_end.back();
726 p.pad_b = isPool1D ? 0 : pads_end[pads_end.size() - 2];
727
728 p.avePoolPaddedArea = avePoolPaddedArea;
729 p.nstripes = nstripes;
730 p.computeMaxIdx = computeMaxIdx;
731 p.poolingType = poolingType;
732 p.spatialScale = spatialScale;
733
734 if( !computeMaxIdx )
735 {
736 int height = isPool1D ? 1 : src.size[src.dims - 2];
737 int width = src.size[src.dims - 1];
738
739 int kernel_d = isPool3D ? kernel_size[0] : 1;
740 int kernel_h = isPool1D ? 1 : kernel_size[kernel_size.size() - 2];
741 int kernel_w = kernel_size.back();
742
743 p.ofsbuf.resize(kernel_d * kernel_h * kernel_w);
744 for (int i = 0; i < kernel_d; ++i) {
745 for (int j = 0; j < kernel_h; ++j) {
746 for (int k = 0; k < kernel_w; ++k) {
747 p.ofsbuf[i * kernel_h * kernel_w + j * kernel_w + k] = width * height * i + width * j + k;
748 }
749 }
750 }
751 }
752
753 parallel_for_(Range(0, nstripes), p, nstripes);
754 }
755
756 void operator()(const Range& r) const CV_OVERRIDE
757 {
758 int channels = dst->size[1];
759
760 bool isPool3D = src->dims == 5;
761 bool isPool2D = src->dims == 4;
762 bool isPool1D = src->dims == 3;
763 int depth = isPool3D? dst->size[2] : 1;
764 int height = isPool1D? 1 : dst->size[dst->dims - 2];
765 int width = dst->size[dst->dims - 1];
766
767 int inp_depth = isPool3D? src->size[2] : 1;
768 int inp_height = isPool1D? 1 : src->size[src->dims - 2];
769 int inp_width = src->size[src->dims - 1];
770
771 size_t total = dst->total();
772 size_t stripeSize = (total + nstripes - 1)/nstripes;
773 size_t stripeStart = r.start*stripeSize;
774 size_t stripeEnd = std::min(r.end*stripeSize, total);
775
776 int kernel_d = isPool3D? kernel_size[0] : 1;
777 int kernel_h = isPool1D? 1 : kernel_size[kernel_size.size() - 2];
778 int kernel_w = kernel_size.back();
779
780 int stride_d = isPool3D? strides[0] : 0;
781 int stride_h = isPool1D? 1 :strides[strides.size() - 2];
782 int stride_w = strides.back();
783 bool compMaxIdx = computeMaxIdx;
784
785#if CV_SIMD128
786 const int* ofsptr = ofsbuf.empty() ? 0 : (const int*)&ofsbuf[0];
787 if (poolingType == MAX && !compMaxIdx && !ofsptr)
788 CV_Error(Error::StsBadArg, "ofsbuf should be initialized in this mode");
789 v_float32x4 idx00(0.f, (float)stride_w, (float)(stride_w*2), (float)(stride_w*3));
790 v_float32x4 ones = v_setall_f32(1.f);
791 v_float32x4 idx_delta = v_setall_f32((float)(inp_width - kernel_w));
792#endif
793
794 for( size_t ofs0 = stripeStart; ofs0 < stripeEnd; )
795 {
796 size_t ofs = ofs0;
797 int x0 = (int)(ofs % width);
798 ofs /= width;
799 int y0 = (int)(ofs % height);
800 ofs /= height;
801
802 int d0 = (int)(ofs % depth);
803 ofs /= depth;
804
805 int c = (int)(ofs % channels);
806 int n = (int)(ofs / channels);
807 int ystart, yend;
808 int dstart = 0, dend = 1;
809
810 const float *srcData = 0;
811 if (poolingType == ROI)
812 {
813 const float *roisData = rois->ptr<float>(n);
814 int ystartROI = roundRoiSize(roisData[2] * spatialScale);
815 int yendROI = roundRoiSize(roisData[4] * spatialScale);
816 int roiHeight = std::max(yendROI - ystartROI + 1, 1);
817 float roiRatio = (float)roiHeight / height;
818
819 ystart = ystartROI + y0 * roiRatio;
820 yend = ystartROI + std::ceil((y0 + 1) * roiRatio);
821
822 CV_Assert(roisData[0] < src->size[0]);
823 srcData = src->ptr<float>(roisData[0], c);
824 }
825 else if (poolingType == PSROI)
826 {
827 const float *roisData = rois->ptr<float>(n);
828 float ystartROI = roundRoiSize(roisData[2]) * spatialScale;
829 float yendROI = roundRoiSize(roisData[4] + 1) * spatialScale;
830 float roiHeight = std::max(yendROI - ystartROI, 0.1f);
831 float roiRatio = roiHeight / height;
832
833 ystart = (int)std::floor(ystartROI + y0 * roiRatio);
834 yend = (int)std::ceil(ystartROI + (y0 + 1) * roiRatio);
835 }
836 else
837 {
838 int pad_d_begin = (pads_begin.size() == 3) ? pads_begin[0] : 0;
839 dstart = d0 * stride_d - pad_d_begin;
840 dend = min(dstart + kernel_d, (int)(inp_depth + pads_end[0]));
841
842 ystart = y0 * stride_h - pad_t;
843 yend = min(ystart + kernel_h, inp_height + pad_b);
844 srcData = src->ptr<float>(n, c);
845 }
846 int ddelta = dend - dstart;
847 dstart = max(dstart, 0);
848 dend = min(dend, inp_depth);
849 int ydelta = yend - ystart;
850 ystart = max(ystart, 0);
851 yend = min(yend, inp_height);
852 float *dstData = &dst->ptr<float>(n, c, d0)[y0 * width];
853 float *dstMaskData = mask->data ? &mask->ptr<float>(n, c, d0)[y0 * width] : 0;
854
855 int delta = std::min((int)(stripeEnd - ofs0), width - x0);
856 ofs0 += delta;
857 int x1 = x0 + delta;
858
859 if( poolingType == MAX)
860 for( ; x0 < x1; x0++ )
861 {
862 int xstart = x0 * stride_w - pad_l;
863 int xend = min(xstart + kernel_w, inp_width);
864 xstart = max(xstart, 0);
865 if (xstart >= xend || ystart >= yend)
866 {
867 dstData[x0] = 0;
868 if (compMaxIdx && dstMaskData)
869 dstMaskData[x0] = -1;
870 continue;
871 }
872#if CV_SIMD128
873 if( isPool2D && xstart > 0 && x0 + 7 < x1 && (x0 + 7) * stride_w - pad_l + kernel_w < inp_width )
874 {
875 if( compMaxIdx )
876 {
877 v_float32x4 max_val0 = v_setall_f32(-FLT_MAX);
878 v_float32x4 max_val1 = max_val0;
879 v_float32x4 max_idx0 = v_setall_f32(-1.f);
880 v_float32x4 max_idx1 = max_idx0;
881 int index0 = ystart * inp_width + xstart;
882 v_float32x4 idx0 = v_add(idx00, v_setall_f32((float)index0));
883 v_float32x4 idx1 = v_add(idx0, v_setall_f32((float)(stride_w * 4)));
884
885 for (int y = ystart; y < yend; ++y)
886 {
887 for (int x = xstart; x < xend; ++x, idx0 = v_add(idx0, ones), idx1 = v_add(idx1, ones))
888 {
889 const int index = y * inp_width + x;
890 v_float32x4 v0(srcData[index], srcData[index + stride_w],
891 srcData[index + stride_w*2], srcData[index + stride_w*3]);
892 v_float32x4 v1(srcData[index + stride_w*4], srcData[index + stride_w*5],
893 srcData[index + stride_w*6], srcData[index + stride_w*7]);
894 max_idx0 = v_select(v_gt(v0, max_val0), idx0, max_idx0);
895 max_idx1 = v_select(v_gt(v1, max_val1), idx1, max_idx1);
896 max_val0 = v_max(max_val0, v0);
897 max_val1 = v_max(max_val1, v1);
898 }
899 idx0 = v_add(idx0, idx_delta);
900 idx1 = v_add(idx1, idx_delta);
901 }
902 v_store(dstData + x0, max_val0);
903 v_store(dstData + x0 + 4, max_val1);
904 if (dstMaskData)
905 {
906 v_store(dstMaskData + x0, max_idx0);
907 v_store(dstMaskData + x0 + 4, max_idx1);
908 }
909 x0 += 7;
910 }
911 else
912 {
913 v_float32x4 max_val0 = v_setall_f32(-FLT_MAX);
914 v_float32x4 max_val1 = max_val0;
915
916 if( yend - ystart == kernel_h )
917 {
918 const float* srcData1 = srcData + ystart*inp_width + xstart;
919 if( stride_w == 1 )
920 for (int k = 0; k < kernel_w*kernel_h; k++)
921 {
922 int index = ofsptr[k];
923 v_float32x4 v0 = v_load(srcData1 + index);
924 v_float32x4 v1 = v_load(srcData1 + index + 4);
925 max_val0 = v_max(max_val0, v0);
926 max_val1 = v_max(max_val1, v1);
927 }
928 else if( stride_w == 2 )
929 for (int k = 0; k < kernel_w*kernel_h; k++)
930 {
931 int index = ofsptr[k];
932 v_float32x4 v0, v1, dummy;
933 v_load_deinterleave(srcData1 + index, v0, dummy); // f0 f2 f4 f6 ,f1 f3 f5 f7
934 v_load_deinterleave(srcData1 + index + 8, v1, dummy); // f8 f10 f12 f14 ,f9 f11 f13 f15
935 max_val0 = v_max(max_val0, v0);
936 max_val1 = v_max(max_val1, v1);
937 }
938 else
939 for (int k = 0; k < kernel_w*kernel_h; k++)
940 {
941 int index = ofsptr[k];
942 v_float32x4 v0(srcData1[index], srcData1[index + stride_w],
943 srcData1[index + stride_w*2], srcData1[index + stride_w*3]);
944 v_float32x4 v1(srcData1[index + stride_w*4], srcData1[index + stride_w*5],
945 srcData1[index + stride_w*6], srcData1[index + stride_w*7]);
946 max_val0 = v_max(max_val0, v0);
947 max_val1 = v_max(max_val1, v1);
948 }
949 }
950 else
951 {
952 for (int y = ystart; y < yend; ++y)
953 {
954 for (int x = xstart; x < xend; ++x)
955 {
956 const int index = y * inp_width + x;
957 v_float32x4 v0(srcData[index], srcData[index + stride_w],
958 srcData[index + stride_w*2], srcData[index + stride_w*3]);
959 v_float32x4 v1(srcData[index + stride_w*4], srcData[index + stride_w*5],
960 srcData[index + stride_w*6], srcData[index + stride_w*7]);
961 max_val0 = v_max(max_val0, v0);
962 max_val1 = v_max(max_val1, v1);
963 }
964 }
965 }
966 v_store(dstData + x0, max_val0);
967 v_store(dstData + x0 + 4, max_val1);
968 x0 += 7;
969 }
970 }
971 else
972#else
973 CV_UNUSED(isPool2D);
974#endif
975 if( isPool1D )
976 {
977 const float* first = srcData + xstart;
978 const float* last = srcData + xend;
979 const float* max_elem = std::max_element(first, last);
980 if (max_elem!=last)
981 {
982 dstData[x0] = *max_elem;
983 if( compMaxIdx && dstMaskData )
984 {
985 dstMaskData[x0] = std::distance(first, max_elem);
986 }
987 }
988 }
989 else
990 {
991 float max_val = -FLT_MAX;
992 if( compMaxIdx )
993 {
994 int max_index = -1;
995 for (int d = dstart; d < dend; ++d)
996 for (int y = ystart; y < yend; ++y)
997 for (int x = xstart; x < xend; ++x)
998 {
999 const int index = d * inp_width * inp_height + y * inp_width + x;
1000 float val = srcData[index];
1001 if (val > max_val)
1002 {
1003 max_val = val;
1004 max_index = index;
1005 }
1006 }
1007 dstData[x0] = max_val;
1008 if (dstMaskData)
1009 dstMaskData[x0] = max_index;
1010 }
1011 else
1012 {
1013 for (int d = dstart; d < dend; ++d) {
1014 for (int y = ystart; y < yend; ++y) {
1015 for (int x = xstart; x < xend; ++x) {
1016 const int index = d * inp_width * inp_height + y * inp_width + x;
1017 float val = srcData[index];
1018 max_val = std::max(max_val, val);
1019 }
1020 }
1021 }
1022 dstData[x0] = max_val;
1023 }
1024 }
1025 }
1026 else if (poolingType == AVE || poolingType == SUM)
1027 {
1028 for( ; x0 < x1; ++x0)
1029 {
1030 int xstart = x0 * stride_w - pad_l;
1031 int xend = min(xstart + kernel_w, inp_width + pad_r);
1032 int xdelta = xend - xstart;
1033 xstart = max(xstart, 0);
1034 xend = min(xend, inp_width);
1035 float inv_kernel_area = avePoolPaddedArea ? xdelta * ydelta * ddelta :
1036 ((dend - dstart) * (yend - ystart) * (xend - xstart));
1037 inv_kernel_area = poolingType == AVE ? 1.0 / inv_kernel_area : 1.0;
1038#if CV_SIMD128
1039 if( isPool2D && xstart > 0 && x0 + 7 < x1 && (x0 + 7) * stride_w - pad_l + kernel_w < inp_width )
1040 {
1041 v_float32x4 sum_val0 = v_setzero_f32(), sum_val1 = v_setzero_f32();
1042 v_float32x4 ikarea = v_setall_f32(inv_kernel_area);
1043
1044 for (int y = ystart; y < yend; ++y)
1045 {
1046 for (int x = xstart; x < xend; ++x)
1047 {
1048 const int index = y * inp_width + x;
1049 v_float32x4 v0(srcData[index], srcData[index + stride_w],
1050 srcData[index + stride_w*2], srcData[index + stride_w*3]);
1051 v_float32x4 v1(srcData[index + stride_w*4], srcData[index + stride_w*5],
1052 srcData[index + stride_w*6], srcData[index + stride_w*7]);
1053 sum_val0 = v_add(sum_val0, v0);
1054 sum_val1 = v_add(sum_val1, v1);
1055 }
1056 }
1057 v_store(dstData + x0, v_mul(sum_val0, ikarea));
1058 v_store(dstData + x0 + 4, v_mul(sum_val1, ikarea));
1059 x0 += 7;
1060 }
1061 else
1062#endif
1063 if( isPool1D )
1064 {
1065 const float* first = srcData + xstart;
1066 const float* last = srcData + xend;
1067 float sum_val = std::accumulate(first, last, 0.f);
1068 dstData[x0] = sum_val*inv_kernel_area;
1069 }
1070 else
1071 {
1072 float sum_val = 0.f;
1073 for (int d = dstart; d < dend; ++d) {
1074 for (int y = ystart; y < yend; ++y) {
1075 for (int x = xstart; x < xend; ++x) {
1076 const int index = d * inp_width * inp_height + y * inp_width + x;
1077 float val = srcData[index];
1078 sum_val += val;
1079 }
1080 }
1081 }
1082 dstData[x0] = sum_val*inv_kernel_area;
1083 }
1084 }
1085 }
1086 else if (poolingType == ROI)
1087 {
1088 const float *roisData = rois->ptr<float>(n);
1089 int xstartROI = roundRoiSize(roisData[1] * spatialScale);
1090 int xendROI = roundRoiSize(roisData[3] * spatialScale);
1091 int roiWidth = std::max(xendROI - xstartROI + 1, 1);
1092 float roiRatio = (float)roiWidth / width;
1093 for( ; x0 < x1; x0++ )
1094 {
1095 int xstart = xstartROI + x0 * roiRatio;
1096 int xend = xstartROI + std::ceil((x0 + 1) * roiRatio);
1097 xstart = max(xstart, 0);
1098 xend = min(xend, inp_width);
1099 if (xstart >= xend || ystart >= yend)
1100 {
1101 dstData[x0] = 0;
1102 if (compMaxIdx && dstMaskData)
1103 dstMaskData[x0] = -1;
1104 continue;
1105 }
1106 float max_val = -FLT_MAX;
1107 for (int y = ystart; y < yend; ++y)
1108 for (int x = xstart; x < xend; ++x)
1109 {
1110 const int index = y * inp_width + x;
1111 float val = srcData[index];
1112 max_val = std::max(max_val, val);
1113 }
1114 dstData[x0] = max_val;
1115 }
1116 }
1117 else // PSROI
1118 {
1119 const float *roisData = rois->ptr<float>(n);
1120 CV_Assert(roisData[0] < src->size[0]);
1121 float xstartROI = roundRoiSize(roisData[1]) * spatialScale;
1122 float xendROI = roundRoiSize(roisData[3] + 1) * spatialScale;
1123 float roiWidth = std::max(xendROI - xstartROI, 0.1f);
1124 float roiRatio = roiWidth / width;
1125 for( ; x0 < x1; x0++ )
1126 {
1127 int xstart = (int)std::floor(xstartROI + x0 * roiRatio);
1128 int xend = (int)std::ceil(xstartROI + (x0 + 1) * roiRatio);
1129 xstart = max(xstart, 0);
1130 xend = min(xend, inp_width);
1131 if (xstart >= xend || ystart >= yend)
1132 {
1133 dstData[x0] = 0;
1134 continue;
1135 }
1136
1137 srcData = src->ptr<float>(roisData[0], (c * height + y0) * width + x0);
1138 float sum_val = 0.f;
1139 for (int y = ystart; y < yend; ++y)
1140 for (int x = xstart; x < xend; ++x)
1141 {
1142 const int index = y * inp_width + x;
1143 float val = srcData[index];
1144 sum_val += val;
1145 }
1146 dstData[x0] = sum_val / ((yend - ystart) * (xend - xstart));
1147 }
1148 }
1149 }
1150 }
1151 };

1233 }

1238 }
1239 }

1242 CV_Assert(inputs.size() == 2);
1243 outShape[0] = inputs[1][0]; // Number of proposals;
1244 }
1245 else if (type == PSROI)

1249 outShape[0] = inputs[1][0]; // Number of proposals;
1250 outShape[1] = psRoiOutChannels;
1251 }

1253 CV_Assert(numOutputs == 1 || (numOutputs == 2 && type == MAX));
1254
