98def make_optional_arg(*arg_names: str) -> Callable[[NamespaceNode, SymbolName], None]:
99 def _make_optional_arg(root_node: NamespaceNode,
100 function_symbol_name: SymbolName) -> None:
101 function = find_function_node(root_node, function_symbol_name)
102 for arg_name in arg_names:
103 found_overload_with_arg = False
104
105 for overload in function.overloads:
106 arg_idx = _find_argument_index(overload.arguments, arg_name)
107
108 # skip overloads without this argument
109 if arg_idx is None:
110 continue
111
112 # Avoid multiplying optional qualification
113 if isinstance(overload.arguments[arg_idx].type_node, OptionalTypeNode):
114 continue
115
116 overload.arguments[arg_idx].type_node = OptionalTypeNode(
117 cast(TypeNode, overload.arguments[arg_idx].type_node)
118 )
119
120 found_overload_with_arg = True
121
122 if not found_overload_with_arg:
123 raise RuntimeError(
124 f"Failed to find argument with name: '{arg_name}'"
125 f" in '{function_symbol_name.name}' overloads"
126 )
127
128 return _make_optional_arg

357def _trim_class_name_from_argument_types(
358 overloads: Iterable[FunctionNode.Overload],
359 class_name: str
360) -> None:
361 separator = f"{class_name}_"
362 for overload in overloads:
363 for arg in [arg for arg in overload.arguments
364 if arg.type_node is not None]:
365 ast_node = cast(ASTNodeTypeNode, arg.type_node)
366 if class_name in ast_node.ctype_name:
367 fixed_name = ast_node._typename.split(separator)[-1]
368 ast_node._typename = fixed_name

379def make_matlike_or_scalar_arg(*arg_names: str) -> Callable[[NamespaceNode, SymbolName], None]:
380 """Make arguments accept both MatLike and Scalar types.
381
382 This is used for functions like inRange where the C++ InputArray parameter
383 can accept both Mat objects and Scalar values (tuples, floats, etc.).
384
385 Example: cv2.inRange(img, (0, 0, 0), (255, 255, 255)) should be valid.
386 """
387 def _make_matlike_or_scalar_arg(root_node: NamespaceNode,
388 function_symbol_name: SymbolName) -> None:
389 from .predefined_types import PREDEFINED_TYPES
390
391 function = find_function_node(root_node, function_symbol_name)
392 for arg_name in arg_names:
393 found_overload_with_arg = False
394
395 for overload in function.overloads:
396 arg_idx = _find_argument_index(overload.arguments, arg_name)
397
398 # skip overloads without this argument
399 if arg_idx is None:
400 continue
401
402 current_type = overload.arguments[arg_idx].type_node
403
404 # Check if it's already a union or if it already includes Scalar
405 if isinstance(current_type, UnionTypeNode):
406 # Check if Scalar is already in the union
407 has_scalar = any(
408 isinstance(item, AliasRefTypeNode) and item.typename == "Scalar"
409 for item in current_type.items
410 )
411 if has_scalar:
412 continue
413 # Add Scalar to existing union
414 scalar_ref = AliasRefTypeNode("Scalar")
415 current_type.items = current_type.items + (scalar_ref,)
416 else:
417 # Create a union of current type and Scalar
418 scalar_ref = AliasRefTypeNode("Scalar")
419 overload.arguments[arg_idx].type_node = UnionTypeNode(
420 f"{arg_name}_type",
421 (cast(TypeNode, current_type), scalar_ref)
422 )
423
424 found_overload_with_arg = True
425
426 if not found_overload_with_arg:
427 raise RuntimeError(
428 f"Failed to find argument with name: '{arg_name}'"
429 f" in '{function_symbol_name.name}' overloads"
430 )
431
432 return _make_matlike_or_scalar_arg
