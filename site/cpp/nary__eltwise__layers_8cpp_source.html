70 void init(const std::vector<Mat>& inputs, const std::vector<Mat>& outputs)
71 {
72 narrays = 0;
73 max_ndims = 0;
74 all_ndims.clear();
75 orig_shapes.clear();
76 orig_steps.clear();
77 shapes.clear();
78 steps.clear();
79 elemsize.clear();
80
81 ninputs = inputs.size();
82 narrays = ninputs + 1;
83
84 // collect ndims
85 std::vector<int> v_inp_dims;
86 std::transform(inputs.begin(), inputs.end(), std::back_inserter(v_inp_dims), [] (const Mat& m) { return m.dims; });
87 const int* inp_ndims = v_inp_dims.data();
88 int out_ndims = outputs[0].dims;
89
90 // find max ndims for broadcasting
91 int i;
92 max_ndims = out_ndims > 2 ? out_ndims : 2;
93 for(i = 0; i < ninputs; i++)
94 max_ndims = max_ndims > inp_ndims[i] ? max_ndims : inp_ndims[i];
95
96 shapes = std::vector<std::vector<int>>(narrays, std::vector<int>(max_ndims, 0));
97 steps = std::vector<std::vector<size_t>>(narrays, std::vector<size_t>(max_ndims, 0));
98
99 for(i = 0; i <= ninputs; i++) {
100 all_ndims.push_back(i == 0 ? out_ndims : inp_ndims[i-1]);
101 std::vector<int> _size;
102 std::vector<size_t> _step;
103 if (!i) {
104 std::transform(outputs[0].size.p, outputs[0].size.p + outputs[0].dims, std::back_inserter(_size), [](int s) { return s; });
105 std::transform(outputs[0].step.p, outputs[0].step.p + outputs[0].dims, std::back_inserter(_step), [](size_t s) { return s; });
106 }
107 else {
108 std::transform(inputs[i-1].size.p, inputs[i-1].size.p + inputs[i-1].dims, std::back_inserter(_size), [](int s) { return s; });
109 std::transform(inputs[i-1].step.p, inputs[i-1].step.p + inputs[i-1].dims, std::back_inserter(_step), [](size_t s) { return s; });
110 }
111 orig_shapes.push_back(_size);
112 orig_steps.push_back(_step);
113
114 int esz = i == 0 ? outputs[0].elemSize() : inputs[i - 1].elemSize();
115 elemsize.push_back(esz);
116 }
117 }

119 bool prepare_for_broadcast_op()
120 {
121 int i, j, k;
122
123 // step 1.
124 // * make all inputs and the output max_ndims-dimensional.
125 // ** prepend dimension 1 to the mat of less dims
126 // * compute proper step's
127 for (i = this->max_ndims-1; i >= 0; i--) {
128 for (k = 0; k < this->narrays; k++) {
129 j = this->all_ndims[k] - (this->max_ndims - i);
130 int sz_i = j >= 0 ? this->orig_shapes[k][j] : 1;
131 size_t st_i = j >= 0 && this->orig_steps[k][j] > 0 ? this->orig_steps[k][j] :
132 i == this->max_ndims-1 ? elemsize[k] : this->steps[k][i+1]*this->shapes[k][i+1];
133 assert(st_i % elemsize[k] == 0);
134 this->shapes[k][i] = sz_i;
135 this->steps[k][i] = st_i;
136 }
137 }
138
139 // step 3. Let's do the flattening first,
140 // since we'd need proper values of steps to check continuity.
141 // this loop is probably the most tricky part
142 // in the whole implementation of broadcasting.
143 j = this->max_ndims > 0 ? this->max_ndims-1 : 0;
144 for (i = j - 1; i >= 0; i--) {
145 bool all_contiguous = true, all_scalars = true, all_consistent = true;
146 for(k = 0; k < this->narrays; k++) {
147 size_t st = this->steps[k][j]*this->shapes[k][j];
148 bool prev_scalar = this->shapes[k][j] == 1;
149 bool scalar = this->shapes[k][i] == 1;
150 all_contiguous = all_contiguous && (st == this->steps[k][i]);
151 all_scalars = all_scalars && scalar;
152 all_consistent = all_consistent && (scalar == prev_scalar);
153 }
154 if (all_contiguous && (all_consistent || all_scalars)) {
155 for(k = 0; k < this->narrays; k++)
156 this->shapes[k][j] *= this->shapes[k][i];
157 } else {
158 j--;
159 if (i < j) {
160 for(k = 0; k < this->narrays; k++) {
161 this->shapes[k][j] = this->shapes[k][i];
162 this->steps[k][j] = this->steps[k][i];
163 }
164 }
165 }
166 }
167
168 // step 2. Set some step's to 0's.
169 for (i = this->max_ndims-1; i >= j; i--) {
170 for (k = 0; k < this->narrays; k++)
171 this->steps[k][i] = this->shapes[k][i] == 1 ? 0 : this->steps[k][i];
172 }
173 if (this->max_ndims == 0)
174 i = 0;
175 for (; i >= 0; i--) {
176 for (k = 0; k < this->narrays; k++) {
177 this->steps[k][i] = 0;
178 this->shapes[k][i] = 1;
179 }
180 }
181 return true;
182 }

294 static MatShape findCommonShape(const std::vector<MatShape>& shapes)
295 {
296 size_t i, ninputs = shapes.size();
297 CV_Assert(ninputs > 0u);
298
299 int C0 = shapes[0].C, dims0 = shapes[0].dims, maxdims = dims0;
300 bool constC = true;
301 bool allBlock = true;
302 bool constDims = true;
303 for (i = 0; i < ninputs; i++) {
304 const MatShape& inpShape = shapes[i];
305 int dims = inpShape.dims;
306 allBlock = allBlock && inpShape.layout == DATA_LAYOUT_BLOCK;
307 constC = constC && inpShape.C == C0;
308 constDims = constDims && dims == dims0;
309 maxdims = std::max(maxdims, dims);
310 }
311
312 MatShape outShape(maxdims, 1);
313 if (allBlock && constC && constDims) {
314 outShape.layout = DATA_LAYOUT_BLOCK;
315 outShape.C = C0;
316 }
317
318 for (i = 0; i < ninputs; i++)
319 {
320 const MatShape& inpShape = shapes[i];
321 int dims = inpShape.dims, delta = maxdims - dims;
322 for (int j = 0; j < maxdims; j++) {
323 int inpsz = j < delta ? 1 : inpShape[j - delta];
324 int outsz = outShape[j];
325 if (inpsz == outsz || inpsz == 1 || outsz == 1) {
326 outShape[j] = inpsz != 1 ? inpsz : outsz;
327 } else {
328 // Clamp to minimum to tolerate off-by-1 spatial mismatches from Caffe→ONNX export.
329 CV_LOG_WARNING(NULL, "NaryEltwiseLayer: Shape mismatch detected at dimension " << j << " (" << inpsz << " vs " << outsz << "). Clamping to " << std::min(inpsz, outsz) << ".");
330 outShape[j] = std::min(inpsz, outsz);
331 }
332 }
333 }
334
335 return outShape;
336 }

476 template <typename T, typename RESULT_T, typename Functor>
477 void binary_forward_impl(const Functor& op, int ndims, const std::vector<int>& shape_in,
478 const char* data1, const std::vector<size_t>& step1_in,
479 const char* data2, const std::vector<size_t>& step2_in,
480 char* data, const std::vector<size_t>& step_in, size_t block_size) {
481 // Collapse consecutive dims that are contiguous in all three tensors.
482 std::vector<int> shape(shape_in.begin(), shape_in.begin() + ndims);
483 std::vector<size_t> step1(step1_in.begin(), step1_in.begin() + ndims);
484 std::vector<size_t> step2(step2_in.begin(), step2_in.begin() + ndims);
485 std::vector<size_t> step (step_in .begin(), step_in .begin() + ndims);
486 for (int k = (int)shape.size() - 2; k >= 0; k--) {
487 size_t e1 = (size_t)shape[k + 1] * step1[k + 1];
488 size_t e2 = (size_t)shape[k + 1] * step2[k + 1];
489 size_t e = (size_t)shape[k + 1] * step [k + 1];
490 if (step1[k] == e1 && step2[k] == e2 && step[k] == e) {
491 shape[k] *= shape[k + 1];
492 step1[k] = step1[k + 1];
493 step2[k] = step2[k + 1];
494 step [k] = step [k + 1];
495 shape.erase(shape.begin() + k + 1);
496 step1.erase(step1.begin() + k + 1);
497 step2.erase(step2.begin() + k + 1);
498 step .erase(step .begin() + k + 1);
499 }
500 }
501 ndims = (int)shape.size();
502
503 size_t dp1 = 0, dp2 = 0, dp = 0;
504 int plane_size = 1;
505 size_t nplanes = 1;
506
507 if (ndims >= 1) {
508 dp1 = step1.back() / sizeof(T);
509 dp2 = step2.back() / sizeof(T);
510 dp = step.back() / sizeof(RESULT_T);
511 plane_size = shape.back();
512
513 if (ndims >= 2) {
514 nplanes = std::accumulate(shape.begin(), shape.end() - 1, 1, std::multiplies<size_t>());
515 }
516 }
517
518 const bool is_add = (this->op == OPERATION::SUM || this->op == OPERATION::ADD);
519 const bool is_sub = (this->op == OPERATION::SUB);
520 const bool is_mul = (this->op == OPERATION::PROD);
521 const bool is_div = (this->op == OPERATION::DIV);
522 const bool simd_f32 = std::is_same<T, float>::value && std::is_same<RESULT_T, float>::value &&
523 (is_add || is_sub || is_mul || is_div);
524 const int simd_bin_op = is_add ? cpu_baseline::SIMD_BIN_ADD :
525 is_sub ? cpu_baseline::SIMD_BIN_SUB :
526 is_mul ? cpu_baseline::SIMD_BIN_MUL :
527 is_div ? cpu_baseline::SIMD_BIN_DIV : -1;
528
529 // Fast path: fully-flat contiguous float binop -> one big parallel SIMD loop.
530 if (simd_f32 && ndims == 1 && dp1 == 1 && dp2 == 1 && dp == 1) {
531 int64_t total = (int64_t)nplanes * plane_size;
532 const float* p1 = (const float*)data1;
533 const float* p2 = (const float*)data2;
534 float* po = (float*)data;
535 const int64_t chunk = 4096;
536 int64_t nchunks = (total + chunk - 1) / chunk;
537 parallel_for_(Range(0, (int)nchunks), [&](const Range& r) {
538 for (int c = r.start; c < r.end; c++) {
539 int64_t start = c * chunk;
540 int64_t end = std::min(start + chunk, total);
541 int n = (int)(end - start);
542 simd_binop_f32_dispatch(p1 + start, p2 + start, po + start, n, simd_bin_op);
543 }
544 });
545 return;
546 }
547
548 if (nplanes == 1) { // parallelize within the plane
549 const T* ptr1 = (const T*)data1;
550 const T* ptr2 = (const T*)data2;
551 RESULT_T* ptr = (RESULT_T*)data;
552 auto worker = [&](const Range &r) {
553 if (dp1 == 1 && dp2 == 1 && dp == 1) {
554 if (simd_f32) {
555 const float* p1 = (const float*)(const void*)&ptr1[r.start];
556 const float* p2 = (const float*)(const void*)&ptr2[r.start];
557 float* po = (float*)(void*)&ptr[r.start];
558 simd_binop_f32_dispatch(p1, p2, po, r.end - r.start, simd_bin_op);
559 } else {
560 for (int i = r.start; i < r.end; i++)
561 ptr[i] = op(ptr1[i], ptr2[i]);
562 }
563 } else if (dp1 == 1 && dp2 == 0 && dp == 1){
564 T x2 = *ptr2;
565 for(int i = r.start; i < r.end; i++) {
566 ptr[i] = op(ptr1[i], x2);
567 }
568 } else if (dp1 == 0 && dp2 == 1 && dp == 1){
569 T x1 = *ptr1;
570 for(int i = r.start; i < r.end; i++) {
571 ptr[i] = op(x1, ptr2[i]);
572 }
573 } else {
574 for(int i = r.start; i < r.end; i++, ptr1 += dp1, ptr2 += dp2, ptr += dp) {
575 *ptr = op(*ptr1, *ptr2);
576 }
577 }
578 };
579
580 double nstripes = plane_size * (1.0 / double(block_size));
581 parallel_for_(Range(0, plane_size), worker, nstripes);
582 } else { // parallelize across planes
583 auto worker = [&](const Range &r) {
584 for (int plane_idx = r.start; plane_idx < r.end; plane_idx++) {
585 const char* ptr1_ = data1;
586 const char* ptr2_ = data2;
587 char* ptr_ = data;
588 size_t idx = plane_idx;
589 for (int k = ndims - 2; k >= 0; k--) {
590 size_t next_idx = idx / shape[k];
591 size_t i_k = (int)(idx - next_idx * shape[k]);
592 ptr1_ += i_k * step1[k];
593 ptr2_ += i_k * step2[k];
594 ptr_ += i_k * step[k];
595 idx = next_idx;
596 }
597
598 const T* ptr1 = (const T*)ptr1_;
599 const T* ptr2 = (const T*)ptr2_;
600 RESULT_T* ptr = (RESULT_T*)ptr_;
601 if (dp1 == 1 && dp2 == 1 && dp == 1) {
602 if (simd_f32) {
603 simd_binop_f32_dispatch((const float*)(const void*)ptr1,
604 (const float*)(const void*)ptr2,
605 (float*)(void*)ptr,
606 plane_size, simd_bin_op);
607 } else {
608 for (int i = 0; i < plane_size; i++)
609 ptr[i] = op(ptr1[i], ptr2[i]);
610 }
611 } else if (dp1 == 1 && dp2 == 0 && dp == 1){
612 T x2 = *ptr2;
613 for(int i = 0; i < plane_size; i++) {
614 ptr[i] = op(ptr1[i], x2);
615 }
616 } else if (dp1 == 0 && dp2 == 1 && dp == 1){
617 T x1 = *ptr1;
618 for(int i = 0; i < plane_size; i++) {
619 ptr[i] = op(x1, ptr2[i]);
620 }
621 } else {
622 for(int i = 0; i < plane_size; i++, ptr1 += dp1, ptr2 += dp2, ptr += dp) {
623 *ptr = op(*ptr1, *ptr2);
624 }
625 }
626 }
627 };
628 // Use total work (planes * plane_size) so large broadcasts parallelize
629 // even when nplanes alone is below the old nstripes threshold. Target
630 // ~16KB of output work per stripe so even tiny plane_size ops (e.g.
631 // 2-element broadcast rows) spread across threads instead of running
632 // serially.
633 double nstripes = (double)nplanes * plane_size * sizeof(T) / 16384.0;
634 if (nstripes < 1.0) nstripes = 1.0;
635 (void)block_size;
636 parallel_for_(Range(0, nplanes), worker, nstripes);
637 }
638 }

659 template<typename T, typename Functor>
660 void nary_forward_impl(const Functor& op, const T scale, int ninputs, int ndims, const std::vector<int>& shape,
661 const char** inp, char* out, const std::vector<std::vector<size_t>>& steps, size_t block_size) {
662 CV_Assert(ndims >= 2);
663 size_t dp = steps[0].back() / sizeof(T);
664 size_t dp1 = steps[1].back() / sizeof(T);
665 size_t dp2 = steps[2].back() / sizeof(T);
666
667 int plane_size = shape.back();
668 int nplanes = std::accumulate(shape.begin(), shape.end() - 1, 1, std::multiplies<int>());
669
670 if (nplanes == 1) { // parallelize within the plane
671 AutoBuffer<char*> buf_ptrs(steps.size());
672 char** ptrs = buf_ptrs.data();
673 ptrs[0] = out;
674 for (int i = 0; i < ninputs; i++) {
675 ptrs[i+1] = (char*)inp[i];
676 }
677 auto worker = [&](const Range &r) {
678 const T* ptr1 = (const T*)(ptrs[1]);
679 const T* ptr2 = (const T*)(ptrs[2]);
680 T* ptr = (T*)(ptrs[0]);
681 if (dp == 1 && dp1 == 1 && dp2 == 1) {
682 for (int i = r.start; i < r.end; i++) {
683 ptr[i] = op(ptr1[i], ptr2[i]);
684 }
685 for (int j = 2; j < ninputs; j++) {
686 size_t dpj = steps[j + 1].back() / sizeof(T);
687 const T* ptrj = (const T*)(ptrs[j + 1]);
688 if (dpj == 1) {
689 for (int i = r.start; i < r.end; i++) {
690 ptr[i] = saturate_cast<T>(op(ptr[i], ptrj[i]) * scale);
691 }
692 } else {
693 for (int i = r.start; i < r.end; i++, ptrj += dpj) {
694 ptr[i] = saturate_cast<T>(op(ptr[i], *ptrj) * scale);
695 }
696 }
697 }
698 } else {
699 auto *tmp = ptr;
700 for (int i = r.start; i < r.end; i++, ptr += dp, ptr1 += dp1, ptr2 += dp2) {
701 *ptr = op(*ptr1, *ptr2);
702 }
703 ptr = tmp;
704 for (int j = 2; j < ninputs; j++) {
705 size_t dpj = steps[j + 1].back() / sizeof(T);
706 const T* ptr_j = (const T*)(ptrs[j + 1]);
707 for (int i = r.start; i < r.end; i++, ptr += dp, ptr_j += dpj) {
708 *ptr = saturate_cast<T>(op(*ptr, *ptr_j) * scale);
709 }
710 }
711 }
712 };
713 double nstripes = plane_size * (1.0 / double(block_size));
714 parallel_for_(Range(0, plane_size), worker, nstripes);
715 } else { // parallelize across the plane
716 auto worker = [&](const Range &r) {
717 AutoBuffer<char*> buf_ptrs(steps.size());
718 char** ptrs = buf_ptrs.data();
719 for (int plane_idx = r.start; plane_idx < r.end; plane_idx++) {
720 ptrs[0] = out;
721 for (int i = 0; i < ninputs; i++) ptrs[i+1] = (char*)inp[i];
722 size_t idx = plane_idx;
723 for (int k = ndims - 2; k >= 0; k--) {
724 size_t next_idx = idx / shape[k];
725 int i_k = (int)(idx - next_idx * shape[k]);
726 for (int i = 0; i <= ninputs; i++) {
727 ptrs[i] += i_k * steps[i][k];
728 }
729 idx = next_idx;
730 }
731
732 const T* ptr1 = (const T*)(ptrs[1]);
733 const T* ptr2 = (const T*)(ptrs[2]);
734 T* ptr = (T*)(ptrs[0]);
735 if (dp == 1 && dp1 == 1 && dp2 == 1) {
736 for (int i = 0; i < plane_size; i++) {
737 ptr[i] = saturate_cast<T>(op(ptr1[i], ptr2[i]) * scale);
738 }
739 for (int j = 2; j < ninputs; j++) {
740 size_t dpj = steps[j + 1].back() / sizeof(T);
741 const T* ptrj = (const T*)(ptrs[j + 1]);
742 if (dpj == 1) {
743 for (int i = 0; i < plane_size; i++) {
744 ptr[i] = op(ptr[i], saturate_cast<T>(ptrj[i] * scale));
745 }
746 } else {
747 for (int i = 0; i < plane_size; i++, ptrj += dpj) {
748 ptr[i] = op(ptr[i], saturate_cast<T>(*ptrj * scale));
749 }
750 }
751 }
752 } else {
753 auto *tmp = ptr;
754 for (int i = 0; i < plane_size; i++, ptr += dp, ptr1 += dp1, ptr2 += dp2) {
755 *ptr = saturate_cast<T>(op(*ptr1, *ptr2) * scale);
756 }
757 ptr = tmp;
758 for (int j = 2; j < ninputs; j++) {
759 size_t dpj = steps[j + 1].back() / sizeof(T);
760 const T* ptrj = (const T*)(ptrs[j + 1]);
761 for (int i = 0; i < plane_size; i++, ptr += dp, ptrj += dpj) {
762 *ptr = op(*ptr, saturate_cast<T>(*ptrj * scale));
763 }
764 }
765 }
766 }
767 };
768 double nstripes = (double)nplanes * plane_size * sizeof(T) / 16384.0;
769 if (nstripes < 1.0) nstripes = 1.0;
770 (void)block_size;
771 parallel_for_(Range(0, nplanes), worker, nstripes);
772 }
773 }

812 template <typename T_INP1, typename T_INP2, typename T_INP3, typename T_OUT, typename Functor>
813 void ternary_forward_impl(
814 const Functor& op, int ndims, const std::vector<int>& shape,
815 const char* data1, const std::vector<size_t>& step1,
816 const char* data2, const std::vector<size_t>& step2,
817 const char* data3, const std::vector<size_t>& step3,
818 char* data, const std::vector<size_t>& step, size_t block_size) {
819 CV_Assert(ndims >= 2);
820 size_t dp1 = step1.back() / sizeof(T_INP1);
821 size_t dp2 = step2.back() / sizeof(T_INP2);
822 size_t dp3 = step3.back() / sizeof(T_INP3);
823 size_t dp = step.back() / sizeof(T_OUT);
824 int plane_size = shape.back();
825 int nplanes = std::accumulate(shape.begin(), shape.end() - 1, 1, std::multiplies<int>());
826
827 if (nplanes == 1) { // parallelize within the plane
828 const auto *ptr1 = (const T_INP1*)data1;
829 const auto *ptr2 = (const T_INP2*)data2;
830 const auto *ptr3 = (const T_INP3*)data3;
831 auto* ptr = (T_OUT*)data;
832 auto worker = [&](const Range &r) {
833 if (dp1 == 1 && dp2 == 1 && dp3 == 1 && dp == 1) {
834 for (int i = r.start; i < r.end; i++) {
835 ptr[i] = op(ptr1[i], ptr2[i], ptr3[i]);
836 }
837 } else if (dp1 == 0 && dp2 == 1 && dp3 == 1 && dp == 1){
838 auto x1 = *ptr1;
839 for (int i = r.start; i < r.end; i++) {
840 ptr[i] = op(x1, ptr2[i], ptr3[i]);
841 }
842 } else if (dp1 == 1 && dp2 == 0 && dp3 == 1 && dp == 1){
843 auto x2 = *ptr2;
844 for (int i = r.start; i < r.end; i++) {
845 ptr[i] = op(ptr1[i], x2, ptr3[i]);
846 }
847 } else if (dp1 == 1 && dp2 == 1 && dp3 == 1 && dp == 1) {
848 auto x3 = *ptr3;
849 for (int i = r.start; i < r.end; i++) {
850 ptr[i] = op(ptr1[i], ptr2[i], x3);
851 }
852 } else {
853 for(int i = r.start; i < r.end; i++, ptr1 += dp1, ptr2 += dp2, ptr3 += dp3, ptr += dp) {
854 *ptr = op(*ptr1, *ptr2, *ptr3);
855 }
856 }
857 };
858 double nstripes = plane_size * (1.0 / double(block_size));
859 parallel_for_(Range(0, plane_size), worker, nstripes);
860 } else { // parallelize across planes
861 auto worker = [&](const Range &r) {
862 for (int plane_idx = r.start; plane_idx < r.end; plane_idx++) {
863 const char* ptr1_ = data1;
864 const char* ptr2_ = data2;
865 const char* ptr3_ = data3;
866 char* ptr_ = data;
867 size_t idx = plane_idx;
868 for (int k = ndims - 2; k >= 0; k--)
869 {
870 size_t next_idx = idx / shape[k];
871 int i_k = (int)(idx - next_idx * shape[k]);
872 ptr1_ += i_k * step1[k];
873 ptr2_ += i_k * step2[k];
874 ptr3_ += i_k * step3[k];
875 ptr_ += i_k * step[k];
876 idx = next_idx;
877 }
878
879 const auto *ptr1 = (const T_INP1*)ptr1_;
880 const auto *ptr2 = (const T_INP2*)ptr2_;
881 const auto *ptr3 = (const T_INP3*)ptr3_;
882 auto* ptr = (T_OUT*)ptr_;
883 if (dp1 == 1 && dp2 == 1 && dp3 == 1 && dp == 1) {
884 for (int i = 0; i < plane_size; i++) {
885 ptr[i] = op(ptr1[i], ptr2[i], ptr3[i]);
886 }
887 } else if (dp1 == 0 && dp2 == 1 && dp3 == 1 && dp == 1){
888 auto x1 = *ptr1;
889 for (int i = 0; i < plane_size; i++) {
890 ptr[i] = op(x1, ptr2[i], ptr3[i]);
891 }
892 } else if (dp1 == 1 && dp2 == 0 && dp3 == 1 && dp == 1){
893 auto x2 = *ptr2;
894 for (int i = 0; i < plane_size; i++) {
895 ptr[i] = op(ptr1[i], x2, ptr3[i]);
896 }
897 } else if (dp1 == 1 && dp2 == 1 && dp3 == 1 && dp == 1) {
898 auto x3 = *ptr3;
899 for (int i = 0; i < plane_size; i++) {
900 ptr[i] = op(ptr1[i], ptr2[i], x3);
901 }
902 } else {
903 for(int i = 0; i < plane_size; i++, ptr1 += dp1, ptr2 += dp2, ptr3 += dp3, ptr += dp) {
904 *ptr = op(*ptr1, *ptr2, *ptr3);
905 }
906 }
907 }
908 };
909 double nstripes = (double)nplanes * plane_size * (sizeof(T_INP1) + sizeof(T_OUT)) / 16384.0;
910 if (nstripes < 1.0) nstripes = 1.0;
911 (void)block_size;
912 parallel_for_(Range(0, nplanes), worker, nstripes);
913 }

916 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
917 {
918 CV_TRACE_FUNCTION();
919 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
920
921 if (inputs_arr.depth() == CV_16F)
922 {
923 forward_fallback(inputs_arr, outputs_arr, internals_arr);
924 return;
925 }
926
927 std::vector<Mat> inputs, outputs;
928 inputs_arr.getMatVector(inputs);
929 outputs_arr.getMatVector(outputs);
930
931 if (inputs.size() == 1) {
932 inputs[0].copyTo(outputs[0]);
933 return;
934 }
935
936 std::vector<Mat> used_inputs = inputs;
937
938 const Mat& out0 = outputs[0];
939 bool needsCrop = false;
940 for (const auto& inp : used_inputs) {
941 int delta = out0.dims - inp.dims;
942 for (int d = 0; d < inp.dims; d++) {
943 int od = d + delta;
944 if (od >= 0 && inp.size[d] > 1 && out0.size[od] > 1 &&
945 inp.size[d] != out0.size[od]) {
946 needsCrop = true; break;
947 }
948 }
949 if (needsCrop) break;
950 }
951 if (needsCrop) {
952 std::vector<Range> ranges;
953 for (size_t k = 0; k < used_inputs.size(); k++) {
954 Mat& inp = used_inputs[k];
955 int delta = out0.dims - inp.dims;
956 bool modified = false;
957 ranges.assign(inp.dims, Range::all());
958 for (int d = 0; d < inp.dims; d++) {
959 int od = d + delta;
960 if (od >= 0 && inp.size[d] > out0.size[od] &&
961 inp.size[d] > 1 && out0.size[od] > 1) {
962 ranges[d] = Range(0, out0.size[od]);
963 modified = true;
964 }
965 }
966 if (modified)
967 inp = inp(ranges).clone();
968 }
969 helper.init(used_inputs, outputs);
970 CV_CheckTrue(helper.prepare_for_broadcast_op(), "NaryEltwiseLayer: Preparation for broadcasting failed");
971 }
972
973 if (op == OPERATION::POW) {
974 CV_Assert(used_inputs.size() == 2);
975 const int out_type = outputs[0].type();
976 if (used_inputs[0].type() != out_type || used_inputs[1].type() != out_type) {
977 Mat a_conv, b_conv;
978 used_inputs[0].convertTo(a_conv, out_type);
979 used_inputs[1].convertTo(b_conv, out_type);
980 used_inputs = {a_conv, b_conv};
981 helper.init(used_inputs, outputs);
982 CV_CheckTrue(helper.prepare_for_broadcast_op(), "NaryEltwiseLayer: Preparation for broadcasting failed");
983 }
984 }
985
986 int type_for_dispatch = (op == OPERATION::WHERE || op == OPERATION::POW) ? outputs.front().type() : used_inputs.front().type();
987 typeDispatch(type_for_dispatch, used_inputs.size(), used_inputs, outputs);
988 }

990 template<typename T, typename... Args>
991 inline typename std::enable_if<std::is_integral<T>::value && !std::is_same<T, bool>::value, void>::type opDispatch(size_t ninputs, Args&&... args)
992 {
993 if (ninputs == 2) {
994 switch (op) {
995 case OPERATION::EQUAL: {
996 auto equal = [](const T &a, const T &b) { return a == b; };
997 binary_forward<T, bool>(equal, std::forward<Args>(args)...);
998 break;
999 }
1000 case OPERATION::GREATER: {
1001 auto greater = [](const T &a, const T &b) { return a > b; };
1002 binary_forward<T, bool>(greater, std::forward<Args>(args)...);
1003 break;
1004 }
1005 case OPERATION::GREATER_EQUAL: {
1006 auto greater_equal = [](const T &a, const T &b) { return a >= b; };
1007 binary_forward<T, bool>(greater_equal, std::forward<Args>(args)...);
1008 break;
1009 }
1010 case OPERATION::LESS: {
1011 auto less = [](const T &a, const T &b) { return a < b; };
1012 binary_forward<T, bool>(less, std::forward<Args>(args)...);
1013 break;
1014 }
1015 case OPERATION::LESS_EQUAL: {
1016 auto less_equal = [](const T &a, const T &b) { return a <= b; };
1017 binary_forward<T, bool>(less_equal, std::forward<Args>(args)...);
1018 break;
1019 }
1020 case OPERATION::POW: {
1021 auto pow = [] (const T& a, const T& b) { return saturate_cast<T>(std::pow((double)a, (double)b)); };
1022 binary_forward<T, T>(pow, std::forward<Args>(args)..., 1e5);
1023 break;
1024 }
1025 case OPERATION::BITSHIFT: {
1026 auto bitshift = [] (const uint8_t &a, const uint8_t &b) { return a << b; };
1027 binary_forward<T, T>(bitshift, std::forward<Args>(args)...);
1028 break;
1029 }
1030 case OPERATION::MAX: {
1031 auto max = [](const T &a, const T &b) { return std::max(a, b); };
1032 binary_forward<T, T>(max, std::forward<Args>(args)...);
1033 break;
1034 }
1035 case OPERATION::MEAN: {
1036 auto mean = [](const T &a, const T &b) { return (a + b) / T{2}; };
1037 binary_forward<T, T>(mean, std::forward<Args>(args)...);
1038 break;
1039 }
1040 case OPERATION::MIN: {
1041 auto min = [](const T &a, const T &b) { return std::min(a, b); };
1042 binary_forward<T, T>(min, std::forward<Args>(args)...);
1043 break;
1044 }
1045 case OPERATION::MOD: {
1046 auto mod = [] (const T &a, const T &b) { return static_cast<T>(_mod(int(a), int(b))); };
1047 binary_forward<T, T>(mod, std::forward<Args>(args)...);
1048 break;
1049 }
1050 case OPERATION::FMOD: {
1051 auto fmod = [](const T &a, const T &b) { return std::fmod(a, b); };
1052 binary_forward<T, T>(fmod, std::forward<Args>(args)...);
1053 break;
1054 }
1055 case OPERATION::PROD: {
1056 auto prod = [](const T &a, const T &b) { return a * b; };
1057 binary_forward<T, T>(prod, std::forward<Args>(args)...);
1058 break;
1059 }
1060 case OPERATION::SUB: {
1061 auto sub = [](const T &a, const T &b) { return a - b; };
1062 binary_forward<T, T>(sub, std::forward<Args>(args)...);
1063 break;
1064 }
1065 case OPERATION::ADD:
1066 case OPERATION::SUM: {
1067 auto sum = [](const T &a, const T &b) { return a + b; };
1068 binary_forward<T, T>(sum, std::forward<Args>(args)...);
1069 break;
1070 }
1071 case OPERATION::DIV: {
1072 auto div = [](const T &a, const T &b) { return a / b; };
1073 binary_forward<T, T>(div, std::forward<Args>(args)...);
1074 break;
1075 }
1076 case OPERATION::BITWISE_AND: {
1077 auto band = [](const T &a, const T &b) { return (T)(a & b); };
1078 binary_forward<T, T>(band, std::forward<Args>(args)...);
1079 break;
1080 }
1081 case OPERATION::BITWISE_OR: {
1082 auto bor = [](const T &a, const T &b) { return (T)(a | b); };
1083 binary_forward<T, T>(bor, std::forward<Args>(args)...);
1084 break;
1085 }
1086 case OPERATION::BITWISE_XOR: {
1087 auto bxor = [](const T &a, const T &b) { return (T)(a ^ b); };
1088 binary_forward<T, T>(bxor, std::forward<Args>(args)...);
1089 break;
1090 }
1091 default: CV_Error(Error::StsBadArg, "Unsupported operation");
1092 }
1093 } else if (ninputs == 3 && op == OPERATION::WHERE) {
1094 auto where = [](const T &a, const T &b, const T &c) { return a ? b : c; };
1095 ternary_forward<bool, T, T, T>(where, std::forward<Args>(args)...);
1096 } else {
1097 switch (op)
1098 {
1099 case OPERATION::MAX: {
1100 auto max = [](const T &a, const T &b) { return std::max(a, b); };
1101 nary_forward<T>(max, T{1}, std::forward<Args>(args)...);
1102 break;
1103 }
1104 case OPERATION::MEAN: {
1105 auto sum = [](const T &a, const T &b) { return a + b; };
1106 nary_forward<T>(sum, T{1} / ninputs, std::forward<Args>(args)...);
1107 break;
