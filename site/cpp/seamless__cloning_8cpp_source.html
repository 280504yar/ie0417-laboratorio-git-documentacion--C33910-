51static Mat checkMask(InputArray mask, Size size)
52{
53 Mat gray;
54 if (mask.channels() == 3 || mask.channels() == 4)
55 cvtColor(mask, gray, COLOR_BGRA2GRAY);
56 else
57 {
58 if (mask.empty())
59 gray = Mat(size.height, size.width, CV_8UC1, Scalar(255));
60 else
61 return mask.getMat();
62 }
63
64 return gray;
65}

67void cv::seamlessClone(InputArray _src, InputArray _dst, InputArray _mask, Point p, OutputArray _blend, int flags)
68{
69 CV_INSTRUMENT_REGION();
70 CV_Assert(!_src.empty());
71 CV_Assert(!_dst.empty());
72
73 const Mat src = _src.getMat();
74 const Mat dest = _dst.getMat();
75
76 Mat mask = checkMask(_mask, src.size());
77 dest.copyTo(_blend);
78 Mat blend = _blend.getMat();
79
80 Mat mask_inner = mask(Rect(1, 1, mask.cols - 2, mask.rows - 2));
81 copyMakeBorder(mask_inner, mask, 1, 1, 1, 1, BORDER_ISOLATED | BORDER_CONSTANT, Scalar(0));
82
83 Rect roi_s = boundingRect(mask);
84 if (roi_s.empty()) return;
85
86 int l_from_center = p.x - roi_s.width / 2;
87 int t_from_center = p.y - roi_s.height / 2;
88
89 if (flags >= NORMAL_CLONE_WIDE)
90 {
91 l_from_center = p.x - (mask.cols / 2 - roi_s.x);
92 t_from_center = p.y - (mask.rows / 2 - roi_s.y);
93 }
94
95 Rect roi_d(l_from_center, t_from_center, roi_s.width, roi_s.height);
96 Mat destinationROI = dest(roi_d);
97 Mat sourceROI = Mat::zeros(roi_s.height, roi_s.width, src.type());
98 src(roi_s).copyTo(sourceROI,mask(roi_s));
99
100 Mat maskROI = mask(roi_s);
101 Mat recoveredROI = blend(roi_d);
102
103 Cloning obj;
104 obj.normalClone(destinationROI,sourceROI,maskROI,recoveredROI,flags);
105}
