9void reprojectImageTo3D( InputArray _disparity, OutputArray __3dImage,
10 InputArray _Qmat, bool handleMissingValues, int dtype )
11{
12 CV_INSTRUMENT_REGION();
13
14 Mat disparity = _disparity.getMat(), Q = _Qmat.getMat();
15 int stype = disparity.type();
16
17 CV_Assert( stype == CV_8UC1 || stype == CV_16SC1 ||
18 stype == CV_32SC1 || stype == CV_32FC1 );
19 CV_Assert( Q.size() == Size(4,4) );
20
21 if( dtype >= 0 )
22 dtype = CV_MAKETYPE(CV_MAT_DEPTH(dtype), 3);
23
24 if( __3dImage.fixedType() )
25 {
26 int dtype_ = __3dImage.type();
27 CV_Assert( dtype == -1 || dtype == dtype_ );
28 dtype = dtype_;
29 }
30
31 if( dtype < 0 )
32 dtype = CV_32FC3;
33 else
34 CV_Assert( dtype == CV_16SC3 || dtype == CV_32SC3 || dtype == CV_32FC3 );
35
36 __3dImage.create(disparity.size(), dtype);
37 Mat _3dImage = __3dImage.getMat();
38
39 const float bigZ = 10000.f;
40 Matx44d _Q;
41 Q.convertTo(_Q, CV_64F);
42
43 int x, cols = disparity.cols;
44 CV_Assert( cols >= 0 );
45
46 std::vector<float> _sbuf(cols);
47 std::vector<Vec3f> _dbuf(cols);
48 float* sbuf = &_sbuf[0];
49 Vec3f* dbuf = &_dbuf[0];
50 double minDisparity = FLT_MAX;
51
52 // NOTE: here we quietly assume that at least one pixel in the disparity map is not defined.
53 // and we set the corresponding Z's to some fixed big value.
54 if( handleMissingValues )
55 cv::minMaxIdx( disparity, &minDisparity, 0, 0, 0 );
56
57 for( int y = 0; y < disparity.rows; y++ )
58 {
59 float* sptr = sbuf;
60 Vec3f* dptr = dbuf;
61
62 if( stype == CV_8UC1 )
63 {
64 const uchar* sptr0 = disparity.ptr<uchar>(y);
65 for( x = 0; x < cols; x++ )
66 sptr[x] = (float)sptr0[x];
67 }
68 else if( stype == CV_16SC1 )
69 {
70 const short* sptr0 = disparity.ptr<short>(y);
71 for( x = 0; x < cols; x++ )
72 sptr[x] = (float)sptr0[x];
73 }
74 else if( stype == CV_32SC1 )
75 {
76 const int* sptr0 = disparity.ptr<int>(y);
77 for( x = 0; x < cols; x++ )
78 sptr[x] = (float)sptr0[x];
79 }
80 else
81 sptr = disparity.ptr<float>(y);
82
83 if( dtype == CV_32FC3 )
84 dptr = _3dImage.ptr<Vec3f>(y);
85
86 for( x = 0; x < cols; x++)
87 {
88 double d = sptr[x];
89 Vec4d homg_pt = _Q*Vec4d(x, y, d, 1.0);
90 dptr[x] = Vec3d(homg_pt.val);
91 dptr[x] /= homg_pt[3];
92
93 if( fabs(d-minDisparity) <= FLT_EPSILON )
94 dptr[x][2] = bigZ;
95 }
96
97 if( dtype == CV_16SC3 )
98 {
99 Vec3s* dptr0 = _3dImage.ptr<Vec3s>(y);
100 for( x = 0; x < cols; x++ )
101 {
102 dptr0[x] = dptr[x];
103 }
104 }
105 else if( dtype == CV_32SC3 )
106 {
107 Vec3i* dptr0 = _3dImage.ptr<Vec3i>(y);
108 for( x = 0; x < cols; x++ )
109 {
110 dptr0[x] = dptr[x];
111 }
112 }
113 }
114}

116void stereoRectify( InputArray _cameraMatrix1, InputArray _distCoeffs1,
117 InputArray _cameraMatrix2, InputArray _distCoeffs2,
118 Size imageSize, InputArray R, InputArray T,
119 OutputArray _R1, OutputArray _R2,
120 OutputArray _P1, OutputArray _P2,
121 OutputArray _Qmat, int flags,
122 double alpha, Size newImgSize,
123 Rect* roi1, Rect* roi2 )
124{
125 Mat matR = Mat_<double>(R.getMat()), matT = Mat_<double>(T.getMat());
126
127 Mat om, r_r;
128 Mat Z = Mat::zeros(3, 1, CV_64F);
129 double nx = imageSize.width, ny = imageSize.height;
130
131 if( matR.rows == 3 && matR.cols == 3 )
132 Rodrigues(matR, om); // get vector rotation
133 else
134 matR.copyTo(om);
135 om *= -0.5; // get average rotation
136 Rodrigues(om, r_r);
137 Mat t = r_r * matT; // rotate cameras to same orientation by averaging
138
139 int idx = fabs(t.at<double>(0)) > fabs(t.at<double>(1)) ? 0 : 1;
140 double c = t.at<double>(idx), nt = norm(t, NORM_L2);
141 double _uu[3]={0, 0, 0};
142 _uu[idx] = c > 0 ? 1 : -1;
143
144 CV_Assert(nt > 0.0);
145
146 // calculate global Z rotation
147 Mat ww = t.cross(Mat(3, 1, CV_64F, _uu)), wR;
148 double nw = norm(ww, NORM_L2);
149 if (nw > 0.0)
150 ww *= std::acos(fabs(c)/nt)/nw;
151 Rodrigues(ww, wR);
152
153 Mat Ri;
154 // apply to both views
155 gemm(wR, r_r, 1, Mat(), 0, Ri, GEMM_2_T);
156 Ri.copyTo(_R1);
157 gemm(wR, r_r, 1, Mat(), 0, Ri, 0);
158 Ri.copyTo(_R2);
159 t = Ri * matT;
160
161 // calculate projection/camera matrices
162 // these contain the relevant rectified image internal params (fx, fy=fx, cx, cy)
163 Point2d cc_new[2]={};
164
165 newImgSize = newImgSize.width * newImgSize.height != 0 ? newImgSize : imageSize;
166 const double ratio_x = (double)newImgSize.width / imageSize.width / 2;
167 const double ratio_y = (double)newImgSize.height / imageSize.height / 2;
168 const double ratio = idx == 1 ? ratio_x : ratio_y;
169
170 Mat cameraMatrix1 = Mat_<double>(_cameraMatrix1.getMat());
171 Mat cameraMatrix2 = Mat_<double>(_cameraMatrix2.getMat());
172 Mat distCoeffs1, distCoeffs2;
173 if (!_distCoeffs1.empty())
174 distCoeffs1 = Mat_<double>(_distCoeffs1.getMat());
175 if (!_distCoeffs2.empty())
176 distCoeffs2 = Mat_<double>(_distCoeffs2.getMat());
177
178 double fc_new = (cameraMatrix1.at<double>(idx ^ 1, idx ^ 1) + cameraMatrix2.at<double>(idx ^ 1, idx ^ 1)) * ratio;
179
180 for( int k = 0; k < 2; k++ )
181 {
182 const Mat& A = k == 0 ? cameraMatrix1 : cameraMatrix2;
183 const Mat& Dk = k == 0 ? distCoeffs1 : distCoeffs2;
184 Point2f _pts[4] = {};
185 Point3f _pts_3[4] = {};
186 Mat pts(1, 4, CV_32FC2, _pts);
187 Mat pts_3(1, 4, CV_32FC3, _pts_3);
188
189 for( int i = 0; i < 4; i++ )
190 {
191 int j = (i<2) ? 0 : 1;
192 _pts[i].x = (float)((i % 2)*(nx-1));
193 _pts[i].y = (float)(j*(ny-1));
194 }
195 undistortPoints(pts, pts, A, Dk, Mat(), Mat());
196 convertPointsToHomogeneous(pts, pts_3);
197
198 // Change the camera matrix to have cc=[0,0] and fc = fc_new
199 double _a_tmp[3][3] = {{fc_new, 0, 0}, {0, fc_new, 0}, {0, 0, 1}};
200 Mat A_tmp(3, 3, CV_64F, _a_tmp);
201 projectPoints(pts_3, (k == 0 ? _R1 : _R2), Z, A_tmp, Mat(), pts);
202 Scalar avg = mean(pts);
203 cc_new[k].x = (nx-1)/2 - avg.val[0];
204 cc_new[k].y = (ny-1)/2 - avg.val[1];
205 }
206
207 // vertical focal length must be the same for both images to keep the epipolar constraint
208 // (for horizontal epipolar lines -- TBD: check for vertical epipolar lines)
209 // use fy for fx also, for simplicity
210
211 // For simplicity, set the principal points for both cameras to be the average
212 // of the two principal points (either one of or both x- and y- coordinates)
213 if( flags & STEREO_ZERO_DISPARITY )
214 {
215 cc_new[0].x = cc_new[1].x = (cc_new[0].x + cc_new[1].x)*0.5;
216 cc_new[0].y = cc_new[1].y = (cc_new[0].y + cc_new[1].y)*0.5;
217 }
218 else if( idx == 0 ) // horizontal stereo
219 cc_new[0].y = cc_new[1].y = (cc_new[0].y + cc_new[1].y)*0.5;
220 else // vertical stereo
221 cc_new[0].x = cc_new[1].x = (cc_new[0].x + cc_new[1].x)*0.5;
222
223 double t_idx = t.at<double>(idx);
224
225 Mat pp = Mat::zeros(3, 4, CV_64F);
226 pp.at<double>(0, 0) = pp.at<double>(1, 1) = fc_new;
227 pp.at<double>(0, 2) = cc_new[0].x;
228 pp.at<double>(1, 2) = cc_new[0].y;
229 pp.at<double>(2, 2) = 1.;
230 pp.copyTo(_P1);
231
232 pp.at<double>(0, 2) = cc_new[1].x;
233 pp.at<double>(1, 2) = cc_new[1].y;
234 pp.at<double>(idx, 3) = t_idx*fc_new; // baseline * focal length
235 pp.copyTo(_P2);
236
237 alpha = MIN(alpha, 1.);
238
239 cv::Rect_<double> inner1, inner2, outer1, outer2;
240 getUndistortRectangles(cameraMatrix1, distCoeffs1, _R1, _P1, imageSize, inner1, outer1);
241 getUndistortRectangles(cameraMatrix2, distCoeffs2, _R2, _P2, imageSize, inner2, outer2);
242
243 {
244 newImgSize = newImgSize.width*newImgSize.height != 0 ? newImgSize : imageSize;
245 double cx1_0 = cc_new[0].x;
246 double cy1_0 = cc_new[0].y;
247 double cx2_0 = cc_new[1].x;
248 double cy2_0 = cc_new[1].y;
249 double cx1 = newImgSize.width*cx1_0/imageSize.width;
250 double cy1 = newImgSize.height*cy1_0/imageSize.height;
251 double cx2 = newImgSize.width*cx2_0/imageSize.width;
252 double cy2 = newImgSize.height*cy2_0/imageSize.height;
253 double s = 1.;
254
255 if( alpha >= 0 )
256 {
257 double s0 = std::max(std::max(std::max((double)cx1/(cx1_0 - inner1.x), (double)cy1/(cy1_0 - inner1.y)),
258 (double)(newImgSize.width - 1 - cx1)/(inner1.x + inner1.width - cx1_0)),
259 (double)(newImgSize.height - 1 - cy1)/(inner1.y + inner1.height - cy1_0));
260 s0 = std::max(std::max(std::max(std::max((double)cx2/(cx2_0 - inner2.x), (double)cy2/(cy2_0 - inner2.y)),
261 (double)(newImgSize.width - 1 - cx2)/(inner2.x + inner2.width - cx2_0)),
262 (double)(newImgSize.height - 1 - cy2)/(inner2.y + inner2.height - cy2_0)),
263 s0);
264
265 double s1 = std::min(std::min(std::min((double)cx1/(cx1_0 - outer1.x), (double)cy1/(cy1_0 - outer1.y)),
266 (double)(newImgSize.width - 1 - cx1)/(outer1.x + outer1.width - cx1_0)),
267 (double)(newImgSize.height - 1 - cy1)/(outer1.y + outer1.height - cy1_0));
268 s1 = std::min(std::min(std::min(std::min((double)cx2/(cx2_0 - outer2.x), (double)cy2/(cy2_0 - outer2.y)),
269 (double)(newImgSize.width - 1 - cx2)/(outer2.x + outer2.width - cx2_0)),
270 (double)(newImgSize.height - 1 - cy2)/(outer2.y + outer2.height - cy2_0)),
271 s1);
272
273 s = s0*(1 - alpha) + s1*alpha;
274 }
275
276 fc_new *= s;
277 cc_new[0] = Point2d(cx1, cy1);
278 cc_new[1] = Point2d(cx2, cy2);
279
280 pp.at<double>(0, 0) = pp.at<double>(1, 1) = fc_new;
281 pp.at<double>(0, 2) = cx2;
282 pp.at<double>(1, 2) = cy2;
283 pp.at<double>(idx, 3) *= s;
284 pp.copyTo(_P2);
285
286 pp.at<double>(0, 2) = cx1;
287 pp.at<double>(1, 2) = cy1;
288 pp.at<double>(idx, 3) = 0.;
289 pp.copyTo(_P1);
290
291 if(roi1)
292 {
293 *roi1 =
294 cv::Rect(cvCeil((inner1.x - cx1_0)*s + cx1),
295 cvCeil((inner1.y - cy1_0)*s + cy1),
296 cvFloor(inner1.width*s), cvFloor(inner1.height*s))
297 & cv::Rect(0, 0, newImgSize.width, newImgSize.height)
298 ;
299 }
300
301 if(roi2)
302 {
303 *roi2 =
304 cv::Rect(cvCeil((inner2.x - cx2_0)*s + cx2),
305 cvCeil((inner2.y - cy2_0)*s + cy2),
306 cvFloor(inner2.width*s), cvFloor(inner2.height*s))
307 & cv::Rect(0, 0, newImgSize.width, newImgSize.height)
308 ;
309 }
310 }
311
312 if( _Qmat.needed() )
313 {
314 double q[] =
315 {
316 1, 0, 0, -cc_new[0].x,
317 0, 1, 0, -cc_new[0].y,
318 0, 0, 0, fc_new,
319 0, 0, -1./t_idx,
320 (idx == 0 ? cc_new[0].x - cc_new[1].x : cc_new[0].y - cc_new[1].y)/t_idx
321 };
322 Mat Q(4, 4, CV_64F, q);
323 Q.copyTo(_Qmat);
324 }
325}

327/*
328CV_IMPL int cvStereoRectifyUncalibrated(
329 const CvMat* _points1, const CvMat* _points2,
330 const CvMat* F0, CvSize imgSize,
331 CvMat* _H1, CvMat* _H2, double threshold )
332*/
333bool stereoRectifyUncalibrated( InputArray _points1, InputArray _points2,
334 InputArray _Fmat, Size imgSize,
335 OutputArray _Hmat1, OutputArray _Hmat2, double threshold )
336{
337 Mat points1 = _points1.getMat(), points2 = _points2.getMat();
338 CV_Assert( points1.size() == points2.size() );
339
340 int npoints = points1.checkVector(2);
341 CV_Assert(npoints > 0);
342
343 Mat _m1, _m2;
344
345 points1.convertTo(_m1, CV_64F);
346 points2.convertTo(_m2, CV_64F);
347 _m1 = _m1.reshape(2, 1);
348 _m2 = _m2.reshape(2, 1);
349
350 Mat F0 = _Fmat.getMat(), F, Wdiag, U, Vt;
351 F0.convertTo(F, CV_64F);
352
353 SVDecomp(F, Wdiag, U, Vt, 0);
354 Wdiag.at<double>(2) = 0.;
355 Mat W = Mat::diag(Wdiag), UW;
356 gemm(U, W, 1, Mat(), 0, UW);
357 gemm(UW, Vt, 1, Mat(), 0, F);
358
359 double cx = cvRound( (imgSize.width-1)*0.5 );
360 double cy = cvRound( (imgSize.height-1)*0.5 );
361
362 if( threshold > 0 )
363 {
364 Mat _lines1, _lines2;
365 computeCorrespondEpilines(_m1, 1, F, _lines1);
366 computeCorrespondEpilines(_m2, 2, F, _lines2);
367 CV_Assert(_m1.isContinuous() && _m2.isContinuous() &&
368 _lines1.isContinuous() && _lines2.isContinuous());
369 Point2d* m1 = (Point2d*)_m1.data;
370 Point2d* m2 = (Point2d*)_m2.data;
371 Point3d* lines1 = (Point3d*)_lines1.data;
372 Point3d* lines2 = (Point3d*)_lines2.data;
373
374 // measure distance from points to the corresponding epilines, mark outliers
375 int i, j;
376 for( i = j = 0; i < npoints; i++ )
377 {
378 if( fabs(m1[i].x*lines2[i].x +
379 m1[i].y*lines2[i].y +
380 lines2[i].z) <= threshold &&
381 fabs(m2[i].x*lines1[i].x +
382 m2[i].y*lines1[i].y +
383 lines1[i].z) <= threshold )
384 {
385 if( j < i )
386 {
387 m1[j] = m1[i];
388 m2[j] = m2[i];
389 }
390 j++;
391 }
392 }
393
394 npoints = j;
395 if( npoints == 0 )
396 return false;
397 _m1.cols = _m2.cols = npoints;
398 }
399
400 Mat E2 = U.col(2).clone();
401 if (E2.at<double>(2) < 0)
402 E2 *= -1.0;
403
404 double t[] =
405 {
406 1, 0, -cx,
407 0, 1, -cy,
408 0, 0, 1
409 };
410 Mat T(3, 3, CV_64F, t);
411 E2 = T*E2;
412
413 double* e2 = (double*)E2.data;
414 int mirror = e2[0] < 0;
415 double d = std::sqrt(e2[0]*e2[0] + e2[1]*e2[1]);
416 d = MAX(d, DBL_EPSILON);
417 double alpha = e2[0]/d;
418 double beta = e2[1]/d;
419 double r[] =
420 {
421 alpha, beta, 0,
422 -beta, alpha, 0,
423 0, 0, 1
424 };
425 Mat R(3, 3, CV_64F, r);
426 T = R*T;
427 E2 = R*E2;
428 double invf = fabs(e2[2]) < 1e-6*fabs(e2[0]) ? 0 : -e2[2]/e2[0];
429 double k[] =
430 {
431 1, 0, 0,
432 0, 1, 0,
433 invf, 0, 1
434 };
435 Mat K(3, 3, CV_64F, k);
436 Mat H2 = K*T;
437 E2 = K*E2;
438
439 double it[] =
440 {
441 1, 0, cx,
442 0, 1, cy,
443 0, 0, 1
444 };
445 Mat iT( 3, 3, CV_64F, it );
446 H2 = iT*H2;
447
448 U.col(2).copyTo(E2);
449 if (E2.at<double>(2) < 0)
450 E2 *= -1.0;
451
452 double e2_x[] =
453 {
454 0, -e2[2], e2[1],
455 e2[2], 0, -e2[0],
456 -e2[1], e2[0], 0
457 };
458 double e2_111[] =
459 {
460 e2[0], e2[0], e2[0],
461 e2[1], e2[1], e2[1],
462 e2[2], e2[2], e2[2],
463 };
464 Mat E2_x(3, 3, CV_64F, e2_x);
465 Mat E2_111(3, 3, CV_64F, e2_111);
466 Mat H0 = E2_x*F + E2_111;
467 H0 = H2*H0;
468 Mat E1(3, 1, CV_64F, (double*)Vt.data+6);
469 E1 = H0*E1;
470
471 perspectiveTransform( _m1, _m1, H0 );
472 perspectiveTransform( _m2, _m2, H2 );
473 Mat A, X;
474 convertPointsToHomogeneous(_m1, A, CV_64F);
475 A = A.reshape(1, npoints);
476 Mat BxBy = _m2.reshape(1, npoints);
477 Mat B = BxBy.col(0);
478 solve(A, B, X, DECOMP_SVD);
479 CV_Assert(X.isContinuous());
480 double* x = X.ptr<double>();
481
482 double ha[] =
483 {
484 x[0], x[1], x[2],
485 0, 1, 0,
486 0, 0, 1
487 };
488 Mat Ha(3, 3, CV_64F, ha);
489 Mat H1 = Ha*H0;
490 perspectiveTransform( _m1, _m1, Ha );
491
492 if( mirror )
493 {
494 double mm[] = { -1, 0, cx*2, 0, -1, cy*2, 0, 0, 1 };
495 Mat MM(3, 3, CV_64F, mm);
496 H1 = MM*H1;
497 H2 = MM*H2;
498 }
499
500 H1.copyTo(_Hmat1);
501 H2.copyTo(_Hmat2);
502 return true;
503}

506static void adjust3rdMatrix(InputArrayOfArrays _imgpt1_0,
507 InputArrayOfArrays _imgpt3_0,
508 const Mat& cameraMatrix1, const Mat& distCoeffs1,
509 const Mat& cameraMatrix3, const Mat& distCoeffs3,
510 const Mat& R1, const Mat& R3, const Mat& P1, Mat& P3 )
511{
512 size_t n1 = _imgpt1_0.total(), n3 = _imgpt3_0.total();
513 std::vector<Point2f> imgpt1, imgpt3;
514
515 for( int i = 0; i < (int)std::min(n1, n3); i++ )
516 {
517 Mat pt1 = _imgpt1_0.getMat(i), pt3 = _imgpt3_0.getMat(i);
518 int ni1 = pt1.checkVector(2, CV_32F), ni3 = pt3.checkVector(2, CV_32F);
519 CV_Assert( ni1 > 0 && ni1 == ni3 );
520 const Point2f* pt1data = pt1.ptr<Point2f>();
521 const Point2f* pt3data = pt3.ptr<Point2f>();
522 std::copy(pt1data, pt1data + ni1, std::back_inserter(imgpt1));
523 std::copy(pt3data, pt3data + ni3, std::back_inserter(imgpt3));
524 }
525
526 undistortPoints(imgpt1, imgpt1, cameraMatrix1, distCoeffs1, R1, P1);
527 undistortPoints(imgpt3, imgpt3, cameraMatrix3, distCoeffs3, R3, P3);
528
529 double y1_ = 0, y2_ = 0, y1y1_ = 0, y1y2_ = 0;
530 size_t n = imgpt1.size();
531 CV_DbgAssert(n > 0);
532
533 for( size_t i = 0; i < n; i++ )
534 {
535 double y1 = imgpt3[i].y, y2 = imgpt1[i].y;
536
537 y1_ += y1; y2_ += y2;
538 y1y1_ += y1*y1; y1y2_ += y1*y2;
539 }
540
541 y1_ /= n;
542 y2_ /= n;
543 y1y1_ /= n;
544 y1y2_ /= n;
545
546 double a = (y1y2_ - y1_*y2_)/(y1y1_ - y1_*y1_);
547 double b = y2_ - a*y1_;
548
549 P3.at<double>(0,0) *= a;
550 P3.at<double>(1,1) *= a;
551 P3.at<double>(0,2) = P3.at<double>(0,2)*a;
552 P3.at<double>(1,2) = P3.at<double>(1,2)*a + b;
553 P3.at<double>(0,3) *= a;
554 P3.at<double>(1,3) *= a;
555}

557float rectify3Collinear( InputArray _cameraMatrix1, InputArray _distCoeffs1,
558 InputArray _cameraMatrix2, InputArray _distCoeffs2,
559 InputArray _cameraMatrix3, InputArray _distCoeffs3,
560 InputArrayOfArrays _imgpt1,
561 InputArrayOfArrays _imgpt3,
562 Size imageSize, InputArray _Rmat12, InputArray _Tmat12,
563 InputArray _Rmat13, InputArray _Tmat13,
564 OutputArray _Rmat1, OutputArray _Rmat2, OutputArray _Rmat3,
565 OutputArray _Pmat1, OutputArray _Pmat2, OutputArray _Pmat3,
566 OutputArray _Qmat,
567 double alpha, Size newImgSize,
568 Rect* roi1, Rect* roi2, int flags )
569{
570 // first, rectify the 1-2 stereo pair
571 stereoRectify( _cameraMatrix1, _distCoeffs1, _cameraMatrix2, _distCoeffs2,
572 imageSize, _Rmat12, _Tmat12, _Rmat1, _Rmat2, _Pmat1, _Pmat2, _Qmat,
573 flags, alpha, newImgSize, roi1, roi2 );
574
575 Mat R12 = _Rmat12.getMat(), R13 = _Rmat13.getMat(), T12 = _Tmat12.getMat(), T13 = _Tmat13.getMat();
576
577 _Rmat3.create(3, 3, CV_64F);
578 _Pmat3.create(3, 4, CV_64F);
579
580 Mat P1 = _Pmat1.getMat(), P2 = _Pmat2.getMat();
581 Mat R3 = _Rmat3.getMat(), P3 = _Pmat3.getMat();
582
583 // recompute rectification transforms for cameras 1 & 2.
584 Mat om, r_r, r_r13;
585
586 if( R13.size() != Size(3,3) )
587 Rodrigues(R13, r_r13);
588 else
589 R13.copyTo(r_r13);
590
591 if( R12.size() == Size(3,3) )
592 Rodrigues(R12, om);
593 else
594 R12.copyTo(om);
595
596 om *= -0.5;
597 Rodrigues(om, r_r); // rotate cameras to same orientation by averaging
598 Mat_<double> t12 = r_r * T12;
599
600 int idx = fabs(t12(0,0)) > fabs(t12(1,0)) ? 0 : 1;
601 double c = t12(idx,0), nt = norm(t12, NORM_L2);
602 CV_Assert(fabs(nt) > 0);
603 Mat_<double> uu = Mat_<double>::zeros(3,1);
604 uu(idx, 0) = c > 0 ? 1 : -1;
605
606 // calculate global Z rotation
607 Mat_<double> ww = t12.cross(uu), wR;
608 double nw = norm(ww, NORM_L2);
609 CV_Assert(fabs(nw) > 0);
610 ww *= std::acos(fabs(c)/nt)/nw;
611 Rodrigues(ww, wR);
612
613 // now rotate camera 3 to make its optical axis parallel to cameras 1 and 2.
614 R3 = wR*r_r.t()*r_r13.t();
615 Mat_<double> t13 = R3 * T13;
616
617 P2.copyTo(P3);
618 Mat t = P3.col(3);
619 t13.copyTo(t);
620 P3.at<double>(0,3) *= P3.at<double>(0,0);
621 P3.at<double>(1,3) *= P3.at<double>(1,1);
622
623 if( !_imgpt1.empty() && !_imgpt3.empty() )
624 adjust3rdMatrix(_imgpt1, _imgpt3, _cameraMatrix1.getMat(), _distCoeffs1.getMat(),
625 _cameraMatrix3.getMat(), _distCoeffs3.getMat(), _Rmat1.getMat(), R3, P1, P3);
626
627 return (float)((P3.at<double>(idx,3)/P3.at<double>(idx,idx))/
628 (P2.at<double>(idx,3)/P2.at<double>(idx,idx)));
629}

631void cv::fisheye::stereoRectify( InputArray K1, InputArray D1, InputArray K2, InputArray D2, const Size& imageSize,
632 InputArray _R, InputArray _tvec, OutputArray R1, OutputArray R2, OutputArray P1, OutputArray P2,
633 OutputArray Q, int flags, const Size& newImageSize, double balance, double fov_scale)
634{
635 CV_INSTRUMENT_REGION();
636
637 CV_Assert((_R.size() == Size(3, 3) || _R.total() * _R.channels() == 3) && (_R.depth() == CV_32F || _R.depth() == CV_64F));
638 CV_Assert(_tvec.total() * _tvec.channels() == 3 && (_tvec.depth() == CV_32F || _tvec.depth() == CV_64F));
639
640
641 Mat aaa = _tvec.getMat().reshape(3, 1);
642
643 Vec3d rvec; // Rodrigues vector
644 if (_R.size() == Size(3, 3))
645 {
646 Matx33d rmat;
647 _R.getMat().convertTo(rmat, CV_64F);
648 rvec = Affine3d(rmat).rvec();
649 }
650 else if (_R.total() * _R.channels() == 3)
651 _R.getMat().convertTo(rvec, CV_64F);
652
653 Vec3d tvec;
654 _tvec.getMat().convertTo(tvec, CV_64F);
655
656 // rectification algorithm
657 rvec *= -0.5; // get average rotation
658
659 Matx33d r_r;
660 Rodrigues(rvec, r_r); // rotate cameras to same orientation by averaging
661
662 Vec3d t = r_r * tvec;
663 Vec3d uu(t[0] > 0 ? 1 : -1, 0, 0);
664
665 // calculate global Z rotation
666 Vec3d ww = t.cross(uu);
667 double nw = norm(ww);
668 if (nw > 0.0)
669 ww *= std::acos(fabs(t[0])/cv::norm(t))/nw;
670
671 Matx33d wr;
672 Rodrigues(ww, wr);
673
674 // apply to both views
675 Matx33d ri1 = wr * r_r.t();
676 Mat(ri1, false).convertTo(R1, R1.empty() ? CV_64F : R1.type());
677 Matx33d ri2 = wr * r_r;
678 Mat(ri2, false).convertTo(R2, R2.empty() ? CV_64F : R2.type());
679 Vec3d tnew = ri2 * tvec;
680
681 // calculate projection/camera matrices. these contain the relevant rectified image internal params (fx, fy=fx, cx, cy)
682 Matx33d newK1, newK2;
683 fisheye::estimateNewCameraMatrixForUndistortRectify(K1, D1, imageSize, R1, newK1, balance, newImageSize, fov_scale);
684 fisheye::estimateNewCameraMatrixForUndistortRectify(K2, D2, imageSize, R2, newK2, balance, newImageSize, fov_scale);
685
686 double fc_new = std::min(newK1(1,1), newK2(1,1));
687 Point2d cc_new[2] = { Vec2d(newK1(0, 2), newK1(1, 2)), Vec2d(newK2(0, 2), newK2(1, 2)) };
688
689 // Vertical focal length must be the same for both images to keep the epipolar constraint use fy for fx also.
690 // For simplicity, set the principal points for both cameras to be the average
691 // of the two principal points (either one of or both x- and y- coordinates)
692 if( flags & STEREO_ZERO_DISPARITY )
693 cc_new[0] = cc_new[1] = (cc_new[0] + cc_new[1]) * 0.5;
694 else
695 cc_new[0].y = cc_new[1].y = (cc_new[0].y + cc_new[1].y)*0.5;
696
697 Mat(Matx34d(fc_new, 0, cc_new[0].x, 0,
698 0, fc_new, cc_new[0].y, 0,
699 0, 0, 1, 0), false).convertTo(P1, P1.empty() ? CV_64F : P1.type());
700
701 Mat(Matx34d(fc_new, 0, cc_new[1].x, tnew[0]*fc_new, // baseline * focal length;,
702 0, fc_new, cc_new[1].y, 0,
703 0, 0, 1, 0), false).convertTo(P2, P2.empty() ? CV_64F : P2.type());
704
705 if (Q.needed())
706 Mat(Matx44d(1, 0, 0, -cc_new[0].x,
707 0, 1, 0, -cc_new[0].y,
708 0, 0, 0, fc_new,
709 0, 0, -1./tnew[0], (cc_new[0].x - cc_new[1].x)/tnew[0]), false).convertTo(Q, Q.empty() ? CV_64F : Q.depth());
710}
