31 template <class T>
32 struct RegionConfiguration {
33 /* The image is divided into (H, W) cells.
34 *
35 * Each cell is interested in exactly one object and predicts `boxes_per_cell` bounding boxes
36 * for that object.
37 *
38 * Each bounding box contains:
39 * - 4 box coordinates
40 * - objectness confidence score
41 * - `classes` number of class scores
42 *
43 * The object score is reduced to a probability using sigmoid and the class scores are reduced to
44 * probabilities by either applying sigmoid or softmax (which is a configuration option).
45 *
46 * object_prob = sigmoid(object_score)
47 * conditional_class_prob = sigmoid, softmax across all classes
48 *
49 * actual class probability = conditional_class_prob * object_prob
50 */
51 std::size_t classes, boxes_per_cell;
52 std::size_t width_norm, height_norm;
53 T scale_x_y;
54
55 /* method for reducing class scores to probabilities */
56 SquashMethod squash_method;
57
58 /* prob cutoffs below which the prediction is nulled */
59 T object_prob_cutoff;
60 T class_prob_cutoff;
61
62 T nms_iou_threshold;
63 bool new_coords;
64 };

94 void forward(
95 const std::vector<cv::Ptr<BackendWrapper>>& inputs,
96 const std::vector<cv::Ptr<BackendWrapper>>& outputs,
97 csl::Workspace& workspace) override
98 {
99 CV_Assert(outputs.size() == 1);
100
101 auto input_wrapper = inputs[0].dynamicCast<wrapper_type>();
102 auto input = input_wrapper->getView();
103
104 auto output_wrapper = outputs[0].dynamicCast<wrapper_type>();
105 auto output = output_wrapper->getSpan();
106
107 auto rows = input.get_axis_size(1);
108 auto cols = input.get_axis_size(2);
109
110 auto cell_box_size = classes + 4 + 1;
111
112 /* we squash class scores into probabilities using softmax or sigmoid */
113 bool if_true_sigmoid_else_softmax = (squash_type == SquashMethod::SIGMOID);
114
115 kernels::region<T>(stream, output, input, biasTensor,
116 object_prob_cutoff, class_prob_cutoff,
117 boxes_per_cell, cell_box_size,
118 rows, cols, scale_x_y,
119 height_norm, width_norm,
120 if_true_sigmoid_else_softmax,
121 new_coords
122 );
123
124 if (nms_iou_threshold > static_cast<T>(0.0f)) {
125 auto output_mat = output_wrapper->getMutableHostMat();
126 CV_Assert(output_mat.type() == CV_32F);
127 for (int i = 0; i < input.get_axis_size(0); i++) {
128 auto sample_size = rows * cols * boxes_per_cell * cell_box_size;
129 do_nms_sort(reinterpret_cast<float*>(output_mat.data) + i * sample_size, rows * cols * boxes_per_cell, class_prob_cutoff, nms_iou_threshold);
130 }
131 }
132 }
133
134 private:
135 void do_nms_sort(float *detections, int total, float score_thresh, float nms_thresh)
136 {
137 std::vector<Rect2d> boxes(total);
138 std::vector<float> scores(total);
139
140 for (int i = 0; i < total; ++i)
141 {
142 Rect2d &b = boxes[i];
143 int box_index = i * (classes + 4 + 1);
144 b.width = detections[box_index + 2];
145 b.height = detections[box_index + 3];
146 b.x = detections[box_index + 0] - b.width / 2;
147 b.y = detections[box_index + 1] - b.height / 2;
148 }
149
150 std::vector<int> indices;
151 for (int k = 0; k < classes; ++k)
152 {
153 for (int i = 0; i < total; ++i)
154 {
155 int box_index = i * (classes + 4 + 1);
156 int class_index = box_index + 5;
157 scores[i] = detections[class_index + k];
158 detections[class_index + k] = 0;
159 }
160 NMSBoxes(boxes, scores, score_thresh, nms_thresh, indices);
161 for (int i = 0, n = indices.size(); i < n; ++i)
162 {
163 int box_index = indices[i] * (classes + 4 + 1);
164 int class_index = box_index + 5;
165 detections[class_index + k] = scores[indices[i]];
166 }
167 }
168 }

171 csl::Stream stream;

174 std::size_t classes, boxes_per_cell;

179 T object_prob_cutoff, class_prob_cutoff;

181 T nms_iou_threshold;
182 bool new_coords;
