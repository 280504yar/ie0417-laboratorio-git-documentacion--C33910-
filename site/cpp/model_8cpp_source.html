266 std::pair<int, float> classify(InputArray frame)
267 {
268 std::vector<Mat> outs;
269 processFrame(frame, outs);
270 CV_Assert(outs.size() == 1);
271
272 Mat out = outs[0].reshape(1, 1);
273
274 if(getEnableSoftmaxPostProcessing())
275 {
276 softmax(out, out);
277 }
278
279 double conf;
280 Point maxLoc;
281 cv::minMaxLoc(out, nullptr, &conf, nullptr, &maxLoc);
282 return {maxLoc.x, static_cast<float>(conf)};
283 }
284
285protected:
286 void softmax(InputArray inblob, OutputArray outblob)
287 {
288 const Mat input = inblob.getMat();
289 outblob.create(inblob.size(), inblob.type());
290
291 Mat exp;
292 const float max = *std::max_element(input.begin<float>(), input.end<float>());
293 cv::exp((input - max), exp);
294 outblob.getMat() = exp / cv::sum(exp)[0];
295 }
296
297protected:
298 bool applySoftmax = false;

349std::vector<Point2f> KeypointsModel::estimate(InputArray frame, float thresh)
350{
351
352 int frameHeight = frame.rows();
353 int frameWidth = frame.cols();
354 std::vector<Mat> outs;
355
356 impl->processFrame(frame, outs);
357 CV_Assert(outs.size() == 1);
358 Mat output = outs[0];
359
360 const int nPoints = output.size[1];
361 std::vector<Point2f> points;
362
363 // If output is a map, extract the keypoints
364 if (output.dims == 4)
365 {
366 int height = output.size[2];
367 int width = output.size[3];
368
369 // find the position of the keypoints (ignore the background)
370 for (int n=0; n < nPoints - 1; n++)
371 {
372 // Probability map of corresponding keypoint
373 Mat probMap(height, width, CV_32F, output.ptr(0, n));
374
375 Point2f p(-1, -1);
376 Point maxLoc;
377 double prob;
378 minMaxLoc(probMap, NULL, &prob, NULL, &maxLoc);
379 if (prob > thresh)
380 {
381 p = maxLoc;
382 p.x *= (float)frameWidth / width;
383 p.y *= (float)frameHeight / height;
384 }
385 points.push_back(p);
386 }
387 }
388 // Otherwise the output is a vector of keypoints and we can just return it
389 else
390 {
391 for (int n=0; n < nPoints; n++)
392 {
393 Point2f p;
394 p.x = *output.ptr<float>(0, n, 0);
395 p.y = *output.ptr<float>(0, n, 1);
396 points.push_back(p);
397 }
398 }
399 return points;
400}

407void SegmentationModel::segment(InputArray frame, OutputArray mask)
408{
409 std::vector<Mat> outs;
410 impl->processFrame(frame, outs);
411 // default output is the first one
412 if(outs.size() > 1)
413 outs.resize(1);
414 Mat score = outs[0];
415
416 const int chns = score.size[1];
417 const int rows = score.size[2];
418 const int cols = score.size[3];
419
420 mask.create(rows, cols, CV_8U);
421 Mat classIds = mask.getMat();
422 classIds.setTo(0);
423 Mat maxVal(rows, cols, CV_32F, score.data);
424
425 for (int ch = 1; ch < chns; ch++)
426 {
427 for (int row = 0; row < rows; row++)
428 {
429 const float *ptrScore = score.ptr<float>(0, ch, row);
430 uint8_t *ptrMaxCl = classIds.ptr<uint8_t>(row);
431 float *ptrMaxVal = maxVal.ptr<float>(row);
432 for (int col = 0; col < cols; col++)
433 {
434 if (ptrScore[col] > ptrMaxVal[col])
435 {
436 ptrMaxVal[col] = ptrScore[col];
437 ptrMaxCl[col] = ch;
438 }
439 }
440 }
441 }

510void DetectionModel::detect(InputArray frame, CV_OUT std::vector<int>& classIds,
511 CV_OUT std::vector<float>& confidences, CV_OUT std::vector<Rect>& boxes,
512 float confThreshold, float nmsThreshold)
513{
514 CV_Assert(impl != nullptr && impl.dynamicCast<DetectionModel_Impl>() != nullptr); // remove once default constructor is removed
515
516 std::vector<Mat> detections;
517 impl->processFrame(frame, detections);
518
519 boxes.clear();
520 confidences.clear();
521 classIds.clear();
522
523 int frameWidth = frame.cols();
524 int frameHeight = frame.rows();
525 if ((getNetwork_().getMainGraph() &&
526 getNetwork_().haveArg("im_info") &&
527 getNetwork_().argKind(getNetwork_().getArg("im_info")) == DNN_ARG_INPUT)
528 ||
529 (!getNetwork_().getMainGraph() &&
530 getNetwork_().getLayer(0)->outputNameToIndex("im_info") != -1))
531 {
532 frameWidth = impl->size.width;
533 frameHeight = impl->size.height;
534 }
535
536 std::vector<String> layerNames = getNetwork_().getLayerNames();
537 int lastLayerId = getNetwork_().getLayerId(layerNames.back());
538 Ptr<Layer> lastLayer = getNetwork_().getLayer(lastLayerId);
539
540 if (lastLayer->type == "DetectionOutput")
541 {
542 // Network produces output blob with a shape 1x1xNx7 where N is a number of
543 // detections and an every detection is a vector of values
544 // [batchId, classId, confidence, left, top, right, bottom]
545 for (int i = 0; i < detections.size(); ++i)
546 {
547 float* data = (float*)detections[i].data;
548 for (int j = 0; j < detections[i].total(); j += 7)
549 {
550 float conf = data[j + 2];
551 if (conf < confThreshold)
552 continue;
553
554 int left = data[j + 3];
555 int top = data[j + 4];
556 int right = data[j + 5];
557 int bottom = data[j + 6];
558 int width = right - left + 1;
559 int height = bottom - top + 1;
560
561 if (width <= 2 || height <= 2)
562 {
563 left = data[j + 3] * frameWidth;
564 top = data[j + 4] * frameHeight;
565 right = data[j + 5] * frameWidth;
566 bottom = data[j + 6] * frameHeight;
567 width = right - left + 1;
568 height = bottom - top + 1;
569 }
570
571 left = std::max(0, std::min(left, frameWidth - 1));
572 top = std::max(0, std::min(top, frameHeight - 1));
573 width = std::max(1, std::min(width, frameWidth - left));
574 height = std::max(1, std::min(height, frameHeight - top));
575 boxes.emplace_back(left, top, width, height);
576
577 classIds.push_back(static_cast<int>(data[j + 1]));
578 confidences.push_back(conf);
579 }
580 }
581 }
582 else if (lastLayer->type == "Region")
583 {
584 std::vector<int> predClassIds;
585 std::vector<Rect> predBoxes;
586 std::vector<float> predConfidences;
587 for (int i = 0; i < detections.size(); ++i)
588 {
589 // Network produces output blob with a shape NxC where N is a number of
590 // detected objects and C is a number of classes + 4 where the first 4
591 // numbers are [center_x, center_y, width, height]
592 float* data = (float*)detections[i].data;
593 for (int j = 0; j < detections[i].rows; ++j, data += detections[i].cols)
594 {
595
596 Mat scores = detections[i].row(j).colRange(5, detections[i].cols);
597 Point classIdPoint;
598 double conf;
599 minMaxLoc(scores, nullptr, &conf, nullptr, &classIdPoint);
600
601 if (static_cast<float>(conf) < confThreshold)
602 continue;
603
604 int centerX = data[0] * frameWidth;
605 int centerY = data[1] * frameHeight;
606 int width = data[2] * frameWidth;
607 int height = data[3] * frameHeight;
608
609 int left = std::max(0, std::min(centerX - width / 2, frameWidth - 1));
610 int top = std::max(0, std::min(centerY - height / 2, frameHeight - 1));
611 width = std::max(1, std::min(width, frameWidth - left));
612 height = std::max(1, std::min(height, frameHeight - top));
613
614 predClassIds.push_back(classIdPoint.x);
615 predConfidences.push_back(static_cast<float>(conf));
616 predBoxes.emplace_back(left, top, width, height);
617 }
618 }
619
620 if (nmsThreshold)
621 {
622 if (getNmsAcrossClasses())
623 {
624 std::vector<int> indices;
625 NMSBoxes(predBoxes, predConfidences, confThreshold, nmsThreshold, indices);
626 for (int idx : indices)
627 {
628 boxes.push_back(predBoxes[idx]);
629 confidences.push_back(predConfidences[idx]);
630 classIds.push_back(predClassIds[idx]);
631 }
632 }
633 else
634 {
635 std::map<int, std::vector<size_t> > class2indices;
636 for (size_t i = 0; i < predClassIds.size(); i++)
637 {
638 if (predConfidences[i] >= confThreshold)
639 {
640 class2indices[predClassIds[i]].push_back(i);
641 }
642 }
643 for (const auto& it : class2indices)
644 {
645 std::vector<Rect> localBoxes;
646 std::vector<float> localConfidences;
647 for (size_t idx : it.second)
648 {
649 localBoxes.push_back(predBoxes[idx]);
650 localConfidences.push_back(predConfidences[idx]);
651 }
652 std::vector<int> indices;
653 NMSBoxes(localBoxes, localConfidences, confThreshold, nmsThreshold, indices);
654 classIds.resize(classIds.size() + indices.size(), it.first);
655 for (int idx : indices)
656 {
657 boxes.push_back(localBoxes[idx]);
658 confidences.push_back(localConfidences[idx]);
659 }
660 }
661 }
662 }
663 else
664 {
665 boxes = std::move(predBoxes);
666 classIds = std::move(predClassIds);
667 confidences = std::move(predConfidences);
668 }
669 }
670 else
671 CV_Error(Error::StsNotImplemented, "Unknown output layer type: \"" + lastLayer->type + "\"");
672}

712 virtual
713 std::string decode(const Mat& prediction)
714 {
715 CV_TRACE_FUNCTION();
716 CV_Assert(!prediction.empty());
717 if (decodeType.empty())
718 CV_Error(Error::StsBadArg, "TextRecognitionModel: decodeType is not specified");
719 if (vocabulary.empty())
720 CV_Error(Error::StsBadArg, "TextRecognitionModel: vocabulary is not specified");
721
722 std::string decodeSeq;
723 if (decodeType == "CTC-greedy") {
724 decodeSeq = ctcGreedyDecode(prediction);
725 } else if (decodeType == "CTC-prefix-beam-search") {
726 decodeSeq = ctcPrefixBeamSearchDecode(prediction);
727 } else if (decodeType.length() == 0) {
728 CV_Error(Error::StsBadArg, "Please set decodeType");
729 } else {
730 CV_Error_(Error::StsBadArg, ("Unsupported decodeType: %s", decodeType.c_str()));
731 }
732
733 return decodeSeq;
734 }

736 static Mat ensureFloat32Prediction(const Mat& prediction)
737 {
738 CV_Assert(!prediction.empty());
739 if (prediction.type() == CV_32FC1)
740 return prediction;
741
742 const int depth = prediction.depth();
743 if (depth == CV_16F || depth == CV_16BF)
744 {
745 Mat prediction32f;
746 prediction.convertTo(prediction32f, CV_32F);
747 return prediction32f;
748 }
749
750 CV_CheckType(prediction.type(), CV_32FC1, "");
751 return prediction;
752 }

754 virtual
755 std::string ctcGreedyDecode(const Mat& prediction)
756 {
757 Mat prediction32f = ensureFloat32Prediction(prediction);
758 const Mat& probs = prediction32f;
759
760 std::string decodeSeq;
761 CV_CheckEQ(probs.dims, 3, "");
762 const int vocLength = (int)(vocabulary.size());
763 CV_CheckLE(probs.size[1], vocLength, "");
764 bool ctcFlag = true;
765 int lastLoc = 0;
766 for (int i = 0; i < probs.size[0]; i++)
767 {
768 const float* pred = probs.ptr<float>(i);
769 int maxLoc = 0;
770 float maxScore = pred[0];
771 for (int j = 1; j < vocLength + 1; j++)
772 {
773 float score = pred[j];
774 if (maxScore < score)
775 {
776 maxScore = score;
777 maxLoc = j;
778 }
779 }
780
781 if (maxLoc > 0)
782 {
783 std::string currentChar = vocabulary.at(maxLoc - 1);
784 if (maxLoc != lastLoc || ctcFlag)
785 {
786 lastLoc = maxLoc;
787 decodeSeq += currentChar;
788 ctcFlag = false;
789 }
790 }
791 else
792 {
793 ctcFlag = true;
794 }
795 }
796 return decodeSeq;
797 }

816 struct PrefixHash
817 {
818 size_t operator()(const std::vector<int>& prefix) const
819 {
820 // BKDR hash
821 unsigned int seed = 131;
822 size_t hash = 0;
823 for (size_t i = 0; i < prefix.size(); i++)
824 {
825 hash = hash * seed + prefix[i];
826 }
827 return hash;
828 }
829 };

831 static
832 std::vector<std::pair<float, int>> TopK(
833 const float* predictions, int length, int k)
834 {
835 std::vector<std::pair<float, int>> results;
836 // No prune.
837 if (k <= 0)
838 {
839 for (int i = 0; i < length; ++i)
840 {
841 results.emplace_back(predictions[i], i);
842 }
843 return results;
844 }
845
846 for (int i = 0; i < k; ++i)
847 {
848 results.emplace_back(predictions[i], i);
849 }
850 std::make_heap(results.begin(), results.end(), std::greater<std::pair<float, int>>{});
851
852 for (int i = k; i < length; ++i)
853 {
854 if (predictions[i] > results.front().first)
855 {
856 std::pop_heap(results.begin(), results.end(), std::greater<std::pair<float, int>>{});
857 results.pop_back();
858 results.emplace_back(predictions[i], i);
859 std::push_heap(results.begin(), results.end(), std::greater<std::pair<float, int>>{});
860 }
861 }
862 return results;
863 }

865 static inline
866 bool PrefixScoreCompare(
867 const std::pair<std::vector<int>, PrefixScore>& a,
868 const std::pair<std::vector<int>, PrefixScore>& b)
869 {
870 float probA = LogAdd(a.second.pB, a.second.pNB);
871 float probB = LogAdd(b.second.pB, b.second.pNB);
872 return probA > probB;
873 }

875 virtual
876 std::string ctcPrefixBeamSearchDecode(const Mat& prediction) {
877 Mat prediction32f = ensureFloat32Prediction(prediction);
878 const Mat& probs = prediction32f;
879
880 // CTC prefix beam search decode.
881 // For more detail, refer to:
882 // https://distill.pub/2017/ctc/#inference
883 // https://gist.github.com/awni/56369a90d03953e370f3964c826ed4b0i
884 using Beam = std::vector<std::pair<std::vector<int>, PrefixScore>>;
885 using BeamInDict = std::unordered_map<std::vector<int>, PrefixScore, PrefixHash>;
886
887 CV_CheckEQ(probs.dims, 3, "");
888 CV_CheckEQ(probs.size[1], 1, "");
889 CV_CheckEQ(probs.size[2], (int)vocabulary.size() + 1, ""); // Length add 1 for ctc blank
890
891 std::string decodeSeq;
892 Beam beam = {std::make_pair(std::vector<int>(), PrefixScore(0.0, kNegativeInfinity))};
893 for (int i = 0; i < probs.size[0]; i++)
894 {
895 // Loop over time
896 BeamInDict nextBeam;
897 const float* pred = probs.ptr<float>(i);
898 std::vector<std::pair<float, int>> topkPreds =
899 TopK(pred, vocabulary.size() + 1, vocPruneSize);
900 for (const auto& each : topkPreds)
901 {
902 // Loop over vocabulary
903 float prob = each.first;
904 int token = each.second;
905 for (const auto& it : beam)
906 {
907 const std::vector<int>& prefix = it.first;
908 const PrefixScore& prefixScore = it.second;
909 if (token == 0) // 0 stands for ctc blank
910 {
911 PrefixScore& nextScore = nextBeam[prefix];
912 nextScore.pB = LogAdd(nextScore.pB,
913 LogAdd(prefixScore.pB + prob, prefixScore.pNB + prob));
914 continue;
915 }
916
917 std::vector<int> nPrefix(prefix);
918 nPrefix.push_back(token);
919 PrefixScore& nextScore = nextBeam[nPrefix];
920 if (prefix.size() > 0 && token == prefix.back())
921 {
922 nextScore.pNB = LogAdd(nextScore.pNB, prefixScore.pB + prob);
923 PrefixScore& mScore = nextBeam[prefix];
924 mScore.pNB = LogAdd(mScore.pNB, prefixScore.pNB + prob);
925 }
926 else
927 {
928 nextScore.pNB = LogAdd(nextScore.pNB,
929 LogAdd(prefixScore.pB + prob, prefixScore.pNB + prob));
930 }
931 }
932 }
933 // Beam prune
934 Beam newBeam(nextBeam.begin(), nextBeam.end());
935 int newBeamSize = std::min(static_cast<int>(newBeam.size()), beamSize);
936 std::nth_element(newBeam.begin(), newBeam.begin() + newBeamSize,
937 newBeam.end(), PrefixScoreCompare);
938 newBeam.resize(newBeamSize);
939 std::sort(newBeam.begin(), newBeam.end(), PrefixScoreCompare);
940 beam = std::move(newBeam);
941 }
942
943 CV_Assert(!beam.empty());
944 for (int token : beam[0].first)
945 {
946 CV_Check(token, token > 0 && token <= vocabulary.size(), "");
947 decodeSeq += vocabulary.at(token - 1);
948 }
949 return decodeSeq;
950 }

1202 virtual
1203 std::vector<cv::RotatedRect> detectTextRectangles(InputArray frame, CV_OUT std::vector<float>& confidences) CV_OVERRIDE
1204 {
1205 CV_TRACE_FUNCTION();
1206 std::vector<cv::RotatedRect> results;
1207
1208 std::vector<Mat> outs;
1209 processFrame(frame, outs);
1210 CV_CheckEQ(outs.size(), (size_t)2, "");
1211 Mat geometry = outs[0];
1212 Mat scoreMap = outs[1];
1213
1214 CV_CheckEQ(scoreMap.dims, 4, "");
1215 CV_CheckEQ(geometry.dims, 4, "");
1216 CV_CheckEQ(scoreMap.size[0], 1, "");
1217 CV_CheckEQ(geometry.size[0], 1, "");
1218
1219 if (geometry.size[1] == 1 && scoreMap.size[1] == 5) {
1220 std::swap(geometry, scoreMap);
1221 }
1222
1223 CV_CheckEQ(scoreMap.size[1], 1, "");
1224 CV_CheckEQ(geometry.size[1], 5, "");
1225 CV_CheckEQ(scoreMap.size[2], geometry.size[2], "");
1226 CV_CheckEQ(scoreMap.size[3], geometry.size[3], "");
1227
1228 CV_CheckType(scoreMap.type(), CV_32FC1, "");
1229 CV_CheckType(geometry.type(), CV_32FC1, "");
1230
1231 std::vector<RotatedRect> boxes;
1232 std::vector<float> scores;
1233 const int height = scoreMap.size[2];
1234 const int width = scoreMap.size[3];
1235 for (int y = 0; y < height; ++y)
1236 {
1237 const float* scoresData = scoreMap.ptr<float>(0, 0, y);
1238 const float* x0_data = geometry.ptr<float>(0, 0, y);
1239 const float* x1_data = geometry.ptr<float>(0, 1, y);
1240 const float* x2_data = geometry.ptr<float>(0, 2, y);
1241 const float* x3_data = geometry.ptr<float>(0, 3, y);
1242 const float* anglesData = geometry.ptr<float>(0, 4, y);
1243 for (int x = 0; x < width; ++x)
1244 {
1245 float score = scoresData[x];
1246 if (score < confThreshold)
1247 continue;
1248
1249 float offsetX = x * 4.0f, offsetY = y * 4.0f;
1250 float angle = anglesData[x];
1251 float cosA = std::cos(angle);
1252 float sinA = std::sin(angle);
1253 float h = x0_data[x] + x2_data[x];
1254 float w = x1_data[x] + x3_data[x];
1255
1256 Point2f offset(offsetX + cosA * x1_data[x] + sinA * x2_data[x],
1257 offsetY - sinA * x1_data[x] + cosA * x2_data[x]);
1258 Point2f p1 = Point2f(-sinA * h, -cosA * h) + offset;
1259 Point2f p3 = Point2f(-cosA * w, sinA * w) + offset;
1260 boxes.push_back(RotatedRect(0.5f * (p1 + p3), Size2f(w, h), -angle * 180.0f / (float)CV_PI));
1261 scores.push_back(score);
1262 }
1263 }
1264
1265 // Apply non-maximum suppression procedure.
1266 std::vector<int> indices;
1267 NMSBoxes(boxes, scores, confThreshold, nmsThreshold, indices);
1268
1269 confidences.clear();
1270 confidences.reserve(indices.size());
1271
1272 // Re-scale
1273 Point2f ratio((float)frame.cols() / size.width, (float)frame.rows() / size.height);
1274 bool isUniformRatio = std::fabs(ratio.x - ratio.y) <= 0.01f;
1275 for (uint i = 0; i < indices.size(); i++)
1276 {
1277 auto idx = indices[i];
1278
1279 auto conf = scores[idx];
1280 confidences.push_back(conf);
1281
1282 RotatedRect& box0 = boxes[idx];
1283
1284 if (isUniformRatio)
1285 {
1286 RotatedRect box = box0;
1287 box.center.x *= ratio.x;
1288 box.center.y *= ratio.y;
1289 box.size.width *= ratio.x;
1290 box.size.height *= ratio.y;
1291 results.emplace_back(box);
1292 }
1293 else
1294 {
1295 Point2f vertices[4] = {};
1296 box0.points(vertices);
1297 for (int j = 0; j < 4; j++)
1298 {
1299 vertices[j].x *= ratio.x;
1300 vertices[j].y *= ratio.y;
1301 }
1302 RotatedRect box = minAreaRect(Mat(4, 1, CV_32FC2, (void*)vertices));
1303
1304 // minArea() rect is not normalized, it may return rectangles rotated by +90/-90
1305 float angle_diff = std::fabs(box.angle - box0.angle);
1306 while (angle_diff >= (90 + 45))
1307 {
1308 box.angle += (box.angle < box0.angle) ? 180 : -180;
1309 angle_diff = std::fabs(box.angle - box0.angle);
1310 }
1311 if (angle_diff > 45) // avoid ~90 degree turns
1312 {
1313 std::swap(box.size.width, box.size.height);
1314 if (box.angle < box0.angle)
1315 box.angle += 90;
1316 else if (box.angle > box0.angle)
1317 box.angle -= 90;
1318 }
1319 // CV_DbgAssert(std::fabs(box.angle - box0.angle) <= 45);
1320
1321 results.emplace_back(box);
1322 }
1323 }
1324
1325 return results;
1326 }

1438
1439 std::vector< std::vector<Point2f> > detect(InputArray frame, CV_OUT std::vector<float>& confidences) CV_OVERRIDE
1440 {
1441 CV_TRACE_FUNCTION();
1442 std::vector< std::vector<Point2f> > results;
1443 confidences.clear();
1444 std::vector<Mat> outs;
1445 processFrame(frame, outs);
1446 CV_Assert(outs.size() == 1);
1447 Mat binary = outs[0];
1448
1449 // Threshold
1450 Mat bitmap;
1451 threshold(binary, bitmap, binaryThreshold, 255, THRESH_BINARY);
1452
1453 // Scale ratio
1454 float scaleHeight = (float)(frame.rows()) / (float)(binary.size[0]);
1455 float scaleWidth = (float)(frame.cols()) / (float)(binary.size[1]);
1456
1457 // Find contours
1458 std::vector< std::vector<Point> > contours;
1459 bitmap.convertTo(bitmap, CV_8UC1);
1460 findContours(bitmap, contours, RETR_LIST, CHAIN_APPROX_SIMPLE);
1461
1462 // Candidate number limitation
1463 size_t numCandidate = std::min(contours.size(), (size_t)(maxCandidates > 0 ? maxCandidates : INT_MAX));
1464
1465 for (size_t i = 0; i < numCandidate; i++)
1466 {
1467 std::vector<Point>& contour = contours[i];
1468
1469 // Calculate text contour score
1470 float score = contourScore(binary, contour);
1471 if (score < polygonThreshold)
1472 continue;
1473
1474 // Rescale
1475 std::vector<Point> contourScaled; contourScaled.reserve(contour.size());
1476 for (size_t j = 0; j < contour.size(); j++)
1477 {
1478 contourScaled.push_back(Point(int(contour[j].x * scaleWidth),
1479 int(contour[j].y * scaleHeight)));
1480 }
1481
1482 // Unclip
1483 RotatedRect box = minAreaRect(contourScaled);
1484 float minLen = std::min(box.size.height/scaleWidth, box.size.width/scaleHeight);
1485
1486 // Filter very small boxes
1487 if (minLen < 3)
1488 continue;
1489

1547 static void unclip(const std::vector<Point2f>& inPoly, std::vector<Point2f> &outPoly, const double unclipRatio)
1548 {
1549 double area = contourArea(inPoly);
1550 double length = arcLength(inPoly, true);
1551
1552 if(length == 0.)
1553 return;
1554
1555 double distance = area * unclipRatio / length;
1556
1557 size_t numPoints = inPoly.size();
1558 std::vector<std::vector<Point2f>> newLines;
1559 for (size_t i = 0; i < numPoints; i++) {
1560 std::vector<Point2f> newLine;
1561 Point pt1 = inPoly[i];
1562 Point pt2 = inPoly[(i - 1) % numPoints];
1563 Point vec = pt1 - pt2;
1564 float unclipDis = (float)(distance / norm(vec));
1565 Point2f rotateVec = Point2f(vec.y * unclipDis, -vec.x * unclipDis);
1566 newLine.push_back(Point2f(pt1.x + rotateVec.x, pt1.y + rotateVec.y));
1567 newLine.push_back(Point2f(pt2.x + rotateVec.x, pt2.y + rotateVec.y));
1568 newLines.push_back(newLine);
1569 }
1570
1571 size_t numLines = newLines.size();
1572 for (size_t i = 0; i < numLines; i++) {
1573 Point2f a = newLines[i][0];
1574 Point2f b = newLines[i][1];
1575 Point2f c = newLines[(i + 1) % numLines][0];
1576 Point2f d = newLines[(i + 1) % numLines][1];
1577 Point2f pt;
1578 Point2f v1 = b - a;
1579 Point2f v2 = d - c;
1580 double cosAngle = (v1.x * v2.x + v1.y * v2.y) / (norm(v1) * norm(v2));
1581
1582 if( fabs(cosAngle) > 0.7 ) {
1583 pt.x = (b.x + c.x) * 0.5;
1584 pt.y = (b.y + c.y) * 0.5;
1585 } else {
1586 double denom = a.x * (double)(d.y - c.y) + b.x * (double)(c.y - d.y) +
1587 d.x * (double)(b.y - a.y) + c.x * (double)(a.y - b.y);
1588 double num = a.x * (double)(d.y - c.y) + c.x * (double)(a.y - d.y) + d.x * (double)(c.y - a.y);
1589 double s = num / denom;
1590
1591 pt.x = a.x + s*(b.x - a.x);
1592 pt.y = a.y + s*(b.y - a.y);
1593 }
1594
1595
1596 outPoly.push_back(pt);
1597 }
