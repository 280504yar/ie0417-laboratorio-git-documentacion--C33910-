22 // ONNX Cast float->int truncates toward zero; Mat::convertTo rounds. Truncate to match the spec.
23 template<typename DT>
24 inline void truncateToIntImpl(const Mat& src, Mat& dst)
25 {
26 const int n = (int)src.total() * src.channels();
27 DT* d = dst.ptr<DT>();
28 if (src.depth() == CV_32F)
29 {
30 const float* s = src.ptr<float>();
31 for (int i = 0; i < n; ++i)
32 d[i] = saturate_cast<DT>(std::trunc(s[i]));
33 }
34 else
35 {
36 const double* s = src.ptr<double>();
37 for (int i = 0; i < n; ++i)
38 d[i] = saturate_cast<DT>(std::trunc(s[i]));
39 }
40 }
41
42 inline void truncateFloatToInt(const Mat& src, Mat& dst)
43 {
44 switch (dst.depth())
45 {
46 case CV_8U: truncateToIntImpl<uchar>(src, dst); break;
47 case CV_8S: truncateToIntImpl<schar>(src, dst); break;
48 case CV_16U: truncateToIntImpl<ushort>(src, dst); break;
49 case CV_16S: truncateToIntImpl<short>(src, dst); break;
50 case CV_32S: truncateToIntImpl<int>(src, dst); break;
51 case CV_64S: truncateToIntImpl<int64_t>(src, dst); break;
52 default: src.convertTo(dst, dst.depth()); break;
53 }
54 }

56 inline void castQuantized(const Mat& src, Mat& dst, int targetDepth)
57 {
58 if (targetDepth == CV_16F)
59 {
60 CV_Assert(dst.depth() == CV_32F);
61 if (src.depth() == CV_32F)
62 {
63 MatConstIterator_<float> sIt = src.begin<float>(), sEnd = src.end<float>();
64 MatIterator_<float> dIt = dst.begin<float>();
65 for (; sIt != sEnd; ++sIt, ++dIt)
66 {
67 *dIt = (float)hfloat(*sIt);
68 }
69 }
70 else if (src.depth() == CV_64F)
71 {
72 MatConstIterator_<double> sIt = src.begin<double>(), sEnd = src.end<double>();
73 MatIterator_<float> dIt = dst.begin<float>();
74 for (; sIt != sEnd; ++sIt, ++dIt)
75 {
76 float v = (float)*sIt;
77 *dIt = (float)hfloat(v);
78 }
79 }
80 else
81 {
82 Mat src32; src.convertTo(src32, CV_32F);
83 MatConstIterator_<float> sIt = src32.begin<float>(), sEnd = src32.end<float>();
84 MatIterator_<float> dIt = dst.begin<float>();
85 for (; sIt != sEnd; ++sIt, ++dIt)
86 {
87 *dIt = (float)hfloat(*sIt);
88 }
89 }
90 return;
91 }
92
93 if (targetDepth == CV_16BF)
94 {
95 const int ddepth = dst.depth();
96 if (!(ddepth == CV_16BF || ddepth == CV_16U))
97 {
98 CV_Error(Error::StsNotImplemented, "Unsupported destination depth for BF16 cast");
99 }
100
101 Mat dst_bits(dst.size(), CV_MAKETYPE(CV_16U, dst.channels()), dst.data, dst.step);
102
103 const Mat* src32p;
104 Mat src32;
105 if (src.depth() == CV_32F)
106 src32p = &src;
107 else
108 {
109 src.convertTo(src32, CV_32F);
110 src32p = &src32;
111 }
112
113 const int rows = src32p->rows;
114 const int cols_x_cn = src32p->cols * src32p->channels();
115 for (int r = 0; r < rows; ++r)
116 {
117 const float* in = src32p->ptr<float>(r);
118 ushort* out = dst_bits.ptr<ushort>(r);
119 for (int i = 0; i < cols_x_cn; ++i)
120 {
121 // float32 -> bfloat16 with round-to-nearest-even (matches ONNX).
122 Cv32suf u; u.f = in[i];
123 const uint32_t x = u.u;
124 if ((x & 0x7fffffffu) > 0x7f800000u) // NaN: keep it NaN
125 out[i] = (ushort)((x >> 16) | 0x0040u);
126 else
127 {
128 const uint32_t bias = 0x7fffu + ((x >> 16) & 1u);
129 out[i] = (ushort)((x + bias) >> 16);
130 }
131 }
132 }
133 return;
134 }
135 src.convertTo(dst, dst.depth());
136 }

226 bool forward_ocl(InputArrayOfArrays inputs_, OutputArrayOfArrays outputs_, OutputArrayOfArrays internals_)
227 {
228 std::vector<UMat> inputs, outputs;
229
230 inputs_.getUMatVector(inputs);
231 outputs_.getUMatVector(outputs);
232 CV_CheckEQ(inputs.size(), (size_t)1, "");
233 CV_CheckEQ(outputs.size(), (size_t)1, "");
234
235 int runtimeTargetDepth = -1;
236 if (hasToParam)
237 {
238 runtimeTargetDepth = toCvDepth_;
239 }
240 else
241 {
242 if (inputs.size() >= 2 && !inputs[1].empty())
243 runtimeTargetDepth = inputs[1].depth();
244 else
245 runtimeTargetDepth = inputs[0].depth();
246 }
247
248 if (runtimeTargetDepth == CV_16F && outputs[0].depth() == CV_32F)
249 {
250 return false;
251 }
252
253 // bf16 needs the float->bfloat16 bit reduction; a plain OpenCL convertTo
254 // would numerically round instead. Fall back to the CPU path.
255 if (runtimeTargetDepth == CV_16BF)
256 {
257 return false;
258 }
259
260 if (inputs[0].depth() == outputs[0].depth())
261 inputs[0].copyTo(outputs[0]);
262 else
263 inputs[0].convertTo(outputs[0], outputs[0].depth());
264 return true;
265 }

268 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
269 {
270 CV_TRACE_FUNCTION();
271 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
272
273 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
274 forward_ocl(inputs_arr, outputs_arr, internals_arr));
275
276 std::vector<Mat> inputs, outputs;
277 inputs_arr.getMatVector(inputs);
278 outputs_arr.getMatVector(outputs);
279
280 CV_Check(inputs.size(), inputs.size() == 1 || inputs.size() == 2, "Cast expects 1 (Cast) or 2 (CastLike) inputs");
281 CV_CheckEQ(outputs.size(), (size_t)1, "");
282
283 const Mat& src0 = inputs[0];
284 Mat& dst0 = outputs[0];
285
286 int runtimeTargetDepth = -1;
287 if (hasToParam)
288 {
289 runtimeTargetDepth = toCvDepth_;
290 }
291 else
292 {
293 Net::Impl* netimpl_ = getNetImpl(this);
294 if (netimpl_ && this->inputs.size() >= 2)
295 {
296 const Arg& in1_arg = this->inputs[1];
297 const ArgData& ad = netimpl_->argData(in1_arg);
298 if (ad.type >= 0)
299 runtimeTargetDepth = CV_MAT_DEPTH(ad.type);
300 }
301 if (runtimeTargetDepth < 0 && inputs.size() >= 2 && !inputs[1].empty())
302 runtimeTargetDepth = inputs[1].depth();
303 if (runtimeTargetDepth < 0)
304 runtimeTargetDepth = src0.depth();
305 }
306 CV_CheckGE(runtimeTargetDepth, 0, "Cast: failed to resolve target data type at runtime");
307
308 int plannedDDepth = (runtimeTargetDepth == CV_16F) ? CV_32F : runtimeTargetDepth;
309 if (dst0.depth() != plannedDDepth)
310 dst0.create(dst0.size(), CV_MAKETYPE(plannedDDepth, src0.channels()));
311
312 Mat src = src0;
313 Mat dst = dst0;
314
315 const int sdepth = src.depth();
316 const int ddepth = dst.depth();
317
318 if (sdepth == runtimeTargetDepth && !(runtimeTargetDepth == CV_16F && ddepth == CV_32F))
319 {
320 src0.copyTo(dst0);
321 return;
322 }
323
324 if (runtimeTargetDepth == CV_16BF && (ddepth == CV_16BF || ddepth == CV_16U))
325 {
326 castQuantized(src, dst, CV_16BF);
327 }
328 else if (sdepth == CV_16BF)
329 {
330 src.convertTo(dst, ddepth);
331 }
332 else if (runtimeTargetDepth == CV_16F && ddepth == CV_32F)
333 {
334 castQuantized(src, dst, CV_16F);
335 }
336 else if (runtimeTargetDepth == CV_64F && ddepth != CV_64F)
337 {
338 if (ddepth == CV_16U || ddepth == CV_16BF)
339 {
340 castQuantized(src, dst, CV_16BF);
341 }
342 else
343 src.convertTo(dst, ddepth);
344 }
345 else if ((sdepth == CV_32F || sdepth == CV_64F) && CV_IS_INT_TYPE(ddepth))
346 {
347 truncateFloatToInt(src, dst);
348 }
349 else
350 {
351 src.convertTo(dst, ddepth);
352 }
353 }

381 // ONNX TensorProto::DataType values (see opencv-onnx.proto); the 'to'

399 static int mapToCvDepth(int v)
400 {
401 switch (v)
402 {
403 case ONNX_DT_FLOAT: return CV_32F;
404 case ONNX_DT_UINT8: return CV_8U;
405 case ONNX_DT_INT8: return CV_8S;
406 case ONNX_DT_UINT16: return CV_16U;
407 case ONNX_DT_INT16: return CV_16S;
408 case ONNX_DT_INT32: return CV_32S;
409 case ONNX_DT_INT64: return CV_64S;
410 case ONNX_DT_BOOL: return CV_Bool;
411 case ONNX_DT_FLOAT16: return CV_16F;
412 case ONNX_DT_DOUBLE: return CV_64F;
413 case ONNX_DT_BFLOAT16: return CV_16BF;
414 default: break;
415 }
416
417 CV_Error(Error::StsNotImplemented, "Cast: unsupported 'to' / dtype value");
418 }
