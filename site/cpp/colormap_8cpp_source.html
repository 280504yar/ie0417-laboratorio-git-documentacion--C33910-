28static Mat linspace(float x0, float x1, int n)
29{
30 Mat pts(n, 1, CV_32FC1);

33 pts.at<float>(i,0) = x0+i*step;

40static void sortMatrixRowsByIndices(InputArray _src, InputArray _indices, OutputArray _dst)
41{
42 if(_indices.getMat().type() != CV_32SC1)
43 CV_Error(Error::StsUnsupportedFormat, "cv::sortRowsByIndices only works on integer indices!");
44 Mat src = _src.getMat();
45 std::vector<int> indices = _indices.getMat();
46 _dst.create(src.rows, src.cols, src.type());
47 Mat dst = _dst.getMat();
48 for(size_t idx = 0; idx < indices.size(); idx++) {
49 Mat originalRow = src.row(indices[idx]);
50 Mat sortedRow = dst.row((int)idx);
51 originalRow.copyTo(sortedRow);
52 }
53}

74template <typename _Tp> static
75Mat interp1_(const Mat& X_, const Mat& Y_, const Mat& XI)
76{
77 int n = XI.rows;
78 // sort input table
79 std::vector<int> sort_indices = argsort(X_);
80
81 Mat X = sortMatrixRowsByIndices(X_,sort_indices);
82 Mat Y = sortMatrixRowsByIndices(Y_,sort_indices);
83 // interpolated values
84 Mat yi = Mat::zeros(XI.size(), XI.type());
85 for(int i = 0; i < n; i++) {
86 int low = 0;
87 int high = X.rows - 1;
88 // set bounds
89 if(XI.at<_Tp>(i,0) < X.at<_Tp>(low, 0))
90 high = 1;
91 if(XI.at<_Tp>(i,0) > X.at<_Tp>(high, 0))
92 low = high - 1;
93 // binary search
94 while((high-low)>1) {
95 const int c = low + ((high - low) >> 1);
96 if(XI.at<_Tp>(i,0) > X.at<_Tp>(c,0)) {
97 low = c;
98 } else {
99 high = c;
100 }
101 }
102 // linear interpolation
103 yi.at<_Tp>(i,0) += Y.at<_Tp>(low,0)
104 + (XI.at<_Tp>(i,0) - X.at<_Tp>(low,0))
105 * (Y.at<_Tp>(high,0) - Y.at<_Tp>(low,0))
106 / (X.at<_Tp>(high,0) - X.at<_Tp>(low,0));
107 }
108 return yi;
109}

111static Mat interp1(InputArray _x, InputArray _Y, InputArray _xi)
112{
113 // get matrices
114 Mat x = _x.getMat();
115 Mat Y = _Y.getMat();
116 Mat xi = _xi.getMat();
117 // check types & alignment
118 CV_Assert((x.type() == Y.type()) && (Y.type() == xi.type()));
119 CV_Assert((x.cols == 1) && (x.rows == Y.rows) && (x.cols == Y.cols));
120 // call templated interp1
121 switch(x.type()) {
122 case CV_8SC1: return interp1_<char>(x,Y,xi); break;
123 case CV_8UC1: return interp1_<unsigned char>(x,Y,xi); break;
124 case CV_16SC1: return interp1_<short>(x,Y,xi); break;
125 case CV_16UC1: return interp1_<unsigned short>(x,Y,xi); break;
126 case CV_32SC1: return interp1_<int>(x,Y,xi); break;
127 case CV_32FC1: return interp1_<float>(x,Y,xi); break;
128 case CV_64FC1: return interp1_<double>(x,Y,xi); break;
129 }
130 CV_Error(Error::StsUnsupportedFormat, "");
131}

736 CV_Error(Error::StsBadArg, "cv::ColorMap only supports source images of type CV_8UC1 or CV_8UC3");

742 CV_CheckType(lut_type, (lut_type == CV_8UC1) || (lut_type == CV_8UC3),

746 if (src.channels() == 1)
747 srcGray = src;
748 else
749 cv::cvtColor(src, srcGray, cv::COLOR_BGR2GRAY);//BGR because of historical cv::LUT() usage

752 Mat dstMat = _dst.getMat();
753
754 //we do not use cv::LUT() which requires src.channels() == dst.channels()
755 const int rows = srcGray.rows;
756 const int cols = srcGray.cols;
757 const int minimalPixelsPerPacket = 1<<12;

759 const int rowsPacketsCount = (rows+rowsPerPacket-1)/rowsPerPacket;

768 lut_pixel_t* dstRow = dstMat.ptr<lut_pixel_t>(row);
769 for(int col = 0 ; col<cols ; ++col)
770 *dstRow++ = srcLUT[*srcRow++];
771 }
772 };
773 parallel_for_(all, body, rowsPacketsCount);
774 }
775 else if (lut_type == CV_8UC3) {
776 typedef Vec3b lut_pixel_t;

782 for(int col = 0 ; col<cols ; ++col)
783 *dstRow++ = srcLUT[*srcRow++];
784 }
785 };
786 parallel_for_(all, body, rowsPacketsCount);
