15// Returns the homography candidate index with the most cheirality-consistent triangulations.
16int bestHomographyCandidate(
17 const std::vector<Mat>& Rs,
18 const std::vector<Mat>& ts,
19 const std::vector<Point2f>& pts1,
20 const std::vector<Point2f>& pts2,
21 const Mat& P1)
22{
23 int bestIdx = 0;
24 int bestCount = -1;
25 const int nSol = (int)Rs.size();
26
27 for (int s = 0; s < nSol; ++s)
28 {
29 Mat Rt(3, 4, CV_64F);
30 Rs[s].copyTo(Rt(Rect(0, 0, 3, 3)));
31 ts[s].reshape(1, 3).copyTo(Rt(Rect(3, 0, 1, 3)));
32
33 Mat Klocal = P1(Rect(0, 0, 3, 3)).clone();
34 Mat P2_s = Klocal * Rt;
35
36 Mat pts4D;
37 triangulatePoints(P1, P2_s, pts1, pts2, pts4D);
38
39 int count = 0;
40 for (int i = 0; i < pts4D.cols; ++i)
41 {
42 float w = pts4D.at<float>(3, i);
43 if (std::abs(w) < 1e-9f) continue;
44 float X = pts4D.at<float>(0, i) / w;
45 float Y = pts4D.at<float>(1, i) / w;
46 float Z = pts4D.at<float>(2, i) / w;
47 float Z2 = (float)(Rs[s].at<double>(2,0)*X + Rs[s].at<double>(2,1)*Y
48 + Rs[s].at<double>(2,2)*Z
49 + ts[s].reshape(1,3).at<double>(2,0));
50 if (Z > 0 && Z2 > 0) ++count;
51 }
52 if (count > bestCount) { bestCount = count; bestIdx = s; }
53 }
54 return bestIdx;
55}

59bool VisualOdometryImpl::bootstrap(Frame& cur)
60{
61 std::vector<DMatch> matches;
62 matchFrames(refFrame.keypoints, refFrame.descriptors, refFrame.imageSize,
63 cur.keypoints, cur.descriptors, cur.imageSize, matches);
64
65 if ((int)matches.size() < params.minInitInliers)
66 {
67 // slide ref forward so we don't keep trying against a frame that's too far
68 refFrame = std::move(cur);
69 return false;
70 }
71
72 std::vector<Point2f> refU, curU;
73 refU.reserve(matches.size());
74 curU.reserve(matches.size());
75 for (const auto& m : matches)
76 {
77 refU.push_back(refFrame.undistKpts[m.queryIdx]);
78 curU.push_back(cur.undistKpts[m.trainIdx]);
79 }
80
81 Mat maskE, maskH;
82 Mat E = findEssentialMat(refU, curU, K, RANSAC,
83 params.essentialRansacConfidence,
84 params.essentialRansacThresh, 1000, maskE);
85 Mat H = findHomography(refU, curU, RANSAC,
86 params.essentialRansacThresh, maskH);
87
88 const int nE = (!E.empty() && !maskE.empty()) ? countNonZero(maskE) : 0;
89 const int nH = (!H.empty() && !maskH.empty()) ? countNonZero(maskH) : 0;
90
91 if (nE < params.minInitInliers && nH < params.minInitInliers)
92 return false;
93
94 const double RH = (double)nH / ((double)nH + (double)nE + 1e-9);
95
96 Mat R, t, modelMask;
97 Mat P1 = K * Mat::eye(3, 4, CV_64F);
98
99 if (RH > params.hfRatioThresh && !H.empty())
100 {
101 std::vector<Mat> Rs, ts, normals;
102 int nSol = decomposeHomographyMat(H, K, Rs, ts, normals);
103 if (nSol <= 0) return false;
104
105 std::vector<Point2f> p1In, p2In;
106 for (size_t i = 0; i < matches.size(); ++i)
107 if (!maskH.empty() && maskH.at<uchar>((int)i))
108 { p1In.push_back(refU[i]); p2In.push_back(curU[i]); }
109 if (p1In.empty()) return false;
110
111 int best = bestHomographyCandidate(Rs, ts, p1In, p2In, P1);
112 R = Rs[best].clone();
113 t = ts[best].clone();
114 modelMask = maskH;
115 }
116 else
117 {
118 if (E.empty() || nE < params.minInitInliers) return false;
119 Mat recoverMask = maskE.clone();
120 int nPose = recoverPose(E, refU, curU, K, R, t, recoverMask);
121 if (nPose < params.minInitInliers) return false;
122 modelMask = recoverMask;
123 }
124
125 std::vector<Point2f> refIn, curIn;
126 std::vector<int> matchIn;
127 for (size_t i = 0; i < matches.size(); ++i)
128 if (!modelMask.empty() && modelMask.at<uchar>((int)i))
129 {
130 refIn.push_back(refU[i]);
131 curIn.push_back(curU[i]);
132 matchIn.push_back((int)i);
133 }
134
135 if ((int)refIn.size() < params.minInitInliers)
136 return false;
137
138 Mat Rt(3, 4, CV_64F);
139 R.copyTo(Rt(Rect(0, 0, 3, 3)));
140 t.reshape(1, 3).copyTo(Rt(Rect(3, 0, 1, 3)));
141 Mat P2 = K * Rt;
142
143 Mat pts4D;
144 triangulatePoints(P1, P2, refIn, curIn, pts4D);
145
146 Matx44d T_ref = Matx44d::eye();
147 Matx44d T_cur = detail::makePose(R, t);
148
149 std::vector<Point3d> goodPts;
150 std::vector<int> goodMatch;
151 int nValid = 0;
152 int nPos = 0;
153
154 for (int i = 0; i < pts4D.cols; ++i)
155 {
156 double w = pts4D.at<float>(3, i);
157 if (std::abs(w) < 1e-9) continue;
158 ++nValid;
159
160 double X = pts4D.at<float>(0, i) / w;
161 double Y = pts4D.at<float>(1, i) / w;
162 double Z = pts4D.at<float>(2, i) / w;
163
164 if (Z <= 0) continue;
165 double Z2 = R.at<double>(2,0)*X + R.at<double>(2,1)*Y
166 + R.at<double>(2,2)*Z + t.reshape(1,3).at<double>(2,0);
167 if (Z2 <= 0) continue;
168 ++nPos;
169
170 Point3d Xw(X, Y, Z);
171 if (detail::parallaxDeg(Xw, T_ref, T_cur) < params.minInitParallaxDeg)
172 continue;
173
174 goodPts.push_back(Xw);
175 goodMatch.push_back(matchIn[i]);
176 }
177
178 // reject degenerate decompositions (< 90% positive-depth ratio)
179 if (nValid > 0 && (double)nPos / nValid < 0.9)
180 return false;
181 if ((int)goodPts.size() < params.minInitPoints)
182 return false;
183
184 // normalize scale: set median scene depth to 1.0
185 std::vector<double> depths;
186 depths.reserve(goodPts.size());
187 for (const auto& p : goodPts) depths.push_back(p.z);
188 std::nth_element(depths.begin(), depths.begin() + depths.size()/2, depths.end());
189 double med = depths[depths.size()/2];
190 if (med < 1e-9) return false;
191
192 double scale = 1.0 / med;
193 for (auto& p : goodPts) { p.x *= scale; p.y *= scale; p.z *= scale; }
194
195 Mat tSc;
196 t.reshape(1, 3).convertTo(tSc, CV_64F);
197 tSc = tSc * scale;
198 T_cur = detail::makePose(R, tSc);
199
200 auto makeKF = [](const Frame& f) -> KeyFrame* {
201 KeyFrame* kf = new KeyFrame();
202 kf->poseCw = Matx44d::eye();
203 kf->keypoints = f.keypoints;
204 kf->descriptors = f.descriptors.clone();
205 kf->undistKpts = f.undistKpts;
206 kf->imageSize = f.imageSize;
207 kf->mapPoints.assign(f.keypoints.size(), nullptr);
208 return kf;
209 };
210
211 KeyFrame* kfRef = makeKF(refFrame);
212 kfRef->poseCw = T_ref;
213
214 KeyFrame* kfCur = makeKF(cur);
215 kfCur->poseCw = T_cur;
216 kfCur->parent = kfRef;
217
218 map.addKeyframe(kfRef);
219 map.addKeyframe(kfCur);
220
221 for (size_t i = 0; i < goodPts.size(); ++i)
222 {
223 MapPoint* mp = new MapPoint();
224 mp->pos = goodPts[i];
225 const DMatch& m = matches[goodMatch[i]];
226 mp->refDesc = refFrame.descriptors.row(m.queryIdx).clone();
227
228 map.addMapPoint(mp);
229 map.addObservation(kfRef, (size_t)m.queryIdx, mp);
230 map.addObservation(kfCur, (size_t)m.trainIdx, mp);
231 }
232
233 detail::updateCovisibility(kfRef);
234
235 map.setRefKeyframe(kfRef);
236 map.setCurrentKeyframe(kfCur);
237
238 lastKf = kfCur;
239 framesSinceKf = 0;
240 lastKfInliers = (int)goodPts.size();
241 lastPoseCw = T_cur;
242 state = TRACKING;
243 hasVelocity = false;
244
245 map.appendPose(T_ref);
246 map.appendPose(T_cur);
247
248 refFrame = Frame();
249 return true;
250}
