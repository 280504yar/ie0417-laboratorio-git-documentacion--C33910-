29// AVX2 8x8 f32 in-register transpose. v_transpose4x4 transposes the two
30// 128-bit halves independently; v_combine_low/high then exchange halves so
31// element k of each target row comes from the right source row.
32static inline void transpose_8x8_f32_avx2_(const float* src, int64_t src_stride,
33 float* dst, int64_t dst_stride)
34{
35 v_float32 r0 = vx_load(src + 0 * src_stride);
36 v_float32 r1 = vx_load(src + 1 * src_stride);
37 v_float32 r2 = vx_load(src + 2 * src_stride);
38 v_float32 r3 = vx_load(src + 3 * src_stride);
39 v_float32 r4 = vx_load(src + 4 * src_stride);
40 v_float32 r5 = vx_load(src + 5 * src_stride);
41 v_float32 r6 = vx_load(src + 6 * src_stride);
42 v_float32 r7 = vx_load(src + 7 * src_stride);
43
44 v_float32 a0, a1, a2, a3, a4, a5, a6, a7;
45 v_transpose4x4(r0, r1, r2, r3, a0, a1, a2, a3);
46 v_transpose4x4(r4, r5, r6, r7, a4, a5, a6, a7);
47
48 v_float32 b0 = v_combine_low (a0, a4);
49 v_float32 b1 = v_combine_low (a1, a5);
50 v_float32 b2 = v_combine_low (a2, a6);
51 v_float32 b3 = v_combine_low (a3, a7);
52 v_float32 b4 = v_combine_high(a0, a4);
53 v_float32 b5 = v_combine_high(a1, a5);
54 v_float32 b6 = v_combine_high(a2, a6);
55 v_float32 b7 = v_combine_high(a3, a7);
56
57 v_store(dst + 0 * dst_stride, b0);
58 v_store(dst + 1 * dst_stride, b1);
59 v_store(dst + 2 * dst_stride, b2);
60 v_store(dst + 3 * dst_stride, b3);
61 v_store(dst + 4 * dst_stride, b4);
62 v_store(dst + 5 * dst_stride, b5);
63 v_store(dst + 6 * dst_stride, b6);
64 v_store(dst + 7 * dst_stride, b7);
65}

68void transpose2D_f32_(const float* inp, float* out,
69 int64_t outer, int64_t rows, int64_t cols)
70{
71 const int TILE = 32;
72 int64_t batchStride = rows * cols;
73
74 int64_t rtiles = (rows + TILE - 1) / TILE;
75 int64_t ctiles = (cols + TILE - 1) / TILE;
76 int64_t total = outer * rtiles * ctiles;
77
78 parallel_for_(Range(0, (int)total), [&](const Range& range) {
79 for (int64_t idx = range.start; idx < range.end; idx++) {
80 int64_t rem = idx;
81 int64_t b = rem / (rtiles * ctiles);
82 rem -= b * rtiles * ctiles;
83 int64_t ti = rem / ctiles;
84 int64_t tj = rem - ti * ctiles;
85 int64_t r0 = ti * TILE, r1 = std::min(r0 + TILE, rows);
86 int64_t c0 = tj * TILE, c1 = std::min(c0 + TILE, cols);
87
88 const float* inB = inp + b * batchStride;
89 float* outB = out + b * batchStride;
90
91#if (CV_SIMD)
92 if (VTraits<v_float32>::vlanes() == 8) {
93 int64_t r = r0;
94 for (; r + 8 <= r1; r += 8) {
95 int64_t c = c0;
96 for (; c + 8 <= c1; c += 8) {
97 transpose_8x8_f32_avx2_(inB + c * rows + r, rows,
98 outB + r * cols + c, cols);
99 }
100 for (; c < c1; c++) {
101 for (int k = 0; k < 8; k++)
102 outB[(r + k) * cols + c] = inB[c * rows + (r + k)];
103 }
104 }
105 for (; r < r1; r++) {
106 for (int64_t c = c0; c < c1; c++)
107 outB[r * cols + c] = inB[c * rows + r];
108 }
109 continue;
110 }
111#endif
112 for (int64_t r = r0; r < r1; r++) {
113 for (int64_t c = c0; c < c1; c++)
114 outB[r * cols + c] = inB[c * rows + r];
115 }
116 }
117 });
118}
