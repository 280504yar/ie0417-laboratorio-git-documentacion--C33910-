157size_t mlasFlashAttentionBufferBytesPerThread(int q_block_size,
158 int kv_block_size,
159 int v_head_size)
160{
161 if (q_block_size <= 0 || kv_block_size <= 0 || v_head_size <= 0) return 0;
162 // flashattn.cpp lays out the per-thread scratch as:
163 // l[q_block_size] + m[q_block_size]
164 // + intermediate[q_block_size * kv_block_size]
165 // + temp_output[q_block_size * v_head_size]
166 const size_t q = static_cast<size_t>(q_block_size);
167 const size_t kv = static_cast<size_t>(kv_block_size);
168 const size_t vd = static_cast<size_t>(v_head_size);
169 return (q * (2 + kv + vd)) * sizeof(float);
170}

172bool mlasFlashAttention(const float* query, const float* key, const float* value,
173 float* output,
174 int batch_size, int num_heads,
175 int q_seq_len, int kv_seq_len,
176 int qk_head_size, int v_head_size,
177 float scale,
178 int q_block_size, int kv_block_size,
179 void* scratch, int thread_count)
180{
181 if (!mlasAvailable()) return false;
182 if (batch_size <= 0 || num_heads <= 0) return false;
183 if (q_seq_len <= 0 || kv_seq_len <= 0) return false;
184 if (qk_head_size <= 0 || v_head_size <= 0) return false;
185 if (q_block_size <= 0 || kv_block_size <= 0) return false;
186 if (thread_count <= 0 || scratch == nullptr) return false;
187 if (query == nullptr || key == nullptr || value == nullptr || output == nullptr)
188 return false;
189
190 MlasFlashAttentionThreadedArgs args;
191 args.batch_size = batch_size;
192 args.num_heads = num_heads;
193 args.q_sequence_length = q_seq_len;
194 args.kv_sequence_length = kv_seq_len;
195 args.qk_head_size = qk_head_size;
196 args.v_head_size = v_head_size;
197 args.q_block_size = q_block_size;
198 args.kv_block_size = kv_block_size;
199 args.scale = scale;
200 args.thread_count = thread_count;
201 args.buffer = static_cast<float*>(scratch);
202 args.buffer_size_per_thread = mlasFlashAttentionBufferBytesPerThread(
203 q_block_size, kv_block_size, v_head_size);
204 args.query = query;
205 args.key = key;
206 args.value = value;
207 args.output = output;
208
209 MlasFlashAttention(&args, /*ThreadPool=*/nullptr);
210 return true;
211}
