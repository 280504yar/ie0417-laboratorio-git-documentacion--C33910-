141bool HomographyDecomp::isRotationValid(const Matx33d& R, const double epsilon)
142{
143 Matx33d RtR = R.t() * R;
144 Matx33d I(1,0,0, 0,1,0, 0,0,1);
145 if (norm(RtR, I, NORM_INF) > epsilon)
146 return false;
147 return (fabs(determinant(R) - 1.0) < epsilon);
148}

150bool HomographyDecomp::passesSameSideOfPlaneConstraint(CameraMotion& motion)
151{
152 typedef Matx<double, 1, 1> Matx11d;
153 Matx31d t = Matx31d(motion.t);
154 Matx31d n = Matx31d(motion.n);
155 Matx11d proj = n.t() * motion.R.t() * t;
156 if ( (1 + proj(0, 0) ) <= 0 )
157 return false;
158 return true;
159}

173/* function computes R&t from tstar, and plane normal(n) using
174 R = H * inv(I + tstar*transpose(n) );
175 t = R * tstar;
176 returns true if computed R&t is a valid solution
177 */
178bool HomographyDecompZhang::findMotionFrom_tstar_n(const cv::Vec3d& tstar, const cv::Vec3d& n, CameraMotion& motion)
179{
180 Matx31d tstar_m = Mat(tstar);
181 Matx31d n_m = Mat(n);
182 Matx33d temp = tstar_m * n_m.t();
183 temp(0, 0) += 1.0;
184 temp(1, 1) += 1.0;
185 temp(2, 2) += 1.0;
186 motion.R = getHnorm() * temp.inv();
187 if (cv::determinant(motion.R) < 0)
188 {
189 motion.R *= -1;
190 }
191 motion.t = motion.R * tstar;
192 motion.n = n;
193 return passesSameSideOfPlaneConstraint(motion);
194}

196void HomographyDecompZhang::decompose(std::vector<CameraMotion>& camMotions)
197{
198 Mat W, U, Vt;
199 SVD::compute(getHnorm(), W, U, Vt);
200 CV_Assert(W.total() > 2 && Vt.total() > 7);
201 double lambda1=W.at<double>(0);
202 double lambda3=W.at<double>(2);
203 double lambda1m3 = (lambda1-lambda3);
204 double lambda1m3_2 = lambda1m3*lambda1m3;
205 double lambda1t3 = lambda1*lambda3;
206
207 double t1 = 1.0/(2.0*lambda1t3);
208 double t2 = sqrt(1.0+4.0*lambda1t3/lambda1m3_2);
209 double t12 = t1*t2;
210
211 double e1 = -t1 + t12; //t1*(-1.0f + t2 );
212 double e3 = -t1 - t12; //t1*(-1.0f - t2);
213 double e1_2 = e1*e1;
214 double e3_2 = e3*e3;
215
216 double nv1p = sqrt(e1_2*lambda1m3_2 + 2*e1*(lambda1t3-1) + 1.0);
217 double nv3p = sqrt(e3_2*lambda1m3_2 + 2*e3*(lambda1t3-1) + 1.0);
218 double v1p[3], v3p[3];
219
220 v1p[0]=Vt.at<double>(0)*nv1p, v1p[1]=Vt.at<double>(1)*nv1p, v1p[2]=Vt.at<double>(2)*nv1p;
221 v3p[0]=Vt.at<double>(6)*nv3p, v3p[1]=Vt.at<double>(7)*nv3p, v3p[2]=Vt.at<double>(8)*nv3p;
222
223 /*The eight solutions are
224 (A): tstar = +- (v1p - v3p)/(e1 -e3), n = +- (e1*v3p - e3*v1p)/(e1-e3)
225 (B): tstar = +- (v1p + v3p)/(e1 -e3), n = +- (e1*v3p + e3*v1p)/(e1-e3)
226 */
227 double v1pmv3p[3], v1ppv3p[3];
228 double e1v3me3v1[3], e1v3pe3v1[3];
229 double inv_e1me3 = 1.0/(e1-e3);
230
231 for(int kk=0;kk<3;++kk){
232 v1pmv3p[kk] = v1p[kk]-v3p[kk];
233 v1ppv3p[kk] = v1p[kk]+v3p[kk];
234 }
235
236 for(int kk=0; kk<3; ++kk){
237 double e1v3 = e1*v3p[kk];
238 double e3v1=e3*v1p[kk];
239 e1v3me3v1[kk] = e1v3-e3v1;
240 e1v3pe3v1[kk] = e1v3+e3v1;
241 }
242
243 Vec3d tstar_p, tstar_n;
244 Vec3d n_p, n_n;
245
247 for(int kk=0; kk<3; ++kk) {
248 tstar_p[kk] = v1pmv3p[kk]*inv_e1me3;
249 tstar_n[kk] = -tstar_p[kk];
250 n_p[kk] = e1v3me3v1[kk]*inv_e1me3;
251 n_n[kk] = -n_p[kk];
252 }
253
254 CameraMotion cmotion;
255 //(A) Four different combinations for solution A
256 // (i) (+, +)
257 if (findMotionFrom_tstar_n(tstar_p, n_p, cmotion))
258 camMotions.push_back(cmotion);
259
260 // (ii) (+, -)
261 if (findMotionFrom_tstar_n(tstar_p, n_n, cmotion))
262 camMotions.push_back(cmotion);
263
264 // (iii) (-, +)
265 if (findMotionFrom_tstar_n(tstar_n, n_p, cmotion))
266 camMotions.push_back(cmotion);
267
268 // (iv) (-, -)
269 if (findMotionFrom_tstar_n(tstar_n, n_n, cmotion))
270 camMotions.push_back(cmotion);
273 for(int kk=0;kk<3;++kk){
274 tstar_p[kk] = v1ppv3p[kk]*inv_e1me3;
275 tstar_n[kk] = -tstar_p[kk];
276 n_p[kk] = e1v3pe3v1[kk]*inv_e1me3;
277 n_n[kk] = -n_p[kk];
278 }
279
280 //(B) Four different combinations for solution B
281 // (i) (+, +)
282 if (findMotionFrom_tstar_n(tstar_p, n_p, cmotion))
283 camMotions.push_back(cmotion);
284
285 // (ii) (+, -)
286 if (findMotionFrom_tstar_n(tstar_p, n_n, cmotion))
287 camMotions.push_back(cmotion);
288
289 // (iii) (-, +)
290 if (findMotionFrom_tstar_n(tstar_n, n_p, cmotion))
291 camMotions.push_back(cmotion);
292
293 // (iv) (-, -)
294 if (findMotionFrom_tstar_n(tstar_n, n_n, cmotion))
295 camMotions.push_back(cmotion);
296}

298double HomographyDecompInria::oppositeOfMinor(const Matx33d& M, const int row, const int col)
299{
300 int x1 = col == 0 ? 1 : 0;
301 int x2 = col == 2 ? 1 : 2;
302 int y1 = row == 0 ? 1 : 0;
303 int y2 = row == 2 ? 1 : 2;
304
305 return (M(y1, x2) * M(y2, x1) - M(y1, x1) * M(y2, x2));
306}

308//computes R = H( I - (2/v)*te_star*ne_t )
309void HomographyDecompInria::findRmatFrom_tstar_n(const cv::Vec3d& tstar, const cv::Vec3d& n, const double v, cv::Matx33d& R)
310{
311 Matx31d tstar_m = Matx31d(tstar);
312 Matx31d n_m = Matx31d(n);
313 Matx33d I(1.0, 0.0, 0.0,
314 0.0, 1.0, 0.0,
315 0.0, 0.0, 1.0);
316
317 R = getHnorm() * (I - (2/v) * tstar_m * n_m.t() );
318 if (cv::determinant(R) < 0)
319 {
320 R *= -1;
321 }
322}

324void HomographyDecompInria::decompose(std::vector<CameraMotion>& camMotions)
325{
326 const double epsilon = 0.001;
327 Matx33d S;
328
329 //S = H'H - I
330 S = getHnorm().t() * getHnorm();
331 S(0, 0) -= 1.0;
332 S(1, 1) -= 1.0;
333 S(2, 2) -= 1.0;
334
335 //check if H is rotation matrix
336 if( norm(S, NORM_INF) < epsilon) {
337 CameraMotion motion;
338 motion.R = Matx33d(getHnorm());
339 motion.t = Vec3d(0, 0, 0);
340 motion.n = Vec3d(0, 0, 0);
341 camMotions.push_back(motion);
342 return;
343 }
344
346 Vec3d npa, npb;
347
348 double M00 = oppositeOfMinor(S, 0, 0);
349 double M11 = oppositeOfMinor(S, 1, 1);
350 double M22 = oppositeOfMinor(S, 2, 2);
351
352 double rtM00 = sqrt(M00);
353 double rtM11 = sqrt(M11);
354 double rtM22 = sqrt(M22);
355
356 double M01 = oppositeOfMinor(S, 0, 1);
357 double M12 = oppositeOfMinor(S, 1, 2);
358 double M02 = oppositeOfMinor(S, 0, 2);
359
360 int e12 = signd(M12);
361 int e02 = signd(M02);
362 int e01 = signd(M01);
363
364 double nS00 = abs(S(0, 0));
365 double nS11 = abs(S(1, 1));
366 double nS22 = abs(S(2, 2));
367
368 //find max( |Sii| ), i=0, 1, 2
369 int indx = 0;
370 if(nS00 < nS11){
371 indx = 1;
372 if( nS11 < nS22 )
373 indx = 2;
374 }
375 else {
376 if(nS00 < nS22 )
377 indx = 2;
378 }
379
380 switch (indx) {
381 case 0:
382 npa[0] = S(0, 0), npb[0] = S(0, 0);
383 npa[1] = S(0, 1) + rtM22, npb[1] = S(0, 1) - rtM22;
384 npa[2] = S(0, 2) + e12 * rtM11, npb[2] = S(0, 2) - e12 * rtM11;
385 break;
386 case 1:
387 npa[0] = S(0, 1) + rtM22, npb[0] = S(0, 1) - rtM22;
388 npa[1] = S(1, 1), npb[1] = S(1, 1);
389 npa[2] = S(1, 2) - e02 * rtM00, npb[2] = S(1, 2) + e02 * rtM00;
390 break;
391 case 2:
392 npa[0] = S(0, 2) + e01 * rtM11, npb[0] = S(0, 2) - e01 * rtM11;
393 npa[1] = S(1, 2) + rtM00, npb[1] = S(1, 2) - rtM00;
394 npa[2] = S(2, 2), npb[2] = S(2, 2);
395 break;
396 default:
397 break;
398 }
399
400 double traceS = S(0, 0) + S(1, 1) + S(2, 2);
401 double v = 2.0 * sqrt(1 + traceS - M00 - M11 - M22);
402
403 double ESii = signd(S(indx, indx)) ;
404 double r_2 = 2 + traceS + v;
405 double nt_2 = 2 + traceS - v;
406
407 double r = sqrt(r_2);
408 double n_t = sqrt(nt_2);
409
410 Vec3d na = npa / norm(npa);
411 Vec3d nb = npb / norm(npb);
412
413 double half_nt = 0.5 * n_t;
414 double esii_t_r = ESii * r;
415
416 Vec3d ta_star = half_nt * (esii_t_r * nb - n_t * na);
417 Vec3d tb_star = half_nt * (esii_t_r * na - n_t * nb);
418
419 camMotions.resize(4);
420
421 Matx33d Ra, Rb;
422 Vec3d ta, tb;
423
424 //Ra, ta, na
425 findRmatFrom_tstar_n(ta_star, na, v, Ra);
426 ta = Ra * ta_star;
427
428 camMotions[0].R = Ra;
429 camMotions[0].t = ta;
430 camMotions[0].n = na;
431
432 //Ra, -ta, -na
433 camMotions[1].R = Ra;
434 camMotions[1].t = -ta;
435 camMotions[1].n = -na;
436
437 //Rb, tb, nb
438 findRmatFrom_tstar_n(tb_star, nb, v, Rb);
439 tb = Rb * tb_star;
440
441 camMotions[2].R = Rb;
442 camMotions[2].t = tb;
443 camMotions[2].n = nb;
444
445 //Rb, -tb, -nb
446 camMotions[3].R = Rb;
447 camMotions[3].t = -tb;
448 camMotions[3].n = -nb;
449}

453// function decomposes image-to-image homography to rotation and translation matrices
454int decomposeHomographyMat(InputArray _H,
455 InputArray _K,
456 OutputArrayOfArrays _rotations,
457 OutputArrayOfArrays _translations,
458 OutputArrayOfArrays _normals)
459{
460 using namespace std;
461 using namespace HomographyDecomposition;
462
463 Mat H = _H.getMat().reshape(1, 3);
464 CV_Assert(H.cols == 3 && H.rows == 3);
465
466 Mat K = _K.getMat().reshape(1, 3);
467 CV_Assert(K.cols == 3 && K.rows == 3);
468
469 cv::Ptr<HomographyDecomp> hdecomp(new HomographyDecompInria);
470
471 vector<CameraMotion> motions;
472 hdecomp->decomposeHomography(H, K, motions);
473
474 int nsols = static_cast<int>(motions.size());
475 int depth = CV_64F; //double precision matrices used in CameraMotion struct
476
477 if (_rotations.needed()) {
478 _rotations.create(nsols, 1, depth);
479 for (int k = 0; k < nsols; ++k ) {
480 _rotations.getMatRef(k) = Mat(motions[k].R);
481 }
482 }
483
484 if (_translations.needed()) {
485 _translations.create(nsols, 1, depth);
486 for (int k = 0; k < nsols; ++k ) {
487 _translations.getMatRef(k) = Mat(motions[k].t);
488 }
489 }
490
491 if (_normals.needed()) {
492 _normals.create(nsols, 1, depth);
493 for (int k = 0; k < nsols; ++k ) {
494 _normals.getMatRef(k) = Mat(motions[k].n);
495 }
496 }
497
498 return nsols;
499}

501void filterHomographyDecompByVisibleRefpoints(InputArrayOfArrays _rotations,
502 InputArrayOfArrays _normals,
503 InputArray _beforeRectifiedPoints,
504 InputArray _afterRectifiedPoints,
505 OutputArray _possibleSolutions,
506 InputArray _pointsMask)
507{
508 CV_Assert(_beforeRectifiedPoints.type() == CV_32FC2 && _afterRectifiedPoints.type() == CV_32FC2);
509 CV_Assert(_pointsMask.empty() || _pointsMask.type() == CV_8U || _pointsMask.type() == CV_8S || _pointsMask.type() == CV_Bool);
510
511 Mat beforeRectifiedPoints = _beforeRectifiedPoints.getMat();
512 Mat afterRectifiedPoints = _afterRectifiedPoints.getMat();
513 Mat pointsMask = _pointsMask.getMat();
514 int nsolutions = (int)_rotations.total();
515 int npoints = (int)beforeRectifiedPoints.total();
516 CV_Assert(pointsMask.empty() || pointsMask.checkVector(1, CV_8U) == npoints);
517 const uchar* pointsMaskPtr = pointsMask.data;
518
519 AutoBuffer<uchar> solutionMask(nsolutions, (uchar)1);
520 std::vector<Mat> normals(nsolutions);
521 std::vector<Mat> rotnorm(nsolutions);
522 Mat R;
523
524 for( int i = 0; i < nsolutions; i++ )
525 {
526 _normals.getMat(i).convertTo(normals[i], CV_64F);
527 CV_Assert(normals[i].total() == 3);
528 _rotations.getMat(i).convertTo(R, CV_64F);
529 rotnorm[i] = R*normals[i];
530 CV_Assert(rotnorm[i].total() == 3);
531 }
532
533 for( int j = 0; j < npoints; j++ )
534 {
535 if( !pointsMaskPtr || pointsMaskPtr[j] )
536 {
537 Point2f prevPoint = beforeRectifiedPoints.at<Point2f>(j);
538 Point2f currPoint = afterRectifiedPoints.at<Point2f>(j);
539
540 for( int i = 0; i < nsolutions; i++ )
541 {
542 if( !solutionMask[i] )
543 continue;
544
545 const double* normal_i = normals[i].ptr<double>();
546 const double* rotnorm_i = rotnorm[i].ptr<double>();
547 double prevNormDot = normal_i[0]*prevPoint.x + normal_i[1]*prevPoint.y + normal_i[2];
548 double currNormDot = rotnorm_i[0]*currPoint.x + rotnorm_i[1]*currPoint.y + rotnorm_i[2];
549
550 if (prevNormDot <= 0 || currNormDot <= 0)
551 solutionMask[i] = (uchar)0;
552 }
553 }
554 }
555
556 std::vector<int> possibleSolutions;
557 for( int i = 0; i < nsolutions; i++ )
558 if( solutionMask[i] )
559 possibleSolutions.push_back(i);
560
561 constexpr int cvType = traits::Type<decltype(possibleSolutions)::value_type>::value;
562 Mat(static_cast<int>(possibleSolutions.size()), 1, cvType, possibleSolutions.data()).copyTo(_possibleSolutions);
563}
