60bool DeviceHandler::init(MFXVideoSession_WRAP &session)
61{
62 mfxStatus res = MFX_ERR_NONE;
63 mfxVersion ver = { {19, 1} };
64
65#ifdef HAVE_ONEVPL
66 res = session.CreateSession();
67 DBG(cout << "MFX CreateSession: " << res << endl);
68#else
69 mfxIMPL impl = getImpl();
70
71 res = session.Init(impl, &ver);
72 DBG(cout << "MFX SessionInit: " << res << endl);
73
74 res = session.QueryIMPL(&impl);
75 DBG(cout << "MFX QueryIMPL: " << res << " => " << asHex(impl) << endl);
76#endif
77
78 res = session.QueryVersion(&ver);
79 DBG(cout << "MFX QueryVersion: " << res << " => " << ver.Major << "." << ver.Minor << endl);
80
81 if (res != MFX_ERR_NONE)
82 return false;
83
84 return initDeviceSession(session);
85}

138SurfacePool::SurfacePool(ushort width_, ushort height_, ushort count, const mfxFrameInfo &frameInfo, uchar bpp)
139 : width(alignSize(width_, 32)),
140 height(alignSize(height_, 32)),
141 oneSize(width * height * bpp / 8),
142 buffers(count * oneSize),
143 surfaces(count)
144{
145 for(int i = 0; i < count; ++i)
146 {
147 mfxFrameSurface1 &surface = surfaces[i];
148 uint8_t * dataPtr = buffers.data() + oneSize * i;
149 memset(&surface, 0, sizeof(mfxFrameSurface1));
150 surface.Info = frameInfo;
151 surface.Data.Y = dataPtr;
152 surface.Data.UV = dataPtr + width * height;
153 surface.Data.PitchLow = width & 0xFFFF;
154 surface.Data.PitchHigh = (width >> 16) & 0xFFFF;
155 DBG(cout << "allocate surface " << (void*)&surface << ", Y = " << (void*)dataPtr << " (" << width << "x" << height << ")" << endl);
156 }
157 DBG(cout << "Allocated: " << endl
158 << "- surface data: " << buffers.size() << " bytes" << endl
159 << "- surface headers: " << surfaces.size() * sizeof(mfxFrameSurface1) << " bytes" << endl);
160}

174mfxFrameSurface1 *SurfacePool::getFreeSurface()
175{
176 const int64 start = cv::getTickCount();
177 do
178 {
179 for(std::vector<mfxFrameSurface1>::iterator i = surfaces.begin(); i != surfaces.end(); ++i)
180 if (!i->Data.Locked)
181 return &(*i);
182 sleep_ms(10);
183 }
184 while((cv::getTickCount() - start) / cv::getTickFrequency() < getPoolTimeoutSec()); // seconds
185 DBG(cout << "No free surface!" << std::endl);
186 return 0;
187}

216bool ReadBitstream::read()
217{
218 memmove(stream.Data, stream.Data + stream.DataOffset, stream.DataLength);
219 stream.DataOffset = 0;
220 input.read((char*)(stream.Data + stream.DataLength), stream.MaxLength - stream.DataLength);
221 if (input.eof() || input.good())
222 {
223 mfxU32 bytesRead = (mfxU32)input.gcount();
224 if (bytesRead > 0)
225 {
226 stream.DataLength += bytesRead;
227 DBG(cout << "read " << bytesRead << " bytes" << endl);
228 return true;
229 }
230 }
231 return false;
232}
