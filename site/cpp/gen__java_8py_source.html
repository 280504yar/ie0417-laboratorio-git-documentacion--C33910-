165 def parseName(self, name, namespaces):
166 '''
167 input: full name and available namespaces
168 returns: (namespace, classpath, classname, name)
169 '''
170 name = name[name.find(" ")+1:].strip() # remove struct/class/const prefix
171 parent = name[:name.rfind('.')].strip()
172 if len(parent) == 0:
173 parent = None
174 spaceName = ""
175 localName = name # <classes>.<name>
176 for namespace in sorted(namespaces, key=len, reverse=True):
177 if name.startswith(namespace + "."):
178 spaceName = namespace
179 localName = name.replace(namespace + ".", "")
180 break
181 pieces = localName.split(".")
182 if len(pieces) > 2: # <class>.<class>.<class>.<name>
183 return name, parent, spaceName, ".".join(pieces[:-1]), pieces[-2], pieces[-1]
184 elif len(pieces) == 2: # <class>.<name>
185 return name, parent, spaceName, pieces[0], pieces[0], pieces[1]
186 elif len(pieces) == 1: # <name>
187 return name, parent, spaceName, "", "", pieces[0]
188 else:
189 return name, parent, spaceName, "", "" # error?!
