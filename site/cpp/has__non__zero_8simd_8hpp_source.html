26static bool funcname( const T* src, size_t len ) \
27{ \
28 size_t i = 0; \
29 SIMD_ONLY( \
30 const int vlanes = VTraits<VT>::vlanes(); \
31 VT v_zero = vx_setzero_##suffix(); \
32 for (i = 0; i + vlanes*8 <= len; i += vlanes*8) \
33 { \
34 VT x0 = vx_load(src + i); \
35 VT x1 = vx_load(src + i + vlanes); \
36 VT x2 = vx_load(src + i + vlanes*2); \
37 VT x3 = vx_load(src + i + vlanes*3); \
38 VT x4 = vx_load(src + i + vlanes*4); \
39 VT x5 = vx_load(src + i + vlanes*5); \
40 VT x6 = vx_load(src + i + vlanes*6); \
41 VT x7 = vx_load(src + i + vlanes*7); \
42 x0 = v_or(x0, x1); \
43 x2 = v_or(x2, x3); \
44 x4 = v_or(x4, x5); \
45 x6 = v_or(x6, x7); \
46 x0 = v_or(x0, x2); \
47 x4 = v_or(x4, x6); \
48 x0 = v_or(x0, x4); \
49 x0 = cmp_op(x0, v_zero); \
50 if (v_check_any(x0)) \
51 return true; \
52 } \
53 for (; i < len; i += vlanes) \
54 { \
55 if (i + vlanes > len) { \
56 if (i == 0) \
57 break; \
58 i = len - vlanes; \
59 } \
60 VT x0 = vx_load(src + i); \
61 x0 = cmp_op(x0, v_zero); \
62 if (v_check_any(x0)) \
63 return true; \
64 } \
65 v_cleanup();) \
66 for( ; i < len; i++ ) \
67 { \
68 T x = src[i]; \
69 if (scalar_nz_op(x) != 0) \
70 return true; \
71 } \
72 return false; \
73}
