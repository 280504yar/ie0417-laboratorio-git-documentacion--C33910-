34 static std::string normalize(const std::string& text) {
35 std::string out;
36 out.reserve(text.size() + text.size() / 4);
37 for (char c : text) {
38 if (c == ' ')
39 out += GEMMA_METASPACE;
40 else
41 out += c;
42 }
43 return out;
44 }

46 static std::vector<std::string> splitUtf8Chars(const std::string& text) {
47 std::vector<std::string> out;
48 out.reserve(text.size());
49 size_t i = 0;
50 while (i < text.size()) {
51 unsigned char c = static_cast<unsigned char>(text[i]);
52 int len;
53 if (c < 0x80) len = 1;
54 else if (c < 0xE0) len = 2;
55 else if (c < 0xF0) len = 3;
56 else len = 4;
57 if (i + static_cast<size_t>(len) > text.size()) len = 1;
58 out.push_back(text.substr(i, len));
59 i += static_cast<size_t>(len);
60 }
61 return out;
62 }

64 static std::vector<std::string> byteFallback(const std::string& ch) {
65 std::vector<std::string> out;
66 out.reserve(ch.size());
67 char buf[8];
68 for (unsigned char b : ch) {
69 std::snprintf(buf, sizeof(buf), "<0x%02X>", static_cast<unsigned>(b));
70 out.push_back(buf);
71 }
72 return out;
73 }

75 std::vector<int> encodePiece(const std::string& text) const {
76 if (text.empty()) return {};
77
78 std::vector<std::string> pieces;
79 for (const std::string& ch : splitUtf8Chars(text)) {
80 if (pieceToId.count(ch)) {
81 pieces.push_back(ch);
82 } else {
83 for (const std::string& fb : byteFallback(ch))
84 pieces.push_back(fb);
85 }
86 }
87
88 static constexpr uint32_t RANK_INF = std::numeric_limits<uint32_t>::max();
89 while (pieces.size() > 1) {
90 uint32_t best_rank = RANK_INF;
91 int best_i = -1;
92
93 for (int i = 0; i + 1 < static_cast<int>(pieces.size()); ++i) {
94 std::string key;
95 key.reserve(pieces[i].size() + 1 + pieces[i + 1].size());
96 key = pieces[i];
97 key += '\0';
98 key += pieces[i + 1];
99
100 auto it = mergeRanks.find(key);
101 if (it != mergeRanks.end() && it->second < best_rank) {
102 best_rank = it->second;
103 best_i = i;
104 }
105 }
106
107 if (best_i == -1) break;
108
109 pieces[best_i] += pieces[best_i + 1];
110 pieces.erase(pieces.begin() + best_i + 1);
111 }
112
113 std::vector<int> ids;
114 ids.reserve(pieces.size());
115 for (const std::string& p : pieces) {
116 auto it = pieceToId.find(p);
117 if (it != pieceToId.end()) {
118 ids.push_back(it->second);
119 } else {
120 for (const std::string& fb : byteFallback(p)) {
121 auto it2 = pieceToId.find(fb);
122 if (it2 != pieceToId.end())
123 ids.push_back(it2->second);
124 }
125 }
126 }
127 return ids;
128 }

130 std::vector<int> encode(const std::string& text,
131 const std::unordered_set<std::string>& allowedSpecial = {}) const {
132 std::vector<int> result;
133 size_t pos = 0;
134
135 while (pos < text.size()) {
136 bool foundSpecial = false;
137 for (const auto& kv : specialToId) {
138 const std::string& sp = kv.first;
139 if (!allowedSpecial.count(sp)) continue;
140 if (text.compare(pos, sp.size(), sp) == 0) {
141 result.push_back(kv.second);
142 pos += sp.size();
143 foundSpecial = true;
144 break;
145 }
146 }
147 if (foundSpecial) continue;
148
149 size_t chunkStart = pos;
150 while (pos < text.size()) {
151 bool atSpecial = false;
152 for (const auto& kv : specialToId) {
153 if (allowedSpecial.count(kv.first) &&
154 text.compare(pos, kv.first.size(), kv.first) == 0) {
155 atSpecial = true;
156 break;
157 }
158 }
159 if (atSpecial) break;
160 ++pos;
161 }
162
163 std::string chunk = text.substr(chunkStart, pos - chunkStart);
164 std::string norm = normalize(chunk);
165 auto ids = encodePiece(norm);
166 result.insert(result.end(), ids.begin(), ids.end());
167 }
168
169 return result;
170 }

172 std::string decode(const std::vector<int>& ids) const {
173 std::string raw;
174 raw.reserve(ids.size() * 4);
175 for (int id : ids) {
176 auto sp = idToSpecial.find(id);
177 if (sp != idToSpecial.end()) {
178 raw += sp->second;
179 continue;
180 }
181 if (id >= 0 && id < static_cast<int>(idToPiece.size()))
182 raw += idToPiece[id];
183 }
184
185 std::string out;
186 out.reserve(raw.size());
187 size_t i = 0;
188 while (i < raw.size()) {
189 if (raw[i] == '<' && i + 5 < raw.size() &&
190 raw[i + 1] == '0' && raw[i + 2] == 'x' &&
191 raw[i + 5] == '>') {
192 char hi = raw[i + 3], lo = raw[i + 4];
193 auto hexVal = [](char c) -> int {
194 if (c >= '0' && c <= '9') return c - '0';
195 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
196 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
197 return -1;
198 };
199 int hv = hexVal(hi), lv = hexVal(lo);
200 if (hv >= 0 && lv >= 0) {
201 out += static_cast<char>((hv << 4) | lv);
202 i += 6;
203 continue;
204 }
205 }
206 if (i + 2 < raw.size() &&
207 static_cast<unsigned char>(raw[i]) == 0xE2 &&
208 static_cast<unsigned char>(raw[i + 1]) == 0x96 &&
209 static_cast<unsigned char>(raw[i + 2]) == 0x81) {
210 out += ' ';
211 i += 3;
212 continue;
213 }
214 out += raw[i++];
215 }
216
217 if (!out.empty() && out[0] == ' ')
218 out.erase(0, 1);
219
220 return out;
221 }
