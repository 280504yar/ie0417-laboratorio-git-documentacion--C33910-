84 void compute(bool trans_a, bool trans_b, MatShape A_shape, MatShape B_shape, MatShape C_shape) {
85 auto A_ndims = A_shape.size(), B_ndims = B_shape.size(), C_ndims = C_shape.size();
86 int ma = A_shape[A_ndims - 2], na = A_shape.back();
87 int mb = B_shape[B_ndims - 2], nb = B_shape.back();
88 lda0 = na, lda1 = 1;
89 ldb0 = nb, ldb1 = 1;
90 ldc = C_shape.back();
91
92 M = trans_a ? na : ma;
93 N = trans_b ? mb : nb;
94 K = trans_a ? ma : na;
95
96 if (trans_a) {
97 std::swap(lda0, lda1);
98 }
99 if (trans_b) {
100 std::swap(ldb0, ldb1);
101 }
102
103 // compute offsets
104 auto batch_ndims = C_ndims - 2;
105
106 batch = total(C_shape, 0, batch_ndims);
107
108 A_offsets.resize(batch, 0);
109 B_offsets.resize(batch, 0);
110 C_offsets.resize(batch, 0);
111 A_rows.resize(batch, 0);
112 B_rows.resize(batch, 0);
113 C_rows.resize(batch, 0);
114
115 // build C_offsets
116 size_t C_step = total(C_shape, C_ndims - 2);
117
118 MatShape A_broadcast_shape(C_ndims, 1);
119 std::memcpy(A_broadcast_shape.data() + (C_ndims - A_ndims), A_shape.data(), A_ndims * sizeof(int));
120 MatShape B_broadcast_shape(C_shape.size(), 1);
121 std::memcpy(B_broadcast_shape.data() + (C_ndims - B_ndims), B_shape.data(), B_shape.size() * sizeof(int));
122 std::vector<size_t> A_steps(C_ndims, 1), B_steps(C_ndims, 1);
123 for (int i = C_ndims - 2; i >= 0; i--) {
124 A_steps[i] = A_steps[i + 1] * A_broadcast_shape[i + 1];
125 B_steps[i] = B_steps[i + 1] * B_broadcast_shape[i + 1];
126 }
127 size_t t, idx;
128 for (size_t i = 0; i < batch; i++) {
129 C_offsets[i] = i * C_step;
130 C_rows[i] = i;
131
132 size_t A_offset = 0, B_offset = 0;
133 t = i;
134 for (int j = batch_ndims - 1; j >= 0; j--) {
135 idx = t / C_shape[j];
136 int idx_offset = (int)(t - idx * C_shape[j]);
137 A_offset += A_broadcast_shape[j] == 1 ? 0 : idx_offset * A_steps[j];
138 B_offset += B_broadcast_shape[j] == 1 ? 0 : idx_offset * B_steps[j];
139 t = idx;
140 }
141 A_offsets[i] = A_offset;
142 B_offsets[i] = B_offset;
143 A_rows[i] = A_offset / (M * K);
144 B_rows[i] = B_offset / (N * K);
145 }
146 }
147
148 // only run after compute
149 void updatePackedBOffsets(size_t packed_B_size) {
150 size_t packed_B_inner_size = packed_B_size / batch;
151 packed_B_offsets.resize(B_offsets.size());
152 for (size_t i = 0; i < packed_B_offsets.size(); i++) {
153 packed_B_offsets[i] = (B_offsets[i] / (N * K)) * packed_B_inner_size;
154 }
155 }
