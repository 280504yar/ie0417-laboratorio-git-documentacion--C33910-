25template<typename MatType>
26class Submap
27{
28public:
29 struct PoseConstraint
30 {
31 Affine3f estimatedPose;
32 int weight;
33
34 PoseConstraint() : weight(0){};
35
36 void accumulatePose(const Affine3f& _pose, int _weight = 1)
37 {
38 DualQuatf accPose = DualQuatf::createFromAffine3(estimatedPose) * float(weight) + DualQuatf::createFromAffine3(_pose) * float(_weight);
39 weight += _weight;
40 accPose = accPose / float(weight);
41 estimatedPose = accPose.toAffine3();
42 }
43 };
44 typedef std::map<int, PoseConstraint> Constraints;
45
46 Submap(int _id, const VolumeSettings& settings, const cv::Affine3f& _pose = cv::Affine3f::Identity(),
47 int _startFrameId = 0)
48 : id(_id), pose(_pose), cameraPose(Affine3f::Identity()), startFrameId(_startFrameId),
49 volume(VolumeType::HashTSDF, settings)
50 { }
51 virtual ~Submap() = default;
52
53 virtual void integrate(InputArray _depth, const int currframeId);
54 virtual void raycast(const cv::Affine3f& cameraPose, cv::Size frameSize, cv::Matx33f K,
55 OutputArray points = noArray(), OutputArray normals = noArray());
56
57 virtual int getTotalAllocatedBlocks() const { return int(volume.getTotalVolumeUnits()); };
58 virtual int getVisibleBlocks(int currFrameId) const
59 {
60 CV_Assert(currFrameId >= startFrameId);
61 //return volume.getVisibleBlocks(currFrameId, FRAME_VISIBILITY_THRESHOLD);
62 return volume.getVisibleBlocks();
63
64 }
65
66 float calcVisibilityRatio(int currFrameId) const
67 {
68 int allocate_blocks = getTotalAllocatedBlocks();
69 int visible_blocks = getVisibleBlocks(currFrameId);
70 return float(visible_blocks) / float(allocate_blocks);
71 }
72
73 // Adding new Edge for Loop Closure Detection. Return true or false to indicate whether adding success.
74 bool addEdgeToSubmap(const int tarSubmapID, const Affine3f& tarPose);
75
77 virtual void setStartFrameId(int _startFrameId) { startFrameId = _startFrameId; };
78 virtual void setStopFrameId(int _stopFrameId) { stopFrameId = _stopFrameId; };
79
80 void composeCameraPose(const cv::Affine3f& _relativePose) { cameraPose = cameraPose * _relativePose; }
81 PoseConstraint& getConstraint(const int _id)
82 {
84 return constraints[_id];
85 }

87public:

89 cv::Affine3f pose;

136template<typename MatType>
137bool Submap<MatType>::addEdgeToSubmap(const int tarSubmapID, const Affine3f& tarPose)
138{
139 auto iter = constraints.find(tarSubmapID);
140
141 // if there is NO edge of currSubmap to tarSubmap.
142 if(iter == constraints.end())
143 {
144 // Frome pose to tarPose transformation
145 Affine3f estimatePose = tarPose * pose.inv();
146
147 // Create new Edge.
148 PoseConstraint& preConstrain = getConstraint(tarSubmapID);
149 preConstrain.accumulatePose(estimatePose, 1);
150
151 return true;
152 } else
153 {
154 return false;
155 }

287template<typename MatType>
288int SubmapManager<MatType>::estimateConstraint(int fromSubmapId, int toSubmapId, int& inliers, Affine3f& inlierPose)
289{
290 static constexpr int MAX_ITER = 10;
291 static constexpr float CONVERGE_WEIGHT_THRESHOLD = 0.01f;
292 static constexpr float INLIER_WEIGHT_THRESH = 0.8f;
293 static constexpr int MIN_INLIERS = 10;
294 static constexpr int MAX_TRACKING_ATTEMPTS = 25;
295
297 auto huberWeight = [](float residual, float thresh = 0.1f) -> float {
298 float rAbs = abs(residual);
299 if (rAbs < thresh)
300 return 1.0;
301 float numerator = sqrt(2 * thresh * rAbs - thresh * thresh);
302 return numerator / rAbs;
303 };
304
305 Ptr<SubmapT> fromSubmap = getSubmap(fromSubmapId);
306 Ptr<SubmapT> toSubmap = getSubmap(toSubmapId);
307 ActiveSubmapData& fromSubmapData = activeSubmaps.at(fromSubmapId);
308
309 Affine3f TcameraToFromSubmap = fromSubmap->cameraPose;
310 Affine3f TcameraToToSubmap = toSubmap->cameraPose;
311
312 // FromSubmap -> ToSubmap transform
313 Affine3f candidateConstraint = TcameraToToSubmap * TcameraToFromSubmap.inv();
314 fromSubmapData.trackingAttempts++;
315 fromSubmapData.constraints.push_back(candidateConstraint);
316
317 std::vector<float> weights(fromSubmapData.constraints.size() + 1, 1.0f);
318
319 Affine3f prevConstraint = fromSubmap->getConstraint(toSubmap->id).estimatedPose;
320 int prevWeight = fromSubmap->getConstraint(toSubmap->id).weight;
321
322 // Iterative reweighted least squares with huber threshold to find the inliers in the past observations
323 Vec6f meanConstraint;
324 float sumWeight = 0.0f;
325 for (int i = 0; i < MAX_ITER; i++)
326 {
327 Vec6f constraintVec;
328 for (int j = 0; j < int(weights.size() - 1); j++)
329 {
330 Affine3f currObservation = fromSubmapData.constraints[j];
331 cv::vconcat(currObservation.rvec(), currObservation.translation(), constraintVec);
332 meanConstraint += weights[j] * constraintVec;
333 sumWeight += weights[j];
334 }
335 // Heavier weight given to the estimatedPose
336 cv::vconcat(prevConstraint.rvec(), prevConstraint.translation(), constraintVec);
337 meanConstraint += weights.back() * prevWeight * constraintVec;
338 sumWeight += prevWeight;
339 meanConstraint /= float(sumWeight);
340
341 float residual = 0.0f;
342 float diff = 0.0f;
343 for (int j = 0; j < int(weights.size()); j++)
344 {
345 int w;
346 if (j == int(weights.size() - 1))
347 {
348 cv::vconcat(prevConstraint.rvec(), prevConstraint.translation(), constraintVec);
349 w = prevWeight;
350 }
351 else
352 {
353 Affine3f currObservation = fromSubmapData.constraints[j];
354 cv::vconcat(currObservation.rvec(), currObservation.translation(), constraintVec);
355 w = 1;
356 }
357
358 cv::Vec6f residualVec = (constraintVec - meanConstraint);
359 residual = float(norm(residualVec));
360 float newWeight = huberWeight(residual);
361 diff += w * abs(newWeight - weights[j]);
362 weights[j] = newWeight;
363 }
364
365 if (diff / (prevWeight + weights.size() - 1) < CONVERGE_WEIGHT_THRESHOLD)
366 break;
367 }
368
369 int localInliers = 0;
370 DualQuatf inlierConstraint;
371 for (int i = 0; i < int(weights.size()); i++)
372 {
373 if (weights[i] > INLIER_WEIGHT_THRESH)
374 {
375 localInliers++;
376 if (i == int(weights.size() - 1))
377 inlierConstraint += DualQuatf::createFromMat(prevConstraint.matrix);
378 else
379 inlierConstraint += DualQuatf::createFromMat(fromSubmapData.constraints[i].matrix);
380 }
381 }
382 inlierConstraint = inlierConstraint * 1.0f/float(max(localInliers, 1));
383 inlierPose = inlierConstraint.toAffine3();
384 inliers = localInliers;
385
386 if (inliers >= MIN_INLIERS)
387 {
388 return 1;
389 }
390 if(fromSubmapData.trackingAttempts - inliers > (MAX_TRACKING_ATTEMPTS - MIN_INLIERS))
391 {
392 return -1;
393 }
394
395 return 0;
396}

398template<typename MatType>
399bool SubmapManager<MatType>::shouldChangeCurrSubmap(int _frameId, int toSubmapId)
400{
401 auto toSubmap = getSubmap(toSubmapId);
402 auto toSubmapData = activeSubmaps.at(toSubmapId);
403 auto currActiveSubmap = getCurrentSubmap();
404
405 int blocksInNewMap = toSubmap->getTotalAllocatedBlocks();
406 float newRatio = toSubmap->calcVisibilityRatio(_frameId);
407
408 float currRatio = currActiveSubmap->calcVisibilityRatio(_frameId);
409
411 if (blocksInNewMap <= 0)
412 return false;
413 if ((newRatio > currRatio) && (toSubmapData.type == Type::NEW))
414 return true;
415
416 return false;
417}

428template<typename MatType>
429bool SubmapManager<MatType>::updateMap(int _frameId, const OdometryFrame& _frame)
430{
431 bool mapUpdated = false;
432 int changedCurrentMapId = -1;
433
434 const int currSubmapId = getCurrentSubmap()->id;
435
436 for (auto& it : activeSubmaps)
437 {
438 int submapId = it.first;
439 auto& submapData = it.second;
440 if (submapData.type == Type::NEW || submapData.type == Type::LOOP_CLOSURE)
441 {
442 // Check with previous estimate
443 int inliers;
444 Affine3f inlierPose;
445 int constraintUpdate = estimateConstraint(submapId, currSubmapId, inliers, inlierPose);
446 if (constraintUpdate == 1)
447 {
448 typename SubmapT::PoseConstraint& submapConstraint = getSubmap(submapId)->getConstraint(currSubmapId);
449 submapConstraint.accumulatePose(inlierPose, inliers);
450 submapData.constraints.clear();
451 submapData.trackingAttempts = 0;
452
453 if (shouldChangeCurrSubmap(_frameId, submapId))
454 {
455 changedCurrentMapId = submapId;
456 }
457 mapUpdated = true;
458 }
459 else if(constraintUpdate == -1)
460 {
461 submapData.type = Type::LOST;
462 }
463 }
464 }
465
466 std::vector<int> createNewConstraintsList;
467 for (auto& it : activeSubmaps)
468 {
469 int submapId = it.first;
470 auto& submapData = it.second;
471
472 if (submapId == changedCurrentMapId)
473 {
474 submapData.type = Type::CURRENT;
475 }
476 if ((submapData.type == Type::CURRENT) && (changedCurrentMapId >= 0) && (submapId != changedCurrentMapId))
477 {
478 submapData.type = Type::LOST;
479 createNewConstraintsList.push_back(submapId);
480 }
481 if ((submapData.type == Type::NEW || submapData.type == Type::LOOP_CLOSURE) && (changedCurrentMapId >= 0))
482 {
484 submapData.type = Type::LOST;
485 createNewConstraintsList.push_back(submapId);
486 }
487 }
488
489 for (typename IdToActiveSubmaps::iterator it = activeSubmaps.begin(); it != activeSubmaps.end();)
490 {
491 auto& submapData = it->second;
492 if (submapData.type == Type::LOST)
493 it = activeSubmaps.erase(it);
494 else
495 it++;
496 }
497
498 for (std::vector<int>::const_iterator it = createNewConstraintsList.begin(); it != createNewConstraintsList.end(); ++it)
499 {
500 int dataId = *it;
501 ActiveSubmapData newSubmapData;
502 newSubmapData.trackingAttempts = 0;
503 newSubmapData.type = Type::LOOP_CLOSURE;
504 activeSubmaps[dataId] = newSubmapData;
505 }
506
507 if (shouldCreateSubmap(_frameId))
508 {
509 Ptr<SubmapT> currActiveSubmap = getCurrentSubmap();
510 Affine3f newSubmapPose = currActiveSubmap->pose * currActiveSubmap->cameraPose;
511 int submapId = createNewSubmap(false, _frameId, newSubmapPose);
512 auto newSubmap = getSubmap(submapId);
513 newSubmap->frame = _frame;
514 }
515
516 return mapUpdated;
517}

519template<typename MatType>
520Ptr<detail::PoseGraph> SubmapManager<MatType>::MapToPoseGraph()
521{
522 Ptr<detail::PoseGraph> localPoseGraph = detail::PoseGraph::create();
523
524 for(const auto& currSubmap : submapList)
525 {
526 const typename SubmapT::Constraints& constraintList = currSubmap->constraints;
527 for(const auto& currConstraintPair : constraintList)
528 {
529 // TODO: Handle case with duplicate constraints A -> B and B -> A
530 /* Matx66f informationMatrix = Matx66f::eye() * (currConstraintPair.second.weight/10); */
531 Matx66f informationMatrix = Matx66f::eye();
532 localPoseGraph->addEdge(currSubmap->id, currConstraintPair.first, currConstraintPair.second.estimatedPose, informationMatrix);
533 }
534 }
535
536 for(const auto& currSubmap : submapList)
537 {
538 localPoseGraph->addNode(currSubmap->id, currSubmap->pose, (currSubmap->id == 0));
539 }
540
541 return localPoseGraph;
542}
