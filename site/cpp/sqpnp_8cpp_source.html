74void PoseSolver::solve(InputArray objectPoints, InputArray imagePoints, OutputArrayOfArrays rvecs,
75 OutputArrayOfArrays tvecs)
76{
77 // Input checking
78 int objType = objectPoints.getMat().type();
79 CV_CheckType(objType, objType == CV_32FC3 || objType == CV_64FC3,
80 "Type of objectPoints must be CV_32FC3 or CV_64FC3");
81
82 int imgType = imagePoints.getMat().type();
83 CV_CheckType(imgType, imgType == CV_32FC2 || imgType == CV_64FC2,
84 "Type of imagePoints must be CV_32FC2 or CV_64FC2");
85
86 CV_Assert(objectPoints.rows() == 1 || objectPoints.cols() == 1);
87 CV_Assert(objectPoints.rows() >= 3 || objectPoints.cols() >= 3);
88 CV_Assert(imagePoints.rows() == 1 || imagePoints.cols() == 1);
89 CV_Assert(imagePoints.rows() * imagePoints.cols() == objectPoints.rows() * objectPoints.cols());
90
91 Mat _imagePoints;
92 if (imgType == CV_32FC2)
93 {
94 imagePoints.getMat().convertTo(_imagePoints, CV_64F);
95 }
96 else
97 {
98 _imagePoints = imagePoints.getMat();
99 }
100
101 Mat _objectPoints;
102 if (objType == CV_32FC3)
103 {
104 objectPoints.getMat().convertTo(_objectPoints, CV_64F);
105 }
106 else
107 {
108 _objectPoints = objectPoints.getMat();
109 }
110
111 num_null_vectors_ = -1;
112 num_solutions_ = 0;
113
114 computeOmega(_objectPoints, _imagePoints);
115 solveInternal(_objectPoints);
116
117 int depthRot = rvecs.fixedType() ? rvecs.depth() : CV_64F;
118 int depthTrans = tvecs.fixedType() ? tvecs.depth() : CV_64F;
119
120 rvecs.create(num_solutions_, 1, CV_MAKETYPE(depthRot, rvecs.fixedType() && rvecs.kind() == _InputArray::STD_VECTOR ? 3 : 1));
121 tvecs.create(num_solutions_, 1, CV_MAKETYPE(depthTrans, tvecs.fixedType() && tvecs.kind() == _InputArray::STD_VECTOR ? 3 : 1));
122
123 for (int i = 0; i < num_solutions_; i++)
124 {
125
126 Mat rvec;
127 Mat rotation = Mat(solutions_[i].r_hat).reshape(1, 3);
128 Rodrigues(rotation, rvec);
129
130 rvecs.getMatRef(i) = rvec;
131 tvecs.getMatRef(i) = Mat(solutions_[i].t);
132 }
133}

135void PoseSolver::computeOmega(InputArray objectPoints, InputArray imagePoints)
136{
137 omega_ = cv::Matx<double, 9, 9>::zeros();
138 cv::Matx<double, 3, 9> qa_sum = cv::Matx<double, 3, 9>::zeros();
139
140 cv::Point2d sum_img(0, 0);
141 cv::Point3d sum_obj(0, 0, 0);
142 double sq_norm_sum = 0;
143
144 Mat _imagePoints = imagePoints.getMat();
145 Mat _objectPoints = objectPoints.getMat();
146
147 int n = _objectPoints.cols * _objectPoints.rows;
148
149 for (int i = 0; i < n; i++)
150 {
151 const cv::Point2d& img_pt = _imagePoints.at<cv::Point2d>(i);
152 const cv::Point3d& obj_pt = _objectPoints.at<cv::Point3d>(i);
153
154 sum_img += img_pt;
155 sum_obj += obj_pt;
156
157 const double x = img_pt.x, y = img_pt.y;
158 const double X = obj_pt.x, Y = obj_pt.y, Z = obj_pt.z;
159 double sq_norm = x * x + y * y;
160 sq_norm_sum += sq_norm;
161
162 const double X2 = X * X,
163 XY = X * Y,
164 XZ = X * Z,
165 Y2 = Y * Y,
166 YZ = Y * Z,
167 Z2 = Z * Z;
168
169 omega_(0, 0) += X2;
170 omega_(0, 1) += XY;
171 omega_(0, 2) += XZ;
172 omega_(1, 1) += Y2;
173 omega_(1, 2) += YZ;
174 omega_(2, 2) += Z2;
175
176
177 // Populating this manually saves operations by only calculating upper triangle
178 omega_(0, 6) -= x * X2; omega_(0, 7) -= x * XY; omega_(0, 8) -= x * XZ;
179 omega_(1, 7) -= x * Y2; omega_(1, 8) -= x * YZ;
180 omega_(2, 8) -= x * Z2;
181
182 omega_(3, 6) -= y * X2; omega_(3, 7) -= y * XY; omega_(3, 8) -= y * XZ;
183 omega_(4, 7) -= y * Y2; omega_(4, 8) -= y * YZ;
184 omega_(5, 8) -= y * Z2;
185
186
187 omega_(6, 6) += sq_norm * X2; omega_(6, 7) += sq_norm * XY; omega_(6, 8) += sq_norm * XZ;
188 omega_(7, 7) += sq_norm * Y2; omega_(7, 8) += sq_norm * YZ;
189 omega_(8, 8) += sq_norm * Z2;
190
191 // Compute qa_sum. Certain pairs of elements are equal, so filling them outside the loop saves some operations
192 qa_sum(0, 0) += X; qa_sum(0, 1) += Y; qa_sum(0, 2) += Z;
193
194 qa_sum(0, 6) -= x * X; qa_sum(0, 7) -= x * Y; qa_sum(0, 8) -= x * Z;
195 qa_sum(1, 6) -= y * X; qa_sum(1, 7) -= y * Y; qa_sum(1, 8) -= y * Z;
196
197 qa_sum(2, 6) += sq_norm * X; qa_sum(2, 7) += sq_norm * Y; qa_sum(2, 8) += sq_norm * Z;
198 }
199
200 // Complete qa_sum
201 qa_sum(1, 3) = qa_sum(0, 0); qa_sum(1, 4) = qa_sum(0, 1); qa_sum(1, 5) = qa_sum(0, 2);
202 qa_sum(2, 0) = qa_sum(0, 6); qa_sum(2, 1) = qa_sum(0, 7); qa_sum(2, 2) = qa_sum(0, 8);
203 qa_sum(2, 3) = qa_sum(1, 6); qa_sum(2, 4) = qa_sum(1, 7); qa_sum(2, 5) = qa_sum(1, 8);
204
205
206 // lower triangles of omega_'s off-diagonal blocks (0:2, 6:8), (3:5, 6:8) and (6:8, 6:8)
207 omega_(1, 6) = omega_(0, 7); omega_(2, 6) = omega_(0, 8); omega_(2, 7) = omega_(1, 8);
208 omega_(4, 6) = omega_(3, 7); omega_(5, 6) = omega_(3, 8); omega_(5, 7) = omega_(4, 8);
209 omega_(7, 6) = omega_(6, 7); omega_(8, 6) = omega_(6, 8); omega_(8, 7) = omega_(7, 8);
210
211 // upper triangle of omega_'s block (3:5, 3:5)
212 omega_(3, 3) = omega_(0, 0); omega_(3, 4) = omega_(0, 1); omega_(3, 5) = omega_(0, 2);
213 omega_(4, 4) = omega_(1, 1); omega_(4, 5) = omega_(1, 2);
214 omega_(5, 5) = omega_(2, 2);
215
216 // Mirror omega_'s upper triangle to lower triangle
217 // Note that elements (7, 6), (8, 6) & (8, 7) have already been assigned above
218 omega_(1, 0) = omega_(0, 1);
219 omega_(2, 0) = omega_(0, 2); omega_(2, 1) = omega_(1, 2);
220 omega_(3, 0) = omega_(0, 3); omega_(3, 1) = omega_(1, 3); omega_(3, 2) = omega_(2, 3);
221 omega_(4, 0) = omega_(0, 4); omega_(4, 1) = omega_(1, 4); omega_(4, 2) = omega_(2, 4); omega_(4, 3) = omega_(3, 4);
222 omega_(5, 0) = omega_(0, 5); omega_(5, 1) = omega_(1, 5); omega_(5, 2) = omega_(2, 5); omega_(5, 3) = omega_(3, 5); omega_(5, 4) = omega_(4, 5);
223 omega_(6, 0) = omega_(0, 6); omega_(6, 1) = omega_(1, 6); omega_(6, 2) = omega_(2, 6); omega_(6, 3) = omega_(3, 6); omega_(6, 4) = omega_(4, 6); omega_(6, 5) = omega_(5, 6);
224 omega_(7, 0) = omega_(0, 7); omega_(7, 1) = omega_(1, 7); omega_(7, 2) = omega_(2, 7); omega_(7, 3) = omega_(3, 7); omega_(7, 4) = omega_(4, 7); omega_(7, 5) = omega_(5, 7);
225 omega_(8, 0) = omega_(0, 8); omega_(8, 1) = omega_(1, 8); omega_(8, 2) = omega_(2, 8); omega_(8, 3) = omega_(3, 8); omega_(8, 4) = omega_(4, 8); omega_(8, 5) = omega_(5, 8);
226
227 cv::Matx<double, 3, 3> q;
228 q(0, 0) = n; q(0, 1) = 0; q(0, 2) = -sum_img.x;
229 q(1, 0) = 0; q(1, 1) = n; q(1, 2) = -sum_img.y;
230 q(2, 0) = -sum_img.x; q(2, 1) = -sum_img.y; q(2, 2) = sq_norm_sum;
231
232 double inv_n = 1.0 / n;
233 double detQ = n * (n * sq_norm_sum - sum_img.y * sum_img.y - sum_img.x * sum_img.x);
234 double point_coordinate_variance = detQ * inv_n * inv_n * inv_n;
235
236 CV_Assert(point_coordinate_variance >= POINT_VARIANCE_THRESHOLD);
237
238 Matx<double, 3, 3> q_inv;
239 if (!invertSPD3x3(q, q_inv)) analyticalInverse3x3Symm(q, q_inv);
240
241 p_ = -q_inv * qa_sum;
242
243 omega_ += qa_sum.t() * p_;
244
245#ifdef HAVE_EIGEN
246 // Rank revealing QR nullspace computation with full pivoting.
247 // This is slightly less accurate compared to SVD but x2-x3 faster
248 Eigen::Matrix<double, 9, 9> omega_eig, tmp_eig;
249 cv::cv2eigen(omega_, omega_eig);
250 Eigen::FullPivHouseholderQR<Eigen::Matrix<double, 9, 9> > rrqr(omega_eig);
251 tmp_eig = rrqr.matrixQ();
252 cv::eigen2cv(tmp_eig, u_);
253
254 tmp_eig = rrqr.matrixQR().template triangularView<Eigen::Upper>(); // R
255 Eigen::Matrix<double, 9, 1> S_eig = tmp_eig.diagonal().array().abs();
256 cv::eigen2cv(S_eig, s_);
257#else
258 // Use OpenCV's SVD
259 cv::SVD omega_svd(omega_, cv::SVD::FULL_UV);
260 s_ = omega_svd.w;
261 u_ = cv::Mat(omega_svd.vt.t());
262#if 0
263 // EVD equivalent of the SVD; less accurate
264 cv::eigen(omega_, s_, u_);
265 u_ = u_.t(); // eigenvectors were returned as rows
266#endif // EVD
267
268#endif // HAVE_EIGEN
269
270 CV_Assert(s_(0) >= 1e-7);
271
272 while (s_(7 - num_null_vectors_) < RANK_TOLERANCE) num_null_vectors_++;
273
274 CV_Assert(++num_null_vectors_ <= 6);
275
276 point_mean_ = cv::Vec3d(sum_obj.x / n, sum_obj.y / n, sum_obj.z / n);
277}

279void PoseSolver::solveInternal(InputArray objectPoints)
280{
281 double min_sq_err = std::numeric_limits<double>::max();
282 int num_eigen_points = num_null_vectors_ > 0 ? num_null_vectors_ : 1;
283
284 for (int i = 9 - num_eigen_points; i < 9; i++)
285 {
286 const cv::Matx<double, 9, 1> e = SQRT3 * u_.col(i);
287 double orthogonality_sq_err = orthogonalityError(e);
288
289 SQPSolution solutions[2];
290
291 // If e is orthogonal, we can skip SQP
292 if (orthogonality_sq_err < ORTHOGONALITY_SQUARED_ERROR_THRESHOLD)
293 {
294 solutions[0].r_hat = det3x3(e) * e;
295 solutions[0].t = p_ * solutions[0].r_hat;
296 checkSolution(solutions[0], objectPoints, min_sq_err);
297 }
298 else
299 {
300 Matx<double, 9, 1> r;
301 nearestRotationMatrixFOAM(e, r);
302 solutions[0] = runSQP(r);
303 solutions[0].t = p_ * solutions[0].r_hat;
304 checkSolution(solutions[0], objectPoints, min_sq_err);
305
306 nearestRotationMatrixFOAM(-e, r);
307 solutions[1] = runSQP(r);
308 solutions[1].t = p_ * solutions[1].r_hat;
309 checkSolution(solutions[1], objectPoints, min_sq_err);
310 }
311 }
312
313 int index, c = 1;
314 while ((index = 9 - num_eigen_points - c) > 0 && min_sq_err > 3 * s_[index])
315 {
316 const cv::Matx<double, 9, 1> e = u_.col(index);
317 SQPSolution solutions[2];
318
319 Matx<double, 9, 1> r;
320 nearestRotationMatrixFOAM(e, r);
321 solutions[0] = runSQP(r);
322 solutions[0].t = p_ * solutions[0].r_hat;
323 checkSolution(solutions[0], objectPoints, min_sq_err);
324
325 nearestRotationMatrixFOAM(-e, r);
326 solutions[1] = runSQP(r);
327 solutions[1].t = p_ * solutions[1].r_hat;
328 checkSolution(solutions[1], objectPoints, min_sq_err);
329
330 c++;
331 }
332}

334PoseSolver::SQPSolution PoseSolver::runSQP(const cv::Matx<double, 9, 1>& r0)
335{
336 cv::Matx<double, 9, 1> r = r0;
337
338 double delta_squared_norm = std::numeric_limits<double>::max();
339 cv::Matx<double, 9, 1> delta;
340
341 int step = 0;
342 while (delta_squared_norm > SQP_SQUARED_TOLERANCE && step++ < SQP_MAX_ITERATION)
343 {
344 solveSQPSystem(r, delta);
345 r += delta;
346 delta_squared_norm = cv::norm(delta, cv::NORM_L2SQR);
347 }
348
349 SQPSolution solution;
350
351 double det_r = det3x3(r);
352 if (det_r < 0)
353 {
354 r = -r;
355 det_r = -det_r;
356 }
357
358 if (det_r > SQP_DET_THRESHOLD)
359 {
360 nearestRotationMatrixFOAM(r, solution.r_hat);
361 }
362 else
363 {
364 solution.r_hat = r;
365 }
366
367 return solution;
368}

370void PoseSolver::solveSQPSystem(const cv::Matx<double, 9, 1>& r, cv::Matx<double, 9, 1>& delta)
371{
372 double sqnorm_r1 = r(0) * r(0) + r(1) * r(1) + r(2) * r(2),
373 sqnorm_r2 = r(3) * r(3) + r(4) * r(4) + r(5) * r(5),
374 sqnorm_r3 = r(6) * r(6) + r(7) * r(7) + r(8) * r(8);
375 double dot_r1r2 = r(0) * r(3) + r(1) * r(4) + r(2) * r(5),
376 dot_r1r3 = r(0) * r(6) + r(1) * r(7) + r(2) * r(8),
377 dot_r2r3 = r(3) * r(6) + r(4) * r(7) + r(5) * r(8);
378
379 cv::Matx<double, 9, 3> N;
380 cv::Matx<double, 9, 6> H;
381 cv::Matx<double, 6, 6> JH;
382
383 computeRowAndNullspace(r, H, N, JH);
384
385 cv::Matx<double, 6, 1> g;
386 g(0) = 1 - sqnorm_r1; g(1) = 1 - sqnorm_r2; g(2) = 1 - sqnorm_r3; g(3) = -dot_r1r2; g(4) = -dot_r2r3; g(5) = -dot_r1r3;
387
388 cv::Matx<double, 6, 1> x;
389 x(0) = g(0) / JH(0, 0);
390 x(1) = g(1) / JH(1, 1);
391 x(2) = g(2) / JH(2, 2);
392 x(3) = (g(3) - JH(3, 0) * x(0) - JH(3, 1) * x(1)) / JH(3, 3);
393 x(4) = (g(4) - JH(4, 1) * x(1) - JH(4, 2) * x(2) - JH(4, 3) * x(3)) / JH(4, 4);
394 x(5) = (g(5) - JH(5, 0) * x(0) - JH(5, 2) * x(2) - JH(5, 3) * x(3) - JH(5, 4) * x(4)) / JH(5, 5);
395
396 delta = H * x;
397
398
399 cv::Matx<double, 3, 9> nt_omega = N.t() * omega_;
400 cv::Matx<double, 3, 3> W = nt_omega * N, W_inv;
401
402 analyticalInverse3x3Symm(W, W_inv);
403
404 cv::Matx<double, 3, 1> y = -W_inv * nt_omega * (delta + r);
405 delta += N * y;
406}

408// Inverse of SPD 3x3 A via a lower triangular sqrt-free Cholesky
409// factorization A=L*D*L' (L has ones on its diagonal, D is diagonal).
410//
411// Only the lower triangular part of A is accessed.
412//
413// The function returns true if successful
414//
415// see http://euler.nmt.edu/~brian/ldlt.html
416//
417bool PoseSolver::invertSPD3x3(const cv::Matx<double, 3, 3>& A, cv::Matx<double, 3, 3>& A1)
418{
419 double L[3*3], D[3], v[2], x[3];
420
421 v[0]=D[0]=A(0, 0);
422 if(v[0]<=1E-10) return false;
423 v[1]=1.0/v[0];
424 L[3]=A(1, 0)*v[1];
425 L[6]=A(2, 0)*v[1];
426 //L[0]=1.0;
427 //L[1]=L[2]=0.0;
428
429 v[0]=L[3]*D[0];
430 v[1]=D[1]=A(1, 1)-L[3]*v[0];
431 if(v[1]<=1E-10) return false;
432 L[7]=(A(2, 1)-L[6]*v[0])/v[1];
433 //L[4]=1.0;
434 //L[5]=0.0;
435
436 v[0]=L[6]*D[0];
437 v[1]=L[7]*D[1];
438 D[2]=A(2, 2)-L[6]*v[0]-L[7]*v[1];
439 if(D[2]<=1E-10) return false;
440 //L[8]=1.0;
441
442 D[0]=1.0/D[0];
443 D[1]=1.0/D[1];
444 D[2]=1.0/D[2];
445
446 /* Forward solve Lx = e0 */
447 //x[0]=1.0;
448 x[1]=-L[3];
449 x[2]=-L[6]+L[7]*L[3];
450
451 /* Backward solve D*L'x = y */
452 A1(0, 2)=x[2]=x[2]*D[2];
453 A1(0, 1)=x[1]=x[1]*D[1]-L[7]*x[2];
454 A1(0, 0) = D[0]-L[3]*x[1]-L[6]*x[2];
455
456 /* Forward solve Lx = e1 */
457 //x[0]=0.0;
458 //x[1]=1.0;
459 x[2]=-L[7];
460
461 /* Backward solve D*L'x = y */
462 A1(1, 2)=x[2]=x[2]*D[2];
463 A1(1, 1)=x[1]= D[1]-L[7]*x[2];
464 A1(1, 0) = -L[3]*x[1]-L[6]*x[2];
465
466 /* Forward solve Lx = e2 */
467 //x[0]=0.0;
468 //x[1]=0.0;
469 //x[2]=1.0;
470
471 /* Backward solve D*L'x = y */
472 A1(2, 2)=x[2]=D[2];
473 A1(2, 1)=x[1]= -L[7]*x[2];
474 A1(2, 0) = -L[3]*x[1]-L[6]*x[2];
475
476 return true;
477}

479bool PoseSolver::analyticalInverse3x3Symm(const cv::Matx<double, 3, 3>& Q,
480 cv::Matx<double, 3, 3>& Qinv,
481 const double& threshold)
482{
483 // 1. Get the elements of the matrix
484 double a = Q(0, 0),
485 b = Q(1, 0), d = Q(1, 1),
486 c = Q(2, 0), e = Q(2, 1), f = Q(2, 2);
487
488 // 2. Determinant
489 double t2, t4, t7, t9, t12;
490 t2 = e * e;
491 t4 = a * d;
492 t7 = b * b;
493 t9 = b * c;
494 t12 = c * c;
495 double det = -t4 * f + a * t2 + t7 * f - 2.0 * t9 * e + t12 * d;
496
497 if (fabs(det) < threshold) { cv::invert(Q, Qinv, cv::DECOMP_SVD); return false; } // fall back to pseudoinverse
498
499 // 3. Inverse
500 double t15, t20, t24, t30;
501 t15 = 1.0 / det;
502 t20 = (-b * f + c * e) * t15;
503 t24 = (b * e - c * d) * t15;
504 t30 = (a * e - t9) * t15;
505 Qinv(0, 0) = (-d * f + t2) * t15;
506 Qinv(0, 1) = Qinv(1, 0) = -t20;
507 Qinv(0, 2) = Qinv(2, 0) = -t24;
508 Qinv(1, 1) = -(a * f - t12) * t15;
509 Qinv(1, 2) = Qinv(2, 1) = t30;
510 Qinv(2, 2) = -(t4 - t7) * t15;
511
512 return true;
513}

515void PoseSolver::computeRowAndNullspace(const cv::Matx<double, 9, 1>& r,
516 cv::Matx<double, 9, 6>& H,
517 cv::Matx<double, 9, 3>& N,
518 cv::Matx<double, 6, 6>& K,
519 const double& norm_threshold)
520{
521 H = cv::Matx<double, 9, 6>::zeros();
522
523 // 1. q1
524 double norm_r1 = sqrt(r(0) * r(0) + r(1) * r(1) + r(2) * r(2));
525 double inv_norm_r1 = norm_r1 > 1e-5 ? 1.0 / norm_r1 : 0.0;
526 H(0, 0) = r(0) * inv_norm_r1;
527 H(1, 0) = r(1) * inv_norm_r1;
528 H(2, 0) = r(2) * inv_norm_r1;
529 K(0, 0) = 2 * norm_r1;
530
531 // 2. q2
532 double norm_r2 = sqrt(r(3) * r(3) + r(4) * r(4) + r(5) * r(5));
533 double inv_norm_r2 = 1.0 / norm_r2;
534 H(3, 1) = r(3) * inv_norm_r2;
535 H(4, 1) = r(4) * inv_norm_r2;
536 H(5, 1) = r(5) * inv_norm_r2;
537 K(1, 0) = 0;
538 K(1, 1) = 2 * norm_r2;
539
540 // 3. q3 = (r3'*q2)*q2 - (r3'*q1)*q1 ; q3 = q3/norm(q3)
541 double norm_r3 = sqrt(r(6) * r(6) + r(7) * r(7) + r(8) * r(8));
542 double inv_norm_r3 = 1.0 / norm_r3;
543 H(6, 2) = r(6) * inv_norm_r3;
544 H(7, 2) = r(7) * inv_norm_r3;
545 H(8, 2) = r(8) * inv_norm_r3;
546 K(2, 0) = K(2, 1) = 0;
547 K(2, 2) = 2 * norm_r3;
548
549 // 4. q4
550 double dot_j4q1 = r(3) * H(0, 0) + r(4) * H(1, 0) + r(5) * H(2, 0),
551 dot_j4q2 = r(0) * H(3, 1) + r(1) * H(4, 1) + r(2) * H(5, 1);
552
553 H(0, 3) = r(3) - dot_j4q1 * H(0, 0);
554 H(1, 3) = r(4) - dot_j4q1 * H(1, 0);
555 H(2, 3) = r(5) - dot_j4q1 * H(2, 0);
556 H(3, 3) = r(0) - dot_j4q2 * H(3, 1);
557 H(4, 3) = r(1) - dot_j4q2 * H(4, 1);
558 H(5, 3) = r(2) - dot_j4q2 * H(5, 1);
559 double inv_norm_j4 = 1.0 / sqrt(H(0, 3) * H(0, 3) + H(1, 3) * H(1, 3) + H(2, 3) * H(2, 3) +
560 H(3, 3) * H(3, 3) + H(4, 3) * H(4, 3) + H(5, 3) * H(5, 3));
561
562 H(0, 3) *= inv_norm_j4;
563 H(1, 3) *= inv_norm_j4;
564 H(2, 3) *= inv_norm_j4;
565 H(3, 3) *= inv_norm_j4;
566 H(4, 3) *= inv_norm_j4;
567 H(5, 3) *= inv_norm_j4;
568
569 K(3, 0) = r(3) * H(0, 0) + r(4) * H(1, 0) + r(5) * H(2, 0);
570 K(3, 1) = r(0) * H(3, 1) + r(1) * H(4, 1) + r(2) * H(5, 1);
571 K(3, 2) = 0;
572 K(3, 3) = r(3) * H(0, 3) + r(4) * H(1, 3) + r(5) * H(2, 3) + r(0) * H(3, 3) + r(1) * H(4, 3) + r(2) * H(5, 3);
573
574 // 5. q5
575 double dot_j5q2 = r(6) * H(3, 1) + r(7) * H(4, 1) + r(8) * H(5, 1);
576 double dot_j5q3 = r(3) * H(6, 2) + r(4) * H(7, 2) + r(5) * H(8, 2);
577 double dot_j5q4 = r(6) * H(3, 3) + r(7) * H(4, 3) + r(8) * H(5, 3);
578
579 H(0, 4) = -dot_j5q4 * H(0, 3);
580 H(1, 4) = -dot_j5q4 * H(1, 3);
581 H(2, 4) = -dot_j5q4 * H(2, 3);
582 H(3, 4) = r(6) - dot_j5q2 * H(3, 1) - dot_j5q4 * H(3, 3);
583 H(4, 4) = r(7) - dot_j5q2 * H(4, 1) - dot_j5q4 * H(4, 3);
584 H(5, 4) = r(8) - dot_j5q2 * H(5, 1) - dot_j5q4 * H(5, 3);
585 H(6, 4) = r(3) - dot_j5q3 * H(6, 2); H(7, 4) = r(4) - dot_j5q3 * H(7, 2); H(8, 4) = r(5) - dot_j5q3 * H(8, 2);
586
587 cv::Matx<double, 9, 1> q4 = H.col(4);
588 const double inv_norm_q4 = 1.0 / cv::norm(q4);
589 for (int i = 0; i < 9; ++i)
590 H(i, 4) = q4(i) * inv_norm_q4;
591
592 K(4, 0) = 0;
593 K(4, 1) = r(6) * H(3, 1) + r(7) * H(4, 1) + r(8) * H(5, 1);
594 K(4, 2) = r(3) * H(6, 2) + r(4) * H(7, 2) + r(5) * H(8, 2);
595 K(4, 3) = r(6) * H(3, 3) + r(7) * H(4, 3) + r(8) * H(5, 3);
596 K(4, 4) = r(6) * H(3, 4) + r(7) * H(4, 4) + r(8) * H(5, 4) + r(3) * H(6, 4) + r(4) * H(7, 4) + r(5) * H(8, 4);
597
598
599 // 4. q6
600 double dot_j6q1 = r(6) * H(0, 0) + r(7) * H(1, 0) + r(8) * H(2, 0);
601 double dot_j6q3 = r(0) * H(6, 2) + r(1) * H(7, 2) + r(2) * H(8, 2);
602 double dot_j6q4 = r(6) * H(0, 3) + r(7) * H(1, 3) + r(8) * H(2, 3);
603 double dot_j6q5 = r(0) * H(6, 4) + r(1) * H(7, 4) + r(2) * H(8, 4) + r(6) * H(0, 4) + r(7) * H(1, 4) + r(8) * H(2, 4);
604
605 H(0, 5) = r(6) - dot_j6q1 * H(0, 0) - dot_j6q4 * H(0, 3) - dot_j6q5 * H(0, 4);
606 H(1, 5) = r(7) - dot_j6q1 * H(1, 0) - dot_j6q4 * H(1, 3) - dot_j6q5 * H(1, 4);
607 H(2, 5) = r(8) - dot_j6q1 * H(2, 0) - dot_j6q4 * H(2, 3) - dot_j6q5 * H(2, 4);
608
609 H(3, 5) = -dot_j6q5 * H(3, 4) - dot_j6q4 * H(3, 3);
610 H(4, 5) = -dot_j6q5 * H(4, 4) - dot_j6q4 * H(4, 3);
611 H(5, 5) = -dot_j6q5 * H(5, 4) - dot_j6q4 * H(5, 3);
612
613 H(6, 5) = r(0) - dot_j6q3 * H(6, 2) - dot_j6q5 * H(6, 4);
614 H(7, 5) = r(1) - dot_j6q3 * H(7, 2) - dot_j6q5 * H(7, 4);
615 H(8, 5) = r(2) - dot_j6q3 * H(8, 2) - dot_j6q5 * H(8, 4);
616
617 cv::Matx<double, 9, 1> q5 = H.col(5);
618 const double inv_norm_q5 = 1.0 / cv::norm(q5);
619 for (int i = 0; i < 9; ++i)
620 H(i, 5) = q5(i) * inv_norm_q5;
621
622 K(5, 0) = r(6) * H(0, 0) + r(7) * H(1, 0) + r(8) * H(2, 0);
623 K(5, 1) = 0; K(5, 2) = r(0) * H(6, 2) + r(1) * H(7, 2) + r(2) * H(8, 2);
624 K(5, 3) = r(6) * H(0, 3) + r(7) * H(1, 3) + r(8) * H(2, 3);
625 K(5, 4) = r(6) * H(0, 4) + r(7) * H(1, 4) + r(8) * H(2, 4) + r(0) * H(6, 4) + r(1) * H(7, 4) + r(2) * H(8, 4);
626 K(5, 5) = r(6) * H(0, 5) + r(7) * H(1, 5) + r(8) * H(2, 5) + r(0) * H(6, 5) + r(1) * H(7, 5) + r(2) * H(8, 5);
627
628 // Great! Now H is an orthogonalized, sparse basis of the Jacobian row space and K is filled.
629 //
630 // Now get a projector onto the null space H:
631 const cv::Matx<double, 9, 9> Pn = cv::Matx<double, 9, 9>::eye() - (H * H.t());
632
633 // Now we need to pick 3 columns of P with non-zero norm (> 0.3) and some angle between them (> 0.3).
634 //
635 // Find the 3 columns of Pn with largest norms
636 int index1 = 0,
637 index2 = 0,
638 index3 = 0;
639 double max_norm1 = std::numeric_limits<double>::min();
640 double min_dot12 = std::numeric_limits<double>::max();
641 double min_dot1323 = std::numeric_limits<double>::max();
642
643
644 double col_norms[9];
645 for (int i = 0; i < 9; i++)
646 {
647 col_norms[i] = cv::norm(Pn.col(i));
648 if (col_norms[i] >= norm_threshold)
649 {
650 if (max_norm1 < col_norms[i])
651 {
652 max_norm1 = col_norms[i];
653 index1 = i;
654 }
655 }
656 }
657
658 const cv::Matx<double, 9, 1>& v1 = Pn.col(index1);
659 const double inv_max_norm1 = 1.0 / max_norm1;
660 for (int i = 0; i < 9; ++i)
661 N(i, 0) = v1(i) * inv_max_norm1;
662 col_norms[index1] = -1.0; // mark to avoid use in subsequent loops
663
664 for (int i = 0; i < 9; i++)
665 {
666 //if (i == index1) continue;
667 if (col_norms[i] >= norm_threshold)
668 {
669 double cos_v1_x_col = fabs(Pn.col(i).dot(v1) / col_norms[i]);
670
671 if (cos_v1_x_col <= min_dot12)
672 {
673 index2 = i;
674 min_dot12 = cos_v1_x_col;
675 }
676 }
677 }
678
679 const cv::Matx<double, 9, 1>& v2 = Pn.col(index2);
680 const cv::Matx<double, 9, 1>& n0 = N.col(0);
681 cv::Matx<double, 9, 1> u2 = v2 - v2.dot(n0) * n0;
682 const double inv_norm_u2 = 1.0 / cv::norm(u2);
683 for (int i = 0; i < 9; ++i)
684 N(i, 1) = u2(i) * inv_norm_u2;
685 col_norms[index2] = -1.0; // mark to avoid use in subsequent loops
686
687 for (int i = 0; i < 9; i++)
688 {
689 //if (i == index2 || i == index1) continue;
690 if (col_norms[i] >= norm_threshold)
691 {
692 double inv_norm = 1.0 / col_norms[i];
693 double cos_v1_x_col = fabs(Pn.col(i).dot(v1) * inv_norm);
694 double cos_v2_x_col = fabs(Pn.col(i).dot(v2) * inv_norm);
695
696 if (cos_v1_x_col + cos_v2_x_col <= min_dot1323)
697 {
698 index3 = i;
699 min_dot1323 = cos_v1_x_col + cos_v2_x_col;
700 }
701 }
702 }
703
704 const cv::Matx<double, 9, 1>& v3 = Pn.col(index3);
705 const cv::Matx<double, 9, 1>& n1 = N.col(1);
706 cv::Matx<double, 9, 1> u3 = v3 - (v3.dot(n1) * n1) - (v3.dot(n0) * n0);
707 const double inv_norm_u3 = 1.0 / cv::norm(u3);
708 for (int i = 0; i < 9; ++i)
709 N(i, 2) = u3(i) * inv_norm_u3;
710}

712// if e = u*w*vt then r=u*diag([1, 1, det(u)*det(v)])*vt
713void PoseSolver::nearestRotationMatrixSVD(const cv::Matx<double, 9, 1>& e,
714 cv::Matx<double, 9, 1>& r)
715{
716 cv::Matx<double, 3, 3> e33 = e.reshape<3, 3>();
717 cv::SVD e33_svd(e33, cv::SVD::FULL_UV);
718 double detuv = cv::determinant(e33_svd.u)*cv::determinant(e33_svd.vt);
719 cv::Matx<double, 3, 3> diag = cv::Matx33d::eye();
720 diag(2, 2) = detuv;
721 cv::Matx<double, 3, 3> r33 = cv::Mat(e33_svd.u*diag*e33_svd.vt);
722 r = r33.reshape<9, 1>();
723}

725// Faster nearest rotation computation based on FOAM. See M. Lourakis: "An Efficient Solution to Absolute Orientation", ICPR 2016
726// and M. Lourakis, G. Terzakis: "Efficient Absolute Orientation Revisited", IROS 2018.
727/* Solve the nearest orthogonal approximation problem
728 * i.e., given e, find R minimizing ||R-e||_F
729 *
730 * The computation borrows from Markley's FOAM algorithm
731 * "Attitude Determination Using Vector Observations: A Fast Optimal Matrix Algorithm", J. Astronaut. Sci. 1993.
732 *
733 * See also M. Lourakis: "An Efficient Solution to Absolute Orientation", ICPR 2016
734 *
735 * Copyright (C) 2019 Manolis Lourakis (lourakis **at** ics forth gr)
736 * Institute of Computer Science, Foundation for Research & Technology - Hellas
737 * Heraklion, Crete, Greece.
738 */
739void PoseSolver::nearestRotationMatrixFOAM(const cv::Matx<double, 9, 1>& e,
740 cv::Matx<double, 9, 1>& r)
741{
742 int i;
743 double l, lprev, det_e, e_sq, adj_e_sq, adj_e[9];
744
745 // det(e)
746 det_e = ( e(0) * e(4) * e(8) - e(0) * e(5) * e(7) - e(1) * e(3) * e(8) ) + ( e(2) * e(3) * e(7) + e(1) * e(6) * e(5) - e(2) * e(6) * e(4) );
747 if (fabs(det_e) < 1E-04) { // singular, handle it with SVD
748 PoseSolver::nearestRotationMatrixSVD(e, r);
749 return;
750 }
751
752 // e's adjoint
753 adj_e[0] = e(4) * e(8) - e(5) * e(7); adj_e[1] = e(2) * e(7) - e(1) * e(8); adj_e[2] = e(1) * e(5) - e(2) * e(4);
754 adj_e[3] = e(5) * e(6) - e(3) * e(8); adj_e[4] = e(0) * e(8) - e(2) * e(6); adj_e[5] = e(2) * e(3) - e(0) * e(5);
755 adj_e[6] = e(3) * e(7) - e(4) * e(6); adj_e[7] = e(1) * e(6) - e(0) * e(7); adj_e[8] = e(0) * e(4) - e(1) * e(3);
756
757 // ||e||^2, ||adj(e)||^2
758 e_sq = ( e(0) * e(0) + e(1) * e(1) + e(2) * e(2) ) + ( e(3) * e(3) + e(4) * e(4) + e(5) * e(5) ) + ( e(6) * e(6) + e(7) * e(7) + e(8) * e(8) );
759 adj_e_sq = ( adj_e[0] * adj_e[0] + adj_e[1] * adj_e[1] + adj_e[2] * adj_e[2] ) + ( adj_e[3] * adj_e[3] + adj_e[4] * adj_e[4] + adj_e[5] * adj_e[5] ) + ( adj_e[6] * adj_e[6] + adj_e[7] * adj_e[7] + adj_e[8] * adj_e[8] );
760
761 // compute l_max with Newton-Raphson from FOAM's characteristic polynomial, i.e. eq.(23) - (26)
762 l = 0.5*(e_sq + 3.0); // 1/2*(trace(mat(e)*mat(e)') + trace(eye(3)))
763 if (det_e < 0.0) l = -l;
764 for (i = 15, lprev = 0.0; fabs(l - lprev) > 1E-12 * fabs(lprev) && i > 0; --i) {
765 double tmp, p, pp;
766
767 tmp = (l * l - e_sq);
768 p = (tmp * tmp - 8.0 * l * det_e - 4.0 * adj_e_sq);
769 pp = 8.0 * (0.5 * tmp * l - det_e);
770
771 lprev = l;
772 l -= p / pp;
773 }
774
775 // the rotation matrix equals ((l^2 + e_sq)*e + 2*l*adj(e') - 2*e*e'*e) / (l*(l*l-e_sq) - 2*det(e)), i.e. eq.(14) using (18), (19)
776 {
777 // compute (l^2 + e_sq)*e
778 double tmp[9], e_et[9], denom;
779 const double a = l * l + e_sq;
780
781 // e_et=e*e'
782 e_et[0] = e(0) * e(0) + e(1) * e(1) + e(2) * e(2);
783 e_et[1] = e(0) * e(3) + e(1) * e(4) + e(2) * e(5);
784 e_et[2] = e(0) * e(6) + e(1) * e(7) + e(2) * e(8);
785
786 e_et[3] = e_et[1];
787 e_et[4] = e(3) * e(3) + e(4) * e(4) + e(5) * e(5);
788 e_et[5] = e(3) * e(6) + e(4) * e(7) + e(5) * e(8);
789
790 e_et[6] = e_et[2];
791 e_et[7] = e_et[5];
792 e_et[8] = e(6) * e(6) + e(7) * e(7) + e(8) * e(8);
793
794 // tmp=e_et*e
795 tmp[0] = e_et[0] * e(0) + e_et[1] * e(3) + e_et[2] * e(6);
796 tmp[1] = e_et[0] * e(1) + e_et[1] * e(4) + e_et[2] * e(7);
797 tmp[2] = e_et[0] * e(2) + e_et[1] * e(5) + e_et[2] * e(8);
798
799 tmp[3] = e_et[3] * e(0) + e_et[4] * e(3) + e_et[5] * e(6);
800 tmp[4] = e_et[3] * e(1) + e_et[4] * e(4) + e_et[5] * e(7);
801 tmp[5] = e_et[3] * e(2) + e_et[4] * e(5) + e_et[5] * e(8);
802
803 tmp[6] = e_et[6] * e(0) + e_et[7] * e(3) + e_et[8] * e(6);
804 tmp[7] = e_et[6] * e(1) + e_et[7] * e(4) + e_et[8] * e(7);
805 tmp[8] = e_et[6] * e(2) + e_et[7] * e(5) + e_et[8] * e(8);
806
807 // compute R as (a*e + 2*(l*adj(e)' - tmp))*denom; note that adj(e')=adj(e)'
808 denom = l * (l * l - e_sq) - 2.0 * det_e;
809 denom = 1.0 / denom;
810 r(0) = (a * e(0) + 2.0 * (l * adj_e[0] - tmp[0])) * denom;
811 r(1) = (a * e(1) + 2.0 * (l * adj_e[3] - tmp[1])) * denom;
812 r(2) = (a * e(2) + 2.0 * (l * adj_e[6] - tmp[2])) * denom;
813
814 r(3) = (a * e(3) + 2.0 * (l * adj_e[1] - tmp[3])) * denom;
815 r(4) = (a * e(4) + 2.0 * (l * adj_e[4] - tmp[4])) * denom;
816 r(5) = (a * e(5) + 2.0 * (l * adj_e[7] - tmp[5])) * denom;
817
818 r(6) = (a * e(6) + 2.0 * (l * adj_e[2] - tmp[6])) * denom;
819 r(7) = (a * e(7) + 2.0 * (l * adj_e[5] - tmp[7])) * denom;
820 r(8) = (a * e(8) + 2.0 * (l * adj_e[8] - tmp[8])) * denom;
821 }
822}

824double PoseSolver::det3x3(const cv::Matx<double, 9, 1>& e)
825{
826 return ( e(0) * e(4) * e(8) + e(1) * e(5) * e(6) + e(2) * e(3) * e(7) )
827 - ( e(6) * e(4) * e(2) + e(7) * e(5) * e(0) + e(8) * e(3) * e(1) );

830inline bool PoseSolver::positiveDepth(const SQPSolution& solution) const
831{
832 const cv::Matx<double, 9, 1>& r = solution.r_hat;
833 const cv::Matx<double, 3, 1>& t = solution.t;
834 const cv::Vec3d& mean = point_mean_;
835 return (r(6) * mean(0) + r(7) * mean(1) + r(8) * mean(2) + t(2) > 0);
836}

838inline bool PoseSolver::positiveMajorityDepths(const SQPSolution& solution, InputArray objectPoints) const
839{
840 const cv::Matx<double, 9, 1>& r = solution.r_hat;
841 const cv::Matx<double, 3, 1>& t = solution.t;
842 int npos = 0, nneg = 0;
843
844 Mat _objectPoints = objectPoints.getMat();
845
846 int n = _objectPoints.cols * _objectPoints.rows;
847
848 for (int i = 0; i < n; i++)
849 {
850 const cv::Point3d& obj_pt = _objectPoints.at<cv::Point3d>(i);
851 if (r(6) * obj_pt.x + r(7) * obj_pt.y + r(8) * obj_pt.z + t(2) > 0) ++npos;
852 else ++nneg;
853 }
854
855 return npos >= nneg;
856}

859void PoseSolver::checkSolution(SQPSolution& solution, InputArray objectPoints, double& min_error)
860{
861 bool cheirok = positiveDepth(solution) || positiveMajorityDepths(solution, objectPoints); // check the majority if the check with centroid fails
862 if (cheirok)
863 {
864 solution.sq_error = (omega_ * solution.r_hat).ddot(solution.r_hat);
865 if (fabs(min_error - solution.sq_error) > EQUAL_SQUARED_ERRORS_DIFF)
866 {
867 if (min_error > solution.sq_error)
868 {
869 min_error = solution.sq_error;
870 solutions_[0] = solution;
871 num_solutions_ = 1;
872 }
873 }
874 else
875 {
876 bool found = false;
877 for (int i = 0; i < num_solutions_; i++)
878 {
879 if (cv::norm(solutions_[i].r_hat - solution.r_hat, cv::NORM_L2SQR) < EQUAL_VECTORS_SQUARED_DIFF)
880 {
881 if (solutions_[i].sq_error > solution.sq_error)
882 {
883 solutions_[i] = solution;
884 }
885 found = true;
886 break;
887 }
888 }
889
890 if (!found)
891 {
892 solutions_[num_solutions_++] = solution;
893 }
894 if (min_error > solution.sq_error) min_error = solution.sq_error;
895 }
896 }

899double PoseSolver::orthogonalityError(const cv::Matx<double, 9, 1>& e)
