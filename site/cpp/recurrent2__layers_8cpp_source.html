17template<typename Dtype>
18static void tanh(const Mat &src, Mat &dst)
19{
20 MatConstIterator_<Dtype> itSrc = src.begin<Dtype>();
21 MatIterator_<Dtype> itDst = dst.begin<Dtype>();
22
23 for (; itSrc != src.end<Dtype>(); itSrc++, itDst++)
24 *itDst = std::tanh(*itSrc);
25}

27static void tanh(const Mat &src, Mat &dst)
28{
29 dst.create(src.size, src.type());
30 if (src.type() == CV_32F)
31 tanh<float>(src, dst);
32 else if (src.type() == CV_64F)
33 tanh<double>(src, dst);
34 else
35 CV_Error(Error::StsUnsupportedFormat, "Function supports only floating point types");
36}

38static void sigmoid(const Mat &src, Mat &dst)
39{
40 cv::exp(-src, dst);
41 cv::pow(1 + dst, -1, dst);
42}

44// ONNX gate order is I,O,F,C but the recurrence wants I,F,O,C: swap the O/F blocks.
45template<typename T>
46static void reorderGatesIOFCtoIFOC_(Mat& in)
47{
48 int first = in.size[0];
49 int rest = in.total() / first / 4;
50 Mat m = in.reshape(1, {first, 4, rest});
51 Mat outputGate = m.col(1), forgetGate = m.col(2);
52 std::swap_ranges(outputGate.begin<T>(), outputGate.end<T>(), forgetGate.begin<T>());
53}
54static void reorderGatesIOFCtoIFOC(Mat& in)
55{
56 if (in.depth() == CV_64F)
57 reorderGatesIOFCtoIFOC_<double>(in);
58 else
59 reorderGatesIOFCtoIFOC_<float>(in);
60}

151 bool getMemoryShapes(const std::vector<MatShape> &inputs,
152 const int requiredOutputs,
153 std::vector<MatShape> &outputs,
154 std::vector<MatShape> &internals) const CV_OVERRIDE
155 {
156 const MatShape& inp0 = inputs[0];
157 const MatShape& Wx = inputs[1];
158 const MatShape& Wh = inputs[2];
159
160 int _hidSize = Wh[2];
161 int _inpSize = Wx[2];
162 MatShape outTailShape_(outTailShape), outResShape;
163
164 if (!outTailShape_.empty())
165 CV_Assert(total(outTailShape_) == _hidSize);
166 else
167 outTailShape_.assign(1, _hidSize);
168
169 // compute output shape of y
170 // figure out batch size
171 int _batchSize;
172 int _seqLen;
173 if (useTimestampDim)
174 {
175 CV_Assert(inp0.size() >= 2 && total(inp0, 2) == _inpSize);
176 if (layout == SEQ_BATCH_HID) {
177 _batchSize = inp0[1];
178 _seqLen = inp0[0];
179 outResShape.push_back(_seqLen);
180 outResShape.push_back(1 + static_cast<int>(bidirectional));
181 outResShape.push_back(_batchSize);
182 } else {
183 // ONNX layout=1: Y is (batch, seq, dirs, hid) - this must match what forward()
184 // actually writes, the graph engine preallocates the output by this shape
185 _batchSize = inp0[0];
186 _seqLen = inp0[1];
187 outResShape.push_back(_batchSize);
188 outResShape.push_back(_seqLen);
189 outResShape.push_back(1 + static_cast<int>(bidirectional));
190 }
191 }
192 else
193 {
194 CV_Assert(inp0.size() >= 2 && total(inp0, 1) == _inpSize);
195 _batchSize = inp0[0];
196 outResShape.push_back(1 + static_cast<int>(bidirectional));
197 outResShape.push_back(_batchSize);
198 }
199
200 outResShape.push_back(_hidSize);
201 outputs.assign(1, outResShape);
202
203 // Yh / Yc: ONNX layout=0 -> (dirs, batch, hid), layout=1 -> (batch, dirs, hid)
204 int shp[] = {1 + static_cast<int>(bidirectional), _batchSize, numHidden};
205 if (layout == BATCH_SEQ_HID)
206 std::swap(shp[0], shp[1]);
207 MatShape newShape(shp, shp + sizeof(shp)/sizeof(shp[0]));
208
209 // compute output shape of yc
210 if (produceCellOutput)
211 {
212 outputs.push_back(newShape);
213 }
214 // compute output shape of yh
215 if (produceOutputYh)
216 {
217 outputs.push_back(newShape);
218 }
219
220 // forward() allocates its own per-direction scratch, so no engine internals are needed
221 internals.clear();
222
223 return false;
224 }

237 virtual int64 getFLOPS(const std::vector<MatShape> &inputs,
238 const std::vector<MatShape> &outputs) const CV_OVERRIDE
239 {
240 // LSTM: 4 gates, each gate = input_size*hidden_size + hidden_size*hidden_size MACs
241 const MatShape& inp0 = inputs[0];
242 const MatShape& Wx = inputs[1];
243 int _hidSize = numHidden;
244 int _inpSize = Wx[2];
245 int _seqLen = inp0[0];
246 int _batchSize = inp0[1];
247 int numDirs = 1 + static_cast<int>(bidirectional);
248
249 // Per timestep: 4 gates * (2*input_size*hidden_size + 2*hidden_size*hidden_size + hidden_size)
250 int64 flopsPerStep = CV_BIG_INT(4) * (CV_BIG_INT(2) * _inpSize * _hidSize +
251 CV_BIG_INT(2) * _hidSize * _hidSize +
252 _hidSize);
253 return (int64)numDirs * _seqLen * _batchSize * flopsPerStep;
254 }

256 // Run one direction's recurrence into its own scratch so both directions run concurrently.
257 // Writes hOutAll columns [i*H, (i+1)*H) and the matching cOut slice.
258 void forwardDirection(int i, int numDirs, const Mat& xTs, const Mat& h0All, const Mat& c0All,
259 Mat& hOutAll, Mat& cOut) const
260 {
261 const int batchSizeTotal = seqLenth * batchSize;
262 const int dtype = xTs.type();
263
264 Mat Wx = weightBlobs[0].rowRange(i * weightBlobs[0].rows / numDirs, (i + 1) * weightBlobs[0].rows / numDirs);
265 Mat Wh = weightBlobs[1].rowRange(i * weightBlobs[1].rows / numDirs, (i + 1) * weightBlobs[1].rows / numDirs);
266 Mat bias = weightBlobs[2].colRange(i * weightBlobs[2].cols / numDirs, (i + 1) * weightBlobs[2].cols / numDirs);
267 Mat pI, pF, pO;
268 if (usePeephole)
269 {
270 pI = weightBlobs[3].rowRange(i * weightBlobs[3].rows / numDirs, (i + 1) * weightBlobs[3].rows / numDirs);
271 pF = weightBlobs[4].rowRange(i * weightBlobs[4].rows / numDirs, (i + 1) * weightBlobs[4].rows / numDirs);
272 pO = weightBlobs[5].rowRange(i * weightBlobs[5].rows / numDirs, (i + 1) * weightBlobs[5].rows / numDirs);
273 }
274
275 Mat hInternal(batchSize, numHidden, dtype);
276 Mat cInternal(batchSize, numHidden, dtype);
277 h0All.rowRange(i * batchSize, (i + 1) * batchSize).copyTo(hInternal);
278 c0All.rowRange(i * batchSize, (i + 1) * batchSize).copyTo(cInternal);
279
280 Mat hOutTs(batchSizeTotal, numHidden, dtype);
281 Mat cOutTs;
282 if (produceCellOutput)
283 {
284 cOutTs = cOut.reshape(1, batchSizeTotal);
285 cOutTs = cOutTs.colRange(i * cOutTs.cols / numDirs, (i + 1) * cOutTs.cols / numDirs);
286 }
287
288 // Batched projection: gatesAll = Wx*x + bias for the whole sequence.
289 const int gateN = 4 * numHidden, projK = xTs.cols;
290 Mat gatesAll;
291 if (dtype == CV_32F && (int)packedWx.size() == numDirs && !packedWx[i].empty())
292 {
293 FastGemmOpt opt = gemmOpt;
294 opt.multi_thread = (numDirs == 1); // directions are parallelized when bidirectional
295 repeat(bias, batchSizeTotal, 1, gatesAll); // each row starts as the bias
296 fastGemm(false, batchSizeTotal, gateN, projK, 1.f, xTs.ptr<float>(), projK,
297 packedWx[i].data(), 1.f, gatesAll.ptr<float>(), gateN, opt);
298 }
299 else // fp64 / unpacked fallback
300 {
301 Mat onesCol(batchSizeTotal, 1, dtype, Scalar(1));
302 gatesAll.create(batchSizeTotal, gateN, dtype);
303 gemm(onesCol, bias, 1.0, noArray(), 0.0, gatesAll);
304 gemm(xTs, Wx, 1.0, gatesAll, 1.0, gatesAll, GEMM_2_T);
305 }
306
307 // fastGEMM1T (recurrent step) needs contiguous fp32 operands
308#if CV_TRY_AVX2 || CV_TRY_AVX
309 bool canUseAvxH = hInternal.isContinuous() && gatesAll.isContinuous()
310 && Wh.depth() == CV_32F && hInternal.depth() == CV_32F && gatesAll.depth() == CV_32F && Wh.cols >= 8;
311#endif
312#if CV_TRY_SVE
313 bool canUseSveH = hInternal.isContinuous() && gatesAll.isContinuous()
314 && Wh.depth() == CV_32F && hInternal.depth() == CV_32F && gatesAll.depth() == CV_32F;
315#endif
316#if CV_TRY_NEON
317 bool canUseNeonH = hInternal.isContinuous() && gatesAll.isContinuous()
318 && Wh.depth() == CV_32F && hInternal.depth() == CV_32F && gatesAll.depth() == CV_32F && Wh.cols >= 4;
319#endif
320
321 int tsStart, tsEnd, tsInc;
322 if (reverse || i == 1) {
323 tsStart = seqLenth - 1;
324 tsEnd = -1;
325 tsInc = -1;
326 }
327 else {
328 tsStart = 0;
329 tsEnd = seqLenth;
330 tsInc = 1;
331 }
332
333 for (int ts = tsStart; ts != tsEnd; ts += tsInc)
334 {
335 Range curRowRange(ts*batchSize, (ts + 1)*batchSize);
336 Mat gates = gatesAll.rowRange(curRowRange); // already holds Wx * x_t + b
337
338 // gates += Wh * h_{t-1}
339#if CV_TRY_AVX2
340 if (useAVX2 && canUseAvxH)
341 {
342 for (int n = 0; n < hInternal.rows; n++)
343 opt_AVX2::fastGEMM1T(hInternal.ptr<float>(n), Wh.ptr<float>(), Wh.step1(),
344 gates.ptr<float>(n), gates.ptr<float>(n), Wh.rows, Wh.cols);
345 }
346 else
347#endif
348#if CV_TRY_AVX
349 if (useAVX && canUseAvxH)
350 {
351 for (int n = 0; n < hInternal.rows; n++)
352 opt_AVX::fastGEMM1T(hInternal.ptr<float>(n), Wh.ptr<float>(), Wh.step1(),
353 gates.ptr<float>(n), gates.ptr<float>(n), Wh.rows, Wh.cols);
354 }
355 else
356#endif
357#if CV_TRY_SVE
358 if (useSVE && canUseSveH)
359 {
360 for (int n = 0; n < hInternal.rows; n++)
361 opt_SVE::fastGEMM1T(hInternal.ptr<float>(n), Wh.ptr<float>(), Wh.step1(),
362 gates.ptr<float>(n), gates.ptr<float>(n), Wh.rows, Wh.cols);
363 }
364 else
365#endif
366#if CV_TRY_NEON
367 if (useNEON && canUseNeonH)
368 {
369 for (int n = 0; n < hInternal.rows; n++)
370 opt_NEON::fastGEMM1T(hInternal.ptr<float>(n), Wh.ptr<float>(), Wh.step1(),
371 gates.ptr<float>(n), gates.ptr<float>(n), Wh.rows, Wh.cols);
372 }
373 else
374#endif
375 {
376 gemm(hInternal, Wh, 1, gates, 1, gates, GEMM_2_T);
377 }
378
379 Mat gateI = gates.colRange(0*numHidden, 1*numHidden);
380 Mat gateF = gates.colRange(1*numHidden, 2*numHidden);
381 Mat gateO = gates.colRange(2*numHidden, 3*numHidden);
382 Mat gateG = gates.colRange(3*numHidden, 4*numHidden);
383
384 if (forgetBias){
385 add(gateF, forgetBias, gateF);
386 }
387
388 if (usePeephole)
389 {
390 Mat gatesIF = gates.colRange(0, 2*numHidden);
391 gemm(cInternal, pI, 1, gateI, 1, gateI);
392 gemm(cInternal, pF, 1, gateF, 1, gateF);
393 f_activation(gatesIF, gatesIF);
394 }
395 else
396 {
397 Mat gatesIFO = gates.colRange(0, 3*numHidden);
398 f_activation(gatesIFO, gatesIFO);
399 }
400
401 g_activation(gateG, gateG);
402
403 //compute c_t
404 multiply(gateF, cInternal, gateF); // f_t (*) c_{t-1}
405 multiply(gateI, gateG, gateI); // i_t (*) g_t
406 add(gateF, gateI, cInternal); // c_t = f_t (*) c_{t-1} + i_t (*) g_t
407
408 if (useCellClip)
409 {
410 min(cInternal, cellClip, cInternal);
411 max(cInternal, -cellClip, cInternal);
412 }
413
414 if (usePeephole)
415 {
416 gemm(cInternal, pO, 1, gateO, 1, gateO);
417 f_activation(gateO, gateO);
418 }
419
420 //compute h_t
421 h_activation(cInternal, hInternal);
422 multiply(gateO, hInternal, hInternal);
423
424 hInternal.copyTo(hOutTs.rowRange(curRowRange));
425
426 if (produceCellOutput)
427 cInternal.copyTo(cOutTs.rowRange(curRowRange));
428 }
429
430 // slice this direction's result into the assembly buffer
431 hOutTs.copyTo(hOutAll.colRange(i * numHidden, (i + 1) * numHidden));
432 }

434 void forward(InputArrayOfArrays inputs_arr,
435 OutputArrayOfArrays outputs_arr,
436 OutputArrayOfArrays internals_arr) CV_OVERRIDE
437 {
438
439 std::vector<Mat> input, output;
440 inputs_arr.getMatVector(input);
441 outputs_arr.getMatVector(output);
442
443 int numInputs = input.size();
444 int inpSize = input[0].size[2];
445
446 // determine seqLen and batchSize
447 if (useTimestampDim)
448 {
449 CV_Assert(input[0].dims >= 2 && (int)input[0].total(2) == inpSize);
450 if (layout == SEQ_BATCH_HID){
451 seqLenth = input[0].size[0];
452 batchSize = input[0].size[1];
453 }else{
454 seqLenth = input[0].size[1];
455 batchSize = input[0].size[0];
456 }
457 } else {
458 CV_Assert(input[0].dims >= 2 && (int)input[0].total(1) == inpSize);
459 seqLenth = 1;
460 batchSize = input[0].size[0];
461 }
462
463 // ONNX LSTM inputs 0..7: X,W,R,B,sequence_lens,initial_h,initial_c,P; slots 3..7 optional.
464 // Test non-emptiness, not count: some exports keep empty slots for unused inputs.
465 auto hasInput = [&](int idx) {
466 return idx < numInputs && !input[idx].empty();
467 };
468
469 CV_Assert(numInputs >= 3); // X, W, R are mandatory
470
471 const int numDirs = 1 + static_cast<int>(bidirectional);
472 const int batchSizeTotal = seqLenth * batchSize;
473
474 // Weight transform is shape-independent: once for const weights, else every forward.
475 if (!constWeights || !weightsPacked)
476 {
477 packWeights(input);
478 weightsPacked = true;
479 }
480
481 // Initial states are shape dependent, so (re)build them every forward.
482 Mat h0All = hasInput(5) ? input[5].reshape(1, input[5].size[0] * input[5].size[1])
483 : Mat::zeros(numDirs * batchSize, numHidden, input[0].type());
484 Mat c0All = hasInput(6) ? input[6].reshape(1, input[6].size[0] * input[6].size[1])
485 : Mat::zeros(numDirs * batchSize, numHidden, input[0].type());
486
487 // set outputs to 0
488 for (auto& out : output)
489 out.setTo(0);
490
491 // seq-major cell-state scratch: (seq, batch, dirs, hid), matching the recurrence.
492 int cOutShape[] = {seqLenth, batchSize, numDirs, numHidden};
493 Mat cOut = produceCellOutput ? Mat::zeros(4, cOutShape, output[0].type()) : Mat();
494
495 // the recurrence below slices X by timestep, so it needs the seq-major order;
496 // under ONNX layout=1 the input arrives as (batch, seq, ...)
497 Mat xSeqFirst = input[0];
498 if (layout == BATCH_SEQ_HID)
499 {
500 std::vector<int> perm(input[0].dims);
501 std::iota(perm.begin(), perm.end(), 0);
502 std::swap(perm[0], perm[1]);
503 cv::transposeND(input[0], perm, xSeqFirst);
504 }
505 Mat xTs = xSeqFirst.reshape(1, batchSizeTotal);
506
507 // seq-major Y assembly buffer; transposed into output[0] below.
508 // Never reallocate output[0]'s header or it detaches from the graph.
509 Mat hOutAll(batchSizeTotal, numDirs * numHidden, output[0].type());
510
511 // Directions are independent (disjoint output columns): run them in parallel.
512 parallel_for_(Range(0, numDirs), [&](const Range& r)
513 {
514 for (int i = r.start; i < r.end; i++)
515 forwardDirection(i, numDirs, xTs, h0All, c0All, hOutAll, cOut);
516 }, numDirs);
517
518 // Reshape to (seq, batch, dirs, hid), then transpose into output[0] per `layout`.
519 int shp1[] = {seqLenth, batchSize, numDirs, numHidden};
520 Mat y4d = hOutAll.reshape(1, sizeof(shp1)/sizeof(shp1[0]), shp1);
521 Mat ySeqFirst; // (seq, dirs, batch, hid): the layout=0 Y; Yh sliced from it
522 if (layout == SEQ_BATCH_HID) {
523 cv::transposeND(y4d, {0, 2, 1, 3}, output[0]);
524 ySeqFirst = output[0];
525 } else {
526 cv::transposeND(y4d, {0, 2, 1, 3}, ySeqFirst);
527 cv::transposeND(y4d, {1, 0, 2, 3}, output[0]); // (batch, seq, dirs, hid)
528 }
529
530 if (produceOutputYh){
531 getCellStateYh(ySeqFirst, output[1], numDirs);
532 }
533
534 if (produceCellOutput){
535 getCellStateYc(cOut, output[2], numDirs);
536 }
537 }

539 void getCellStateYh(Mat& scr, Mat& dst, int numDirs)
540 {
541 // TODO: implement
542 if (numDirs == 1){
543 // take a slice of output[0]
544 Mat hOut = scr.rowRange(scr.size[0] - 1, scr.size[0]);
545
546 // reshape 1x1xBxH -> 1xBxH
547 int shp[] = {1, batchSize, numHidden};
548 hOut = hOut.reshape(1, sizeof(shp)/sizeof(shp[0]), shp);
549
550 if (layout == BATCH_SEQ_HID){
551 cv::transposeND(hOut, {1, 0, 2}, dst);
552 }
553 else{
554 hOut.copyTo(dst);
555 }
556
557 } else {
558 // Slice: SxDxBxH -> last sequence, first direction
559 Range ranges1[] = {cv::Range(scr.size[0] - 1, scr.size[0]), cv::Range(0, 1), cv::Range::all(), cv::Range::all()};
560 Mat part1 = scr(ranges1);
561
562 // Slice: SxDxBxH -> first sequence, last direction
563 Range ranges2[] = {cv::Range(0, 1), cv::Range(scr.size[1] - 1, scr.size[1]), cv::Range::all(), cv::Range::all()};
564 Mat part2 = scr(ranges2);
565
566 int shp[] = {1, part1.size[2] * part1.size[3]};
567 part1 = part1.reshape(1, sizeof(shp)/sizeof(shp[0]), shp);
568 part2 = part2.reshape(1, sizeof(shp)/sizeof(shp[0]), shp);
569
570 // build into a temp, then write into the preallocated dst in place (vconcat straight
571 // into dst would replace its header and detach it from the graph's output tensor)
572 Mat tmp;
573 vconcat(part1, part2, tmp);
574
575 int finalShape[] = {2, batchSize, numHidden};
576 tmp = tmp.reshape(1, sizeof(finalShape)/sizeof(finalShape[0]), finalShape);
577
578 if (layout == BATCH_SEQ_HID){
579 cv::transposeND(tmp, {1, 0, 2}, dst);
580 } else {
581 tmp.copyTo(dst);
582 }
583 }

587 void getCellStateYc(Mat& cOut, Mat& dst, int numDirs)
588 {
589 // seq, batch, dirs, hidden
590 int shp[] = {0, batchSize, numDirs, numHidden};
591 cOut = cOut.reshape(1, sizeof(shp)/sizeof(shp[0]), shp);
592
593 // permute to (seq, dirs, batch, hidden); the `layout` only affects the FINAL Yc order
594 // below, the last-timestep/last-direction slicing is layout-independent
595 cv::Mat newCellState;
596 cv::transposeND(cOut, {0, 2, 1, 3}, newCellState);
597 cOut = newCellState;
598
599 if (numDirs == 1)
600 {
601 // Slice: Yh = Y[-1, :, :, :]
602 Range ranges[] = {cv::Range(cOut.size[0] - 1, cOut.size[0]), cv::Range::all(), cv::Range::all(), cv::Range::all()};
603 cOut = cOut(ranges);
604 // Reshape: 1x1xBxH -> 1xBxH
605 int shp[] = {1, batchSize, numHidden};
606 cOut = cOut.reshape(1, sizeof(shp)/sizeof(shp[0]), shp);
607 }
608 else
609 {
610 // Slice: SxDxBxH -> last sequence, first direction
611 Range ranges1[] = {cv::Range(cOut.size[0] - 1, cOut.size[0]), cv::Range(0, 1), cv::Range::all(), cv::Range::all()};
612 Mat part1 = cOut(ranges1);
613
614 // Slice: SxDxBxH -> first sequence, last direction
615 Range ranges2[] = {cv::Range(0, 1), cv::Range(cOut.size[1] - 1, cOut.size[1]), cv::Range::all(), cv::Range::all()};
616 Mat part2 = cOut(ranges2);
617
618 int shp[] = {1, part1.size[2] * part1.size[3]};
619 part1 = part1.reshape(1, sizeof(shp)/sizeof(shp[0]), shp);
620 part2 = part2.reshape(1, sizeof(shp)/sizeof(shp[0]), shp);
621
622 vconcat(part1, part2, cOut);
623
624 // Reshape: 1x2xBxH -> 2xBxH
625 int finalShape[] = {2, batchSize, numHidden};
626 cOut = cOut.reshape(1, sizeof(finalShape)/sizeof(finalShape[0]), finalShape);
627 }
628
629 // (dirs, batch, hid), or (batch, dirs, hid) for the batch-first layout - written into the
630 // preallocated dst in place
631 if (layout == BATCH_SEQ_HID){
632 cv::transposeND(cOut, {1, 0, 2}, dst);
633 } else {
634 cOut.copyTo(dst);
635 }
636 }
637
638 // Fill weightBlobs: [0]=Wx, [1]=Wh, [2]=bias, and (peephole) [3]=pI, [4]=pF, [5]=pO.
639 // Shape-independent, so it runs once for constant weights.
640 void packWeights(const std::vector<Mat>& input)
641 {
642 int biasShape[] = {1 + static_cast<int>(bidirectional), 8 * numHidden};
643
644 weightBlobs.clear();
645 weightBlobs.push_back(input[1].clone()); // W -> Wx
646 weightBlobs.push_back(input[2].clone()); // R -> Wh
647 weightBlobs.push_back((input.size() > 3 && !input[3].empty())
648 ? input[3].clone()
649 : Mat::zeros(2, biasShape, input[0].type())); // B
650 bool hasP = usePeephole && input.size() > 7 && !input[7].empty();
651 if (hasP)
652 weightBlobs.push_back(input[7].clone()); // P
653
654 Mat& Wx = weightBlobs[0];
655 Mat& Wh = weightBlobs[1];
656 Mat& b = weightBlobs[2];
657
658 b = b.reshape(1, b.size[0]);
659 Mat bx = b.colRange(0, b.cols / 2);
660 Mat bh = b.colRange(b.cols / 2, b.cols);
661 b = bx + bh;
662
663 reorderGatesIOFCtoIFOC(Wx);
664 reorderGatesIOFCtoIFOC(Wh);
665 reorderGatesIOFCtoIFOC(b);
666
667 weightBlobs[0] = Wx.reshape(1, Wx.size[0] * Wx.size[1]);
668 weightBlobs[1] = Wh.reshape(1, Wh.size[0] * Wh.size[1]);
669 weightBlobs[2] = b.reshape(1, 1);
670
671 // Prepack Wx per direction (fp32) so the projection is one batched GEMM.
672 packedWx.clear();
673 if (weightBlobs[0].type() == CV_32F)
674 {
675 const int numDirs = 1 + static_cast<int>(bidirectional);
676 const int N = weightBlobs[0].rows / numDirs; // 4 * numHidden
677 packedWx.resize(numDirs);
678 for (int d = 0; d < numDirs; d++)
679 {
680 Mat WxDir = weightBlobs[0].rowRange(d * N, (d + 1) * N);
681 fastGemmPackB(WxDir, packedWx[d], true, gemmOpt);
682 }
683 }
684
685 if (!hasP)
686 return;
687
688 Mat P = weightBlobs[3];
689 weightBlobs[3] = P.colRange(0, numHidden);
690 weightBlobs[3] = weightBlobs[3].clone().reshape(1, weightBlobs[3].total()); // Single column.
691 weightBlobs[3] = Mat::diag(weightBlobs[3]);
692
693 weightBlobs.push_back(P.colRange(numHidden, 2 * numHidden));
694 weightBlobs[4] = weightBlobs[4].clone().reshape(1, weightBlobs[4].total()); // Single column.
695 weightBlobs[4] = Mat::diag(weightBlobs[4]);
696
697 weightBlobs.push_back(P.colRange(2 * numHidden, 3 * numHidden));
698 weightBlobs[5] = weightBlobs[5].clone().reshape(1, weightBlobs[5].total()); // Single column.
699 weightBlobs[5] = Mat::diag(weightBlobs[5]);
700 }
701};
