29static void depthwiseConv32f(const void* inp__, const void* residual__,
30 void* out__, const ConvState& cs,
31 const void* weights__, const float* scale__,
32 const float* bias__)
33{
34 int NC1 = cs.inpshape[0]*cs.inpshape[1];
35
36 CV_Assert(cs.inpshape.layout == DATA_LAYOUT_BLOCK);
37 CV_Assert(cs.outshape.layout == DATA_LAYOUT_BLOCK);
38 CV_Assert(cs.inpshape.dims == cs.outshape.dims);
39
40 parallel_for_(Range(0, NC1), [&](const Range& range)
41 {
42 // Offer this task range to an accelerated HAL first, flattening the descriptor into a
43 // stable C argument list (no dnn types cross the boundary). The HAL fuses conv, scale,
44 // bias, residual and the fast-activation (out = min(s>=0 ? s : s*alpha, maxval)). A
45 // generic activation is a function pointer that cannot cross the ABI, so when one is
46 // present we still run the HAL for the heavy conv and apply the (elementwise) activation
47 // over this range afterwards -- matching the built-in fast-epilogue-then-activation
48 // order below. On NOT_IMPLEMENTED fall through to the built-in path.
49 {
50 int sd = cs.nspatialdims;
51 int insize[3] = { sd > 2 ? cs.inpshape[sd-1] : 1, sd > 1 ? cs.inpshape[sd] : 1, cs.inpshape[sd+1] };
52 int outsize[3] = { sd > 2 ? cs.outshape[sd-1] : 1, sd > 1 ? cs.outshape[sd] : 1, cs.outshape[sd+1] };
53 float maxval = FLT_MAX, default_alpha = 0.f;
54 const float* prelu_slope = nullptr;
55 switch (cs.fastActivation) {
56 case FAST_ACTIV_CLIP: maxval = cs.activParams[1]; break;
57 case FAST_ACTIV_LEAKY_RELU: default_alpha = cs.activParams[0]; break;
58 case FAST_ACTIV_PRELU: prelu_slope = cs.activParams.data(); break;
59 case FAST_ACTIV_NONE: default_alpha = 1.f; break;
60 default: break; // FAST_ACTIV_RELU: maxval = FLT_MAX, default_alpha = 0
61 }
62 int hal_res = cv_hal_dnn_depthwise_conv32f(
63 (const float*)inp__, (const float*)residual__, (float*)out__,
64 (const float*)weights__, scale__, bias__,
65 cs.inpshape.C, cs.inpshape.back(), cs.inpshape[1],
66 insize, outsize, cs.strides, cs.pads, cs.inner,
67 cs.coordtab.data(), cs.ofstab.data(), (int)cs.ofstab.size(),
68 maxval, default_alpha, prelu_slope,
69 range.start, range.end);
70 if (hal_res == CV_HAL_ERROR_OK) {
71 if (cs.activation) {
72 // The HAL wrote conv + scale/bias/residual with an identity fast-epilogue;
73 // apply the generic activation in place over this range's output planes.
74 int planesz = outsize[0]*outsize[1]*outsize[2]*cs.inpshape.back();
75 const float* activParams = cs.activParams.data();
76 float* outp = (float*)out__ + range.start*planesz;
77 for (int nc1 = range.start; nc1 < range.end; nc1++, outp += planesz)
78 cs.activation(outp, outp, planesz, activParams);
79 }
80 return;
81 }
82 else if (hal_res != CV_HAL_ERROR_NOT_IMPLEMENTED)
83 CV_Error_(cv::Error::StsInternal,
84 ("HAL implementation dnn_depthwise_conv32f ==> cv_hal_dnn_depthwise_conv32f returned %d (0x%08x)", hal_res, hal_res));
85 }
86
87 constexpr int MAX_CONV_DIMS = ConvState::MAX_CONV_DIMS;
88 constexpr int C0 = 8;
89
90 CV_Assert(cs.nspatialdims <= MAX_CONV_DIMS && MAX_CONV_DIMS == 3);
91 CV_Assert(C0 == cs.inpshape.back());
92
93 int sdims = cs.nspatialdims;
94 int C = cs.inpshape.C;
95 int C1 = cs.inpshape[1];
96 int Di = sdims > 2 ? cs.inpshape[sdims - 1] : 1;
97 int Hi = sdims > 1 ? cs.inpshape[sdims] : 1;
98 int Wi = cs.inpshape[sdims + 1];
99 int D = sdims > 2 ? cs.outshape[sdims - 1] : 1;
100 int H = sdims > 1 ? cs.outshape[sdims] : 1;
101 int W = cs.outshape[sdims + 1];
102 int iplanesize = Di*Hi*Wi*C0;
103 int planesize = D*H*W*C0;
104 int SZ = cs.strides[0], SY = cs.strides[1], SX = cs.strides[2];
105 int padZ0 = cs.pads[0], padY0 = cs.pads[1], padX0 = cs.pads[2];
106 int inner_z0 = cs.inner[0], inner_z1 = cs.inner[MAX_CONV_DIMS];
107 int inner_y0 = cs.inner[1], inner_y1 = cs.inner[MAX_CONV_DIMS + 1];
108 int inner_x0 = cs.inner[2], inner_x1 = cs.inner[MAX_CONV_DIMS + 2];
109 int ksize = (int)cs.ofstab.size();
110 const int* zyxtab = cs.coordtab.data();
111 const int* ofstab = cs.ofstab.data();
112 const float* scale_ = scale__;
113 const float* bias_ = bias__;
114
115 const float* inp = (const float*)inp__ + range.start*iplanesize;
116 float* out = (float*)out__ + range.start*planesize;
117 const float* residual = residual__ ? (const float*)residual__ + range.start*planesize : nullptr;
118
119 FastActivation fastActivation = cs.fastActivation;
120 const float* activParams = cs.activParams.data();
121 ActivationFunc activation = cs.activation;
122 float maxval = FLT_MAX, defaultAlpha = 0.f;
123 float scalebuf[C0], biasbuf[C0], alphabuf[C0];
124 if (fastActivation == FAST_ACTIV_CLIP) {
125 CV_Assert(cs.activParams.size() == 2u);
126 maxval = activParams[1];
127 } else if (fastActivation == FAST_ACTIV_RELU) {
128 CV_Assert(!activParams);
129 } else if (fastActivation == FAST_ACTIV_LEAKY_RELU) {
130 CV_Assert(cs.activParams.size() == 1u);
131 defaultAlpha = activParams[0];
132 } else if (fastActivation == FAST_ACTIV_PRELU) {
133 CV_Assert(cs.activParams.size() == size_t(C));
134 } else {
135 // FAST_ACTIV_NONE: activation (if any) is handled via function pointer
136 defaultAlpha = 1.f;
137 }
138
139 #if CV_SIMD
140 v_float32 v_maxval = vx_setall_f32(maxval);
141 v_float32 z = vx_setzero_f32();
142 const int nlanes = VTraits<v_float32>::vlanes();
143 // RVV (CV_SIMD_SCALABLE) disabled for the m1 switch: with the fixed C0=8 the
144 // block is narrower than the register at VLEN>=512, tripping this assert. Runs
145 // scalar on RVV; re-enable via v_setvlmax<v_float32>(C0) (#29493, cf #29180).
146 CV_Assert(C0 == nlanes || C0 == nlanes*2 || C0 % (nlanes*4) == 0);
147 #endif
148
149 for (int nc1 = range.start; nc1 < range.end; nc1++, inp += iplanesize) {
150 int n = nc1 / C1;
151 int c_base = (nc1 - n*C1)*C0;
152 int c_count = std::min(C0, C - c_base);
153 const float* weights = (const float*)weights__ + (c_base/C0)*ksize*C0;
154
155 {
156 int c = 0;
157 for (; c < c_count; c++) {
158 scalebuf[c] = scale_ ? scale_[c_base + c] : 1.f;
159 biasbuf[c] = bias_ ? bias_[c_base + c] : 0.f;
160 alphabuf[c] = fastActivation == FAST_ACTIV_PRELU ? activParams[c_base + c] : defaultAlpha;
161 }
162 for (; c < C0; c++) {
163 scalebuf[c] = 0.f;
164 biasbuf[c] = 0.f;
165 alphabuf[c] = 0.f;
166 }
167 }
168
169 for (int z0 = 0; z0 < D; z0++) {
170 int zi_ = z0*SZ - padZ0;
171 for (int y0 = 0; y0 < H; y0++, out += W*C0,
172 residual += (residual ? W*C0 : 0)) {
173 //int64_t x0 = 0, x1 = W;
174 int x0 = 0;
175 int x1 = z0 >= inner_z0 && z0 < inner_z1 &&
176 y0 >= inner_y0 && y0 < inner_y1 ? inner_x0 : W;
177 int yi_ = y0*SY - padY0;
178
179 #if !(CV_SIMD)
180 memset(out, 0, W*C0*sizeof(out[0]));
181 #endif
182
183 for(;;) {
184 #if CV_SIMD
185 if (nlanes == C0) {
186 v_float32 sc0 = vx_load(scalebuf), b0 = vx_load(biasbuf);
187 v_float32 alpha0 = vx_load(alphabuf);
188 for (; x0 < x1; x0++) {
189 int xi_ = x0*SX - padX0;
190 v_float32 s0 = z;
191 for (int k = 0; k < ksize; k++) {
192 int zi = zi_ + zyxtab[k*MAX_CONV_DIMS];
193 int yi = yi_ + zyxtab[k*MAX_CONV_DIMS + 1];
194 int xi = xi_ + zyxtab[k*MAX_CONV_DIMS + 2];
195 v_float32 v0, w0;
196 if ((unsigned)zi >= (unsigned)Di ||
197 (unsigned)yi >= (unsigned)Hi ||
198 (unsigned)xi >= (unsigned)Wi)
199 continue;
200 v0 = vx_load(inp + ((zi*Hi + yi)*Wi + xi)*C0);
201 w0 = vx_load(weights + k*C0);
202 s0 = v_fma(v0, w0, s0);
203 }
204 s0 = v_fma(s0, sc0, b0);
205 if (residual)
206 s0 = v_add(s0, vx_load(residual + x0*C0));
207 s0 = v_min(v_select(v_ge(s0, z), s0, v_mul(s0, alpha0)), v_maxval);
208 vx_store(out + x0*C0, s0);
209 }
210 } else {
211 for (; x0 < x1; x0++) {
212 int xi_ = x0*SX - padX0;
213 for (int c = 0; c < C0; c += nlanes*2) {
214 v_float32 s0 = z, s1 = z;
215 for (int k = 0; k < ksize; k++) {
216 int zi = zi_ + zyxtab[k*MAX_CONV_DIMS];
217 int yi = yi_ + zyxtab[k*MAX_CONV_DIMS + 1];
218 int xi = xi_ + zyxtab[k*MAX_CONV_DIMS + 2];
219 v_float32 v0, v1, w0, w1;
220 if ((unsigned)zi >= (unsigned)Di ||
221 (unsigned)yi >= (unsigned)Hi ||
222 (unsigned)xi >= (unsigned)Wi)
223 continue;
224 int ofs_k = ((zi*Hi + yi)*Wi + xi)*C0 + c;
225 int ofs_w = k*C0;
226 v0 = vx_load(inp + ofs_k);
227 v1 = vx_load(inp + ofs_k + nlanes);
228 w0 = vx_load(weights + ofs_w);
229 w1 = vx_load(weights + ofs_w + nlanes);
230 s0 = v_fma(v0, w0, s0);
231 s1 = v_fma(v1, w1, s1);
232 }
233 s0 = v_fma(s0, vx_load(scalebuf + c), vx_load(biasbuf + c));
234 s1 = v_fma(s1, vx_load(scalebuf + c + nlanes), vx_load(biasbuf + c + nlanes));
235 if (residual) {
236 s0 = v_add(s0, vx_load(residual + x0*C0 + c));
237 s1 = v_add(s1, vx_load(residual + x0*C0 + c + nlanes));
238 }
239 v_float32 alpha0 = vx_load(alphabuf + c);
240 v_float32 alpha1 = vx_load(alphabuf + c + nlanes);
241
242 s0 = v_min(v_select(v_ge(s0, z), s0, v_mul(s0, alpha0)), v_maxval);
243 s1 = v_min(v_select(v_ge(s1, z), s1, v_mul(s1, alpha1)), v_maxval);
244 vx_store(out + x0*C0 + c, s0);
245 vx_store(out + x0*C0 + c + nlanes, s1);
246 }
247 }
248 }
249 #else
250 for (; x0 < x1; x0++) {
251 int xi_ = x0*SX - padX0;
252 for (int k = 0; k < ksize; k++) {
253 int zi = zi_ + zyxtab[k*MAX_CONV_DIMS];
254 int yi = yi_ + zyxtab[k*MAX_CONV_DIMS + 1];
255 int xi = xi_ + zyxtab[k*MAX_CONV_DIMS + 2];
256 if ((unsigned)zi >= (unsigned)Di ||
257 (unsigned)yi >= (unsigned)Hi ||
258 (unsigned)xi >= (unsigned)Wi)
259 continue;
260 const float* inptr = inp + ((zi*Hi + yi)*Wi + xi)*C0;
261 for (int c = 0; c < C0; c++)
262 out[x0*C0 + c] += inptr[c]*weights[k*C0 + c];
263 }
264 }
265 #endif
266
267 if (x0 == W)
268 break;
269 x1 = inner_x1;
270
271 #if CV_SIMD
272 if (nlanes == C0) {
273 v_float32 sc0 = vx_load(scalebuf), b0 = vx_load(biasbuf), alpha0 = vx_load(alphabuf);
274 for (; x0 < x1; x0++) {
275 int xi_ = x0*SX - padX0;
276 const float* inp_xi = inp + ((Hi*zi_ + yi_)*Wi + xi_)*C0;
277
278 v_float32 s0 = z;
279 for (int k = 0; k < ksize; k++) {
280 v_float32 v0 = vx_load(inp_xi + ofstab[k]);
281 v_float32 w0 = vx_load(weights + k*C0);
282 s0 = v_fma(v0, w0, s0);
283 }
284 s0 = v_fma(s0, sc0, b0);
285 if (residual)
286 s0 = v_add(s0, vx_load(residual + x0*C0));
287 s0 = v_min(v_select(v_ge(s0, z), s0, v_mul(s0, alpha0)), v_maxval);
288 vx_store(out + x0*C0, s0);
289 }
290 } else if (nlanes*2 == C0) {
291 v_float32 sc0 = vx_load(scalebuf), sc1 = vx_load(scalebuf + nlanes);
292 v_float32 b0 = vx_load(biasbuf), b1 = vx_load(biasbuf + nlanes);
293 v_float32 alpha0 = vx_load(alphabuf), alpha1 = vx_load(alphabuf + nlanes);
294 for (; x0 < x1; x0++) {
295 int xi_ = x0*SX - padX0;
296 const float* inp_xi = inp + ((Hi*zi_ + yi_)*Wi + xi_)*C0;
297
298 v_float32 s0 = z, s1 = z;
299 for (int k = 0; k < ksize; k++) {
300 int ofs_k = ofstab[k], ofs_w = k*C0;
301 v_float32 v0 = vx_load(inp_xi + ofs_k);
302 v_float32 v1 = vx_load(inp_xi + ofs_k + nlanes);
303 v_float32 w0 = vx_load(weights + ofs_w);
304 v_float32 w1 = vx_load(weights + ofs_w + nlanes);
305 s0 = v_fma(v0, w0, s0);
306 s1 = v_fma(v1, w1, s1);
307 }
308 s0 = v_fma(s0, sc0, b0);
309 s1 = v_fma(s1, sc1, b1);
310 if (residual) {
311 s0 = v_add(s0, vx_load(residual + x0*C0));
312 s1 = v_add(s1, vx_load(residual + x0*C0 + nlanes));
313 }
314 s0 = v_min(v_select(v_ge(s0, z), s0, v_mul(s0, alpha0)), v_maxval);
315 s1 = v_min(v_select(v_ge(s1, z), s1, v_mul(s1, alpha1)), v_maxval);
316 vx_store(out + x0*C0, s0);
317 vx_store(out + x0*C0 + nlanes, s1);
318 }
319 } else {
320 for (; x0 < x1; x0++) {
321 int xi_ = x0*SX - padX0;
322 for (int c = 0; c < C0; c += nlanes*4) {
323 const float* inp_xi = inp + ((Hi*zi_ + yi_)*Wi + xi_)*C0 + c;
324 v_float32 s0 = z, s1 = z, s2 = z, s3 = z;
325 for (int k = 0; k < ksize; k++) {
326 int ofs_k = ofstab[k], ofs_w = k*C0 + c;
327 v_float32 v0 = vx_load(inp_xi + ofs_k);
328 v_float32 v1 = vx_load(inp_xi + ofs_k + nlanes);
329 v_float32 v2 = vx_load(inp_xi + ofs_k + nlanes*2);
330 v_float32 v3 = vx_load(inp_xi + ofs_k + nlanes*3);
331 v_float32 w0 = vx_load(weights + ofs_w);
332 v_float32 w1 = vx_load(weights + ofs_w + nlanes);
333 v_float32 w2 = vx_load(weights + ofs_w + nlanes*2);
334 v_float32 w3 = vx_load(weights + ofs_w + nlanes*3);
335 s0 = v_fma(v0, w0, s0);
336 s1 = v_fma(v1, w1, s1);
337 s2 = v_fma(v2, w2, s2);
338 s3 = v_fma(v3, w3, s3);
339 }
340 s0 = v_fma(s0, vx_load(scalebuf + c),
341 vx_load(biasbuf + c));
342 s1 = v_fma(s1, vx_load(scalebuf + c + nlanes),
343 vx_load(biasbuf + c + nlanes));
344 s2 = v_fma(s2, vx_load(scalebuf + c + nlanes*2),
345 vx_load(biasbuf + c + nlanes*2));
346 s3 = v_fma(s3, vx_load(scalebuf + c + nlanes*3),
347 vx_load(biasbuf + c + nlanes*3));
348
349 if (residual) {
350 s0 = v_add(s0, vx_load(residual + x0*C0 + c));
351 s1 = v_add(s1, vx_load(residual + x0*C0 + c + nlanes));
352 s2 = v_add(s2, vx_load(residual + x0*C0 + c + nlanes*2));
353 s3 = v_add(s3, vx_load(residual + x0*C0 + c + nlanes*3));
354 }
355
356 v_float32 alpha0 = vx_load(alphabuf + c);
357 v_float32 alpha1 = vx_load(alphabuf + c + nlanes);
358 v_float32 alpha2 = vx_load(alphabuf + c + nlanes*2);
359 v_float32 alpha3 = vx_load(alphabuf + c + nlanes*3);
360
361 s0 = v_min(v_select(v_ge(s0, z), s0, v_mul(s0, alpha0)), v_maxval);
362 s1 = v_min(v_select(v_ge(s1, z), s1, v_mul(s1, alpha1)), v_maxval);
363 s2 = v_min(v_select(v_ge(s2, z), s2, v_mul(s2, alpha2)), v_maxval);
364 s3 = v_min(v_select(v_ge(s3, z), s3, v_mul(s3, alpha3)), v_maxval);
365 vx_store(out + x0*C0 + c, s0);
366 vx_store(out + x0*C0 + c + nlanes, s1);
367 vx_store(out + x0*C0 + c + nlanes*2, s2);
368 vx_store(out + x0*C0 + c + nlanes*3, s3);
369 }
370 }
371 }
372 #else
373 for (; x0 < x1; x0++) {
374 int xi_ = x0*SX - padX0;
375 const float* inp_xi = inp + ((Hi*zi_ + yi_)*Wi + xi_)*C0;
376
377 for (int k = 0; k < ksize; k++) {
378 const float* inptr = inp_xi + ofstab[k];
379 for (int c = 0; c < C0; c++)
380 out[x0*C0 + c] += inptr[c]*weights[k*C0 + c];
381 }
382 }
383 #endif
384
385 x1 = W;
386 }
387
388 #if !(CV_SIMD)
389 if (residual) {
390 for (int x = 0; x < W*C0; x += C0) {
391 for (int c = 0; c < C0; c++) {
392 float v = out[x + c]*scalebuf[c] + biasbuf[c] + residual[x + c];
393 v = std::min(v*(v >= 0 ? 1.f : alphabuf[c]), maxval);
394 out[x + c] = v;
395 }
396 }
397 } else {
398 for (int x = 0; x < W*C0; x += C0) {
399 for (int c = 0; c < C0; c++) {
400 float v = out[x + c]*scalebuf[c] + biasbuf[c];
401 v = std::min(v*(v >= 0 ? 1.f : alphabuf[c]), maxval);
402 out[x + c] = v;
403 }
404 }
405 }
406 #endif
407 }
408
409 if (activation) {
410 activation(out - planesize, out - planesize, planesize, activParams);
411 }
412 }
413 }
414 });
