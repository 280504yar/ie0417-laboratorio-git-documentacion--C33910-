177void Subdiv2D::splice( int edgeA, int edgeB )
178{
179 int& a_next = qedges[edgeA >> 2].next[edgeA & 3];
180 int& b_next = qedges[edgeB >> 2].next[edgeB & 3];
181 int a_rot = rotateEdge(a_next, 1);
182 int b_rot = rotateEdge(b_next, 1);
183 int& a_rot_next = qedges[a_rot >> 2].next[a_rot & 3];
184 int& b_rot_next = qedges[b_rot >> 2].next[b_rot & 3];
185 std::swap(a_next, b_next);
186 std::swap(a_rot_next, b_rot_next);
187}

197int Subdiv2D::connectEdges( int edgeA, int edgeB )
198{
199 int edge = newEdge();
200
201 splice(edge, getEdge(edgeA, NEXT_AROUND_LEFT));
202 splice(symEdge(edge), edgeB);
203
204 setEdgePoints(edge, edgeDst(edgeA), edgeOrg(edgeB));
205 return edge;
206}

208void Subdiv2D::swapEdges( int edge )
209{
210 int sedge = symEdge(edge);
211 int a = getEdge(edge, PREV_AROUND_ORG);
212 int b = getEdge(sedge, PREV_AROUND_ORG);
213
214 splice(edge, a);
215 splice(sedge, b);
216
217 setEdgePoints(edge, edgeDst(a), edgeDst(b));
218
219 splice(edge, getEdge(a, NEXT_AROUND_LEFT));
220 splice(sedge, getEdge(b, NEXT_AROUND_LEFT));
221}

223static double triangleArea( Point2f a, Point2f b, Point2f c )
224{
225 return ((double)b.x - a.x) * ((double)c.y - a.y) - ((double)b.y - a.y) * ((double)c.x - a.x);

252{
253 CV_DbgAssert((size_t)(edge >> 2) < (size_t)qedges.size());
254 splice( edge, getEdge(edge, PREV_AROUND_ORG) );
255 int sedge = symEdge(edge);
256 splice(sedge, getEdge(sedge, PREV_AROUND_ORG) );

260 qedges[edge].next[1] = freeQEdge;

286int Subdiv2D::locate(Point2f pt, int& _edge, int& _vertex)
287{
288 CV_INSTRUMENT_REGION();
289
290 int vertex = 0;
291
292 int i, maxEdges = (int)(qedges.size() * 4);
293
294 if( qedges.size() < (size_t)4 )
295 CV_Error( cv::Error::StsError, "Subdivision is empty" );
296
297 if( pt.x < topLeft.x || pt.y < topLeft.y || pt.x >= bottomRight.x || pt.y >= bottomRight.y )
298 CV_Error( cv::Error::StsOutOfRange, "" );
299
300 int edge = recentEdge;
301 CV_Assert(edge > 0);
302
303 int location = PTLOC_ERROR;
304
305 int right_of_curr = isRightOf(pt, edge);
306 if( right_of_curr > 0 )
307 {
308 edge = symEdge(edge);
309 right_of_curr = -right_of_curr;
310 }
311
312 for( i = 0; i < maxEdges; i++ )
313 {
314 int onext_edge = nextEdge( edge );
315 int dprev_edge = getEdge( edge, PREV_AROUND_DST );
316
317 int right_of_onext = isRightOf( pt, onext_edge );
318 int right_of_dprev = isRightOf( pt, dprev_edge );
319
320 if( right_of_dprev > 0 )
321 {
322 if( right_of_onext > 0 || (right_of_onext == 0 && right_of_curr == 0) )
323 {
324 location = PTLOC_INSIDE;
325 break;
326 }
327 else
328 {
329 right_of_curr = right_of_onext;
330 edge = onext_edge;
331 }
332 }
333 else
334 {
335 if( right_of_onext > 0 )
336 {
337 if( right_of_dprev == 0 && right_of_curr == 0 )
338 {
339 location = PTLOC_INSIDE;
340 break;
341 }
342 else
343 {
344 right_of_curr = right_of_dprev;
345 edge = dprev_edge;
346 }
347 }
348 else if( right_of_curr == 0 &&
349 isRightOf( vtx[edgeDst(onext_edge)].pt, edge ) >= 0 )
350 {
351 edge = symEdge( edge );
352 }
353 else
354 {
355 right_of_curr = right_of_onext;
356 edge = onext_edge;
357 }
358 }
359 }
360
361 recentEdge = edge;
362
363 if( location == PTLOC_INSIDE )
364 {
365 Point2f org_pt, dst_pt;
366 edgeOrg(edge, &org_pt);
367 edgeDst(edge, &dst_pt);
368
369 double t1 = fabs( pt.x - org_pt.x );
370 t1 += fabs( pt.y - org_pt.y );
371 double t2 = fabs( pt.x - dst_pt.x );
372 t2 += fabs( pt.y - dst_pt.y );
373 double t3 = fabs( org_pt.x - dst_pt.x );
374 t3 += fabs( org_pt.y - dst_pt.y );
375
376 if( t1 < FLT_EPSILON )
377 {
378 location = PTLOC_VERTEX;
379 vertex = edgeOrg( edge );
380 edge = 0;
381 }
382 else if( t2 < FLT_EPSILON )
383 {
384 location = PTLOC_VERTEX;
385 vertex = edgeDst( edge );
386 edge = 0;
387 }
388 else if( (t1 < t3 || t2 < t3) &&
389 fabs( triangleArea( pt, org_pt, dst_pt )) < FLT_EPSILON )
390 {
391 location = PTLOC_ON_EDGE;
392 vertex = 0;
393 }
394 }
395
396 if( location == PTLOC_ERROR )
397 {
398 edge = 0;
399 vertex = 0;
400 }
401
402 _edge = edge;
403 _vertex = vertex;
404
405 return location;
406}

409inline int
410isPtInCircle3( Point2f pt, Point2f a, Point2f b, Point2f c)
411{
412 const double eps = FLT_EPSILON*0.125;
413 double val = ((double)a.x * a.x + (double)a.y * a.y) * triangleArea( b, c, pt );
414 val -= ((double)b.x * b.x + (double)b.y * b.y) * triangleArea( a, c, pt );
415 val += ((double)c.x * c.x + (double)c.y * c.y) * triangleArea( a, b, pt );
416 val -= ((double)pt.x * pt.x + (double)pt.y * pt.y) * triangleArea( a, b, c );
417
418 return val > eps ? 1 : val < -eps ? -1 : 0;
419}

422int Subdiv2D::insert(Point2f pt)
423{
424 CV_INSTRUMENT_REGION();
425
426 int curr_point = 0, curr_edge = 0, deleted_edge = 0;
427 int location = locate( pt, curr_edge, curr_point );
428
429 if( location == PTLOC_ERROR )
430 CV_Error( cv::Error::StsBadSize, "" );
431
432 if( location == PTLOC_OUTSIDE_RECT )
433 CV_Error( cv::Error::StsOutOfRange, "" );
434
435 if( location == PTLOC_VERTEX )
436 return curr_point;
437
438 if( location == PTLOC_ON_EDGE )
439 {
440 deleted_edge = curr_edge;
441 recentEdge = curr_edge = getEdge( curr_edge, PREV_AROUND_ORG );
442 deleteEdge(deleted_edge);
443 }
444 else if( location == PTLOC_INSIDE )
445 ;
446 else
447 CV_Error_(cv::Error::StsError, ("Subdiv2D::locate returned invalid location = %d", location) );
448
449 CV_Assert( curr_edge != 0 );
450 validGeometry = false;
451
452 curr_point = newPoint(pt, false);
453 int base_edge = newEdge();
454 int first_point = edgeOrg(curr_edge);
455 setEdgePoints(base_edge, first_point, curr_point);
456 splice(base_edge, curr_edge);
457
458 do
459 {
460 base_edge = connectEdges( curr_edge, symEdge(base_edge) );
461 curr_edge = getEdge(base_edge, PREV_AROUND_ORG);
462 }
463 while( edgeDst(curr_edge) != first_point );
464
465 curr_edge = getEdge( base_edge, PREV_AROUND_ORG );
466
467 int i, max_edges = (int)(qedges.size()*4);
468
469 for( i = 0; i < max_edges; i++ )
470 {
471 int temp_dst = 0, curr_org = 0, curr_dst = 0;
472 int temp_edge = getEdge( curr_edge, PREV_AROUND_ORG );
473
474 temp_dst = edgeDst( temp_edge );
475 curr_org = edgeOrg( curr_edge );
476 curr_dst = edgeDst( curr_edge );
477
478 if( isRightOf( vtx[temp_dst].pt, curr_edge ) > 0 &&
479 isPtInCircle3( vtx[curr_org].pt, vtx[temp_dst].pt,
480 vtx[curr_dst].pt, vtx[curr_point].pt ) < 0 )
481 {
482 swapEdges( curr_edge );
483 curr_edge = getEdge( curr_edge, PREV_AROUND_ORG );
484 }
485 else if( curr_org == first_point )
486 break;
487 else
488 curr_edge = getEdge( nextEdge( curr_edge ), PREV_AROUND_LEFT );
489 }
490
491 return curr_point;
492}

613static Point2f computeVoronoiPoint(Point2f org0, Point2f dst0, Point2f org1, Point2f dst1)
614{
615 double a0 = dst0.x - org0.x;
616 double b0 = dst0.y - org0.y;
617 double c0 = -0.5*(a0 * (dst0.x + org0.x) + b0 * (dst0.y + org0.y));
618
619 double a1 = dst1.x - org1.x;
620 double b1 = dst1.y - org1.y;
621 double c1 = -0.5*(a1 * (dst1.x + org1.x) + b1 * (dst1.y + org1.y));
622
623 double det = a0 * b1 - a1 * b0;
624
625 if( det != 0 )
626 {
627 det = 1. / det;
628 return Point2f((float) ((b0 * c1 - b1 * c0) * det),
629 (float) ((a1 * c0 - a0 * c1) * det));
630 }
631
632 return Point2f(FLT_MAX, FLT_MAX);
633}

636void Subdiv2D::calcVoronoi()
637{
638 // check if it is already calculated
639 if( validGeometry )
640 return;
641
642 clearVoronoi();
643 int i, total = (int)qedges.size();
644
645 // loop through all quad-edges, except for the first 3 (#1, #2, #3 - 0 is reserved for "NULL" pointer)
646 for( i = 4; i < total; i++ )
647 {
648 QuadEdge& quadedge = qedges[i];
649
650 if( quadedge.isfree() )
651 continue;
652
653 int edge0 = (int)(i*4);
654 Point2f org0, dst0, org1, dst1;
655
656 if( !quadedge.pt[3] )
657 {
658 int edge1 = getEdge( edge0, NEXT_AROUND_LEFT );
659 int edge2 = getEdge( edge1, NEXT_AROUND_LEFT );
660
661 edgeOrg(edge0, &org0);
662 edgeDst(edge0, &dst0);
663 edgeOrg(edge1, &org1);
664 edgeDst(edge1, &dst1);
665
666 Point2f virt_point = computeVoronoiPoint(org0, dst0, org1, dst1);
667
668 if( fabs( virt_point.x ) < FLT_MAX * 0.5 &&
669 fabs( virt_point.y ) < FLT_MAX * 0.5 )
670 {
671 quadedge.pt[3] = qedges[edge1 >> 2].pt[3 - (edge1 & 2)] =
672 qedges[edge2 >> 2].pt[3 - (edge2 & 2)] = newPoint(virt_point, true);
673 }
674 }
675
676 if( !quadedge.pt[1] )
677 {
678 int edge1 = getEdge( edge0, NEXT_AROUND_RIGHT );
679 int edge2 = getEdge( edge1, NEXT_AROUND_RIGHT );
680
681 edgeOrg(edge0, &org0);
682 edgeDst(edge0, &dst0);
683 edgeOrg(edge1, &org1);
684 edgeDst(edge1, &dst1);
685
686 Point2f virt_point = computeVoronoiPoint(org0, dst0, org1, dst1);
687
688 if( fabs( virt_point.x ) < FLT_MAX * 0.5 &&
689 fabs( virt_point.y ) < FLT_MAX * 0.5 )
690 {
691 quadedge.pt[1] = qedges[edge1 >> 2].pt[1 + (edge1 & 2)] =
692 qedges[edge2 >> 2].pt[1 + (edge2 & 2)] = newPoint(virt_point, true);
693 }
694 }
695 }
696
697 validGeometry = true;
698}

709int Subdiv2D::findNearest(Point2f pt, Point2f* nearestPt)
710{
711 CV_INSTRUMENT_REGION();
712
713 if( !validGeometry )
714 calcVoronoi();
715
716 int vertex = 0, edge = 0;
717 int loc = locate( pt, edge, vertex );
718
719 if( loc != PTLOC_ON_EDGE && loc != PTLOC_INSIDE )
720 return vertex;
721
722 vertex = 0;
723
724 Point2f start;
725 edgeOrg(edge, &start);
726 Point2f diff = pt - start;
727
728 edge = rotateEdge(edge, 1);
729
730 int i, total = (int)vtx.size();
731
732 for( i = 0; i < total; i++ )
733 {
734 Point2f t;
735
736 for(;;)
737 {
738 CV_Assert( edgeDst(edge, &t) > 0 );
739 if( isRightOf2( t, start, diff ) >= 0 )
740 break;
741
742 edge = getEdge( edge, NEXT_AROUND_LEFT );
743 }
744
745 for(;;)
746 {
747 CV_Assert( edgeOrg( edge, &t ) > 0 );
748
749 if( isRightOf2( t, start, diff ) < 0 )
750 break;
751
752 edge = getEdge( edge, PREV_AROUND_LEFT );
753 }
754
755 Point2f tempDiff;
756 edgeDst(edge, &tempDiff);
757 edgeOrg(edge, &t);
758 tempDiff -= t;
759
760 if( isRightOf2( pt, t, tempDiff ) >= 0 )
761 {
762 vertex = edgeOrg(rotateEdge( edge, 3 ));
763 break;
764 }
765
766 edge = symEdge( edge );
767 }
768
769 if( nearestPt && vertex > 0 )
770 *nearestPt = vtx[vertex].pt;
771
772 return vertex;
773}

817 //use a std::vector<bool> to benefit from the "bitset size/8" implementation
818 std::vector<bool> edgemask(total, false);
819 Rect2f rect(topLeft.x, topLeft.y, bottomRight.x - topLeft.x, bottomRight.y - topLeft.y);

821 for( i = 4; i < total; i += 2 )
822 {

831 edgeOrg(edge_b, &b);
832 if ( !rect.contains(b) )
833 continue;
834 int edge_c = getEdge(edge_b, NEXT_AROUND_LEFT);
835 edgeOrg(edge_c, &c);
836 if ( !rect.contains(c) )
837 continue;
838 edgemask[edge_a] = true;
839 edgemask[edge_b] = true;
840 edgemask[edge_c] = true;
841 triangleList.emplace_back(a.x, a.y, b.x, b.y, c.x, c.y);

845void Subdiv2D::getVoronoiFacetList(const std::vector<int>& idx,
846 CV_OUT std::vector<std::vector<Point2f> >& facetList,
847 CV_OUT std::vector<Point2f>& facetCenters)
848{
849 calcVoronoi();
850 facetList.clear();
851 facetCenters.clear();
852
853 std::vector<Point2f> buf;
854
855 size_t i, total;
856 if( idx.empty() )
857 i = 4, total = vtx.size();
858 else
859 i = 0, total = idx.size();
860
861 for( ; i < total; i++ )
862 {
863 int k = idx.empty() ? (int)i : idx[i];
864
865 if( vtx[k].isfree() || vtx[k].isvirtual() )
866 continue;
867 int edge = rotateEdge(vtx[k].firstEdge, 1), t = edge;
868
869 // gather points
870 buf.clear();
871 do
872 {
873 buf.push_back(vtx[edgeOrg(t)].pt);
874 t = getEdge( t, NEXT_AROUND_LEFT );
875 }
876 while( t != edge );
877
878 facetList.push_back(buf);
879 facetCenters.push_back(vtx[k].pt);
880 }

885{
886 int i, j, total = (int)qedges.size();
887
888 for( i = 0; i < total; i++ )
889 {
890 const QuadEdge& qe = qedges[i];
891
892 if( qe.isfree() )
893 continue;
894
895 for( j = 0; j < 4; j++ )
896 {
897 int e = (int)(i*4 + j);
898 int o_next = nextEdge(e);
899 int o_prev = getEdge(e, PREV_AROUND_ORG );
900 int d_prev = getEdge(e, PREV_AROUND_DST );
901 int d_next = getEdge(e, NEXT_AROUND_DST );
902
903 // check points
904 CV_Assert( edgeOrg(e) == edgeOrg(o_next));
905 CV_Assert( edgeOrg(e) == edgeOrg(o_prev));
906 CV_Assert( edgeDst(e) == edgeDst(d_next));
907 CV_Assert( edgeDst(e) == edgeDst(d_prev));
908
909 if( j % 2 == 0 )
910 {
911 CV_Assert( edgeDst(o_next) == edgeOrg(d_prev));
912 CV_Assert( edgeDst(o_prev) == edgeOrg(d_next));
913 CV_Assert( getEdge(getEdge(getEdge(e,NEXT_AROUND_LEFT),NEXT_AROUND_LEFT),NEXT_AROUND_LEFT) == e );
914 CV_Assert( getEdge(getEdge(getEdge(e,NEXT_AROUND_RIGHT),NEXT_AROUND_RIGHT),NEXT_AROUND_RIGHT) == e);
915 }
916 }
917 }
