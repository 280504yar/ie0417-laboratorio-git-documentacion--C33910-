28// Combining several things that require static dynamic initialization in a

31struct GlobalLoggingInitStruct
32{
33public:

61 void handleMalformed()
62 {
63 // need to print warning for malformed log tag config strings?
64 if (m_isDebugBuild)
65 {
66 const auto& parser = logTagManager.getConfigParser();
67 if (parser.hasMalformed())
68 {
69 const auto& malformedList = parser.getMalformed();
70 for (const auto& malformed : malformedList)

72 std::cout << "Malformed log level config: \"" << malformed << "\"\n";

74 std::cout.flush();
75 }
76 }

85// Static dynamic initialization guard function for the combined struct
86// just defined above
87//
88// An initialization guard function guarantees that outside code cannot
89// accidentally see not-yet-dynamically-initialized data, by routing
90// all outside access request to this function, so that this function
91// has a chance to run the initialization code if necessary.
92//
93// An initialization guard function only guarantees initialization upon
94// the first call to this function.
95//
96static GlobalLoggingInitStruct& getGlobalLoggingInitStruct()
97{
98 CV_SINGLETON_LAZY_INIT_REF(GlobalLoggingInitStruct, new GlobalLoggingInitStruct());
99}
100
101// To ensure that the combined struct defined above is initialized even
102// if the initialization guard function wasn't called, a dummy static
103// instance of a struct is defined below, which will call the
104// initialization guard function.
105//
106struct GlobalLoggingInitCall
107{
108 GlobalLoggingInitCall()
109 {
110 getGlobalLoggingInitStruct();
111 (void)getGlobalLogTag(); // complete initialization of logger structures
112 }
113};

198void writeLogMessage(LogLevel logLevel, const char* message)
199{
200 WriteLogMessageFuncType userFunc = stc_userWriteLogMessageFunc.load();
201 if (userFunc && userFunc != writeLogMessage)
202 {
203 (*userFunc)(logLevel, message);
204 return;
205 }
206
207 const int threadID = cv::utils::getThreadID();
208
209 std::string message_id;
210 switch (getShowTimestampMode())
211 {
212 case 1: message_id = cv::format("%d@%0.3f", threadID, getTimestampNS() * 1e-9); break;
213 case 1+2: message_id = cv::format("%d@%llu", threadID, (long long unsigned int)getTimestampNS()); break;
214 default: message_id = cv::format("%d", threadID); break;
215 }
216
217 std::ostringstream ss;
218 switch (logLevel)
219 {
220 case LOG_LEVEL_FATAL: ss << "[FATAL:" << message_id << "] " << message << std::endl; break;
221 case LOG_LEVEL_ERROR: ss << "[ERROR:" << message_id << "] " << message << std::endl; break;
222 case LOG_LEVEL_WARNING: ss << "[ WARN:" << message_id << "] " << message << std::endl; break;
223 case LOG_LEVEL_INFO: ss << "[ INFO:" << message_id << "] " << message << std::endl; break;
224 case LOG_LEVEL_DEBUG: ss << "[DEBUG:" << message_id << "] " << message << std::endl; break;
225 case LOG_LEVEL_VERBOSE: ss << message << std::endl; break;
226 case LOG_LEVEL_SILENT: return; // avoid compiler warning about incomplete switch
227 case ENUM_LOG_LEVEL_FORCE_INT: return; // avoid compiler warning about incomplete switch
228 }
229#ifdef __ANDROID__
230 int android_logLevel = ANDROID_LOG_INFO;
231 switch (logLevel)
232 {
233 case LOG_LEVEL_FATAL: android_logLevel = ANDROID_LOG_FATAL; break;
234 case LOG_LEVEL_ERROR: android_logLevel = ANDROID_LOG_ERROR; break;
235 case LOG_LEVEL_WARNING: android_logLevel = ANDROID_LOG_WARN; break;
236 case LOG_LEVEL_INFO: android_logLevel = ANDROID_LOG_INFO; break;
237 case LOG_LEVEL_DEBUG: android_logLevel = ANDROID_LOG_DEBUG; break;
238 case LOG_LEVEL_VERBOSE: android_logLevel = ANDROID_LOG_VERBOSE; break;
239 default:
240 break;
241 }
242 __android_log_print(android_logLevel, "OpenCV/" CV_VERSION, "%s", ss.str().c_str());
243#endif
244 std::ostream* out = (logLevel <= LOG_LEVEL_WARNING) ? &std::cerr : &std::cout;
245 (*out) << ss.str();
246 if (logLevel <= LOG_LEVEL_WARNING)
247 {
248 (*out) << std::flush;
249 }
250}

252static const char* stripSourceFilePathPrefix(const char* file)
253{
254 CV_Assert(file);
255 const char* pos = file;
256 const char* strip_pos = NULL;
257 char ch = 0;
258 while ((ch = pos[0]) != 0)
259 {
260 ++pos;
261 if (ch == '/' || ch == '\\')
262 strip_pos = pos;
263 }
264 if (strip_pos == NULL || strip_pos == pos/*eos*/)
265 return file;
266 return strip_pos;
267}

269void writeLogMessageEx(LogLevel logLevel, const char* tag, const char* file, int line, const char* func, const char* message)
270{
271 WriteLogMessageExFuncType userFunc = stc_userWriteLogMessageExFunc.load();
272 if (userFunc && userFunc != writeLogMessageEx)
273 {
274 (*userFunc)(logLevel, tag, file, line, func, message);
275 return;
276 }
277
278 std::ostringstream strm;
279 if (tag)
280 {
281 strm << tag << ' ';
282 }
283 if (file)
284 {
285 strm << stripSourceFilePathPrefix(file);
286 if (line > 0)
287 {
288 strm << ':' << line;
289 }
290 strm << ' ';
291 }
292 if (func)
293 {
294 strm << func << ' ';
295 }
296 strm << message;
297 writeLogMessage(logLevel, strm.str().c_str());
298}
