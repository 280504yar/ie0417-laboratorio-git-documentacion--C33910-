236void OdometrySettingsImplCommon::setCameraMatrix(InputArray val)
237{
238 if (!val.empty())
239 {
240 CV_Assert(val.rows() == 3 && val.cols() == 3 && val.channels() == 1);
241 CV_Assert(val.type() == CV_32F);
242 val.copyTo(cameraMatrix);
243 }
244 else
245 {
246 DefaultSets ds;
247 this->cameraMatrix = ds.defaultCameraMatrix;
248 }

256void OdometrySettingsImplCommon::setIterCounts(InputArray val)
257{
258 if (!val.empty())
259 {
260 size_t nLevels = val.size(-1).width;
261 std::vector<Mat> pyramids;
262 val.getMatVector(pyramids);
263 this->iterCounts.clear();
264 for (size_t i = 0; i < nLevels; i++)
265 this->iterCounts.push_back(pyramids[i].at<int>(0));
266 }
