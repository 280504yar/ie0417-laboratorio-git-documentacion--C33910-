108inline Mat getPlane(const Mat &m, int n, int cn)
109{
110 CV_Assert(m.dims > 2);
111 int sz[CV_MAX_DIM];
112 for(int i = 2; i < m.dims; i++)
113 {
114 sz[i-2] = m.size.p[i];
115 }
116 return Mat(m.dims - 2, sz, m.type(), (void*)m.ptr<float>(n, cn));

155inline size_t total(const MatShape& shape, int start = -1, int end = -1)
156{
157 //if (shape.empty())
158 // return 0;
159
160 int dims = (int)shape.size();
161
162 if (start == -1) start = 0;
163 if (end == -1) end = dims;
164
165 CV_CheckLE(0, start, "");
166 CV_CheckLE(start, end, "");
167 CV_CheckLE(end, dims, "");
168
169 size_t elems = 1;

200inline MatShape concat(const MatShape& a, const MatShape& b)
201{
202 MatShape c = a;
203 size_t a_size = a.size(), b_size = b.size(), c_size = a_size + b_size;
204 c.resize(c_size);
205 for (size_t i = 0; i < b_size; i++) {
206 c[i+a_size] = b[i];
207 }
208 return c;
209}

247int normalize_axis(int axis, int dims)
248{
249 CV_Assert(dims >= 0);
250 CV_Check(axis, axis >= -dims && axis <= dims, "");
251 axis = (unsigned)axis < (unsigned)dims ? axis : axis < 0 ? axis + dims : axis - dims;
252 return axis;
253}

261inline
262Range normalize_axis_range(const Range& r, int axisSize)
263{
264 if (r == Range::all() || r == Range(0, INT_MAX))
265 return Range(0, axisSize);
266 CV_CheckGE(r.start, 0, "");
267 Range clamped(r.start,
268 r.end > 0 ? std::min(r.end, axisSize) : axisSize + r.end + 1);
269 CV_DbgCheckGE(clamped.start, 0, "");
270 CV_CheckLT(clamped.start, clamped.end, "");
271 CV_CheckLE(clamped.end, axisSize, "");
272 return clamped;
273}

275inline
276bool isAllOnes(const MatShape &inputShape, int startPos, int endPos)
277{
278 CV_CheckGE(startPos, 0, "");
279 CV_CheckLE(startPos, endPos, "");
280 CV_CheckLE(endPos, inputShape.dims, "");
281
282 for (int i = startPos; i < endPos; i++)
283 {
284 if (inputShape[i] != 1)
285 return false;
286 }
287 return true;
288}
