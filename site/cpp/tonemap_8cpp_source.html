56class TonemapImpl CV_FINAL : public Tonemap
57{
58public:
59 TonemapImpl(float _gamma) : name("Tonemap"), gamma(_gamma)
60 {
61 }
62
63 void process(InputArray _src, OutputArray _dst) CV_OVERRIDE
64 {
65 CV_INSTRUMENT_REGION();
66
67 Mat src = _src.getMat();
68 CV_Assert(!src.empty());
69 CV_Assert(_src.dims() == 2 && _src.type() == CV_32FC3);
70 _dst.create(src.size(), CV_32FC3);
71 Mat dst = _dst.getMat();
72
73 double min, max;
74 minMaxLoc(src, &min, &max);
75 float fmin = static_cast<float>(min);
76 float fmax = static_cast<float>(max);
77 if(max - min > DBL_EPSILON) {
78 dst = (src - fmin) / (fmax - fmin);
79 } else {
80 src.copyTo(dst);
81 }
82
83 pow(dst, 1.0f / gamma, dst);
84 }

91 writeFormat(fs);
92 fs << "name" << name
93 << "gamma" << gamma;
94 }

96 void read(const FileNode& fn) CV_OVERRIDE
97 {
98 FileNode n = fn["name"];
99 CV_Assert(n.isString() && String(n) == name);

101 }

103protected:
104 String name;

113class TonemapDragoImpl CV_FINAL : public TonemapDrago
114{
115public:
116 TonemapDragoImpl(float _gamma, float _saturation, float _bias) :
117 name("TonemapDrago"),
118 gamma(_gamma),
119 saturation(_saturation),
120 bias(_bias)
121 {
122 }
123
124 void process(InputArray _src, OutputArray _dst) CV_OVERRIDE
125 {
126 CV_INSTRUMENT_REGION();
127
128 Mat src = _src.getMat();
129 CV_Assert(!src.empty());
130 _dst.create(src.size(), CV_32FC3);
131 Mat img = _dst.getMat();
132
133 Ptr<Tonemap> linear = createTonemap(1.0f);
134 linear->process(src, img);
135
136 Mat gray_img;
137 cvtColor(img, gray_img, COLOR_RGB2GRAY);
138 Mat log_img;
139 log_(gray_img, log_img);
140 float mean = expf(static_cast<float>(sum(log_img)[0]) / log_img.total());
141 gray_img /= mean;
142 log_img.release();
143
144 double dmax;
145 minMaxLoc(gray_img, NULL, &dmax);
146 float max = static_cast<float>(dmax);
147 CV_Assert(max > 0);
148
149 Mat map;
150 log(gray_img + 1.0f, map);
151 Mat div;
152 pow(gray_img / static_cast<float>(max), logf(bias) / logf(0.5f), div);
153 log(2.0f + 8.0f * div, div);
154 map = map.mul(1.0f / div);
155 div.release();
156 mapLuminance(img, img, gray_img, map, saturation);
157
158 linear->setGamma(gamma);
159 linear->process(img, img);
160 }

199class TonemapReinhardImpl CV_FINAL : public TonemapReinhard
200{
201public:
202 TonemapReinhardImpl(float _gamma, float _intensity, float _light_adapt, float _color_adapt) :
203 name("TonemapReinhard"),
204 gamma(_gamma),
205 intensity(_intensity),
206 light_adapt(_light_adapt),
207 color_adapt(_color_adapt)
208 {
209 }
210
211 void process(InputArray _src, OutputArray _dst) CV_OVERRIDE
212 {
213 CV_INSTRUMENT_REGION();
214
215 Mat src = _src.getMat();
216 CV_Assert(!src.empty());
217 _dst.create(src.size(), CV_32FC3);
218 Mat img = _dst.getMat();
219 Ptr<Tonemap> linear = createTonemap(1.0f);
220 linear->process(src, img);
221
222 Mat gray_img;
223 cvtColor(img, gray_img, COLOR_RGB2GRAY);
224 Mat log_img;
225 log_(gray_img, log_img);
226
227 float log_mean = static_cast<float>(sum(log_img)[0] / log_img.total());
228 double dlog_min, dlog_max;
229 minMaxLoc(log_img, &dlog_min, &dlog_max);
230 float log_max = static_cast<float>(dlog_max);
231 float log_min = static_cast<float>(dlog_min);
232 log_img.release();
233
234 float key = (log_max - log_mean) / (log_max - log_min);
235 float map_key = 0.3f + 0.7f * std::pow(key, 1.4f);
236 intensity = exp(-intensity);
237 Scalar chan_mean = mean(img);
238 float gray_mean = static_cast<float>(mean(gray_img)[0]);
239
240 std::vector<Mat> channels(3);
241 split(img, channels);
242
243 for(int i = 0; i < 3; i++) {
244 float global = color_adapt * static_cast<float>(chan_mean[i]) + (1.0f - color_adapt) * gray_mean;
245 Mat adapt = color_adapt * channels[i] + (1.0f - color_adapt) * gray_img;
246 adapt = light_adapt * adapt + (1.0f - light_adapt) * global;
247 pow(intensity * adapt, map_key, adapt);
248 channels[i] = channels[i].mul(1.0f / (adapt + channels[i]));
249 }
250 gray_img.release();
251 merge(channels, img);
252
253 linear->setGamma(gamma);
254 linear->process(img, img);
255 }

299class TonemapMantiukImpl CV_FINAL : public TonemapMantiuk
300{
301public:
302 TonemapMantiukImpl(float _gamma, float _scale, float _saturation) :
303 name("TonemapMantiuk"),
304 gamma(_gamma),
305 scale(_scale),
306 saturation(_saturation)
307 {
308 }
309
310 void process(InputArray _src, OutputArray _dst) CV_OVERRIDE
311 {
312 CV_INSTRUMENT_REGION();
313
314 Mat src = _src.getMat();
315 CV_Assert(!src.empty());
316 _dst.create(src.size(), CV_32FC3);
317 Mat img = _dst.getMat();
318 Ptr<Tonemap> linear = createTonemap(1.0f);
319 linear->process(src, img);
320
321 Mat gray_img;
322 cvtColor(img, gray_img, COLOR_RGB2GRAY);
323 Mat log_img;
324 log_(gray_img, log_img);
325
326 std::vector<Mat> x_contrast, y_contrast;
327 getContrast(log_img, x_contrast, y_contrast);
328
329 for(size_t i = 0; i < x_contrast.size(); i++) {
330 mapContrast(x_contrast[i]);
331 mapContrast(y_contrast[i]);
332 }
333
334 Mat right(src.size(), CV_32F);
335 calculateSum(x_contrast, y_contrast, right);
336
337 Mat p, r, product, x = log_img;
338 calculateProduct(x, r);
339 r = right - r;
340 r.copyTo(p);
341
342 const float target_error = 1e-3f;
343 float target_norm = static_cast<float>(right.dot(right)) * powf(target_error, 2.0f);
344 int max_iterations = 100;
345 float rr = static_cast<float>(r.dot(r));
346
347 for(int i = 0; i < max_iterations; i++)
348 {
349 calculateProduct(p, product);
350 double dprod = p.dot(product);
351 CV_Assert(fabs(dprod) > 0);
352 float alpha = rr / static_cast<float>(dprod);
353
354 r -= alpha * product;
355 x += alpha * p;
356
357 float new_rr = static_cast<float>(r.dot(r));
358 CV_Assert(fabs(rr) > 0);
359 p = r + (new_rr / rr) * p;
360 rr = new_rr;
361
362 if(rr < target_norm) {
363 break;
364 }
365 }
366 exp(x, x);
367 mapLuminance(img, img, gray_img, x, saturation);
368
369 linear = createTonemap(gamma);
370 linear->process(img, img);
371 }

404 void signedPow(Mat src, float power, Mat& dst)
405 {
406 Mat sign = (src > 0);
407 sign.convertTo(sign, CV_32F, 1.0f/255.0f);
408 sign = sign * 2.0f - 1.0f;
409 pow(abs(src), power, dst);
410 dst = dst.mul(sign);
411 }
412
413 void mapContrast(Mat& contrast)
414 {
415 const float response_power = 0.4185f;
416 signedPow(contrast, response_power, contrast);
417 contrast *= scale;
418 signedPow(contrast, 1.0f / response_power, contrast);
419 }

421 void getGradient(Mat src, Mat& dst, int pos)
422 {
423 dst = Mat::zeros(src.size(), CV_32F);
424 Mat a, b;
425 Mat grad = src.colRange(1, src.cols) - src.colRange(0, src.cols - 1);
426 grad.copyTo(dst.colRange(pos, src.cols + pos - 1));
427 if(pos == 1) {
428 src.col(0).copyTo(dst.col(0));
429 }
430 }
431
432 void getContrast(Mat src, std::vector<Mat>& x_contrast, std::vector<Mat>& y_contrast)
433 {
434 int levels = static_cast<int>(logf(static_cast<float>(min(src.rows, src.cols))) / logf(2.0f));
435 x_contrast.resize(levels);
436 y_contrast.resize(levels);
437
438 Mat layer;
439 src.copyTo(layer);
440 for(int i = 0; i < levels; i++) {
441 getGradient(layer, x_contrast[i], 0);
442 getGradient(layer.t(), y_contrast[i], 0);
443 resize(layer, layer, Size(layer.cols / 2, layer.rows / 2), 0, 0, INTER_LINEAR);
444 }
445 }
446
447 void calculateSum(std::vector<Mat>& x_contrast, std::vector<Mat>& y_contrast, Mat& sum)
448 {
449 if (x_contrast.empty())
450 return;
451 const int last = (int)x_contrast.size() - 1;
452 sum = Mat::zeros(x_contrast[last].size(), CV_32F);
453 for(int i = last; i >= 0; i--)
454 {
455 Mat grad_x, grad_y;
456 getGradient(x_contrast[i], grad_x, 1);
457 getGradient(y_contrast[i], grad_y, 1);
458 resize(sum, sum, x_contrast[i].size(), 0, 0, INTER_LINEAR);
459 sum += grad_x + grad_y.t();
460 }
461 }
462
463 void calculateProduct(Mat src, Mat& dst)
464 {
465 std::vector<Mat> x_contrast, y_contrast;
466 getContrast(src, x_contrast, y_contrast);
467 calculateSum(x_contrast, y_contrast, dst);
468 }
