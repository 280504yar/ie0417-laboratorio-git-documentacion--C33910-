99 EltwiseLayerImpl(const LayerParams& params)
100 : outputChannels(0)
101 {
102 setParamsFrom(params);
103 hasVecInput = false;
104 op = SUM;
105 if (params.has("operation"))
106 {
107 String operation = toLowerCase(params.get<String>("operation"));
108 if (operation == "prod" || operation == "mul")
109 op = PROD;
110 else if (operation == "sum")
111 op = SUM;
112 else if (operation == "max")
113 op = MAX;
114 else if (operation == "min")
115 op = MIN;
116 else if (operation == "div")
117 op = DIV;
118 else
119 CV_Error(cv::Error::StsBadArg, "Unknown operation type \"" + operation + "\"");
120 }
121
122 if (params.has("coeff"))
123 {
124 DictValue paramCoeff = params.get("coeff");
125 int i, n = paramCoeff.size();
126 coeffs.resize(n);
127 for (i = 0; i < n; i++)
128 {
129 coeffs[i] = paramCoeff.get<float>(i);
130 }
131 }
132
133 channelsModeInput = ELTWISE_CHANNNELS_SAME;
134 if (params.has("output_channels_mode"))
135 {
136 String v = toLowerCase(params.get<String>("output_channels_mode"));
137 if (v == "same")
138 {
139 channelsModeInput = ELTWISE_CHANNNELS_SAME;
140 }
141 else if (v == "input_0")
142 {
143 channelsModeInput = ELTWISE_CHANNNELS_INPUT_0;
144 }
145 else if (v == "input_0_truncate")
146 {
147 channelsModeInput = ELTWISE_CHANNNELS_INPUT_0_TRUNCATE;
148 }
149 else if (v == "max_input_channels")
150 {
151 channelsModeInput = ELTWISE_CHANNNELS_USE_MAX;
152 if (op != SUM)
153 CV_Error(cv::Error::StsBadArg, "[" + type + "]:(" + name + ") 'max' channels mode is limited to SUM operation only");
154 }
155 else
156 CV_Error(cv::Error::StsBadArg, "[" + type + "]:(" + name + ") unknown channels mode: \"" + v + "\"");
157 }
158 channelsMode = channelsModeInput;
159
160 // TODO Must have checks for other unknown options
161 }

163 virtual bool supportBackend(int backendId) CV_OVERRIDE
164 {
165 if (hasVecInput && ELTWISE_CHANNNELS_SAME)
166 return backendId == DNN_BACKEND_OPENCV;
167
168#ifdef HAVE_INF_ENGINE
169 if (backendId == DNN_BACKEND_INFERENCE_ENGINE_NGRAPH)
170 return channelsMode == ELTWISE_CHANNNELS_SAME;
171#endif

173#ifdef HAVE_CANN
174 if (backendId == DNN_BACKEND_CANN)
175 return channelsMode == ELTWISE_CHANNNELS_SAME && coeffs.empty();
176#endif
177
178 if (backendId == DNN_BACKEND_CUDA)
179 {
180 if(channelsModeInput == ELTWISE_CHANNNELS_INPUT_0 || channelsModeInput == ELTWISE_CHANNNELS_INPUT_0_TRUNCATE)
181 return op == SUM && coeffs.empty();
182 return channelsModeInput == ELTWISE_CHANNNELS_SAME;
183 }
184
185 return backendId == DNN_BACKEND_OPENCV;
186 }

188 bool getMemoryShapes(const std::vector<MatShape> &inputs,
189 const int requiredOutputs,
190 std::vector<MatShape> &outputs,
191 std::vector<MatShape> &internals) const CV_OVERRIDE
192 {
193 CV_Assert(inputs.size() >= 2);
194 if (inputs[0].size() == 0){
195 outputs.assign(1, inputs[0]);
196 return false;
197 }
198 CV_Assert(inputs[0].size() >= 1);
199 CV_Assert(coeffs.size() == 0 || coeffs.size() == inputs.size());
200 CV_Assert(op == SUM || coeffs.size() == 0);
201
202 int dims = inputs[0].size();
203 // Number of channels in output shape is determined by the first input tensor.
204 bool variableChannels = false;
205 int numChannels = (dims == 1) ? inputs[0][0] : inputs[0][1];
206 for (size_t i = 1; i < inputs.size(); i++)
207 {
208 CV_Assert(inputs[0][0] == inputs[i][0]); // batch sizes are equal
209
210 int input_channels = (dims == 1) ? inputs[i][0] : inputs[i][1];
211 if (numChannels != input_channels)
212 variableChannels = true;
213
214 if (channelsModeInput == ELTWISE_CHANNNELS_SAME)
215 {
216 CV_Assert(numChannels == input_channels);
217 }
218 else if (channelsModeInput == ELTWISE_CHANNNELS_INPUT_0)
219 {
220 CV_Assert(numChannels >= input_channels);
221 }
222 else if (channelsModeInput == ELTWISE_CHANNNELS_INPUT_0_TRUNCATE)
223 {
224 // nothing to check
225 }
226 else if (channelsModeInput == ELTWISE_CHANNNELS_USE_MAX)
227 {
228 numChannels = std::max(numChannels, input_channels);
229 }
230 else
231 {
232 CV_Assert(0 && "Internal error");
233 }
234 }
235
236 channelsMode = variableChannels ? channelsModeInput : ELTWISE_CHANNNELS_SAME;
237 outputChannels = numChannels;
238
239 outputs.assign(1, inputs[0]);
240 outputs[0][(dims == 1) ? 0 : 1] = numChannels;
241
242 if (dims >= 1)
243 {
244 size_t vecIdx = 0;
245 bool isVecFound = false;
246 for (size_t i = 0; i < inputs.size(); i++)
247 {
248 bool allOnes = isAllOnes(inputs[i], (dims != 1) ? 2 : 1, dims);
249 if (!allOnes && !isVecFound)
250 {
251 vecIdx = i;
252 isVecFound = true;
253 }
254
255 if (!allOnes && i != vecIdx)
256 {
257 for (size_t j = 2; j < dims; j++)
258 {
259 CV_Assert(inputs[vecIdx][j] == inputs[i][j]);
260 }
261 }
262 }
263
264 if (channelsModeInput == ELTWISE_CHANNNELS_SAME && isVecFound)
265 {
266 for (size_t j = 2; j < dims; j++)
267 {
268 outputs[0][j] = inputs[vecIdx][j];
269 }
270 }
271 }
272
273 return false;
274 }

276 void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays) CV_OVERRIDE
277 {
278 std::vector<Mat> inputs;
279 inputs_arr.getMatVector(inputs);

284 if (isAllOnes(inpShape, 0, inputs[i].dims))

286 hasVecInput = true;
287 return;
288 }
289 }

292 class EltwiseInvoker : public ParallelLoopBody
293 {
294 EltwiseLayerImpl& self;
295 std::vector<const Mat*> srcs;
296 std::vector<int> srcNumChannels;
297 int nsrcs;
298 Mat* dst;
299 std::vector<float> coeffs;
300 int nstripes;
301 const ActivationLayer* activ;
302 int channels;
303 size_t planeSize;
304
305 EltwiseInvoker(EltwiseLayerImpl& self_)
306 : self(self_)
307 , nsrcs(0), dst(0), nstripes(0), activ(0), channels(0)
308 , planeSize(0)
309 {}
310
311 public:
312 static void run(EltwiseLayerImpl& self,
313 const Mat* srcs, int nsrcs, Mat& dst,
314 int nstripes)
315 {
316 const EltwiseOp op = self.op;
317 CV_Check(dst.dims, 0 <= dst.dims && dst.dims <= 5, "");
318 CV_CheckTypeEQ(dst.type(), CV_32FC1, "");
319 CV_Assert(dst.isContinuous());
320 CV_Assert(self.coeffs.empty() || self.coeffs.size() == (size_t)nsrcs);
321 CV_CheckGE(nsrcs, 2, "");
322
323 if (dst.dims > 1)
324 CV_Assert(self.outputChannels == dst.size[1]);
325
326 EltwiseInvoker p(self);
327 p.srcs.resize(nsrcs);
328 p.srcNumChannels.resize(nsrcs);
329 p.coeffs = self.coeffs; // can be sorted
330
331 bool sortInputs = false;
332 for( int i = 0; i < nsrcs; i++ )
333 {
334 p.srcs[i] = &srcs[i];
335 CV_CheckEQ(srcs[i].dims, dst.dims, "");
336 CV_Assert(srcs[i].isContinuous());
337 CV_Assert(srcs[i].type() == dst.type());
338 p.srcNumChannels[i] = (srcs[i].dims >= 4) ? srcs[i].size[1] : 1;
339
340 if (self.channelsMode == ELTWISE_CHANNNELS_SAME)
341 {
342 CV_Assert(srcs[i].size == dst.size);
343 }
344 else if (self.channelsMode == ELTWISE_CHANNNELS_INPUT_0)
345 {
346 if (i == 0)
347 CV_Assert(srcs[0].size == dst.size);
348 CV_Assert(self.outputChannels >= p.srcNumChannels[i]);
349 sortInputs = true;
350 }
351 else if (self.channelsMode == ELTWISE_CHANNNELS_INPUT_0_TRUNCATE)
352 {
353 if (i == 0)
354 CV_Assert(srcs[0].size == dst.size);
355 sortInputs = true;
356 }
357 else if (self.channelsMode == ELTWISE_CHANNNELS_USE_MAX)
358 {
359 CV_Assert(op == SUM);
360 CV_Assert(self.outputChannels >= p.srcNumChannels[i]);
361 sortInputs = true;
362 }
363 else
364 {
365 CV_Assert(0 && "Internal error");
366 }
367
368 if (sortInputs)
369 {
370 // Sort srcs and coefficients in the desc order by number of channels
371 for (int j = i; j >= 1; j--)
372 {
373 if (std::min(self.outputChannels, p.srcs[j - 1]->size[1]) < std::min(self.outputChannels, p.srcs[j]->size[1]))
374 {
375 std::swap(p.srcs[j - 1], p.srcs[j]);
376 std::swap(p.srcNumChannels[j - 1], p.srcNumChannels[j]);
377 if (!p.coeffs.empty())
378 std::swap(p.coeffs[j - 1], p.coeffs[j]);
379 }
380 else
381 break;
382 }
383 }
384 }
385
386 p.nsrcs = nsrcs;
387 p.dst = &dst;
388 p.nstripes = nstripes;
389 p.channels = (dst.dims >= 4 ? dst.size[1] : 1);
390
391 p.planeSize = dst.total(dst.dims >= 4 ? 2 : 1);
392 CV_CheckEQ(dst.total(), dst.size[0] * p.channels * p.planeSize, "");
393
394 bool simpleCoeffs = true;
395 if (op == SUM && !p.coeffs.empty())
396 {
397 CV_CheckEQ(p.coeffs.size(), (size_t)nsrcs, "");
398
399 for (size_t i = 0; i < p.coeffs.size(); i++)
400 {
401 if (p.coeffs[i] != 1)
402 {
403 simpleCoeffs = false;
404 break;
405 }
406 }
407 }
408 if (simpleCoeffs)
409 p.coeffs.clear();
410 p.activ = self.activ.get();
411
412 parallel_for_(Range(0, nstripes), p, nstripes);
413 }
414
415 void operator()(const Range& r) const CV_OVERRIDE
416 {
417 const EltwiseOp op = self.op;
418 size_t total = dst->size[0]*planeSize;
419 size_t stripeSize = (total + nstripes - 1)/nstripes;
420 size_t stripeStart = r.start*stripeSize;
421 size_t stripeEnd = std::min(r.end*stripeSize, total);
422 const float* coeffsptr = !coeffs.empty() ? &coeffs[0] : 0;
423 float* dstptr0 = dst->ptr<float>();
424 int blockSize0 = 1 << 12;
425
426 for (size_t ofs = stripeStart; ofs < stripeEnd; )
427 {
428 int sampleIdx = (int)(ofs / planeSize);
429 int delta = (int)ofs - sampleIdx * planeSize;
430 int blockSize = std::min(blockSize0, std::min((int)(stripeEnd - ofs), (int)planeSize - delta));
431 if( blockSize <= 0 )
432 break;
433 ofs += blockSize;
434
435 for (int c = 0; c < channels; c++)
436 {
437 size_t dstIdx = delta + (sampleIdx*channels + c)*planeSize;
438 float* dstptr = dstptr0 + dstIdx;
439
440 // process first two inputs
441 {
442 const float* srcptr0 = srcs[0]->ptr<float>() + dstIdx;
443
444 const int inputIdx = 1;
445 int src1_channels = srcNumChannels[inputIdx];
446 if (c >= src1_channels)
447 {
448 // no data from second input
449 if (!coeffsptr || coeffsptr[0] == 1.0f)
450 {
451 for (int j = 0; j < blockSize; j++)
452 {
453 dstptr[j] = srcptr0[j];
454 }
455 }
456 else
457 {
458 float c0 = coeffsptr[0];
459 for (int j = 0; j < blockSize; j++)
460 {
461 dstptr[j] = c0*srcptr0[j];
462 }
463 }
464 }
465 else
466 {
467 size_t srcIdx = delta + (sampleIdx * src1_channels + c) * planeSize;
468 const float* srcptrI = srcs[inputIdx]->ptr<float>() + srcIdx;
469
470 if (op == PROD)
471 {
472 for (int j = 0; j < blockSize; j++)
473 {
474 dstptr[j] = srcptr0[j] * srcptrI[j];
475 }
476 }
477 else if (op == DIV)
478 {
479 for (int j = 0; j < blockSize; j++)
480 {
481 dstptr[j] = srcptr0[j] / srcptrI[j];
482 }
483 }
484 else if (op == MAX)
485 {
486 for (int j = 0; j < blockSize; j++)
487 {
488 dstptr[j] = std::max(srcptr0[j], srcptrI[j]);
489 }
490 }
491 else if (op == MIN)
492 {
493 for (int j = 0; j < blockSize; j++)
494 {
495 dstptr[j] = std::min(srcptr0[j], srcptrI[j]);
496 }
497 }
498 else if (op == SUM)
499 {
500 if (!coeffsptr || (coeffsptr[0] == 1.0f && coeffsptr[1] == 1.0f))
501 {
502 for (int j = 0; j < blockSize; j++)
503 {
504 dstptr[j] = srcptr0[j] + srcptrI[j];
505 }
506 }
507 else
508 {
509 float c0 = coeffsptr[0];
510 float c1 = coeffsptr[1];
511 for (int j = 0; j < blockSize; j++)
512 {
513 dstptr[j] = c0*srcptr0[j] + c1*srcptrI[j];
514 }
515 }
516 }
517 else
518 CV_Error(Error::StsInternal, "");
519 }
520 }
521
522 // aggregate other inputs (3+)
523 for (size_t inputIdx = 2; inputIdx < nsrcs; inputIdx++)
524 {
525 int srcI_channels = srcNumChannels[inputIdx];
526 if (c >= srcI_channels)
527 continue; // no data from second input
528 size_t srcIdx = delta + (sampleIdx * srcI_channels + c) * planeSize;
529 const float* srcptrI = srcs[inputIdx]->ptr<float>() + srcIdx;
530
531 if (op == PROD)
532 {
533 for (int j = 0; j < blockSize; j++)
534 {
535 dstptr[j] *= srcptrI[j];
536 }
537 }
538 else if (op == DIV)
539 {
540 for (int j = 0; j < blockSize; j++)
541 {
542 dstptr[j] /= srcptrI[j];
543 }
544 }
545 else if (op == MAX)
546 {
547 for (int j = 0; j < blockSize; j++)
548 {
549 dstptr[j] = std::max(dstptr[j], srcptrI[j]);
550 }
551 }
552 else if (op == MIN)
553 {
554 for (int j = 0; j < blockSize; j++)
555 {
556 dstptr[j] = std::min(dstptr[j], srcptrI[j]);
557 }
558 }
559 else if (op == SUM)
560 {
561 if (!coeffsptr || coeffsptr[inputIdx] == 1.0f)
562 {
563 for (int j = 0; j < blockSize; j++)
564 {
565 dstptr[j] += srcptrI[j];
566 }
567 }
568 else
569 {
570 float cI = coeffsptr[inputIdx];
571 for (int j = 0; j < blockSize; j++)
572 {
573 dstptr[j] += cI * srcptrI[j];
574 }
575 }
576 }
577 else
578 CV_Error(Error::StsInternal, "");
579 }
580 }
581
582 if( activ )
583 {
584 float* ptr = dstptr0 + delta + sampleIdx*channels*planeSize;
585 activ->forwardSlice(ptr, ptr, blockSize, planeSize, 0, channels);
586 }
587 }
588 }

591#ifdef HAVE_OPENCL
592 bool forward_ocl(InputArrayOfArrays inputs_, OutputArrayOfArrays outputs_, OutputArrayOfArrays internals_)
593 {
594 std::vector<UMat> inputs;
595 std::vector<UMat> outputs;
596
597 if ((inputs_.depth() == CV_16F && op != SUM) || (channelsMode != ELTWISE_CHANNNELS_SAME))
598 return false;
599
600 if (hasVecInput)
601 return false; // TODO not implemented yet: https://github.com/opencv/opencv/pull/19477
602
603 inputs_.getUMatVector(inputs);
604 outputs_.getUMatVector(outputs);
605
606 switch (op)
607 {
608 case SUM:
609 {
610 int channels = total(shape(outputs[0]), 0, 2);
611 int plane_size = total(shape(outputs[0]), 2);
612 if (channels % 4 == 0 && plane_size % 4 == 0)
613 {
614 size_t localsize[] = { 128 };
615 size_t globalsize[] = { (size_t)channels / 4 * localsize[0] };
616 String opts;
617 if (inputs_.depth() == CV_16F)
618 opts = " -DDtype=half -DDtype4=half4 -DDtype8=half8";
619 else
620 opts = " -DDtype=float -DDtype4=float4 -DDtype8=float8";
621
622 for (int i = 0; i < (inputs.size() - 1); ++i)
623 {
624 String buildopt = format("-DLOOP=%d", i) + opts;
625 ocl::Kernel kernel("op_sum4", ocl::dnn::eltwise_oclsrc, buildopt);
626 int idx = 0;
627 UMat inpMat = (i == 0) ? inputs[0] : UMat();
628 float coeff1 = (coeffs.empty() || i > 0) ? 1.0f : coeffs[i];
629 float coeff2 = coeffs.empty() ? 1.0f : coeffs[i + 1];
630 kernel.set(idx++, ocl::KernelArg::PtrReadOnly(inputs[0]));
631 kernel.set(idx++, ocl::KernelArg::PtrReadOnly(inputs[1]));
632 kernel.set(idx++, (int)plane_size);
633 kernel.set(idx++, (float)coeff1);
634 kernel.set(idx++, (float)coeff2);
635 kernel.set(idx++, ocl::KernelArg::PtrReadWrite(outputs[0]));
636 bool ret = kernel.run(1, globalsize, localsize, false);
637 if (!ret)
638 return false;
639 }
640 }
641 else
642 {
643 if (inputs_.depth() == CV_16F)
644 return false;
645
646 float coeff1 = coeffs.empty() ? 1.f : coeffs[0];
647 float coeff2 = coeffs.empty() ? 1.f : coeffs[1];
648 UMat mul0, mul1;
649 multiply(coeff1, inputs[0], mul0);
650 multiply(coeff2, inputs[1], mul1);
651 add(mul0, mul1, outputs[0]);
652 for (int i = 2; i < inputs.size(); ++i)
653 {
654 float coeff = coeffs.empty() ? 1.f : coeffs[i];
655 multiply(coeff, inputs[i], mul0);
656 add(mul0, outputs[0], outputs[0]);
657 }
658 }
659 }
660 break;
661 case PROD:
662 multiply(inputs[0], inputs[1], outputs[0]);
663 for (int i = 2; i < inputs.size(); ++i)
664 multiply(inputs[i], outputs[0], outputs[0]);
665 break;
666 case DIV:
667 divide(inputs[0], inputs[1], outputs[0]);
668 for (int i = 2; i < inputs.size(); ++i)
669 divide(outputs[0], inputs[i], outputs[0]);
670 break;
671 case MAX:
672 max(inputs[0], inputs[1], outputs[0]);
673 for (int i = 2; i < inputs.size(); ++i)
674 max(inputs[i], outputs[0], outputs[0]);
675 break;
676 case MIN:
677 min(inputs[0], inputs[1], outputs[0]);
678 for (int i = 2; i < inputs.size(); ++i)
679 min(inputs[i], outputs[0], outputs[0]);
680 break;
681 default:
682 return false;
683 }
684 return true;
685 }

688 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
689 {
690 CV_TRACE_FUNCTION();
691 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
692
693 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
694 forward_ocl(inputs_arr, outputs_arr, internals_arr))
695
696 if (inputs_arr.depth() == CV_16F)
697 {
698 forward_fallback(inputs_arr, outputs_arr, internals_arr);
699 return;
700 }
701
702 std::vector<Mat> inputs, outputs;
703 inputs_arr.getMatVector(inputs);
704 outputs_arr.getMatVector(outputs);
705
706 CV_Assert(outputs.size() == 1);
707 const int nstripes = getNumThreads();
708
709 if (channelsModeInput == ELTWISE_CHANNNELS_SAME && inputs[0].dims > 2)
710 {
711 for (size_t i = 0; i < inputs.size(); i++)
712 {
713 MatShape inpShape = inputs[i].shape();
714 bool allOnes = isAllOnes(inpShape, 2, inputs[i].dims);
715
716 if (allOnes)
717 {
718 Mat tmpInput = inputs[i];
719 MatShape outShape = outputs[0].shape();
720 size_t xSize = outShape[2];
721 for (int j = 3; j < outShape.dims; j++)
722 xSize *= outShape[j];
723

727

729 std::vector<int> outIdx(inpShape.size(), 0);
730

732 {
733 outIdx[0] = idx[0] = j;
734 for(size_t k = 0; k < outShape[1]; k++)
735 {
736 outIdx[1] = idx[1] = k;
737 for (size_t x = 0; x < xSize; x++)
738 {
739 outIdx[2] = x;
740 inputs[i].at<float>(outIdx.data()) = tmpInput.at<float>(idx.data());
741 }
742 }
743 }
744 inputs[i] = inputs[i].reshape(0, outShape);
745 }
746 }
