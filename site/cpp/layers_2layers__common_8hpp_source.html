116inline int onnxDataTypeToCV(OnnxDataType dt)
117{
118 switch (dt)
119 {
120 case ONNX_UINT8: return CV_8U;
121 case ONNX_INT8: return CV_8S;
122 case ONNX_UINT16: return CV_16U;
123 case ONNX_INT16: return CV_16S;
124 case ONNX_UINT32: return CV_32U;
125 case ONNX_INT32: return CV_32S;
126 case ONNX_UINT64: return CV_64U;
127 case ONNX_INT64: return CV_64S;
128 case ONNX_FLOAT: return CV_32F;
129 case ONNX_DOUBLE: return CV_64F;
130 case ONNX_FLOAT16: return CV_16F;
131 case ONNX_BFLOAT16: return CV_16BF;
132 case ONNX_BOOL: return CV_Bool;
133 default:
134 // Fallback to default ONNX FLOAT if value is unknown.
135 return CV_32F;
136 }
137}
