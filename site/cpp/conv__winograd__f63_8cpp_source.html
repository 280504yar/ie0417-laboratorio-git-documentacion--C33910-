17enum { VEC_ALIGN = 32, DFT_TYPE = CV_32F }; // Memory alignment.
18
19int runWinograd63(InputArray _input, InputArray _fusedAddMat, OutputArray _output, const Ptr<FastConv>& conv,
20 int ntasks, float minval, float maxval, ActivationLayer* activ, bool ifMinMaxAct)
21{
22 const cv::dnn::Winofunc func =
23 conv->useFP16 ? cv::dnn::getWinofunc_F16()
24 : (conv->useAVX || conv->useAVX2 || conv->useNEON || conv->useRVV || conv->useSIMD128) ? cv::dnn::getWinofunc_F32()
25 : cv::dnn::Winofunc::empty();
26
27 if (!func.isGood())
28 return 0;
29
30 Mat input = _input.getMat();
31 Mat output = _output.getMat();
32 Mat fusedAddMat = _fusedAddMat.getMat();
33
34 MatShape inputShape = shape(input);
35 MatShape outputShape = shape(output);
36 CV_Assert(inputShape.size() == 4 && outputShape.size() == 4);
37
38 int N = inputShape[0], C = inputShape[1], Hi = inputShape[2], Wi = inputShape[3]; // [N, C, H, W]
39 int K = conv->K;
40 int H0 = outputShape[2], W0 = outputShape[3];
41
42 int pad_top = conv->pad_top;
43 int pad_left = conv->pad_left;
44
45 int ngroups = conv->ngroups, Cg = C/ngroups, Kg = K/ngroups;
46
47 const int CONV_WINO_KBLOCK = 4;
48 const int CONV_WINO_IBLOCK = func.iblock;
49 const int CONV_WINO_ATOM = func.natom;
50 const int CONV_WINO_NATOMS = CONV_WINO_AREA / CONV_WINO_ATOM;
51 const int esz = func.esz;
52
53 int Kg_nblocks = (Kg + CONV_WINO_KBLOCK - 1)/CONV_WINO_KBLOCK;
54 const size_t inp_planesize = (size_t)Hi*Wi;
55 const size_t out_planesize = (size_t)H0*W0;
56
57 int blocks_per_row = (W0+CONV_WINO_STEP-1)/CONV_WINO_STEP;
58 int blocks_per_plane = ((H0+CONV_WINO_STEP-1)/CONV_WINO_STEP)*blocks_per_row;
59 int blocks_per_plane_aligned = ((blocks_per_plane +
60 CONV_WINO_IBLOCK-1)/CONV_WINO_IBLOCK)*CONV_WINO_IBLOCK;
61
62 size_t totalbufsize = (size_t)N*C*blocks_per_plane_aligned*CONV_WINO_AREA;
63
64 AutoBuffer<char> _buf;
65 _buf.allocate((totalbufsize + VEC_ALIGN) * esz);
66 char* wbuf_all = alignPtr(_buf.data(), VEC_ALIGN * esz);
67
68 float* inp = input.ptr<float>();
69 float* out = output.ptr<float>();
70
71 float* fusedAddPtr = fusedAddMat.empty() ? nullptr : fusedAddMat.ptr<float>();
72
73 // Phase 1. compute forward Winograd transforms for all input blocks,
74 // all input planes, all samples in the batch.
75 // [TODO]: maybe, if there are too many input channels, it makes sense to
76 // transform only part of input channels at once and then compute the partial
77 // accumulated sums (i.e. update the output buffers several times,
78 // rather than compute them in one pass).
79 parallel_for_(Range(0, ntasks), [&](const Range& r0) {
80 for (int task_id = r0.start; task_id < r0.end; task_id++)
81 {
82 int nc0 = (N*C)*task_id/ntasks;
83 int nc1 = (N*C)*(task_id+1)/ntasks;
84 for(; nc0 < nc1; nc0++)
85 {
86 int n = nc0 / C;
87 int c = nc0 - n*C;
88 int g = c / Cg;
89 c -= g*Cg;
90
91 for (int block_id = 0; block_id < blocks_per_plane; block_id += CONV_WINO_IBLOCK)
92 {
93 for (int db = 0; db < CONV_WINO_IBLOCK; db++)
94 {
95 size_t inwofs = ((n*ngroups + g)*blocks_per_plane_aligned +
96 block_id)*Cg*CONV_WINO_AREA +
97 (c*CONV_WINO_IBLOCK + db) * CONV_WINO_ATOM;
98 char* inwptr = wbuf_all + inwofs * esz;
99
100 if (block_id + db < blocks_per_plane)
101 {
102 int y0 = (block_id + db) / blocks_per_row;
103 int x0 = (block_id + db) - y0 * blocks_per_row;
104 y0 = y0*CONV_WINO_STEP - pad_top;
105 x0 = x0*CONV_WINO_STEP - pad_left;
106 bool partial = y0 < 0 || y0 + CONV_WINO_SIZE > Hi ||
107 x0 < 0 || x0 + CONV_WINO_SIZE > Wi;
108 int dx1 = 0, dx2 = CONV_WINO_SIZE, dy1 = 0, dy2 = CONV_WINO_SIZE;
109 int inpstep = Wi;
110
111 float inpbuf[CONV_WINO_AREA];
112 float* inptr0 = (float*)inp + nc0*inp_planesize + y0*Wi + x0;
113 float* inptr = inptr0;
114
115 if (partial)
116 {
117 memset(inpbuf, 0, sizeof(inpbuf));
118 dy1 = -y0 > 0 ? -y0 : 0;
119 dy2 = Hi - y0 < CONV_WINO_SIZE ? Hi - y0 : CONV_WINO_SIZE;
120
121 if (dy2 < dy1) {dy2 = dy1 = 0;}
122 dx1 = -x0 > 0 ? -x0 : 0;
123 dx2 = Wi - x0 < CONV_WINO_SIZE ? Wi - x0 : CONV_WINO_SIZE;
124
125 if (dx2 < dx1) {dx2 = dx1 = 0;}
126 inptr0 -= y0*Wi + x0;
127
128 if (dx1 < dx2 && dy1 < dy2)
129 {
130 for(int dy = dy1; dy < dy2; dy++)
131 memcpy(&inpbuf[dy*CONV_WINO_SIZE + dx1],
132 inptr0 + (y0+dy)*Wi + (x0+dx1),
133 (dx2-dx1)*sizeof(inpbuf[0]));
134 }
135
136 inptr = inpbuf;
137 inpstep = CONV_WINO_SIZE;
138 }
139 func.BtXB_8x8(inptr, inpstep, (uchar*)inwptr, Cg, CONV_WINO_IBLOCK, CONV_WINO_ATOM);
140 }
141 else
142 {
143 for (int i = 0; i < CONV_WINO_NATOMS; i++, inwptr += CONV_WINO_IBLOCK * CONV_WINO_ATOM * esz)
144 memset(inwptr, 0, CONV_WINO_ATOM * esz);
145 }
146 }
147 }
148 }
149 }});
150
151 // Phase 2. compute elemwise-weighted sums of transformed blocks,
152 // apply inverse Winograd transforms to the sums,
153 // add bias, apply activation function if any and store the results.
154 char* wptr0 = nullptr;
155 if (esz == 2)
156 {
157 CV_Assert(!conv->weightsWinoBuf_FP16.empty());
158 wptr0 = (char *)conv->getWeightsWinoFP16();
159 }
160 else if (esz == 4)
161 {
162 CV_Assert(!conv->weightsWinoBuf.empty());
163 wptr0 = (char *)conv->getWeightsWino();
164 }
165 else
166 {
167 CV_Error(Error::StsError, "Impossible configuration");
168 }
169
170 parallel_for_(Range(0, ntasks), [&](const Range& r0) {
171 for (int task_id = r0.start; task_id < r0.end; task_id++)
172 {
173 size_t out_wbuf_size = CONV_WINO_AREA * CONV_WINO_KBLOCK * CONV_WINO_IBLOCK;
174 size_t outbuf_size = CONV_WINO_AREA;
175
176 // For saving the accumulation output.
177 AutoBuffer<char> out_wbuf_;
178 out_wbuf_.allocate((out_wbuf_size + VEC_ALIGN) * esz);
179 char* out_wbuf = alignPtr(out_wbuf_.data(), VEC_ALIGN * esz);
180 memset(out_wbuf, 0, out_wbuf_size * esz);
181
182 // For saving the fuse_Add data.
183 AutoBuffer<float> outbuf_;
184 outbuf_.allocate(outbuf_size + VEC_ALIGN);
185 float* outbuf = alignPtr(outbuf_.data(), VEC_ALIGN);
186 memset(outbuf, 0, outbuf_size * sizeof(outbuf[0]));
187
188 int ngk0 = (int)(((int64_t)N*Kg_nblocks*ngroups)*task_id/ntasks);
189 int ngk1 = (int)(((int64_t)N*Kg_nblocks*ngroups)*(task_id+1)/ntasks);
190
191 for(; ngk0 < ngk1; ngk0++)
192 {
193 int n = ngk0 / (Kg_nblocks*ngroups);
194 int gk0 = ngk0 % (Kg_nblocks*ngroups);
195 int g = gk0 / Kg_nblocks;
196 int k0 = (gk0 % Kg_nblocks)*CONV_WINO_KBLOCK;
197 int k1 = k0 + CONV_WINO_KBLOCK <= Kg ? k0 + CONV_WINO_KBLOCK : Kg;
198
199 for (int block_id0 = 0; block_id0 < blocks_per_plane; block_id0 += CONV_WINO_IBLOCK)
200 {
201 int block_id1 = block_id0 + CONV_WINO_IBLOCK;
202 block_id1 = block_id1 < blocks_per_plane ? block_id1 : blocks_per_plane;
203 size_t inwofs = ((n*ngroups + g)*blocks_per_plane_aligned + block_id0)*Cg*CONV_WINO_AREA;
204 size_t wofs = (g*Kg_nblocks*CONV_WINO_KBLOCK + k0)*Cg*CONV_WINO_AREA;
205
206 char* inwptr = wbuf_all + inwofs * esz;
207 char* wptr = wptr0 + wofs * esz;
208
209 func.accum((uchar*)inwptr, (uchar*)wptr, (uchar*)out_wbuf, Cg,
210 block_id1 - block_id0, CONV_WINO_IBLOCK,
211 CONV_WINO_KBLOCK, CONV_WINO_ATOM, CONV_WINO_NATOMS);
212
213 for (int k = k0; k < k1; k++)
214 {
215 float biasv = conv->biasBuf[g*Kg + k];
216 for (int block_id = block_id0; block_id < block_id1; block_id++)
217 {
218 int y0 = block_id / blocks_per_row;
219 int x0 = block_id - y0 * blocks_per_row;
220 y0 = y0*CONV_WINO_STEP;
221 x0 = x0*CONV_WINO_STEP;
222 int dy1 = H0 - y0;
223 if (dy1 > CONV_WINO_STEP) dy1 = CONV_WINO_STEP;
224 int dx1 = W0 - x0;
225 if (dx1 > CONV_WINO_STEP) dx1 = CONV_WINO_STEP;
226 assert(dx1 > 0 && dy1 > 0);
227 bool partial = activ || dy1 < CONV_WINO_STEP || dx1 < CONV_WINO_STEP;
228 size_t outofs = (n*K + g*Kg + k)*out_planesize + y0*W0 + x0;
229 int outstep = W0;
230
231 float* outptr0 = (float*)out + outofs;
232 float* pbptr0 = fusedAddPtr ? fusedAddPtr + outofs : nullptr;
233 float *outptr = outptr0, *bpptr = pbptr0;
234
235 if (partial)
236 {
237 outptr = outbuf;
238 outstep = CONV_WINO_SIZE;
239 if (pbptr0)
240 {
241 bpptr = outbuf;
242 for (int y = 0; y < dy1; y++)
243 memcpy(outbuf + y*CONV_WINO_SIZE, pbptr0 + y*W0,
244 dx1*sizeof(pbptr0[0]));
245 }
246 }
247
248 const int count = ((k - k0)*CONV_WINO_IBLOCK + (block_id - block_id0))*CONV_WINO_AREA;
249 func.AtXA_8x8((uchar*)out_wbuf + count * esz, CONV_WINO_SIZE,
250 bpptr, outstep, outptr, outstep, biasv, minval, maxval, ifMinMaxAct);
251
252 if (partial)
253 {
254 if (activ)
255 activ->forwardSlice(outptr, outptr, CONV_WINO_SIZE*CONV_WINO_STEP, 0, g*Kg + k, g*Kg + k + 1);
256 for (int y = 0; y < dy1; y++)
257 memcpy(outptr0 + y*W0, outptr + y*CONV_WINO_SIZE, dx1*sizeof(outptr0[0]));
258 }
259 }
260 }
261 }
262 }
263 }});
264 return 1;
265}
