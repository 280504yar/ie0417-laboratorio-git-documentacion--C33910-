5// The new element-wise expression engine - low-level contract shared by the kernels
6// (arithm.simd.hpp), the per-op dispatchers (arithm.dispatch.cpp) and the
7// graph compiler / executor / parser (arithm_expr.cpp).
8//
9// Private core header for now; once cv::add() is rebuilt on top of it, the public-facing parts
10// (cv::expression, the get*Func entry points) move to external headers. Assumes precomp.hpp (Mat,
11// MatShape, AutoBuffer, Scalar) is already included.
12//
13// Design notes (agreed):
14// - Universal arity: every instruction is {fptr, arg0, arg1, arg2, result}; unused operands
15// reference the reserved "none" arg slot (index 0).
16// - A kernel processes one 2D tile of a single slice; broadcasting is per-operand y/x steps
17// (step 0 = re-read). Steps are in ELEMENTS; dst is contiguous in x (dst.stepx == 1).

30// Sentinel depth for an unused operand (note: CV_8U == 0 is a *valid* depth,
31// so the "no operand" marker must be negative).
32enum { EW_DEPTH_NONE = -1 };
33
34// Each op carries a fixed numerical value whose high bits encode its arity, so the arity
35// can be recovered with a shift and no lookup table: arity = (op >> OP_ARITY_SHIFT) & 7.
36// The ops are grouped into contiguous arity blocks (unary = 1<<10, binary = 2<<10,
37// ternary = 3<<10); within a block the low bits are just a running index.
38enum { OP_ARITY_SHIFT = 10 };
39enum
40{
41 OP_UNARY_BASE = 1 << OP_ARITY_SHIFT, // 0x400
42 OP_BINARY_BASE = 2 << OP_ARITY_SHIFT, // 0x800
43 OP_TERNARY_BASE = 3 << OP_ARITY_SHIFT // 0xC00
44};
45
46// The single enumeration of element-wise operations, used both in the IR and by the
47// kernel dispatcher.
48enum TOp
49{
50 OP_NOP = 0,
51
52 // ---------------- unary (arity 1) ----------------
53 OP_NEG = OP_UNARY_BASE, OP_ABS, OP_NOT,
54 OP_SQRT, OP_EXP, OP_LOG,
55 OP_SIN, OP_COS, OP_TANH, OP_ERF, OP_RELU,
56 OP_CAST, // saturating type conversion, no scaling
57
58 // ---------------- binary (arity 2) ----------------
59 OP_ADD = OP_BINARY_BASE, OP_SUB, OP_MUL, OP_DIV, OP_POW,
60 OP_MIN, OP_MAX, OP_ABSDIFF,
61 OP_AND, OP_OR, OP_XOR,
62 // compare -> mask (result depth given explicitly, e.g. CV_Bool/CV_8U)
63 OP_CMP_EQ, OP_CMP_NE, OP_CMP_LT, OP_CMP_LE, OP_CMP_GT, OP_CMP_GE,
64 // addWeighted: a*alpha + b*beta + gamma (params = {alpha, beta, gamma}). A fused composite, not a
65 // kernel - emitBinary expands it. Placed last in the binary group so it doesn't renumber the ops
66 // above it (some dispatch is by enum value).
67 OP_ADDW,
68 // hypot(x, y) = sqrt(x^2 + y^2), the magnitude of a 2D vector. NAIVE evaluation (matching
69 // cv::magnitude), NOT the overflow-safe std::hypot. Computes in the float domain like OP_POW.
70 OP_HYPOT,
71 // atan2(y, x) in RADIANS, standard C range (-pi, pi] (NB: cv::phase/fastAtan2 use degrees
72 // [0, 360)). arg0 = y, arg1 = x, like std::atan2. Float domain, like OP_POW/OP_HYPOT.
73 OP_ATAN2,
74
75 // ---------------- ternary (arity 3) ----------------
76 OP_CLAMP = OP_TERNARY_BASE, // clamp(x, lo, hi)
77 // select(mask, a, b) (a.k.a. where): dst = (mask != 0) ? a : b; mask is 1 byte (bool/u8/s8),
78 // never cast. Also the engine's masked-op tail: cv::add(..., mask) computes into a temp r,
79 // then select(mask, r, dst) -> dst overwrites only the masked subset of the (pre-existing)
80 // output (dst rides as both arg2 and the result slot; the kernel is alias-safe).
81 OP_SELECT,
82 OP_CONVERT_SCALE // cast<rdepth>(src*scale + offset); scale/offset may be tensors
83};
84
85// Arity from the encoding above (0 for OP_NOP). No table to keep in lock-step with the enum.
86inline int opArity(TOp op) { return ((int)op >> OP_ARITY_SHIFT) & 7; }

88// Operation category — the graph compiler's type-inference rules differ per category.
89enum ElemwiseCategory
90{
91 CAT_ARITH = 0, // numeric, result type follows promotion rules
92 CAT_BITWISE, // integer-only, same-type
93 CAT_COMPARE, // produces a mask, result depth is explicit
94 CAT_MATH, // transcendental, float domain
95 CAT_CAST, // type conversion (with/without scaling)
96 CAT_SELECT // data-routing (select/where)
97};

109// 2^width-1 (|(-128)-127| = 255), so it needs the UNSIGNED type of the same width (8s->8u, 16s->16u,
110// 32s->32u, 64s->64u) to hold it without saturation. Unsigned/float depths keep their type.

113// ---------------------------------------------------------------------------
114// Steps for one operand, in elemsize1 units, one entry per shape dim (parallel to a
115// MatShape). A 0 entry means broadcast along that axis. Heap-free, like MatShape.
116// ---------------------------------------------------------------------------
117typedef std::array<size_t, MatShape::MAX_DIMS> EwSteps;

119// ---------------------------------------------------------------------------
120// The low-level kernel contract.
121//
122// Processes a width x height tile of one slice. Each source operand carries its own
123// (stepy, stepx) in elements; a 0 step means broadcast along that axis. stepx is restricted
124// to {0,1} (1 = contiguous, 0 = broadcast-scalar along x): the general strided/gather case is
125// excluded and the executor guarantees the invariant. The result is contiguous in x (dst stepx == 1).
126// Returns >= 0 on success, < 0 (a CV_HAL_ERROR_* code) to let the caller fall back.
127//
128// The trailing `params` points at the instruction's scalar parameter block (Insn::params, a cv::Scalar's
129// 4 doubles): mul/div read params[0] as a scale (1.0 = none). `flags` carries small per-kernel options
130// (e.g. a compare op's 0/1-vs-0/255 mask value); `userdata` carries a wrapped core BinaryFunc for casts.
131// ---------------------------------------------------------------------------
132typedef int (*KernelFunc)(
133 const void* src0, size_t step0y, size_t step0x,
134 const void* src1, size_t step1y, size_t step1x,
135 const void* src2, size_t step2y, size_t step2x,
136 void* dst, size_t dstepy,
137 int width, int height, const double* params,
138 int flags, void* userdata);
139
140// TKernel::flags - small per-kernel options, interpreted by the kernel itself (so the meaning is
141// per-kernel: the cast kernels read it as the dst element size, the compare kernels as the bits below).
142enum TKernelFlags
143{
144 EW_KERNEL_MASK1 = 1, // compare: emit a 0/1 mask instead of the default 0/255 (cv::compare)
145 EW_KERNEL_SWAP01 = 2, // compare: the kernel swaps its own src0<->src1 (with their steps), so
146 // LT/LE reuse the GT/GE kernels (a<b == b>a, a<=b == b>=a)
147 // compare fuses its post-op fix-up: the u8 result is (rawmask & M) | V per channel. Uniform (this
148 // bit clear): M = trueVal (MASK1 ? 1 : 255), V = 0 - an ordinary compare. Per-channel (this bit
149 // set, only the divergent multi-channel scalar case): M/V come from the 4-bit fields below. This
150 // folds the former separate patch pass into the compare kernel (one pass, no extra kernel).
151 EW_CMP_PATCH = 4,
152 EW_CMP_PATCH_SHIFT = 8, // per channel c in [0,4): bits [SHIFT+c*4 .. +4) = 2 bits M then 2 bits V;
153 // each 2-bit field decodes 0->0x00, 1->0x01, 2|3->0xFF.
154};
155
156// Decode a 2-bit patch field (0->0, 1->1, else 255) - shared by the compare kernel and its builder.
157inline int cmpPatchByte(int twoBits) { return twoBits == 0 ? 0 : twoBits == 1 ? 1 : 255; }

159struct TKernel

161 KernelFunc fptr = nullptr;
162 void* userdata = nullptr;
163 int flags = 0;

202// An element-wise expression as ONE flat program (the analogue of cv::MatExpr for element-wise ops):
203// an arg table (`arginfo`, the typed operands) + an instruction list (`prog`). The program IS the
204// representation - it is built directly:
205// - declare operands with addInput()/addConst()/addOutput() (and addTemp() for intermediates);
206// - append operations with addInsn() (a single op, you pick the slots) or, for automatic type
207// inference + cast insertion, with emitUnary()/emitBinary()/emitTernary().
208// Operand types are known at build time, so addInsn resolves each instruction's kernel on the spot;
209// compile() is a cheap finalize pass (pack temps into physical buffers via liveness).
210//
211// Heap-free for the common case: to back cv::add() the program is (re)built every call, so the
212// containers must not allocate for typical (small) expressions. AutoBuffer keeps a handful of
213// insns/slots inline on the stack and only spills to the heap for large expressions; it is copyable,
214// so TExpr is still returned/passed by value. A default-constructed AutoBuffer is empty (size()==0)
215// and grows like std::vector (push_back amortized 1.5x); clear() resets the size to 0.
216struct CV_EXPORTS TExpr
217{
218 // Static (shape-independent) classification of an arg slot.
219 enum ArgKind
220 {
221 NONE = 0, // the reserved empty operand (slot 0)
222 INPUT,
223 CONST,
224 TEMP,
225 OUTPUT
226 };
227
228 struct Arg
229 {
230 ArgKind kind = NONE;
231 int depth = EW_DEPTH_NONE; // EW_DEPTH_NONE on a CONST = "flexible" (the emit* layers type it per use)
232 int channels = 0; // for CONST: # of per-channel values (0/1 => single broadcast value)
233 int index = -1; // input#/output#/temp-id depending on kind
234 // CONST only: the constant's values live in TExpr::constbuf (in `srcdepth` until compile(),
235 // which converts them to the resolved `depth`). constofs = offset into constbuf, in uint64_t
236 // units. A per-channel scalar of any width is carried this way (no 4-channel Scalar limit).
237 int srcdepth = EW_DEPTH_NONE;
238 size_t constofs = 0;
239 };
240
241 // One compiled instruction: the op + arg-table indices + a resolved kernel (TKernel: fptr +
242 // userdata + flags). Every kernel uses the SAME calling convention (the universal KernelFunc);
243 // OP_CAST / OP_CONVERT_SCALE bind castKernel, which carries the core convert BinaryFunc in
244 // kernel.userdata. `params` is the per-instruction scalar block (mul/div scale in params[0];
245 // convert_scale {scale, offset} in params[0..1]). The kernel is bound by addInsn at build time.
246 struct Insn
247 {
248 TKernel kernel;
249 TOp op = OP_NOP;
250 int arg0 = 0, arg1 = 0, arg2 = 0, result = 0;
251 Scalar params = Scalar(1); // op scalars; params[0]=scale defaults to 1 (identity)
252 };
253
254 AutoBuffer<Insn, 16> prog; // instructions, in execution order (kernels bound by addInsn)
255 AutoBuffer<Arg, 16> arginfo; // slot 0 is always NONE
256 AutoBuffer<uint64_t, 16> constbuf; // CONST value store (uint64-aligned slots); Arg::constofs
257 // indexes it. Source values on build, resolved-type after compile()
258 int ninputs = 0;
259 int noutputs = 0;
260 int ntemps = 0;
261 int nbuffers = 0; // distinct physical temp buffers after liveness
262 int nconsts = 0; // # materialized CONST slots (set by compile()) - sizes
263 // the const store; nconsts==0 enables exec()'s fast path
264 int capElems = 0; // # elements one ~16KB L1 scratch fragment holds (set by
265 // compile()); INT_MAX when there are no temps.
266 AutoBuffer<int, 16> bufferOfTemp; // temp-id -> physical buffer id
267 AutoBuffer<int, 8> bufEszPrefix; // [nbuffers+1] prefix sums of each physical temp buffer's
268 // elem size (set by compile()); [nbuffers] = temp bytes
269 // per output element.
270
271 TExpr();
272 void clear(); // reset to an empty program (slot 0 = NONE)
273 void dump(std::ostream& os) const; // human-readable slot table + instruction list (debug)
274
275 // ---- operand / instruction builders (return the new slot / instruction index) ----
276 int addInput(int depth);
277 int addConst(int depth, const Scalar& v, int channels = 1); // source = Scalar (f64); depth NONE => flexible
278 int addConst(int depth, int srcdepth, const void* data, int channels); // source = native bytes
279 // A typed copy of a flexible CONST `srcSlot` at the resolved `depth` (shares its source values;
280 // compile() converts them). Used by the emit* layers / parser cast where addConst(depth, cval) was.
281 int typedConstFrom(int srcSlot, int depth);
282 int addTemp(int depth);
283 int addOutput(int depth);
284 // addInsn resolves the instruction's kernel NOW from the operand/result depths (final at build
285 // time). The 2nd form takes a pre-resolved kernel, for callers that already probed getElemwiseFunc.
286 int addInsn(TOp op, int a0, int a1, int a2, int result, const Scalar& params = Scalar(1));
287 int addInsn(TOp op, int a0, int a1, int a2, int result, const TKernel& kernel,
288 const Scalar& params = Scalar(1));
289
290 // Return `arg` unchanged if it is already of depth `depth`; otherwise append an OP_CAST into a
291 // fresh temp of that depth and return the temp's slot. The one place casts are inserted.
292 int maybeAddCast(int arg, int depth);
293
294 // ---- type-inference + cast-insertion policy layers (parser + hand builders) ----
295 // Each derives the result depth (promotion, or a forced `rdepth`), picks the compute depth and a
296 // wide fallback per op family, materializes flexible CONST operands, casts every operand to the
297 // compute depth, then emits `op` (direct when a kernel exists, else compute wide and cast down).
298 // Returns the slot holding the result. `rdepth` EW_DEPTH_NONE = auto.
299 int emitUnary(TOp op, int a, int rdepth = EW_DEPTH_NONE, const Scalar& params = Scalar(1));
300 int emitBinary(TOp op, int a, int b, int rdepth = EW_DEPTH_NONE, const Scalar& params = Scalar(1));
301 int emitTernary(TOp op, int a, int b, int c, int rdepth = EW_DEPTH_NONE);
302
303 // Land `temp` in the existing slot `out`: redirect temp's single producer to write `out` directly
304 // (dropping a dead last temp so compile() keeps its no-temp fast exit); otherwise copy via OP_CAST.
305 int moveToOutput(int temp, int out);
306
307 // Declare a result tensor fed by `rootSlot` (a fresh OUTPUT of its depth) and moveToOutput into it.
308 int output(int rootSlot);
309
310 // Finalize: pack temps into physical buffers (liveness), count consts, size the L1 fragment cap.
311 void compile();
312
313 // The broadcast output geometry for the given inputs (spatial dims + channel count, channels
314 // innermost). All outputs share it; their depth comes from arginfo. Lets a caller pre-create the
315 // destination (dst.create(spatial, CV_MAKETYPE(depth, channels))) before exec writes into it.
316 // Inputs are passed as an array of pointers (no Mat-header copies in the hot path).
317 void outputShape(const Mat* const* inputs, MatShape& spatial, int& channels) const;
318
319 // Execute the compiled program over a set of input Mats, producing the broadcast result(s). If an
320 // output Mat already has the right shape/type it is reused (not reallocated). Inputs are passed as
321 // an array of pointers.
322 void exec(const Mat* const* inputs, Mat* outputs);
323
324 // Convenience overloads: inputs as a contiguous array of Mats (builds the pointer array + forwards).
325 // Handy for callers holding a Mat[]/vector<Mat>; the hot path should pass pointers directly.
326 void outputShape(const Mat* inputs, MatShape& spatial, int& channels) const;
327 void exec(const Mat* inputs, Mat* outputs);
328};
