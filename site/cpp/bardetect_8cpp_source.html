16 top_left = static_cast<float>(*((ptr) + left_col + integral_cols * top_row));\

23inline bool Detect::isValidCoord(const Point &coord, const Size &limit)
24{
25 if ((coord.x < 0) || (coord.y < 0))
26 {
27 return false;
28 }
29
30 if ((unsigned) coord.x > (unsigned) (limit.width - 1) || ((unsigned) coord.y > (unsigned) (limit.height - 1)))
31 {
32 return false;
33 }
34
35 return true;
36}

52inline void GetMaxScoreIndex(const std::vector<float>& scores, const float threshold, const int top_k,
53 std::vector<std::pair<float, int> >& score_index_vec)
54{
55 CV_DbgAssert(score_index_vec.empty());
56 // Generate index score pairs.
57 for (size_t i = 0; i < scores.size(); ++i)
58 {
59 if (scores[i] > threshold)
60 {
61 score_index_vec.push_back(std::make_pair(scores[i], (int)i));
62 }
63 }
64
65 // Sort the score pair according to the scores in descending order
66 std::stable_sort(score_index_vec.begin(), score_index_vec.end(),
67 SortScorePairDescend<int>);
68
69 // Keep top_k scores if needed.
70 if (top_k > 0 && top_k < (int)score_index_vec.size())
71 {
72 score_index_vec.resize(top_k);
73 }
74}

76template <typename BoxType>
77inline void NMSFast_(const std::vector<BoxType>& bboxes,
78 const std::vector<float>& scores, const float score_threshold,
79 const float nms_threshold, const float eta, const int top_k,
80 std::vector<int>& indices,
81 float (*computeOverlap)(const BoxType&, const BoxType&),
82 size_t limit = std::numeric_limits<int>::max())
83{
84 CV_Assert(bboxes.size() == scores.size());
85
86 // Get top_k scores (with corresponding indices).
87 std::vector<std::pair<float, int> > score_index_vec;
88 GetMaxScoreIndex(scores, score_threshold, top_k, score_index_vec);
89
90 // Do nms.
91 float adaptive_threshold = nms_threshold;
92 indices.clear();
93 for (size_t i = 0; i < score_index_vec.size(); ++i) {
94 const int idx = score_index_vec[i].second;
95 bool keep = true;
96 for (int k = 0; k < (int)indices.size() && keep; ++k) {
97 const int kept_idx = indices[k];
98 float overlap = computeOverlap(bboxes[idx], bboxes[kept_idx]);
99 keep = overlap <= adaptive_threshold;
100 }
101 if (keep) {
102 indices.push_back(idx);
103 if (indices.size() >= limit) {
104 break;
105 }
106 }
107 if (keep && eta < 1 && adaptive_threshold > 0.5) {
108 adaptive_threshold *= eta;
109 }
110 }
111}

113static inline float rotatedRectIOU(const RotatedRect& a, const RotatedRect& b)
114{
115 std::vector<Point2f> inter;
116 int res = rotatedRectangleIntersection(a, b, inter);
117 if (inter.empty() || res == INTERSECT_NONE)
118 return 0.0f;
119 if (res == INTERSECT_FULL)
120 return 1.0f;
121 float interArea = (float)contourArea(inter);
122 return interArea / (a.size.area() + b.size.area() - interArea);
123}

125static void NMSBoxes(const std::vector<RotatedRect>& bboxes, const std::vector<float>& scores,
126 const float score_threshold, const float nms_threshold,
127 std::vector<int>& indices, const float eta = 1.f, const int top_k = 0)
128{
129 CV_Assert_N(bboxes.size() == scores.size(), score_threshold >= 0,
130 nms_threshold >= 0, eta > 0);
131 NMSFast_(bboxes, scores, score_threshold, nms_threshold, eta, top_k, indices, rotatedRectIOU);
132}

175{
176

200bool Detect::computeTransformationPoints()
201{
202
203 bbox_indices.clear();
204 transformation_points.clear();
205 transformation_points.reserve(bbox_indices.size());
206 RotatedRect rect;
207 Point2f temp[4];
208
218 const float THRESHOLD_WSCALE = (purpose != UNCHANGED) ? 20 : (20 * width / 512.f);
219 const float THRESHOLD_HSCALE = (purpose != UNCHANGED) ? 15 : (15 * height / 512.f);
220 const float THRESHOLD_SCORE = (width / THRESHOLD_WSCALE) * (height / THRESHOLD_HSCALE);
221
222 NMSBoxes(localization_bbox, bbox_scores, THRESHOLD_SCORE, 0.1f, bbox_indices);
223
224 for (const auto &bbox_index : bbox_indices)
225 {
226 rect = localization_bbox[bbox_index];
227 if (purpose == ZOOMING)
228 {
229 rect.center /= coeff_expansion;
230 rect.size.height /= static_cast<float>(coeff_expansion);
231 rect.size.width /= static_cast<float>(coeff_expansion);
232 }
233 else if (purpose == SHRINKING)
234 {
235 rect.center *= coeff_expansion;
236 rect.size.height *= static_cast<float>(coeff_expansion);
237 rect.size.width *= static_cast<float>(coeff_expansion);
238 }
239 rect.points(temp);
240 transformation_points.emplace_back(vector<Point2f>{temp[0], temp[1], temp[2], temp[3]});
241 }
242
243 return !transformation_points.empty();
244}

247void Detect::preprocess(double detectorGradientMagnitudeThresh)
248{
249 Mat scharr_x, scharr_y, temp;
250 Scharr(resized_barcode, scharr_x, CV_32F, 1, 0);
251 Scharr(resized_barcode, scharr_y, CV_32F, 0, 1);
252 // calculate magnitude of gradient and truncate
253 magnitude(scharr_x, scharr_y, temp);
254 threshold(temp, temp, detectorGradientMagnitudeThresh, 1, THRESH_BINARY);
255 temp.convertTo(gradient_magnitude, CV_8U);
256 integral(gradient_magnitude, integral_edges, CV_32F);
257
258
259 for (int y = 0; y < height; y++)
260 {
261 auto *const x_row = scharr_x.ptr<float_t>(y);
262 auto *const y_row = scharr_y.ptr<float_t>(y);
263 auto *const magnitude_row = gradient_magnitude.ptr<uint8_t>(y);
264 for (int pos = 0; pos < width; pos++)
265 {
266 if (magnitude_row[pos] == 0)
267 {
268 x_row[pos] = 0;
269 y_row[pos] = 0;
270 continue;
271 }
272 if (x_row[pos] < 0)
273 {
274 x_row[pos] *= -1;
275 y_row[pos] *= -1;
276 }
277 }
278 }
279 integral(scharr_x, temp, integral_x_sq, CV_32F, CV_32F);
280 integral(scharr_y, temp, integral_y_sq, CV_32F, CV_32F);
281 integral(scharr_x.mul(scharr_y), integral_xy, temp, CV_32F, CV_32F);
282}

285// Change coherence orientation edge_nums
286// depend on width height integral_edges integral_x_sq integral_y_sq integral_xy
287void Detect::calCoherence(int window_size)
288{
289 static constexpr float THRESHOLD_COHERENCE = 0.9f;
290 int right_col, left_col, top_row, bottom_row;
291 float xy, x_sq, y_sq, d, rect_area;
292 const float THRESHOLD_AREA = float(window_size * window_size) * 0.42f;
293 Size new_size(width / window_size, height / window_size);
294 coherence = Mat(new_size, CV_8U), orientation = Mat(new_size, CV_32F), edge_nums = Mat(new_size, CV_32F);
295
296 float top_left, top_right, bottom_left, bottom_right;
297 int integral_cols = width + 1;
298 const auto *edges_ptr = integral_edges.ptr<float_t>(), *x_sq_ptr = integral_x_sq.ptr<float_t>(), *y_sq_ptr = integral_y_sq.ptr<float_t>(), *xy_ptr = integral_xy.ptr<float_t>();
299 for (int y = 0; y < new_size.height; y++)
300 {
301 auto *coherence_row = coherence.ptr<uint8_t>(y);
302 auto *orientation_row = orientation.ptr<float_t>(y);
303 auto *edge_nums_row = edge_nums.ptr<float_t>(y);
304 if (y * window_size >= height)
305 {
306 continue;
307 }
308 top_row = y * window_size;
309 bottom_row = min(height, (y + 1) * window_size);
310
311 for (int pos = 0; pos < new_size.width; pos++)
312 {
313
314 // then calculate the column locations of the rectangle and set them to -1
315 // if they are outside the matrix bounds
316 if (pos * window_size >= width)
317 {
318 continue;
319 }
320 left_col = pos * window_size;
321 right_col = min(width, (pos + 1) * window_size);
322
323 //we had an integral image to count non-zero elements
324 CALCULATE_SUM(edges_ptr, rect_area)
325 if (rect_area < THRESHOLD_AREA)
326 {
327 // smooth region
328 coherence_row[pos] = 0;
329 continue;
330 }
331
332 CALCULATE_SUM(x_sq_ptr, x_sq)
333 CALCULATE_SUM(y_sq_ptr, y_sq)
334 CALCULATE_SUM(xy_ptr, xy)
335
336 // get the values of the rectangle corners from the integral image - 0 if outside bounds
337 d = sqrt((x_sq - y_sq) * (x_sq - y_sq) + 4 * xy * xy) / (x_sq + y_sq);
338 if (d > THRESHOLD_COHERENCE)
339 {
340 coherence_row[pos] = 255;
341 orientation_row[pos] = atan2(x_sq - y_sq, 2 * xy) / 2.0f;
342 edge_nums_row[pos] = rect_area;
343 }
344 else
345 {
346 coherence_row[pos] = 0;
347 }
348
349 }
350
351 }
352}

354// will change localization_bbox bbox_scores
355// will change coherence,
356// depend on coherence orientation edge_nums
357void Detect::regionGrowing(int window_size)
358{
359 static constexpr float LOCAL_THRESHOLD_COHERENCE = 0.95f, THRESHOLD_RADIAN =
360 PI / 30, LOCAL_RATIO = 0.5f, EXPANSION_FACTOR = 1.2f;
361 static constexpr uint THRESHOLD_BLOCK_NUM = 35;
362 Point pt_to_grow, pt; //point to grow
363
364 float src_value;
365 float cur_value;
366 float edge_num;
367 float rect_orientation;
368 float sin_sum, cos_sum;
369 uint counter;
370 //grow direction
371 static constexpr int DIR[8][2] = {{-1, -1},
372 {0, -1},
373 {1, -1},
374 {1, 0},
375 {1, 1},
376 {0, 1},
377 {-1, 1},
378 {-1, 0}};
379 vector<Point2f> growingPoints, growingImgPoints;
380 for (int y = 0; y < coherence.rows; y++)
381 {
382 auto *coherence_row = coherence.ptr<uint8_t>(y);
383
384 for (int x = 0; x < coherence.cols; x++)
385 {
386 if (coherence_row[x] == 0)
387 {
388 continue;
389 }
390 // flag
391 coherence_row[x] = 0;
392 growingPoints.clear();
393 growingImgPoints.clear();
394
395 pt = Point(x, y);
396 cur_value = orientation.at<float_t>(pt);
397 sin_sum = sin(2 * cur_value);
398 cos_sum = cos(2 * cur_value);
399 counter = 1;
400 edge_num = edge_nums.at<float_t>(pt);
401 growingPoints.push_back(pt);
402 growingImgPoints.push_back(Point(pt));
403 while (!growingPoints.empty())
404 {
405 pt = growingPoints.back();
406 growingPoints.pop_back();
407 src_value = orientation.at<float_t>(pt);
408
409 //growing in eight directions
410 for (auto i : DIR)
411 {
412 pt_to_grow = Point(pt.x + i[0], pt.y + i[1]);
413
414 //check if out of boundary
415 if (!isValidCoord(pt_to_grow, coherence.size()))
416 {
417 continue;
418 }
419
420 if (coherence.at<uint8_t>(pt_to_grow) == 0)
421 {
422 continue;
423 }
424 cur_value = orientation.at<float_t>(pt_to_grow);
425 if (abs(cur_value - src_value) < THRESHOLD_RADIAN ||
426 abs(cur_value - src_value) > PI - THRESHOLD_RADIAN)
427 {
428 coherence.at<uint8_t>(pt_to_grow) = 0;
429 sin_sum += sin(2 * cur_value);
430 cos_sum += cos(2 * cur_value);
431 counter += 1;
432 edge_num += edge_nums.at<float_t>(pt_to_grow);
433 growingPoints.push_back(pt_to_grow); //push next point to grow back to stack
434 growingImgPoints.push_back(pt_to_grow);
435 }
436 }
437 }
438 //minimum block num
439 if (counter < THRESHOLD_BLOCK_NUM)
440 {
441 continue;
442 }
443 float local_coherence = (sin_sum * sin_sum + cos_sum * cos_sum) / static_cast<float>(counter * counter);
444 // minimum local gradient orientation_arg coherence_arg
445 if (local_coherence < LOCAL_THRESHOLD_COHERENCE)
446 {
447 continue;
448 }
449 RotatedRect minRect = minAreaRect(growingImgPoints);
450 if (edge_num < minRect.size.area() * float(window_size * window_size) * LOCAL_RATIO ||
451 static_cast<float>(counter) < minRect.size.area() * LOCAL_RATIO)
452 {
453 continue;
454 }
455 const float local_orientation = atan2(cos_sum, sin_sum) / 2.0f;
456 // only orientation_arg is approximately equal to the rectangle orientation_arg
457 rect_orientation = (minRect.angle) * PI / 180.f;
458 if (minRect.size.width < minRect.size.height)
459 {
460 rect_orientation += (rect_orientation <= 0.f ? HALF_PI : -HALF_PI);
461 std::swap(minRect.size.width, minRect.size.height);
462 }
463 if (abs(local_orientation - rect_orientation) > THRESHOLD_RADIAN &&
464 abs(local_orientation - rect_orientation) < PI - THRESHOLD_RADIAN)
465 {
466 continue;
467 }
468 minRect.angle = local_orientation * 180.f / PI;
469 minRect.size.width *= static_cast<float>(window_size) * EXPANSION_FACTOR;
470 minRect.size.height *= static_cast<float>(window_size);
471 minRect.center.x = (minRect.center.x + 0.5f) * static_cast<float>(window_size);
472 minRect.center.y = (minRect.center.y + 0.5f) * static_cast<float>(window_size);
473 localization_bbox.push_back(minRect);
474 bbox_scores.push_back(edge_num);
475
476 }
477 }

492// Change mat
493void Detect::barcodeErode()
494{
495 static const std::array<Mat, 4> &structuringElement = getStructuringElement();
496 Mat m0, m1, m2, m3;
497 dilate(coherence, m0, structuringElement[0]);
498 dilate(coherence, m1, structuringElement[1]);
499 dilate(coherence, m2, structuringElement[2]);
500 dilate(coherence, m3, structuringElement[3]);
501 int sum;
502 for (int y = 0; y < coherence.rows; y++)
503 {
504 auto coherence_row = coherence.ptr<uint8_t>(y);
505 auto m0_row = m0.ptr<uint8_t>(y);
506 auto m1_row = m1.ptr<uint8_t>(y);
507 auto m2_row = m2.ptr<uint8_t>(y);
508 auto m3_row = m3.ptr<uint8_t>(y);
509
510 for (int pos = 0; pos < coherence.cols; pos++)
511 {
512 if (coherence_row[pos] != 0)
513 {
514 sum = m0_row[pos] + m1_row[pos] + m2_row[pos] + m3_row[pos];
515 //more than 2 group
516 coherence_row[pos] = sum > 600 ? 255 : 0;
517 }
518 }
519 }
520}
