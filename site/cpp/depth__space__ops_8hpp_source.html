20 template <class T>
21 class DepthSpaceOps final : public CUDABackendNode {
22 public:
23 using wrapper_type = GetCUDABackendWrapperType<T>;
24
25 DepthSpaceOps(csl::Stream stream_, const MatShape& internal_shape_,
26 const std::vector<size_t> &permutation_)
27 : stream(std::move(stream_)), internal_shape(internal_shape_),
28 permutation(permutation_)
29 {
30 int dims = internal_shape.dims;
31 int nperm = (int)permutation_.size();
32 transposed_internal_shape = MatShape(dims);
33 for (int i = 0; i < nperm; i++) {
34 transposed_internal_shape[i] = internal_shape[(int)permutation[i]];
35 }
36
37 size_t num_elements = internal_shape.total();
38 csl::WorkspaceBuilder builder;
39 builder.require<T>(num_elements);
40 scratch_mem_in_bytes = builder.required_workspace_size();
41 }
42
43 void forward(const std::vector<cv::Ptr<BackendWrapper>> &inputs,
44 const std::vector<cv::Ptr<BackendWrapper>> &outputs,
45 csl::Workspace &workspace) override {
46 CV_CheckEQ(inputs.size(), size_t(1), "DepthSpaceOps: only one input is accepted");
47 CV_CheckEQ(outputs.size(), size_t(1), "DepthSpaceOps: only one output is accepted");
48
49 auto input_wrapper = inputs.front().dynamicCast<wrapper_type>();
50 auto input = input_wrapper->getView();
51 CV_CheckEQ(input.rank(), size_t(4), "DepthSpaceOps: input needs to be 4-dimensional [N, C, H, W]");
52 auto output_wrapper = outputs.front().dynamicCast<wrapper_type>();
53 auto output = output_wrapper->getSpan();
54 auto ws_allocator = csl::WorkspaceAllocator(workspace);
55 auto transposed_internal = ws_allocator.get_tensor_span<T>(transposed_internal_shape.begin(), transposed_internal_shape.end());
56
57 // Call reshape on input so that it has the correct shape for permutation
58 input.reshape(internal_shape.begin(), internal_shape.end());
59 kernels::permute(stream, transposed_internal, input, permutation);
60 // Only copying is needed as output already has the expected shape
61 auto t = csl::TensorView<T>(transposed_internal);
62 csl::memcpy(output.get(), t.get(), output.size(), stream);
63 }
64
65 std::size_t get_workspace_memory_in_bytes() const noexcept override { return scratch_mem_in_bytes; }
66
67 private:
68 csl::Stream stream;
69 MatShape internal_shape;
70 std::vector<size_t> permutation;
71 MatShape transposed_internal_shape;
72
73 std::size_t scratch_mem_in_bytes;
74 };
