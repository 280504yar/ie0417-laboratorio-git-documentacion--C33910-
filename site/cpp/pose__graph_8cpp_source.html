24// matrix form of quaternion multiplication from left side
25static inline cv::Matx44d m_left(cv::Quatd q)
26{
27 // M_left(a)* V(b) =
28 // = (I_4 * a0 + [ 0 | -av [ 0 | 0_1x3
29 // av | 0_3] + 0_3x1 | skew(av)]) * V(b)
30
31 double w = q.w, x = q.x, y = q.y, z = q.z;
32 return { w, -x, -y, -z,
33 x, w, -z, y,
34 y, z, w, -x,
35 z, -y, x, w };
36}
37
38// matrix form of quaternion multiplication from right side
39static inline cv::Matx44d m_right(cv::Quatd q)
40{
41 // M_right(b)* V(a) =
42 // = (I_4 * b0 + [ 0 | -bv [ 0 | 0_1x3
43 // bv | 0_3] + 0_3x1 | skew(-bv)]) * V(a)
44
45 double w = q.w, x = q.x, y = q.y, z = q.z;
46 return { w, -x, -y, -z,
47 x, w, z, -y,
48 y, -z, w, x,
49 z, y, -x, w };
50}
51
52// jacobian of quaternionic (exp(x)*q) : R_3 -> H near x == 0
53static inline cv::Matx43d expQuatJacobian(cv::Quatd q)
54{
55 double w = q.w, x = q.x, y = q.y, z = q.z;
56 return cv::Matx43d(-x, -y, -z,
57 w, z, -y,
58 -z, w, x,
59 y, -x, w);
60}

62// concatenate matrices vertically
63template<typename _Tp, int m, int n, int k> static inline
64cv::Matx<_Tp, m + k, n> concatVert(const cv::Matx<_Tp, m, n>& a, const cv::Matx<_Tp, k, n>& b)
65{
66 cv::Matx<_Tp, m + k, n> res;
67 for (int i = 0; i < m; i++)
68 {
69 for (int j = 0; j < n; j++)
70 {
71 res(i, j) = a(i, j);
72 }
73 }
74 for (int i = 0; i < k; i++)
75 {
76 for (int j = 0; j < n; j++)
77 {
78 res(m + i, j) = b(i, j);
79 }
80 }
81 return res;
82}

84// concatenate matrices horizontally
85template<typename _Tp, int m, int n, int k> static inline
86cv::Matx<_Tp, m, n + k> concatHor(const cv::Matx<_Tp, m, n>& a, const cv::Matx<_Tp, m, k>& b)
87{
88 cv::Matx<_Tp, m, n + k> res;
89
90 for (int i = 0; i < m; i++)
91 {
92 for (int j = 0; j < n; j++)
93 {
94 res(i, j) = a(i, j);
95 }
96 }
97 for (int i = 0; i < m; i++)
98 {
99 for (int j = 0; j < k; j++)
100 {
101 res(i, n + j) = b(i, j);
102 }
103 }
104 return res;

122 struct Pose3d
123 {
124 Vec3d t;
125 Quatd q;
126
127 Pose3d() : t(), q(1, 0, 0, 0) { }
128
129 Pose3d(const Matx33d& rotation, const Vec3d& translation)
130 : t(translation), q(Quatd::createFromRotMat(rotation).normalize())
131 { }
132
133 explicit Pose3d(const Matx44d& pose) :
134 Pose3d(pose.get_minor<3, 3>(0, 0), Vec3d(pose(0, 3), pose(1, 3), pose(2, 3)))
135 { }
136
137 inline Pose3d operator*(const Pose3d& otherPose) const
138 {
139 Pose3d out(*this);
140 out.t += q.toRotMat3x3(QUAT_ASSUME_UNIT) * otherPose.t;
141 out.q = out.q * otherPose.q;
142 return out;
143 }
144
145 Affine3d getAffine() const
146 {
147 return Affine3d(q.toRotMat3x3(QUAT_ASSUME_UNIT), t);
148 }
149
150 inline Pose3d inverse() const
151 {
152 Pose3d out;
153 out.q = q.conjugate();
154 out.t = -(out.q.toRotMat3x3(QUAT_ASSUME_UNIT) * t);
155 return out;
156 }
157
158 inline void normalizeRotation()
159 {
160 q = q.normalize();
161 }
162
163 // jacobian of exponential (exp(x)* q) : R_6->SE(3) near x == 0
164 inline Matx<double, 7, 6> expJacobian()
165 {
166 Matx43d qj = expQuatJacobian(q);
167 // x node layout is (rot_x, rot_y, rot_z, trans_x, trans_y, trans_z)
168 // pose layout is (q_w, q_x, q_y, q_z, trans_x, trans_y, trans_z)
169 return concatVert(concatHor(qj, Matx43d()),
170 concatHor(Matx33d(), Matx33d::eye()));
171 }

173 inline Pose3d oplus(const Vec6d dx)
174 {
175 Vec3d deltaRot(dx[0], dx[1], dx[2]), deltaTrans(dx[3], dx[4], dx[5]);
176 Pose3d p;
177 p.q = Quatd(0, deltaRot[0], deltaRot[1], deltaRot[2]).exp() * this->q;
178 p.t = this->t + deltaTrans;
179 return p;
180 }

391// Cholesky decomposition of symmetrical 6x6 matrix
392static inline cv::Matx66d llt6(Matx66d m)
393{
394 Matx66d L;
395 for (int i = 0; i < 6; i++)
396 {
397 for (int j = 0; j < (i + 1); j++)
398 {
399 double sum = 0;
400 for (int k = 0; k < j; k++)
401 sum += L(i, k) * L(j, k);
402
403 if (i == j)
404 L(i, i) = sqrt(m(i, i) - sum);
405 else
406 L(i, j) = (1.0 / L(j, j) * (m(i, j) - sum));
407 }
408 }
409 return L;
410}

421bool PoseGraphImpl::isValid() const
422{
423 size_t numNodes = getNumNodes();
424 size_t numEdges = getNumEdges();
425
426 if (!numNodes || !numEdges)
427 return false;
428
429 std::unordered_set<size_t> nodesVisited;
430 std::vector<size_t> nodesToVisit;
431
432 nodesToVisit.push_back(nodes.begin()->first);
433
434 bool isGraphConnected = false;
435 while (!nodesToVisit.empty())
436 {
437 size_t currNodeId = nodesToVisit.back();
438 nodesToVisit.pop_back();
439 nodesVisited.insert(currNodeId);
440 // Since each node does not maintain its neighbor list
441 for (size_t i = 0; i < numEdges; i++)
442 {
443 const Edge& potentialEdge = edges.at(i);
444 size_t nextNodeId = (size_t)(-1);
445
446 if (potentialEdge.sourceNodeId == currNodeId)
447 {
448 nextNodeId = potentialEdge.targetNodeId;
449 }
450 else if (potentialEdge.targetNodeId == currNodeId)
451 {
452 nextNodeId = potentialEdge.sourceNodeId;
453 }
454 if (nextNodeId != (size_t)(-1))
455 {
456 if (nodesVisited.count(nextNodeId) == 0)
457 {
458 nodesToVisit.push_back(nextNodeId);
459 }
460 }
461 }
462 }
463
464 isGraphConnected = (nodesVisited.size() == numNodes);
465
466 CV_LOG_INFO(NULL, "nodesVisited: " << nodesVisited.size() << " IsGraphConnected: " << isGraphConnected);
467
468 bool invalidEdgeNode = false;
469 for (size_t i = 0; i < numEdges; i++)
470 {
471 const Edge& edge = edges.at(i);
472 // edges have spurious source/target nodes
473 if ((nodesVisited.count(edge.sourceNodeId) != 1) ||
474 (nodesVisited.count(edge.targetNodeId) != 1))
475 {
476 invalidEdgeNode = true;
477 break;
478 }
479 }
480 return isGraphConnected && !invalidEdgeNode;
481}

483double PoseGraphImpl::calculateWeight(const PoseGraphImpl::Edge& e, double lambda) const
484{
485 double translationNorm = cv::norm(e.pose.t);
486
487 cv::Matx33d R = e.pose.q.toRotMat3x3(cv::QUAT_ASSUME_UNIT);
488 cv::Vec3d rvec;
489 cv::Rodrigues(R, rvec);
490 double rotationAngle = cv::norm(rvec);
491
492 double weight = translationNorm + lambda * rotationAngle;
493
494 return weight;
495}

497void PoseGraphImpl::applyMST(const std::vector<cv::MSTEdge>& resultingEdges, const PoseGraphImpl::Node& rootNode)
498{
499 std::unordered_map<size_t, std::vector<std::pair<size_t, PoseGraphImpl::Pose3d>>> adj;
500 for (const auto& e: resultingEdges)
501 {
502 auto it = std::find_if(edges.begin(), edges.end(), [&](const PoseGraphImpl::Edge& edge)
503 {
504 return (edge.sourceNodeId == static_cast<size_t>(e.source) && edge.targetNodeId == static_cast<size_t>(e.target)) ||
505 (edge.sourceNodeId == static_cast<size_t>(e.target) && edge.targetNodeId == static_cast<size_t>(e.source));
506 });
507 if (it != edges.end())
508 {
509 size_t src = it->sourceNodeId;
510 size_t tgt = it->targetNodeId;
511 const PoseGraphImpl::Pose3d& relPose = it->pose;
512
513 adj[src].emplace_back(tgt, relPose);
514 adj[tgt].emplace_back(src, relPose.inverse());
515 }
516 }
517
518 std::unordered_map<size_t, PoseGraphImpl::Pose3d> newPoses;
519 std::stack<size_t> toVisit;
520 std::unordered_set<size_t> visited;
521
522 newPoses[rootNode.id] = rootNode.pose;
523 toVisit.push(rootNode.id);
524
525 while (!toVisit.empty())
526 {
527 size_t current = toVisit.top();
528 toVisit.pop();
529 visited.insert(current);
530
531 const auto& currentPose = newPoses[current];
532
533 auto it = adj.find(current);
534 if (it == adj.end())
535 continue;
536
537 for (const auto& pr : it->second)
538 {
539 auto neighbor = pr.first;
540 auto relativePose = pr.second;
541
542 if (visited.count(neighbor))
543 continue;
544 newPoses[neighbor] = currentPose * relativePose;
545 toVisit.push(neighbor);
546 }
547 }
548
549 // Apply the new poses
550 for (const auto& np : newPoses)
551 {
552 auto nodeId = np.first;
553 auto pose = np.second;
554
555 if (!nodes.at(nodeId).isFixed)
556 nodes.at(nodeId).setPose(pose.getAffine());
557 }
558}

560void PoseGraphImpl::initializePosesWithMST(double lambda)
561{
562 size_t numNodes = getNumNodes();
563
564 std::vector<MSTEdge> MSTedges;
565 for (const auto& e: edges)
566 {
567 double weight = calculateWeight(e, lambda);
568 MSTedges.push_back({static_cast<int>(e.sourceNodeId), static_cast<int>(e.targetNodeId), weight});
569 }
570
571 size_t rootId = 0;
572 PoseGraphImpl::Node rootNode = nodes.begin()->second;
573
574 for (const auto& n: nodes)
575 {
576 if (isNodeFixed(n.second.id))
577 {
578 rootNode = n.second;
579 rootId = n.second.id;
580 break;
581 }
582 }
583
584 std::vector<MSTEdge> resultingEdges;
585 if (!cv::buildMST(numNodes, MSTedges, resultingEdges, MST_PRIM, static_cast<int>(rootId)))
586 {
587 CV_LOG_INFO(NULL, "Failed to build MST: graph may be disconnected.");
588 return;
589 }
590
591 applyMST(resultingEdges, rootNode);
592}

595// Optimization itself //

598static inline double poseError(Quatd sourceQuat, Vec3d sourceTrans, Quatd targetQuat, Vec3d targetTrans,
599 Quatd rotMeasured, Vec3d transMeasured, Matx66d sqrtInfoMatrix, bool needJacobians,
600 Matx<double, 6, 4>& sqj, Matx<double, 6, 3>& stj,
601 Matx<double, 6, 4>& tqj, Matx<double, 6, 3>& ttj,
602 Vec6d& res)
603{
604 // err_r = 2*Im(conj(rel_r) * measure_r) = 2*Im(conj(target_r) * source_r * measure_r)
605 // err_t = conj(source_r) * (target_t - source_t) * source_r - measure_t
606
607 Quatd sourceQuatInv = sourceQuat.conjugate();
608 Vec3d deltaTrans = targetTrans - sourceTrans;
609
610 Quatd relativeQuat = sourceQuatInv * targetQuat;
611 Vec3d relativeTrans = sourceQuatInv.toRotMat3x3(cv::QUAT_ASSUME_UNIT) * deltaTrans;
612
614 Quatd deltaRot = relativeQuat.conjugate() * rotMeasured;
615
616 Vec3d terr = relativeTrans - transMeasured;
617 Vec3d rerr = 2.0 * Vec3d(deltaRot.x, deltaRot.y, deltaRot.z);
618 Vec6d rterr(terr[0], terr[1], terr[2], rerr[0], rerr[1], rerr[2]);
619
620 res = sqrtInfoMatrix * rterr;
621
622 if (needJacobians)
623 {
624 // d(err_r) = 2*Im(d(conj(target_r) * source_r * measure_r)) = < measure_r is constant > =
625 // 2*Im((conj(d(target_r)) * source_r + conj(target_r) * d(source_r)) * measure_r)
626 // d(target_r) == 0:
627 // # d(err_r) = 2*Im(conj(target_r) * d(source_r) * measure_r)
628 // # V(d(err_r)) = 2 * M_Im * M_right(measure_r) * M_left(conj(target_r)) * V(d(source_r))
629 // # d(err_r) / d(source_r) = 2 * M_Im * M_right(measure_r) * M_left(conj(target_r))
630 Matx34d drdsq = 2.0 * (m_right(rotMeasured) * m_left(targetQuat.conjugate())).get_minor<3, 4>(1, 0);
631
632 // d(source_r) == 0:
633 // # d(err_r) = 2*Im(conj(d(target_r)) * source_r * measure_r)
634 // # V(d(err_r)) = 2 * M_Im * M_right(source_r * measure_r) * M_Conj * V(d(target_r))
635 // # d(err_r) / d(target_r) = 2 * M_Im * M_right(source_r * measure_r) * M_Conj
636 Matx34d drdtq = 2.0 * (m_right(sourceQuat * rotMeasured) * M_Conj).get_minor<3, 4>(1, 0);
637
638 // d(err_t) = d(conj(source_r) * (target_t - source_t) * source_r) =
639 // conj(source_r) * (d(target_t) - d(source_t)) * source_r +
640 // conj(d(source_r)) * (target_t - source_t) * source_r +
641 // conj(source_r) * (target_t - source_t) * d(source_r) =
642 // <conj(a*b) == conj(b)*conj(a), conj(target_t - source_t) = - (target_t - source_t), 2 * Im(x) = (x - conj(x))>
643 // conj(source_r) * (d(target_t) - d(source_t)) * source_r +
644 // 2 * Im(conj(source_r) * (target_t - source_t) * d(source_r))
645 // d(*_t) == 0:
646 // # d(err_t) = 2 * Im(conj(source_r) * (target_t - source_t) * d(source_r))
647 // # V(d(err_t)) = 2 * M_Im * M_left(conj(source_r) * (target_t - source_t)) * V(d(source_r))
648 // # d(err_t) / d(source_r) = 2 * M_Im * M_left(conj(source_r) * (target_t - source_t))
649 Matx34d dtdsq = 2 * m_left(sourceQuatInv * Quatd(0, deltaTrans[0], deltaTrans[1], deltaTrans[2])).get_minor<3, 4>(1, 0);
650 // deltaTrans is rotated by sourceQuatInv, so the jacobian is rot matrix of sourceQuatInv by +1 or -1
651 Matx33d dtdtt = sourceQuatInv.toRotMat3x3(QUAT_ASSUME_UNIT);
652 Matx33d dtdst = -dtdtt;
653
654 Matx33d z;
655 sqj = concatVert(dtdsq, drdsq);
656 tqj = concatVert(Matx34d(), drdtq);
657 stj = concatVert(dtdst, z);
658 ttj = concatVert(dtdtt, z);
659
660 stj = sqrtInfoMatrix * stj;
661 ttj = sqrtInfoMatrix * ttj;
662 sqj = sqrtInfoMatrix * sqj;
663 tqj = sqrtInfoMatrix * tqj;
664 }
665
666 return res.ddot(res);
667}

676// estimate current energy
677double PoseGraphImpl::calcEnergyNodes(const std::map<size_t, Node>& newNodes) const
678{
679 double totalErr = 0;
680 for (const auto& e : edges)
681 {
682 Pose3d srcP = newNodes.at(e.sourceNodeId).pose;
683 Pose3d tgtP = newNodes.at(e.targetNodeId).pose;
684
685 Vec6d res;
686 Matx<double, 6, 3> stj, ttj;
687 Matx<double, 6, 4> sqj, tqj;
688 double err = poseError(srcP.q, srcP.t, tgtP.q, tgtP.t, e.pose.q, e.pose.t, e.sqrtInfo,
689 /* needJacobians = */ false, sqj, stj, tqj, ttj, res);
690
691 totalErr += err;
692 }
693 return totalErr * 0.5;
694}

696
697// J := J * d_inv, d_inv = make_diag(di)
698// J^T*J := (J * d_inv)^T * J * d_inv = diag(di) * (J^T * J) * diag(di) = eltwise_mul(J^T*J, di*di^T)
699// J^T*b := (J * d_inv)^T * b = d_inv^T * J^T*b = eltwise_mul(J^T*b, di)
700static void doJacobiScalingSparse(BlockSparseMat<double, 6, 6>& jtj, Mat_<double>& jtb, const Mat_<double>& di)
701{
702 // scaling J^T*J
703 for (auto& ijv : jtj.ijValue)
704 {
705 Point2i bpt = ijv.first;
706 Matx66d& m = ijv.second;
707 for (int i = 0; i < 6; i++)
708 {
709 for (int j = 0; j < 6; j++)
710 {
711 Point2i pt(bpt.x * 6 + i, bpt.y * 6 + j);
712 m(i, j) *= di(pt.x) * di(pt.y);
713 }
714 }
715 }
716
717 // scaling J^T*b
718 jtb = jtb.mul(di);
719}

777 virtual bool calcFunc(double& energy, bool calcEnergy = true, bool calcJacobian = false) CV_OVERRIDE
778 {
779 std::map<size_t, PoseGraphImpl::Node>& nodes = tempNodes;
780
781 std::vector<cv::Matx<double, 7, 6>> cachedJac;
782 if (calcJacobian)
783 {
784 jtj.clear();
785 std::fill(jtb.begin(), jtb.end(), 0.0);
786
787 // caching nodes jacobians
788 for (auto id : placesIds)
789 {
790 cachedJac.push_back(nodes.at(id).pose.expJacobian());
791 }
792
793 if (useGeo)
794 jtCached.clear();
795 }
796
797 double totalErr = 0.0;
798 for (const auto& e : pg->edges)
799 {
800 size_t srcId = e.sourceNodeId, dstId = e.targetNodeId;
801 const PoseGraphImpl::Node& srcNode = nodes.at(srcId);
802 const PoseGraphImpl::Node& dstNode = nodes.at(dstId);
803
804 const PoseGraphImpl::Pose3d& srcP = srcNode.pose;
805 const PoseGraphImpl::Pose3d& tgtP = dstNode.pose;
806 bool srcFixed = srcNode.isFixed;
807 bool dstFixed = dstNode.isFixed;
808
809 Vec6d res;
810 Matx<double, 6, 3> stj, ttj;
811 Matx<double, 6, 4> sqj, tqj;
812
813 double err = poseError(srcP.q, srcP.t, tgtP.q, tgtP.t, e.pose.q, e.pose.t, e.sqrtInfo,
814 /* needJacobians = */ calcJacobian, sqj, stj, tqj, ttj, res);
815 totalErr += err;
816
817 if (calcJacobian)
818 {
819 size_t srcPlace = (size_t)(-1), dstPlace = (size_t)(-1);
820 Matx66d sj, tj;
821 if (!srcFixed)
822 {
823 srcPlace = idToPlace.at(srcId);
824 sj = concatHor(sqj, stj) * cachedJac[srcPlace];
825
826 jtj.refBlock(srcPlace, srcPlace) += sj.t() * sj;
827
828 Vec6d jtbSrc = sj.t() * res;
829 for (int i = 0; i < 6; i++)
830 {
831 jtb(6 * (int)srcPlace + i) += jtbSrc[i];
832 }
833 }
834
835 if (!dstFixed)
836 {
837 dstPlace = idToPlace.at(dstId);
838 tj = concatHor(tqj, ttj) * cachedJac[dstPlace];
839
840 jtj.refBlock(dstPlace, dstPlace) += tj.t() * tj;
841
842 Vec6d jtbDst = tj.t() * res;
843 for (int i = 0; i < 6; i++)
844 {
845 jtb(6 * (int)dstPlace + i) += jtbDst[i];
846 }
847 }
848
849 if (!(srcFixed || dstFixed))
850 {
851 Matx66d sjttj = sj.t() * tj;
852 jtj.refBlock(srcPlace, dstPlace) += sjttj;
853 jtj.refBlock(dstPlace, srcPlace) += sjttj.t();
854 }
855
856 if (useGeo)
857 {
858 jtCached.push_back({ sj, tj });
859 }
860 }
861 }
862
863 if (calcEnergy)
864 {
865 energy = totalErr * 0.5;
866 }
867
868 return true;
869 }
