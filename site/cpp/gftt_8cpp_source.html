112 void detect( InputArray _image, std::vector<KeyPoint>& keypoints, InputArray _mask ) CV_OVERRIDE
113 {
114 CV_INSTRUMENT_REGION();
115
116 if(_image.empty())
117 {
118 keypoints.clear();
119 return;
120 }
121
122 std::vector<Point2f> corners;
123 std::vector<float> cornersQuality;
124
125 if (_image.isUMat())
126 {
127 UMat ugrayImage;
128 if( _image.type() != CV_8U )
129 cvtColor( _image, ugrayImage, COLOR_BGR2GRAY );
130 else
131 ugrayImage = _image.getUMat();
132
133 goodFeaturesToTrack( ugrayImage, corners, nfeatures, qualityLevel, minDistance, _mask,
134 cornersQuality, blockSize, gradSize, useHarrisDetector, k );
135 }
136 else
137 {
138 Mat image = _image.getMat(), grayImage = image;
139 if( image.type() != CV_8U )
140 cvtColor( image, grayImage, COLOR_BGR2GRAY );
141
142 goodFeaturesToTrack( grayImage, corners, nfeatures, qualityLevel, minDistance, _mask,
143 cornersQuality, blockSize, gradSize, useHarrisDetector, k );
144 }
145
146 CV_Assert(corners.size() == cornersQuality.size());
147
148 keypoints.resize(corners.size());
149 for (size_t i = 0; i < corners.size(); i++)
150 keypoints[i] = KeyPoint(corners[i], (float)blockSize, -1, cornersQuality[i]);
151
152 }
153
154 int nfeatures;
155 double qualityLevel;
156 double minDistance;
157 int blockSize;
158 int gradSize;
159 bool useHarrisDetector;
160 double k;
161};
