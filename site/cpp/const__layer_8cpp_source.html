106 virtual Ptr<BackendNode> initCann(const std::vector<Ptr<BackendWrapper> > &inputs,
107 const std::vector<Ptr<BackendWrapper> > &outputs,
108 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
109 {
110 auto mat_shape = shape(blobs[0]);
111 std::vector<int64_t> mat_shape_{mat_shape.begin(), mat_shape.end()};
112
113 auto ge_shape = ge::Shape(mat_shape_);
114 auto ge_dtype = ge::DT_FLOAT;
115 switch (blobs[0].type())
116 {
117 case CV_32F: break;
118 case CV_32S: ge_dtype = ge::DT_INT32; break;
119 default: CV_Error(Error::StsNotImplemented, "Unsupported data type");
120 }
121 auto size_of_type = sizeof(float);
122 switch (blobs[0].type())
123 {
124 case CV_32F: break;
125 case CV_32S: size_of_type = sizeof(int); break;
126 default: CV_Error(Error::StsNotImplemented, "Unsupported data type");
127 }
128
129 auto desc = std::make_shared<ge::TensorDesc>(ge_shape, ge::FORMAT_NCHW, ge_dtype);
130 auto ge_tensor = std::make_shared<ge::Tensor>();
131 ge_tensor->SetTensorDesc(*desc);
132 ge_tensor->SetData(blobs[0].data, ge_shape.GetShapeSize() * size_of_type);
133
134 auto op = std::make_shared<ge::op::Const>(name);
135 op->set_attr_value(*ge_tensor);
136
137 return Ptr<BackendNode>(new CannBackendNode(op));
138 }
