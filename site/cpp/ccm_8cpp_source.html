114 {
115 Mat ccm_(ccmLoss->shape, 1, CV_64F);
116 for (int i = 0; i < ccmLoss->shape; i++)

118 ccm_.at<double>(i, 0) = x[i];

143Mat ColorCorrectionModel::Impl::prepare(const Mat& inp)
144{
145 switch (ccmType)
146 {
147 case cv::ccm::CCM_LINEAR:
148 shape = 9;
149 return inp;
150 case cv::ccm::CCM_AFFINE:
151 {
152 shape = 12;
153 Mat ones(inp.size(), CV_64F, Scalar(1));
154 Mat out(inp.size(), CV_64FC4);
155 const Mat srcs[] = { inp, ones };
156 const int fromTo[] = { 0,0, 1,1, 2,2, 3,3 }; // inp[ch] → out[ch]
157 mixChannels(srcs, 2, &out, 1, fromTo, 4);
158 return out;
159 }
160 default:
161 CV_Error(Error::StsBadArg, "Wrong ccmType!");
162 break;
163 }
164}

166void ColorCorrectionModel::Impl::calWeightsMasks(const Mat& weightsList_, double weightsCoeff_, Mat saturateMask)
167{
168 // weights
169 if (!weightsList_.empty())
170 {
171 weights = weightsList_;
172 }
173 else if (weightsCoeff_ != 0)
174 {
175 pow(ref.toLuminant(cs.illumobserver), weightsCoeff_, weights);
176 }
177
178 // masks
179 Mat weight_mask = Mat::ones(src.rows, 1, CV_8U);
180 if (!weights.empty())
181 {
182 weight_mask = weights > 0;
183 }
184 this->mask = (weight_mask) & (saturateMask);
185
186 // weights' mask
187 if (!weights.empty())
188 {
189 Mat weights_masked = maskCopyTo(this->weights, this->mask);
190 weights = weights_masked / mean(weights_masked)[0];
191 }
192 maskedLen = (int)sum(mask)[0];
193}

195void ColorCorrectionModel::Impl::initialWhiteBalance()
196{
197 // sum over all pixels – Scalar holds per-channel sums
198 const cv::Scalar srcSum = cv::sum(srcRgbl);
199 const cv::Scalar dstSum = cv::sum(dstRgbl);
200
201 // channel-wise gain factors
202 const double gR = dstSum[0] / srcSum[0];
203 const double gG = dstSum[1] / srcSum[1];
204 const double gB = dstSum[2] / srcSum[2];
205
206 // shape == 9 for a 3×3 linear CCM, or 12 for a 3×4 affine CCM
207 if (shape == 9) {
208 // 3×3 diagonal matrix
209 ccm0 = cv::Mat::zeros(3, 3, CV_64F);
210 ccm0.at<double>(0, 0) = gR;
211 ccm0.at<double>(1, 1) = gG;
212 ccm0.at<double>(2, 2) = gB;
213 }
214 else {
215 // 3×4 affine matrix (last column = zeros)
216 ccm0 = cv::Mat::zeros(3, 4, CV_64F);
217 ccm0.at<double>(0, 0) = gR;
218 ccm0.at<double>(1, 1) = gG;
219 ccm0.at<double>(2, 2) = gB;
220 }
221}

224void ColorCorrectionModel::Impl::initialLeastSquare(bool fit)
225{
226 Mat A, B, w;
227 if (weights.empty())
228 {
229 A = srcRgbl;
230 B = dstRgbl;
231 }
232 else
233 {
234 pow(weights, 0.5, w);
235 Mat w_;
236 merge(std::vector<Mat> { w, w, w }, w_);
237 A = w_.mul(srcRgbl);
238 B = w_.mul(dstRgbl);
239 }
240 solve(A.reshape(1, A.rows), B.reshape(1, B.rows), ccm0, DECOMP_SVD);
241
242 // if fit is True, return optimalization for rgbl distance function.
243 if (fit)
244 {
245 ccm = ccm0;
246 Mat residual = A.reshape(1, A.rows) * ccm.reshape(0, shape / 3) - B.reshape(1, B.rows);
247 Scalar s = residual.dot(residual);
248 double sum = s[0];
249 loss = sqrt(sum / maskedLen);
250 }
251}

253double ColorCorrectionModel::Impl::calcLoss_(Color color)
254{
255 Mat distlist = color.diff(ref, distance);
256 Color lab = color.to(COLOR_SPACE_LAB_D50_2);
257 Mat dist_;
258 pow(distlist, 2, dist_);
259 if (!weights.empty())
260 {
261 dist_ = weights.mul(dist_);
262 }
263 Scalar ss = sum(dist_);
264 return ss[0];
265}

274void ColorCorrectionModel::Impl::fitting(void)
275{
276 cv::Ptr<DownhillSolver> solver = cv::DownhillSolver::create();
277 cv::Ptr<LossFunction> ptr_F(new LossFunction(this));
278 solver->setFunction(ptr_F);
279 Mat reshapeCcm = ccm0.clone().reshape(0, 1);
280 Mat step = Mat::ones(reshapeCcm.size(), CV_64F);
281 solver->setInitStep(step);
282 TermCriteria termcrit = TermCriteria(TermCriteria::MAX_ITER + TermCriteria::EPS, maxCount, epsilon);
283 solver->setTermCriteria(termcrit);
284 double res = solver->minimize(reshapeCcm);
285 ccm = reshapeCcm.reshape(0, shape / 3);
286 loss = sqrt(res / maskedLen);
287}

293void ColorCorrectionModel::correctImage(InputArray src, OutputArray ref, bool islinear)
294{
295 if (!p->ccm.data)
296 {
297 CV_Error(Error::StsBadArg, "No CCM values!" );
298 }
299 Mat img, normImg;
300 if (p->rgb){
301 cvtColor(src.getMat(), img, COLOR_BGR2RGB);
302 } else {
303 img = src.getMat();
304 }
305
306 double scale;
307 int type = img.type();
308 switch (type) {
309 case CV_8UC3:
310 scale = 1.0 / 255.0;
311 break;
312 case CV_16UC3:
313 scale = 1.0 / 65535.0;
314 break;
315 case CV_32FC3:
316 scale = 1.0; // Already in [0,1] range
317 break;
318 default:
319 CV_Error( cv::Error::StsUnsupportedFormat, "8-bit, 16-bit unsigned or 32-bit float 3-channel input images are supported");
320 }
321
322 img.convertTo(normImg, CV_64F, scale);
323 Mat linearImg = (p->linear)->linearize(normImg);
324 Mat ccm = p->ccm.reshape(0, p->shape / 3);
325 Mat imgCcm = multiple(p->prepare(linearImg), ccm);
326 if (islinear == true)
327 {
328 imgCcm.copyTo(ref);
329 }
330 Mat imgCorrected = p->cs.fromLFunc(imgCcm, linearImg);
331
332 imgCorrected *= 1.0/scale;
333 imgCorrected.convertTo(imgCorrected, type);
334
335 if (p->rgb)
336 cvtColor(imgCorrected, imgCorrected, COLOR_RGB2BGR);
337 imgCorrected.copyTo(ref);
338}

425Mat ColorCorrectionModel::compute()
426{
427
428 Mat saturateMask = saturate(p->src, p->saturatedThreshold[0], p->saturatedThreshold[1]);
429 p->linear = getLinear(p->gamma, p->deg, p->src, p->ref, saturateMask, (p->cs), p->linearizationType);
430 p->calWeightsMasks(p->weightsList, p->weightsCoeff, saturateMask);
431 p->srcRgbl = p->linear->linearize(maskCopyTo(p->src, p->mask));
432 p->ref.colors = maskCopyTo(p->ref.colors, p->mask);
433 p->dstRgbl = p->ref.to(*(p->cs.l)).colors;
434
435 // make no change for CCM_LINEAR, make change for CCM_AFFINE.
436 p->srcRgbl = p->prepare(p->srcRgbl);
437
438 // distance function may affect the loss function and the fitting function
439 switch (p->distance)
440 {
441 case cv::ccm::DISTANCE_RGBL:
442 p->initialLeastSquare(true);
443 break;
444 default:
445 switch (p->initialMethodType)
446 {
447 case cv::ccm::INITIAL_METHOD_WHITE_BALANCE:
448 p->initialWhiteBalance();
449 break;
450 case cv::ccm::INITIAL_METHOD_LEAST_SQUARE:
451 p->initialLeastSquare();
452 break;
453 default:
454 CV_Error(Error::StsBadArg, "Wrong initial_methoddistance_type!" );
455 break;
456 }
457 break;
458 }
459 p->fitting();
460
461 return p->ccm;
462}

501void ColorCorrectionModel::read(const FileNode& node)
502{
503 node["ccm"] >> p->ccm;
504 node["loss"] >> p->loss;
505 node["ccm_type"] >> p->ccmType;
506 node["shape"] >> p->shape;
507 node["distance"] >> p->distance;
508 node["gamma"] >> p->gamma;

515
516 node["linear_type"] >> p->linearizationType;
517 switch (p->linearizationType) {
518 case cv::ccm::LINEARIZATION_GAMMA:
519 p->linear = std::shared_ptr<Linear>(new LinearGamma());
520 break;
521 case cv::ccm::LINEARIZATION_COLORPOLYFIT:
522 p->linear = std::shared_ptr<Linear>(new LinearColor<Polyfit>());
523 break;
524 case cv::ccm::LINEARIZATION_IDENTITY:
525 p->linear = std::shared_ptr<Linear>(new LinearIdentity());
526 break;
527 case cv::ccm::LINEARIZATION_COLORLOGPOLYFIT:
528 p->linear = std::shared_ptr<Linear>(new LinearColor<LogPolyfit>());
529 break;
530 case cv::ccm::LINEARIZATION_GRAYPOLYFIT:
531 p->linear = std::shared_ptr<Linear>(new LinearGray<Polyfit>());
532 break;
533 case cv::ccm::LINEARIZATION_GRAYLOGPOLYFIT:
534 p->linear = std::shared_ptr<Linear>(new LinearGray<LogPolyfit>());
535 break;
536 default:
537 CV_Error(Error::StsBadArg, "Wrong linear_type!");
538 break;
539 }
540 node["linear"] >> *p->linear;
541}
