48template<typename _Tp> static int solveQuadratic(_Tp a, _Tp b, _Tp c, _Tp& x1, _Tp& x2)
49{
50 if( a == 0 )
51 {
52 if( b == 0 )
53 {
54 x1 = x2 = 0;
55 return c == 0;
56 }
57 x1 = x2 = -c/b;
58 return 1;
59 }
60
61 _Tp d = b*b - 4*a*c;
62 if( d < 0 )
63 {
64 x1 = x2 = 0;
65 return 0;
66 }
67 if( d > 0 )
68 {
69 d = std::sqrt(d);
70 double s = 1/(2*a);
71 x1 = (-b - d)*s;
72 x2 = (-b + d)*s;
73 if( x1 > x2 )
74 std::swap(x1, x2);
75 return 2;
76 }
77 x1 = x2 = -b/(2*a);
78 return 1;
79}

83static inline Point2f applyHomography( const Mat_<double>& H, const Point2f& pt )
84{
85 double z = H(2,0)*pt.x + H(2,1)*pt.y + H(2,2);
86 if( z )
87 {
88 double w = 1./z;
89 return Point2f( (float)((H(0,0)*pt.x + H(0,1)*pt.y + H(0,2))*w), (float)((H(1,0)*pt.x + H(1,1)*pt.y + H(1,2))*w) );
90 }
91 return Point2f( std::numeric_limits<float>::max(), std::numeric_limits<float>::max() );
92}

94static inline void linearizeHomographyAt( const Mat_<double>& H, const Point2f& pt, Mat_<double>& A )
95{
96 A.create(2,2);
97 double p1 = H(0,0)*pt.x + H(0,1)*pt.y + H(0,2),
98 p2 = H(1,0)*pt.x + H(1,1)*pt.y + H(1,2),
99 p3 = H(2,0)*pt.x + H(2,1)*pt.y + H(2,2),
100 p3_2 = p3*p3;
101 if( p3 )
102 {
103 A(0,0) = H(0,0)/p3 - p1*H(2,0)/p3_2; // fxdx
104 A(0,1) = H(0,1)/p3 - p1*H(2,1)/p3_2; // fxdy
105
106 A(1,0) = H(1,0)/p3 - p2*H(2,0)/p3_2; // fydx
107 A(1,1) = H(1,1)/p3 - p2*H(2,1)/p3_2; // fydx
108 }
109 else
110 A.setTo(Scalar::all(std::numeric_limits<double>::max()));
111}

139EllipticKeyPoint::EllipticKeyPoint( const Point2f& _center, const Scalar& _ellipse )
140{
141 center = _center;
142 ellipse = _ellipse;
143
144 double a = ellipse[0], b = ellipse[1], c = ellipse[2];
145 double ac_b2 = a*c - b*b;
146 double x1, x2;
147 solveQuadratic(1., -(a+c), ac_b2, x1, x2);
148 axes.width = (float)(1/sqrt(x1));
149 axes.height = (float)(1/sqrt(x2));
150
151 boundingBox.width = (float)sqrt(ellipse[2]/ac_b2);
152 boundingBox.height = (float)sqrt(ellipse[0]/ac_b2);
153}

157 Mat_<double> M(2, 2);
158 M(0,0) = _ellipse[0];

160 M(1,1) = _ellipse[2];

169void EllipticKeyPoint::calcProjection( const Mat_<double>& H, EllipticKeyPoint& projection ) const
170{
171 Point2f dstCenter = applyHomography(H, center);
172
173 Mat_<double> invM; invert(getSecondMomentsMatrix(), invM);
174 Mat_<double> Aff; linearizeHomographyAt(H, center, Aff);
175 Mat_<double> dstM; invert(Aff*invM*Aff.t(), dstM);
176
177 projection = EllipticKeyPoint( dstCenter, Scalar(dstM(0,0), dstM(0,1), dstM(1,1)) );
178}

245struct IntersectAreaCounter
246{
247 IntersectAreaCounter( float _dr, int _minx,
248 int _miny, int _maxy,
249 const Point2f& _diff,
250 const Scalar& _ellipse1, const Scalar& _ellipse2 ) :
251 dr(_dr), bua(0), bna(0), minx(_minx), miny(_miny), maxy(_maxy),
252 diff(_diff), ellipse1(_ellipse1), ellipse2(_ellipse2) {}
253 IntersectAreaCounter( const IntersectAreaCounter& counter, Split )
254 {
255 *this = counter;
256 bua = 0;
257 bna = 0;
258 }
259
260 void operator()( const BlockedRange& range )
261 {
262 CV_Assert( miny < maxy );
263 CV_Assert( dr > FLT_EPSILON );
264
265 int temp_bua = bua, temp_bna = bna;
266 for( int i = range.begin(); i != range.end(); i++ )
267 {
268 float rx1 = minx + i*dr;
269 float rx2 = rx1 - diff.x;
270 for( float ry1 = (float)miny; ry1 <= (float)maxy; ry1 += dr )
271 {
272 float ry2 = ry1 - diff.y;
273 //compute the distance from the ellipse center
274 float e1 = (float)(ellipse1[0]*rx1*rx1 + 2*ellipse1[1]*rx1*ry1 + ellipse1[2]*ry1*ry1);
275 float e2 = (float)(ellipse2[0]*rx2*rx2 + 2*ellipse2[1]*rx2*ry2 + ellipse2[2]*ry2*ry2);
276 //compute the area
277 if( e1<1 && e2<1 ) temp_bna++;
278 if( e1<1 || e2<1 ) temp_bua++;
279 }
280 }
281 bua = temp_bua;
282 bna = temp_bna;
283 }
284
285 void join( IntersectAreaCounter& ac )
286 {
287 bua += ac.bua;
288 bna += ac.bna;
289 }

291 float dr;
292 int bua, bna;
293
294 int minx;
295 int miny, maxy;
296
297 Point2f diff;
298 Scalar ellipse1, ellipse2;

323static void computeOneToOneMatchedOverlaps( const std::vector<EllipticKeyPoint>& keypoints1, const std::vector<EllipticKeyPoint>& keypoints2t,
324 bool commonPart, std::vector<SIdx>& overlaps, float minOverlap )
325{
326 CV_Assert( minOverlap >= 0.f );
327 overlaps.clear();
328 if( keypoints1.empty() || keypoints2t.empty() )
329 return;
330
331 overlaps.clear();
332 overlaps.reserve(cvRound(keypoints1.size() * keypoints2t.size() * 0.01));
333
334 for( size_t i1 = 0; i1 < keypoints1.size(); i1++ )
335 {
336 EllipticKeyPoint kp1 = keypoints1[i1];
337 float maxDist = sqrt(kp1.axes.width*kp1.axes.height),
338 fac = 30.f/maxDist;
339 if( !commonPart )
340 fac=3;
341
342 maxDist = maxDist*4;
343 fac = 1.f/(fac*fac);
344
345 EllipticKeyPoint keypoint1a = EllipticKeyPoint( kp1.center, Scalar(fac*kp1.ellipse[0], fac*kp1.ellipse[1], fac*kp1.ellipse[2]) );
346
347 for( size_t i2 = 0; i2 < keypoints2t.size(); i2++ )
348 {
349 EllipticKeyPoint kp2 = keypoints2t[i2];
350 Point2f diff = kp2.center - kp1.center;
351
352 if( norm(diff) < maxDist )
353 {
354 EllipticKeyPoint keypoint2a = EllipticKeyPoint( kp2.center, Scalar(fac*kp2.ellipse[0], fac*kp2.ellipse[1], fac*kp2.ellipse[2]) );
355 //find the largest eigenvalue
356 int maxx = (int)ceil(( keypoint1a.boundingBox.width > (diff.x+keypoint2a.boundingBox.width)) ?
357 keypoint1a.boundingBox.width : (diff.x+keypoint2a.boundingBox.width));
358 int minx = (int)floor((-keypoint1a.boundingBox.width < (diff.x-keypoint2a.boundingBox.width)) ?
359 -keypoint1a.boundingBox.width : (diff.x-keypoint2a.boundingBox.width));
360
361 int maxy = (int)ceil(( keypoint1a.boundingBox.height > (diff.y+keypoint2a.boundingBox.height)) ?
362 keypoint1a.boundingBox.height : (diff.y+keypoint2a.boundingBox.height));
363 int miny = (int)floor((-keypoint1a.boundingBox.height < (diff.y-keypoint2a.boundingBox.height)) ?
364 -keypoint1a.boundingBox.height : (diff.y-keypoint2a.boundingBox.height));
365 int mina = (maxx-minx) < (maxy-miny) ? (maxx-minx) : (maxy-miny) ;
366
367 //compute the area
368 float dr = (float)mina/50.f;
369 int N = (int)floor((float)(maxx - minx) / dr);
370 IntersectAreaCounter ac( dr, minx, miny, maxy, diff, keypoint1a.ellipse, keypoint2a.ellipse );
371 parallel_reduce( BlockedRange(0, N+1), ac );
372 if( ac.bna > 0 )
373 {
374 float ov = (float)ac.bna / (float)ac.bua;
375 if( ov >= minOverlap )
376 overlaps.push_back(SIdx(ov, (int)i1, (int)i2));
377 }
378 }
379 }
380 }
381
382 std::sort( overlaps.begin(), overlaps.end() );
383
384 typedef std::vector<SIdx>::iterator It;
385
386 It pos = overlaps.begin();
387 It end = overlaps.end();
388
389 while(pos != end)
390 {
391 It prev = pos++;
392 end = std::remove_if(pos, end, SIdx::UsedFinder(*prev));
393 }
394 overlaps.erase(pos, overlaps.end());
395}

397static void calculateRepeatability( const Mat& img1, const Mat& img2, const Mat& H1to2,
398 const std::vector<KeyPoint>& _keypoints1, const std::vector<KeyPoint>& _keypoints2,
399 float& repeatability, int& correspondencesCount,
400 Mat* thresholdedOverlapMask=0 )
401{
402 std::vector<EllipticKeyPoint> keypoints1, keypoints2, keypoints1t, keypoints2t;
403 EllipticKeyPoint::convert( _keypoints1, keypoints1 );
404 EllipticKeyPoint::convert( _keypoints2, keypoints2 );
405
406 // calculate projections of key points
407 EllipticKeyPoint::calcProjection( keypoints1, H1to2, keypoints1t );
408 Mat H2to1; invert(H1to2, H2to1);
409 EllipticKeyPoint::calcProjection( keypoints2, H2to1, keypoints2t );
410
411 float overlapThreshold;
412 bool ifEvaluateDetectors = thresholdedOverlapMask == 0;
413 if( ifEvaluateDetectors )
414 {
415 overlapThreshold = 1.f - 0.4f;
416
417 // remove key points from outside of the common image part
418 Size sz1 = img1.size(), sz2 = img2.size();
419 filterEllipticKeyPointsByImageSize( keypoints1, sz1 );
420 filterEllipticKeyPointsByImageSize( keypoints1t, sz2 );
421 filterEllipticKeyPointsByImageSize( keypoints2, sz2 );
422 filterEllipticKeyPointsByImageSize( keypoints2t, sz1 );
423 }
424 else
425 {
426 overlapThreshold = 1.f - 0.5f;
427
428 thresholdedOverlapMask->create( (int)keypoints1.size(), (int)keypoints2t.size(), CV_8UC1 );
429 thresholdedOverlapMask->setTo( Scalar::all(0) );
430 }
431 size_t size1 = keypoints1.size(), size2 = keypoints2t.size();
432 size_t minCount = MIN( size1, size2 );
433
434 // calculate overlap errors
435 std::vector<SIdx> overlaps;
436 computeOneToOneMatchedOverlaps( keypoints1, keypoints2t, ifEvaluateDetectors, overlaps, overlapThreshold/*min overlap*/ );
437
438 correspondencesCount = -1;
439 repeatability = -1.f;
440 if( overlaps.empty() )
441 return;
442
443 if( ifEvaluateDetectors )
444 {
445 // regions one-to-one matching
446 correspondencesCount = (int)overlaps.size();
447 repeatability = minCount ? (float)correspondencesCount / minCount : -1;
448 }
449 else
450 {
451 for( size_t i = 0; i < overlaps.size(); i++ )
452 {
453 int y = overlaps[i].i1;
454 int x = overlaps[i].i2;
455 thresholdedOverlapMask->at<uchar>(y,x) = 1;
456 }
457 }
458}

499void cv::computeRecallPrecisionCurve( const std::vector<std::vector<DMatch> >& matches1to2,
500 const std::vector<std::vector<uchar> >& correctMatches1to2Mask,
501 std::vector<Point2f>& recallPrecisionCurve )
502{
503 CV_INSTRUMENT_REGION();
504
505 CV_Assert( matches1to2.size() == correctMatches1to2Mask.size() );
506
507 std::vector<DMatchForEvaluation> allMatches;
508 int correspondenceCount = 0;
509 for( size_t i = 0; i < matches1to2.size(); i++ )
510 {
511 for( size_t j = 0; j < matches1to2[i].size(); j++ )
512 {
513 DMatchForEvaluation match = matches1to2[i][j];
514 match.isCorrect = correctMatches1to2Mask[i][j] ;
515 allMatches.push_back( match );
516 correspondenceCount += match.isCorrect != 0 ? 1 : 0;
517 }
518 }
519
520 std::sort( allMatches.begin(), allMatches.end() );
521
522 int correctMatchCount = 0, falseMatchCount = 0;
523 recallPrecisionCurve.resize( allMatches.size() );
524 for( size_t i = 0; i < allMatches.size(); i++ )
525 {
526 if( allMatches[i].isCorrect )
527 correctMatchCount++;
528 else
529 falseMatchCount++;
530
531 float r = recall( correctMatchCount, correspondenceCount );
532 float p = precision( correctMatchCount, falseMatchCount );
533 recallPrecisionCurve[i] = Point2f(1-p, r);
534 }
