19// ONNX ImageDecoder operator
20// Spec: https://onnx.ai/onnx/operators/onnx__ImageDecoder.html
21// Supported opsets: 20

23class ImageDecoderLayerImpl CV_FINAL : public ImageDecoderLayer
24{
25public:
26 enum PixelFormat { PF_RGB, PF_BGR, PF_GRAYSCALE };
27 PixelFormat pixelFormat;
28
29 ImageDecoderLayerImpl(const LayerParams& params)
30 {
31 setParamsFrom(params);
32 String pf = params.get<String>("pixel_format", "RGB");
33 if (pf == "RGB")
34 pixelFormat = PF_RGB;
35 else if (pf == "BGR")
36 pixelFormat = PF_BGR;
37 else if (pf == "Grayscale")
38 pixelFormat = PF_GRAYSCALE;
39 else
40 CV_Error_(Error::StsBadArg, ("DNN/ImageDecoder: unsupported pixel_format '%s'", pf.c_str()));
41 }
42
43 virtual bool dynamicOutputShapes() const CV_OVERRIDE
44 {
45 return true;
46 }
47
48 bool supportBackend(int backendId) CV_OVERRIDE
49 {
50 return backendId == DNN_BACKEND_OPENCV;
51 }
52
53 bool getMemoryShapes(const std::vector<MatShape>& inputs, const int /*requiredOutputs*/,
54 std::vector<MatShape>& outputs, std::vector<MatShape>& /*internals*/) const CV_OVERRIDE
55 {
56 CV_Assert(inputs.size() == 1);
57 int channels = pixelFormat == PF_GRAYSCALE ? 1 : 3;
58 MatShape out({-1, -1, channels});
59 outputs.assign(1, out);
60 return false;
61 }
62
63 void getTypes(const std::vector<MatType>& /*inputs*/, const int requiredOutputs,
64 const int /*requiredInternals*/, std::vector<MatType>& outputs,
65 std::vector<MatType>& /*internals*/) const CV_OVERRIDE
66 {
67 outputs.assign(requiredOutputs, CV_8U);
68 }
69
70 void forward(InputArrayOfArrays in_arr, OutputArrayOfArrays out_arr, OutputArrayOfArrays) CV_OVERRIDE
71 {
72 CV_Assert(in_arr.size().area() == 1);
73 Mat encoded = in_arr.getMat(0);
74 CV_CheckTypeEQ(encoded.type(), CV_8UC1, "DNN/ImageDecoder: input must be a uint8 tensor");
75 CV_Assert(encoded.isContinuous());
76
77#ifndef HAVE_OPENCV_IMGCODECS
78 CV_UNUSED(out_arr);
79 CV_Error(Error::StsNotImplemented, "DNN/ImageDecoder: OpenCV was built without imgcodecs support");
80#else
81 Mat buf(1, (int)encoded.total(), CV_8UC1, encoded.data);
82 Mat decoded;
83 if (pixelFormat == PF_GRAYSCALE)
84 {
85 Mat color = imdecode(buf, IMREAD_COLOR_BGR);
86 cvtColor(color, decoded, COLOR_BGR2GRAY);
87 }
88 else
89 {
90 decoded = imdecode(buf, pixelFormat == PF_RGB ? IMREAD_COLOR_RGB : IMREAD_COLOR_BGR);
91 }
92
93 const int channels = pixelFormat == PF_GRAYSCALE ? 1 : 3;
94 MatShape outShape({decoded.rows, decoded.cols, channels});
95
96 // Zero-copy reshape; ternary guards the empty case since reshape() throws on it.
97 Mat Y = decoded.empty() ? Mat(outShape, CV_8U) : decoded.reshape(1, outShape);
98
99 auto kind = out_arr.kind();
100 if (kind == _InputArray::STD_VECTOR_MAT)
101 {
102 std::vector<Mat>& out_mats = out_arr.getMatVecRef();
103 out_mats.resize(1);
104 out_mats[0] = Y;
105 }
106 else
107 {
108 CV_Assert(kind == _InputArray::STD_VECTOR_UMAT);
109 std::vector<UMat>& out_umats = out_arr.getUMatVecRef();
110 out_umats.resize(1);
111 Y.copyTo(out_umats[0]);
112 }
113#endif
114 }
115};

117Ptr<ImageDecoderLayer> ImageDecoderLayer::create(const LayerParams& params)
118{
119 return makePtr<ImageDecoderLayerImpl>(params);
120}
