60static bool ocl_boxFilter3x3_8UC1( InputArray _src, OutputArray _dst, int ddepth,
61 Size ksize, Point anchor, int borderType, bool normalize )
62{
63 const ocl::Device & dev = ocl::Device::getDefault();
64 int type = _src.type(), sdepth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type);
65
66 if (ddepth < 0)
67 ddepth = sdepth;
68
69 if (anchor.x < 0)
70 anchor.x = ksize.width / 2;
71 if (anchor.y < 0)
72 anchor.y = ksize.height / 2;
73
74 if ( !(dev.isIntel() && (type == CV_8UC1) &&
75 (_src.offset() == 0) && (_src.step() % 4 == 0) &&
76 (_src.cols() % 16 == 0) && (_src.rows() % 2 == 0) &&
77 (anchor.x == 1) && (anchor.y == 1) &&
78 (ksize.width == 3) && (ksize.height == 3)) )
79 return false;
80
81 float alpha = 1.0f / (ksize.height * ksize.width);
82 Size size = _src.size();
83 size_t globalsize[2] = { 0, 0 };
84 size_t localsize[2] = { 0, 0 };
85 const char * const borderMap[] = { "BORDER_CONSTANT", "BORDER_REPLICATE", "BORDER_REFLECT", 0, "BORDER_REFLECT_101" };
86
87 globalsize[0] = size.width / 16;
88 globalsize[1] = size.height / 2;
89
90 char build_opts[1024];
91 snprintf(build_opts, sizeof(build_opts), "-D %s %s", borderMap[borderType], normalize ? "-D NORMALIZE" : "");
92
93 ocl::Kernel kernel("boxFilter3x3_8UC1_cols16_rows2", cv::ocl::imgproc::boxFilter3x3_oclsrc, build_opts);
94 if (kernel.empty())
95 return false;
96
97 UMat src = _src.getUMat();
98 _dst.create(size, CV_MAKETYPE(ddepth, cn));
99 if (!(_dst.offset() == 0 && _dst.step() % 4 == 0))
100 return false;
101 UMat dst = _dst.getUMat();
102
103 int idxArg = kernel.set(0, ocl::KernelArg::PtrReadOnly(src));
104 idxArg = kernel.set(idxArg, (int)src.step);
105 idxArg = kernel.set(idxArg, ocl::KernelArg::PtrWriteOnly(dst));
106 idxArg = kernel.set(idxArg, (int)dst.step);
107 idxArg = kernel.set(idxArg, (int)dst.rows);
108 idxArg = kernel.set(idxArg, (int)dst.cols);
109 if (normalize)
110 idxArg = kernel.set(idxArg, (float)alpha);
111
112 return kernel.run(2, globalsize, (localsize[0] == 0) ? NULL : localsize, false);
113}

115static bool ocl_boxFilter( InputArray _src, OutputArray _dst, int ddepth,
116 Size ksize, Point anchor, int borderType, bool normalize, bool sqr = false )
117{
118 const ocl::Device & dev = ocl::Device::getDefault();
119 int type = _src.type(), sdepth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type), esz = CV_ELEM_SIZE(type);
120 bool doubleSupport = dev.doubleFPConfig() > 0;
121
122 if (ddepth < 0)
123 ddepth = sdepth;
124
125 if (cn > 4 || (!doubleSupport && (sdepth == CV_64F || ddepth == CV_64F)) ||
126 _src.offset() % esz != 0 || _src.step() % esz != 0)
127 return false;
128
129 if (anchor.x < 0)
130 anchor.x = ksize.width / 2;
131 if (anchor.y < 0)
132 anchor.y = ksize.height / 2;
133
134 int computeUnits = ocl::Device::getDefault().maxComputeUnits();
135 float alpha = 1.0f / (ksize.height * ksize.width);
136 Size size = _src.size(), wholeSize;
137 bool isolated = (borderType & BORDER_ISOLATED) != 0;
138 borderType &= ~BORDER_ISOLATED;
139 int wdepth = std::max(CV_32F, std::max(ddepth, sdepth)),
140 wtype = CV_MAKE_TYPE(wdepth, cn), dtype = CV_MAKE_TYPE(ddepth, cn);
141
142 const char * const borderMap[] = { "BORDER_CONSTANT", "BORDER_REPLICATE", "BORDER_REFLECT", 0, "BORDER_REFLECT_101" };
143 size_t globalsize[2] = { (size_t)size.width, (size_t)size.height };
144 size_t localsize_general[2] = { 0, 1 }, * localsize = NULL;
145
146 UMat src = _src.getUMat();
147 if (!isolated)
148 {
149 Point ofs;
150 src.locateROI(wholeSize, ofs);
151 }
152
153 int h = isolated ? size.height : wholeSize.height;
154 int w = isolated ? size.width : wholeSize.width;
155
156 size_t maxWorkItemSizes[32];
157 ocl::Device::getDefault().maxWorkItemSizes(maxWorkItemSizes);
158 int tryWorkItems = (int)maxWorkItemSizes[0];
159
160 ocl::Kernel kernel;
161
162 if (dev.isIntel() && !(dev.type() & ocl::Device::TYPE_CPU) &&
163 ((ksize.width < 5 && ksize.height < 5 && esz <= 4) ||
164 (ksize.width == 5 && ksize.height == 5 && cn == 1)))
165 {
166 if (w < ksize.width || h < ksize.height)
167 return false;
168
169 // Figure out what vector size to use for loading the pixels.
170 int pxLoadNumPixels = cn != 1 || size.width % 4 ? 1 : 4;
171 int pxLoadVecSize = cn * pxLoadNumPixels;
172
173 // Figure out how many pixels per work item to compute in X and Y
174 // directions. Too many and we run out of registers.
175 int pxPerWorkItemX = 1, pxPerWorkItemY = 1;
176 if (cn <= 2 && ksize.width <= 4 && ksize.height <= 4)
177 {
178 pxPerWorkItemX = size.width % 8 ? size.width % 4 ? size.width % 2 ? 1 : 2 : 4 : 8;
179 pxPerWorkItemY = size.height % 2 ? 1 : 2;
180 }
181 else if (cn < 4 || (ksize.width <= 4 && ksize.height <= 4))
182 {
183 pxPerWorkItemX = size.width % 2 ? 1 : 2;
184 pxPerWorkItemY = size.height % 2 ? 1 : 2;
185 }
186 globalsize[0] = size.width / pxPerWorkItemX;
187 globalsize[1] = size.height / pxPerWorkItemY;
188
189 // Need some padding in the private array for pixels
190 int privDataWidth = roundUp(pxPerWorkItemX + ksize.width - 1, pxLoadNumPixels);
191
192 // Make the global size a nice round number so the runtime can pick
193 // from reasonable choices for the workgroup size
194 const int wgRound = 256;
195 globalsize[0] = roundUp(globalsize[0], wgRound);
196
197 char build_options[1024], cvt[2][50];
198 snprintf(build_options, sizeof(build_options), "-D cn=%d "
199 "-D ANCHOR_X=%d -D ANCHOR_Y=%d -D KERNEL_SIZE_X=%d -D KERNEL_SIZE_Y=%d "
200 "-D PX_LOAD_VEC_SIZE=%d -D PX_LOAD_NUM_PX=%d "
201 "-D PX_PER_WI_X=%d -D PX_PER_WI_Y=%d -D PRIV_DATA_WIDTH=%d -D %s -D %s "
202 "-D PX_LOAD_X_ITERATIONS=%d -D PX_LOAD_Y_ITERATIONS=%d "
203 "-D srcT=%s -D srcT1=%s -D dstT=%s -D dstT1=%s -D WT=%s -D WT1=%s "
204 "-D convertToWT=%s -D convertToDstT=%s%s%s -D PX_LOAD_FLOAT_VEC_CONV=convert_%s -D OP_BOX_FILTER",
205 cn, anchor.x, anchor.y, ksize.width, ksize.height,
206 pxLoadVecSize, pxLoadNumPixels,
207 pxPerWorkItemX, pxPerWorkItemY, privDataWidth, borderMap[borderType],
208 isolated ? "BORDER_ISOLATED" : "NO_BORDER_ISOLATED",
209 privDataWidth / pxLoadNumPixels, pxPerWorkItemY + ksize.height - 1,
210 ocl::typeToStr(type), ocl::typeToStr(sdepth), ocl::typeToStr(dtype),
211 ocl::typeToStr(ddepth), ocl::typeToStr(wtype), ocl::typeToStr(wdepth),
212 ocl::convertTypeStr(sdepth, wdepth, cn, cvt[0], sizeof(cvt[0])),
213 ocl::convertTypeStr(wdepth, ddepth, cn, cvt[1], sizeof(cvt[1])),
214 normalize ? " -D NORMALIZE" : "", sqr ? " -D SQR" : "",
215 ocl::typeToStr(CV_MAKE_TYPE(wdepth, pxLoadVecSize)) //PX_LOAD_FLOAT_VEC_CONV
216 );
217
218
219 if (!kernel.create("filterSmall", cv::ocl::imgproc::filterSmall_oclsrc, build_options))
220 return false;
221 }
222 else
223 {
224 localsize = localsize_general;
225 for ( ; ; )
226 {
227 int BLOCK_SIZE_X = tryWorkItems, BLOCK_SIZE_Y = std::min(ksize.height * 10, size.height);
228
229 while (BLOCK_SIZE_X > 32 && BLOCK_SIZE_X >= ksize.width * 2 && BLOCK_SIZE_X > size.width * 2)
230 BLOCK_SIZE_X /= 2;
231 while (BLOCK_SIZE_Y < BLOCK_SIZE_X / 8 && BLOCK_SIZE_Y * computeUnits * 32 < size.height)
232 BLOCK_SIZE_Y *= 2;
233
234 if (ksize.width > BLOCK_SIZE_X || w < ksize.width || h < ksize.height)
235 return false;
236
237 char cvt[2][50];
238 String opts = format("-D LOCAL_SIZE_X=%d -D BLOCK_SIZE_Y=%d -D ST=%s -D DT=%s -D WT=%s -D convertToDT=%s -D convertToWT=%s"
239 " -D ANCHOR_X=%d -D ANCHOR_Y=%d -D KERNEL_SIZE_X=%d -D KERNEL_SIZE_Y=%d -D %s%s%s%s%s"
240 " -D ST1=%s -D DT1=%s -D cn=%d",
241 BLOCK_SIZE_X, BLOCK_SIZE_Y, ocl::typeToStr(type), ocl::typeToStr(CV_MAKE_TYPE(ddepth, cn)),
242 ocl::typeToStr(CV_MAKE_TYPE(wdepth, cn)),
243 ocl::convertTypeStr(wdepth, ddepth, cn, cvt[0], sizeof(cvt[0])),
244 ocl::convertTypeStr(sdepth, wdepth, cn, cvt[1], sizeof(cvt[1])),
245 anchor.x, anchor.y, ksize.width, ksize.height, borderMap[borderType],
246 isolated ? " -D BORDER_ISOLATED" : "", doubleSupport ? " -D DOUBLE_SUPPORT" : "",
247 normalize ? " -D NORMALIZE" : "", sqr ? " -D SQR" : "",
248 ocl::typeToStr(sdepth), ocl::typeToStr(ddepth), cn);
249
250 localsize[0] = BLOCK_SIZE_X;
251 globalsize[0] = divUp(size.width, BLOCK_SIZE_X - (ksize.width - 1)) * BLOCK_SIZE_X;
252 globalsize[1] = divUp(size.height, BLOCK_SIZE_Y);
253
254 kernel.create("boxFilter", cv::ocl::imgproc::boxFilter_oclsrc, opts);
255 if (kernel.empty())
256 return false;
257
258 size_t kernelWorkGroupSize = kernel.workGroupSize();
259 if (localsize[0] <= kernelWorkGroupSize)
260 break;
261 if (BLOCK_SIZE_X < (int)kernelWorkGroupSize)
262 return false;
263
264 tryWorkItems = (int)kernelWorkGroupSize;
265 }
266 }
267
268 _dst.create(size, CV_MAKETYPE(ddepth, cn));
269 UMat dst = _dst.getUMat();
270
271 int idxArg = kernel.set(0, ocl::KernelArg::PtrReadOnly(src));
272 idxArg = kernel.set(idxArg, (int)src.step);
273 int srcOffsetX = (int)((src.offset % src.step) / src.elemSize());
274 int srcOffsetY = (int)(src.offset / src.step);
275 int srcEndX = isolated ? srcOffsetX + size.width : wholeSize.width;
276 int srcEndY = isolated ? srcOffsetY + size.height : wholeSize.height;
277 idxArg = kernel.set(idxArg, srcOffsetX);
278 idxArg = kernel.set(idxArg, srcOffsetY);
279 idxArg = kernel.set(idxArg, srcEndX);
280 idxArg = kernel.set(idxArg, srcEndY);
281 idxArg = kernel.set(idxArg, ocl::KernelArg::WriteOnly(dst));
282 if (normalize)
283 idxArg = kernel.set(idxArg, (float)alpha);
284
285 return kernel.run(2, globalsize, localsize, false);
286}

318void boxFilter(InputArray _src, OutputArray _dst, int ddepth,
319 Size ksize, Point anchor,
320 bool normalize, int borderType)
321{
322 CV_INSTRUMENT_REGION();
323
324 CV_Assert(!_src.empty());
325 CV_Assert(ksize.width > 0 && ksize.height > 0);
326
327 CV_OCL_RUN(_dst.isUMat() &&
328 (borderType == BORDER_REPLICATE || borderType == BORDER_CONSTANT ||
329 borderType == BORDER_REFLECT || borderType == BORDER_REFLECT_101),
330 ocl_boxFilter3x3_8UC1(_src, _dst, ddepth, ksize, anchor, borderType, normalize))
331
332 CV_OCL_RUN(_dst.isUMat(), ocl_boxFilter(_src, _dst, ddepth, ksize, anchor, borderType, normalize))
333
334 Mat src = _src.getMat();
335 int stype = src.type(), sdepth = CV_MAT_DEPTH(stype), cn = CV_MAT_CN(stype);
336 if( ddepth < 0 )
337 ddepth = sdepth;
338 _dst.create( src.size(), CV_MAKETYPE(ddepth, cn) );
339 Mat dst = _dst.getMat();
340 if( borderType != BORDER_CONSTANT && normalize && (borderType & BORDER_ISOLATED) != 0 )
341 {
342 if( src.rows == 1 )
343 ksize.height = 1;
344 if( src.cols == 1 )
345 ksize.width = 1;
346 }
347
348 Point ofs;
349 Size wsz(src.cols, src.rows);
350 if(!(borderType&BORDER_ISOLATED))
351 src.locateROI( wsz, ofs );
352
353 CALL_HAL(boxFilter, cv_hal_boxFilter, src.ptr(), src.step, dst.ptr(), dst.step, src.cols, src.rows, sdepth, ddepth, cn,
354 ofs.x, ofs.y, wsz.width - src.cols - ofs.x, wsz.height - src.rows - ofs.y, ksize.width, ksize.height,
355 anchor.x, anchor.y, normalize, borderType&~BORDER_ISOLATED);
356
357 borderType = (borderType&~BORDER_ISOLATED);
358
359 if(sdepth >= CV_32F && src.type() == dst.type() && (ksize.height <= 5 && ksize.width <= 5))
360 {
361 CV_CPU_DISPATCH(blockSum, (src, dst, ksize, anchor, wsz, ofs, normalize, borderType),
362 CV_CPU_DISPATCH_MODES_ALL);
363 return;
364 }
365
366 Ptr<FilterEngine> f = createBoxFilter( src.type(), dst.type(),
367 ksize, anchor, normalize, borderType );
368
369 f->apply( src, dst, wsz, ofs );
370}

394void sqrBoxFilter(InputArray _src, OutputArray _dst, int ddepth,
395 Size ksize, Point anchor,
396 bool normalize, int borderType)
397{
398 CV_INSTRUMENT_REGION();
399
400 CV_Assert(!_src.empty());
401
402 int srcType = _src.type(), sdepth = CV_MAT_DEPTH(srcType), cn = CV_MAT_CN(srcType);
403 Size size = _src.size();
404
405 if( ddepth < 0 )
406 ddepth = sdepth < CV_32F ? CV_32F : CV_64F;
407
408 if( borderType != BORDER_CONSTANT && normalize )
409 {
410 if( size.height == 1 )
411 ksize.height = 1;
412 if( size.width == 1 )
413 ksize.width = 1;
414 }
415
416 CV_OCL_RUN(_dst.isUMat() && _src.dims() <= 2,
417 ocl_boxFilter(_src, _dst, ddepth, ksize, anchor, borderType, normalize, true))
418
419 int sumDepth = CV_64F;
420 if( sdepth == CV_8U )
421 sumDepth = CV_32S;
422 int sumType = CV_MAKETYPE( sumDepth, cn ), dstType = CV_MAKETYPE(ddepth, cn);
423
424 Mat src = _src.getMat();
425 _dst.create( size, dstType );
426 Mat dst = _dst.getMat();
427
428 Ptr<BaseRowFilter> rowFilter = getSqrRowSumFilter(srcType, sumType, ksize.width, anchor.x );
429 Ptr<BaseColumnFilter> columnFilter = getColumnSumFilter(sumType,
430 dstType, ksize.height, anchor.y,
431 normalize ? 1./(ksize.width*ksize.height) : 1);
432
433 Ptr<FilterEngine> f = makePtr<FilterEngine>(Ptr<BaseFilter>(), rowFilter, columnFilter,
434 srcType, dstType, sumType, borderType );
435 Point ofs;
436 Size wsz(src.cols, src.rows);
437 src.locateROI( wsz, ofs );
438
439 f->apply( src, dst, wsz, ofs );
440}
