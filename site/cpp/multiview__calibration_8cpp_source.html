47static double robustWrapper (const Mat& ptsErrors, Mat& weights, const RobustFunction &fnc) {
48 Mat errs;
49 ptsErrors.convertTo(errs, CV_32F);
50 weights.create((int)ptsErrors.total()*ptsErrors.channels(), 1, CV_32FC1);
51 const Point2f * errs_ptr = errs.ptr<Point2f>();
52 float * weights_ptr = weights.ptr<float>();
53 double robust_sum_sqr_errs = 0.0;
54 for (int pt = 0; pt < (int)errs.total(); pt++) {
55 Point2f p = errs_ptr[pt];
56 float sqr_err = p.dot(p);
57 float w = fnc.getError(sqr_err);
58 weights_ptr[pt*2 + 0] = w;
59 weights_ptr[pt*2 + 1] = w;
60 robust_sum_sqr_errs += w * sqr_err;
61 }
62 return robust_sum_sqr_errs;
63}

65static double computeReprojectionMSE(const Mat &obj_points_, const Mat &img_points_, const Matx33d &K, const Mat &distortion,
66 const Mat &rvec, const Mat &tvec, InputArray rvec2, InputArray tvec2, int model) {
67 Mat r, t;
68 if (!rvec2.empty() && !tvec2.empty()) {
69 composeRT(rvec, tvec, rvec2, tvec2, r, t);
70 } else {
71 r = rvec; t = tvec;
72 }
73 Mat tmpImagePoints, obj_points = obj_points_, img_points = img_points_;
74 if (model == cv::CALIB_MODEL_FISHEYE) {
75 obj_points = obj_points.reshape(3); // must be 3 channels
76 fisheye::projectPoints(obj_points, tmpImagePoints, r, t, K, distortion);
77 } else if (model == cv::CALIB_MODEL_PINHOLE) {
78 projectPoints(obj_points, r, t, K, distortion, tmpImagePoints);
79 } else {
80 CV_Error(Error::StsBadArg, "Unsupported camera model!");
81 }
82
83 if (img_points.channels() != tmpImagePoints.channels())
84 img_points = img_points.reshape(tmpImagePoints.channels());
85 if (img_points.rows != tmpImagePoints.rows)
86 img_points = img_points.t();
87 subtract (tmpImagePoints, img_points, tmpImagePoints);
88 return norm(tmpImagePoints, NORM_L2SQR) / tmpImagePoints.rows;
89}

91static bool maximumSpanningTree (int NUM_CAMERAS, int NUM_FRAMES, const std::vector<std::vector<bool>> &detection_mask,
92 std::vector<int> &parent, std::vector<std::vector<int>> &overlap,
93 std::vector<std::vector<Vec3d>> &opt_axes,
94 const std::vector<std::vector<bool>> &is_valid_angle2pattern,
95 const std::vector<std::vector<float>> &points_area_ratio,
96 double WEIGHT_ANGLE_PATTERN, double WEIGHT_CAMERAS_ANGLES) {
97 const double THR_CAMERAS_ANGLES = 160*M_PI/180;
98 // build weights matrix
99 overlap = std::vector<std::vector<int>>(NUM_CAMERAS, std::vector<int>(NUM_CAMERAS, 0));
100 std::vector<std::vector<double>> weights(NUM_CAMERAS, std::vector<double>(NUM_CAMERAS, DBL_MIN));
101 for (int c1 = 0; c1 < NUM_CAMERAS; c1++) {
102 for (int c2 = c1+1; c2 < NUM_CAMERAS; c2++) {
103 double weight = 0;
104 int overlaps = 0;
105 for (int f = 0; f < NUM_FRAMES; f++) {
106 if (detection_mask[c1][f] && detection_mask[c2][f]) {
107 overlaps += 1;
108 weight += points_area_ratio[c1][f] + points_area_ratio[c2][f];
109 weight += WEIGHT_ANGLE_PATTERN * ((int)is_valid_angle2pattern[c1][f] + (int)is_valid_angle2pattern[c2][f]);
110 if (WEIGHT_CAMERAS_ANGLES > 0) {
111 // angle between cameras optical axes
112 weight += WEIGHT_CAMERAS_ANGLES * int(acos(opt_axes[c1][f].dot(opt_axes[c2][f])) < THR_CAMERAS_ANGLES);
113 }
114 }
115 }
116 if (overlaps > 0) {
117 overlap[c1][c2] = overlap[c2][c1] = overlaps;
118 weights[c1][c2] = weights[c2][c1] = overlaps + weight;
119 }
120 }
121 }
122
123 // find maximum spanning tree using Prim's algorithm
124 std::vector<bool> visited(NUM_CAMERAS, false);
125 std::vector<double> weight(NUM_CAMERAS, DBL_MIN);
126 parent = std::vector<int>(NUM_CAMERAS, -1);
127 weight[0] = DBL_MAX;
128 for (int cam = 0; cam < NUM_CAMERAS-1; cam++) {
129 int max_weight_idx = -1;
130 auto max_weight = DBL_MIN;
131 for (int cam2 = 0; cam2 < NUM_CAMERAS; cam2++) {
132 if (!visited[cam2] && max_weight < weight[cam2]) {
133 max_weight = weight[cam2];
134 max_weight_idx = cam2;
135 }
136 }
137 if (max_weight_idx == -1)
138 return false;
139 visited[max_weight_idx] = true;
140 for (int cam2 = 0; cam2 < NUM_CAMERAS; cam2++) {
141 if (!visited[cam2] && overlap[max_weight_idx][cam2] > 0) {
142 if (weight[cam2] < weights[max_weight_idx][cam2]) {
143 weight[cam2] = weights[max_weight_idx][cam2];
144 parent[cam2] = max_weight_idx;
145 }
146 }
147 }
148 }
149 return true;
150}

152static void imagePointsArea (const std::vector<Size> &imageSize, const std::vector<std::vector<bool>> &detection_mask_mat,
153 const std::vector<std::vector<Mat>> &imagePoints, std::vector<std::vector<float>> &points_ratio_area) {
154 const int NUM_CAMERAS = (int) imageSize.size(), NUM_FRAMES = (int)detection_mask_mat[0].size();
155 for (int c = 0; c < NUM_CAMERAS; c++) {
156 const auto img_area = (float)(imageSize[c].width * imageSize[c].height);
157 for (int f = 0; f < NUM_FRAMES; f++) {
158 if (!detection_mask_mat[c][f])
159 continue;
160 CV_Assert((imagePoints[c][f].type() == CV_32F && imagePoints[c][f].cols == 2) || imagePoints[c][f].type() == CV_32FC2);
161 std::vector<int> hull;
162 const auto img_pts = imagePoints[c][f];
163 const auto * const image_pts_ptr = (float *) img_pts.data;
164 convexHull(img_pts, hull, true/*has to be clockwise*/, false/*indices*/);
165 float area = 0;
166 int j = hull.back();
167 // http://alienryderflex.com/polygon_area/
168 for (int i : hull) {
169 area += (image_pts_ptr[j*2] + image_pts_ptr[i*2])*(image_pts_ptr[j*2+1] - image_pts_ptr[i*2+1]);
170 j = i;
171 }
172 points_ratio_area[c][f] = area*.5f / img_area;
173 }
174 }

177static void selectPairsBFS (std::vector<std::pair<int,int>> &pairs, int NUM_CAMERAS, const std::vector<int> &parent) {
178 // find pairs using Breadth First Search graph traversing
179 // it is important to keep this order of pairs, since it is easier
180 // to find relative views wrt to 0-th camera.
181 std::vector<int> nodes = {0};
182 pairs.reserve(NUM_CAMERAS-1);
183 while (!nodes.empty()) {
184 std::vector<int> new_nodes;
185 for (int n : nodes) {
186 for (int c = 0; c < NUM_CAMERAS; c++) {
187 if (parent[c] == n) {
188 pairs.emplace_back(std::make_pair(n, c));
189 new_nodes.emplace_back(c);
190 }
191 }
192 }
193 nodes = new_nodes;
194 }

197static double getScaleOfObjPoints (int NUM_PATTERN_PTS, const Mat &obj_pts, bool obj_points_in_rows) {
198 double scale_3d_pts = 0.0;
199 // compute scale of 3D points as the maximum pairwise distance
200 for (int i = 0; i < NUM_PATTERN_PTS; i++) {
201 for (int j = i+1; j < NUM_PATTERN_PTS; j++) {
202 double dist;
203 if (obj_points_in_rows) {
204 dist = norm(obj_pts.row(i)-obj_pts.row(j), NORM_L2SQR);
205 } else {
206 dist = norm(obj_pts.col(i)-obj_pts.col(j), NORM_L2SQR);
207 }
208 if (scale_3d_pts < dist) {
209 scale_3d_pts = dist;
210 }
211 }
212 }
213 return scale_3d_pts;
214}

216static void thresholdPatternCameraAngles (int NUM_PATTERN_PTS, double THR_PATTERN_CAMERA_ANGLES,
217 const std::vector<Mat> &objPoints_norm, const std::vector<std::vector<Vec3d>> &rvecs_all,
218 std::vector<std::vector<Vec3d>> &opt_axes, std::vector<std::vector<bool>> &is_valid_angle2pattern) {
219 const int NUM_FRAMES = (int)objPoints_norm.size(), NUM_CAMERAS = (int)rvecs_all.size();
220 is_valid_angle2pattern = std::vector<std::vector<bool>>(NUM_CAMERAS, std::vector<bool>(NUM_FRAMES, true));
221 int pattern1 = -1, pattern2 = -1, pattern3 = -1;
222 for (int f = 0; f < NUM_FRAMES; f++) {
223 double norm_normal = 0;
224 if (pattern1 == -1) {
225 // take non colinear 3 points and save them
226 for (int p1 = 0; p1 < NUM_PATTERN_PTS; p1++) {
227 for (int p2 = p1+1; p2 < NUM_PATTERN_PTS; p2++) {
228 for (int p3 = NUM_PATTERN_PTS-1; p3 > p2; p3--) { // start from the last point
229 Mat pattern_normal = (objPoints_norm[f].row(p2)-objPoints_norm[f].row(p1))
230 .cross(objPoints_norm[f].row(p3)-objPoints_norm[f].row(p1));
231 norm_normal = norm(pattern_normal, NORM_L2SQR);
232 if (norm_normal > 1e-6) {
233 pattern1 = p1;
234 pattern2 = p2;
235 pattern3 = p3;
236 norm_normal = sqrt(norm_normal);
237 break;
238 }
239 }
240 if (pattern1 != -1) break;
241 }
242 if (pattern1 != -1) break;
243 }
244 if (pattern1 == -1) {
245 CV_Error(Error::StsBadArg, "Pattern points are collinear!");
246 }
247 }
248 Vec3d pattern_normal = (objPoints_norm[f].row(pattern2)-objPoints_norm[f].row(pattern1)).
249 cross(objPoints_norm[f].row(pattern3)-objPoints_norm[f].row(pattern1));
250 norm_normal = norm(pattern_normal);
251 pattern_normal /= norm_normal;
252
253 for (int c = 0; c < NUM_CAMERAS; c++) {
254 Matx33d R;
255 Rodrigues(rvecs_all[c][f], R);
256 opt_axes[c][f] = Vec3d(Mat(R.row(2)));
257 const double angle = acos(opt_axes[c][f].dot(pattern_normal));
258 is_valid_angle2pattern[c][f] = min(M_PI-angle, angle) < THR_PATTERN_CAMERA_ANGLES;
259 }
260 }

263static void pairwiseRegistration (const std::vector<std::pair<int,int>> &pairs,
264 const cv::Mat &models, const std::vector<Mat> &objPoints_norm,
265 const std::vector<std::vector<Mat>> &imagePoints, const std::vector<std::vector<int>> &overlaps,
266 const std::vector<std::vector<bool>> &detection_mask_mat, const std::vector<Mat> &Ks,
267 const std::vector<Mat> &distortions, std::vector<Matx33d> &Rs_vec, std::vector<Vec3d> &Ts_vec,
268 Mat &intrinsic_flags, int extrinsic_flags, TermCriteria criteria) {
269 CV_UNUSED(intrinsic_flags);
270 const int NUM_FRAMES = (int)objPoints_norm.size();
271 const int NUM_CAMERAS = (int)detection_mask_mat.size();
272 std::vector<Matx33d> Rs_prior;
273 std::vector<Vec3d> Ts_prior;
274 if (extrinsic_flags & cv::CALIB_USE_EXTRINSIC_GUESS) {
275 Rs_prior.resize(NUM_CAMERAS);
276 Ts_prior.resize(NUM_CAMERAS);
277 for (int i = 0; i < NUM_CAMERAS; i++) {
278 Rs_vec[i].copyTo(Rs_prior[i]);
279 Ts_vec[i].copyTo(Ts_prior[i]);
280 }
281 }
282
283 for (const auto &pair : pairs) {
284 const int c1 = pair.first, c2 = pair.second, overlap = overlaps[c1][c2];
285 // prepare image points of two cameras and grid points
286 std::vector<Mat> image_points1, image_points2, grid_points1, grid_points2;
287 grid_points1.reserve(overlap);
288 grid_points2.reserve(overlap);
289 image_points1.reserve(overlap);
290 image_points2.reserve(overlap);
291 for (int f = 0; f < NUM_FRAMES; f++) {
292 if (detection_mask_mat[c1][f] && detection_mask_mat[c2][f]) {
293 grid_points1.emplace_back(objPoints_norm[f]);
294 grid_points2.emplace_back(objPoints_norm[f]);
295 image_points1.emplace_back(imagePoints[c1][f]);
296 image_points2.emplace_back(imagePoints[c2][f]);
297 }
298 }
299 Matx33d R;
300 Vec3d T;
301
302 if (extrinsic_flags & cv::CALIB_USE_EXTRINSIC_GUESS) {
303 R = Rs_prior[c2] * Rs_prior[c1].t();
304 T = -R * Ts_prior[c1] + Ts_prior[c2];
305 }
306
307 extrinsic_flags |= CALIB_FIX_INTRINSIC;
308 double err = registerCameras(grid_points1, grid_points2, image_points1, image_points2,
309 Ks[c1], distortions[c1], cv::CameraModel(models.at<uchar>(c1)),
310 Ks[c2], distortions[c2], cv::CameraModel(models.at<uchar>(c2)),
311 R, T, noArray(), noArray(), noArray(), noArray(), noArray(),
312 extrinsic_flags, criteria);
313 CV_LOG_INFO(NULL, "Pair " << c1 << "-" << c2 << " registration RMS " << err);
314 // R_0 = I
315 // R_ij = R_i R_j^T => R_i = R_ij R_j
316 // t_ij = ti - R_ij tj => t_i = t_ij + R_ij t_j
317 if (c1 == 0) {
318 Rs_vec[c2] = R;
319 Ts_vec[c2] = T;
320 } else {
321 Rs_vec[c2] = Matx33d(Mat(R * Rs_vec[c1]));
322 Ts_vec[c2] = Vec3d(Mat(T + R * Ts_vec[c1]));
323 }
324 }

327static void pairwiseStereoCalibration (const std::vector<std::pair<int,int>> &pairs,
328 const cv::Mat &models, const std::vector<Mat> &objPoints_norm,
329 const std::vector<std::vector<Mat>> &imagePoints, const std::vector<std::vector<int>> &overlaps,
330 const std::vector<std::vector<bool>> &detection_mask_mat, const std::vector<Mat> &Ks,
331 const std::vector<Mat> &distortions, std::vector<Matx33d> &Rs_vec, std::vector<Vec3d> &Ts_vec,
332 Mat &intrinsic_flags, int extrinsic_flags, TermCriteria criteria) {
333 const int NUM_FRAMES = (int)objPoints_norm.size();
334 const int NUM_CAMERAS = (int)detection_mask_mat.size();
335 std::vector<Matx33d> Rs_prior;
336 std::vector<Vec3d> Ts_prior;
337 if (extrinsic_flags & cv::CALIB_USE_EXTRINSIC_GUESS) {
338 Rs_prior.resize(NUM_CAMERAS);
339 Ts_prior.resize(NUM_CAMERAS);
340 for (int i = 0; i < NUM_CAMERAS; i++) {
341 Rs_vec[i].copyTo(Rs_prior[i]);
342 Ts_vec[i].copyTo(Ts_prior[i]);
343 }
344 }
345
346 for (const auto &pair : pairs) {
347 const int c1 = pair.first, c2 = pair.second, overlap = overlaps[c1][c2];
348 // prepare image points of two cameras and grid points
349 std::vector<Mat> image_points1, image_points2, grid_points;
350 grid_points.reserve(overlap);
351 image_points1.reserve(overlap);
352 image_points2.reserve(overlap);
353 const bool are_fisheye_cams = models.at<uchar>(c1) == cv::CALIB_MODEL_FISHEYE &&
354 models.at<uchar>(c2) == cv::CALIB_MODEL_FISHEYE;
355 for (int f = 0; f < NUM_FRAMES; f++) {
356 if (detection_mask_mat[c1][f] && detection_mask_mat[c2][f]) {
357 grid_points.emplace_back((are_fisheye_cams && objPoints_norm[f].channels() != 3) ?
358 objPoints_norm[f].reshape(3): objPoints_norm[f]);
359 image_points1.emplace_back((are_fisheye_cams && imagePoints[c1][f].channels() != 2) ?
360 imagePoints[c1][f].reshape(2) : imagePoints[c1][f]);
361 image_points2.emplace_back((are_fisheye_cams && imagePoints[c2][f].channels() != 2) ?
362 imagePoints[c2][f].reshape(2) : imagePoints[c2][f]);
363 }
364 }
365 Matx33d R;
366 Vec3d T;
367
368 if (extrinsic_flags & cv::CALIB_USE_EXTRINSIC_GUESS) {
369 R = Rs_prior[c2] * Rs_prior[c1].t();
370 T = -R * Ts_prior[c1] + Ts_prior[c2];
371 }
372
373 // stereoCalibrate tries to overwrite distortion coefficients
374 Mat dist1 = distortions[c1].clone();
375 Mat dist2 = distortions[c2].clone();
376
377 // image size does not matter since intrinsics are used
378 if (are_fisheye_cams) {
379 extrinsic_flags |= CALIB_FIX_INTRINSIC;
380 double err = fisheye::stereoCalibrate(grid_points, image_points1, image_points2,
381 Ks[c1], dist1,
382 Ks[c2], dist2,
383 Size(), R, T,
384 extrinsic_flags, criteria);
385 CV_LOG_INFO(NULL, "Stereo pair " << c1 << "-" << c2 << " registration RMS " << err);
386 } else {
387 extrinsic_flags |= CALIB_FIX_INTRINSIC;
388 if ((intrinsic_flags.at<int>(c1) & CALIB_RATIONAL_MODEL) || (intrinsic_flags.at<int>(c2) & CALIB_RATIONAL_MODEL))
389 extrinsic_flags |= CALIB_RATIONAL_MODEL;
390 if ((intrinsic_flags.at<int>(c1) & CALIB_THIN_PRISM_MODEL) || (intrinsic_flags.at<int>(c2) & CALIB_THIN_PRISM_MODEL))
391 extrinsic_flags |= CALIB_THIN_PRISM_MODEL;
392
393 double err = stereoCalibrate(grid_points, image_points1, image_points2,
394 Ks[c1], dist1,
395 Ks[c2], dist2,
396 Size(), R, T, noArray(), noArray(), noArray(),
397 extrinsic_flags, criteria);
398 CV_LOG_INFO(NULL, "Stereo pair " << c1 << "-" << c2 << " registration RMS " << err);
399 }
400
401 // R_0 = I
402 // R_ij = R_i R_j^T => R_i = R_ij R_j
403 // t_ij = ti - R_ij tj => t_i = t_ij + R_ij t_j
404 if (c1 == 0) {
405 Rs_vec[c2] = R;
406 Ts_vec[c2] = T;
407 } else {
408 Rs_vec[c2] = Matx33d(Mat(R * Rs_vec[c1]));
409 Ts_vec[c2] = Vec3d(Mat(T + R * Ts_vec[c1]));
410 }
411 }

414static void optimizeLM (std::vector<double> &param, const RobustFunction &robust_fnc, const TermCriteria &termCrit,
415 const std::vector<bool> &valid_frames, const std::vector<std::vector<bool>> &detection_mask_mat,
416 const std::vector<Mat> &objPoints_norm, const std::vector<std::vector<Mat>> &imagePoints,
417 const std::vector<Mat> &Ks, const std::vector<Mat> &distortions,
418 const Mat& models, int NUM_PATTERN_PTS) {
419 const int NUM_FRAMES = (int) objPoints_norm.size(), NUM_CAMERAS = (int)detection_mask_mat.size();
420 int iters_lm = 0, cnt_valid_frame = 0;
421 auto lmcallback = [&](InputOutputArray _param, OutputArray JtErr_, OutputArray JtJ_, double& errnorm) {
422 auto * param_p = _param.getMat().ptr<double>();
423 errnorm = 0;
424 cnt_valid_frame = 0;
425 for (int i = 0; i < NUM_FRAMES; i++ ) {
426 if (!valid_frames[i]) continue;
427 for (int k = 0; k < NUM_CAMERAS; k++ ) {
428 // Pose for camera #0 is not optimized, but it's re-projection error is taken into account
429 if (!detection_mask_mat[k][i]) continue;
430 const int cam_idx = (k-1)*6; // camera extrinsics
431 const auto * const pose_k = (k > 0)? (param_p + cam_idx) : nullptr;
432 Vec3d om_0ToK = (k > 0)? Vec3d(pose_k[0], pose_k[1], pose_k[2]) : Vec3d(0., 0., 0.), om[2];
433 Vec3d T_0ToK = (k > 0)? Vec3d(pose_k[3], pose_k[4], pose_k[5]) : Vec3d(0., 0., 0.), T[2];
434 Matx33d dr3dr1, dr3dr2, dt3dr2, dt3dt1, dt3dt2;
435
436 auto * pi = param_p + (cnt_valid_frame+NUM_CAMERAS-1)*6; // get rvecs / tvecs for frame pose
437 om[0] = Vec3d(pi[0], pi[1], pi[2]);
438 T[0] = Vec3d(pi[3], pi[4], pi[5]);
439
440 if( JtJ_.needed() || JtErr_.needed() )
441 composeRT( om[0], T[0], om_0ToK, T_0ToK, om[1], T[1], dr3dr1, noArray(),
442 dr3dr2, noArray(), noArray(), dt3dt1, dt3dr2, dt3dt2 );
443 else
444 composeRT( om[0], T[0], om_0ToK, T_0ToK, om[1], T[1] );
445
446 // get object points
447 Mat objpt_i = objPoints_norm[i].reshape(3, 1);
448 objpt_i.convertTo(objpt_i, CV_64FC3);
449
450 Mat err( NUM_PATTERN_PTS*2, 1, CV_64F ), tmpImagePoints = err.reshape(2, 1);
451 Mat Je( NUM_PATTERN_PTS*2, 6, CV_64F ), J_0ToK( NUM_PATTERN_PTS*2, 6, CV_64F );
452 Mat dpdrot = Je.colRange(0, 3), dpdt = Je.colRange(3, 6); // num_points*2 x 3 each
453 // get image points
454 Mat imgpt_ik = imagePoints[k][i].reshape(2, 1);
455 imgpt_ik.convertTo(imgpt_ik, CV_64FC2);
456
457 if (models.at<uchar>(k)) {
458 if( JtJ_.needed() || JtErr_.needed() ) {
459 Mat jacobian; // of size num_points*2 x 15 (2 + 2 + 1 + 4 + 3 + 3; // f, c, alpha, k, om, T)
460 fisheye::projectPoints(objpt_i, tmpImagePoints, om[1], T[1], Ks[k], distortions[k], 0, jacobian);
461 jacobian.colRange(8,11).copyTo(dpdrot);
462 jacobian.colRange(11,14).copyTo(dpdt);
463 } else
464 fisheye::projectPoints(objpt_i, tmpImagePoints, om[1], T[1], Ks[k], distortions[k]);
465 } else {
466 if( JtJ_.needed() || JtErr_.needed() )
467 projectPoints(objpt_i, om[1], T[1], Ks[k], distortions[k],
468 tmpImagePoints, dpdrot, dpdt, noArray(), noArray(), noArray(), noArray());
469 else
470 projectPoints(objpt_i, om[1], T[1], Ks[k], distortions[k], tmpImagePoints);
471 }
472 subtract( tmpImagePoints, imgpt_ik, tmpImagePoints);
473 Mat weights;
474 const double robust_l2_norm = multiview::robustWrapper(tmpImagePoints, weights, robust_fnc);
475 errnorm += robust_l2_norm;
476
477 if (JtJ_.needed()) {
478 Mat JtErr = JtErr_.getMat(), JtJ = JtJ_.getMat();
479 const int eofs = (cnt_valid_frame+NUM_CAMERAS-1)*6;
480 assert( JtJ_.needed() && JtErr_.needed() );
481 // JtJ : NUM_PARAMS x NUM_PARAMS, JtErr : NUM_PARAMS x 1
482 // d(err_{x|y}R) ~ de3
483 // convert de3/{dr3,dt3} => de3{dr1,dt1} & de3{dr2,dt2}
484
485 Mat wd;
486 Mat::diag(weights).convertTo(wd, CV_64F);
487 if (k > 0) { // if not camera #0
488 for (int p = 0; p < NUM_PATTERN_PTS * 2; p++) {
489 Matx13d de3dr3, de3dt3, de3dr2, de3dt2, de3dr1, de3dt1;
490 for (int j = 0; j < 3; j++)
491 de3dr3(j) = Je.at<double>(p, j);
492
493 for (int j = 0; j < 3; j++)
494 de3dt3(j) = Je.at<double>(p, 3 + j);
495
496 for (int j = 0; j < 3; j++)
497 de3dr2(j) = J_0ToK.at<double>(p, j);
498
499 for (int j = 0; j < 3; j++)
500 de3dt2(j) = J_0ToK.at<double>(p, 3 + j);
501
502 de3dr1 = de3dr3 * dr3dr1;
503 de3dt1 = de3dt3 * dt3dt1;
504 de3dr2 = de3dr3 * dr3dr2 + de3dt3 * dt3dr2;
505 de3dt2 = de3dt3 * dt3dt2;
506
507 for (int j = 0; j < 3; j++)
508 Je.at<double>(p, j) = de3dr1(j);
509
510 for (int j = 0; j < 3; j++)
511 Je.at<double>(p, 3 + j) = de3dt1(j);
512
513 for (int j = 0; j < 3; j++)
514 J_0ToK.at<double>(p, j) = de3dr2(j);
515
516 for (int j = 0; j < 3; j++)
517 J_0ToK.at<double>(p, 3 + j) = de3dt2(j);
518 }
519
520 // 6 x (ni*2) * (ni*2 x ni*2) * (ni*2) x 6
521 JtJ(Rect((k - 1) * 6, (k - 1) * 6, 6, 6)) += (J_0ToK.t() * wd * J_0ToK);
522 JtJ(Rect(eofs, (k - 1) * 6, 6, 6)) = (J_0ToK.t() * wd * Je);
523 JtErr.rowRange((k - 1) * 6, (k - 1) * 6 + 6) += (J_0ToK.t() * wd * err);
524 }
525 JtJ(Rect(eofs, eofs, 6, 6)) += Je.t() * wd * Je;
526 JtErr.rowRange(eofs, eofs + 6) += Je.t() * wd * err;
527 }
528 }
529 cnt_valid_frame++;
530 }
531 iters_lm += 1;
532 return true;
533 };
534 LevMarq solver(param, lmcallback,
535 LevMarq::Settings()
536 .setMaxIterations(termCrit.maxCount)
537 .setStepNormTolerance(termCrit.epsilon)
538 .setSmallEnergyTolerance(termCrit.epsilon * termCrit.epsilon),
539 noArray()/*mask, all variables to optimize*/);
540 cv::LevMarq::Report status = solver.optimize();
541 CV_LOG_INFO(NULL, "LevMarq finished with status " << status.found << " energy " << status.energy << " after " << status.iters << " iterations");
542}

545 const int NUM_CAMERAS = (int)detection_mask_mat.size(), NUM_FRAMES = (int)detection_mask_mat[0].size();
546 std::vector<bool> visited(NUM_CAMERAS, false);

548 dfs_search = [&] (int cam) {
549 visited[cam] = true;
550 for (int cam2 = 0; cam2 < NUM_CAMERAS; cam2++) {
551 if (!visited[cam2]) {

556 }

558 }
559 }
560 };

569 isolated_cameras += std::to_string(i);
570 } else {
571 if (visited_str != "")
572 visited_str += ", ";
573 visited_str += std::to_string(i);
574 }
575 }
576 CV_Error(Error::StsBadArg, "Isolated cameras (or components) "+isolated_cameras+" from the connected component "+visited_str+"!");
577 }
578 }

582double calibrateMultiview(
583 InputArrayOfArrays objPoints, const std::vector<std::vector<Mat>> &imagePoints,
584 const std::vector<cv::Size>& imageSize, InputArray detectionMask, InputArray models,
585 InputOutputArrayOfArrays Ks, InputOutputArrayOfArrays distortions,
586 InputOutputArrayOfArrays Rs, InputOutputArrayOfArrays Ts,
587 OutputArray initializationPairs, OutputArrayOfArrays rvecs0,
588 OutputArrayOfArrays tvecs0, OutputArray perFrameErrors,
589 InputArray flagsForIntrinsics, int flags, TermCriteria criteria) {
590 CV_CheckFalse(objPoints.empty(), "Objects points must not be empty!");
591 CV_CheckFalse(imagePoints.empty(), "Image points must not be empty!");
592 CV_CheckFalse(imageSize.empty(), "Image size per camera must not be empty!");
593 CV_CheckFalse(detectionMask.empty(), "detectionMask matrix must not be empty!");
594 CV_CheckFalse(models.empty(), "Fisheye mask must not be empty!");
595
596 Mat detection_mask_ = detectionMask.getMat();
597 Mat models_mat = models.getMat();
598
599 CV_Assert(detection_mask_.type() == CV_8UC1 || detection_mask_.type() == CV_BoolC1);
600 CV_CheckEQ(models_mat.type(), CV_8U, "models must be of type CV_8U");
601
602 if(flags & cv::CALIB_STEREO_REGISTRATION)
603 {
604 bool is_fisheye = false;
605 bool is_pinhole = false;
606
607 for (int i = 0; i < (int)models_mat.total(); i++) {
608 if (models_mat.at<uchar>(i) == cv::CALIB_MODEL_FISHEYE) {
609 is_fisheye = true;
610 } else if (models_mat.at<uchar>(i) == cv::CALIB_MODEL_PINHOLE) {
611 is_pinhole = true;
612 } else {
613 CV_Error(Error::StsBadArg, "Unsupported camera model");
614 }
615 }
616 CV_CheckEQ(is_fisheye && is_pinhole, false, "Mix of pinhole and fisheye cameras is not supported with CALIB_STEREO_REGISTRATION flag");
617 }
618
619 // equal number of cameras
620 CV_Assert(imageSize.size() == imagePoints.size());
621 CV_Assert(detection_mask_.rows == std::max(models.rows(), models.cols()));
622 CV_Assert(detection_mask_.rows == (int)imageSize.size());
623 CV_Assert(detection_mask_.cols == std::max(objPoints.rows(), objPoints.cols())); // equal number of frames
624 CV_Assert(Rs.isMatVector() == Ts.isMatVector());
625 if (flags & cv::CALIB_USE_INTRINSIC_GUESS) {
626 CV_Assert(Ks.isMatVector() && distortions.isMatVector());
627 CV_Assert(Ks.total() == distortions.total() && Ks.total() == imageSize.size());
628 }
629 if (flags & cv::CALIB_USE_EXTRINSIC_GUESS) {
630 CV_Assert(Rs.isMatVector() && Ts.isMatVector());
631 CV_Assert(Rs.total() == Ts.total() && Rs.total() == imageSize.size());
632 }
633 // normalize object points
634 const Mat obj_pts_0 = objPoints.getMat(0);
635 CV_Assert((obj_pts_0.type() == CV_32F && (obj_pts_0.rows == 3 || obj_pts_0.cols == 3)) ||
636 (obj_pts_0.type() == CV_32FC3 && (obj_pts_0.rows == 1 || obj_pts_0.cols == 1)));
637 const bool obj_points_in_rows = obj_pts_0.cols == 3;
638 const int NUM_CAMERAS = (int)detection_mask_.rows, NUM_FRAMES = (int)detection_mask_.cols;
639 CV_Assert((NUM_CAMERAS > 1) && (NUM_FRAMES > 0));
640 const int NUM_PATTERN_PTS = obj_points_in_rows ? obj_pts_0.rows : obj_pts_0.cols;
641 const double scale_3d_pts = multiview::getScaleOfObjPoints(NUM_PATTERN_PTS, obj_pts_0, obj_points_in_rows);
642
643 Mat flagsForIntrinsics_mat = flagsForIntrinsics.getMat();
644 if (flagsForIntrinsics_mat.empty())
645 {
646 flagsForIntrinsics_mat = Mat(Size(1, NUM_CAMERAS), CV_32SC1, cv::Scalar(0));
647 }
648
649 CV_Assert(flagsForIntrinsics_mat.total() == size_t(NUM_CAMERAS));
650 CV_CheckEQ(flagsForIntrinsics_mat.type(), CV_32S, "flagsForIntrinsics should be of type 32SC1");
651 CV_CheckEQ(flagsForIntrinsics_mat.channels(), 1, "flagsForIntrinsics should be of type 32SC1");
652
653 std::vector<Mat> objPoints_norm;
654 objPoints_norm.reserve(NUM_FRAMES);
655 for (int i = 0; i < NUM_FRAMES; i++) {
656 if (obj_points_in_rows)
657 objPoints_norm.emplace_back(objPoints.getMat(i)*(1/scale_3d_pts));
658 else
659 objPoints_norm.emplace_back(objPoints.getMat(i).t()*(1/scale_3d_pts));
660 objPoints_norm[i] = objPoints_norm[i].reshape(1);
661 }
662
664 std::vector<int> num_visible_frames_per_camera(NUM_CAMERAS);
665 std::vector<bool> valid_frames(NUM_FRAMES, false);
666 // process input and count all visible frames and points
667
668 std::vector<std::vector<bool>> detection_mask_mat(NUM_CAMERAS, std::vector<bool>(NUM_FRAMES));
669 const auto * const detection_mask_ptr = detection_mask_.data;
670 for (int c = 0; c < NUM_CAMERAS; c++) {
671 int num_visible_frames = 0;
672 for (int f = 0; f < NUM_FRAMES; f++) {
673 detection_mask_mat[c][f] = detection_mask_ptr[c*NUM_FRAMES + f] != 0;
674 if (detection_mask_mat[c][f]) {
675 num_visible_frames++;
676 valid_frames[f] = true; // if frame is visible by at least one camera then count is as a valid one
677 }
678 }
679 if (num_visible_frames == 0) {
680 CV_Error(Error::StsBadArg, "camera "+std::to_string(c)+" has no visible frames!");
681 }
682 num_visible_frames_per_camera[c] = num_visible_frames;
683 }
684
685 multiview::checkConnected(detection_mask_mat);
686
687 std::vector<std::vector<float>> points_ratio_area(NUM_CAMERAS, std::vector<float>(NUM_FRAMES));
688 multiview::imagePointsArea(imageSize, detection_mask_mat, imagePoints, points_ratio_area);
689
690 // constant threshold for angle between two camera axes in radians (=160*M_PI/180).
691 // if angle exceeds this threshold then a weight of a camera pair is lowered.
692 const double THR_PATTERN_CAMERA_ANGLES = 160*M_PI/180;
693 std::vector<std::vector<Vec3d>> rvecs_all(NUM_CAMERAS, std::vector<Vec3d>(NUM_FRAMES)),
694 tvecs_all(NUM_CAMERAS, std::vector<Vec3d>(NUM_FRAMES)),
695 opt_axes(NUM_CAMERAS, std::vector<Vec3d>(NUM_FRAMES));
696
697 std::vector<int> camera_rt_best(NUM_FRAMES, -1);
698 std::vector<double> camera_rt_errors(NUM_FRAMES, std::numeric_limits<double>::max());
699 const double WARNING_RMSE = 15.;
700 if ((flags & cv::CALIB_USE_INTRINSIC_GUESS) == 0) {
701 Ks.create(NUM_CAMERAS, 1, CV_64F);
702 distortions.create(NUM_CAMERAS, 1, CV_64F);
703
704 // calibrate each camera independently to find intrinsic parameters - K and distortion coefficients
705 parallel_for_(Range(0, NUM_CAMERAS), [&](const Range& range) {
706 for (int camera = range.start; camera < range.end; camera++) {
707 Mat K, dist;
708 Mat rvecs, tvecs;
709 std::vector<Mat> obj_points_, img_points_;
710 std::vector<double> errors_per_view;
711 obj_points_.reserve(num_visible_frames_per_camera[camera]);
712 img_points_.reserve(num_visible_frames_per_camera[camera]);
713 for (int f = 0; f < NUM_FRAMES; f++) {
714 if (detection_mask_mat[camera][f]) {
715 obj_points_.emplace_back((models_mat.at<uchar>(camera) == cv::CALIB_MODEL_FISHEYE && objPoints_norm[f].channels() != 3) ?
716 objPoints_norm[f].reshape(3): objPoints_norm[f]);
717 img_points_.emplace_back((models_mat.at<uchar>(camera) == cv::CALIB_MODEL_FISHEYE && imagePoints[camera][f].channels() != 2) ?
718 imagePoints[camera][f].reshape(2) : imagePoints[camera][f]);
719 }
720 }
721 double repr_err;
722 if (models_mat.at<uchar>(camera) == cv::CALIB_MODEL_FISHEYE) {
723 repr_err = fisheye::calibrate(obj_points_, img_points_, imageSize[camera],
724 K, dist, rvecs, tvecs, flagsForIntrinsics_mat.at<int>(camera));
725 CV_LOG_INFO(NULL, "Camera " << camera << " intrinsics calibration RMS " << repr_err);
726 // calibrate does not compute error per view, so compute it manually
727 errors_per_view = std::vector<double>(obj_points_.size());
728 for (int f = 0; f < (int) obj_points_.size(); f++) {
729 double err2 = multiview::computeReprojectionMSE(obj_points_[f],
730 img_points_[f], K, dist, rvecs.row(f), tvecs.row(f), noArray(), noArray(), cv::CALIB_MODEL_FISHEYE);
731 errors_per_view[f] = sqrt(err2);
732 }
733 } else {
734 repr_err = calibrateCamera(obj_points_, img_points_, imageSize[camera], K, dist,
735 rvecs, tvecs, noArray(), noArray(), errors_per_view, flagsForIntrinsics_mat.at<int>(camera));
736 CV_LOG_INFO(NULL, "Camera " << camera << " intrinsics calibration RMS " << repr_err);
737 }
738 CV_LOG_IF_WARNING(NULL, repr_err > WARNING_RMSE, "Warning! Mean RMSE of intrinsics calibration is higher than "+std::to_string(WARNING_RMSE)+" pixels!");
739 int cnt_visible_frame = 0;
740 for (int f = 0; f < NUM_FRAMES; f++) {
741 if (detection_mask_mat[camera][f]) {
742 rvecs_all[camera][f] = Vec3d(Mat(3, 1, CV_64F, rvecs.row(cnt_visible_frame).data));
743 tvecs_all[camera][f] = Vec3d(Mat(3, 1, CV_64F, tvecs.row(cnt_visible_frame).data));
744 double err = errors_per_view[cnt_visible_frame];
745 double err2 = err * err;
746 if (camera_rt_errors[f] > err2) {
747 camera_rt_errors[f] = err2;
748 camera_rt_best[f] = camera;
749 }
750 cnt_visible_frame++;
751 }
752 }
753
754 Ks.create(K.rows, K.cols, CV_64F, camera);
755 distortions.create(dist.rows, dist.cols, CV_64F, camera, true);
756 K.copyTo(Ks.getMat(camera));
757 dist.copyTo(distortions.getMat(camera));
758 }
759 });
760 } else {
761 // use PnP to compute rvecs and tvecs
762 for (int i = 0; i < NUM_FRAMES; i++) {
763 for (int k = 0; k < NUM_CAMERAS; k++) {
764 if (!detection_mask_mat[k][i]) continue;
765 Vec3d rvec, tvec;
766 solvePnP(objPoints_norm[i], imagePoints[k][i], Ks.getMat(k), distortions.getMat(k), rvec, tvec, false, SOLVEPNP_ITERATIVE);
767 rvecs_all[k][i] = rvec;
768 tvecs_all[k][i] = tvec;
769 const double err2 = multiview::computeReprojectionMSE(objPoints_norm[i], imagePoints[k][i], Ks.getMat(k), distortions.getMat(k), Mat(rvec), Mat(tvec), noArray(), noArray(), models_mat.at<uchar>(k));
770 if (camera_rt_errors[i] > err2) {
771 camera_rt_errors[i] = err2;
772 camera_rt_best[i] = k;
773 }
774 }
775 }
776 }
777
778 std::vector<cv::Mat> Ks_vec, distortions_vec;
779 Ks.getMatVector(Ks_vec);
780 distortions.getMatVector(distortions_vec);
781
782 std::vector<std::vector<bool>> is_valid_angle2pattern;
783 multiview::thresholdPatternCameraAngles(NUM_PATTERN_PTS, THR_PATTERN_CAMERA_ANGLES, objPoints_norm, rvecs_all, opt_axes, is_valid_angle2pattern);
784
785 std::vector<Matx33d> Rs_vec(NUM_CAMERAS);
786 std::vector<Vec3d> Ts_vec(NUM_CAMERAS);
787 Rs_vec[0] = Matx33d ::eye();
788 Ts_vec[0] = Vec3d::zeros();
789
790 std::vector<int> parent;
791 std::vector<std::vector<int>> overlaps;
792 if (! multiview::maximumSpanningTree(NUM_CAMERAS, NUM_FRAMES, detection_mask_mat, parent, overlaps, opt_axes,
793 is_valid_angle2pattern, points_ratio_area, .5, 1.0)) {
794 // failed to find suitable pairs with constraints!
795 CV_Error(Error::StsInternal, "Failed to build tree for stereo calibration.");
796 }
797
798 std::vector<std::pair<int,int>> pairs;
799 multiview::selectPairsBFS (pairs, NUM_CAMERAS, parent);
800
801 if ((int)pairs.size() != NUM_CAMERAS-1) {
802 CV_Error(Error::StsInternal, "Failed to build tree for stereo calibration. Incorrect number of pairs.");
803 }
804 if (initializationPairs.needed()) {
805 Mat pairs_mat = Mat_<int>(NUM_CAMERAS-1, 2);
806 auto * pairs_ptr = (int *) pairs_mat.data;
807 for (const auto &p : pairs) {
808 (*pairs_ptr++) = p.first;
809 (*pairs_ptr++) = p.second;
810 }
811 pairs_mat.copyTo(initializationPairs);
812 }
813
814 if(flags & cv::CALIB_STEREO_REGISTRATION) {
815 multiview::pairwiseStereoCalibration(pairs, models_mat, objPoints_norm, imagePoints,
816 overlaps, detection_mask_mat, Ks_vec, distortions_vec, Rs_vec, Ts_vec, flagsForIntrinsics_mat, 0, criteria);
817 } else {
818 multiview::pairwiseRegistration(pairs, models_mat, objPoints_norm, imagePoints,
819 overlaps, detection_mask_mat, Ks_vec, distortions_vec, Rs_vec, Ts_vec, flagsForIntrinsics_mat, 0, criteria);
820 }
821
822 const int NUM_VALID_FRAMES = countNonZero(valid_frames);
823 const int nparams = (NUM_VALID_FRAMES + NUM_CAMERAS - 1) * 6; // rvecs + tvecs (6)
824 std::vector<double> param(nparams, 0.);
825
826 // use found relative extrinsics to initialize parameters
827 for (int c = 1; c < NUM_CAMERAS; c++) {
828 Vec3d rvec;
829 Rodrigues(Rs_vec[c], rvec);
830 memcpy(&param[0]+(c-1)*6 , rvec.val, 3*sizeof(double));
831 memcpy(&param[0]+(c-1)*6+3, Ts_vec[c].val, 3*sizeof(double));
832 }
833
834 // use found rvecs / tvecs or estimate them to initialize rest of parameters
835 int cnt_valid_frame = 0;
836 for (int i = 0; i < NUM_FRAMES; i++ ) {
837 if (!valid_frames[i]) continue;
838 Vec3d rvec_0, tvec_0;
839 if (camera_rt_best[i] != 0) {
840 // convert rvecs / tvecs from k-th camera to the first one
841
842 // formulas for relative rotation / translation
843 // R = R_k R0^T => R_k = R R_0
844 // t = t_k - R t_0 => t_k = t + R t_0
845
846 // initial camera R_0 = I, t_0 = 0 is fixed to R(rvec_0) and tvec_0
847 // R_0 = R(rvec_0)
848 // t_0 = tvec_0
849
850 // R'_k = R(rvec_k) = R_k R_0 => R_0 = R_k^T R(rvec_k)
851 // t'_k = tvec_k = t_k + R_k t_0 => t_0 = R_k^T (tvec_k - t_k)
852 const int rt_best_idx = camera_rt_best[i];
853 Matx33d R_k;
854 Rodrigues(rvecs_all[rt_best_idx][i], R_k);
855 tvec_0 = Rs_vec[rt_best_idx].t() * (tvecs_all[rt_best_idx][i] - Ts_vec[rt_best_idx]);
856 Rodrigues(Rs_vec[rt_best_idx].t() * R_k, rvec_0);
857 } else {
858 rvec_0 = rvecs_all[0][i];
859 tvec_0 = tvecs_all[0][i];
860 }
861
862 // save rvecs0 / tvecs0 parameters
863 memcpy(&param[0]+(cnt_valid_frame+NUM_CAMERAS-1)*6 , rvec_0.val, 3*sizeof(double));
864 memcpy(&param[0]+(cnt_valid_frame+NUM_CAMERAS-1)*6+3, tvec_0.val, 3*sizeof(double));
865 cnt_valid_frame++;
866 }
867
868 const float RBS_FNC_SCALE = 30;
869 multiview::RobustExpFunction robust_fnc(RBS_FNC_SCALE);
870 multiview::optimizeLM(param, robust_fnc, criteria, valid_frames, detection_mask_mat, objPoints_norm,
871 imagePoints, Ks_vec, distortions_vec, models_mat, NUM_PATTERN_PTS);
872 const auto * const params = &param[0];
873
874 // extract extrinsics (R_i, t_i) for i = 1 ... NUM_CAMERAS:
875 const bool rt_mat_vec = Rs.isMatVector();
876 Mat rs, ts;
877 if (rt_mat_vec) {
878 Rs.create(NUM_CAMERAS, 1, CV_64F);
879 Ts.create(NUM_CAMERAS, 1, CV_64F);
880 } else {
881 rs = Mat_<double>(NUM_CAMERAS, 3);
882 ts = Mat_<double>(NUM_CAMERAS, 3);
883 }
884 for (int c = 0; c < NUM_CAMERAS; c++) {
885 Mat r_store, t_store;
886 if (rt_mat_vec) {
887 Rs.create(3, 1, CV_64F, c, true);
888 Ts.create(3, 1, CV_64F, c, true);
889 r_store = Rs.getMat(c);
890 t_store = Ts.getMat(c);
891 } else {
892 r_store = rs.row(c);
893 t_store = ts.row(c);
894 }
895 if (c == 0) {
896 memcpy(r_store.ptr(), Vec3d(0,0,0).val, 3*sizeof(double));
897 memcpy(t_store.ptr(), Vec3d(0,0,0).val, 3*sizeof(double));
898 } else {
899 memcpy(r_store.ptr(), params + (c-1)*6, 3*sizeof(double));
900 memcpy(t_store.ptr(), params + (c-1)*6+3, 3*sizeof(double)); // and de-normalize translation
901 t_store *= scale_3d_pts;
902 }
903 Mat R;
904 Rodrigues(r_store, R);
905 }
906 if (! rt_mat_vec) {
907 rs.copyTo(Rs);
908 ts.copyTo(Ts);
909 }
910 Mat rvecs0_, tvecs0_;
911 bool is_mat_vec = rvecs0.needed() && rvecs0.isMatVector();
912 if (is_mat_vec) {
913 rvecs0.create(NUM_FRAMES, 1, CV_64F);
914 } else {
915 rvecs0_ = Mat_<double>(NUM_FRAMES, 3);
916 }
917 cnt_valid_frame = 0;
918 for (int f = 0; f < NUM_FRAMES; f++) {
919 if (!valid_frames[f]) continue;
920 if (is_mat_vec)
921 rvecs0.create(3, 1, CV_64F, f, true);
922 Mat store = is_mat_vec ? rvecs0.getMat(f) : rvecs0_.row(f);
923 memcpy(store.ptr(), params + (cnt_valid_frame + NUM_CAMERAS - 1)*6, 3*sizeof(double));
924 cnt_valid_frame += 1;
925 }
926 if (!is_mat_vec && rvecs0.needed())
927 rvecs0_.copyTo(rvecs0);
928
929 is_mat_vec = tvecs0.needed() && tvecs0.isMatVector();
930 if (is_mat_vec) {
931 tvecs0.create(NUM_FRAMES, 1, CV_64F);
932 } else {
933 tvecs0_ = Mat_<double>(NUM_FRAMES, 3);
934 }
935 cnt_valid_frame = 0;
936 for (int f = 0; f < NUM_FRAMES; f++) {
937 if (!valid_frames[f]) continue;
938 if (is_mat_vec)
939 tvecs0.create(3, 1, CV_64F, f, true);
940 Mat store = is_mat_vec ? tvecs0.getMat(f) : tvecs0_.row(f);
941 memcpy(store.ptr(), params + (cnt_valid_frame + NUM_CAMERAS - 1)*6+3, 3*sizeof(double));
