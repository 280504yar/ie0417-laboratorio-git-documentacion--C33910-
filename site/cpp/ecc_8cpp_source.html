50static void image_jacobian_homo_ECC(const Mat& src1, const Mat& src2, const Mat& src3, const Mat& src4, const Mat& src5,
51 Mat& dst) {
52 CV_Assert(src1.size() == src2.size());
53 CV_Assert(src1.size() == src3.size());
54 CV_Assert(src1.size() == src4.size());
55
56 CV_Assert(src1.rows == dst.rows);
57 CV_Assert(dst.cols == (src1.cols * 8));
58 CV_Assert(dst.type() == CV_MAKETYPE(CV_32F, src1.channels()));
59
60 CV_Assert(src5.isContinuous());
61
62 const float* hptr = src5.ptr<float>(0);
63
64 const float h0_ = hptr[0];
65 const float h1_ = hptr[3];
66 const float h2_ = hptr[6];
67 const float h3_ = hptr[1];
68 const float h4_ = hptr[4];
69 const float h5_ = hptr[7];
70 const float h6_ = hptr[2];
71 const float h7_ = hptr[5];
72
73 const int w = src1.cols;
74
75 // create denominator for all points as a block
76 Mat den_;
77 addWeighted(src3, h2_, src4, h5_, 1.0, den_);
78
79 // create projected points
80 Mat hatX_, hatY_;
81 addWeighted(src3, h0_, src4, h3_, 0.0, hatX_);
82 hatX_ += h6_;
83
84 addWeighted(src3, h1_, src4, h4_, 0.0, hatY_);
85 hatY_ += h7_;
86
87 divide(-hatY_, den_, hatY_);
88 divide(-hatX_, den_, hatX_);
89
90 // instead of dividing each block with den,
91 // just pre-divide the block of gradients (it's more efficient)
92
93 Mat src1Divided_;
94 Mat src2Divided_;
95
96 divide(src1, den_, src1Divided_);
97 divide(src2, den_, src2Divided_);
98
99 // compute Jacobian blocks (8 blocks)
100
101 dst.colRange(0, w) = src1Divided_.mul(src3); // 1
102
103 dst.colRange(w, 2 * w) = src2Divided_.mul(src3); // 2
104
105 Mat temp_ = (hatX_.mul(src1Divided_) + hatY_.mul(src2Divided_));
106 dst.colRange(2 * w, 3 * w) = temp_.mul(src3); // 3
107
108 hatX_.release();
109 hatY_.release();
110
111 dst.colRange(3 * w, 4 * w) = src1Divided_.mul(src4); // 4
112
113 dst.colRange(4 * w, 5 * w) = src2Divided_.mul(src4); // 5
114
115 dst.colRange(5 * w, 6 * w) = temp_.mul(src4); // 6
116
117 src1Divided_.copyTo(dst.colRange(6 * w, 7 * w)); // 7
118
119 src2Divided_.copyTo(dst.colRange(7 * w, 8 * w)); // 8
120}

122static void image_jacobian_euclidean_ECC(const Mat& src1, const Mat& src2, const Mat& src3, const Mat& src4,
123 const Mat& src5, Mat& dst) {
124 CV_Assert(src1.size() == src2.size());
125 CV_Assert(src1.size() == src3.size());
126 CV_Assert(src1.size() == src4.size());
127
128 CV_Assert(src1.rows == dst.rows);
129 CV_Assert(dst.cols == (src1.cols * 3));
130 CV_Assert(dst.type() == CV_MAKETYPE(CV_32F, src1.channels()));
131
132 CV_Assert(src5.isContinuous());
133
134 const float* hptr = src5.ptr<float>(0);
135
136 const float h0 = hptr[0]; // cos(theta)
137 const float h1 = hptr[3]; // sin(theta)
138
139 const int w = src1.cols;
140
141 // create -sin(theta)*X -cos(theta)*Y for all points as a block -> hatX
142 Mat hatX = -(src3 * h1) - (src4 * h0);
143
144 // create cos(theta)*X -sin(theta)*Y for all points as a block -> hatY
145 Mat hatY = (src3 * h0) - (src4 * h1);
146
147 // compute Jacobian blocks (3 blocks)
148 dst.colRange(0, w) = (src1.mul(hatX)) + (src2.mul(hatY)); // 1
149
150 src1.copyTo(dst.colRange(w, 2 * w)); // 2
151 src2.copyTo(dst.colRange(2 * w, 3 * w)); // 3
152}

154static void image_jacobian_affine_ECC(const Mat& src1, const Mat& src2, const Mat& src3, const Mat& src4, Mat& dst) {
155 CV_Assert(src1.size() == src2.size());
156 CV_Assert(src1.size() == src3.size());
157 CV_Assert(src1.size() == src4.size());
158
159 CV_Assert(src1.rows == dst.rows);
160 CV_Assert(dst.cols == (6 * src1.cols));
161
162 CV_Assert(dst.type() == CV_MAKETYPE(CV_32F, src1.channels()));
163
164 const int w = src1.cols;
165
166 // compute Jacobian blocks (6 blocks)
167
168 dst.colRange(0, w) = src1.mul(src3); // 1
169 dst.colRange(w, 2 * w) = src2.mul(src3); // 2
170 dst.colRange(2 * w, 3 * w) = src1.mul(src4); // 3
171 dst.colRange(3 * w, 4 * w) = src2.mul(src4); // 4
172 src1.copyTo(dst.colRange(4 * w, 5 * w)); // 5
173 src2.copyTo(dst.colRange(5 * w, 6 * w)); // 6
174}

176static void image_jacobian_translation_ECC(const Mat& src1, const Mat& src2, Mat& dst) {
177 CV_Assert(src1.size() == src2.size());
178
179 CV_Assert(src1.rows == dst.rows);
180 CV_Assert(dst.cols == (src1.cols * 2));
181 CV_Assert(dst.type() == CV_MAKETYPE(CV_32F, src1.channels()));
182
183 const int w = src1.cols;
184
185 // compute Jacobian blocks (2 blocks)
186 src1.copyTo(dst.colRange(0, w));
187 src2.copyTo(dst.colRange(w, 2 * w));
188}

190static void project_onto_jacobian_ECC(const Mat& src1, const Mat& src2, Mat& dst) {
191 /* this functions is used for two types of projections. If src1.cols ==src.cols
192 it does a blockwise multiplication (like in the outer product of vectors)
193 of the blocks in matrices src1 and src2 and dst
194 has size (number_of_blcks x number_of_blocks), otherwise dst is a vector of size
195 (number_of_blocks x 1) since src2 is "multiplied"(dot) with each block of src1.
196
197 The number_of_blocks is equal to the number of parameters we are lloking for
198 (i.e. rtanslation:2, euclidean: 3, affine: 6, homography: 8)
199
200 */
201 CV_Assert(src1.rows == src2.rows);
202 CV_Assert((src1.cols % src2.cols) == 0);
203 int w;
204
205 float* dstPtr = dst.ptr<float>(0);
206
207 if (src1.cols != src2.cols) { // dst.cols==1
208 w = src2.cols;
209 for (int i = 0; i < dst.rows; i++) {
210 dstPtr[i] = (float)src2.dot(src1.colRange(i * w, (i + 1) * w));
211 }
212 }
213
214 else {
215 CV_Assert(dst.cols == dst.rows); // dst is square (and symmetric)
216 w = src2.cols / dst.cols;
217 Mat mat;
218 for (int i = 0; i < dst.rows; i++) {
219 mat = Mat(src1.colRange(i * w, (i + 1) * w));
220 dstPtr[i * (dst.rows + 1)] = (float)std::pow(norm(mat), 2); // diagonal elements
221
222 for (int j = i + 1; j < dst.cols; j++) { // j starts from i+1
223 dstPtr[i * dst.cols + j] = (float)mat.dot(src2.colRange(j * w, (j + 1) * w));
224 dstPtr[j * dst.cols + i] = dstPtr[i * dst.cols + j]; // due to symmetry
225 }
226 }
227 }

230static void update_warping_matrix_ECC(Mat& map_matrix, const Mat& update, const int motionType) {
231 CV_Assert(map_matrix.type() == CV_32FC1);
232 CV_Assert(update.type() == CV_32FC1);
233
234 CV_Assert(motionType == MOTION_TRANSLATION || motionType == MOTION_EUCLIDEAN || motionType == MOTION_AFFINE ||
235 motionType == MOTION_HOMOGRAPHY);
236
237 if (motionType == MOTION_HOMOGRAPHY)
238 CV_Assert(map_matrix.rows == 3 && update.rows == 8);
239 else if (motionType == MOTION_AFFINE)
240 CV_Assert(map_matrix.rows == 2 && update.rows == 6);
241 else if (motionType == MOTION_EUCLIDEAN)
242 CV_Assert(map_matrix.rows == 2 && update.rows == 3);
243 else
244 CV_Assert(map_matrix.rows == 2 && update.rows == 2);
245
246 CV_Assert(update.cols == 1);
247
248 CV_Assert(map_matrix.isContinuous());
249 CV_Assert(update.isContinuous());
250
251 float* mapPtr = map_matrix.ptr<float>(0);
252 const float* updatePtr = update.ptr<float>(0);
253
254 if (motionType == MOTION_TRANSLATION) {
255 mapPtr[2] += updatePtr[0];
256 mapPtr[5] += updatePtr[1];
257 }
258 if (motionType == MOTION_AFFINE) {
259 mapPtr[0] += updatePtr[0];
260 mapPtr[3] += updatePtr[1];
261 mapPtr[1] += updatePtr[2];
262 mapPtr[4] += updatePtr[3];
263 mapPtr[2] += updatePtr[4];
264 mapPtr[5] += updatePtr[5];
265 }
266 if (motionType == MOTION_HOMOGRAPHY) {
267 mapPtr[0] += updatePtr[0];
268 mapPtr[3] += updatePtr[1];
269 mapPtr[6] += updatePtr[2];
270 mapPtr[1] += updatePtr[3];
271 mapPtr[4] += updatePtr[4];
272 mapPtr[7] += updatePtr[5];
273 mapPtr[2] += updatePtr[6];
274 mapPtr[5] += updatePtr[7];
275 }
276 if (motionType == MOTION_EUCLIDEAN) {
277 double new_theta = updatePtr[0];
278 new_theta += asin(mapPtr[3]);
279
280 mapPtr[2] += updatePtr[1];
281 mapPtr[5] += updatePtr[2];
282 mapPtr[0] = mapPtr[4] = (float)cos(new_theta);
283 mapPtr[3] = (float)sin(new_theta);
284 mapPtr[1] = -mapPtr[3];
285 }

291double cv::computeECC(InputArray templateImage, InputArray inputImage, InputArray inputMask) {
292 CV_Assert(!templateImage.empty());
293 CV_Assert(!inputImage.empty());
294
295 CV_Assert(templateImage.channels() == 1 || templateImage.channels() == 3);
296
297 if (!(templateImage.type() == inputImage.type()))
298 CV_Error(Error::StsUnmatchedFormats, "Both input images must have the same data type");
299
300 Scalar meanTemplate, sdTemplate;
301
302 int active_pixels = inputMask.empty() ? templateImage.size().area() : countNonZero(inputMask);
303 int type = templateImage.type();
304 meanStdDev(templateImage, meanTemplate, sdTemplate, inputMask);
305 Mat templateImage_zeromean = Mat::zeros(templateImage.size(), templateImage.type());
306 Mat templateMat = templateImage.getMat();
307 Mat inputMat = inputImage.getMat();
308
309 /*
310 * For unsigned ints, when the mean is computed and subtracted, any values less than the mean
311 * will be set to 0 (since there are no negatives values). This impacts the norm and dot product, which
312 * ultimately results in an incorrect ECC. To circumvent this problem, if unsigned ints are provided,
313 * we convert them to a signed ints with larger resolution for the subtraction step.
314 */
315 if (type == CV_8U || type == CV_16U) {
316 int newType = type == CV_8U ? CV_16S : CV_32S;
317 Mat templateMatConverted, inputMatConverted;
318 templateMat.convertTo(templateMatConverted, newType);
319 cv::swap(templateMat, templateMatConverted);
320 inputMat.convertTo(inputMatConverted, newType);
321 cv::swap(inputMat, inputMatConverted);
322 }
323 subtract(templateMat, meanTemplate, templateImage_zeromean, inputMask);
324 double templateImagenorm = std::sqrt(active_pixels * cv::norm(sdTemplate, NORM_L2SQR));
325
326 Scalar meanInput, sdInput;
327
328 Mat inputImage_zeromean = Mat::zeros(inputImage.size(), inputImage.type());
329 meanStdDev(inputImage, meanInput, sdInput, inputMask);
330 subtract(inputMat, meanInput, inputImage_zeromean, inputMask);
331 double inputImagenorm = std::sqrt(active_pixels * norm(sdInput, NORM_L2SQR));
332
333 return templateImage_zeromean.dot(inputImage_zeromean) / (templateImagenorm * inputImagenorm);
334}

337double cv::findTransformECCWithMask( InputArray templateImage,
338 InputArray inputImage,
339 InputArray templateMask,
340 InputArray inputMask,
341 InputOutputArray warpMatrix,
342 int motionType,
343 TermCriteria criteria,
344 int gaussFiltSize) {
345 Mat src = templateImage.getMat(); // template image
346 Mat dst = inputImage.getMat(); // input image (to be warped)
347 Mat map = warpMatrix.getMat(); // warp (transformation)
348
349 CV_Assert(!src.empty());
350 CV_Assert(!dst.empty());
351
352 CV_Assert(src.channels() == 1 || src.channels() == 3);
353 CV_Assert(src.channels() == dst.channels());
354 CV_Assert(src.depth() == dst.depth());
355 CV_Assert(src.depth() == CV_8U || src.depth() == CV_16U || src.depth() == CV_32F || src.depth() == CV_64F);
356
357 // If the user passed an un-initialized warpMatrix, initialize to identity
358 if (map.empty()) {
359 int rowCount = 2;
360 if (motionType == MOTION_HOMOGRAPHY)
361 rowCount = 3;
362
363 warpMatrix.create(rowCount, 3, CV_32FC1);
364 map = warpMatrix.getMat();
365 map = Mat::eye(rowCount, 3, CV_32F);
366 }
367
368 if (!(src.type() == dst.type()))
369 CV_Error(Error::StsUnmatchedFormats, "Both input images must have the same data type");
370
371 if (map.type() != CV_32FC1)
372 CV_Error(Error::StsUnsupportedFormat, "warpMatrix must be single-channel floating-point matrix");
373
374 CV_Assert(map.cols == 3);
375 CV_Assert(map.rows == 2 || map.rows == 3);
376
377 CV_Assert(motionType == MOTION_AFFINE || motionType == MOTION_HOMOGRAPHY || motionType == MOTION_EUCLIDEAN ||
378 motionType == MOTION_TRANSLATION);
379
380 if (motionType == MOTION_HOMOGRAPHY) {
381 CV_Assert(map.rows == 3);
382 }
383
384 CV_Assert(criteria.type & TermCriteria::COUNT || criteria.type & TermCriteria::EPS);
385 const int numberOfIterations = (criteria.type & TermCriteria::COUNT) ? criteria.maxCount : 200;
386 const double termination_eps = (criteria.type & TermCriteria::EPS) ? criteria.epsilon : -1;
387
388 int paramTemp = 6; // default: affine
389 switch (motionType) {
390 case MOTION_TRANSLATION:
391 paramTemp = 2;
392 break;
393 case MOTION_EUCLIDEAN:
394 paramTemp = 3;
395 break;
396 case MOTION_HOMOGRAPHY:
397 paramTemp = 8;
398 break;
399 }
400
401 const int numberOfParameters = paramTemp;
402
403 const int ws = src.cols;
404 const int hs = src.rows;
405 const int wd = dst.cols;
406 const int hd = dst.rows;
407
408 Mat Xcoord = Mat(1, ws, CV_32F);
409 Mat Ycoord = Mat(hs, 1, CV_32F);
410 Mat Xgrid = Mat(hs, ws, CV_32F);
411 Mat Ygrid = Mat(hs, ws, CV_32F);
412
413 float* XcoPtr = Xcoord.ptr<float>(0);
414 float* YcoPtr = Ycoord.ptr<float>(0);
415 int j;
416 for (j = 0; j < ws; j++) XcoPtr[j] = (float)j;
417 for (j = 0; j < hs; j++) YcoPtr[j] = (float)j;
418
419 repeat(Xcoord, hs, 1, Xgrid);
420 repeat(Ycoord, 1, ws, Ygrid);
421
422 Xcoord.release();
423 Ycoord.release();
424
425 const int channels = src.channels();
426 int type = CV_MAKETYPE(CV_32F, channels);
427
428 std::vector<cv::Mat> XgridCh(channels, Xgrid);
429 cv::merge(XgridCh, Xgrid);
430
431 std::vector<cv::Mat> YgridCh(channels, Ygrid);
432 cv::merge(YgridCh, Ygrid);
433
434 Mat templateZM = Mat(hs, ws, type); // to store the (smoothed)zero-mean version of template
435 Mat templateFloat = Mat(hs, ws, type); // to store the (smoothed) template
436 Mat imageFloat = Mat(hd, wd, type); // to store the (smoothed) input image
437 Mat imageWarped = Mat(hs, ws, type); // to store the warped zero-mean input image
438 Mat imageMask = Mat(hs, ws, CV_8U); // to store the final mask
439
440 // Gaussian filtering is optional
441 src.convertTo(templateFloat, templateFloat.type());
442 GaussianBlur(templateFloat, templateFloat, Size(gaussFiltSize, gaussFiltSize), 0, 0);
443
444 dst.convertTo(imageFloat, imageFloat.type());
445 GaussianBlur(imageFloat, imageFloat, Size(gaussFiltSize, gaussFiltSize), 0, 0);
446
447 // needed matrices for gradients and warped gradients
448 Mat gradientX = Mat::zeros(hd, wd, type);
449 Mat gradientY = Mat::zeros(hd, wd, type);
450 Mat gradientXWarped = Mat(hs, ws, type);
451 Mat gradientYWarped = Mat(hs, ws, type);
452
453 // calculate first order image derivatives
454 Matx13f dx(-0.5f, 0.0f, 0.5f);
455
456 filter2D(imageFloat, gradientX, -1, dx);
457 filter2D(imageFloat, gradientY, -1, dx.t());
458
459 // To use in mask warping
460 Mat templtMask;
461 if(templateMask.empty())
462 {
463 templtMask = Mat::ones(hs, ws, CV_8U);
464 }
465 else
466 {
467 threshold(templateMask, templtMask, 0, 1, THRESH_BINARY);
468 templtMask.convertTo(templtMask, CV_32F);
469 GaussianBlur(templtMask, templtMask, Size(gaussFiltSize, gaussFiltSize), 0, 0);
470 templtMask *= (0.5/0.95);
471 templtMask.convertTo(templtMask, CV_8U);
472 }
473
474 //to use it for mask warping
475 Mat preMask;
476 if(inputMask.empty())
477 {
478 preMask = Mat::ones(hd, wd, CV_8U);
479 }
480 else
481 {
482 Mat preMaskFloat;
483 Mat inputMaskMat = inputMask.getMat();
484 if (inputMaskMat.depth() == CV_Bool)
485 {
486 inputMaskMat.convertTo(preMaskFloat, CV_32F);
487 }
488 else
489 {
490 threshold(inputMaskMat, preMask, 0, 1, THRESH_BINARY);
491 preMask.convertTo(preMaskFloat, CV_32F);
492 }
493 GaussianBlur(preMaskFloat, preMaskFloat, Size(gaussFiltSize, gaussFiltSize), 0, 0);
494 // Change threshold.
495 preMaskFloat *= (0.5/0.95);
496 // Rounding conversion.
497 preMaskFloat.convertTo(preMask, CV_8U);
498
499 // If there's no template mask, we can apply image masks to gradients only once.
500 // Otherwise, we'll need to combine the template and image masks at each iteration.
501 if (templateMask.empty())
502 {
503 cv::Mat zeroMask = (preMask == 0);
504 gradientX.setTo(0, zeroMask);
505 gradientY.setTo(0, zeroMask);
506 }
507 }
508
509 // matrices needed for solving linear equation system for maximizing ECC
510 Mat jacobian = Mat(hs, ws * numberOfParameters, type);
511 Mat hessian = Mat(numberOfParameters, numberOfParameters, CV_32F);
512 Mat hessianInv = Mat(numberOfParameters, numberOfParameters, CV_32F);
513 Mat imageProjection = Mat(numberOfParameters, 1, CV_32F);
514 Mat templateProjection = Mat(numberOfParameters, 1, CV_32F);
515 Mat imageProjectionHessian = Mat(numberOfParameters, 1, CV_32F);
516 Mat errorProjection = Mat(numberOfParameters, 1, CV_32F);
517
518 Mat deltaP = Mat(numberOfParameters, 1, CV_32F); // transformation parameter correction
519 Mat error = Mat(hs, ws, CV_32F); // error as 2D matrix
520
521 const int imageFlags = INTER_LINEAR + WARP_INVERSE_MAP;
522 const int maskFlags = INTER_NEAREST + WARP_INVERSE_MAP;
523
524 // iteratively update map_matrix
525 double rho = -1;
526 double last_rho = -termination_eps;
527 for (int i = 1; (i <= numberOfIterations) && (fabs(rho - last_rho) >= termination_eps); i++) {
528 // warp-back portion of the inputImage and gradients to the coordinate space of the templateImage
529 if (motionType != MOTION_HOMOGRAPHY) {
530 warpAffine(imageFloat, imageWarped, map, imageWarped.size(), imageFlags);
531 warpAffine(gradientX, gradientXWarped, map, gradientXWarped.size(), imageFlags);
532 warpAffine(gradientY, gradientYWarped, map, gradientYWarped.size(), imageFlags);
533 warpAffine(preMask, imageMask, map, imageMask.size(), maskFlags);
534 } else {
535 warpPerspective(imageFloat, imageWarped, map, imageWarped.size(), imageFlags);
536 warpPerspective(gradientX, gradientXWarped, map, gradientXWarped.size(), imageFlags);
537 warpPerspective(gradientY, gradientYWarped, map, gradientYWarped.size(), imageFlags);
538 warpPerspective(preMask, imageMask, map, imageMask.size(), maskFlags);
539 }
540
541 if (!templateMask.empty())
542 {
543 cv::bitwise_and(imageMask, templtMask, imageMask);
544
545 cv::Mat zeroMask = (imageMask == 0);
546 gradientXWarped.setTo(0, zeroMask);
547 gradientYWarped.setTo(0, zeroMask);
548 }
549
550 Scalar imgMean, imgStd, tmpMean, tmpStd;
551 meanStdDev(imageWarped, imgMean, imgStd, imageMask);
552 meanStdDev(templateFloat, tmpMean, tmpStd, imageMask);
553
554 subtract(imageWarped, imgMean, imageWarped, imageMask); // zero-mean input
555 templateZM = Mat::zeros(templateZM.rows, templateZM.cols, templateZM.type());
556 subtract(templateFloat, tmpMean, templateZM, imageMask); // zero-mean template
557
558 int validPixels = countNonZero(imageMask);
559 double tmpNorm = std::sqrt(validPixels * cv::norm(tmpStd, cv::NORM_L2SQR));
560 double imgNorm = std::sqrt(validPixels * cv::norm(imgStd, cv::NORM_L2SQR));
561
562 // calculate jacobian of image wrt parameters
563 switch (motionType) {
564 case MOTION_AFFINE:
565 image_jacobian_affine_ECC(gradientXWarped, gradientYWarped, Xgrid, Ygrid, jacobian);
566 break;
567 case MOTION_HOMOGRAPHY:
568 image_jacobian_homo_ECC(gradientXWarped, gradientYWarped, Xgrid, Ygrid, map, jacobian);
569 break;
570 case MOTION_TRANSLATION:
571 image_jacobian_translation_ECC(gradientXWarped, gradientYWarped, jacobian);
572 break;
573 case MOTION_EUCLIDEAN:
574 image_jacobian_euclidean_ECC(gradientXWarped, gradientYWarped, Xgrid, Ygrid, map, jacobian);
575 break;
576 }
577
578 // calculate Hessian and its inverse
579 project_onto_jacobian_ECC(jacobian, jacobian, hessian);
580
581 hessianInv = hessian.inv();
582
583 const double correlation = templateZM.dot(imageWarped);
584
585 // calculate enhanced correlation coefficient (ECC)->rho
586 last_rho = rho;
587 rho = correlation / (imgNorm * tmpNorm);
588 if (cvIsNaN(rho)) {
589 CV_Error(Error::StsNoConv, "NaN encountered.");
590 }
591
592 // project images into jacobian
593 project_onto_jacobian_ECC(jacobian, imageWarped, imageProjection);
594 project_onto_jacobian_ECC(jacobian, templateZM, templateProjection);
595
596 // calculate the parameter lambda to account for illumination variation
597 imageProjectionHessian = hessianInv * imageProjection;
598 const double lambda_n = (imgNorm * imgNorm) - imageProjection.dot(imageProjectionHessian);
599 const double lambda_d = correlation - templateProjection.dot(imageProjectionHessian);
600 if (lambda_d <= 0.0) {
601 rho = -1;
602 CV_Error(Error::StsNoConv,
603 "The algorithm stopped before its convergence. The correlation is going to be minimized. Images "
604 "may be uncorrelated or non-overlapped");
605 }
606 const double lambda = (lambda_n / lambda_d);
607
608 // estimate the update step delta_p
609 error = lambda * templateZM - imageWarped;
610 project_onto_jacobian_ECC(jacobian, error, errorProjection);
611 deltaP = hessianInv * errorProjection;
612
613 // update warping matrix
614 update_warping_matrix_ECC(map, deltaP, motionType);
615 }
616
617 // return final correlation coefficient
618 return rho;
619}
