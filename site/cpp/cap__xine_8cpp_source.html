108 bool open(const char *filename)
109 {
110 CV_Assert_N(!xine, !stream, !vo_port);
111 char configfile[2048] = {0};
112
113 xine = xine_new();
114 snprintf(configfile, sizeof(configfile), "%s%s", xine_get_homedir(), "/.xine/config");
115 xine_config_load(xine, configfile);
116 xine_init(xine);
117 xine_engine_set_param(xine, 0, 0);
118
119 vo_port = xine_new_framegrab_video_port(xine);
120 if (!vo_port)
121 return false;
122
123 stream = xine_stream_new(xine, NULL, vo_port);
124 if (!xine_open(stream, filename))
125 return false;
126
127 // reset stream...
128 if (!xine_play(stream, 0, 0))
129 return false;
130
131 // initialize some internals...
132 frame_number = 0;
133
134
135 if ( !xine_get_next_video_frame( vo_port, &xine_frame ) )
136 return false;
137
138 size = Size( xine_frame.width, xine_frame.height );
139
140 xine_free_video_frame( vo_port, &xine_frame );
141 xine_frame.data = 0;
142
143 {
144 xine_video_frame_t tmp;
145 if (!xine_play( stream, 0, 300 )) /* 300msec */
146 return false;
147 if (!xine_get_next_video_frame( vo_port, &tmp ))
148 return false;
149 seekable = ( tmp.frame_number != 0 );
150 xine_free_video_frame( vo_port, &tmp );
151 if (!xine_play( stream, 0, 0 ))
152 return false;
153 }
154
155 frame_duration = xine_get_stream_info( stream, XINE_STREAM_INFO_FRAME_DURATION ) / 90.;
156 frame_rate = frame_duration > 0 ? 1000 / frame_duration : 0.;
157 return true;
158 }

169 bool retrieveFrame(int, OutputArray out) CV_OVERRIDE
170 {
171 CV_Assert(stream);
172 CV_Assert(vo_port);
173
174 if (xine_frame.data == 0)
175 return false;
176
177 bool res = false;
178 Mat frame_bgr;
179
180 switch (xine_frame.colorspace)
181 {
182 case XINE_IMGFMT_YV12: // actual format seems to be I420 (or IYUV)
183 {
184 Mat frame(Size(xine_frame.width, xine_frame.height * 3 / 2), CV_8UC1, xine_frame.data);
185 cv::cvtColor(frame, out, cv::COLOR_YUV2BGR_I420);
186 res = true;
187 }
188 break;
189
190 case XINE_IMGFMT_YUY2:
191 {
192 Mat frame(Size(xine_frame.width, xine_frame.height), CV_8UC2, xine_frame.data);
193 cv::cvtColor(frame, out, cv::COLOR_YUV2BGR_YUY2);
194 res = true;
195 }
196 break;
197
198 default:
199 break;
200 }
201
202 // always release last xine_frame, not needed anymore
203 xine_free_video_frame(vo_port, &xine_frame);
204 xine_frame.data = 0;
205 return res;
206 }

208 double getProperty(int property_id) const CV_OVERRIDE
209 {
210 CV_Assert_N(xine, vo_port, stream);
211 int pos_t, pos_l, length;
212 bool res = (bool)xine_get_pos_length(stream, &pos_l, &pos_t, &length);
213
214 switch (property_id)
215 {
216 case CAP_PROP_POS_MSEC: return res ? pos_t : CAP_PROP_UNKNOWN;
217 case CAP_PROP_POS_FRAMES: return frame_number;
218 case CAP_PROP_POS_AVI_RATIO: return length && res ? pos_l / 65535.0 : CAP_PROP_UNKNOWN;
219 case CAP_PROP_FRAME_WIDTH: return size.width;
220 case CAP_PROP_FRAME_HEIGHT: return size.height;
221 case CAP_PROP_FPS: return frame_rate;
222 case CAP_PROP_FOURCC: return (double)xine_get_stream_info(stream, XINE_STREAM_INFO_VIDEO_FOURCC);
223 }
224 return CAP_PROP_UNKNOWN;
225 }

240protected:
241 bool oldSeekFrame(int f)
242 {
243 CV_Assert_N(xine, vo_port, stream);
244 // no need to seek if we are already there...
245 if (f == frame_number)
246 {
247 return true;
248 }
249 else if (f > frame_number)
250 {
251 // if the requested position is behind out actual position,
252 // we just need to read the remaining amount of frames until we are there.
253 for (; frame_number < f; frame_number++)
254 {
255 // un-increment framenumber grabbing failed
256 if (!xine_get_next_video_frame(vo_port, &xine_frame))
257 {
258 frame_number--;
259 break;
260 }
261 else
262 {
263 xine_free_video_frame(vo_port, &xine_frame);
264 }
265 }
266 }
267 else // f < frame_number
268 {
269 // otherwise we need to reset the stream and
270 // start reading frames from the beginning.
271 // reset stream, should also work with non-seekable input
272 xine_play(stream, 0, 0);
273 // read frames until we are at the requested frame
274 for (frame_number = 0; frame_number < f; frame_number++)
275 {
276 // un-increment last framenumber if grabbing failed
277 if (!xine_get_next_video_frame(vo_port, &xine_frame))
278 {
279 frame_number--;
280 break;
281 }
282 else
283 {
284 xine_free_video_frame(vo_port, &xine_frame);
285 }
286 }
287 }
288 return f == frame_number;
289 }

291 bool seekFrame(int f)
292 {
293 CV_Assert_N(xine, vo_port, stream);
294 if (seekable)
295 {
296 int new_time = (int)((f + 1) * (float)frame_duration);
297 if (xine_play(stream, 0, new_time))
298 {
299 frame_number = f;
300 return true;
301 }
302 }
303 else
304 {
305 return oldSeekFrame(f);
306 }
307 return false;
308 }
309
310 bool seekTime(int t)
311 {
312 CV_Assert_N(xine, vo_port, stream);
313 if (seekable)
314 {
315 if (xine_play(stream, 0, t))
316 {
317 frame_number = (int)((double)t * frame_rate / 1000);
318 return true;
319 }
320 }
321 else
322 {
323 int new_frame = (int)((double)t * frame_rate / 1000);
324 return oldSeekFrame(new_frame);
325 }
326 return false;
327 }
328
329 bool seekRatio(double ratio)
330 {
331 CV_Assert_N(xine, vo_port, stream);
332 if (ratio > 1 || ratio < 0)
333 return false;
334 if (seekable)
335 {
336 // TODO: FIX IT, DOESN'T WORK PROPERLY, YET...!

338 bool res = (bool)xine_get_pos_length(stream, &pos_l, &pos_t, &length);
339 if (res && xine_play(stream, (int)(ratio * (double)length), 0))
340 {
341 frame_number = (int)(ratio * length / frame_duration);
342 return true;
343 }
344 }
345 return false;
346 }
