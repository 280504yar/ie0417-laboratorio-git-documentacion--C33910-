52/*
53 Multiply-with-carry generator is used here:
54 temp = ( A*X(n) + carry )
55 X(n+1) = temp mod (2^32)
56 carry = floor (temp / (2^32))
57*/
58
59#define RNG_NEXT(x) ((uint64)(unsigned)(x)*CV_RNG_COEFF + ((x) >> 32))
60// make it jump-less
61#define CN_NEXT(k) (((k) + 1) & (((k) >= cn) - 1))

73template<typename T> static void
74randBits_( T* arr, int len, int cn, uint64* state, const Vec2l* p, int flags )
75{
76 bool small_flag = (flags & RNG_FLAG_SMALL) != 0;
77 uint64 temp = *state;
78 int i, k = 0;
79 len *= cn;
80 --cn;
81
82 if( !small_flag )
83 {
84 for( i = 0; i <= len - 4; i += 4 )
85 {
86 int64_t t0, t1;
87
88 temp = RNG_NEXT(temp);
89 t0 = ((int64_t)temp & p[k][0]) + p[k][1];
90 k = CN_NEXT(k);
91 temp = RNG_NEXT(temp);
92 t1 = ((int64_t)temp & p[k][0]) + p[k][1];
93 k = CN_NEXT(k);
94 arr[i] = saturate_cast<T>(t0);
95 arr[i+1] = saturate_cast<T>(t1);
96
97 temp = RNG_NEXT(temp);
98 t0 = ((int64_t)temp & p[k][0]) + p[k][1];
99 k = CN_NEXT(k);
100 temp = RNG_NEXT(temp);
101 t1 = ((int64_t)temp & p[k][0]) + p[k][1];
102 k = CN_NEXT(k);
103 arr[i+2] = saturate_cast<T>(t0);
104 arr[i+3] = saturate_cast<T>(t1);
105 }
106 }
107 else
108 {
109 for( i = 0; i <= len - 4; i += 4 )
110 {
111 int64_t t0, t1, t;
112 temp = RNG_NEXT(temp);
113 t = temp;
114 // p[i+...][0] is within 0..255 in this branch (small_flag==true),
115 // so we don't need to do (t>>...)&255,
116 // the upper bits will be cleaned with ... & p[i+...][0].
117 t0 = (t & p[k][0]) + p[k][1];
118 k = CN_NEXT(k);
119 t1 = ((t >> 8) & p[k][0]) + p[k][1];
120 k = CN_NEXT(k);
121 arr[i] = saturate_cast<T>(t0);
122 arr[i+1] = saturate_cast<T>(t1);
123
124 t0 = ((t >> 16) & p[k][0]) + p[k][1];
125 k = CN_NEXT(k);
126 t1 = ((t >> 24) & p[k][0]) + p[k][1];
127 k = CN_NEXT(k);
128 arr[i+2] = saturate_cast<T>(t0);
129 arr[i+3] = saturate_cast<T>(t1);
130 }
131 }
132
133 for( ; i < len; i++ )
134 {
135 int64_t t0;
136 temp = RNG_NEXT(temp);
137
138 t0 = ((int64_t)temp & p[k][0]) + p[k][1];
139 k = CN_NEXT(k);
140 arr[i] = saturate_cast<T>(t0);
141 }
142
143 *state = temp;
144}

146struct DivStruct
147{
148 unsigned d;
149 unsigned M;
150 int sh1, sh2;
151 int64_t delta;
152 uint64_t diff;
153};

155template<typename T> static void
156randi_( T* arr, int len, int cn, uint64* state, const DivStruct* p )
157{
158 uint64 temp = *state;
159 int k = 0;
160 len *= cn;
161 cn--;
162 for( int i = 0; i < len; i++ )
163 {
164 temp = RNG_NEXT(temp);
165 unsigned t = (unsigned)temp;
166 unsigned v = (unsigned)(((uint64)t * p[k].M) >> 32);
167 v = (v + ((t - v) >> p[k].sh1)) >> p[k].sh2;
168 int64_t res = (int64_t)(t - v*p[k].d) + p[k].delta;
169 k = CN_NEXT(k);
170 arr[i] = saturate_cast<T>(res);
171 }
172 *state = temp;
173}

176randi_( int64_t* arr, int len, int cn, uint64* state, const DivStruct* p )
177{
178 uint64 temp = *state;

181 cn--;

195static void
196randi_( uint64_t* arr, int len, int cn, uint64* state, const DivStruct* p )
197{
198 uint64 temp = *state;

244{
245 int depth = CV_MAT_DEPTH(flags);
246 uint64 temp = *state;
247 int k = 0, len = len_*cn;

262 else if (isFP8Depth(depth))
263 cvt32fToFP8(fbuf, dst, len, depth);

304/*
305 The code below implements the algorithm described in
306 "The Ziggurat Method for Generating Random Variables"
307 by George Marsaglia and Wai Wan Tsang, Journal of Statistical Software, 2007.
308*/
309static void
310randn_0_1_32f( float* arr, int len, uint64* state )
311{
312 const float r = 3.442620f; // The start of the right tail
313 const float rng_flt = 2.3283064365386962890625e-10f; // 2^-32
314 static unsigned kn[128];
315 static float wn[128], fn[128];
316 uint64 temp = *state;
317 static bool initialized=false;
318 int i;
319
320 if( !initialized )
321 {
322 const double m1 = 2147483648.0;
323 double dn = 3.442619855899, tn = dn, vn = 9.91256303526217e-3;
324
325 // Set up the tables
326 double q = vn/std::exp(-.5*dn*dn);
327 kn[0] = (unsigned)((dn/q)*m1);
328 kn[1] = 0;
329
330 wn[0] = (float)(q/m1);
331 wn[127] = (float)(dn/m1);
332
333 fn[0] = 1.f;
334 fn[127] = (float)std::exp(-.5*dn*dn);
335
336 for(i=126;i>=1;i--)
337 {
338 dn = std::sqrt(-2.*std::log(vn/dn+std::exp(-.5*dn*dn)));
339 kn[i+1] = (unsigned)((dn/tn)*m1);
340 tn = dn;
341 fn[i] = (float)std::exp(-.5*dn*dn);
342 wn[i] = (float)(dn/m1);
343 }
344 initialized = true;
345 }
346
347 for( i = 0; i < len; i++ )
348 {
349 float x, y;
350 for(;;)
351 {
352 int hz = (int)temp;
353 temp = RNG_NEXT(temp);
354 int iz = hz & 127;
355 x = hz*wn[iz];
356 if( (unsigned)std::abs(hz) < kn[iz] )
357 break;
358 if( iz == 0) // iz==0, handles the base strip
359 {
360 do
361 {
362 x = (unsigned)temp*rng_flt;
363 temp = RNG_NEXT(temp);
364 y = (unsigned)temp*rng_flt;
365 temp = RNG_NEXT(temp);
366 x = (float)(-std::log(x+FLT_MIN)*0.2904764);
367 y = (float)-std::log(y+FLT_MIN);
368 } // .2904764 is 1/r
369 while( y + y < x*x );
370 x = hz > 0 ? r + x : -r - x;
371 break;
372 }
373 // iz > 0, handle the wedges of other strips
374 y = (unsigned)temp*rng_flt;
375 temp = RNG_NEXT(temp);
376 if( fn[iz] + y*(fn[iz - 1] - fn[iz]) < std::exp(-.5*x*x) )
377 break;
378 }
379 arr[i] = x;
380 }
381 *state = temp;
382}

392template<typename T, typename PT> static void
393randnScale_(float* src, T* dst, int len, int cn,
394 const PT* mean, const PT* stddev, int flags )
395{
396 bool stdmtx = (flags & RNG_FLAG_STDMTX) != 0;
397 int i, j, k;
398 if( !stdmtx || cn == 1 )
399 {
400 if( cn == 1 )
401 {
402 PT a = stddev[0], b = mean[0];
403 for( i = 0; i < len; i++ )
404 dst[i] = saturate_cast<T>(src[i]*a + b);
405 }
406 else
407 {
408 len *= cn;
409 cn--;
410 for( i = k = 0; i < len; i++ ) {
411 dst[i] = saturate_cast<T>(src[i]*stddev[k] + mean[k]);
412 k = CN_NEXT(k);
413 }
414 }
415 }
416 else
417 {
418 len *= cn;
419 cn--;
420 for( i = j = 0; i < len; i++ )
421 {
422 PT s = mean[j];
423 int i0 = i - j;
424 for( k = 0; k <= cn; k++ )
425 s += src[i0 + k]*stddev[j*(cn+1) + k];
426 dst[i] = saturate_cast<T>(s);
427 j = CN_NEXT(j);
428 }
429 }
430}

433static void
434randnScale_16_or_32f(float* fbuf, float* dst, int len, int cn,
435 const float* mean, const float* stddev, int flags)
436{
437 bool stdmtx = (flags & RNG_FLAG_STDMTX) != 0;
438 int depth = CV_MAT_DEPTH(flags);
439 float* arr = depth == CV_16F || depth == CV_16BF || isFP8Depth(depth) ? fbuf : dst;
440 int i, j, k;
441
442 if( !stdmtx || cn == 1 )
443 {
444 if( cn == 1 )
445 {
446 float a = stddev[0], b = mean[0];
447 for( i = 0; i < len; i++ )
448 arr[i] = fbuf[i]*a + b;
449 }
450 else
451 {
452 len *= cn;
453 cn--;
454 for( i = k = 0; i < len; i++ ) {
455 arr[i] = fbuf[i]*stddev[k] + mean[k];
456 k = CN_NEXT(k);
457 }
458 }
459 }
460 else if( depth == CV_32F )
461 {
462 len *= cn;
463 cn--;
464 for( i = j = 0; i < len; i++ )
465 {
466 float s = mean[j];
467 int i0 = i - j;
468 for( k = 0; k <= cn; k++ )
469 s += fbuf[i0 + k]*stddev[j*(cn+1) + k];
470 dst[i] = s;
471 j = CN_NEXT(j);
472 }
473 }
474 else
475 {
476 float elembuf[CV_CN_MAX];
477 len *= cn;
478 for( i = 0; i < len; i += cn )
479 {
480 // since we process fbuf in-place,
481 // we need to copy each cn-channel element
482 // prior to matrix multiplication
483 for (j = 0; j < cn; j++)
484 elembuf[j] = fbuf[i + j];
485 for (j = 0; j < cn; j++) {
486 float s = mean[j];
487 for( k = 0; k < cn; k++ )
488 s += elembuf[k]*stddev[j*cn + k];
489 fbuf[i + j] = s;
490 }
491 }
492 }
493 if (depth == CV_16F)
494 hal::cvt32f16f(fbuf, (hfloat*)dst, len);
495 else if (depth == CV_16BF)
496 hal::cvt32f16bf(fbuf, (bfloat*)dst, len);
497 else if (isFP8Depth(depth))
498 cvt32fToFP8(fbuf, dst, len, depth);
499}

530void RNG::fill( InputOutputArray _mat, int disttype,
531 InputArray _param1arg, InputArray _param2arg,
532 bool saturateRange )
533{
534 CV_Assert(!_mat.empty());
535
536 Mat mat = _mat.getMat(), _param1 = _param1arg.getMat(), _param2 = _param2arg.getMat();
537 int j, depth = mat.depth(), cn = mat.channels();
538 int esz1 = CV_ELEM_SIZE(depth);
539 AutoBuffer<double> _parambuf;
540 bool fast_int_mode = false;
541 bool small_flag = false;
542 RandFunc func = 0;
543 RandnScaleFunc scaleFunc = 0;
544
545 CV_Assert(_param1.channels() == 1 && (_param1.rows == 1 || _param1.cols == 1) &&
546 (_param1.rows + _param1.cols - 1 == cn || _param1.rows + _param1.cols - 1 == 1 ||
547 (_param1.size() == Size(1, 4) && _param1.type() == CV_64F && cn <= 4)));
548 CV_Assert( _param2.channels() == 1 &&
549 (((_param2.rows == 1 || _param2.cols == 1) &&
550 (_param2.rows + _param2.cols - 1 == cn || _param2.rows + _param2.cols - 1 == 1 ||
551 (_param1.size() == Size(1, 4) && _param1.type() == CV_64F && cn <= 4))) ||
552 (_param2.rows == cn && _param2.cols == cn && disttype == NORMAL)));
553
554 const void* uni_param = 0;
555 uchar* mean = 0;
556 uchar* stddev = 0;
557 bool stdmtx = false;
558 int n1 = (int)_param1.total();
559 int n2 = (int)_param2.total();
560
561 if( disttype == UNIFORM )
562 {
563 _parambuf.allocate(cn*(sizeof(DivStruct)+sizeof(double)-1)/sizeof(double) + cn*4);
564 double* parambuf = _parambuf.data();
565 double* p1 = _param1.ptr<double>();
566 double* p2 = _param2.ptr<double>();
567
568 if( !_param1.isContinuous() || _param1.type() != CV_64F || n1 != cn )
569 {
570 p1 = parambuf;
571 Mat tmp(_param1.size(), CV_64F, p1);
572 _param1.convertTo(tmp, CV_64F);
573 for( j = n1; j < cn; j++ )
574 p1[j] = p1[j-n1];
575 }
576
577 if( !_param2.isContinuous() || _param2.type() != CV_64F || n2 != cn )
578 {
579 p2 = parambuf + cn;
580 Mat tmp(_param2.size(), CV_64F, p2);
581 _param2.convertTo(tmp, CV_64F);
582 for( j = n2; j < cn; j++ )
583 p2[j] = p2[j-n2];
584 }
585
586 if( CV_IS_INT_TYPE(depth) )
587 {
588 Vec2l* ip = (Vec2l*)(parambuf + cn*2);
589 CV_DbgCheckLT((size_t)(cn*4 - 1), _parambuf.size(), "");
590 for( j = 0, fast_int_mode = true; j < cn; j++ )
591 {
592 double a = std::min(p1[j], p2[j]);
593 double b = std::max(p1[j], p2[j]);
594 if( saturateRange )
595 {
596 a = std::max(a, depth == CV_8U || depth == CV_16U || depth == CV_32U ||
597 depth == CV_64U || depth == CV_Bool ? 0. :
598 depth == CV_8S ? -128. : depth == CV_16S ? -32768. :
599 depth == CV_32S ? (double)INT_MIN : (double)INT64_MIN);
600 b = std::min(b, depth == CV_8U ? 256. : depth == CV_Bool ? 2. : depth == CV_16U ? 65536. :
601 depth == CV_8S ? 128. : depth == CV_16S ? 32768. : depth == CV_32U ? (double)UINT_MAX :
602 depth == CV_32S ? (double)INT_MAX : (double)INT64_MAX);
603 }
604 ip[j][1] = (int64_t)ceil(a);
605 int64_t idiff = ip[j][0] = (int64_t)floor(b) - ip[j][1] - 1;
606 if (idiff < 0)
607 {
608 idiff = 0;
609 ip[j][0] = 0;
610 }
611 double diff = b - a;
612
613 fast_int_mode = fast_int_mode && diff <= 4294967296. && (idiff & (idiff+1)) == 0;
614 if( fast_int_mode )
615 small_flag = idiff <= 255;
616 else
617 {
618 int64_t minval = INT32_MIN/2, maxval = INT32_MAX;
619 if (depth == CV_64S || depth == CV_64U)
620 {
621 minval = INT64_MIN/2;
622 maxval = INT64_MAX;
623 }
624 if( diff > (double)maxval )
625 ip[j][0] = maxval;
626 if( a < (double)minval )
627 ip[j][1] = minval;
628 }
629 }
630
631 uni_param = ip;
632 if( !fast_int_mode )
633 {
634 DivStruct* ds = (DivStruct*)(ip + cn);
635 CV_DbgCheckLE((void*)(ds + cn), (void*)(parambuf + _parambuf.size()), "Last byte check");
636 for( j = 0; j < cn; j++ )
637 {
638 ds[j].delta = ip[j][1];
639 ds[j].diff = ip[j][0];
640 if (depth != CV_64U && depth != CV_64S) {
641 unsigned d = ds[j].d = (unsigned)(ip[j][0]+1);
642 int l = 0;
643 while(((uint64)1 << l) < d)
644 l++;
645 ds[j].M = (unsigned)(((uint64)1 << 32)*(((uint64)1 << l) - d)/d) + 1;
646 ds[j].sh1 = std::min(l, 1);
647 ds[j].sh2 = std::max(l - 1, 0);
648 }
649 }
650 uni_param = ds;
651 }
652
653 func = randTab[fast_int_mode ? 1 : 0][depth];
654 }
655 else
656 {
657 double scale = depth == CV_64F ?
658 5.4210108624275221700372640043497e-20 : // 2**-64
659 2.3283064365386962890625e-10; // 2**-32
660 double maxdiff = saturateRange ? (double)FLT_MAX : DBL_MAX;
661
662 // for each channel i compute such dparam[0][i] & dparam[1][i],
663 // so that a signed 32/64-bit integer X is transformed to
664 // the range [param1.val[i], param2.val[i]) using
665 // dparam[0][i]*X + dparam[1][i]
666 CV_DbgCheckLT((size_t)(cn*4 - 1), _parambuf.size(), "");
667 if( depth != CV_64F )
668 {
669 Vec2f* fp = (Vec2f*)(parambuf + cn*2);
670 for( j = 0; j < cn; j++ )
671 {
672 fp[j][0] = (float)(std::min(maxdiff, p2[j] - p1[j])*scale);
673 fp[j][1] = (float)((p2[j] + p1[j])*0.5);
674 }
675 uni_param = fp;
676 }
677 else
678 {
679 Vec2d* dp = (Vec2d*)(parambuf + cn*2);
680 for( j = 0; j < cn; j++ )
681 {
682 dp[j][0] = std::min(DBL_MAX, p2[j] - p1[j])*scale;
683 dp[j][1] = ((p2[j] + p1[j])*0.5);
684 }
685 uni_param = dp;
686 }
687
688 func = randTab[0][depth];
689 }
690 CV_Assert( func != 0 );
691 }
692 else if( disttype == RNG::NORMAL )
693 {
694 _parambuf.allocate(MAX(n1, cn) + MAX(n2, cn));
695 double* parambuf = _parambuf.data();
696
697 int ptype = esz1 == 8 ? CV_64F : CV_32F;
698
699 if( _param1.isContinuous() && _param1.type() == ptype && n1 >= cn)
700 mean = _param1.ptr();
701 else
702 {
703 Mat tmp(_param1.size(), ptype, parambuf);
704 _param1.convertTo(tmp, ptype);
705 mean = (uchar*)parambuf;
706 }
707
708 if( n1 < cn )
709 for( j = n1*esz1; j < cn*esz1; j++ )
710 mean[j] = mean[j - n1*esz1];
711
712 if( _param2.isContinuous() && _param2.type() == ptype && n2 >= cn)
713 stddev = _param2.ptr();
714 else
715 {
716 Mat tmp(_param2.size(), ptype, parambuf + MAX(n1, cn));
717 _param2.convertTo(tmp, ptype);
718 stddev = (uchar*)(parambuf + MAX(n1, cn));
719 }
720
721 if( n2 < cn )
722 for( j = n2*esz1; j < cn*esz1; j++ )
723 stddev[j] = stddev[j - n2*esz1];
724
725 stdmtx = _param2.rows == cn && _param2.cols == cn;
726 scaleFunc = randnScaleTab[depth];
727 CV_Assert( scaleFunc != 0 );
728 }
729 else
730 CV_Error( cv::Error::StsBadArg, "Unknown distribution type" );
731
732 const Mat* arrays[] = {&mat, 0};
733 uchar* ptr = 0;
734 NAryMatIterator it(arrays, &ptr, 1);
735 float fbuf[BLOCK_SIZE + CV_CN_MAX];
736 int total = (int)it.size;
737 int blockSize = std::min((BLOCK_SIZE + cn - 1)/cn, total);
738 size_t esz = (size_t)esz1*cn;
739 int flags = mat.type();
740
741 if( disttype == UNIFORM )
742 flags |= (small_flag ? (int)RNG_FLAG_SMALL : 0);
743 else
744 flags |= (stdmtx ? (int)RNG_FLAG_STDMTX : 0);
745
746 for( size_t i = 0; i < it.nplanes; i++, ++it )
747 {
748 for( j = 0; j < total; j += blockSize )
749 {
750 int len = std::min(total - j, blockSize);
751
752 if( disttype == UNIFORM )
753 func(ptr + j*esz, len, cn, &state, uni_param, fbuf, flags);
754 else
755 {
756 randn_0_1_32f(fbuf, len*cn, &state);
757 scaleFunc(fbuf, ptr + j*esz, len, cn, mean, stddev, flags);
758 }
759 }
760 }
761}

793template<typename T> static void
794randShuffle_( Mat& _arr, RNG& rng, double )
795{
796 unsigned sz = (unsigned)_arr.total();
797 if( _arr.isContinuous() )
798 {
799 T* arr = _arr.ptr<T>();
800 for( unsigned i = 0; i < sz; i++ )
801 {
802 unsigned j = (unsigned)rng % sz;
803 std::swap( arr[j], arr[i] );
804 }
805 }
806 else
807 {
808 CV_Assert( _arr.dims <= 2 );
809 uchar* data = _arr.ptr();
810 size_t step = _arr.step;
811 int rows = _arr.rows;
812 int cols = _arr.cols;
813 for( int i0 = 0; i0 < rows; i0++ )
814 {
815 T* p = _arr.ptr<T>(i0);
816 for( int j0 = 0; j0 < cols; j0++ )
817 {
818 unsigned k1 = (unsigned)rng % sz;
819 int i1 = (int)(k1 / cols);
820 int j1 = (int)(k1 - (unsigned)i1*(unsigned)cols);
821 std::swap( p[j0], ((T*)(data + step*i1))[j1] );
822 }
823 }
824 }
825}

915void cv::RNG_MT19937::seed(unsigned s)
916{
917 state[0]= s;
918 for (mti = 1; mti < N; mti++)
919 {
920 /* See Knuth TAOCP Vol2. 3rd Ed. P.106 for multiplier. */
921 state[mti] = (1812433253U * (state[mti - 1] ^ (state[mti - 1] >> 30)) + mti);

925unsigned cv::RNG_MT19937::next()
926{
927 /* mag01[x] = x * MATRIX_A for x=0,1 */
928 static unsigned mag01[2] = { 0x0U, /*MATRIX_A*/ 0x9908b0dfU};
929
930 const unsigned UPPER_MASK = 0x80000000U;
931 const unsigned LOWER_MASK = 0x7fffffffU;
932
933 /* generate N words at one time */
934 if (mti >= N)
935 {
936 int kk = 0;
937
938 for (; kk < N - M; ++kk)
939 {
940 unsigned y = (state[kk] & UPPER_MASK) | (state[kk + 1] & LOWER_MASK);
941 state[kk] = state[kk + M] ^ (y >> 1) ^ mag01[y & 0x1U];
942 }
943
944 for (; kk < N - 1; ++kk)
945 {
946 unsigned y = (state[kk] & UPPER_MASK) | (state[kk + 1] & LOWER_MASK);
947 state[kk] = state[kk + (M - N)] ^ (y >> 1) ^ mag01[y & 0x1U];
948 }
949
950 unsigned y = (state[N - 1] & UPPER_MASK) | (state[0] & LOWER_MASK);
951 state[N - 1] = state[M - 1] ^ (y >> 1) ^ mag01[y & 0x1U];
952
953 mti = 0;
954 }
955
956 unsigned y = state[mti++];
957
958 /* Tempering */
959 y ^= (y >> 11);
960 y ^= (y << 7) & 0x9d2c5680U;
961 y ^= (y << 15) & 0xefc60000U;
962 y ^= (y >> 18);
963
964 return y;
965}
