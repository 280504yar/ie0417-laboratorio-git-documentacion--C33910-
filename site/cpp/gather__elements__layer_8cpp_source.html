12static inline int calculateOffset(int outer_dim, const MatShape &shape_indices, int axis_skip, const MatStep &step_data) {
13 int offset = 0;
14 for (int axis = static_cast<int>(shape_indices.size()) - 2; axis >= 0; axis--) {
15 int dim = shape_indices[axis];
16 if (axis != axis_skip) {
17 offset += (outer_dim % dim) * step_data[axis];
18 }
19 outer_dim /= dim;
20 }
21 return offset;
22}

100 template <typename T, typename T_INDEX>
101 void forward_impl(const Mat& data_, const Mat& indices_, Mat& out_)
102 {
103 const auto *ptr_data = data_.ptr<const T>();
104 const auto *ptr_indices = indices_.ptr<const T_INDEX>();
105 auto *ptr_out = out_.ptr<T>();
106
107 const auto shape_data = shape(data_);
108 const auto &step_data = data_.step;
109 const auto shape_indices = shape(indices_);
110
111 int inner_most_dim = shape_indices.back();
112 int axis_dim = shape_data[axis];
113 size_t axis_step = static_cast<size_t>(step_data[axis] / sizeof(T));
114
115 bool innermost_axis = axis == static_cast<int>(shape_data.size() - 1);
116
117 auto fn = [&](const Range &r) {
118 for (int i = r.start; i < r.end; i++) {
119 auto *data = ptr_data + static_cast<size_t>(calculateOffset(i, shape_indices, axis, step_data) / sizeof(T));
120 auto *indices = ptr_indices + i * inner_most_dim;
121 auto *out = ptr_out + i * inner_most_dim;
122
123 if (innermost_axis) {
124 for (int j = 0; j < inner_most_dim; j++) {
125 int index = (indices[j] + axis_dim) % axis_dim; // TODO: Check out-of-range index
126 out[j] = data[index];
127 }
128 } else {
129 for (int j = 0; j < inner_most_dim; j++) {
130 int index = (indices[j] + axis_dim) % axis_dim; // TODO: Check out-of-range index
131 out[j] = data[index * axis_step + j];
132 }
133 }
134 }
135 };
136
137 int outer_dims = total(shape_indices, 0, shape_indices.size() - 1);
138 double nstripes = static_cast<size_t>(outer_dims * inner_most_dim * (1 / 1024.0));
139 parallel_for_(Range(0, outer_dims), fn, nstripes);
140 }

144 {
145 switch (index_type)

158 // Gathering copies elements without interpreting them, so only the width matters.
159 template<typename T_INDEX, typename... Args>
160 inline void typeDispatch(const int type, Args&&... args)
161 {
162 switch (CV_ELEM_SIZE(type))
163 {
164 case 1:
165 forward_impl<uint8_t, T_INDEX>(std::forward<Args>(args)...);
166 break;
167 case 2:
168 forward_impl<uint16_t, T_INDEX>(std::forward<Args>(args)...);
169 break;
170 case 4:
171 forward_impl<uint32_t, T_INDEX>(std::forward<Args>(args)...);
172 break;
173 case 8:
174 forward_impl<uint64_t, T_INDEX>(std::forward<Args>(args)...);
175 break;
176 default:
177 CV_Error(cv::Error::BadDepth, "DNN/GatherElements: Unsupported type.");
178 };
