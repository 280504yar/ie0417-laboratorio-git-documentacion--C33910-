98void Cloning::dst(const Mat& src, Mat& dest, bool invert)
99{
100 Mat temp = Mat::zeros(src.rows, 2 * src.cols + 2, CV_32F);
101
102 int flag = invert ? DFT_ROWS + DFT_SCALE + DFT_INVERSE: DFT_ROWS;
103
104 src.copyTo(temp(Rect(1,0, src.cols, src.rows)));
105
106 for(int j = 0 ; j < src.rows ; ++j)
107 {
108 float * tempLinePtr = temp.ptr<float>(j);
109 const float * srcLinePtr = src.ptr<float>(j);
110 for(int i = 0 ; i < src.cols ; ++i)
111 {
112 tempLinePtr[src.cols + 2 + i] = - srcLinePtr[src.cols - 1 - i];
113 }
114 }
115
116 Mat planes[] = {temp, Mat::zeros(temp.size(), CV_32F)};
117 Mat complex;
118
119 merge(planes, 2, complex);
120 dft(complex, complex, flag);
121 split(complex, planes);
122 temp = Mat::zeros(src.cols, 2 * src.rows + 2, CV_32F);
123
124 for(int j = 0 ; j < src.cols ; ++j)
125 {
126 float * tempLinePtr = temp.ptr<float>(j);
127 for(int i = 0 ; i < src.rows ; ++i)
128 {
129 float val = planes[1].ptr<float>(i)[j + 1];
130 tempLinePtr[i + 1] = val;
131 tempLinePtr[temp.cols - 1 - i] = - val;
132 }
133 }
134
135 Mat planes2[] = {temp, Mat::zeros(temp.size(), CV_32F)};
136
137 merge(planes2, 2, complex);
138 dft(complex, complex, flag);
139 split(complex, planes2);
140
141 temp = planes2[1].t();
142 temp(Rect( 0, 1, src.cols, src.rows)).copyTo(dest);
143}

145void Cloning::solve(const Mat &img, Mat& mod_diff, Mat &result)
146{
147 const int w = img.cols;
148 const int h = img.rows;
149
150 Mat res;
151 dst(mod_diff, res);
152
153 for(int j = 0 ; j < h-2; j++)
154 {
155 float * resLinePtr = res.ptr<float>(j);
156 for(int i = 0 ; i < w-2; i++)
157 {
158 resLinePtr[i] /= (filter_X[i] + filter_Y[j] - 4);
159 }
160 }
161
162 dst(res, mod_diff, true);
163
164 unsigned char * resLinePtr = result.ptr<unsigned char>(0);
165 const unsigned char * imgLinePtr = img.ptr<unsigned char>(0);
166 const float * interpLinePtr = NULL;
167
168 //first col
169 for(int i = 0 ; i < w ; ++i)
170 result.ptr<unsigned char>(0)[i] = img.ptr<unsigned char>(0)[i];
171
172 for(int j = 1 ; j < h-1 ; ++j)
173 {
174 resLinePtr = result.ptr<unsigned char>(j);
175 imgLinePtr = img.ptr<unsigned char>(j);
176 interpLinePtr = mod_diff.ptr<float>(j-1);
177
178 //first row
179 resLinePtr[0] = imgLinePtr[0];
180
181 for(int i = 1 ; i < w-1 ; ++i)
182 {
183 //saturate cast is not used here, because it behaves differently from the previous implementation
184 //most notable, saturate_cast rounds before truncating, here it's the opposite.
185 float value = interpLinePtr[i-1];
186 if(value < 0.)
187 resLinePtr[i] = 0;
188 else if (value > 255.0)
189 resLinePtr[i] = 255;
190 else
191 resLinePtr[i] = static_cast<unsigned char>(value);
192 }
193
194 //last row
195 resLinePtr[w-1] = imgLinePtr[w-1];
196 }
197
198 //last col
199 resLinePtr = result.ptr<unsigned char>(h-1);
200 imgLinePtr = img.ptr<unsigned char>(h-1);
201 for(int i = 0 ; i < w ; ++i)
202 resLinePtr[i] = imgLinePtr[i];
203}

205void Cloning::poissonSolver(const Mat &img, Mat &laplacianX , Mat &laplacianY, Mat &result)
206{
207 const int w = img.cols;
208 const int h = img.rows;
209
210 Mat lap = laplacianX + laplacianY;
211
212 Mat bound = img.clone();
213
214 rectangle(bound, Point(1, 1), Point(img.cols-2, img.rows-2), Scalar::all(0), -1);
215 Mat boundary_points;
216 Laplacian(bound, boundary_points, CV_32F);
217
218 boundary_points = lap - boundary_points;
219
220 Mat mod_diff = boundary_points(Rect(1, 1, w-2, h-2));
221
222 solve(img,mod_diff,result);
223}

225void Cloning::initVariables(const Mat &destination, const Mat &binaryMask)
226{
227 destinationGradientX = Mat(destination.size(),CV_32FC3);
228 destinationGradientY = Mat(destination.size(),CV_32FC3);
229 patchGradientX = Mat(destination.size(),CV_32FC3);
230 patchGradientY = Mat(destination.size(),CV_32FC3);
231
232 binaryMaskFloat = Mat(binaryMask.size(),CV_32FC1);
233 binaryMaskFloatInverted = Mat(binaryMask.size(),CV_32FC1);
234
235 //init of the filters used in the dst
236 const int w = destination.cols;
237 filter_X.resize(w - 2);
238 double scale = CV_PI / (w - 1);
239 for(int i = 0 ; i < w-2 ; ++i)
240 filter_X[i] = 2.0f * (float)std::cos(scale * (i + 1));
241
242 const int h = destination.rows;
243 filter_Y.resize(h - 2);
244 scale = CV_PI / (h - 1);
245 for(int j = 0 ; j < h - 2 ; ++j)
246 filter_Y[j] = 2.0f * (float)std::cos(scale * (j + 1));
247}

249void Cloning::computeDerivatives(const Mat& destination, const Mat &patch, Mat &binaryMask)
250{
251 initVariables(destination, binaryMask);
252
253 computeGradientX(destination, destinationGradientX);
254 computeGradientY(destination, destinationGradientY);
255
256 computeGradientX(patch, patchGradientX);
257 computeGradientY(patch, patchGradientY);
258
259 Mat Kernel(Size(3, 3), CV_8UC1);
260 Kernel.setTo(Scalar(1));
261 erode(binaryMask, binaryMask, Kernel, Point(-1,-1), 3);
262
263 binaryMask.convertTo(binaryMaskFloat, CV_32FC1, 1.0/255.0);
264}

273 merge(channels,mat);

277{
278 vector <Mat> lhs_channels;
279 vector <Mat> result_channels;

281 split(lhs,lhs_channels);
282 split(result,result_channels);
283

285 multiply(lhs_channels[chan],rhs,result_channels[chan]);

287 merge(result_channels,result);

290void Cloning::poisson(const Mat &destination)
291{
292 Mat laplacianX = destinationGradientX + patchGradientX;
293 Mat laplacianY = destinationGradientY + patchGradientY;
294
295 computeLaplacianX(laplacianX,laplacianX);
296 computeLaplacianY(laplacianY,laplacianY);
297
298 split(laplacianX,rgbx_channel);
299 split(laplacianY,rgby_channel);
300
301 split(destination,output);
302
303 for(int chan = 0 ; chan < 3 ; ++chan)
304 {
305 poissonSolver(output[chan], rgbx_channel[chan], rgby_channel[chan], output[chan]);
306 }

309void Cloning::evaluate(const Mat &I, Mat &wmask, const Mat &cloned)
310{
311 bitwise_not(wmask,wmask);
312
313 wmask.convertTo(binaryMaskFloatInverted,CV_32FC1,1.0/255.0);
314
315 arrayProduct(destinationGradientX, binaryMaskFloatInverted, destinationGradientX);
316 arrayProduct(destinationGradientY, binaryMaskFloatInverted, destinationGradientY);
317
318 poisson(I);
319
320 merge(output,cloned);
321}

323void Cloning::normalClone(const Mat &destination, const Mat &patch, Mat &binaryMask, Mat &cloned, int flag)
324{
325 const int w = destination.cols;
326 const int h = destination.rows;
327 const int channel = destination.channels();
328 const int n_elem_in_line = w * channel;
329
330 computeDerivatives(destination,patch,binaryMask);
331
332 switch(flag)
333 {
334 case NORMAL_CLONE:
335 case NORMAL_CLONE_WIDE:
336 arrayProduct(patchGradientX, binaryMaskFloat, patchGradientX);
337 arrayProduct(patchGradientY, binaryMaskFloat, patchGradientY);
338 break;
339
340 case MIXED_CLONE:
341 case MIXED_CLONE_WIDE:
342 {
343 AutoBuffer<int> maskIndices(n_elem_in_line);
344 for (int i = 0; i < n_elem_in_line; ++i)
345 maskIndices[i] = i / channel;
346
347 for(int i=0;i < h; i++)
348 {
349 float * patchXLinePtr = patchGradientX.ptr<float>(i);
350 float * patchYLinePtr = patchGradientY.ptr<float>(i);
351 const float * destinationXLinePtr = destinationGradientX.ptr<float>(i);
352 const float * destinationYLinePtr = destinationGradientY.ptr<float>(i);
353 const float * binaryMaskLinePtr = binaryMaskFloat.ptr<float>(i);
354
355 for(int j=0; j < n_elem_in_line; j++)
356 {
357 int maskIndex = maskIndices[j];
358
359 if(abs(patchXLinePtr[j] - patchYLinePtr[j]) >
360 abs(destinationXLinePtr[j] - destinationYLinePtr[j]))
361 {
362 patchXLinePtr[j] *= binaryMaskLinePtr[maskIndex];
363 patchYLinePtr[j] *= binaryMaskLinePtr[maskIndex];
364 }
365 else
366 {
367 patchXLinePtr[j] = destinationXLinePtr[j]
368 * binaryMaskLinePtr[maskIndex];
369 patchYLinePtr[j] = destinationYLinePtr[j]
370 * binaryMaskLinePtr[maskIndex];
371 }
372 }
373 }
374 }
375 break;
376
377 case MONOCHROME_TRANSFER:
378 case MONOCHROME_TRANSFER_WIDE:
379 Mat gray;
380 cvtColor(patch, gray, COLOR_BGR2GRAY );
381
382 computeGradientX(gray,patchGradientX);
383 computeGradientY(gray,patchGradientY);
384
385 arrayProduct(patchGradientX, binaryMaskFloat, patchGradientX);
386 arrayProduct(patchGradientY, binaryMaskFloat, patchGradientY);
387 break;
388
389 }
390
391 evaluate(destination,binaryMask,cloned);
392}

407void Cloning::illuminationChange(Mat &I, Mat &mask, Mat &wmask, Mat &cloned, float alpha, float beta)
408{
409 CV_INSTRUMENT_REGION();
410
411 computeDerivatives(I,mask,wmask);
412
413 arrayProduct(patchGradientX,binaryMaskFloat, patchGradientX);
414 arrayProduct(patchGradientY,binaryMaskFloat, patchGradientY);
415
416 Mat mag;
417 magnitude(patchGradientX,patchGradientY,mag);
418
419 Mat multX, multY, multx_temp, multy_temp;
420
421 multiply(patchGradientX,pow(alpha,beta),multX);
422 pow(mag,-1*beta, multx_temp);
423 multiply(multX,multx_temp, patchGradientX);
424 patchNaNs(patchGradientX);
425
426 multiply(patchGradientY,pow(alpha,beta),multY);
427 pow(mag,-1*beta, multy_temp);
428 multiply(multY,multy_temp,patchGradientY);
429 patchNaNs(patchGradientY);
430
431 evaluate(I,wmask,cloned);
432}

434void Cloning::textureFlatten(Mat &I, Mat &mask, Mat &wmask, float low_threshold,
435 float high_threshold, int kernel_size, Mat &cloned)

447 arrayProduct(patchGradientX,binaryMaskFloat, patchGradientX);
448 arrayProduct(patchGradientY,binaryMaskFloat, patchGradientY);
449
450 evaluate(I,wmask,cloned);
