24template<typename _Tp, size_t blockM, size_t blockN>

27 struct Point2iHash
28 {
29 size_t operator()(const cv::Point2i& point) const noexcept
30 {
31 size_t seed = 0;
32 constexpr uint32_t GOLDEN_RATIO = 0x9e3779b9;
33 seed ^= std::hash<int>()(point.x) + GOLDEN_RATIO + (seed << 6) + (seed >> 2);
34 seed ^= std::hash<int>()(point.y) + GOLDEN_RATIO + (seed << 6) + (seed >> 2);
35 return seed;
36 }
37 };
38 typedef Matx<_Tp, blockM, blockN> MatType;
39 typedef std::unordered_map<Point2i, MatType, Point2iHash> IDtoBlockValueMap;

48 inline MatType& refBlock(size_t i, size_t j)
49 {
50 Point2i p((int)i, (int)j);
51 auto it = ijValue.find(p);
52 if (it == ijValue.end())
53 {
54 it = ijValue.insert({ p, MatType::zeros() }).first;
55 }
56 return it->second;
57 }

59 inline _Tp& refElem(size_t i, size_t j)
60 {
61 Point2i ib((int)(i / blockM), (int)(j / blockN));
62 Point2i iv((int)(i % blockM), (int)(j % blockN));
63 return refBlock(ib.x, ib.y)(iv.x, iv.y);
64 }
65
66 inline MatType valBlock(size_t i, size_t j) const
67 {
68 Point2i p((int)i, (int)j);
69 auto it = ijValue.find(p);
70 if (it == ijValue.end())
71 return MatType::zeros();
72 else
73 return it->second;
74 }
75
76 inline _Tp valElem(size_t i, size_t j) const
77 {
78 Point2i ib((int)(i / blockM), (int)(j / blockN));
79 Point2i iv((int)(i % blockM), (int)(j % blockN));
80 return valBlock(ib.x, ib.y)(iv.x, iv.y);
81 }

83 Mat diagonal() const
84 {
85 // Diagonal max length is the number of columns in the sparse matrix
86 int diagLength =int( blockN * nBlocks );
87 cv::Mat diag = cv::Mat::zeros(diagLength, 1, cv::DataType<_Tp>::type);
88
89 for (int i = 0; i < diagLength; i++)
90 {
91 diag.at<_Tp>(i, 0) = valElem(i, i);
92 }
93 return diag;
94 }

96#if defined(HAVE_EIGEN)
97 Eigen::SparseMatrix<_Tp> toEigen() const
98 {
99 std::vector<Eigen::Triplet<_Tp>> tripletList;
100 tripletList.reserve(ijValue.size() * blockM * blockN);
101 for (const auto& ijv : ijValue)
102 {
103 int xb = ijv.first.x, yb = ijv.first.y;
104 MatType vblock = ijv.second;
105 for (size_t i = 0; i < blockM; i++)
106 {
107 for (size_t j = 0; j < blockN; j++)
108 {
109 _Tp val = vblock((int)i, (int)j);
110 if (abs(val) >= NON_ZERO_VAL_THRESHOLD)
111 {
112 tripletList.push_back(Eigen::Triplet<_Tp>((int)(blockM * xb + i), (int)(blockN * yb + j), val));
113 }
114 }
115 }
116 }
117 Eigen::SparseMatrix<_Tp> EigenMat(blockM * nBlocks, blockN * nBlocks);
118 EigenMat.setFromTriplets(tripletList.begin(), tripletList.end());
119 EigenMat.makeCompressed();
120
121 return EigenMat;
122 }
123#endif
124 inline size_t nonZeroBlocks() const { return ijValue.size(); }

126 BlockSparseMat<_Tp, blockM, blockN>& operator+=(const BlockSparseMat<_Tp, blockM, blockN>& other)

128 for (const auto& oijv : other.ijValue)
129 {

140 // Decomposes matrix for further solution
141 // Sometimes it's required to consequently solve A*x = b then A*x = c then A*x = d...
142 // Splitting the solution procedure into two parts let us reuse the matrix' decomposition
143 struct Decomposition
144 {
145 Eigen::SparseMatrix<_Tp> bigA;
146 Eigen::SimplicialLDLT<Eigen::SparseMatrix<_Tp>> solver;
147 };
148
149 bool decompose(Decomposition& d, bool checkSymmetry = true) const
150 {
151 d.bigA = this->toEigen();
152
153 Eigen::SparseMatrix<_Tp> bigAtranspose = d.bigA.transpose();
154 if (checkSymmetry && !d.bigA.isApprox(bigAtranspose))
155 {
156 CV_Error(Error::StsBadArg, "H matrix is not symmetrical");
157 return false;
158 }
159
160 d.solver.compute(d.bigA);
161 bool r = (d.solver.info() == Eigen::Success);
162 if (!r)
163 {
164 CV_LOG_INFO(NULL, "Failed to eigen-decompose");
165 }
166
167 return r;
168 }
169
170 static bool solveDecomposed(const Decomposition& d, InputArray B, OutputArray X, OutputArray predB = cv::noArray())
171 {
172 Mat mb = B.getMat();
173 mb = mb.cols == 1 ? mb : mb.t();
174 Eigen::Matrix<_Tp, -1, 1> bigB;
175 cv2eigen(mb, bigB);
176
177 Eigen::Matrix<_Tp, -1, 1> solutionX = d.solver.solve(bigB);
178 if (d.solver.info() != Eigen::Success)
179 {
180 CV_LOG_INFO(NULL, "Failed to eigen-solve");
181 return false;
182 }
183 else
184 {
185 eigen2cv(solutionX, X);
186 if (predB.needed())
187 {
188 Eigen::Matrix<_Tp, -1, 1> predBEigen = d.bigA * solutionX;
189 eigen2cv(predBEigen, predB);
190 }
191 return true;
192 }
193 }
