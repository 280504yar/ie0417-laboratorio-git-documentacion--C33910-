184inline void convertToShow(const cv::Mat &src, cv::Mat &dst, bool toRGB = true)
185{
186 const int src_depth = src.depth();
187 CV_Assert(src_depth != CV_16F && src_depth != CV_32S);
188 cv::Mat tmp;
189 switch(src_depth)
190 {
191 case CV_8U:
192 tmp = src;
193 break;
194 case CV_8S:
195 cv::convertScaleAbs(src, tmp, 1, 127);
196 break;
197 case CV_16S:
198 cv::convertScaleAbs(src, tmp, 1/255., 127);
199 break;
200 case CV_16U:
201 cv::convertScaleAbs(src, tmp, 1/255.);
202 break;
203 case CV_32F:
204 case CV_64F: // assuming image has values in range [0, 1)
205 src.convertTo(tmp, CV_8U, 255., 0.);
206 break;
207 }
208 cv::cvtColor(tmp, dst, toRGB ? cv::COLOR_BGR2RGB : cv::COLOR_BGRA2BGR, dst.channels());
209}
