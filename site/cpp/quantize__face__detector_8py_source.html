41def scale(x, name):
42 with tf.variable_scope(name):
43 layer = dnnLayer(name)
44 w = tf.Variable(layer.blobs[0].flatten(), dtype=dtype, name='mul')
45 if len(layer.blobs) > 1:
46 b = tf.Variable(layer.blobs[1].flatten(), dtype=dtype, name='add')
47 return tf.nn.bias_add(tf.multiply(x, w), b)
48 else:
49 return tf.multiply(x, w, name)

51def conv(x, name, stride=1, pad='SAME', dilation=1, activ=None):
52 with tf.variable_scope(name):
53 layer = dnnLayer(name)
54 w = tf.Variable(layer.blobs[0].transpose(2, 3, 1, 0), dtype=dtype, name='weights')
55 if dilation == 1:
56 conv = tf.nn.conv2d(x, filter=w, strides=(1, stride, stride, 1), padding=pad)
57 else:
58 assert(stride == 1)
59 conv = tf.nn.atrous_conv2d(x, w, rate=dilation, padding=pad)
60
61 if len(layer.blobs) > 1:
62 b = tf.Variable(layer.blobs[1].flatten(), dtype=dtype, name='bias')
63 conv = tf.nn.bias_add(conv, b)
64 return activ(conv) if activ else conv

66def batch_norm(x, name):
67 with tf.variable_scope(name):
68 # Unfortunately, TensorFlow's batch normalization layer doesn't work with fp16 input.
69 # Here we do a cast to fp32 but remove it in the frozen graph.
70 if x.dtype != tf.float32:
71 x = tf.cast(x, tf.float32)
72
73 layer = dnnLayer(name)
74 assert(len(layer.blobs) >= 3)
75
76 mean = layer.blobs[0].flatten()
77 std = layer.blobs[1].flatten()
78 scale = layer.blobs[2].flatten()
79
80 eps = 1e-5
81 hasBias = len(layer.blobs) > 3
82 hasWeights = scale.shape != (1,)
83
84 if not hasWeights and not hasBias:
85 mean /= scale[0]
86 std /= scale[0]
87
88 mean = tf.Variable(mean, dtype=tf.float32, name='mean')
89 std = tf.Variable(std, dtype=tf.float32, name='std')
90 gamma = tf.Variable(scale if hasWeights else np.ones(mean.shape), dtype=tf.float32, name='gamma')
91 beta = tf.Variable(layer.blobs[3].flatten() if hasBias else np.zeros(mean.shape), dtype=tf.float32, name='beta')
92 bn = tf.nn.fused_batch_norm(x, gamma, beta, mean, std, eps,
93 is_training=False)[0]
94 if bn.dtype != dtype:
95 bn = tf.cast(bn, dtype)
96 return bn

98def l2norm(x, name):

101 w = tf.Variable(layer.blobs[0].flatten(), dtype=dtype, name='mul')

236 # By default, float16 weights are stored in repeated tensor's field called
237 # `half_val`. It has type int32 with leading zeros for unused bytes.
238 # This type is encoded by Variant that means only 7 bits are used for value
239 # representation but the last one is indicated the end of encoding. This way
240 # float16 might takes 1 or 2 or 3 bytes depends on value. To improve compression,
241 # we replace all `half_val` values to `tensor_content` using only 2 bytes for everyone.
242 for node in graph_def.node:
243 if 'value' in node.attr:
244 halfs = node.attr["value"].tensor.half_val
245 if not node.attr["value"].tensor.tensor_content and halfs:
246 node.attr["value"].tensor.tensor_content = struct.pack('H' * len(halfs), *halfs)
247 node.attr["value"].tensor.ClearField('half_val')
