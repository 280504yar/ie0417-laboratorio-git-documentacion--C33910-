60struct ByteVecHash {
61 std::size_t operator()(const std::vector<std::uint8_t>& v) const noexcept {
62 std::size_t h = 0;
63 for (auto b : v) h = h * 31u + static_cast<std::size_t>(b);
64 return h;
65 }
66};
