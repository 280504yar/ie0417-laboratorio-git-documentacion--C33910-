99#ifdef HAVE_OPENCL
100 bool forward_ocl(InputArrayOfArrays inputs_, OutputArrayOfArrays outputs_, OutputArrayOfArrays internals_)
101 {
102 std::vector<UMat> inputs;
103 std::vector<UMat> outputs;
104
105 inputs_.getUMatVector(inputs);
106 outputs_.getUMatVector(outputs);
107
108 for (int i = 0, n = outputs.size(); i < n; ++i)
109 {
110 void *src_handle = inputs[i].handle(ACCESS_READ);
111 void *dst_handle = outputs[i].handle(ACCESS_WRITE);
112 if (src_handle != dst_handle)
113 inputs[i].copyTo(outputs[i]);
114 }
115
116 return true;
117 }
118#endif

120 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
121 {
122 CV_TRACE_FUNCTION();
123 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
124
125 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
126 forward_ocl(inputs_arr, outputs_arr, internals_arr))
127
128 std::vector<Mat> inputs, outputs;
129 inputs_arr.getMatVector(inputs);
130 outputs_arr.getMatVector(outputs);
131
132 size_t i, ninputs = inputs.size(), noutputs = outputs.size();
133 for (i = 0; i < noutputs; ++i) {
134 const Mat& inp = inputs[i < ninputs ? i : 0];
135 if (outputs[i].data != inp.data)
136 inp.copyTo(outputs[i]);
137 }
138 }

164#ifdef HAVE_DNN_NGRAPH
165 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> >& inputs,
166 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
167 {
168 auto ieInpNode = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
169 // We use Reshape as blank layer.
170 // Previously, we used Concat layer, but it didn't work with scalar tensor.
171 // Also we used ConvertLike layer, but it did't work with old OpenVINO versions.
172 auto shape = std::make_shared<ov::op::v0::ShapeOf>(ieInpNode);
173 auto blank = std::make_shared<ov::op::v1::Reshape>(ieInpNode, shape, false);
174 return Ptr<BackendNode>(new InfEngineNgraphNode(blank));
175 }
176#endif // HAVE_DNN_NGRAPH

195Ptr<Layer> BlankLayer::create(const LayerParams& params)
196{
197 // In case of Caffe's Dropout layer from Faster-RCNN framework,
198 // https://github.com/rbgirshick/caffe-fast-rcnn/tree/faster-rcnn
199 // return Power layer.
200 if (!params.get<bool>("scale_train", true))
201 {
202 float scale = 1 - params.get<float>("dropout_ratio", 0.5f);
203 CV_Assert(scale > 0);
204
205 LayerParams powerParams;
206 powerParams.name = params.name;
207 powerParams.type = "Power";
208 powerParams.set("scale", scale);
209
210 return PowerLayer::create(powerParams);
211 }
212 else
213 return Ptr<BlankLayer>(new BlankLayerImpl(params));
214}
