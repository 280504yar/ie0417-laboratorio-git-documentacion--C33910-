75static bool genericWarp(const Mat& src, const Mat& M_, const Mat& mapx, const Mat& mapy, Mat& dst,
76 int interpolation, int borderType_, const Scalar& borderValue, bool relativeMap_)
77{
78 constexpr int MAX_CHANNELS = 4;
79 int srctype = src.type();
80
81 ImgWarpFunc func = getImgWarpFunc(srctype, interpolation);
82 if (!func) {
83 return false; // not implemented;
84 }
85
86 CV_Assert(src.type() == dst.type());
87 CV_Assert(src.channels() <= MAX_CHANNELS);
88 Matx33f Mdata(1.f, 0.f, 0.f, 0.f, 1.f, 0.f, 0.f, 0.f, 1.f);
89
90 if (!M_.empty()) {
91 if (!mapx.empty() || !mapy.empty()) {
92 CV_Error(Error::StsBadArg, "Either M or both mapx and mapy must be empty");
93 }
94 CV_Assert(M_.size() == Size(3, 2) || M_.size() == Size(3, 3));
95 CV_Assert(M_.type() == CV_32F || M_.type() == CV_64F);
96 Mat Mcopy(M_.size(), CV_32F, Mdata.val);
97 M_.convertTo(Mcopy, CV_32F);
98 } else {
99 if (mapx.empty()) {
100 CV_Error(Error::StsBadArg, "When M is empty, mapx and/or mapy must be non-empty");
101 }
102 CV_Assert(mapx.size() == dst.size());
103 if (mapy.empty()) {
104 CV_Assert(mapx.type() == CV_32FC2);
105 } else {
106 if (mapx.type() == mapy.type()) {
107 CV_Assert(mapx.type() == CV_32FC1);
108 } else {
109 CV_Assert(mapx.type() == CV_16SC2 && (mapy.type() == CV_16UC1 || mapy.type() == CV_16SC1));
110 }
111 CV_Assert(mapx.size() == mapy.size());
112 }
113 }
114
115 int64_t borderValBuf_[MAX_CHANNELS];
116 scalarToRawData(borderValue, borderValBuf_, src.type());
117
118 parallel_for_(Range(0, dst.rows), [&](const Range& range) {
119 constexpr float FIXPT_SCALE = 1.f/32;
120 constexpr int BLOCK_SIZE = 128;
121 const uint8_t* srcdata = src.data;
122 const float* fparams = nullptr;
123 size_t srcstep = src.step;
124 Size srcsize = src.size();
125 int dstcols = dst.cols;
126 int borderType = borderType_;
127 size_t bpp = src.elemSize();
128
129 bool warping = !M_.empty();
130 bool warpAffine = warping && M_.rows == 2;
131 bool warpPerspective = warping && M_.rows == 3;
132
133 bool relativeMap = relativeMap_ && !warping;
134 float relscale = relativeMap ? 1.f : 0.f;
135 bool interleavedmap = !warping && (mapx.type() == CV_32FC2);
136 bool planarmap = !warping && (mapx.type() == mapy.type());
137 bool fixedmap = !warping && (mapx.type() == CV_16SC2);
138
139 CV_Assert(warping || interleavedmap || planarmap || fixedmap);
140
141 Matx33f M = Mdata;
142 const uint8_t* borderValBuf = (const uint8_t*)borderValBuf_;
143 float M_xx = M(0, 0), M_yx = M(1, 0), M_zx = M(2, 0);
144 float xbuf[BLOCK_SIZE], ybuf[BLOCK_SIZE];
145 const float* xbufptr = xbuf;
146 const float* ybufptr = ybuf;
147
148 for (int y = range.start; y < range.end; y++) {
149 uint8_t* dstptr = dst.ptr(y);
150 float M_x = float(y)*M(0, 1) + M(0, 2);
151 float M_y = float(y)*M(1, 1) + M(1, 2);
152 float M_z = float(y)*M(2, 1) + M(2, 2);
153 const Vec2s* xysptr = fixedmap ? mapx.ptr<Vec2s>(y) : nullptr;
154 const ushort* idxptr = fixedmap ? mapy.ptr<ushort>(y) : nullptr;
155 const Vec2f* xyfptr = interleavedmap ? mapx.ptr<Vec2f>(y) : nullptr;
156 const float* xfptr = planarmap ? mapx.ptr<float>(y) : nullptr;
157 const float* yfptr = planarmap ? mapy.ptr<float>(y) : nullptr;
158 float y0 = relativeMap ? float(y) : 0.f;
159
160 for (int x = 0; x < dstcols; x += BLOCK_SIZE) {
161 int blocksize = std::min(BLOCK_SIZE, dstcols - x);
162 if (warpAffine) {
163 for (int dx = 0; dx < blocksize; dx++) {
164 float xf = float(x + dx);
165 xbuf[dx] = M_x + M_xx*xf;
166 ybuf[dx] = M_y + M_yx*xf;
167 }
168 } else if (warpPerspective) {
169 for (int dx = 0; dx < blocksize; dx++) {
170 double xf = double(x + dx);
171 double invz = 1./(M_z + M_zx*xf);
172 xbuf[dx] = float((M_x + M_xx*xf)*invz);
173 ybuf[dx] = float((M_y + M_yx*xf)*invz);
174 }
175 } else if (fixedmap) {
176 for (int dx = 0; dx < blocksize; dx++) {
177 Vec2s xy = xysptr[x + dx];
178 ushort idx = idxptr[x + dx];
179 float xf = float(xy[0]) + (idx & 31)*FIXPT_SCALE + (x + dx)*relscale;
180 float yf = float(xy[1]) + ((idx >> 5) & 31)*FIXPT_SCALE + y0;
181 xbuf[dx] = xf;
182 ybuf[dx] = yf;
183 }
184 } else if (interleavedmap) {
185 for (int dx = 0; dx < blocksize; dx++) {
186 Vec2f xy = xyfptr[x + dx];
187 float xf = xy[0] + (x + dx)*relscale;
188 float yf = xy[1] + y0;
189 xbuf[dx] = xf;
190 ybuf[dx] = yf;
191 }
192 } else if (relativeMap) {
193 for (int dx = 0; dx < blocksize; dx++) {
194 float xf = xfptr[x + dx] + (x + dx)*relscale;
195 float yf = yfptr[x + dx] + y0;
196 xbuf[dx] = xf;
197 ybuf[dx] = yf;
198 }
199 } else {
200 // planar absolute map — just use it as-is without copying
201 xbufptr = xfptr + x;
202 ybufptr = yfptr + x;
203 }
204
205 func(xbufptr, ybufptr, blocksize, srcdata, srcstep, srcsize,
206 dstptr + x*bpp, fparams, borderType, borderValBuf);
207 }
208 }
209 });
210
211 return true;
212}

232static inline void interpolateLinear( float x, float* coeffs )
233{
234 coeffs[0] = 1.f - x;
235 coeffs[1] = x;
236}

238static inline void interpolateLanczos4( float x, float* coeffs )
239{
240 static const double s45 = 0.70710678118654752440084436210485;
241 static const double cs[][2]=
242 {{1, 0}, {-s45, -s45}, {0, 1}, {s45, -s45}, {-1, 0}, {s45, s45}, {0, -1}, {-s45, s45}};
243
244 if( x < FLT_EPSILON )
245 {
246 for( int i = 0; i < 8; i++ )
247 coeffs[i] = 0;
248 coeffs[3] = 1;
249 return;
250 }
251
252 float sum = 0;
253 double y0=-(x+3)*CV_PI*0.25, s0 = std::sin(y0), c0= std::cos(y0);
254 for(int i = 0; i < 8; i++ )
255 {
256 double y = -(x+3-i)*CV_PI*0.25;
257 coeffs[i] = (float)((cs[i][0]*s0 + cs[i][1]*c0)/(y*y));
258 sum += coeffs[i];
259 }
260
261 sum = 1.f/sum;
262 for(int i = 0; i < 8; i++ )
263 coeffs[i] *= sum;
264}

266static void initInterTab1D(int method, float* tab, int tabsz)
267{
268 float scale = 1.f/tabsz;
269 if( method == INTER_LINEAR )
270 {
271 for( int i = 0; i < tabsz; i++, tab += 2 )
272 interpolateLinear( i*scale, tab );
273 }
274 else if( method == INTER_CUBIC )
275 {
276 ; // pass
277 }
278 else if( method == INTER_LANCZOS4 )
279 {
280 for( int i = 0; i < tabsz; i++, tab += 8 )
281 interpolateLanczos4( i*scale, tab );
282 }
283 else
284 CV_Error( cv::Error::StsBadArg, "Unknown interpolation method" );
285}

288static const void* initInterTab2D( int method, bool fixpt )
289{
290 static bool inittab[INTER_MAX+1] = {false};
291
292 if( method == INTER_CUBIC )
293 return nullptr;
294
295 float* tab = 0;
296 short* itab = 0;
297 int ksize = 0;
298 if( method == INTER_LINEAR )
299 tab = BilinearTab_f[0][0], itab = BilinearTab_i[0][0], ksize=2;
300 else if( method == INTER_LANCZOS4 )
301 tab = Lanczos4Tab_f[0][0], itab = Lanczos4Tab_i[0][0], ksize=8;
302 else
303 CV_Error( cv::Error::StsBadArg, "Unknown/unsupported interpolation type" );
304
305 if( !inittab[method] )
306 {
307 AutoBuffer<float> _tab(8*INTER_TAB_SIZE);
308 int i, j, k1, k2;
309 initInterTab1D(method, _tab.data(), INTER_TAB_SIZE);
310 for( i = 0; i < INTER_TAB_SIZE; i++ )
311 for( j = 0; j < INTER_TAB_SIZE; j++, tab += ksize*ksize, itab += ksize*ksize )
312 {
313 int isum = 0;
314 NNDeltaTab_i[i*INTER_TAB_SIZE+j][0] = j >= INTER_TAB_SIZE/2;
315 NNDeltaTab_i[i*INTER_TAB_SIZE+j][1] = i >= INTER_TAB_SIZE/2;
316
317 for( k1 = 0; k1 < ksize; k1++ )
318 {
319 float vy = _tab[i*ksize + k1];
320 for( k2 = 0; k2 < ksize; k2++ )
321 {
322 float v = vy*_tab[j*ksize + k2];
323 tab[k1*ksize + k2] = v;
324 isum += itab[k1*ksize + k2] = saturate_cast<short>(v*INTER_REMAP_COEF_SCALE);
325 }
326 }
327
328 if( isum != INTER_REMAP_COEF_SCALE )
329 {
330 int diff = isum - INTER_REMAP_COEF_SCALE;
331 int ksize2 = ksize/2, Mk1=ksize2, Mk2=ksize2, mk1=ksize2, mk2=ksize2;
332 for( k1 = ksize2; k1 < ksize2+2; k1++ )
333 for( k2 = ksize2; k2 < ksize2+2; k2++ )
334 {
335 if( itab[k1*ksize+k2] < itab[mk1*ksize+mk2] )
336 mk1 = k1, mk2 = k2;
337 else if( itab[k1*ksize+k2] > itab[Mk1*ksize+Mk2] )
338 Mk1 = k1, Mk2 = k2;
339 }
340 if( diff < 0 )
341 itab[Mk1*ksize + Mk2] = (short)(itab[Mk1*ksize + Mk2] - diff);
342 else
343 itab[mk1*ksize + mk2] = (short)(itab[mk1*ksize + mk2] - diff);
344 }
345 }
346 tab -= INTER_TAB_SIZE2*ksize*ksize;
347 itab -= INTER_TAB_SIZE2*ksize*ksize;
348#if CV_SIMD128
349 if( method == INTER_LINEAR )
350 {
351 for( i = 0; i < INTER_TAB_SIZE2; i++ )
352 for( j = 0; j < 4; j++ )
353 {
354 BilinearTab_iC4[i][0][j*2] = BilinearTab_i[i][0][0];
355 BilinearTab_iC4[i][0][j*2+1] = BilinearTab_i[i][0][1];
356 BilinearTab_iC4[i][1][j*2] = BilinearTab_i[i][1][0];
357 BilinearTab_iC4[i][1][j*2+1] = BilinearTab_i[i][1][1];
358 }
359 }
360#endif
361 inittab[method] = true;
362 }
363 return fixpt ? (const void*)itab : (const void*)tab;
364}

404template<typename T, bool isRelative>
405static void remapNearest( const Mat& _src, Mat& _dst, const Mat& _xy,
406 int borderType, const Scalar& _borderValue, const Point& _offset )
407{
408 Size ssize = _src.size(), dsize = _dst.size();
409 const int cn = _src.channels();
410 const T* S0 = _src.ptr<T>();
411 T cval[CV_CN_MAX];
412 size_t sstep = _src.step/sizeof(S0[0]);
413
414 for(int k = 0; k < cn; k++ )
415 cval[k] = saturate_cast<T>(_borderValue[k & 3]);
416
417 unsigned width1 = ssize.width, height1 = ssize.height;
418
419 if( _dst.isContinuous() && _xy.isContinuous() && !isRelative )
420 {
421 dsize.width *= dsize.height;
422 dsize.height = 1;
423 }
424
425 for(int dy = 0; dy < dsize.height; dy++ )
426 {
427 T* D = _dst.ptr<T>(dy);
428 const short* XY = _xy.ptr<short>(dy);
429 const int off_y = isRelative ? (_offset.y+dy) : 0;
430 if( cn == 1 )
431 {
432 for(int dx = 0; dx < dsize.width; dx++ )
433 {
434 const int off_x = isRelative ? (_offset.x+dx) : 0;
435 int sx = XY[dx*2]+off_x, sy = XY[dx*2+1]+off_y;
436 if( (unsigned)sx < width1 && (unsigned)sy < height1 )
437 D[dx] = S0[sy*sstep + sx];
438 else
439 {
440 if( borderType == BORDER_REPLICATE )
441 {
442 sx = clip(sx, 0, ssize.width);
443 sy = clip(sy, 0, ssize.height);
444 D[dx] = S0[sy*sstep + sx];
445 }
446 else if( borderType == BORDER_CONSTANT )
447 D[dx] = cval[0];
448 else if( borderType != BORDER_TRANSPARENT )
449 {
450 sx = borderInterpolate(sx, ssize.width, borderType);
451 sy = borderInterpolate(sy, ssize.height, borderType);
452 D[dx] = S0[sy*sstep + sx];
453 }
454 }
455 }
456 }
457 else
458 {
459 for(int dx = 0; dx < dsize.width; dx++, D += cn )
460 {
461 const int off_x = isRelative ? (_offset.x+dx) : 0;
462 int sx = XY[dx*2]+off_x, sy = XY[dx*2+1]+off_y;
463 const T *S;
464 if( (unsigned)sx < width1 && (unsigned)sy < height1 )
465 {
466 if( cn == 3 )
467 {
468 S = S0 + sy*sstep + sx*3;
469 D[0] = S[0], D[1] = S[1], D[2] = S[2];
470 }
471 else if( cn == 4 )
472 {
473 S = S0 + sy*sstep + sx*4;
474 D[0] = S[0], D[1] = S[1], D[2] = S[2], D[3] = S[3];
475 }
476 else
477 {
478 S = S0 + sy*sstep + sx*cn;
479 for(int k = 0; k < cn; k++ )
480 D[k] = S[k];
481 }
482 }
483 else if( borderType != BORDER_TRANSPARENT )
484 {
485 if( borderType == BORDER_REPLICATE )
486 {
487 sx = clip(sx, 0, ssize.width);
488 sy = clip(sy, 0, ssize.height);
489 S = S0 + sy*sstep + sx*cn;
490 }
491 else if( borderType == BORDER_CONSTANT )
492 S = &cval[0];
493 else
494 {
495 sx = borderInterpolate(sx, ssize.width, borderType);
496 sy = borderInterpolate(sy, ssize.height, borderType);
497 S = S0 + sy*sstep + sx*cn;
498 }
499 for(int k = 0; k < cn; k++ )
500 D[k] = S[k];
501 }
502 }
503 }
504 }

750template<class CastOp, class VecOp, typename AT, bool isRelative>
751static void remapBilinear( const Mat& _src, Mat& _dst, const Mat& _xy,
752 const Mat& _fxy, const void* _wtab,
753 int borderType, const Scalar& _borderValue, const Point& _offset )
754{
755 typedef typename CastOp::rtype T;
756 typedef typename CastOp::type1 WT;
757 Size ssize = _src.size(), dsize = _dst.size();
758 const int cn = _src.channels();
759 const AT* wtab = (const AT*)_wtab;
760 const T* S0 = _src.ptr<T>();
761 size_t sstep = _src.step/sizeof(S0[0]);
762 T cval[CV_CN_MAX];
763 CastOp castOp;
764 VecOp vecOp;
765
766 for(int k = 0; k < cn; k++ )
767 cval[k] = saturate_cast<T>(_borderValue[k & 3]);
768
769 unsigned width1 = std::max(ssize.width-1, 0), height1 = std::max(ssize.height-1, 0);
770 CV_Assert( !ssize.empty() );
771#if CV_SIMD128
772 if( _src.type() == CV_8UC3 )
773 width1 = std::max(ssize.width-2, 0);
774#endif
775
776 for(int dy = 0; dy < dsize.height; dy++ )
777 {
778 T* D = _dst.ptr<T>(dy);
779 const short* XY = _xy.ptr<short>(dy);
780 const ushort* FXY = _fxy.ptr<ushort>(dy);
781 int X0 = 0;
782 bool prevInlier = false;
783 const int off_y = (isRelative ? (_offset.y+dy) : 0);
784 for(int dx = 0; dx <= dsize.width; dx++ )
785 {
786 bool curInlier = dx < dsize.width ?
787 (unsigned)XY[dx*2]+(isRelative ? (_offset.x+dx) : 0) < width1 &&
788 (unsigned)XY[dx*2+1]+off_y < height1 : !prevInlier;
789 if( curInlier == prevInlier )
790 continue;
791
792 int X1 = dx;
793 dx = X0;
794 X0 = X1;
795 prevInlier = curInlier;
796
797 if( !curInlier )
798 {
799 Point subOffset(_offset.x+dx, _offset.y+dy);
800 int len = vecOp( _src, D, XY + dx*2, FXY + dx, wtab, X1 - dx, subOffset );
801 D += len*cn;
802 dx += len;
803
804 if( cn == 1 )
805 {
806 for( ; dx < X1; dx++, D++ )
807 {
808 int sx = XY[dx*2]+(isRelative ? (_offset.x+dx) : 0), sy = XY[dx*2+1]+off_y;
809 const AT* w = wtab + FXY[dx]*4;
810 const T* S = S0 + sy*sstep + sx;
811 *D = castOp(WT(S[0]*w[0] + S[1]*w[1] + S[sstep]*w[2] + S[sstep+1]*w[3]));
812 }
813 }
814 else if( cn == 2 )
815 for( ; dx < X1; dx++, D += 2 )
816 {
817 int sx = XY[dx*2]+(isRelative ? (_offset.x+dx) : 0), sy = XY[dx*2+1]+off_y;
818 const AT* w = wtab + FXY[dx]*4;
819 const T* S = S0 + sy*sstep + sx*2;
820 WT t0 = S[0]*w[0] + S[2]*w[1] + S[sstep]*w[2] + S[sstep+2]*w[3];
821 WT t1 = S[1]*w[0] + S[3]*w[1] + S[sstep+1]*w[2] + S[sstep+3]*w[3];
822 D[0] = castOp(t0); D[1] = castOp(t1);
823 }
824 else if( cn == 3 )
825 for( ; dx < X1; dx++, D += 3 )
826 {
827 int sx = XY[dx*2]+(isRelative ? (_offset.x+dx) : 0), sy = XY[dx*2+1]+off_y;
828 const AT* w = wtab + FXY[dx]*4;
829 const T* S = S0 + sy*sstep + sx*3;
830 WT t0 = S[0]*w[0] + S[3]*w[1] + S[sstep]*w[2] + S[sstep+3]*w[3];
831 WT t1 = S[1]*w[0] + S[4]*w[1] + S[sstep+1]*w[2] + S[sstep+4]*w[3];
832 WT t2 = S[2]*w[0] + S[5]*w[1] + S[sstep+2]*w[2] + S[sstep+5]*w[3];
833 D[0] = castOp(t0); D[1] = castOp(t1); D[2] = castOp(t2);
834 }
835 else if( cn == 4 )
836 for( ; dx < X1; dx++, D += 4 )
837 {
838 int sx = XY[dx*2]+(isRelative ? (_offset.x+dx) : 0), sy = XY[dx*2+1]+off_y;
839 const AT* w = wtab + FXY[dx]*4;
840 const T* S = S0 + sy*sstep + sx*4;
841 WT t0 = S[0]*w[0] + S[4]*w[1] + S[sstep]*w[2] + S[sstep+4]*w[3];
842 WT t1 = S[1]*w[0] + S[5]*w[1] + S[sstep+1]*w[2] + S[sstep+5]*w[3];
843 D[0] = castOp(t0); D[1] = castOp(t1);
844 t0 = S[2]*w[0] + S[6]*w[1] + S[sstep+2]*w[2] + S[sstep+6]*w[3];
845 t1 = S[3]*w[0] + S[7]*w[1] + S[sstep+3]*w[2] + S[sstep+7]*w[3];
846 D[2] = castOp(t0); D[3] = castOp(t1);
847 }
848 else
849 for( ; dx < X1; dx++, D += cn )
850 {
851 int sx = XY[dx*2]+(isRelative ? (_offset.x+dx) : 0), sy = XY[dx*2+1]+off_y;
852 const AT* w = wtab + FXY[dx]*4;
853 const T* S = S0 + sy*sstep + sx*cn;
854 for(int k = 0; k < cn; k++ )
855 {
856 WT t0 = S[k]*w[0] + S[k+cn]*w[1] + S[sstep+k]*w[2] + S[sstep+k+cn]*w[3];
857 D[k] = castOp(t0);
858 }
859 }
860 }
861 else
862 {
863 if (borderType == BORDER_TRANSPARENT) {
864 for (; dx < X1; dx++, D += cn) {
865 if (dx >= dsize.width) continue;
866 const int sx = XY[dx * 2]+(isRelative ? (_offset.x+dx) : 0), sy = XY[dx * 2 + 1]+off_y;
867 // If the mapped point is still within bounds, it did not get computed
868 // because it lacked 4 neighbors. Still, it can be computed with an
869 // approximate formula. If it is outside, the point is left untouched.
870 if (sx >= 0 && sx <= ssize.width - 1 && sy >= 0 && sy <= ssize.height - 1) {
871 const AT* w = wtab + FXY[dx] * 4;
872 WT w_tot = 0;
873 if (sx >= 0 && sy >= 0) w_tot += w[0];
874 if (sy >= 0 && sx < ssize.width - 1) w_tot += w[1];
875 if (sx >= 0 && sy < ssize.height - 1) w_tot += w[2];
876 if (sx < ssize.width - 1 && sy < ssize.height - 1) w_tot += w[3];
877 if (w_tot == 0.f) continue;
878 const WT w_tot_ini = (WT)w[0] + w[1] + w[2] + w[3];
879 const T* S = S0 + sy * sstep + sx * cn;
880 for (int k = 0; k < cn; k++) {
881 WT t0 = 0;
882 if (sx >= 0 && sy >= 0) t0 += S[k] * w[0];
883 if (sy >= 0 && sx < ssize.width - 1) t0 += S[k + cn] * w[1];
884 if (sx >= 0 && sy < ssize.height - 1) t0 += S[sstep + k] * w[2];
885 if (sx < ssize.width - 1 && sy < ssize.height - 1) t0 += S[sstep + k + cn] * w[3];
886 t0 = (WT)(t0 * (float)w_tot_ini / w_tot);
887 D[k] = castOp(t0);
888 }
889 }
890 }
891 continue;
892 }
893
894 if( cn == 1 )
895 for( ; dx < X1; dx++, D++ )
896 {
897 int sx = XY[dx*2]+(isRelative ? (_offset.x+dx) : 0), sy = XY[dx*2+1]+off_y;
898 if( borderType == BORDER_CONSTANT &&
899 (sx >= ssize.width || sx+1 < 0 ||
900 sy >= ssize.height || sy+1 < 0) )
901 {
902 D[0] = cval[0];
903 }
904 else
905 {
906 int sx0, sx1, sy0, sy1;

908 const AT* w = wtab + FXY[dx]*4;
909 if( borderType == BORDER_REPLICATE )

912 sx1 = clip(sx+1, 0, ssize.width);
913 sy0 = clip(sy, 0, ssize.height);
914 sy1 = clip(sy+1, 0, ssize.height);
915 v0 = S0[sy0*sstep + sx0];
916 v1 = S0[sy0*sstep + sx1];
917 v2 = S0[sy1*sstep + sx0];
918 v3 = S0[sy1*sstep + sx1];
919 }
920 else
921 {
922 sx0 = borderInterpolate(sx, ssize.width, borderType);
923 sx1 = borderInterpolate(sx+1, ssize.width, borderType);
924 sy0 = borderInterpolate(sy, ssize.height, borderType);
925 sy1 = borderInterpolate(sy+1, ssize.height, borderType);
926 v0 = sx0 >= 0 && sy0 >= 0 ? S0[sy0*sstep + sx0] : cval[0];
927 v1 = sx1 >= 0 && sy0 >= 0 ? S0[sy0*sstep + sx1] : cval[0];
928 v2 = sx0 >= 0 && sy1 >= 0 ? S0[sy1*sstep + sx0] : cval[0];
929 v3 = sx1 >= 0 && sy1 >= 0 ? S0[sy1*sstep + sx1] : cval[0];
930 }
931 D[0] = castOp(WT(v0*w[0] + v1*w[1] + v2*w[2] + v3*w[3]));
932 }
933 }
934 else
935 for( ; dx < X1; dx++, D += cn )
936 {
937 int sx = XY[dx*2]+(isRelative ? (_offset.x+dx) : 0), sy = XY[dx*2+1]+off_y;
938 if( borderType == BORDER_CONSTANT &&
939 (sx >= ssize.width || sx+1 < 0 ||
940 sy >= ssize.height || sy+1 < 0) )
941 {
942 for(int k = 0; k < cn; k++ )
943 D[k] = cval[k];
944 }
945 else
946 {
947 int sx0, sx1, sy0, sy1;
948 const T *v0, *v1, *v2, *v3;
949 const AT* w = wtab + FXY[dx]*4;
950 if( borderType == BORDER_REPLICATE )
951 {
952 sx0 = clip(sx, 0, ssize.width);
953 sx1 = clip(sx+1, 0, ssize.width);
954 sy0 = clip(sy, 0, ssize.height);
955 sy1 = clip(sy+1, 0, ssize.height);
956 v0 = S0 + sy0*sstep + sx0*cn;
957 v1 = S0 + sy0*sstep + sx1*cn;
958 v2 = S0 + sy1*sstep + sx0*cn;
959 v3 = S0 + sy1*sstep + sx1*cn;
960 }
961 else
962 {
963 sx0 = borderInterpolate(sx, ssize.width, borderType);
964 sx1 = borderInterpolate(sx+1, ssize.width, borderType);
965 sy0 = borderInterpolate(sy, ssize.height, borderType);
966 sy1 = borderInterpolate(sy+1, ssize.height, borderType);
967 v0 = sx0 >= 0 && sy0 >= 0 ? S0 + sy0*sstep + sx0*cn : &cval[0];
968 v1 = sx1 >= 0 && sy0 >= 0 ? S0 + sy0*sstep + sx1*cn : &cval[0];
969 v2 = sx0 >= 0 && sy1 >= 0 ? S0 + sy1*sstep + sx0*cn : &cval[0];
970 v3 = sx1 >= 0 && sy1 >= 0 ? S0 + sy1*sstep + sx1*cn : &cval[0];
971 }
972 for(int k = 0; k < cn; k++ )
973 D[k] = castOp(WT(v0[k]*w[0] + v1[k]*w[1] + v2[k]*w[2] + v3[k]*w[3]));
974 }
975 }
976 }
977 }
978 }

1021 if( (unsigned)sx < width1 && (unsigned)sy < height1 )

1024 for(int k = 0; k < cn; k++ )

1026 WT sum = 0;

1029 S[cn*4]*w[4] + S[cn*5]*w[5] + S[cn*6]*w[6] + S[cn*7]*w[7];
1030 w -= 64;
1031 S -= sstep*8 - 1;
1032 D[k] = castOp(sum);
1033 }
1034 }
1035 else
1036 {
1037 int x[8], y[8];
1038 if( borderType == BORDER_TRANSPARENT &&
1039 ((unsigned)(sx+3) >= (unsigned)ssize.width ||
1040 (unsigned)(sy+3) >= (unsigned)ssize.height) )

1050 }

1052 for(int i = 0; i < 8; i++ )

1055 y[i] = borderInterpolate(sy + i, ssize.height, borderType1);
1056 }

1058 for(int k = 0; k < cn; k++, S0++, w -= 64 )

1061 for(int i = 0; i < 8; i++, w += 8 )
1062 {
1063 int yi = y[i];
1064 if( yi < 0 )
1065 continue;
1066 const T* S1 = S0 + yi*sstep;
1067 if( x[0] >= 0 )
1068 sum += (S1[x[0]] - cv)*w[0];
1069 if( x[1] >= 0 )
1070 sum += (S1[x[1]] - cv)*w[1];
1071 if( x[2] >= 0 )
1072 sum += (S1[x[2]] - cv)*w[2];
1073 if( x[3] >= 0 )
1074 sum += (S1[x[3]] - cv)*w[3];
1075 if( x[4] >= 0 )
1076 sum += (S1[x[4]] - cv)*w[4];
1077 if( x[5] >= 0 )
1078 sum += (S1[x[5]] - cv)*w[5];
1079 if( x[6] >= 0 )
1080 sum += (S1[x[6]] - cv)*w[6];
1081 if( x[7] >= 0 )
1082 sum += (S1[x[7]] - cv)*w[7];
1083 }
1084 D[k] = castOp(sum);
1085 }
1086 S0 -= cn;
1087 }
1088 }
