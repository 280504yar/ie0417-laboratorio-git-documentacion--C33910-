59template<typename _uint_t> inline size_t
60to_binary(_uint_t val, uchar * cur)
61{
62 size_t delta = CHAR_BIT;
63 size_t cnt = sizeof(_uint_t);
64 while (cnt --> static_cast<size_t>(0U)) {
65 *cur++ = static_cast<uchar>(val);
66 val >>= delta;
67 }
68 return sizeof(_uint_t);
69}
