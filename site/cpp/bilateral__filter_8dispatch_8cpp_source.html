61static inline int bilateralRadius(int d, double sigmaSpace)
62{
63 int radius = (d <= 0) ? cvRound(sigmaSpace * 1.5) : d / 2;
64 return std::max(radius, 1);
65}

69static bool ocl_bilateralFilter_8u(InputArray _src, OutputArray _dst, int d,
70 double sigma_color, double sigma_space,
71 int borderType)
72{
73 CV_INSTRUMENT_REGION();
74#ifdef __ANDROID__
75 if (ocl::Device::getDefault().isNVidia())
76 return false;
77#endif
78
79 int type = _src.type(), depth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type);
80 int i, j, maxk, radius;
81
82 if (depth != CV_8U || cn > 4)
83 return false;
84
85 double gauss_color_coeff = -0.5 / (sigma_color * sigma_color);
86 double gauss_space_coeff = -0.5 / (sigma_space * sigma_space);
87
88 radius = bilateralRadius(d, sigma_space);
89 d = radius * 2 + 1;
90
91 UMat src = _src.getUMat(), dst = _dst.getUMat(), temp;
92 if (src.u == dst.u)
93 return false;
94
95 copyMakeBorder(src, temp, radius, radius, radius, radius, borderType);
96 AutoBuffer<float> _space_weight(d * d);
97 AutoBuffer<int> _space_ofs(d * d);
98 float * const space_weight = &_space_weight[0];
99 int * const space_ofs = &_space_ofs[0];
100
101 // initialize space-related bilateral filter coefficients
102 for( i = -radius, maxk = 0; i <= radius; i++ )
103 for( j = -radius; j <= radius; j++ )
104 {
105 double r = std::sqrt((double)i * i + (double)j * j);
106 if ( r > radius )
107 continue;
108 space_weight[maxk] = (float)std::exp(r * r * gauss_space_coeff);
109 space_ofs[maxk++] = (int)(i * temp.step + j * cn);
110 }
111
112 char cvt[3][50];
113 String cnstr = cn > 1 ? format("%d", cn) : "";
114 String kernelName("bilateral");
115 size_t sizeDiv = 1;
116 if ((ocl::Device::getDefault().isIntel()) &&
117 (ocl::Device::getDefault().type() == ocl::Device::TYPE_GPU))
118 {
119 //Intel GPU
120 if (dst.cols % 4 == 0 && cn == 1) // For single channel x4 sized images.
121 {
122 kernelName = "bilateral_float4";
123 sizeDiv = 4;
124 }
125 }
126 ocl::Kernel k(kernelName.c_str(), ocl::imgproc::bilateral_oclsrc,
127 format("-D radius=%d -D maxk=%d -D cn=%d -D int_t=%s -D uint_t=uint%s -D convert_int_t=%s"
128 " -D uchar_t=%s -D float_t=%s -D convert_float_t=%s -D convert_uchar_t=%s -D gauss_color_coeff=(float)%f",
129 radius, maxk, cn, ocl::typeToStr(CV_32SC(cn)), cnstr.c_str(),
130 ocl::convertTypeStr(CV_8U, CV_32S, cn, cvt[0], sizeof(cvt[0])),
131 ocl::typeToStr(type), ocl::typeToStr(CV_32FC(cn)),
132 ocl::convertTypeStr(CV_32S, CV_32F, cn, cvt[1], sizeof(cvt[1])),
133 ocl::convertTypeStr(CV_32F, CV_8U, cn, cvt[2], sizeof(cvt[2])), gauss_color_coeff));
134 if (k.empty())
135 return false;
136
137 Mat mspace_weight(1, d * d, CV_32FC1, space_weight);
138 Mat mspace_ofs(1, d * d, CV_32SC1, space_ofs);
139 UMat ucolor_weight, uspace_weight, uspace_ofs;
140
141 mspace_weight.copyTo(uspace_weight);
142 mspace_ofs.copyTo(uspace_ofs);
143
144 k.args(ocl::KernelArg::ReadOnlyNoSize(temp), ocl::KernelArg::WriteOnly(dst),
145 ocl::KernelArg::PtrReadOnly(uspace_weight),
146 ocl::KernelArg::PtrReadOnly(uspace_ofs));
147
148 size_t globalsize[2] = { (size_t)dst.cols / sizeDiv, (size_t)dst.rows };
149 return k.run(2, globalsize, NULL, false);
150}

154static void
155bilateralFilter_8u( const Mat& src, Mat& dst, int d,
156 double sigma_color, double sigma_space,
157 int borderType )
158{
159 CV_INSTRUMENT_REGION();
160
161 int cn = src.channels();
162 int i, j, maxk, radius;
163
164 CV_Assert( (src.type() == CV_8UC1 || src.type() == CV_8UC3) && src.data != dst.data );
165
166 float gauss_color_coeff = (float)(-0.5/(sigma_color*sigma_color));
167 float gauss_space_coeff = (float)(-0.5/(sigma_space*sigma_space));
168
169 radius = bilateralRadius(d, sigma_space);
170 d = radius*2 + 1;
171
172 Mat temp;
173 copyMakeBorder( src, temp, radius, radius, radius, radius, borderType );
174
175 AutoBuffer<float> _color_weight(cn*256);
176 AutoBuffer<float> _space_weight(d*d);
177 AutoBuffer<int> _space_ofs(d*d);
178 float* color_weight = &_color_weight[0];
179 float* space_weight = &_space_weight[0];
180 int* space_ofs = &_space_ofs[0];
181
182 // initialize color-related bilateral filter coefficients
183 i = 0;
184#if (CV_SIMD || CV_SIMD_SCALABLE)
185 int nlanes = VTraits<v_float32>::vlanes();
186 v_float32 v_gauss_color_coeff = vx_setall_f32(gauss_color_coeff);
187 float counter[16] = {0.0, 1., 2., 3., 4., 5., 6., 7.,
188 8., 9., 10., 11., 12., 13., 14., 15.};
189 v_float32 v_i = vx_load(counter);
190 v_float32 v_inc = vx_setall_f32(float(nlanes));
191
192 for( ; i < (256*cn) - nlanes; i += nlanes )
193 {
194 v_float32 v_color_weight = v_mul(v_mul(v_i,v_i), v_gauss_color_coeff);
195 v_store(color_weight + i, v_exp(v_color_weight));
196 v_i = v_add(v_i, v_inc);
197 }
198#endif
199 for(; i < 256*cn; i++ )
200 {
201 color_weight[i] = (float)std::exp(i*i*gauss_color_coeff);
202 }
203
204 // initialize space-related bilateral filter coefficients
205 for( i = -radius, maxk = 0; i <= radius; i++ )
206 {
207 j = -radius;
208 for( ; j <= radius; j++ )
209 {
210 double r = std::sqrt((double)i*i + (double)j*j);
211 if( r > radius )
212 continue;
213 space_weight[maxk] = (float)std::exp(r*r*gauss_space_coeff);
214 space_ofs[maxk++] = (int)(i*temp.step + j*cn);
215 }
216 }
217
218 CV_CPU_DISPATCH(bilateralFilterInvoker_8u, (dst, temp, radius, maxk, space_ofs, space_weight, color_weight),
219 CV_CPU_DISPATCH_MODES_ALL);
220}

223static void
224bilateralFilter_32f( const Mat& src, Mat& dst, int d,
225 double sigma_color, double sigma_space,
226 int borderType )
227{
228 CV_INSTRUMENT_REGION();
229
230 int cn = src.channels();
231 int i, j, maxk, radius;
232 double minValSrc=-1, maxValSrc=1;
233 const int kExpNumBinsPerChannel = 1 << 12;
234 int kExpNumBins = 0;
235 float len, scale_index;
236
237 CV_Assert( (src.type() == CV_32FC1 || src.type() == CV_32FC3) && src.data != dst.data );
238
239 double gauss_color_coeff = -0.5/(sigma_color*sigma_color);
240 double gauss_space_coeff = -0.5/(sigma_space*sigma_space);
241
242 radius = bilateralRadius(d, sigma_space);
243 d = radius*2 + 1;
244 // compute the min/max range for the input image (even if multichannel)
245
246 minMaxLoc( src.reshape(1), &minValSrc, &maxValSrc );
247 if(std::abs(minValSrc - maxValSrc) < FLT_EPSILON)
248 {
249 src.copyTo(dst);
250 return;
251 }
252
253 // temporary copy of the image with borders for easy processing
254 Mat temp;
255 copyMakeBorder( src, temp, radius, radius, radius, radius, borderType );
256
257 // allocate lookup tables
258 AutoBuffer<float> _space_weight(d*d);
259 AutoBuffer<int> _space_ofs(d*d);
260 float* space_weight = &_space_weight[0];
261 int* space_ofs = &_space_ofs[0];
262
263 // assign a length which is slightly more than needed
264 len = (float)(maxValSrc - minValSrc) * cn;
265 kExpNumBins = kExpNumBinsPerChannel * cn;
266 AutoBuffer<float> _expLUT(kExpNumBins+2);
267 float* expLUT = &_expLUT[0];
268
269 scale_index = kExpNumBins/len;
270
271 // initialize the exp LUT
272 i = 0;
273#if (CV_SIMD || CV_SIMD_SCALABLE)
274 int nlanes = VTraits<v_float32>::vlanes();
275 v_float32 v_scale_index = vx_setall_f32((float)scale_index);
276 v_float32 v_gauss_color_coeff = vx_setall_f32((float)gauss_color_coeff);
277 float counter[16] = {0., 1., 2., 3., 4., 5., 6., 7.,
278 8., 9., 10., 11., 12., 13., 14., 15.};
279 v_float32 v_i = vx_load(counter);
280 v_float32 v_inc = vx_setall_f32(float(nlanes));
281
282 for( ; i < (kExpNumBins+2) - nlanes; i += nlanes )
283 {
284 v_float32 v_val = v_div(v_i, v_scale_index);
285 v_store(expLUT + i, v_exp(v_mul(v_mul(v_val, v_val), v_gauss_color_coeff)));
286 v_i = v_add(v_i, v_inc);
287 }
288#endif
289 for( ; i < kExpNumBins+2; i++ )
290 {
291 double val = i / scale_index;
292 expLUT[i] = (float)std::exp(val * val * gauss_color_coeff);
293 }
294
295 // initialize space-related bilateral filter coefficients
296 for( i = -radius, maxk = 0; i <= radius; i++ )
297 for( j = -radius; j <= radius; j++ )
298 {
299 double r = std::sqrt((double)i*i + (double)j*j);
300 if( r > radius || ( i == 0 && j == 0 ) )
301 continue;
302 space_weight[maxk] = (float)std::exp(r*r*gauss_space_coeff);
303 space_ofs[maxk++] = (int)(i*(temp.step/sizeof(float)) + j*cn);
304 }
305
306 // parallel_for usage
307 CV_CPU_DISPATCH(bilateralFilterInvoker_32f, (cn, radius, maxk, space_ofs, temp, dst, scale_index, space_weight, expLUT, kExpNumBins),
308 CV_CPU_DISPATCH_MODES_ALL);
309}

311void bilateralFilter( InputArray _src, OutputArray _dst, int d,
312 double sigmaColor, double sigmaSpace,
313 int borderType )
314{
315 CV_INSTRUMENT_REGION();
316
317 CV_Assert(!_src.empty());
318
319 _dst.create( _src.size(), _src.type() );
320
321 constexpr double eps = 1e-6;
322 if( sigmaColor <= eps || sigmaSpace <= eps )
323 {
324 _src.copyTo(_dst);
325 return;
326 }
327
328 CV_OCL_RUN(_src.dims() <= 2 && _dst.isUMat(),
329 ocl_bilateralFilter_8u(_src, _dst, d, sigmaColor, sigmaSpace, borderType))
330
331 Mat src = _src.getMat(), dst = _dst.getMat();
332
333 {
334 Point ofs;
335 Size wsz(src.cols, src.rows);
336 if( !(borderType & BORDER_ISOLATED) )
337 src.locateROI( wsz, ofs );
338
339 CALL_HAL(bilateralFilter, cv_hal_bilateralFilter_offset, src.data, src.step, dst.data, dst.step,
340 src.cols, src.rows, src.depth(), src.channels(),
341 ofs.x, ofs.y, wsz.width - src.cols - ofs.x, wsz.height - src.rows - ofs.y,
342 d, sigmaColor, sigmaSpace, borderType & (~BORDER_ISOLATED));
343 }
344
345 CALL_HAL(bilateralFilter, cv_hal_bilateralFilter, src.data, src.step, dst.data, dst.step,
346 src.cols, src.rows, src.depth(), src.channels(), d, sigmaColor, sigmaSpace, borderType);
347
348 if( src.depth() == CV_8U )
349 bilateralFilter_8u( src, dst, d, sigmaColor, sigmaSpace, borderType );
350 else if( src.depth() == CV_32F )
351 bilateralFilter_32f( src, dst, d, sigmaColor, sigmaSpace, borderType );
352 else
353 CV_Error( cv::Error::StsUnsupportedFormat,
354 "Bilateral filtering is only implemented for 8u and 32f images" );
355}
