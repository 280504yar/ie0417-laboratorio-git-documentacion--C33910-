33void LUT8u_( const uchar* src, const uchar* lut, uchar* dst, int len, int cn, int lutcn )
34{
35 const int total = len * cn;
36
37 if( lutcn == 1 )
38 {
39 int i = 0;
40#if CV_LUT8U_SIMD
41 const int nlanes = VTraits<v_uint8>::vlanes();
42 for( ; i <= total - nlanes; i += nlanes )
43 {
44 v_uint8 idx = vx_load(src + i);
45 v_uint8 result = v_lut(lut, idx);
46 v_store(dst + i, result);
47 }
48#endif
49 for( ; i < total; i++ )
50 dst[i] = lut[src[i]];
51 }
52#if CV_LUT8U_SIMD
53 else if( cn == 3 )
54 {
55 // Deinterleave the 3-channel LUT into per-channel tables
56 uchar CV_DECL_ALIGNED(64) lut0[256], lut1[256], lut2[256];
57 const int nlanes = VTraits<v_uint8>::vlanes();
58 {
59 unsigned j = 0;
60 for( ; j + (unsigned)nlanes <= 256; j += (unsigned)nlanes )
61 {
62 v_uint8 a, b, c;
63 v_load_deinterleave(lut + j * 3, a, b, c);
64 v_store(lut0 + j, a);
65 v_store(lut1 + j, b);
66 v_store(lut2 + j, c);
67 }
68 for( ; j < 256; j++ )
69 {
70 const unsigned idx = j * 3;
71 lut0[j] = lut[idx];
72 lut1[j] = lut[idx + 1];
73 lut2[j] = lut[idx + 2];
74 }
75 }
76 int i = 0;
77 for( ; i <= total - nlanes*3; i += nlanes*3 )
78 {
79 v_uint8 r, g, b;
80 v_load_deinterleave(src + i, r, g, b);
81 r = v_lut(lut0, r);
82 g = v_lut(lut1, g);
83 b = v_lut(lut2, b);
84 v_store_interleave(dst + i, r, g, b);
85 }
86 for( ; i < total; i += 3 )
87 {
88 dst[i] = lut0[src[i]];
89 dst[i+1] = lut1[src[i+1]];
90 dst[i+2] = lut2[src[i+2]];
91 }
92 }
93#endif
94 else
95 {
96 for( int i = 0; i < total; i += cn )
97 for( int k = 0; k < cn; k++ )
98 dst[i+k] = lut[src[i+k]*cn + k];
99 }
100}

102void LUT16u_( const uchar* src, const ushort* lut, ushort* dst, int len, int cn, int lutcn )
103{
104 const int total = len * cn;
105
106 if( lutcn == 1 )
107 {
108 int i = 0;
109#if CV_LUT8U_SIMD
110 // Split ushort LUT into low-byte and high-byte tables,
111 // then use two byte v_lut (vpermb on VBMI) + interleave.
112 uchar CV_DECL_ALIGNED(64) lut_lo[256], lut_hi[256];
113 for( int j = 0; j < 256; j++ )
114 {
115 lut_lo[j] = (uchar)(lut[j]);
116 lut_hi[j] = (uchar)(lut[j] >> 8);
117 }
118 const int nlanes8 = VTraits<v_uint8>::vlanes();
119 const int nlanes16 = VTraits<v_uint16>::vlanes();
120 for( ; i <= total - nlanes8; i += nlanes8 )
121 {
122 v_uint8 idx = vx_load(src + i);
123 v_uint8 lo = v_lut(lut_lo, idx);
124 v_uint8 hi = v_lut(lut_hi, idx);
125 // Zip low and high bytes: [l0,h0,l1,h1,...] → ushort values on little-endian
126 v_uint8 res0, res1;
127 v_zip(lo, hi, res0, res1);
128 v_store(dst + i, v_reinterpret_as_u16(res0));
129 v_store(dst + i + nlanes16, v_reinterpret_as_u16(res1));
130 }
131#endif
132 for( ; i < total; i++ )
133 dst[i] = lut[src[i]];
134 }
135 else
136 {
137 for( int i = 0; i < total; i += cn )
138 for( int k = 0; k < cn; k++ )
139 dst[i+k] = lut[src[i+k]*cn + k];
140 }
141}
