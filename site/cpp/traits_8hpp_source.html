484#define CV_CREATE_MEMBER_CHECK(X) \
485template<typename T> class CheckMember_##X { \
486 struct Fallback { int X; }; \
487 struct Derived : T, Fallback { }; \
488 template<typename U, U> struct Check; \
489 typedef char CV_NO[1]; \
490 typedef char CV_YES[2]; \
491 template<typename U> static CV_NO & func(Check<int Fallback::*, &U::X> *); \
492 template<typename U> static CV_YES & func(...); \
493public: \
494 typedef CheckMember_##X type; \
495 enum { value = sizeof(func<Derived>(0)) == sizeof(CV_YES) }; \
496};
