90 protected boolean selectCamera() {
91 Log.i(LOGTAG, "selectCamera");
92 CameraManager manager = (CameraManager) getContext().getSystemService(Context.CAMERA_SERVICE);
93 try {
94 String camList[] = manager.getCameraIdList();
95 if (camList.length == 0) {
96 Log.e(LOGTAG, "Error: camera isn't detected.");
97 return false;
98 }
99 boolean chosen = false; // remember whether the camera ID is set.
100 if (mCameraIndex == CameraBridgeViewBase.CAMERA_ID_ANY) {
101 mCameraID = camList[0];
102 chosen = true;
103 } else {
104 for (String cameraID : camList) {
105 CameraCharacteristics characteristics = manager.getCameraCharacteristics(cameraID);
106 if ((mCameraIndex == CameraBridgeViewBase.CAMERA_ID_BACK &&
107 characteristics.get(CameraCharacteristics.LENS_FACING) == CameraCharacteristics.LENS_FACING_BACK) ||
108 (mCameraIndex == CameraBridgeViewBase.CAMERA_ID_FRONT &&
109 characteristics.get(CameraCharacteristics.LENS_FACING) == CameraCharacteristics.LENS_FACING_FRONT)
110 ) {
111 mCameraID = cameraID;
112 chosen = true;
113 break;
114 }
115 }
116 }
117 if (mCameraID == null || !chosen) { // make JavaCamera2View behaves in the same way as JavaCameraView
118 Log.i(LOGTAG, "Selecting camera by index (" + mCameraIndex + ")");
119 if (mCameraIndex < camList.length) {
120 mCameraID = camList[mCameraIndex];
121 } else {
122 // CAMERA_DISCONNECTED is used when the camera id is no longer valid
123 throw new CameraAccessException(CameraAccessException.CAMERA_DISCONNECTED);
124 }
125 }
126 return true;
127 } catch (CameraAccessException e) {
128 Log.e(LOGTAG, "selectCamera - Camera Access Exception", e);
129 } catch (IllegalArgumentException e) {
130 Log.e(LOGTAG, "selectCamera - Illegal Argument Exception", e);
131 } catch (SecurityException e) {
132 Log.e(LOGTAG, "selectCamera - Security Exception", e);
133 }
134 return false;

189 final int w = mPreviewSize.getWidth(), h = mPreviewSize.getHeight();
190 Log.i(LOGTAG, "createCameraPreviewSession(" + w + "x" + h + ")");
191 if (w < 0 || h < 0)
192 return;
193 try {
194 if (null == mCameraDevice) {
195 Log.e(LOGTAG, "createCameraPreviewSession: camera isn't opened");
196 return;
197 }
198 if (null != mCaptureSession) {
199 Log.e(LOGTAG, "createCameraPreviewSession: mCaptureSession is already started");
200 return;

202
203 mImageReader = ImageReader.newInstance(w, h, mPreviewFormat, 2);
204 mImageReader.setOnImageAvailableListener(new ImageReader.OnImageAvailableListener() {
205 @Override
206 public void onImageAvailable(ImageReader reader) {
207
208 Image image = reader.acquireLatestImage();
209 if (image == null)
210 return;
211
212 // sanity checks - 3 planes
213 Image.Plane[] planes = image.getPlanes();
214 assert (planes.length == 3);
215 assert (image.getFormat() == mPreviewFormat);
216
217 RotatedCameraFrame tempFrame = new RotatedCameraFrame(new JavaCamera2Frame(image), mFrameRotation);
218 deliverAndDrawFrame(tempFrame);
219 tempFrame.mFrame.release();
220 tempFrame.release();
221 image.close();
222 }
223 }, mBackgroundHandler);
224 Surface surface = mImageReader.getSurface();
225
226 mPreviewRequestBuilder = mCameraDevice.createCaptureRequest(mRequestTemplate);
227 mPreviewRequestBuilder.addTarget(surface);

229 mCameraDevice.createCaptureSession(Arrays.asList(surface),

274 Log.i(LOGTAG, "calcPreviewSize: " + width + "x" + height);

285 Size frameSize = calculateCameraFrameSize(sizes_list, new JavaCameraSizeAccessor(), width, height);

355 @Override
356 public Mat gray() {
357 Image.Plane[] planes = mImage.getPlanes();
358 int w = mImage.getWidth();
359 int h = mImage.getHeight();
360 assert(planes[0].getPixelStride() == 1);
361 ByteBuffer y_plane = planes[0].getBuffer();
362 int y_plane_step = planes[0].getRowStride();
363 mGray = new Mat(h, w, CvType.CV_8UC1, y_plane, y_plane_step);
364 return mGray;
365 }
366
367 @Override
368 public Mat rgba() {
369 Image.Plane[] planes = mImage.getPlanes();
370 int w = mImage.getWidth();
371 int h = mImage.getHeight();
372 int chromaPixelStride = planes[1].getPixelStride();
373
374
375 if (chromaPixelStride == 2) { // Chroma channels are interleaved
376 assert(planes[0].getPixelStride() == 1);
377 assert(planes[2].getPixelStride() == 2);
378 ByteBuffer y_plane = planes[0].getBuffer();
379 int y_plane_step = planes[0].getRowStride();
380 ByteBuffer uv_plane1 = planes[1].getBuffer();
381 int uv_plane1_step = planes[1].getRowStride();
382 ByteBuffer uv_plane2 = planes[2].getBuffer();
383 int uv_plane2_step = planes[2].getRowStride();
384 Mat y_mat = new Mat(h, w, CvType.CV_8UC1, y_plane, y_plane_step);
385 Mat uv_mat1 = new Mat(h / 2, w / 2, CvType.CV_8UC2, uv_plane1, uv_plane1_step);
386 Mat uv_mat2 = new Mat(h / 2, w / 2, CvType.CV_8UC2, uv_plane2, uv_plane2_step);
387 long addr_diff = uv_mat2.dataAddr() - uv_mat1.dataAddr();
388 if (addr_diff > 0) {
389 assert(addr_diff == 1);
390 Imgproc.cvtColorTwoPlane(y_mat, uv_mat1, mRgba, Imgproc.COLOR_YUV2RGBA_NV12);
391 } else {
392 assert(addr_diff == -1);
393 Imgproc.cvtColorTwoPlane(y_mat, uv_mat2, mRgba, Imgproc.COLOR_YUV2RGBA_NV21);
394 }
395 return mRgba;
396 } else { // Chroma channels are not interleaved
397 byte[] yuv_bytes = new byte[w*(h+h/2)];
398 ByteBuffer y_plane = planes[0].getBuffer();
399 ByteBuffer u_plane = planes[1].getBuffer();
400 ByteBuffer v_plane = planes[2].getBuffer();
401
402 int yuv_bytes_offset = 0;
403
404 int y_plane_step = planes[0].getRowStride();
405 if (y_plane_step == w) {
406 y_plane.get(yuv_bytes, 0, w*h);
407 yuv_bytes_offset = w*h;
408 } else {
409 int padding = y_plane_step - w;
410 for (int i = 0; i < h; i++){
411 y_plane.get(yuv_bytes, yuv_bytes_offset, w);
412 yuv_bytes_offset += w;
413 if (i < h - 1) {
414 y_plane.position(y_plane.position() + padding);
415 }
416 }
417 assert(yuv_bytes_offset == w * h);
418 }
419
420 int chromaRowStride = planes[1].getRowStride();
421 int chromaRowPadding = chromaRowStride - w/2;
422
423 if (chromaRowPadding == 0){
424 // When the row stride of the chroma channels equals their width, we can copy
425 // the entire channels in one go
426 u_plane.get(yuv_bytes, yuv_bytes_offset, w*h/4);
427 yuv_bytes_offset += w*h/4;
428 v_plane.get(yuv_bytes, yuv_bytes_offset, w*h/4);
429 } else {
430 // When not equal, we need to copy the channels row by row
431 for (int i = 0; i < h/2; i++){
432 u_plane.get(yuv_bytes, yuv_bytes_offset, w/2);
433 yuv_bytes_offset += w/2;
434 if (i < h/2-1){
435 u_plane.position(u_plane.position() + chromaRowPadding);
436 }
437 }
438 for (int i = 0; i < h/2; i++){
439 v_plane.get(yuv_bytes, yuv_bytes_offset, w/2);
440 yuv_bytes_offset += w/2;
441 if (i < h/2-1){
442 v_plane.position(v_plane.position() + chromaRowPadding);
443 }
444 }
445 }
446
447 Mat yuv_mat = new Mat(h+h/2, w, CvType.CV_8UC1);
448 yuv_mat.put(0, 0, yuv_bytes);
449 Imgproc.cvtColor(yuv_mat, mRgba, Imgproc.COLOR_YUV2RGBA_I420, 4);
450 return mRgba;
451 }
452 }
