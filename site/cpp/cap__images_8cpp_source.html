109bool CvCapture_Images::grabFrame()
110{
111 if (length == 1 && currentframe >= length)
112 return false;
113 const cv::String filename = cv::format(filename_pattern.c_str(), (int)(firstframe + currentframe));
114 CV_Assert(!filename.empty());
115
116 if (grabbedInOpen)
117 {
118 grabbedInOpen = false;
119 ++currentframe;
120
121 return !frame.empty();
122 }
123
124 frame = imread(filename, IMREAD_UNCHANGED);
125 if( !frame.empty() )
126 currentframe++;
127
128 return !frame.empty();
129}

138double CvCapture_Images::getProperty(int id) const
139{
140 switch(id)
141 {
142 case cv::CAP_PROP_POS_MSEC:
143 CV_WARN("collections of images don't have framerates");
144 return CAP_PROP_UNKNOWN;
145 case cv::CAP_PROP_POS_FRAMES:
146 return currentframe;
147 case cv::CAP_PROP_FRAME_COUNT:
148 return length;
149 case cv::CAP_PROP_POS_AVI_RATIO:
150 return (double)currentframe / (double)(length - 1);
151 case cv::CAP_PROP_FRAME_WIDTH:
152 return frame.cols;
153 case cv::CAP_PROP_FRAME_HEIGHT:
154 return frame.rows;
155 case cv::CAP_PROP_FPS:
156 CV_WARN("collections of images don't have framerates");
157 return CAP_PROP_UNKNOWN;
158 case cv::CAP_PROP_FOURCC:
159 CV_WARN("collections of images don't have 4-character codes");
160 return CAP_PROP_UNKNOWN;
161 }
162 return CAP_PROP_UNKNOWN;
163}

165bool CvCapture_Images::setProperty(int id, double value)
166{
167 switch(id)
168 {
169 case cv::CAP_PROP_POS_MSEC:
170 case cv::CAP_PROP_POS_FRAMES:
171 if(value < 0) {
172 CV_WARN("seeking to negative positions does not work - clamping");
173 value = 0;
174 }
175 if(value >= length) {
176 CV_WARN("seeking beyond end of sequence - clamping");
177 value = length - 1;
178 }
179 currentframe = cvRound(value);
180 if (currentframe != 0)
181 grabbedInOpen = false; // grabbed frame is not valid anymore
182 return true;
183 case cv::CAP_PROP_POS_AVI_RATIO:
184 if(value > 1) {
185 CV_WARN("seeking beyond end of sequence - clamping");
186 value = 1;
187 } else if(value < 0) {
188 CV_WARN("seeking to negative positions does not work - clamping");
189 value = 0;
190 }
191 currentframe = cvRound((length - 1) * value);
192 if (currentframe != 0)
193 grabbedInOpen = false; // grabbed frame is not valid anymore
194 return true;
195 }
196 CV_WARN("unknown/unhandled property");
197 return false;
198}

200// static
201std::string icvExtractPattern(const std::string& filename, unsigned *offset)
202{
203 size_t len = filename.size();
204 CV_Assert(!filename.empty());
205 CV_Assert(offset);
206
207 *offset = 0;
208
209 // check whether this is a valid image sequence filename
210 std::string::size_type pos = filename.find('%');
211 if (pos != std::string::npos)
212 {
213 pos++; CV_Assert(pos < len);
214 if (filename[pos] == '0') // optional zero prefix
215 {
216 pos++; CV_Assert(pos < len);
217 }
218 if (filename[pos] >= '1' && filename[pos] <= '9') // optional numeric size (1..9) (one symbol only)
219 {
220 pos++; CV_Assert(pos < len);
221 }
222 if (filename[pos] == 'd' || filename[pos] == 'u')
223 {
224 pos++;
225 if (pos == len)
226 return filename; // end of string '...%5d'
227 CV_Assert(pos < len);
228 if (filename.find('%', pos) == std::string::npos)
229 return filename; // no more patterns
230 CV_Error_(Error::StsBadArg, ("CAP_IMAGES: invalid multiple patterns: %s", filename.c_str()));
231 }
232 CV_Error_(Error::StsBadArg, ("CAP_IMAGES: error, expected '0?[1-9][du]' pattern, got: %s", filename.c_str()));
233 }
234 else // no pattern filename was given - extract the pattern
235 {
236 pos = filename.rfind('/');
237#ifdef _WIN32
238 if (pos == std::string::npos)
239 pos = filename.rfind('\\');
240#endif
241 if (pos != std::string::npos)
242 pos++;
243 else
244 pos = 0;
245
246 while (pos < len && !isdigit(filename[pos])) pos++;
247
248 if (pos == len)
249 return "";
250
251 std::string::size_type pos0 = pos;
252
253 const int64_t max_number = 1000000000;
254 CV_Assert(max_number < INT_MAX); // offset is 'int'
255
256 int number_str_size = 0;
257 uint64_t number = 0;
258 while (pos < len && isdigit(filename[pos]))
259 {
260 char ch = filename[pos];
261 number = (number * 10) + (uint64_t)((int)ch - (int)'0');
262 CV_Assert(number < max_number);
263 number_str_size++;
264 CV_Assert(number_str_size <= 64); // don't allow huge zero prefixes
265 pos++;
266 }
267 CV_Assert(number_str_size > 0);
268
269 *offset = (int)number;
270
271 std::string result;
272 if (pos0 > 0)
273 result += filename.substr(0, pos0);
274 result += cv::format("%%0%dd", number_str_size);
275 if (pos < len)
276 result += filename.substr(pos);
277 CV_LOG_INFO(NULL, "Pattern: " << result << " @ " << number);
278 return result;
279 }

283bool CvCapture_Images::open(const std::string& _filename, int image_seq_start)
284{
285 unsigned offset = 0;
286 close();
287
288 CV_Assert(!_filename.empty());
289 filename_pattern = icvExtractPattern(_filename, &offset);
290 const bool explicit_start = (image_seq_start >= 0);
291 if (explicit_start)
292 offset = static_cast<unsigned>(image_seq_start);
293 if (filename_pattern.empty())
294 {
295 filename_pattern = _filename;
296 if (!utils::fs::exists(filename_pattern))
297 {
298 CV_LOG_INFO(NULL, "CAP_IMAGES: File does not exist: " << filename_pattern);
299 close();
300 return false;
301 }
302 if (!haveImageReader(filename_pattern))
303 {
304 CV_LOG_INFO(NULL, "CAP_IMAGES: File is not an image: " << filename_pattern);
305 close();
306 return false;
307 }
308 length = 1;
309 }
310 else
311 {
312 // determine the length of the sequence
313 for (length = 0; ;)
314 {
315 cv::String filename = cv::format(filename_pattern.c_str(), (int)(offset + length));
316 if (!utils::fs::exists(filename))
317 {
318 if (length == 0 && offset == 0 && !explicit_start) // allow starting with 0 or 1
319 {
320 offset++;
321 continue;
322 }
323 CV_LOG_INFO(NULL, "CAP_IMAGES: File does not exist: " << filename);
324 break;
325 }
326
327 if(!haveImageReader(filename))
328 {
329 CV_LOG_INFO(NULL, "CAP_IMAGES: File is not an image: " << filename);
330 break;
331 }
332
333 length++;
334 }
335
336 if (length == 0)
337 {
338 close();
339 return false;
340 }
341
342 firstframe = offset;
343 }
344 // grab frame to enable properties retrieval
345 bool grabRes = CvCapture_Images::grabFrame();
346 grabbedInOpen = true;
347 currentframe = 0;
348 return grabRes;
349}

356Ptr<IVideoCapture> create_Images_capture(const std::string& filename, const VideoCaptureParameters& params)
357{
358 int image_seq_start = -1;
359 if (params.has(CAP_PROP_IMAGE_SEQ_START))
360 {
361 image_seq_start = params.get<int>(CAP_PROP_IMAGE_SEQ_START);
362 }
363 if (params.warnUnusedParameters())
364 {
365 CV_LOG_ERROR(NULL, "VIDEOIO/IMAGES: unsupported parameters in .open(), see logger INFO channel for details. Bailout");
366 return Ptr<IVideoCapture>();
367 }
368 Ptr<CvCapture_Images> cap = makePtr<CvCapture_Images>();
369 if (!cap->open(filename, image_seq_start))
370 return Ptr<IVideoCapture>();
371 return cap;
372}
