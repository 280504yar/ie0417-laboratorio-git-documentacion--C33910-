20static PointCloudDecoder findDecoder(const String &filename)
21{
22 auto file_ext = getExtension(filename);

24 {
25 return std::unique_ptr<ObjDecoder>(new ObjDecoder());
26 }
27 if (file_ext == "ply" || file_ext == "PLY")

29 return std::unique_ptr<PlyDecoder>(new PlyDecoder());

52void loadPointCloud(const String &filename, OutputArray vertices, OutputArray normals, OutputArray rgb)
53{
54#if OPENCV_HAVE_FILESYSTEM_SUPPORT
55 auto decoder = findDecoder(filename);
56 if (!decoder) {
57 String file_ext = getExtension(filename);
58 CV_LOG_ERROR(NULL, "File extension '" << file_ext << "' is not supported");
59 return;
60 }
61
62 decoder->setSource(filename);
63
64 std::vector<Point3f> vec_vertices, vec_normals, vec_rgb;
65
66 decoder->readData(vec_vertices, vec_normals, vec_rgb);
67
68 if (!vec_vertices.empty())
69 Mat(static_cast<int>(vec_vertices.size()), 1, CV_32FC3, vec_vertices.data()).copyTo(vertices);
70
71 if (!vec_normals.empty() && normals.needed())
72 Mat(static_cast<int>(vec_normals.size()), 1, CV_32FC3, vec_normals.data()).copyTo(normals);
73
74 if (!vec_rgb.empty() && rgb.needed())
75 Mat(static_cast<int>(vec_rgb.size()), 1, CV_32FC3, vec_rgb.data()).copyTo(rgb);
76
77#else // OPENCV_HAVE_FILESYSTEM_SUPPORT
78 CV_UNUSED(filename);
79 CV_UNUSED(vertices);
80 CV_UNUSED(normals);
81 CV_UNUSED(rgb);
82 CV_LOG_WARNING(NULL, "File system support is disabled in this OpenCV build!");

125void loadMesh(const String &filename, OutputArray vertices, OutputArrayOfArrays indices,
126 OutputArray normals, OutputArray colors, OutputArray texCoords)
127{
128#if OPENCV_HAVE_FILESYSTEM_SUPPORT
129 CV_Assert(vertices.needed());
130 CV_Assert(indices.needed());
131
132 PointCloudDecoder decoder = findDecoder(filename);
133 String file_ext = getExtension(filename);
134 if (!decoder) {
135 CV_LOG_ERROR(NULL, "File extension '" << file_ext << "' is not supported");
136 return;
137 }
138
139 decoder->setSource(filename);
140
141 std::vector<Point3f> vec_vertices, vec_normals, vec_rgb;
142 std::vector<std::vector<int32_t>> vec_indices;
143
144 std::vector<Point3f> vec_texCoords;
145 int nTexCoords = 0;
146
147 decoder->readData(vec_vertices, vec_normals, vec_rgb, vec_texCoords, nTexCoords, vec_indices, 0);
148
149 if (!vec_vertices.empty())
150 {
151 Mat(1, static_cast<int>(vec_vertices.size()), CV_32FC3, vec_vertices.data()).copyTo(vertices);
152 }
153
154 if (normals.needed() && !vec_normals.empty())
155 {
156 Mat(1, static_cast<int>(vec_normals.size()), CV_32FC3, vec_normals.data()).copyTo(normals);
157 }
158
159 if (colors.needed() && !vec_rgb.empty())
160 {
161 Mat(1, static_cast<int>(vec_rgb.size()), CV_32FC3, vec_rgb.data()).copyTo(colors);
162 }
163
164 if (!vec_indices.empty())
165 {
166 _InputArray::KindFlag kind = indices.kind();
167 int vecsz = (int)vec_indices.size();
168 if (kind == _InputArray::KindFlag::STD_VECTOR_VECTOR)
169 {
170 CV_Assert(indices.depth() == CV_32S);
171 std::vector<std::vector<int32_t>>& vec = *(std::vector<std::vector<int32_t>>*)indices.getObj();
172 vec.resize(vecsz);
173 for (int i = 0; i < vecsz; ++i)
174 {
175 Mat(1, static_cast<int>(vec_indices[i].size()), CV_32SC1, vec_indices[i].data()).copyTo(vec[i]);
176 }
177 }
178 // std::array<Mat> has fixed size, unsupported
179 else if (kind == _InputArray::KindFlag::STD_VECTOR_MAT)
180 {
181 indices.create(vecsz, 1, CV_32S);
182 for (int i = 0; i < vecsz; i++)
183 {
184 std::vector<int> vi = vec_indices[i];
185 indices.create(1, (int)vi.size(), CV_32S, i);
186 Mat(vi).copyTo(indices.getMat(i));
187 }
188 }
189 else
190 {
191 std::vector<Vec3i> vec(vec_indices.size());
192 for (int i = 0; i < vecsz; ++i)
193 {
194 Vec3i tri;
195 size_t sz = vec_indices[i].size();
196 if (sz != 3)
197 {
198 CV_Error(Error::StsBadArg, "Face contains " + std::to_string(sz) + " vertices, can not put it into 3-channel indices array");
199 }
200 else
201 {
202 for (int j = 0; j < 3; j++)
203 {
204 tri[j] = vec_indices[i][j];
205 }
206 }
207 vec[i] = tri;
208 }
209 indices.create(1, (int)vec_indices.size(), CV_32SC3);
210 Mat(1, static_cast<int>(vec_indices.size()), CV_32SC3, vec.data()).copyTo(indices);
211 }
212 }
213
214 if (texCoords.needed())
215 {
216 if (nTexCoords)
217 {
218 CV_Assert(!texCoords.fixedType() || (texCoords.type() == CV_MAKE_TYPE(CV_32F, nTexCoords)));
219
220 Mat tex3(vec_texCoords);
221
222 if (nTexCoords == 3)
223 {
224 tex3.copyTo(texCoords);
225 }
226 else if (nTexCoords == 2)
227 {
228 // if texCoords is empty then channels() can be any number
229 bool has3ch = texCoords.channels() == 3;
230 int ch = has3ch ? 3 : 2;
231 std::vector<int> permut = has3ch ? std::vector<int>{ 0, 0, 1, 1, -1, 2 } : std::vector<int>{ 0, 0, 1, 1 };
232 texCoords.createSameSize(vec_texCoords, CV_MAKE_TYPE(CV_32F, ch));
233 Mat out = texCoords.getMat();
234 cv::mixChannels(tex3, out, permut);
235 }
236 }
237 else
238 {
239 texCoords.clear();
240 }
241 }
242
243#else // OPENCV_HAVE_FILESYSTEM_SUPPORT
244 CV_UNUSED(filename);
245 CV_UNUSED(vertices);
246 CV_UNUSED(normals);
247 CV_UNUSED(colors);
248 CV_UNUSED(indices);
249 CV_UNUSED(texCoords);
250 CV_LOG_WARNING(NULL, "File system support is disabled in this OpenCV build!");
251#endif
252}

254void saveMesh(const String &filename, InputArray vertices, InputArrayOfArrays indices,
255 InputArray normals, InputArray colors, InputArray texCoords)
256{
257#if OPENCV_HAVE_FILESYSTEM_SUPPORT
258 if (vertices.empty()) {
259 CV_LOG_WARNING(NULL, "Have no vertices to save");
260 return;
261 }
262
263 auto encoder = findEncoder(filename);
264 String file_ext = getExtension(filename);
265 if (!encoder) {
266 CV_LOG_ERROR(NULL, "File extension '" << file_ext << "' is not supported");
267 return;
268 }
269
270 encoder->setDestination(filename);
271
272 std::vector<Point3f> vec_vertices(vertices.getMat()), vec_normals, vec_rgb;
273 if (!normals.empty())
274 {
275 vec_normals = normals.getMat();
276 }
277
278 if (!colors.empty())
279 {
280 vec_rgb = colors.getMat();
281 }
282
283 std::vector<std::vector<int32_t>> vec_indices;
284 CV_Assert(indices.depth() == CV_32S);
285 if (indices.kind() == _InputArray::KindFlag::STD_VECTOR_VECTOR ||
286 indices.kind() == _InputArray::KindFlag::STD_VECTOR_MAT)
287 {
288 std::vector<Mat> mat_indices;
289 indices.getMatVector(mat_indices);
290 vec_indices.resize(mat_indices.size());
291 for (size_t i = 0; i < mat_indices.size(); ++i)
292 {
293 mat_indices[i].copyTo(vec_indices[i]);
294 }
295 }
296 else
297 {
298 CV_Assert(indices.channels() == 3);
299 std::vector<Vec3i>& vec = *(std::vector<Vec3i>*)indices.getObj();
300 vec_indices.resize(vec.size());
301 for (size_t i = 0; i < vec.size(); ++i)
302 {
303 for (int j = 0; j < 3; j++)
304 {
305 vec_indices[i].push_back(vec[i][j]);
306 }
307 }
308 }
309
310 std::vector<Point3f> vec_texCoords;
311 int nTexCoords = 0;
312 if (!texCoords.empty())
313 {
314 nTexCoords = texCoords.channels();
315 }
316 if (nTexCoords == 2)
317 {
318 // extend by 3rd zero channel
319 vec_texCoords.resize(texCoords.total());
320 cv::mixChannels(texCoords, vec_texCoords, {0, 0, 1, 1, -1, 2});
321 }
322 if (nTexCoords == 3)
323 {
324 texCoords.copyTo(vec_texCoords);
325 }
326
327 encoder->writeData(vec_vertices, vec_normals, vec_rgb, vec_texCoords, nTexCoords, vec_indices);
328
329#else // OPENCV_HAVE_FILESYSTEM_SUPPORT
330 CV_UNUSED(filename);
331 CV_UNUSED(vertices);
332 CV_UNUSED(colors);
333 CV_UNUSED(normals);
334 CV_UNUSED(indices);
335 CV_UNUSED(texCoords);
336 CV_LOG_WARNING(NULL, "File system support is disabled in this OpenCV build!");
337#endif
338
339}
