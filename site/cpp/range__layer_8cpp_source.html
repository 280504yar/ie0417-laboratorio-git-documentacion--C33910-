21static int rangeSize(double start, double limit, double delta)
22{
23 return std::max((int)ceil((limit - start)/delta), 0);
24}
25
26static int rangeSize(int64_t start, int64_t limit, int64_t delta)
27{
28 return delta > 0 ?
29 std::max((int)((limit - start + delta - 1)/delta), 0) :
30 std::max((int)((start - limit - delta - 1)/-delta), 0);
31}
32
33// out must be pre-allocated
34template <typename _Tp>
35static void makeRange(_Tp start, _Tp limit, _Tp delta, Mat& out)
36{
37 int nout = rangeSize(start, limit, delta);
38 CV_Assert(out.dims == 1);
39 CV_Assert(out.total() == (size_t)nout);
40 uchar* outdata_ = out.data;
41
42 int type = out.type();
43
44 #undef IMPL_RANGE
45 #define IMPL_RANGE(T) \
46 T* outdata = (T*)outdata_; \
47 for (int i = 0; i < nout; i++) \
48 outdata[i] = saturate_cast<T>(start + i*delta)
49
50 if (type == CV_32F) {
51 IMPL_RANGE(float);
52 } else if (type == CV_64F) {
53 IMPL_RANGE(double);
54 } else if (type == CV_32S) {
55 IMPL_RANGE(int32_t);
56 } else if (type == CV_64S) {
57 IMPL_RANGE(int64_t);
58 } else if (type == CV_16S) {
59 IMPL_RANGE(int16_t);
60 } else {
61 CV_Error_(Error::StsNotImplemented, ("invalid/unsupported tensor type: %s", typeToString(out.type()).c_str()));
62 }
63}

88 int getRangeParams(const Mat& startTensor, const Mat& limitTensor, const Mat& deltaTensor,
89 double& fstart, double& flimit, double& fdelta,
90 int64_t& istart, int64_t& ilimit, int64_t& idelta, bool& isflt) const
91 {
92 CV_Assert(startTensor.total() == (size_t)1);
93 CV_Assert(limitTensor.total() == (size_t)1);
94 CV_Assert(deltaTensor.total() == (size_t)1);
95
96 int rtype = startTensor.type();
97 CV_Assert(rtype == limitTensor.type());
98 CV_Assert(rtype == deltaTensor.type());
99
100 fstart = flimit = fdelta = 0.;
101 istart = ilimit = idelta = 0;
102
103 isflt = rtype == CV_32F || rtype == CV_64F || rtype == CV_16F || rtype == CV_16BF;
104
105 if (isflt) {
106 fstart = tensorToScalar<double>(startTensor);
107 flimit = tensorToScalar<double>(limitTensor);
108 fdelta = tensorToScalar<double>(deltaTensor);
109
110 return rangeSize(fstart, flimit, fdelta);
111 } else {
112 istart = tensorToScalar<int64_t>(startTensor);
113 ilimit = tensorToScalar<int64_t>(limitTensor);
114 idelta = tensorToScalar<int64_t>(deltaTensor);
115
116 return rangeSize(istart, ilimit, idelta);
117 }
118 }

120 bool getMemoryShapes(const std::vector<MatShape>& inputs,
121 const int,
122 std::vector<MatShape> &outputs,
123 std::vector<MatShape> &internals) const CV_OVERRIDE
124 {
125 CV_Assert(!dynamicOutputShapes());

127 CV_Assert(inputs.size() == (size_t)3);
128 CV_Assert(inputs.size() == this->inputs.size());
129 Net::Impl* netimpl_ = getNetImpl(this);

133 Mat deltaTensor = netimpl_->argTensor(this->inputs[2]);
134

140 fstart, flimit, fdelta, istart, ilimit, idelta, isflt);
141 MatShape shape(1);
142 shape[0] = nout;
143 outputs.assign(1, shape);
144 internals.clear();

167 void forward(InputArrayOfArrays inputs_arr,
168 OutputArrayOfArrays outputs_arr,
169 OutputArrayOfArrays) CV_OVERRIDE
170 {
171 CV_TRACE_FUNCTION();
172 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
173
174 Size size = inputs_arr.size();
175 int ninputs = size.area();
176 CV_Assert(ninputs == 3);
177
178 Mat startTensor = inputs_arr.getMat(0);
179 Mat limitTensor = inputs_arr.getMat(1);
180 Mat deltaTensor = inputs_arr.getMat(2);
181
182 double fstart, flimit, fdelta;
183 int64_t istart, ilimit, idelta;
184 bool isflt;
185
186 int nout = getRangeParams(startTensor, limitTensor, deltaTensor,
187 fstart, flimit, fdelta, istart, ilimit, idelta, isflt);
188 MatShape shape(1);
189 shape[0] = nout;
190
191 int rtype = startTensor.type();
192
193 auto kind = outputs_arr.kind();
194 if (kind == _InputArray::STD_VECTOR_MAT) {
195 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
196 outs.resize(1);
197 outs[0].fit(shape, rtype);
198 if (isflt) {
199 makeRange(fstart, flimit, fdelta, outs[0]);
200 } else {
201 makeRange(istart, ilimit, idelta, outs[0]);
202 }
203 } else if (kind == _InputArray::STD_VECTOR_UMAT) {
204 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
205 outs.resize(1);
206 outs[0].fit(shape, rtype);
207 Mat temp(shape, rtype);
208 if (isflt) {
209 makeRange(fstart, flimit, fdelta, temp);
210 } else {
211 makeRange(istart, ilimit, idelta, temp);
212 }
213 temp.copyTo(outs[0]);
214 } else {
215 CV_Error(Error::StsNotImplemented, "");
216 }
217 }
