111 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
112 {
113 CV_TRACE_FUNCTION();
114 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
115
116 std::vector<Mat> inputs, outputs;
117 inputs_arr.getMatVector(inputs);
118 outputs_arr.getMatVector(outputs);
119 for (size_t i = 0; i < outputs.size(); i++)
120 {
121 CV_Assert(inputs[0].total() == outputs[i].total());
122 inputs[0].copyTo(outputs[i]);
123 }
124 }
