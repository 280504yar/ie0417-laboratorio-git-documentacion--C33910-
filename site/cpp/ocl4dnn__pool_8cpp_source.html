50template<typename Dtype>
51OCL4DNNPool<Dtype>::OCL4DNNPool(OCL4DNNPoolConfig config)
52{
53 int dims = config.in_shape.size();
54 int spatial_dims = config.in_shape.size()-2;
55
56 channels_ = config.channels;
57 pool_method_ = config.pool_method;
58 avePoolPaddedArea = config.avePoolPaddedArea;
59 computeMaxIdx = config.computeMaxIdx;
60 use_half = config.use_half;
61 kernel_shape_.push_back(config.kernel.height);
62 kernel_shape_.push_back(config.kernel.width);
63 stride_.push_back(config.stride.height);
64 stride_.push_back(config.stride.width);
65
66 for (int i = 0; i < spatial_dims; ++i)
67 {
68 im_in_shape_.push_back(config.in_shape[dims - spatial_dims + i]);
69 im_out_shape_.push_back(config.out_shape[dims - spatial_dims + i]);
70 }
71
72 kernel_h_ = kernel_shape_[0];
73 kernel_w_ = kernel_shape_[1];
74 stride_h_ = stride_[0];
75 stride_w_ = stride_[1];
76 pad_t_ = config.pad_t;
77 pad_l_ = config.pad_l;
78 pad_r_ = config.pad_r;
79 pad_b_ = config.pad_b;
80 height_ = spatial_dims == 1? 1 : im_in_shape_[0];
81 width_ = im_in_shape_.back();
82 pooled_height_ = spatial_dims == 1? 1 : im_out_shape_[0];
83 pooled_width_ = im_out_shape_.back();
84
85 count_ = 1;
86 for (int i = 0; i < config.out_shape.size(); ++i)
87 {
88 count_ *= config.out_shape[i];
89 }
90}

98template<typename Dtype>
99bool OCL4DNNPool<Dtype>::Forward(const UMat& bottom,
100 UMat& top,
101 UMat& top_mask)
102{
103 bool ret = true;
104 size_t global[] = { (size_t)count_ };
105 size_t local[] = { 128 };
106
107 // support 2D case
108 switch (pool_method_)
109 {
110 case LIBDNN_POOLING_METHOD_MAX:
111 {
112 String kname = computeMaxIdx ? "max_pool_forward_mask" : "max_pool_forward";
113 kname += (use_half) ? "_half" : "_float";
114 ocl::Kernel oclk_max_pool_forward(
115 kname.c_str(),
116 ocl::dnn::ocl4dnn_pooling_oclsrc,
117 format(" -D Dtype=%s -D KERNEL_MAX_POOL=1 -D KERNEL_W=%d -D KERNEL_H=%d"
118 " -D STRIDE_W=%d -D STRIDE_H=%d"
119 " -D PAD_L=%d -D PAD_T=%d -D PAD_R=%d -D PAD_B=%d%s",
120 (use_half) ? "half" : "float",
121 kernel_w_, kernel_h_,
122 stride_w_, stride_h_,
123 pad_l_, pad_t_, pad_r_, pad_b_,
124 computeMaxIdx ? " -D HAVE_MASK=1" : ""
125 ));
126 if (oclk_max_pool_forward.empty())
127 return false;
128
129 oclk_max_pool_forward.args(
130 count_,
131 ocl::KernelArg::PtrReadOnly(bottom),
132 channels_,
133 height_,
134 width_,
135 pooled_height_,
136 pooled_width_,
137 ocl::KernelArg::PtrWriteOnly(top)
138 );
139 if (computeMaxIdx)
140 oclk_max_pool_forward.set(8, ocl::KernelArg::PtrWriteOnly(top_mask)); // TODO remove magic number. Extend cv::ocl::Kernel API
141
142 ret = oclk_max_pool_forward.run(1, global, local, false);
143 }
144 break;
145 case LIBDNN_POOLING_METHOD_AVE:
146 {
147 CV_Assert(top_mask.empty());
148
149 String kname = format("ave_pool_forward_%s", (use_half) ? "half" : "float");
150 ocl::Kernel oclk_ave_pool_forward(
151 kname.c_str(),
152 ocl::dnn::ocl4dnn_pooling_oclsrc,
153 format(" -D Dtype=%s -D KERNEL_AVE_POOL=1 -D KERNEL_W=%d -D KERNEL_H=%d"
154 " -D STRIDE_W=%d -D STRIDE_H=%d"
155 " -D PAD_L=%d -D PAD_T=%d -D PAD_R=%d -D PAD_B=%d%s",
156 (use_half) ? "half" : "float",
157 kernel_w_, kernel_h_,
158 stride_w_, stride_h_,
159 pad_l_, pad_t_, pad_r_, pad_b_,
160 avePoolPaddedArea ? " -D AVE_POOL_PADDING_AREA" : ""
161 ));
162
163 if (oclk_ave_pool_forward.empty())
164 return false;
165
166 oclk_ave_pool_forward.args(
167 count_,
168 ocl::KernelArg::PtrReadOnly(bottom),
169 channels_,
170 height_,
171 width_,
172 pooled_height_,
173 pooled_width_,
174 ocl::KernelArg::PtrWriteOnly(top)
175 );
176
177 ret = oclk_ave_pool_forward.run(1, global, local, false);
178 }
179 break;
180 case LIBDNN_POOLING_METHOD_STO:
181 {
182 CV_Assert(top_mask.empty());
183
184 String kname = format("sto_pool_forward_test_%s", (use_half) ? "half" : "float");
185 ocl::Kernel oclk_sto_pool_forward(
186 kname.c_str(),
187 ocl::dnn::ocl4dnn_pooling_oclsrc,
188 format(" -D Dtype=%s -D KERNEL_STO_POOL=1 -D KERNEL_W=%d -D KERNEL_H=%d"
189 " -D STRIDE_W=%d -D STRIDE_H=%d",
190 (use_half) ? "half" : "float",
191 kernel_w_, kernel_h_,
192 stride_w_, stride_h_
193 ));
194
195
196 if (oclk_sto_pool_forward.empty())
197 return false;
198
199 oclk_sto_pool_forward.args(
200 count_,
201 ocl::KernelArg::PtrReadOnly(bottom),
202 channels_,
203 height_,
204 width_,
205 pooled_height_,
206 pooled_width_,
207 ocl::KernelArg::PtrWriteOnly(top)
208 );
209
210 ret = oclk_sto_pool_forward.run(1, global, local, false);
211 }
212 break;
213 default:
214 {
215 ret = false;
216 LOG(FATAL)<< "Unknown pooling method.";
217 }
218 }
219 return ret;
220}
