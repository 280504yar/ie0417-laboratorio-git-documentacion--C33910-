21GpuMatND GpuMatND::operator()(const std::vector<Range>& ranges) const
22{
23 CV_Assert(dims == (int)ranges.size());
24
25 for (int i = 0; i < dims; ++i)
26 {
27 Range r = ranges[i];
28 CV_Assert(r == Range::all() || (0 <= r.start && r.start < r.end && r.end <= size[i]));
29 }
30
31 GpuMatND ret = *this;
32
33 for (int i = 0; i < dims; ++i)
34 {
35 Range r = ranges[i];
36 if (r != Range::all() && r != Range(0, ret.size[i]))
37 {
38 ret.offset += r.start * ret.step[i];
39 ret.size[i] = r.size();
40 ret.flags |= Mat::SUBMATRIX_FLAG;
41 }
42 }
43
44 ret.flags = cv::updateContinuityFlag(ret.flags, dims, ret.size.data(), ret.step.data());
45
46 return ret;
47}

62GpuMat GpuMatND::createGpuMatHeader() const
63{
64 auto Effectively2D = [](GpuMatND m)
65 {
66 for (int i = 0; i < m.dims - 2; ++i)
67 if (m.size[i] > 1)
68 return false;
69 return true;
70 };
71 CV_Assert(Effectively2D(*this));
72 int rows_, cols_ = dims >= 1 ? size[dims - 1] : 1;
73 size_t step_;
74 if (dims >= 2)
75 {
76 rows_ = size[dims - 2];
77 step_ = step[dims - 2];
78 }
79 else
80 {
81 rows_ = 1;
82 step_ = cols_ * elemSize();
83 }
84
85 return GpuMat(rows_, cols_, type(), getDevicePtr(), step_);
86}

98void GpuMatND::setFields(MatShape _size, int _type, StepArray _step)
99{
100 _type &= Mat::TYPE_MASK;
101
102 flags = Mat::MAGIC_VAL + _type;
103 dims = static_cast<int>(_size.size());
104 size = std::move(_size);
105
106 if (_step.empty())
107 {
108 step = StepArray(dims);
109
110 step.back() = elemSize();
111 for (int _i = dims - 2; _i >= 0; --_i)
112 {
113 const size_t i = _i;
114 step[i] = step[i+1] * size[i+1];
115 }
116
117 flags |= Mat::CONTINUOUS_FLAG;
118 }
119 else
120 {
121 step = std::move(_step);
122 if (step.size() < size.size())
123 step.push_back(elemSize());
124
125 flags = cv::updateContinuityFlag(flags, dims, size.data(), step.data());
126 }
127
128 CV_Assert(size.size() == step.size());
129 CV_Assert(step.back() == elemSize());
130}

194void GpuMatND::fit(const MatShape& _shape, int _type)
195{
196 auto elements_nonzero = [](const MatShape& v)
197 {
198 return std::all_of(v.begin(), v.end(), [](int u){ return u > 0; });
199 };
200 CV_Assert(!_shape.empty());
201 CV_Assert(elements_nonzero(_shape));
202
203 _type &= Mat::TYPE_MASK;
204
205 const size_t esz = (size_t)CV_ELEM_SIZE(_type);
206 size_t step0 = esz;
207 for (int i = (int)_shape.size() - 2; i >= 0; --i)
208 step0 *= (size_t)_shape[(size_t)i + 1];
209 const size_t requiredBytes = (size_t)_shape[0] * step0;
210
211 const bool canReuse = !empty() && !external() && isContinuous() && !isSubmatrix() && offset == 0;
212 const size_t oldCapacity = canReuse ? totalMemSize() : 0;
213
214 if (!canReuse || requiredBytes > oldCapacity)
215 {
216 create(_shape, _type);
217 return;
218 }
219
220 setFields(_shape, _type);
221}
