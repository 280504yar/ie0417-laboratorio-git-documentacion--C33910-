21template<typename T, typename ComplexVec, typename FillFn>
22static void dftAlongAxisWorker(const Mat& src,
23 Mat& dst,
24 const std::vector<int>& dimSizesSrc,
25 const std::vector<size_t>& stridesSrc,
26 const std::vector<size_t>& stridesDst,
27 const std::vector<int>& iterDims,
28 const std::vector<int>& outerSizes,
29 const std::vector<size_t>& outerStep,
30 const size_t totalOuter,
31 const int axis,
32 const int N,
33 const int outN,
34 const size_t strideAxisSrc,
35 const size_t strideAxisDst,
36 const int inMatType,
37 const int flagsBase,
38 const bool inverse,
39 FillFn&& fill)
40{
41 const T* sp = src.ptr<T>();
42 T* dp = dst.ptr<T>();
43 cv::parallel_for_(Range(0, (int)totalOuter), [&](const Range& r){
44 Mat inRow(1, N, inMatType);
45 Mat outRow;
46 for (int pos = r.start; pos < r.end; ++pos)
47 {
48 size_t baseSrc = 0;
49 size_t baseDst = 0;
50 for (size_t t = 0; t < iterDims.size(); ++t)
51 {
52 int idxVal = outerStep.empty() ? 0 : (int)((pos / outerStep[t]) % (size_t)outerSizes[t]);
53 int d = iterDims[t];
54 baseSrc += (size_t)idxVal * stridesSrc[d];
55 baseDst += (size_t)idxVal * stridesDst[d];
56 }
57 const T* in = sp + baseSrc;
58 T* out = dp + baseDst;
59 fill(inRow, in, dimSizesSrc[axis], N, strideAxisSrc);
60 int flags = flagsBase | (inverse ? (DFT_INVERSE | DFT_SCALE) : 0);
61 cv::dft(inRow, outRow, flags);
62 const ComplexVec* p = outRow.ptr<ComplexVec>(0);
63 for (int k = 0; k < outN; ++k)
64 {
65 size_t ok = (size_t)k * strideAxisDst;
66 out[ok + 0] = p[k][0];
67 out[ok + 1] = p[k][1];
68 }
69 }
70 });
71}

73template<typename T, typename ComplexVec>
74static void runTypedDFT(const Mat& src,
75 Mat& dst,
76 const std::vector<int>& dimSizesSrc,
77 const std::vector<size_t>& stridesSrc,
78 const std::vector<size_t>& stridesDst,
79 const std::vector<int>& iterDims,
80 const std::vector<int>& outerSizes,
81 const std::vector<size_t>& outerStep,
82 const size_t totalOuter,
83 const int axis,
84 const int N,
85 const int outN,
86 const size_t strideAxisSrc,
87 const size_t strideAxisDst,
88 const bool srcHasComplex,
89 const bool inverse)
90{
91 const int matTypeReal = std::is_same<T, float>::value ? CV_32F : CV_64F;
92 const int matTypeComplex = std::is_same<T, float>::value ? CV_32FC2 : CV_64FC2;
93
94 if (srcHasComplex)
95 {
96 dftAlongAxisWorker<T, ComplexVec>(
97 src, dst, dimSizesSrc, stridesSrc, stridesDst,
98 iterDims, outerSizes, outerStep, totalOuter,
99 axis, N, outN, strideAxisSrc, strideAxisDst,
100 matTypeComplex, 0, inverse,
101 [&](Mat& inRow, const T* in, int origLen, int len, size_t stride){
102 ComplexVec* ptr = inRow.ptr<ComplexVec>(0);
103 for (int n = 0; n < origLen; ++n)
104 {
105 size_t offSrc = (size_t)n * stride;
106 ptr[n][0] = in[offSrc + 0];
107 ptr[n][1] = in[offSrc + 1];
108 }
109 const ComplexVec zeroVal(T(0), T(0));
110 for (int n = origLen; n < len; ++n) ptr[n] = zeroVal;
111 }
112 );
113 }
114 else
115 {
116 dftAlongAxisWorker<T, ComplexVec>(
117 src, dst, dimSizesSrc, stridesSrc, stridesDst,
118 iterDims, outerSizes, outerStep, totalOuter,
119 axis, N, outN, strideAxisSrc, strideAxisDst,
120 matTypeReal, DFT_COMPLEX_OUTPUT, inverse,
121 [&](Mat& inRow, const T* in, int origLen, int len, size_t stride){
122 T* ptr = inRow.ptr<T>(0);
123 for (int n = 0; n < origLen; ++n)
124 {
125 size_t offSrc = (size_t)n * stride;
126 ptr[n] = in[offSrc];
127 }
128 for (int n = origLen; n < len; ++n) ptr[n] = T(0);
129 }
130 );
131 }
132}

144 virtual bool dynamicOutputShapes() const CV_OVERRIDE
145 {
146 if (this->inputs.size() >= 2)
147 {
148 Net::Impl* netimpl_ = getNetImpl(const_cast<DFTLayerImpl*>(this));
149 if (!netimpl_ || !netimpl_->isConstArg(this->inputs[1]))
150 {
151 return true;
152 }
153 }
154 return false;
155 }
156
157private:
158 int getDftLengthFromConstant() const
159 {
160 if (this->inputs.size() < 2)
161 {
162 return -1;
163 }
164
165 Net::Impl* netimpl_ = getNetImpl(const_cast<DFTLayerImpl*>(this));
166 if (!netimpl_)
167 {
168 return -1;
169 }
170
171 Mat dft_length_tensor = netimpl_->argTensor(this->inputs[1]);
172 if (dft_length_tensor.empty() || dft_length_tensor.total() != 1)
173 {
174 return -1;
175 }
176
177 int64_t dft_length64 = 0;
178 tensorToScalar(dft_length_tensor, CV_64S, &dft_length64);
179 if (dft_length64 > 0)
180 {
181 return static_cast<int>(dft_length64);
182 }
183 return -1;
184 }
185
186public:
187 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
188 const int /*requiredOutputs*/,
189 std::vector<MatShape> &outputs,
190 std::vector<MatShape> &/*internals*/) const CV_OVERRIDE
191 {
192 CV_Assert(inputs.size() >= 1);
193 const MatShape &inshape = inputs[0];
194 CV_Assert(!inshape.empty());
195 MatShape out = inshape;
196
197 int last = out.back();
198 if (last == 1)
199 out.back() = 2;
200 else if (last != 2)
201 out.push_back(2);
202
203 int ndims_in = (int)inshape.size();
204 int ax = axis_attr;
205 if (ax == INT_MIN)
206 {
207 ax = (inshape.back() == 2 || inshape.back() == 1) ? ndims_in - 2 : ndims_in - 1;
208 }
209 if (ax < 0) ax += ndims_in;
210 if (ax >= 0 && ax < (int)out.size() - 1)
211 {
212 int dft_length = getDftLengthFromConstant();
213 int signalLen = dft_length > 0 ? dft_length : (ax < (int)inshape.size() ? inshape[ax] : out[ax]);
214 out[ax] = onesided ? (signalLen / 2 + 1) : signalLen;
215 }
216 outputs.assign(1, out);
217 return false;
218 }

220 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays /*internals_arr*/) CV_OVERRIDE
221 {
222 std::vector<Mat> inputs, outputs;
223 inputs_arr.getMatVector(inputs);
224
225 CV_Assert(!inputs.empty());
226 CV_Assert(inputs[0].dims >= 1);
227
228 const Mat &src = inputs[0];
229
230 const int ndims = src.dims;
231 CV_Assert(ndims >= 1);
232 const bool srcHasComplex = (src.size[ndims - 1] == 2);
233 const bool srcLastIsOne = (src.size[ndims - 1] == 1);
234 int axis = (srcHasComplex ? ndims - 2 : ndims - 1);
235 if (axis_attr != INT_MIN)
236 {
237 axis = axis_attr;
238 if (axis < 0) axis += ndims;
239 }
240 else if (!axes.empty())
241 {
242 CV_Assert(axes.size() == 1);
243 axis = axes[0];
244 if (axis < 0) axis += ndims;
245 }
246 CV_Assert(axis >= 0 && axis < (srcHasComplex ? ndims - 1 : ndims));
247 if (onesided)
248 {
249 CV_Assert(!srcHasComplex);
250 CV_Assert(!inverse);
251 }
252
253 int dft_length = -1;
254 if (inputs.size() >= 2 && !inputs[1].empty())
255 {
256 CV_Assert(inputs[1].total() == 1);
257 int64_t dft_length64 = -1;
258 tensorToScalar(inputs[1], CV_64S, &dft_length64);
259 dft_length = static_cast<int>(dft_length64);
260 }
261
262 std::vector<int> outSizesVec;
263 outSizesVec.resize(srcHasComplex ? ndims : ndims + (srcLastIsOne ? 0 : 1));
264 for (int i = 0; i < ndims; ++i) outSizesVec[i] = src.size[i];
265 int complexDim = (int)outSizesVec.size() - 1;
266 outSizesVec[complexDim] = 2;
267 {
268 int dstDimsNoComplex = (int)outSizesVec.size() - 1;
269 if (axis >= 0 && axis < dstDimsNoComplex)
270 {
271 int signalLen = dft_length > 0 ? dft_length : outSizesVec[axis];
272 outSizesVec[axis] = onesided ? (signalLen / 2 + 1) : signalLen;
273 }
274 }
275 MatShape outShape;
276 outShape.assign(outSizesVec.begin(), outSizesVec.end());
277 auto kind = outputs_arr.kind();
278 if (kind == _InputArray::STD_VECTOR_MAT) {
279 outputs_arr.getMatVecRef()[0].fit(outShape, src.type());
280 } else {
281 CV_Assert(kind == _InputArray::STD_VECTOR_UMAT);
282 outputs_arr.getUMatVecRef()[0].fit(outShape, src.type());
283 }
284 outputs_arr.getMatVector(outputs);
285 CV_Assert(outputs.size() == 1);
286 Mat &dst = outputs[0];
287
288 std::vector<int> dimSizesSrc(ndims);
289 for (int i = 0; i < ndims; ++i) {
290 dimSizesSrc[i] = src.size[i];
291 }
292 std::vector<size_t> stridesSrc(ndims, 1);
293 for (int i = ndims - 2; i >= 0; --i) {
294 stridesSrc[i] = stridesSrc[i + 1] * (size_t)dimSizesSrc[i + 1];
295 }
296 const int ndimsDst = (int)outSizesVec.size();
297 std::vector<size_t> stridesDst(ndimsDst, 1);
298 for (int i = ndimsDst - 2; i >= 0; --i) {
299 stridesDst[i] = stridesDst[i + 1] * (size_t)outSizesVec[i + 1];
300 }
301
302 int N = dimSizesSrc[axis];
303 if (dft_length > 0) N = dft_length;
304 int outN = onesided ? (N / 2 + 1) : N;
305 const size_t strideAxisSrc = stridesSrc[axis];
306 const size_t strideAxisDst = stridesDst[axis];
307 std::vector<int> iterDims;
308 for (int i = 0; i < (srcHasComplex ? ndims - 1 : ndims); ++i){
309 if (i != axis){
310 iterDims.push_back(i);
311 }
312 }
313 std::vector<int> outerSizes(iterDims.size(), 0);
314 for (size_t j = 0; j < iterDims.size(); ++j){
315 outerSizes[j] = dimSizesSrc[iterDims[j]];
316 }
317 std::vector<size_t> outerStep(iterDims.size(), 1);
318 for (int j = (int)iterDims.size() - 2; j >= 0; --j){
319 outerStep[j] = outerStep[j + 1] * (size_t)outerSizes[j + 1];
320 }
321 size_t totalOuter = 1;
322 for (int s : outerSizes){
323 totalOuter *= (size_t)s;
324 }
325
326 const int depth = src.depth();
327 if (depth == CV_32F)
328 {
329 runTypedDFT<float, Vec2f>(src, dst, dimSizesSrc, stridesSrc, stridesDst,
330 iterDims, outerSizes, outerStep, totalOuter,
331 axis, N, outN, strideAxisSrc, strideAxisDst,
332 srcHasComplex, inverse);
333 }
334 else if (depth == CV_64F)
335 {
336 runTypedDFT<double, Vec2d>(src, dst, dimSizesSrc, stridesSrc, stridesDst,
337 iterDims, outerSizes, outerStep, totalOuter,
338 axis, N, outN, strideAxisSrc, strideAxisDst,
339 srcHasComplex, inverse);
340 }
341 else
342 {
343 CV_Error(Error::StsNotImplemented, "DFT supports float32/float64 only");
344 }
345 }
