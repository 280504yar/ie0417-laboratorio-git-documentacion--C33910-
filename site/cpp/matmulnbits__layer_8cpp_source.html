12// com.microsoft MatMulNBits (contrib op, since com.microsoft opset 1):
13// https://github.com/microsoft/onnxruntime/blob/main/docs/ContribOperators.md#commicrosoftmatmulnbits
14// Y = A @ dequant(B).T. Weights are block-quantized to {4,8} bits (symmetric, default
15// zero-point 2^(bits-1)) and dequantized per block inside the GEMM (no fp32 copy).

21 MatMulNBitsLayerImpl(const LayerParams& params) {
22 setParamsFrom(params);
23
24 K = params.get<int>("K");
25 N = params.get<int>("N");
26 bits = params.get<int>("bits", 4);
27 block_size = params.get<int>("block_size");
28
29 CV_Check(bits, bits == 4 || bits == 8,
30 "DNN/MatMulNBits: only 4/8-bit weights are supported");
31 CV_Check(block_size, block_size > 0 && (block_size * bits) % 8 == 0,
32 "DNN/MatMulNBits: block_size*bits must be a positive multiple of 8");
33 }

39 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
40 const int,
41 std::vector<MatShape> &outputs,
42 std::vector<MatShape> &internals) const CV_OVERRIDE {
43 CV_CheckEQ(inputs.size(), static_cast<size_t>(1),
44 "DNN/MatMulNBits: expected a single activation input; weights and scales must be constants");
45 CV_CheckEQ(blobs.size(), static_cast<size_t>(2),
46 "DNN/MatMulNBits: expected packed weights and scales as constant blobs");
47
48 const int n_blk = (K + block_size - 1) / block_size; // ceil: K need not divide block_size; last block may be partial
49 CV_CheckEQ(static_cast<int>(blobs[0].total()), N * n_blk * (block_size * bits / 8),
50 "DNN/MatMulNBits: packed weight size mismatch");
51 CV_CheckEQ(static_cast<int>(blobs[1].total()), N * n_blk,
52 "DNN/MatMulNBits: scales size mismatch");
53 CV_CheckType(blobs[0].type(), blobs[0].type() == CV_8U,
54 "DNN/MatMulNBits: packed weights must be CV_8U; int32-packed weights are not supported");
55 CV_CheckType(blobs[1].type(), blobs[1].type() == CV_32F,
56 "DNN/MatMulNBits: scales must be CV_32F");
57
58 MatShape shape_A = inputs[0];
59 CV_CheckGE(shape_A.size(), static_cast<size_t>(1), "DNN/MatMulNBits: invalid shape of input A");
60 CV_CheckEQ(shape_A.back(), K, "DNN/MatMulNBits: last dimension of A must equal K");
61
62 MatShape out = shape_A;
63 out.back() = N;
64 outputs.assign(1, out);
65 internals.clear();
66 return false;
67 }

87 // Dequantize one block into w[0..block_size-1]. BITS is compile-time so the per-byte
88 // unpack unrolls and the shift folds to a constant (the hot half of the kernel).
89 template<int BITS>
90 static inline void dequantBlock(const uchar* packed, int blob_bytes, float scale, float* w) {
91 const int EPB = 8 / BITS; // elements per byte: 4-bit:2 8-bit:1
92 const int QMASK = (1 << BITS) - 1;
93 const int ZP = 1 << (BITS - 1); // symmetric default zero-point
94 for (int j = 0; j < blob_bytes; j++) {
95 const uchar byte = packed[j];
96 for (int e = 0; e < EPB; e++) // LSB-first within the byte
97 w[j * EPB + e] = (((byte >> (e * BITS)) & QMASK) - ZP) * scale;
98 }
99 }
100
101 template<int BITS>
102 void runImpl(const Mat& A, const Mat& B, const Mat& scales, Mat& Y) const {
103 const int M = static_cast<int>(A.total() / K);
104 const int n_blk = (K + block_size - 1) / block_size;
105 const int blob_bytes = block_size * BITS / 8; // packed weights per block
106
107 const float* a = A.ptr<const float>();
108 const uchar* bq = B.ptr<const uchar>();
109 const float* sc = scales.ptr<const float>();
110 float* y = Y.ptr<float>();
111
112 // Each range owns a disjoint set of output columns n, so the column writes are race-free.
113 parallel_for_(Range(0, N), [&](const Range& r) {
114 // Dequant the whole column once so it amortizes across the M rows.
115 AutoBuffer<float, 1024> wcol(n_blk * block_size); // whole blocks (>= K): the last block is dequantized in full
116 for (int n = r.start; n < r.end; n++) {
117 const uchar* bq_n = bq + static_cast<size_t>(n) * n_blk * blob_bytes;
118 const float* sc_n = sc + static_cast<size_t>(n) * n_blk;
119 for (int blk = 0; blk < n_blk; blk++)
120 dequantBlock<BITS>(bq_n + static_cast<size_t>(blk) * blob_bytes, blob_bytes,
121 sc_n[blk], wcol.data() + blk * block_size);
122
123 const float* w = wcol.data();
124 for (int m = 0; m < M; m++) {
125 const float* arow = a + static_cast<size_t>(m) * K;
126 float acc = 0.f;
127 int t = 0;
128#if (CV_SIMD || CV_SIMD_SCALABLE)
129 const int vlanes = VTraits<v_float32>::vlanes();
130 v_float32 vacc0 = vx_setzero_f32(), vacc1 = vx_setzero_f32();
131 v_float32 vacc2 = vx_setzero_f32(), vacc3 = vx_setzero_f32();
132 for (; t <= K - 4 * vlanes; t += 4 * vlanes) {
133 vacc0 = v_fma(vx_load(arow + t), vx_load(w + t), vacc0);
134 vacc1 = v_fma(vx_load(arow + t + vlanes), vx_load(w + t + vlanes), vacc1);
135 vacc2 = v_fma(vx_load(arow + t + 2 * vlanes), vx_load(w + t + 2 * vlanes), vacc2);
136 vacc3 = v_fma(vx_load(arow + t + 3 * vlanes), vx_load(w + t + 3 * vlanes), vacc3);
137 }
138 v_float32 vacc = v_add(v_add(vacc0, vacc1), v_add(vacc2, vacc3));
139 for (; t <= K - vlanes; t += vlanes)
140 vacc = v_fma(vx_load(arow + t), vx_load(w + t), vacc);
141 acc = v_reduce_sum(vacc);
142#endif
143 for (; t < K; t++)
144 acc += arow[t] * w[t];
145 y[static_cast<size_t>(m) * N + n] = acc;
146 }
147 }
148 }, static_cast<double>(N) * M * K * (1 / 1024.0));
149 }

151 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE {
152 CV_TRACE_FUNCTION();
153 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
154
155 std::vector<Mat> inputs, outputs;
156 inputs_arr.getMatVector(inputs);
157 outputs_arr.getMatVector(outputs);
158
159 const Mat& A = inputs[0];
160 const Mat& B = blobs[0]; // packed weights, uint8 [N, n_blk, block_size*bits/8]
161 const Mat& scales = blobs[1]; // fp32 [N, n_blk]
162 Mat& Y = outputs[0];
163
164 CV_Assert(A.isContinuous() && Y.isContinuous());
165
166 switch (bits) { // bits ∈ {4,8} enforced in the constructor
167 case 4: runImpl<4>(A, B, scales, Y); break;
168 default: runImpl<8>(A, B, scales, Y); break;
169 }
170 }
