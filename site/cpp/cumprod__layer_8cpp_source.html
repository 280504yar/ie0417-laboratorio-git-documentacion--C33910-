51 {
52 if (inputs_arr.depth() == CV_16F)
53 {
54 forward_fallback(inputs_arr, outputs_arr, internals_arr);
55 return;
56 }
57 std::vector<Mat> inputs, outputs;
58 inputs_arr.getMatVector(inputs);
59 outputs_arr.getMatVector(outputs);
60 CV_CheckTypeEQ(inputs[0].depth(), outputs[0].depth(), "");

62 switch (inputs[0].depth())
63 {
64 case CV_32F: forwardImpl<float>(inputs, outputs); break;
65 case CV_32S: forwardImpl<int32_t>(inputs, outputs); break;
66 case CV_64S: forwardImpl<int64_t>(inputs, outputs); break;
67 case CV_64F: forwardImpl<double>(inputs, outputs); break;
68 case CV_32U: forwardImpl<uint32_t>(inputs, outputs); break;
69 case CV_64U: forwardImpl<uint64_t>(inputs, outputs); break;
70 default: CV_Error(Error::BadDepth, "");
71 }

74 template <typename T>
75 void forwardImpl(const std::vector<Mat>& inputs, std::vector<Mat>& outputs)
76 {
77 const Mat& src_mat = inputs[0];
78 const T* src_ptr = src_mat.ptr<T>();
79
80 int axis = inputs.size() > 1 ? parseAxis(inputs[1]) : axis_raw;
81 axis = normalize_axis(axis, src_mat.dims);
82
83 Mat& dst_mat = outputs[0];
84 T* dst_ptr = dst_mat.ptr<T>();
85
86 const bool exclusive = exclusive_raw == 1;
87 const bool reverse = reverse_raw == 1;
88
89 // View data as [outer_size, target_size, inner_size] around the scan axis.
90 const size_t outer_size = src_mat.total(0, axis);
91 const size_t target_size = src_mat.size[axis];
92 const size_t inner_size = src_mat.total(axis + 1);
93 const size_t outer_step_length = target_size * inner_size;
94
95 const int target_start = reverse ? (int)target_size - 1 : 0;
96 const int target_stop = reverse ? -1 : (int)target_size;
97 const int target_delta = reverse ? -1 : 1;
98 const int target_step = target_delta * (int)inner_size;
99 const int exclusive_delta = exclusive ? target_step : 0;
100
101 // Each outer slice holds independent scans, so parallelize over it.
102 parallel_for_(Range(0, (int)outer_size), [&](const Range& range)
103 {
104 for (int outer_idx = range.start; outer_idx < range.end; outer_idx++)
105 {
106 const size_t target_offset = (size_t)outer_idx * outer_step_length;
107
108 // First element: multiplicative identity when exclusive, else the source value.
109 size_t first_inner_offset = target_offset + (size_t)target_start * inner_size;
110 if (exclusive)
111 for (size_t inner_idx = 0; inner_idx < inner_size; inner_idx++)
112 dst_ptr[first_inner_offset + inner_idx] = (T)1;
113 else
114 for (size_t inner_idx = 0; inner_idx < inner_size; inner_idx++)
115 dst_ptr[first_inner_offset + inner_idx] = src_ptr[first_inner_offset + inner_idx];
116
117 for (int target_idx = target_start + target_delta; target_idx != target_stop; target_idx += target_delta)
118 {
119 const size_t inner_offset = target_offset + (size_t)target_idx * inner_size;
120 for (size_t inner_idx = 0; inner_idx < inner_size; inner_idx++)
121 {
122 dst_ptr[inner_offset + inner_idx] = dst_ptr[inner_offset - target_step + inner_idx] *
123 src_ptr[inner_offset - exclusive_delta + inner_idx];
124 }
125 }
126 }
127 });
128 }
