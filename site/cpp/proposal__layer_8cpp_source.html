20public:
21 ProposalLayerImpl(const LayerParams& params)
22 {
23 setParamsFrom(params);
24
25 featStride = params.get<uint32_t>("feat_stride", 16);
26 baseSize = params.get<uint32_t>("base_size", 16);
27 // uint32_t minSize = params.get<uint32_t>("min_size", 16);
28 keepTopBeforeNMS = params.get<uint32_t>("pre_nms_topn", 6000);
29 keepTopAfterNMS = params.get<uint32_t>("post_nms_topn", 300);
30 nmsThreshold = params.get<float>("nms_thresh", 0.7);
31 ratios = params.get("ratio");
32 scales = params.get("scale");
33
34 {
35 LayerParams lp;
36 lp.set("step", featStride);
37 lp.set("flip", false);
38 lp.set("clip", false);
39 lp.set("normalized_bbox", false);
40 lp.set("offset", 0.5 * baseSize / featStride);
41
42 // Unused values.
43 float variance[] = {0.1f, 0.1f, 0.2f, 0.2f};
44 lp.set("variance", DictValue::arrayReal<float*>(&variance[0], 4));
45
46 // Compute widths and heights explicitly.
47 std::vector<float> widths, heights;
48 widths.reserve(ratios.size() * scales.size());
49 heights.reserve(ratios.size() * scales.size());
50 for (int i = 0; i < ratios.size(); ++i)
51 {
52 float ratio = ratios.get<float>(i);
53 float width = std::floor(baseSize / sqrt(ratio) + 0.5f);
54 float height = std::floor(width * ratio + 0.5f);
55 for (int j = 0; j < scales.size(); ++j)
56 {
57 float scale = scales.get<float>(j);
58 widths.push_back(scale * width);
59 heights.push_back(scale * height);
60 }
61 }
62 lp.set("width", DictValue::arrayReal<float*>(&widths[0], widths.size()));
63 lp.set("height", DictValue::arrayReal<float*>(&heights[0], heights.size()));
64
65 priorBoxLayer = PriorBoxLayer::create(lp);
66 }
67 {
68 int order[] = {0, 2, 3, 1};
69 LayerParams lp;
70 lp.set("order", DictValue::arrayInt<int*>(&order[0], 4));
71
72 deltasPermute = PermuteLayer::create(lp);
73 scoresPermute = PermuteLayer::create(lp);
74 }
75 {
76 LayerParams lp;
77 lp.set("code_type", "CENTER_SIZE");
78 lp.set("num_classes", 1);
79 lp.set("share_location", true);
80 lp.set("background_label_id", 1); // We won't pass background scores so set it out of range [0, num_classes)
81 lp.set("variance_encoded_in_target", true);
82 lp.set("keep_top_k", keepTopAfterNMS);
83 lp.set("top_k", keepTopBeforeNMS);
84 lp.set("nms_threshold", nmsThreshold);
85 lp.set("normalized_bbox", false);
86 lp.set("clip", true);
87
88 detectionOutputLayer = DetectionOutputLayer::create(lp);
89 }
90 }

104 bool getMemoryShapes(const std::vector<MatShape> &inputs,
105 const int requiredOutputs,
106 std::vector<MatShape> &outputs,
107 std::vector<MatShape> &internals) const CV_OVERRIDE
108 {
109 // We need to allocate the following blobs:
110 // - output priors from PriorBoxLayer
111 // - permuted priors
112 // - permuted scores
113 CV_Assert(inputs.size() == 3);
114
115 const MatShape& scores = inputs[0];
116 const MatShape& bboxDeltas = inputs[1];
117
118 std::vector<MatShape> layerInputs, layerOutputs, layerInternals;
119
120 // Prior boxes layer.
121 layerInputs.assign(1, scores);
122 priorBoxLayer->getMemoryShapes(layerInputs, 1, layerOutputs, layerInternals);
123 CV_Assert(layerOutputs.size() == 1);
124 CV_Assert(layerInternals.empty());
125 internals.push_back(layerOutputs[0]);
126
127 // Scores permute layer.
128 CV_Assert(scores.size() == 4);
129 MatShape objectScores = scores;
130 CV_Assert((scores[1] & 1) == 0); // Number of channels is even.
131 objectScores[1] /= 2;
132 layerInputs.assign(1, objectScores);
133 scoresPermute->getMemoryShapes(layerInputs, 1, layerOutputs, layerInternals);
134 CV_Assert(layerOutputs.size() == 1);
135 CV_Assert(layerInternals.empty());
136 internals.push_back(layerOutputs[0]);
137
138 // BBox predictions permute layer.
139 layerInputs.assign(1, bboxDeltas);
140 deltasPermute->getMemoryShapes(layerInputs, 1, layerOutputs, layerInternals);
141 CV_Assert(layerOutputs.size() == 1);
142 CV_Assert(layerInternals.empty());
143 internals.push_back(layerOutputs[0]);
144
145 // Detections layer.
146 internals.push_back(shape(1, 1, keepTopAfterNMS, 7));
147
148 outputs.resize(2);
149 outputs[0] = shape(keepTopAfterNMS, 5);
150 outputs[1] = shape(keepTopAfterNMS, 1);
151 return false;
152 }

154 void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays) CV_OVERRIDE
155 {
156 std::vector<Mat> inputs;
157 inputs_arr.getMatVector(inputs);
158
159 std::vector<Mat> layerInputs;
160 std::vector<Mat> layerOutputs;
161
162 // Scores permute layer.
163 Mat scores = getObjectScores(inputs[0]);
164 layerInputs.assign(1, scores);
165 layerOutputs.assign(1, Mat({scores.size[0], scores.size[2],
166 scores.size[3], scores.size[1]}, CV_32FC1));
167 scoresPermute->finalize(layerInputs, layerOutputs);
168
169 // BBox predictions permute layer.
170 const Mat& bboxDeltas = inputs[1];
171 CV_Assert(bboxDeltas.dims == 4);
172 layerInputs.assign(1, bboxDeltas);
173 layerOutputs.assign(1, Mat({bboxDeltas.size[0], bboxDeltas.size[2],
174 bboxDeltas.size[3], bboxDeltas.size[1]}, CV_32FC1));
175 deltasPermute->finalize(layerInputs, layerOutputs);
176 }

178#ifdef HAVE_OPENCL
179 bool forward_ocl(InputArrayOfArrays inputs_, OutputArrayOfArrays outputs_, OutputArrayOfArrays internals_)
180 {
181 std::vector<UMat> inputs;
182 std::vector<UMat> outputs;
183 std::vector<UMat> internals;
184
185 if (inputs_.depth() == CV_16F)
186 return false;
187
188 inputs_.getUMatVector(inputs);
189 outputs_.getUMatVector(outputs);
190 internals_.getUMatVector(internals);
191
192 CV_Assert(inputs.size() == 3);
193 CV_Assert(internals.size() == 4);
194 const UMat& scores = inputs[0];
195 const UMat& bboxDeltas = inputs[1];
196 const UMat& imInfo = inputs[2];
197 UMat& priorBoxes = internals[0];
198 UMat& permuttedScores = internals[1];
199 UMat& permuttedDeltas = internals[2];
200 UMat& detections = internals[3];
201
202 CV_Assert(imInfo.total() >= 2);
203 // We've chosen the smallest data type because we need just a shape from it.
204 Mat szMat;
205 imInfo.copyTo(szMat);
206 int rows = (int)szMat.at<float>(0);
207 int cols = (int)szMat.at<float>(1);
208 umat_fakeImageBlob.create(shape(1, 1, rows, cols), CV_8UC1);
209 umat_fakeImageBlob.setTo(0);
210
211 // Generate prior boxes.
212 std::vector<UMat> layerInputs(2), layerOutputs(1, priorBoxes);
213 layerInputs[0] = scores;
214 layerInputs[1] = umat_fakeImageBlob;
215 priorBoxLayer->forward(layerInputs, layerOutputs, internals);
216
217 // Permute scores.
218 layerInputs.assign(1, getObjectScores(scores));
219 layerOutputs.assign(1, permuttedScores);
220 scoresPermute->forward(layerInputs, layerOutputs, internals);
221
222 // Permute deltas.
223 layerInputs.assign(1, bboxDeltas);
224 layerOutputs.assign(1, permuttedDeltas);
225 deltasPermute->forward(layerInputs, layerOutputs, internals);
226
227 // Sort predictions by scores and apply NMS. DetectionOutputLayer allocates
228 // output internally because of different number of objects after NMS.
229 layerInputs.resize(4);
230 layerInputs[0] = permuttedDeltas;
231 layerInputs[1] = permuttedScores;
232 layerInputs[2] = priorBoxes;
233 layerInputs[3] = umat_fakeImageBlob;
234
235 layerOutputs[0] = detections;
236 detectionOutputLayer->forward(layerInputs, layerOutputs, internals);
237
238 // DetectionOutputLayer produces 1x1xNx7 output where N might be less or
239 // equal to keepTopAfterNMS. We fill the rest by zeros.
240 const int numDets = layerOutputs[0].total() / 7;
241 CV_Assert(numDets <= keepTopAfterNMS);
242
243 MatShape s = shape(numDets, 7);
244 layerOutputs[0] = layerOutputs[0].reshape(1, s.size(), &s[0]);
245
246 // The boxes.
247 UMat dst = outputs[0].rowRange(0, numDets);
248 layerOutputs[0].colRange(3, 7).copyTo(dst.colRange(1, 5));
249 dst.col(0).setTo(0); // First column are batch ids. Keep it zeros too.
250
251 // The scores.
252 dst = outputs[1].rowRange(0, numDets);
253 layerOutputs[0].col(2).copyTo(dst);
254
255 return true;
256 }
257#endif

259 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
260 {
261 CV_TRACE_FUNCTION();
262 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
263
264 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget) &&
265 OCL_PERFORMANCE_CHECK(ocl::Device::getDefault().isIntel()),
266 forward_ocl(inputs_arr, outputs_arr, internals_arr))
267
268 if (inputs_arr.depth() == CV_16F)
269 {
270 forward_fallback(inputs_arr, outputs_arr, internals_arr);
271 return;
272 }
273
274 std::vector<Mat> inputs, outputs, internals;
275 inputs_arr.getMatVector(inputs);
276 outputs_arr.getMatVector(outputs);
277 internals_arr.getMatVector(internals);
278
279 CV_Assert(inputs.size() == 3);
280 CV_Assert(internals.size() == 4);
281 const Mat& scores = inputs[0];
282 const Mat& bboxDeltas = inputs[1];
283 const Mat& imInfo = inputs[2];
284 Mat& priorBoxes = internals[0];
285 Mat& permuttedScores = internals[1];
286 Mat& permuttedDeltas = internals[2];
287 Mat& detections = internals[3];
288
289 CV_Assert(imInfo.total() >= 2);
290 int imInfo0, imInfo1;
291 if (imInfo.type() == CV_32F) {
292 imInfo0 = cvRound(imInfo.at<float>(0));
293 imInfo1 = cvRound(imInfo.at<float>(1));
294 } else if (imInfo.type() == CV_32S) {
295 imInfo0 = imInfo.at<int>(0);
296 imInfo1 = imInfo.at<int>(1);
297 } else if (imInfo.type() == CV_64S) {
298 imInfo0 = (int)imInfo.at<int64_t>(0);
299 imInfo1 = (int)imInfo.at<int64_t>(1);
300 } else {
301 CV_Error(Error::StsBadArg, "unsupported type of input[2]: must be 32f, 32s or 64s");
302 }
303 // We've chosen the smallest data type because we need just a shape from it.
304 // We don't allocate memory but just need the shape is correct.
305 Mat fakeImageBlob({1, 1, imInfo0, imInfo1}, CV_8UC1, NULL);
306
307 // Generate prior boxes.
308 std::vector<Mat> layerInputs(2), layerOutputs(1, priorBoxes);
309 layerInputs[0] = scores;
310 layerInputs[1] = fakeImageBlob;
311 priorBoxLayer->forward(layerInputs, layerOutputs, internals);
312
313 // Permute scores.
314 layerInputs.assign(1, getObjectScores(scores));
315 layerOutputs.assign(1, permuttedScores);
316 scoresPermute->forward(layerInputs, layerOutputs, internals);
317
318 // Permute deltas.
319 layerInputs.assign(1, bboxDeltas);
320 layerOutputs.assign(1, permuttedDeltas);
321 deltasPermute->forward(layerInputs, layerOutputs, internals);
322
323 // Sort predictions by scores and apply NMS. DetectionOutputLayer allocates
324 // output internally because of different number of objects after NMS.
325 layerInputs.resize(4);
326 layerInputs[0] = permuttedDeltas;
327 layerInputs[1] = permuttedScores;
328 layerInputs[2] = priorBoxes;
329 layerInputs[3] = fakeImageBlob;
330
331 layerOutputs[0] = detections;
332 detectionOutputLayer->forward(layerInputs, layerOutputs, internals);
333
334 // DetectionOutputLayer produces 1x1xNx7 output where N might be less or
335 // equal to keepTopAfterNMS. We fill the rest by zeros.
336 const int numDets = layerOutputs[0].total() / 7;
337 CV_Assert(numDets <= keepTopAfterNMS);
338
339 // The boxes.
340 layerOutputs[0] = layerOutputs[0].reshape(1, numDets);
341 Mat dst = outputs[0].rowRange(0, numDets);
342 layerOutputs[0].colRange(3, 7).copyTo(dst.colRange(1, 5));
343 dst.col(0).setTo(0); // First column are batch ids. Keep it zeros too.
344
345 // The scores.
346 dst = outputs[1].rowRange(0, numDets);
347 layerOutputs[0].col(2).copyTo(dst);
348 }

352 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> >& inputs,
353 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
354 {
355 CV_Assert(nodes.size() == 3);
356 ov::op::v0::Proposal::Attributes attr;
357 attr.base_size = baseSize;
358 attr.nms_thresh = nmsThreshold;
359 attr.feat_stride = featStride;
360 attr.min_size = 16;
361 attr.pre_nms_topn = keepTopBeforeNMS;
362 attr.post_nms_topn = keepTopAfterNMS;
363
364 std::vector<float> ratiosVec(ratios.size());
365 for (int i = 0; i < ratios.size(); ++i)
366 ratiosVec[i] = ratios.get<float>(i);
367 attr.ratio = ratiosVec;
368
369 std::vector<float> scalesVec(scales.size());
370 for (int i = 0; i < scales.size(); ++i)
371 scalesVec[i] = scales.get<float>(i);
372 attr.scale = scalesVec;
373
374 auto& class_probs = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
375 auto& class_logits = nodes[1].dynamicCast<InfEngineNgraphNode>()->node;
376 auto& image_shape = nodes[2].dynamicCast<InfEngineNgraphNode>()->node;
377
378 CV_Assert_N(image_shape.get_shape().size() == 2, image_shape.get_shape().front() == 1);
379 auto shape = std::make_shared<ov::op::v0::Constant>(ov::element::i64,
380 ov::Shape{1},
381 std::vector<int64_t>{(int64_t)image_shape.get_shape().back()});
382 auto reshape = std::make_shared<ov::op::v1::Reshape>(image_shape, shape, true);
383
384 auto proposal = std::make_shared<ov::op::v0::Proposal>(class_probs, class_logits, reshape, attr);
385 return Ptr<BackendNode>(new InfEngineNgraphNode(proposal));
386 }
387#endif // HAVE_DNN_NGRAPH

389private:
390 // A first half of channels are background scores. We need only a second one.
391 static Mat getObjectScores(const Mat& m)
392 {
393 CV_Assert(m.dims == 4);
394 CV_Assert(m.size[0] == 1);
395 int channels = m.size[1];
396 CV_Assert((channels & 1) == 0);
397 return slice(m, Range::all(), Range(channels / 2, channels));
398 }
399
400#ifdef HAVE_OPENCL
401 static UMat getObjectScores(const UMat& m)
402 {
403 CV_Assert(m.dims == 4);
404 CV_Assert(m.size[0] == 1);
405 int channels = m.size[1];
406 CV_Assert((channels & 1) == 0);
407
408 Range r = Range(channels / 2, channels);
409 Range ranges[4] = { Range::all(), r, Range::all(), Range::all() };
410 return m(&ranges[0]);
411 }
412#endif
