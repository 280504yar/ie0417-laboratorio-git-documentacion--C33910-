76inline static void _zarray_ensure_capacity(zarray_t *za, int capacity){
77 if (capacity <= za->alloc)
78 return;
79
80 while (za->alloc < capacity) {
81 za->alloc *= 2;
82 if (za->alloc < 8)
83 za->alloc = 8;
84 }

86 za->data = (char*) realloc(za->data, za->alloc * za->el_sz);
