25 OneHotLayerImpl(const LayerParams& params)
26 {
27 setParamsFrom(params);
28 axis = params.get<int>("axis", -1);

36private:
37 bool getConstPositiveDepth(int& depthOut) const
38 {
39 Net::Impl* netimpl_ = getNetImpl(this);
40 if (!netimpl_ || !netimpl_->isConstArg(this->inputs[1]))
41 return false;
42
43 try
44 {
45 Mat depthTensor = netimpl_->argTensor(this->inputs[1]);
46 if (!depthTensor.empty() && depthTensor.total() == 1)
47 {
48 int64_t depth64 = 0;
49 tensorToScalar(depthTensor, CV_64S, &depth64);
50 if (depth64 > 0)
51 {
52 depthOut = (int)depth64;
53 return true;
54 }
55 }
56 }
57 catch (const cv::Exception& e)
58 {
59 CV_Error_(Error::StsError, ("OneHot: failed to access constant depth: %s", e.what()));
60 }
61 return false;
62 }
63
64 bool tryInferStaticOutputShape(const MatShape& idxShape, MatShape& outShape) const
65 {
66 outShape.clear();
67 int depth = 0;
68 if (getConstPositiveDepth(depth))
69 {
70 outShape = idxShape;
71 int insAxis = normalize_axis(axis, (int)outShape.size() + 1);
72 outShape.insert(outShape.begin() + insAxis, depth);
73 return true;
74 }
75 return false;
76 }

89 {
90 CV_Assert(inputs.size() == 3);

118private:
119 template<typename Tout>
120 void runKernel(const Mat& indices, int depth, const Mat& values, Mat& Y)
121 {
122 CV_CheckGT(depth, 0, "OneHot: depth must be > 0");
123
124 MatShape outShape = shape(indices);
125 int ndims = (int)outShape.size();
126 int insAxis = normalize_axis(axis, ndims + 1);
127 outShape.insert(outShape.begin() + insAxis, depth);
128
129 if (Y.empty() || Y.type() != DataType<Tout>::type || Y.shape() != outShape)
130 Y.create(outShape, DataType<Tout>::type);
131
132 Tout offVal = (Tout)0;
133 Tout onVal = (Tout)1;
134 if (!values.empty())
135 {
136 CV_Assert(values.total() == 2);
137 Mat flat = values.reshape(1, (int)values.total());
138 offVal = tensorToScalar<Tout>(flat.row(0));
139 onVal = tensorToScalar<Tout>(flat.row(1));
140 }
141
142 Y.setTo(Scalar::all(offVal));
143
144 Mat idx64;
145 if (indices.depth() == CV_64S) idx64 = indices;
146 else indices.convertTo(idx64, CV_64S);
147 const int64_t* idxPtr64 = idx64.ptr<int64_t>();
148 size_t inTotal = indices.total();
149 size_t outStep = 1;
150 for (int k = insAxis + 1; k < (int)outShape.size(); ++k)
151 outStep *= (size_t)outShape[k];
152
153 Tout* outPtr = Y.ptr<Tout>();
154
155 parallel_for_(Range(0, (int)inTotal), [&](const Range& r){
156 const int64_t* localIdxPtr64 = idxPtr64;
157 Tout* localOutPtr = outPtr;
158 const size_t localOutStep = outStep;
159 const int localDepth = depth;
160 const Tout localOnVal = onVal;
161 for (int pos = r.start; pos < r.end; ++pos) {
162 int64_t idxVal = localIdxPtr64[pos];
163 int64_t wrapped = idxVal % localDepth;
164 if (wrapped < 0) wrapped += localDepth;
165 size_t hi = (size_t)pos / localOutStep;
166 size_t lo = (size_t)pos % localOutStep;
167 size_t base = hi * ((size_t)localDepth * localOutStep) + lo;
168 localOutPtr[base + (size_t)wrapped * localOutStep] = localOnVal;
169 }
170 });
171 }

173public:
174 void forward(InputArrayOfArrays in, OutputArrayOfArrays out, OutputArrayOfArrays) CV_OVERRIDE
175 {
176 int inKind = in.kind();
177 int outKind = out.kind();
178 CV_Assert(in.size().area() == 3);
179
180 Mat indices, depthMat, values;
181 if (inKind == _InputArray::STD_VECTOR_MAT)
182 {
183 indices = in.getMat(0);
184 depthMat = in.getMat(1);
185 values = in.getMat(2);
186 }
187 else if (inKind == _InputArray::STD_VECTOR_UMAT)
188 {
189 indices = in.getUMat(0).getMat(ACCESS_READ);
190 depthMat = in.getUMat(1).getMat(ACCESS_READ);
191 values = in.getUMat(2).getMat(ACCESS_READ);
192 }
193 else
194 {
195 CV_Error(Error::StsNotImplemented, "OneHot: unsupported input type");
196 }
197
198 std::vector<Mat>* out_mats = nullptr;
199 std::vector<UMat>* out_umats = nullptr;
200 Mat Y;
201 if (outKind == _InputArray::STD_VECTOR_MAT)
202 {
203 out_mats = &out.getMatVecRef();
204 out_mats->resize(1);
205 }
206 else if (outKind == _InputArray::STD_VECTOR_UMAT)
207 {
208 out_umats = &out.getUMatVecRef();
209 out_umats->resize(1);
210 }
211 else
212 {
213 CV_Error(Error::StsNotImplemented, "OneHot: unsupported output type");
214 }
215
216 int outDepth = values.empty() ? CV_32F : values.depth();
217
218 MatShape outShape = shape(indices);
219 int ndims = (int)outShape.size();
220 int insAxis = normalize_axis(axis, ndims + 1);
221
222 CV_Assert(depthMat.total() == 1);
223 int64_t depth64 = 0;
224 tensorToScalar(depthMat, CV_64S, &depth64);
225 CV_CheckGT(depth64, 0, "OneHot: depth must be > 0");
226 MatShape finalShape;
227 outShape.insert(outShape.begin() + insAxis, (int)depth64);
228 finalShape = outShape;
229 if (!finalShape.empty())
230 {
231 if (outKind == _InputArray::STD_VECTOR_MAT)
232 {
233 out_mats->at(0).fit(finalShape, outDepth);
234 Y = out_mats->at(0);
235 }
236 else
237 {
238 out_umats->at(0).fit(finalShape, outDepth);
239 Y = Mat(finalShape, outDepth);
240 }
241 }
242
243 switch (outDepth) {
244 case CV_8U: runKernel<uchar>(indices, (int)depth64, values, Y); break;
245 case CV_8S: runKernel<schar>(indices, (int)depth64, values, Y); break;
246 case CV_16U: runKernel<uint16_t>(indices, (int)depth64, values, Y); break;
247 case CV_16S: runKernel<int16_t>(indices, (int)depth64, values, Y); break;
248 case CV_32S: runKernel<int>(indices, (int)depth64, values, Y); break;
249 case CV_64S: runKernel<int64_t>(indices, (int)depth64, values, Y); break;
250 case CV_16F: runKernel<hfloat>(indices, (int)depth64, values, Y); break;
251 case CV_16BF: runKernel<bfloat>(indices, (int)depth64, values, Y); break;
252 case CV_32F: runKernel<float>(indices, (int)depth64, values, Y); break;
253 case CV_64F: runKernel<double>(indices, (int)depth64, values, Y); break;
254 default: CV_Error(Error::BadDepth, "OneHot: unsupported output type");
255 }
256
257 if (outKind == _InputArray::STD_VECTOR_UMAT)
258 {
259 Y.copyTo(out_umats->at(0));
260 }
261 }
