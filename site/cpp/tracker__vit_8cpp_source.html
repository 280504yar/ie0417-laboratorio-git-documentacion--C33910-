89static int crop_image(const Mat& src, Mat& dst, Rect box, int factor)
90{
91 int x = box.x, y = box.y, w = box.width, h = box.height;
92 int crop_sz = cvCeil(sqrt(w * h) * factor);
93
94 int x1 = x + (w - crop_sz) / 2;
95 int x2 = x1 + crop_sz;
96 int y1 = y + (h - crop_sz) / 2;
97 int y2 = y1 + crop_sz;
98
99 int x1_pad = std::max(0, -x1);
100 int y1_pad = std::max(0, -y1);
101 int x2_pad = std::max(x2 - src.size[1] + 1, 0);
102 int y2_pad = std::max(y2 - src.size[0] + 1, 0);
103
104 Rect roi(x1 + x1_pad, y1 + y1_pad, x2 - x2_pad - x1 - x1_pad, y2 - y2_pad - y1 - y1_pad);
105 Mat im_crop = src(roi);
106 copyMakeBorder(im_crop, dst, y1_pad, y2_pad, x1_pad, x2_pad, BORDER_CONSTANT);
107
108 return crop_sz;
109}

119static Mat hann1d(int sz, bool centered = true) {
120 Mat hanningWindow(sz, 1, CV_32FC1);
121 float* data = hanningWindow.ptr<float>(0);
122
123 if(centered) {
124 for(int i = 0; i < sz; i++) {
125 float val = 0.5f * (1.f - std::cos(static_cast<float>(2 * M_PI / (sz + 1)) * (i + 1)));
126 data[i] = val;
127 }
128 }
129 else {
130 int half_sz = sz / 2;
131 for(int i = 0; i <= half_sz; i++) {
132 float val = 0.5f * (1.f + std::cos(static_cast<float>(2 * M_PI / (sz + 2)) * i));
133 data[i] = val;
134 data[sz - 1 - i] = val;
135 }
136 }
137
138 return hanningWindow;
139}

141static Mat hann2d(Size size, bool centered = true) {
142 int rows = size.height;
143 int cols = size.width;
144
145 Mat hanningWindowRows = hann1d(rows, centered);
146 Mat hanningWindowCols = hann1d(cols, centered);
147
148 Mat hanningWindow = hanningWindowRows * hanningWindowCols.t();
149
150 return hanningWindow;
151}

153static void updateLastRect(float cx, float cy, float w, float h, int crop_size, Rect &rect_last)
154{
155 int x0 = rect_last.x + (rect_last.width - crop_size) / 2;
156 int y0 = rect_last.y + (rect_last.height - crop_size) / 2;
157
158 float x1 = cx - w / 2, y1 = cy - h / 2;
159 rect_last.x = cvFloor(x1 * crop_size + x0);
160 rect_last.y = cvFloor(y1 * crop_size + y0);
161 rect_last.width = cvFloor(w * crop_size);
162 rect_last.height = cvFloor(h * crop_size);
163}

165void TrackerVitImpl::init(InputArray image_, const Rect &boundingBox_)
166{
167 Mat image = image_.getMat();
168 Mat crop;
169 crop_image(image, crop, boundingBox_, 2);
170 Mat blob;
171 preprocess(crop, blob, templateSize);
172 net.setInput(blob, "template");
173 Size size(16, 16);
174 hanningWindow = hann2d(size, true);
175 rect_last = boundingBox_;

178bool TrackerVitImpl::update(InputArray image_, Rect &boundingBoxRes)
179{
180 Mat image = image_.getMat();
181 Mat crop;
182 int crop_size = crop_image(image, crop, rect_last, 4); // crop: [crop_size, crop_size]
183 Mat blob;
184 preprocess(crop, blob, searchSize);
185 net.setInput(blob, "search");
186 std::vector<String> outputName = {"output1", "output2", "output3"};
187 std::vector<Mat> outs;
188 net.forward(outs, outputName);
189 CV_Assert(outs.size() == 3);
190
191 Mat conf_map = outs[0].reshape(0, {16, 16});
192 Mat size_map = outs[1].reshape(0, {2, 16, 16});
193 Mat offset_map = outs[2].reshape(0, {2, 16, 16});
194
195 multiply(conf_map, hanningWindow, conf_map);
196
197 double maxVal;
198 Point maxLoc;
199 minMaxLoc(conf_map, nullptr, &maxVal, nullptr, &maxLoc);
200 tracking_score = static_cast<float>(maxVal);
201
202 if (tracking_score >= tracking_score_threshold) {
203 float cx = (maxLoc.x + offset_map.at<float>(0, maxLoc.y, maxLoc.x)) / 16;
204 float cy = (maxLoc.y + offset_map.at<float>(1, maxLoc.y, maxLoc.x)) / 16;
205 float w = size_map.at<float>(0, maxLoc.y, maxLoc.x);
206 float h = size_map.at<float>(1, maxLoc.y, maxLoc.x);
207
208 updateLastRect(cx, cy, w, h, crop_size, rect_last);
209 boundingBoxRes = rect_last;
210 return true;
211 } else {
212 return false;
213 }
214}
