25void cv::fisheye::projectPoints(InputArray objectPoints, OutputArray imagePoints, InputArray _rvec,
26 InputArray _tvec, InputArray _K, InputArray _D, double alpha, OutputArray jacobian)
27{
28 CV_INSTRUMENT_REGION();
29
30 // will support only 3-channel data now for points
31 CV_Assert(objectPoints.type() == CV_32FC3 || objectPoints.type() == CV_64FC3);
32 imagePoints.create(objectPoints.size(), CV_MAKETYPE(objectPoints.depth(), 2));
33 size_t n = objectPoints.total();
34
35 CV_Assert(_rvec.total() * _rvec.channels() == 3 && (_rvec.depth() == CV_32F || _rvec.depth() == CV_64F));
36 CV_Assert(_tvec.total() * _tvec.channels() == 3 && (_tvec.depth() == CV_32F || _tvec.depth() == CV_64F));
37 CV_Assert(_tvec.getMat().isContinuous() && _rvec.getMat().isContinuous());
38
39 Vec3d om = _rvec.depth() == CV_32F ? (Vec3d)*_rvec.getMat().ptr<Vec3f>() : *_rvec.getMat().ptr<Vec3d>();
40 Vec3d T = _tvec.depth() == CV_32F ? (Vec3d)*_tvec.getMat().ptr<Vec3f>() : *_tvec.getMat().ptr<Vec3d>();
41
42 CV_Assert(_K.size() == Size(3,3) && (_K.type() == CV_32F || _K.type() == CV_64F) && _D.type() == _K.type() && _D.total() == 4);
43
44 Vec2d f, c;
45 if (_K.depth() == CV_32F)
46 {
47
48 Matx33f K = _K.getMat();
49 f = Vec2f(K(0, 0), K(1, 1));
50 c = Vec2f(K(0, 2), K(1, 2));
51 }
52 else
53 {
54 Matx33d K = _K.getMat();
55 f = Vec2d(K(0, 0), K(1, 1));
56 c = Vec2d(K(0, 2), K(1, 2));
57 }
58
59 Vec4d k = _D.depth() == CV_32F ? (Vec4d)*_D.getMat().ptr<Vec4f>(): *_D.getMat().ptr<Vec4d>();
60
61 const bool isJacobianNeeded = jacobian.needed();
62 JacobianRow *Jn = 0;
63 if (isJacobianNeeded)
64 {
65 int nvars = 2 + 2 + 1 + 4 + 3 + 3; // f, c, alpha, k, om, T,
66 jacobian.create(2*(int)n, nvars, CV_64F);
67 Jn = jacobian.getMat().ptr<JacobianRow>(0);
68 }
69
70 Matx33d R;
71 Matx<double, 3, 9> dRdom;
72 Rodrigues(om, R, dRdom);
73 Affine3d aff(om, T);
74
75 const Vec3f* Xf = objectPoints.getMat().ptr<Vec3f>();
76 const Vec3d* Xd = objectPoints.getMat().ptr<Vec3d>();
77 Vec2f *xpf = imagePoints.getMat().ptr<Vec2f>();
78 Vec2d *xpd = imagePoints.getMat().ptr<Vec2d>();
79
80 for(size_t i = 0; i < n; ++i)
81 {
82 Vec3d Xi = objectPoints.depth() == CV_32F ? (Vec3d)Xf[i] : Xd[i];
83 Vec3d Y = aff*Xi;
84 if (fabs(Y[2]) < DBL_MIN)
85 Y[2] = 1;
86 Vec2d x(Y[0]/Y[2], Y[1]/Y[2]);
87
88 double r2 = x.dot(x);
89 double r = std::sqrt(r2);
90
91 // Angle of the incoming ray:
92 double theta = std::atan(r);
93
94 double theta2 = theta*theta, theta3 = theta2*theta, theta4 = theta2*theta2, theta5 = theta4*theta,
95 theta6 = theta3*theta3, theta7 = theta6*theta, theta8 = theta4*theta4, theta9 = theta8*theta;
96
97 double theta_d = theta + k[0]*theta3 + k[1]*theta5 + k[2]*theta7 + k[3]*theta9;
98
99 double inv_r = r > 1e-8 ? 1.0/r : 1;
100 double cdist = r > 1e-8 ? theta_d * inv_r : 1;
101
102 Vec2d xd1 = x * cdist;
103 Vec2d xd3(xd1[0] + alpha*xd1[1], xd1[1]);
104 Vec2d final_point(xd3[0] * f[0] + c[0], xd3[1] * f[1] + c[1]);
105
106 if (objectPoints.depth() == CV_32F)
107 xpf[i] = final_point;
108 else
109 xpd[i] = final_point;
110
111 if (isJacobianNeeded)
112 {
113 //Vec3d Xi = pdepth == CV_32F ? (Vec3d)Xf[i] : Xd[i];
114 //Vec3d Y = aff*Xi;
115 double dYdR[] = { Xi[0], Xi[1], Xi[2], 0, 0, 0, 0, 0, 0,
116 0, 0, 0, Xi[0], Xi[1], Xi[2], 0, 0, 0,
117 0, 0, 0, 0, 0, 0, Xi[0], Xi[1], Xi[2] };
118
119 Matx33d dYdom_data = Matx<double, 3, 9>(dYdR) * dRdom.t();
120 const Vec3d *dYdom = (Vec3d*)dYdom_data.val;
121
122 Matx33d dYdT_data = Matx33d::eye();
123 const Vec3d *dYdT = (Vec3d*)dYdT_data.val;
124
125 //Vec2d x(Y[0]/Y[2], Y[1]/Y[2]);
126 Vec3d dxdom[2];
127 dxdom[0] = (1.0/Y[2]) * dYdom[0] - x[0]/Y[2] * dYdom[2];
128 dxdom[1] = (1.0/Y[2]) * dYdom[1] - x[1]/Y[2] * dYdom[2];
129
130 Vec3d dxdT[2];
131 dxdT[0] = (1.0/Y[2]) * dYdT[0] - x[0]/Y[2] * dYdT[2];
132 dxdT[1] = (1.0/Y[2]) * dYdT[1] - x[1]/Y[2] * dYdT[2];
133
134 //double r2 = x.dot(x);
135 Vec3d dr2dom = 2 * x[0] * dxdom[0] + 2 * x[1] * dxdom[1];
136 Vec3d dr2dT = 2 * x[0] * dxdT[0] + 2 * x[1] * dxdT[1];
137
138 //double r = std::sqrt(r2);
139 double drdr2 = r > 1e-8 ? 1.0/(2*r) : 1;
140 Vec3d drdom = drdr2 * dr2dom;
141 Vec3d drdT = drdr2 * dr2dT;
142
143 // Angle of the incoming ray:
144 //double theta = atan(r);
145 double dthetadr = 1.0/(1+r2);
146 Vec3d dthetadom = dthetadr * drdom;
147 Vec3d dthetadT = dthetadr * drdT;
148
149 //double theta_d = theta + k[0]*theta3 + k[1]*theta5 + k[2]*theta7 + k[3]*theta9;
150 double dtheta_ddtheta = 1 + 3*k[0]*theta2 + 5*k[1]*theta4 + 7*k[2]*theta6 + 9*k[3]*theta8;
151 Vec3d dtheta_ddom = dtheta_ddtheta * dthetadom;
152 Vec3d dtheta_ddT = dtheta_ddtheta * dthetadT;
153 Vec4d dtheta_ddk = Vec4d(theta3, theta5, theta7, theta9);
154
155 //double inv_r = r > 1e-8 ? 1.0/r : 1;
156 //double cdist = r > 1e-8 ? theta_d / r : 1;
157 Vec3d dcdistdom = inv_r * (dtheta_ddom - cdist*drdom);
158 Vec3d dcdistdT = inv_r * (dtheta_ddT - cdist*drdT);
159 Vec4d dcdistdk = inv_r * dtheta_ddk;
160
161 //Vec2d xd1 = x * cdist;
162 Vec4d dxd1dk[2];
163 Vec3d dxd1dom[2], dxd1dT[2];
164 dxd1dom[0] = x[0] * dcdistdom + cdist * dxdom[0];
165 dxd1dom[1] = x[1] * dcdistdom + cdist * dxdom[1];
166 dxd1dT[0] = x[0] * dcdistdT + cdist * dxdT[0];
167 dxd1dT[1] = x[1] * dcdistdT + cdist * dxdT[1];
168 dxd1dk[0] = x[0] * dcdistdk;
169 dxd1dk[1] = x[1] * dcdistdk;
170
171 //Vec2d xd3(xd1[0] + alpha*xd1[1], xd1[1]);
172 Vec4d dxd3dk[2];
173 Vec3d dxd3dom[2], dxd3dT[2];
174 dxd3dom[0] = dxd1dom[0] + alpha * dxd1dom[1];
175 dxd3dom[1] = dxd1dom[1];
176 dxd3dT[0] = dxd1dT[0] + alpha * dxd1dT[1];
177 dxd3dT[1] = dxd1dT[1];
178 dxd3dk[0] = dxd1dk[0] + alpha * dxd1dk[1];
179 dxd3dk[1] = dxd1dk[1];
180
181 Vec2d dxd3dalpha(xd1[1], 0);
182
183 //final jacobian
184 Jn[0].dom = f[0] * dxd3dom[0];
185 Jn[1].dom = f[1] * dxd3dom[1];
186
187 Jn[0].dT = f[0] * dxd3dT[0];
188 Jn[1].dT = f[1] * dxd3dT[1];
189
190 Jn[0].dk = f[0] * dxd3dk[0];
191 Jn[1].dk = f[1] * dxd3dk[1];
192
193 Jn[0].dalpha = f[0] * dxd3dalpha[0];
194 Jn[1].dalpha = 0; //f[1] * dxd3dalpha[1];
195
196 Jn[0].df = Vec2d(xd3[0], 0);
197 Jn[1].df = Vec2d(0, xd3[1]);
198
199 Jn[0].dc = Vec2d(1, 0);
200 Jn[1].dc = Vec2d(0, 1);
201
202 //step to jacobian rows for next point
203 Jn += 2;
204 }
205 }

269void cv::fisheye::distortPoints(InputArray _undistorted, OutputArray distorted, InputArray Kundistorted, InputArray K, InputArray D, double alpha)
270{
271 CV_INSTRUMENT_REGION();
272
273 CV_Assert(_undistorted.type() == CV_32FC2 || _undistorted.type() == CV_64FC2);
274 CV_Assert(Kundistorted.size() == Size(3,3) && (Kundistorted.type() == CV_32F || Kundistorted.type() == CV_64F));
275
276 cv::Mat undistorted = _undistorted.getMat();
277 cv::Mat normalized(undistorted.size(), CV_64FC2);
278
279 Mat Knew = Kundistorted.getMat();
280
281 double cx, cy, fx, fy;
282 if (Knew.depth() == CV_32F)
283 {
284 fx = (double)Knew.at<float>(0, 0);
285 fy = (double)Knew.at<float>(1, 1);
286 cx = (double)Knew.at<float>(0, 2);
287 cy = (double)Knew.at<float>(1, 2);
288 }
289 else
290 {
291 fx = Knew.at<double>(0, 0);
292 fy = Knew.at<double>(1, 1);
293 cx = Knew.at<double>(0, 2);
294 cy = Knew.at<double>(1, 2);
295 }
296
297 size_t n = undistorted.total();
298 const Vec2f* Xf = undistorted.ptr<Vec2f>();
299 const Vec2d* Xd = undistorted.ptr<Vec2d>();
300 Vec2d* normXd = normalized.ptr<Vec2d>();
301 for (size_t i = 0; i < n; i++)
302 {
303 Vec2d p = undistorted.depth() == CV_32F ? (Vec2d)Xf[i] : Xd[i];
304 normXd[i][0] = (p[0] - cx) / fx;
305 normXd[i][1] = (p[1] - cy) / fy;
306 }
307
308 cv::fisheye::distortPoints(normalized, distorted, K, D, alpha);
309}
310
311// Compensate distortion iteratively using Newton's method. The vectorized loop in
312// undistortPoints() mirrors this solver.
313// theta_d = theta * (1 + k1 * theta^2 + k2 * theta^4 + k3 * theta^6 + k4 * theta^8) =>
314// f(theta) := theta * (1 + k1 * theta^2 + k2 * theta^4 + k3 * theta^6 + k4 * theta^8) - theta_d = 0
315// Newton's method: new_theta = theta - theta_fix, theta_fix := f(theta) / f'(theta)
316// f'(theta) = 1 + 3 * k1 * theta^2 + 5 * k2 * theta^4 + 7 * k3 * theta^6 + 9 * k4 * theta^8
317static inline double solveThetaFromThetaD(double theta_d, const Vec4d& k, int maxCount,
318 bool isEps, double epsilon, bool& converged)
319{
320 double theta = theta_d;
321 converged = false;
322
323 for (int j = 0; j < maxCount; j++)
324 {
325 double theta2 = theta*theta, theta4 = theta2*theta2, theta6 = theta4*theta2, theta8 = theta6*theta2;
326 double k0_theta2 = k[0] * theta2, k1_theta4 = k[1] * theta4, k2_theta6 = k[2] * theta6, k3_theta8 = k[3] * theta8;
327 double theta_fix = (theta * (1 + k0_theta2 + k1_theta4 + k2_theta6 + k3_theta8) - theta_d) /
328 (1 + 3*k0_theta2 + 5*k1_theta4 + 7*k2_theta6 + 9*k3_theta8);
329 theta = theta - theta_fix;
330
331 if (isEps && (fabs(theta_fix) < epsilon))
332 {
333 converged = true;
334 break;
335 }
336 }
337
338 return theta;
339}

341void cv::fisheye::undistortPoints( InputArray distorted, OutputArray undistorted, InputArray K, InputArray D,
342 InputArray R, InputArray P, TermCriteria criteria)
343{
344 CV_INSTRUMENT_REGION();
345
346 // will support only 2-channel data now for points
347 CV_Assert(distorted.type() == CV_32FC2 || distorted.type() == CV_64FC2);
348 undistorted.create(distorted.size(), distorted.type());
349
350 CV_Assert(P.empty() || P.size() == Size(3, 3) || P.size() == Size(4, 3));
351 CV_Assert(R.empty() || R.size() == Size(3, 3) || R.total() * R.channels() == 3);
352 CV_Assert(D.total() == 4 && K.size() == Size(3, 3) && (K.depth() == CV_32F || K.depth() == CV_64F));
353
354 CV_Assert(criteria.isValid());
355
356 Vec2d f, c;
357 if (K.depth() == CV_32F)
358 {
359 Matx33f camMat = K.getMat();
360 f = Vec2f(camMat(0, 0), camMat(1, 1));
361 c = Vec2f(camMat(0, 2), camMat(1, 2));
362 }
363 else
364 {
365 Matx33d camMat = K.getMat();
366 f = Vec2d(camMat(0, 0), camMat(1, 1));
367 c = Vec2d(camMat(0, 2), camMat(1, 2));
368 }
369
370 Vec4d k = D.depth() == CV_32F ? (Vec4d)*D.getMat().ptr<Vec4f>(): *D.getMat().ptr<Vec4d>();
371
372 Matx33d RR = Matx33d::eye();
373 if (!R.empty() && R.total() * R.channels() == 3)
374 {
375 Vec3d rvec;
376 R.getMat().convertTo(rvec, CV_64F);
377 RR = cv::Affine3d(rvec).rotation();
378 }
379 else if (!R.empty() && R.size() == Size(3, 3))
380 R.getMat().convertTo(RR, CV_64F);
381
382 if(!P.empty())
383 {
384 Matx33d PP;
385 P.getMat().colRange(0, 3).convertTo(PP, CV_64F);
386 RR = PP * RR;
387 }
388
389 // start undistorting
390 const Vec2f* srcf = distorted.getMat().ptr<Vec2f>();
391 const Vec2d* srcd = distorted.getMat().ptr<Vec2d>();
392 Vec2f* dstf = undistorted.getMat().ptr<Vec2f>();
393 Vec2d* dstd = undistorted.getMat().ptr<Vec2d>();
394
395 size_t n = distorted.total();
396 int sdepth = distorted.depth();
397
398 const bool isEps = (criteria.type & TermCriteria::EPS) != 0;
399
400 /* Define max count for solver iterations */
401 int maxCount = std::numeric_limits<int>::max();
402 if (criteria.type & TermCriteria::MAX_ITER) {
403 maxCount = criteria.maxCount;
404 }
405
406 size_t i = 0;
407#if (CV_SIMD_SCALABLE_64F || CV_SIMD_64F)
408 // Vectorized fast path: fixed iteration count (no EPS early-exit), no reprojection (R, P empty)
409 // and double I/O. The per-point Newton iteration is independent, so a whole vector of points is
410 // solved at once. Every other case, and the trailing points, go to the scalar loop below.
411 if( !isEps && R.empty() && P.empty() && sdepth == CV_64F )
412 {
413 const int vl = VTraits<v_float64>::vlanes();
414 if( n >= (size_t)vl )
415 {
416 const size_t totalBlocks = n / vl;
417 auto processBlocks = [&](size_t blockBegin, size_t blockEnd)
418 {
419 const v_float64 v_fx = vx_setall_f64(f[0]), v_fy = vx_setall_f64(f[1]);
420 const v_float64 v_cx = vx_setall_f64(c[0]), v_cy = vx_setall_f64(c[1]);
421 const v_float64 v_k0 = vx_setall_f64(k[0]), v_k1 = vx_setall_f64(k[1]);
422 const v_float64 v_k2 = vx_setall_f64(k[2]), v_k3 = vx_setall_f64(k[3]);
423 const v_float64 v_one = vx_setall_f64(1.0), v_zero = vx_setzero_f64();
424 const v_float64 v_three = vx_setall_f64(3.0), v_five = vx_setall_f64(5.0);
425 const v_float64 v_seven = vx_setall_f64(7.0), v_nine = vx_setall_f64(9.0);
426 const v_float64 v_lo = vx_setall_f64(-CV_PI/2.), v_hi = vx_setall_f64(CV_PI/2.);
427 const v_float64 v_sent = vx_setall_f64(-1000000.0);
428 AutoBuffer<double> tbuf(vl);
429
430 for( size_t block = blockBegin; block < blockEnd; block++ )
431 {
432 const size_t j = block * vl;
433 v_float64 px, py;
434 v_load_deinterleave((const double*)(srcd + j), px, py);
435 v_float64 pwx = v_div(v_sub(px, v_cx), v_fx);
436 v_float64 pwy = v_div(v_sub(py, v_cy), v_fy);
437 v_float64 theta_d = v_sqrt(v_add(v_mul(pwx, pwx), v_mul(pwy, pwy)));
438 theta_d = v_min(v_max(theta_d, v_lo), v_hi);
439 v_float64 theta = theta_d;
440
441 for( int it = 0; it < maxCount; it++ )
442 {
443 v_float64 t2 = v_mul(theta, theta), t4 = v_mul(t2, t2);
444 v_float64 t6 = v_mul(t4, t2), t8 = v_mul(t6, t2);
445 v_float64 a0 = v_mul(t2, v_k0), a1 = v_mul(t4, v_k1);
446 v_float64 a2 = v_mul(t6, v_k2), a3 = v_mul(t8, v_k3);
447 v_float64 poly = v_add(v_one, v_add(v_add(a0, a1), v_add(a2, a3)));
448 v_float64 num = v_sub(v_mul(theta, poly), theta_d);
449 v_float64 den = v_add(v_one, v_add(
450 v_add(v_mul(a0, v_three), v_mul(a1, v_five)),
451 v_add(v_mul(a2, v_seven), v_mul(a3, v_nine))));
452 theta = v_sub(theta, v_div(num, den));
453 }
454
455 // scale = tan(theta) / theta_d (scalar pass; there is no vector transcendental)
456 v_store(tbuf.data(), theta);
457 for( int lane = 0; lane < vl; lane++ ) tbuf[lane] = std::tan(tbuf[lane]);
458 v_float64 scale = v_div(vx_load(tbuf.data()), theta_d);
459 v_float64 pux = v_mul(pwx, scale), puy = v_mul(pwy, scale);
460
461 // theta_d is non-negative, so a sign flip is exactly theta < 0.
462 v_float64 flipped = v_lt(theta, v_zero);
463 pux = v_select(flipped, v_sent, pux);
464 puy = v_select(flipped, v_sent, puy);
465 v_store_interleave((double*)(dstd + j), pux, puy);
466 }
467 };
468
469 if( totalBlocks <= (size_t)std::numeric_limits<int>::max() )
470 {
471 // Keep small inputs single-threaded while splitting sufficiently expensive Newton
472 // workloads into multiple stripes.
473 const double nstripes = (double)n * maxCount / (1 << 16);
474 if( nstripes < 1.5 )
475 processBlocks(0, totalBlocks);
476 else
477 {
478 parallel_for_(Range(0, (int)totalBlocks), [&](const Range& range)
479 {
480 processBlocks((size_t)range.start, (size_t)range.end);
481 }, nstripes);
482 }
483 }
484 else
485 processBlocks(0, totalBlocks);
486
487 i = totalBlocks * vl;
488 }
489 }
490#endif
491
492 for( ; i < n; i++ )
493 {
494 Vec2d pi = sdepth == CV_32F ? (Vec2d)srcf[i] : srcd[i]; // image point
495 // u = fx * x' + cx (alpha = 0), v = fy * y' + cy =>
496 // x' = (u - cx) / fx, y' = (v - cy) / fy
497 Vec2d pw((pi[0] - c[0])/f[0], (pi[1] - c[1])/f[1]); // world point
498
499 // x' = (theta_d / r) * a, y' = (theta_d / r) * b =>
500 // x'^2 + y'^2 = theta_d^2 * (a^2 + b^2) / r^2 =>
501 // (r^2 = a^2 + b^2)
502 // x'^2 + y'^2 = theta_d^2 =>
503 // theta_d = sqrt(x'^2 + y'^2)
504 double theta_d = sqrt(pw[0]*pw[0] + pw[1]*pw[1]);
505
506 // the current camera model is only valid up to 180 FOV
507 // for larger FOV the loop below does not converge
508 // clip values so we still get plausible results for super fisheye images > 180 grad
509 theta_d = min(max(-CV_PI/2., theta_d), CV_PI/2.);
510
511 bool converged = false;
512 double theta = theta_d;
513
514 double scale = 0.0;
515
516 if (!isEps || fabs(theta_d) > criteria.epsilon)
517 {
518 // compensate distortion iteratively using Newton method
519 theta = solveThetaFromThetaD(theta_d, k, maxCount, isEps, criteria.epsilon, converged);
520
521 // x' = (theta_d / r) * a, y' = (theta_d / r) * b =>
522 // a = x' * r / theta_d, b = y' * r / theta_d =>
523 // (theta = atan(r) => r = tan(theta), scale := r / theta_d = tan(theta) / theta_d)
524 // a = x' * scale, b = y' * scale
525 scale = std::tan(theta) / theta_d;
526 }
527 else
528 {
529 converged = true;
530 }
531
532 // theta is monotonously increasing or decreasing depending on the sign of theta
533 // if theta has flipped, it might converge due to symmetry but on the opposite of the camera center
534 // so we can check whether theta has changed the sign during the optimization
535 bool theta_flipped = ((theta_d < 0 && theta > 0) || (theta_d > 0 && theta < 0));
536
537 if ((converged || !isEps) && !theta_flipped)
538 {
539 // a = x' * scale, b = y' * scale
540 Vec2d pu = pw * scale; //undistorted point
541 Vec2d fi;
542
543 if (!R.empty() || !P.empty())
544 {
545 // reproject
546 Vec3d pr = RR * Vec3d(pu[0], pu[1], 1.0); // rotated point optionally multiplied by new camera matrix
547 fi = Vec2d(pr[0]/pr[2], pr[1]/pr[2]); // final
548 }
549 else
550 {
551 fi = pu;
552 }
553
554 if( sdepth == CV_32F )
555 dstf[i] = fi;
556 else
557 dstd[i] = fi;
558 }
559 else
560 {
561 // Vec2d fi(std::numeric_limits<double>::quiet_NaN(), std::numeric_limits<double>::quiet_NaN());
562 Vec2d fi(-1000000.0, -1000000.0);
563
564 if( sdepth == CV_32F )
565 dstf[i] = fi;
566 else
567 dstd[i] = fi;
568 }
569 }
570}

572void cv::fisheye::estimateNewCameraMatrixForUndistortRectify(InputArray K, InputArray D, const Size &image_size, InputArray R,
573 OutputArray P, double balance, const Size& new_size, double fov_scale)
574{
575 CV_INSTRUMENT_REGION();
576
577 CV_Assert( K.size() == Size(3, 3) && (K.depth() == CV_32F || K.depth() == CV_64F));
578 CV_Assert(D.empty() || ((D.total() == 4) && (D.depth() == CV_32F || D.depth() == CV_64F)));
579
580 int w = image_size.width, h = image_size.height;
581 balance = std::min(std::max(balance, 0.0), 1.0);
582
583 Mat points(1, 4, CV_64FC2);
584 Vec2d* pptr = points.ptr<Vec2d>();
585 pptr[0] = Vec2d(w/2, 0);
586 pptr[1] = Vec2d(w, h/2);
587 pptr[2] = Vec2d(w/2, h);
588 pptr[3] = Vec2d(0, h/2);
589
590 fisheye::undistortPoints(points, points, K, D, R);
591 cv::Scalar center_mass = mean(points);
592 Vec2d cn(center_mass.val);
593
594 double aspect_ratio = (K.depth() == CV_32F) ? K.getMat().at<float >(0,0)/K.getMat().at<float> (1,1)
595 : K.getMat().at<double>(0,0)/K.getMat().at<double>(1,1);
596
597 // convert to identity ratio
598 cn[1] *= aspect_ratio;
599 for(size_t i = 0; i < points.total(); ++i)
600 pptr[i][1] *= aspect_ratio;
601
602 double minx = DBL_MAX, miny = DBL_MAX, maxx = -DBL_MAX, maxy = -DBL_MAX;
603 for(size_t i = 0; i < points.total(); ++i)
604 {
605 miny = std::min(miny, pptr[i][1]);
606 maxy = std::max(maxy, pptr[i][1]);
607 minx = std::min(minx, pptr[i][0]);
608 maxx = std::max(maxx, pptr[i][0]);
609 }
610
611 double f1 = w * 0.5/(cn[0] - minx);
612 double f2 = w * 0.5/(maxx - cn[0]);
613 double f3 = h * 0.5 * aspect_ratio/(cn[1] - miny);
614 double f4 = h * 0.5 * aspect_ratio/(maxy - cn[1]);
615
616 double fmin = std::min(f1, std::min(f2, std::min(f3, f4)));
617 double fmax = std::max(f1, std::max(f2, std::max(f3, f4)));
618
619 double f = balance * fmin + (1.0 - balance) * fmax;
620 f *= fov_scale > 0 ? 1.0/fov_scale : 1.0;
621
622 Vec2d new_f(f, f), new_c = -cn * f + Vec2d(w, h * aspect_ratio) * 0.5;
623
624 // restore aspect ratio
625 new_f[1] /= aspect_ratio;
626 new_c[1] /= aspect_ratio;
627
628 if (!new_size.empty())
629 {
630 double rx = new_size.width /(double)image_size.width;
631 double ry = new_size.height/(double)image_size.height;
632
633 new_f[0] *= rx; new_f[1] *= ry;
634 new_c[0] *= rx; new_c[1] *= ry;
635 }
636
637 Mat(Matx33d(new_f[0], 0, new_c[0],
638 0, new_f[1], new_c[1],
639 0, 0, 1)).convertTo(P, P.empty() ? K.type() : P.type());
640}
