122 // True if `arg` is produced by the dynamic scale chain Sqrt<-Cast<-Div(1,.)<-Sqrt<-Cast<-Slice<-Shape; visited ops are appended to `chain_ops`.
123 bool isRuntimeQKScaleChain(const vector<Ptr<LayerInfo>>& prog, Arg arg,
124 std::set<int>& chain_ops) const
125 {
126 const std::vector<std::string> expected = {
127 "Sqrt", "Cast2", "NaryEltwise" /*Div*/, "Sqrt",
128 "Cast2", "Slice2", "Shape"
129 };
130 Arg cur = arg;
131 for (const std::string& want : expected) {
132 auto it = producer_.find(cur.idx);
133 if (it == producer_.end()) return false;
134 int idx = it->second;
135 if (idx < 0 || idx >= (int)prog.size() || !prog[idx]) return false;
136 const Ptr<LayerInfo>& l = prog[idx];
137 if (want == "NaryEltwise") {
138 NaryEltwiseLayer* elt = dynamic_cast<NaryEltwiseLayer*>(l.get());
139 if (!elt || elt->op != NaryEltwiseLayer::OPERATION::DIV) return false;
140
141 bool one_seen = false;
142 Arg runtime_in;
143 bool runtime_seen = false;
144 for (Arg in : l->inputs) {
145 if (netimpl->isConstArg(in)) {
146 Mat t = netimpl->argTensor(in);
147 if (t.total() != 1) return false;
148 float v = 0.f;
149 if (t.type() == CV_32F) v = t.at<float>(0);
150 else if (t.type() == CV_64F) v = (float)t.at<double>(0);
151 else return false;
152 if (std::abs(v - 1.f) > 1e-5f) return false;
153 one_seen = true;
154 } else {
155 runtime_in = in;
156 runtime_seen = true;
157 }
158 }
159 if (!one_seen || !runtime_seen) return false;
160 chain_ops.insert(idx);
161 cur = runtime_in;
162 } else {
163 if (l->type != want) return false;
164 chain_ops.insert(idx);
165 if (l->inputs.empty()) return false;
166 cur = l->inputs[0];
167 }
168 }
169 return true;
170 }

187 // Extract a scalar integer from a const-valued arg, possibly wrapped in an
188 // Unsqueeze of a scalar const. Returns -1 if extraction fails.
189 int extractConstInt(const vector<Ptr<LayerInfo>>& prog, Arg a) const
190 {
191 auto readScalar = [&](Arg x) -> int {
192 if (!netimpl->isConstArg(x)) return -1;
193 Mat t = netimpl->argTensor(x);
194 if (t.total() != 1) return -1;
195 if (t.type() == CV_64S) return (int)t.at<int64_t>(0);
196 if (t.type() == CV_32S) return (int)t.at<int32_t>(0);
197 return -1;
198 };
199 int v = readScalar(a);
200 if (v > 0) return v;
201 auto it = producer_.find(a.idx);
202 if (it == producer_.end()) return -1;
203 int pidx = it->second;
204 if (pidx < 0 || pidx >= (int)prog.size() || !prog[pidx]) return -1;
205 if (prog[pidx]->type == "Unsqueeze" && !prog[pidx]->inputs.empty())
206 return readScalar(prog[pidx]->inputs[0]);
207 return -1;
208 }

210 void collectShapeChain(const vector<Ptr<LayerInfo>>& prog, int concat_idx,
211 std::set<int>& chain) const
212 {
213 if (concat_idx < 0 || concat_idx >= (int)prog.size() || !prog[concat_idx])
214 return;
215 chain.insert(concat_idx);
216 for (Arg ci : prog[concat_idx]->inputs) {
217 if (netimpl->isConstArg(ci)) continue;
218 int cur = -1;
219 auto it = producer_.find(ci.idx);
220 if (it != producer_.end()) cur = it->second;
221 while (cur >= 0 && (int)prog.size() > cur && prog[cur]) {
222 const std::string& t = prog[cur]->type;
223 if (t != "Unsqueeze" && t != "Gather2" && t != "Shape") break;
224 Arg cur_out = prog[cur]->outputs[0];
225 auto cit = consumers_.find(cur_out.idx);
226 if (cit == consumers_.end()) break;
227 bool all_in_chain = true;
228 for (int c : cit->second) if (!chain.count(c)) { all_in_chain = false; break; }
229 if (!all_in_chain) break;
230 chain.insert(cur);
231 if (prog[cur]->inputs.empty()) break;
232 Arg prev = prog[cur]->inputs[0];
233 auto pit = producer_.find(prev.idx);
234 if (pit == producer_.end()) break;
235 cur = pit->second;
236 }
237 }

241 int findMatchingConsumer(const vector<Ptr<LayerInfo>>& prog, Arg out,
242 Pred pred, std::set<int>* extra_shape_ops) const
243 {
244 auto it = consumers_.find(out.idx);
245 if (it == consumers_.end()) return -1;
246 int matched = -1;
247 for (int c : it->second) {
248 if (c < 0 || c >= (int)prog.size() || !prog[c]) return -1;
249 if (pred(prog[c].get())) {
250 if (matched >= 0) return -1;
251 matched = c;
252 } else if (prog[c]->type == "Shape") {
253 if (extra_shape_ops) extra_shape_ops->insert(c);
254 } else {
255 return -1;
256 }
257 }
258 return matched;
259 }

261 int followProjChain(const vector<Ptr<LayerInfo>>& prog,
262 int proj_matmul_idx,
263 int* out_reshape_idx,
264 int* out_num_heads,
265 vector<int>* out_perm,
266 std::set<int>* extra_ops_to_remove) const
267 {
268 if (proj_matmul_idx < 0) return -1;
269 Arg proj_out = prog[proj_matmul_idx]->outputs[0];
270 int reshape_idx = findMatchingConsumer(prog, proj_out,
271 [](LayerInfo* L){ return dynamic_cast<Reshape2Layer*>(L) != nullptr; },
272 extra_ops_to_remove);
273 if (!isReshape(prog, reshape_idx)) return -1;
274
275 const auto& reshape_inputs = prog[reshape_idx]->inputs;
276 if (reshape_inputs.size() < 2) return -1;
277 Arg shape_arg = reshape_inputs[1];
278
279 int num_heads = -1;
280 if (netimpl->isConstArg(shape_arg)) {
281 Mat shape_mat = netimpl->argTensor(shape_arg);
282 if (shape_mat.total() != 4) return -1;
283 const int64_t* shape_data = shape_mat.ptr<int64_t>();
284 num_heads = static_cast<int>(shape_data[2]);
285 } else {
286 auto it = producer_.find(shape_arg.idx);
287 if (it == producer_.end()) return -1;
288 int concat_idx = it->second;
289 if (concat_idx < 0 || concat_idx >= (int)prog.size() || !prog[concat_idx])
290 return -1;
291 if (!dynamic_cast<Concat2Layer*>(prog[concat_idx].get()))
292 return -1;
293 const auto& cinputs = prog[concat_idx]->inputs;
294 if (cinputs.size() != 4) return -1;
295 num_heads = extractConstInt(prog, cinputs[2]);
296 if (num_heads <= 0) return -1;
297 if (extra_ops_to_remove)
298 collectShapeChain(prog, concat_idx, *extra_ops_to_remove);
299 }
300 *out_num_heads = num_heads;
301 *out_reshape_idx = reshape_idx;
302
303 Arg reshape_out = prog[reshape_idx]->outputs[0];
304 int transpose_idx = singleConsumer(reshape_out);
305 if (!isTranspose(prog, transpose_idx)) return -1;
306 TransposeLayer* tr = dynamic_cast<TransposeLayer*>(prog[transpose_idx].get());
307 if (!tr) return -1;
308 *out_perm = tr->perm;
309 return transpose_idx;
310 }

312 // Combined-QKV attention: QKV proj -> Reshape ->
313 // Transpose -> 3 Gathers -> QK^T -> Softmax(no mask) -> *V.
314 bool tryFuseCombinedQKV(const vector<Ptr<LayerInfo>>& prog, int qkv_matmul_idx,
315 std::set<int>& removed_ops,
316 vector<std::pair<int, Ptr<LayerInfo>>>& replacements)
317 {
318 if (qkv_matmul_idx < 0 || qkv_matmul_idx >= (int)prog.size() || !prog[qkv_matmul_idx])
319 return false;
320 if (removed_ops.count(qkv_matmul_idx)) return false;
321 if (!isProjCandidate(prog[qkv_matmul_idx])) return false;
322
323 Mat W = getProjWeight(prog[qkv_matmul_idx]);
324 if (W.dims != 2 || W.size[1] % 3 != 0) return false;
325 const int input_hidden = W.size[0];
326 const int proj_hidden = W.size[1] / 3;
327 const int total_hidden = W.size[1];
328
329 Arg cur = prog[qkv_matmul_idx]->outputs[0];
330
331 // Optional bias: separate Add op or Gemm's blobs[1].
332 int add_idx = -1;
333 Mat bias_mat;
334 bool has_bias = false;
335 if (prog[qkv_matmul_idx]->blobs.size() >= 2) {
336 bias_mat = prog[qkv_matmul_idx]->blobs[1];
337 has_bias = bias_mat.total() == (size_t)total_hidden && bias_mat.type() == CV_32F;
338 if (!has_bias) return false;
339 }
340 int next = singleConsumer(cur);
341 if (!has_bias && next >= 0 && next < (int)prog.size() && prog[next]) {
342 NaryEltwiseLayer* e = dynamic_cast<NaryEltwiseLayer*>(prog[next].get());
343 if (e && e->op == NaryEltwiseLayer::OPERATION::ADD &&
344 prog[next]->inputs.size() == 2)
345 {
346 Mat b_candidate;
347 for (Arg in : prog[next]->inputs) {
348 if (netimpl->isConstArg(in)) {
349 Mat t = netimpl->argTensor(in);
350 if (t.type() == CV_32F && (int)t.total() == total_hidden)
351 b_candidate = t;
352 }
353 }
354 if (!b_candidate.empty()) {
355 add_idx = next;
356 bias_mat = b_candidate;
357 has_bias = true;
358 cur = prog[add_idx]->outputs[0];
359 next = singleConsumer(cur);
360 }
361 }
362 }
363
364 if (!isReshape(prog, next)) return false;
365 const int reshape_idx = next;
366 const auto& rinputs = prog[reshape_idx]->inputs;
367 if (rinputs.size() < 2) return false;
368
369 std::set<int> extra_ops;
370 int num_heads = -1, head_dim = -1;
371 Arg shape_arg = rinputs[1];
372 if (netimpl->isConstArg(shape_arg)) {
373 Mat sh = netimpl->argTensor(shape_arg);
374 if (sh.total() != 5) return false;
375 const int64_t* sd = sh.ptr<int64_t>();
376 if ((int)sd[2] != 3) return false;
377 num_heads = (int)sd[3];
378 head_dim = (int)sd[4];
379 } else {
380 auto it = producer_.find(shape_arg.idx);
381 if (it == producer_.end()) return false;
382 int concat_idx = it->second;
383 if (concat_idx < 0 || concat_idx >= (int)prog.size() || !prog[concat_idx])
384 return false;
385 if (!dynamic_cast<Concat2Layer*>(prog[concat_idx].get())) return false;
386 const auto& cinputs = prog[concat_idx]->inputs;
387 if (cinputs.size() != 5) return false;
388 if (extractConstInt(prog, cinputs[2]) != 3) return false;
389 num_heads = extractConstInt(prog, cinputs[3]);
390 head_dim = extractConstInt(prog, cinputs[4]);
391 if (num_heads <= 0 || head_dim <= 0) return false;
392 collectShapeChain(prog, concat_idx, extra_ops);
393 }
394 if (num_heads * head_dim != proj_hidden) return false;
395
396 // Transpose must bring "3" to front; only perm[0]==2 is required.
397 Arg reshape_out = prog[reshape_idx]->outputs[0];
398 int transpose_idx = singleConsumer(reshape_out);
399 if (!isTranspose(prog, transpose_idx)) return false;
400 TransposeLayer* tr = dynamic_cast<TransposeLayer*>(prog[transpose_idx].get());
401 if (!tr || tr->perm.size() != 5 || tr->perm[0] != 2) return false;
402
403 Arg tr_out = prog[transpose_idx]->outputs[0];
404 auto cit = consumers_.find(tr_out.idx);
405 if (cit == consumers_.end() || cit->second.size() != 3) return false;
406
407 int gather_for_idx[3] = { -1, -1, -1 }; // [Q,K,V] = [0,1,2]
408 for (int c : cit->second) {
409 if (c < 0 || c >= (int)prog.size() || !prog[c]) return false;
410 Gather2Layer* g = dynamic_cast<Gather2Layer*>(prog[c].get());
411 if (!g || g->axis != 0 || prog[c]->inputs.size() < 2) return false;
412 Arg idx_arg = prog[c]->inputs[1];
413 if (!netimpl->isConstArg(idx_arg)) return false;
414 Mat t = netimpl->argTensor(idx_arg);
415 if (t.total() != 1) return false;
416 int v;
417 if (t.type() == CV_64S) v = (int)t.at<int64_t>(0);
418 else if (t.type() == CV_32S) v = (int)t.at<int32_t>(0);
419 else return false;
420 if (v < 0 || v > 2 || gather_for_idx[v] != -1) return false;
421 gather_for_idx[v] = c;
422 }
423 const int q_gather = gather_for_idx[0];
424 const int k_gather = gather_for_idx[1];
425 const int v_gather = gather_for_idx[2];
426 if (q_gather < 0 || k_gather < 0 || v_gather < 0) return false;
427
428 Arg q_out = prog[q_gather]->outputs[0];
429 Arg k_out = prog[k_gather]->outputs[0];
430 Arg v_out = prog[v_gather]->outputs[0];
431
432 int q_mul_idx = singleConsumer(q_out);
433 float q_scale = 1.f;
434 if (!isScalarMul(prog, q_mul_idx, &q_scale)) return false;
435 if (q_scale == 0.f) return false;
436 int qk_matmul_idx = singleConsumer(prog[q_mul_idx]->outputs[0]);
437 if (!isMatMul(prog, qk_matmul_idx)) return false;
438
439 int k_trans_idx = singleConsumer(k_out);
440 if (!isTranspose(prog, k_trans_idx)) return false;
441 Arg k_trans_out = prog[k_trans_idx]->outputs[0];
442 bool k_connected = false;
443 for (Arg in : prog[qk_matmul_idx]->inputs)
444 if (in.idx == k_trans_out.idx) { k_connected = true; break; }
445 if (!k_connected) return false;
446
447 int softmax_idx = singleConsumer(prog[qk_matmul_idx]->outputs[0]);
448 if (!isSoftmax(prog, softmax_idx)) return false;
449 int av_matmul_idx = singleConsumer(prog[softmax_idx]->outputs[0]);
450 if (!isMatMul(prog, av_matmul_idx)) return false;
451 bool v_connected = false;
452 for (Arg in : prog[av_matmul_idx]->inputs)
453 if (in.idx == v_out.idx) { v_connected = true; break; }
454 if (!v_connected) return false;
455
456 int out_trans_idx = singleConsumer(prog[av_matmul_idx]->outputs[0]);
457 if (!isTranspose(prog, out_trans_idx)) return false;
458 int out_reshape_idx = findMatchingConsumer(prog, prog[out_trans_idx]->outputs[0],
459 [](LayerInfo* L){ return dynamic_cast<Reshape2Layer*>(L) != nullptr; },
460 &extra_ops);
461 if (!isReshape(prog, out_reshape_idx)) return false;
462
463 const auto& or_inputs = prog[out_reshape_idx]->inputs;
464 if (or_inputs.size() >= 2 && !netimpl->isConstArg(or_inputs[1])) {
465 auto it = producer_.find(or_inputs[1].idx);
466 if (it != producer_.end())
467 collectShapeChain(prog, it->second, extra_ops);
468 }
469
470 // W is already laid out as [Q|K|V] along the output dim.
471 Mat W_qkv = W.clone();
472 Mat bias_qkv;
473 if (has_bias) bias_qkv = bias_mat.clone();
474
475 // Attention `scale` is the pre-softmax divisor; Q was multiplied by
476 // q_scale, so scale = 1/q_scale.
477 const float param_scale = 1.0f / q_scale;
478
479 LayerParams attn_params;
480 attn_params.name = prog[qkv_matmul_idx]->name + "_fused_attention";
481 attn_params.type = "Attention";
482 attn_params.set("num_heads", num_heads);
483 int qkv_sizes[3] = { proj_hidden, proj_hidden, proj_hidden };
484 attn_params.set("qkv_hidden_sizes", DictValue::arrayInt(qkv_sizes, 3));
485 attn_params.set("scale", param_scale);
486 attn_params.set("output_ndims", 3);
487 attn_params.blobs.push_back(W_qkv);
488 if (has_bias) attn_params.blobs.push_back(bias_qkv);
489
490 Ptr<LayerInfo> attn_layer = LayerFactory::createLayerInstance(attn_params.type, attn_params);
491 CV_Assert(attn_layer);
492 Arg shared_input = prog[qkv_matmul_idx]->inputs[0];
493 attn_layer->inputs = { shared_input };
494 attn_layer->outputs = prog[out_reshape_idx]->outputs;
495 attn_layer->netimpl = netimpl;
496
497 std::set<int> to_remove = {
498 qkv_matmul_idx, reshape_idx, transpose_idx,
499 q_gather, k_gather, v_gather,
500 q_mul_idx, k_trans_idx,
501 qk_matmul_idx, softmax_idx,
502 av_matmul_idx, out_trans_idx, out_reshape_idx
503 };
504 if (add_idx >= 0) to_remove.insert(add_idx);
505 for (int op : extra_ops) to_remove.insert(op);
506
507 (void)input_hidden; // input_hidden currently unused; kept for symmetry with existing path.
508
509 std::set<int> attn_out_args;
510 for (const Arg& o : attn_layer->outputs) attn_out_args.insert(o.idx);
511 if (!removalIsSelfContained(prog, to_remove, attn_out_args))
512 return false;
513
514 for (int op : to_remove) removed_ops.insert(op);
515 const int insert_pos = *std::min_element(to_remove.begin(), to_remove.end());
516 replacements.push_back({insert_pos, attn_layer});
517 return true;
518 }

520 // CLIP-branch trace: arg -> [Transpose3D(K^T)] -> Reshape3D -> Transpose ->
521 // Reshape4D -> [Mul(Q scale)] -> [Add(bias)] -> proj_MatMul.
522 int traceClipBranch(const vector<Ptr<LayerInfo>>& prog, Arg arg,
523 bool is_q_branch, bool is_k_branch,
524 Mat& out_W, Mat& out_bias, int& out_num_heads,
525 float& out_q_scale,
526 std::set<int>& ops_consumed) const
527 {
528 Arg cur = arg;
529
530 auto stepProducer = [&](Arg a) -> int {
531 auto it = producer_.find(a.idx);
532 return it == producer_.end() ? -1 : it->second;
533 };
534
535 // K side: peel the (B*H,S,D) -> (B*H,D,S) transpose-3D first.
536 if (is_k_branch) {
537 int idx = stepProducer(cur);
538 if (idx < 0 || !prog[idx]) return -1;
539 TransposeLayer* tr = dynamic_cast<TransposeLayer*>(prog[idx].get());
540 if (!tr || tr->perm.size() != 3) return -1;
541 if (tr->perm[0] != 0 || tr->perm[1] != 2 || tr->perm[2] != 1) return -1;
542 ops_consumed.insert(idx);
543 cur = prog[idx]->inputs[0];
544 }
545
546 // (B,H,S,D) -> (B*H,S,D); shape may be dynamic, validated downstream.
547 int r3d_idx = stepProducer(cur);
548 if (!isReshape(prog, r3d_idx)) return -1;
549 ops_consumed.insert(r3d_idx);
550 if (prog[r3d_idx]->inputs.size() < 2) return -1;
551 Arg shape_arg_r3d = prog[r3d_idx]->inputs[1];
552 if (!netimpl->isConstArg(shape_arg_r3d)) {
553 int sh_idx = stepProducer(shape_arg_r3d);
554 if (sh_idx >= 0)
555 collectShapeChain(prog, sh_idx, ops_consumed);
556 }
557 cur = prog[r3d_idx]->inputs[0];
558
559 int t4d_idx = stepProducer(cur);
560 if (!isTranspose(prog, t4d_idx)) return -1;
561 TransposeLayer* tr4 = dynamic_cast<TransposeLayer*>(prog[t4d_idx].get());
562 if (!tr4 || tr4->perm.size() != 4) return -1;
563 if (tr4->perm[0] != 0 || tr4->perm[1] != 2 ||
564 tr4->perm[2] != 1 || tr4->perm[3] != 3) return -1;
565 ops_consumed.insert(t4d_idx);
566 cur = prog[t4d_idx]->inputs[0];
567
568 // (B,S,H*D) -> (B,S,H,D); num_heads is the third shape entry.
569 int r4d_idx = stepProducer(cur);
570 if (!isReshape(prog, r4d_idx)) return -1;
571 if (prog[r4d_idx]->inputs.size() < 2) return -1;
572 Arg shape_arg_r4d = prog[r4d_idx]->inputs[1];
573 int num_heads = -1;
574 if (netimpl->isConstArg(shape_arg_r4d)) {
575 Mat shape_mat = netimpl->argTensor(shape_arg_r4d);
576 if (shape_mat.total() != 4) return -1;
577 if (shape_mat.type() == CV_64S)
578 num_heads = static_cast<int>(shape_mat.ptr<int64_t>()[2]);
579 else if (shape_mat.type() == CV_32S)
580 num_heads = static_cast<int>(shape_mat.ptr<int32_t>()[2]);
581 else return -1;
582 } else {
583 int concat_idx = stepProducer(shape_arg_r4d);
584 if (concat_idx < 0 || !prog[concat_idx]) return -1;
585 if (!dynamic_cast<Concat2Layer*>(prog[concat_idx].get())) return -1;
586 const auto& cinputs = prog[concat_idx]->inputs;
587 if (cinputs.size() != 4) return -1;
588 num_heads = extractConstInt(prog, cinputs[2]);
589 if (num_heads <= 0) return -1;
590 collectShapeChain(prog, concat_idx, ops_consumed);
591 }
592 if (num_heads <= 0) return -1;
593 out_num_heads = num_heads;
594 ops_consumed.insert(r4d_idx);
595 cur = prog[r4d_idx]->inputs[0];
596
597 out_q_scale = 1.0f;
598 if (is_q_branch) {
599 int mul_idx = stepProducer(cur);
600 if (mul_idx < 0 || !prog[mul_idx]) return -1;
601 NaryEltwiseLayer* mul =
602 dynamic_cast<NaryEltwiseLayer*>(prog[mul_idx].get());
603 if (!mul || mul->op != NaryEltwiseLayer::OPERATION::PROD) return -1;
604 if (prog[mul_idx]->inputs.size() != 2) return -1;
605 Arg runtime_arg;
606 bool got_scale = false, got_runtime = false;
607 for (Arg in : prog[mul_idx]->inputs) {
608 if (netimpl->isConstArg(in)) {
609 Mat t = netimpl->argTensor(in);
610 if (t.total() != 1) return -1;
611 if (t.type() == CV_32F) out_q_scale = t.at<float>(0);
612 else if (t.type() == CV_64F) out_q_scale = (float)t.at<double>(0);
613 else return -1;
614 got_scale = true;
615 } else {
616 runtime_arg = in;
617 got_runtime = true;
618 }
619 }
620 if (!got_scale || !got_runtime) return -1;
621 ops_consumed.insert(mul_idx);
622 cur = runtime_arg;
623 }
624
625 int next_idx = stepProducer(cur);
626 if (next_idx < 0 || !prog[next_idx]) return -1;
627
628 out_bias = Mat();
629 int mm_idx = -1;
630 if (dynamic_cast<NaryEltwiseLayer*>(prog[next_idx].get())) {
631 NaryEltwiseLayer* add =
632 dynamic_cast<NaryEltwiseLayer*>(prog[next_idx].get());
633 if (!add || add->op != NaryEltwiseLayer::OPERATION::ADD) return -1;
634 if (prog[next_idx]->inputs.size() != 2) return -1;
635 Arg bias_arg;
636 Arg matmul_out_arg;
637 bool got_bias = false, got_runtime2 = false;
638 for (Arg in : prog[next_idx]->inputs) {
639 if (netimpl->isConstArg(in)) {
640 bias_arg = in;
641 got_bias = true;
642 } else {
643 matmul_out_arg = in;
644 got_runtime2 = true;
645 }
646 }
647 if (!got_bias || !got_runtime2) return -1;
648 out_bias = netimpl->argTensor(bias_arg).clone();
649 ops_consumed.insert(next_idx);
650 mm_idx = stepProducer(matmul_out_arg);
651 } else {
652 mm_idx = next_idx;
653 }
654
655 if (mm_idx < 0 || !prog[mm_idx]) return -1;
656 if (!dynamic_cast<MatMulLayer*>(prog[mm_idx].get())) return -1;
657 if (prog[mm_idx]->blobs.empty()) return -1;
658 if (prog[mm_idx]->inputs.size() != 1) return -1;
659 out_W = prog[mm_idx]->blobs[0].clone();
660 // Folded MatMul carries bias as a second blob (real_ndims_C >= 1).
661 if (out_bias.empty() && prog[mm_idx]->blobs.size() >= 2)
662 out_bias = prog[mm_idx]->blobs.back().clone();
663 return mm_idx;
664 }

666 // 3 separate q/k/v projections, Q scaled, K^T at
667 // the QK^T matmul, output reshaped+transposed back to (B,S,H*D).
668 bool tryFuseClipAttention(const vector<Ptr<LayerInfo>>& prog, int softmax_idx,
669 std::set<int>& removed_ops,
670 vector<std::pair<int, Ptr<LayerInfo>>>& replacements)
671 {
672 if (softmax_idx < 0 || softmax_idx >= (int)prog.size() || !prog[softmax_idx])
673 return false;
674 if (removed_ops.count(softmax_idx)) return false;
675 SoftmaxLayer* sm = dynamic_cast<SoftmaxLayer*>(prog[softmax_idx].get());
676 if (!sm || sm->logSoftMax) return false;
677 if (prog[softmax_idx]->inputs.size() != 1) return false;
678
679 Arg sm_in = prog[softmax_idx]->inputs[0];
680 auto it = producer_.find(sm_in.idx);
681 if (it == producer_.end()) return false;
682 int qk_matmul_idx = it->second;
683 if (qk_matmul_idx < 0 || !prog[qk_matmul_idx]) return false;
684 if (!dynamic_cast<MatMulLayer*>(prog[qk_matmul_idx].get())) return false;
685 if (!prog[qk_matmul_idx]->blobs.empty()) return false;
686 if (prog[qk_matmul_idx]->inputs.size() != 2) return false;
687 Arg q_arg = prog[qk_matmul_idx]->inputs[0];
688 Arg k_arg = prog[qk_matmul_idx]->inputs[1];
689
690 Arg sm_out = prog[softmax_idx]->outputs[0];
691 int av_matmul_idx = singleConsumer(sm_out);
692 if (av_matmul_idx < 0 || !prog[av_matmul_idx]) return false;
693 if (!dynamic_cast<MatMulLayer*>(prog[av_matmul_idx].get())) return false;
694 if (!prog[av_matmul_idx]->blobs.empty()) return false;
695 if (prog[av_matmul_idx]->inputs.size() != 2) return false;
696 Arg v_arg;
697 bool got_v = false;
698 for (Arg in : prog[av_matmul_idx]->inputs) {
699 if (in.idx == sm_out.idx) continue;
700 v_arg = in; got_v = true;
701 }
702 if (!got_v) return false;
703
704 std::set<int> consumed;
705 Mat Wq, Wk, Wv, bq, bk, bv;
706 int nh_q = 0, nh_k = 0, nh_v = 0;
707 float q_scale = 1.f;
708 float dummy = 1.f;
709 int q_mm_idx = traceClipBranch(prog, q_arg, /*is_q=*/true, /*is_k=*/false,
710 Wq, bq, nh_q, q_scale, consumed);
711 if (q_mm_idx < 0) return false;
712 int k_mm_idx = traceClipBranch(prog, k_arg, /*is_q=*/false, /*is_k=*/true,
713 Wk, bk, nh_k, dummy, consumed);
714 if (k_mm_idx < 0) return false;
715 int v_mm_idx = traceClipBranch(prog, v_arg, /*is_q=*/false, /*is_k=*/false,
716 Wv, bv, nh_v, dummy, consumed);
717 if (v_mm_idx < 0) return false;
718
719 if (q_mm_idx == k_mm_idx || k_mm_idx == v_mm_idx || q_mm_idx == v_mm_idx)
720 return false;
721 if (nh_q != nh_k || nh_k != nh_v) return false;
722 const int num_heads = nh_q;
723 if (q_scale == 0.f) return false;
724
725 // All three projections share the same input.
726 Arg shared_input = prog[q_mm_idx]->inputs[0];
727 if (prog[k_mm_idx]->inputs[0].idx != shared_input.idx) return false;
728 if (prog[v_mm_idx]->inputs[0].idx != shared_input.idx) return false;
729
730 if (Wq.dims != 2 || Wk.dims != 2 || Wv.dims != 2) return false;
731 if (Wq.size[0] != Wk.size[0] || Wk.size[0] != Wv.size[0]) return false;
732 const int hidden_in = Wq.size[0];
733 const int q_hidden = Wq.size[1];
734 const int k_hidden = Wk.size[1];
735 const int v_hidden = Wv.size[1];
736 if (q_hidden != k_hidden || k_hidden != v_hidden) return false;
737 if (q_hidden % num_heads != 0) return false;
738
739 // Output chain: Reshape4D -> Transpose([0,2,1,3]) -> Reshape3D, the
740 // boundary of the fused block.
741 Arg av_out = prog[av_matmul_idx]->outputs[0];
742 int out_r4d = singleConsumer(av_out);
743 if (!isReshape(prog, out_r4d)) return false;
744 Arg out_r4d_out = prog[out_r4d]->outputs[0];
745 int out_t4d = singleConsumer(out_r4d_out);
746 if (!isTranspose(prog, out_t4d)) return false;
747 TransposeLayer* out_tr =
748 dynamic_cast<TransposeLayer*>(prog[out_t4d].get());
749 if (!out_tr || out_tr->perm.size() != 4) return false;
750 if (out_tr->perm[0] != 0 || out_tr->perm[1] != 2 ||
751 out_tr->perm[2] != 1 || out_tr->perm[3] != 3) return false;
752 Arg out_t4d_out = prog[out_t4d]->outputs[0];
753 int out_r3d = singleConsumer(out_t4d_out);
754 if (!isReshape(prog, out_r3d)) return false;
755
756 // Combined [Q|K|V] weight along the output dim.
757 int total_hidden = q_hidden + k_hidden + v_hidden;
758 int wshape[] = {hidden_in, total_hidden};
759 Mat W_qkv(2, wshape, CV_32F);
760 for (int r = 0; r < hidden_in; r++) {
761 float* dst = W_qkv.ptr<float>(r);
762 std::memcpy(dst, Wq.ptr<float>(r), q_hidden * sizeof(float));
763 std::memcpy(dst + q_hidden, Wk.ptr<float>(r), k_hidden * sizeof(float));
764 std::memcpy(dst + q_hidden + k_hidden, Wv.ptr<float>(r), v_hidden * sizeof(float));
765 }
766 Mat bias_qkv;
767 if (!bq.empty() && !bk.empty() && !bv.empty()) {
768 const int bias_total = q_hidden + k_hidden + v_hidden;
769 bias_qkv.create(1, &bias_total, CV_32F);
770 float* dst = bias_qkv.ptr<float>();
771 std::memcpy(dst, bq.ptr<float>(), q_hidden * sizeof(float));
772 std::memcpy(dst + q_hidden, bk.ptr<float>(), k_hidden * sizeof(float));
773 std::memcpy(dst + q_hidden + k_hidden, bv.ptr<float>(), v_hidden * sizeof(float));
774 }
775
776 // scale (pre-softmax divisor) = 1/q_scale, since Q was pre-multiplied.
777 const float param_scale = 1.f / q_scale;
778
779 LayerParams attn_params;
780 attn_params.name = prog[q_mm_idx]->name + "_fused_attention";
781 attn_params.type = "Attention";
782 attn_params.set("num_heads", num_heads);
783 int qkv_sizes[3] = { q_hidden, k_hidden, v_hidden };
784 attn_params.set("qkv_hidden_sizes", DictValue::arrayInt(qkv_sizes, 3));
785 attn_params.set("scale", param_scale);
786 attn_params.set("output_ndims", 3);
787 attn_params.blobs.push_back(W_qkv);
788 if (!bias_qkv.empty()) attn_params.blobs.push_back(bias_qkv);
789
790 Ptr<LayerInfo> attn_layer =
791 LayerFactory::createLayerInstance(attn_params.type, attn_params);
792 if (!attn_layer) return false;
793 attn_layer->inputs = { shared_input };
794 attn_layer->outputs = prog[out_r3d]->outputs;
795 attn_layer->netimpl = netimpl;
796
797 std::set<int> to_remove = {
798 q_mm_idx, k_mm_idx, v_mm_idx,
799 qk_matmul_idx, softmax_idx, av_matmul_idx,
800 out_r4d, out_t4d, out_r3d
801 };
802 for (int op : consumed) to_remove.insert(op);
803
804 std::set<int> attn_out_args;
805 for (const Arg& o : attn_layer->outputs) attn_out_args.insert(o.idx);
806 if (!removalIsSelfContained(prog, to_remove, attn_out_args))
807 return false;
808
809 for (int op : to_remove) removed_ops.insert(op);
810 int insert_pos = *std::min_element(to_remove.begin(), to_remove.end());
811 replacements.push_back({insert_pos, attn_layer});
812 return true;
813 }

815 // A removal set is self-contained if each removed op's output is re-produced by the
816 // replacement layer or consumed only by other removed ops. A graph output, or an output
817 // still feeding a surviving op, would otherwise lose its producer.
818 bool removalIsSelfContained(const vector<Ptr<LayerInfo>>& prog,
819 const std::set<int>& to_remove,
820 const std::set<int>& produced_out_args) const
821 {
822 for (int op_idx : to_remove) {
823 if (op_idx < 0 || op_idx >= (int)prog.size() || !prog[op_idx]) continue;
824 for (const Arg& out : prog[op_idx]->outputs) {
825 if (produced_out_args.count(out.idx)) continue;
826 if (graph_outputs_.count(out.idx)) return false;
827 auto cit = consumers_.find(out.idx);
828 if (cit == consumers_.end()) continue;
829 for (int c : cit->second)
830 if (!to_remove.count(c)) return false;
831 }
832 }
833 return true;
834 }

836 bool fuseGraph(Ptr<Graph>& graph)
837 {
838 const vector<Ptr<LayerInfo>>& prog = graph->prog();
839 size_t nops = prog.size();
840
841 producer_.clear();
842 consumers_.clear();
843 for (size_t i = 0; i < nops; i++) {
844 if (!prog[i]) continue;
845 for (Arg out : prog[i]->outputs)
846 producer_[out.idx] = (int)i;
847 for (Arg inp : prog[i]->inputs)
848 consumers_[inp.idx].push_back((int)i);
849 }
850
851 // A graph output is exported, not consumed by any op, so it never enters consumers_;
852 // track it separately so a removed op cannot silently drop an exported arg.
853 graph_outputs_.clear();
854 for (Arg out : graph->outputs())
855 graph_outputs_.insert(out.idx);
856
857 bool modified = false;
858 std::set<int> removed_ops;
859
860 // Pass 1: combined-QKV blocks.
861 for (size_t i = 0; i < nops; i++) {
862 if (!prog[i] || removed_ops.count((int)i)) continue;
863 if (tryFuseCombinedQKV(prog, (int)i, removed_ops, attention_replacements_))
864 modified = true;
865 }
866
867 // Pass 2: the original 3-separate-projection path.
868 std::map<int, vector<int>> qkv_candidates;
869 for (size_t i = 0; i < nops; i++) {
870 if (!prog[i] || removed_ops.count((int)i)) continue;
871 if (!isProjCandidate(prog[i])) continue;
872 Arg inp = prog[i]->inputs[0];
873 qkv_candidates[inp.idx].push_back((int)i);
874 }
875
876 for (auto& candidate : qkv_candidates) {
877 auto matmul_indices = candidate.second;
878 if (matmul_indices.size() < 3) continue;
879
880 for (size_t qi = 0; qi + 2 < matmul_indices.size(); qi++)
881 for (size_t ki = qi + 1; ki + 1 < matmul_indices.size(); ki++)
882 for (size_t vi = ki + 1; vi < matmul_indices.size(); vi++)
883 {
884 int indices[3] = { matmul_indices[qi], matmul_indices[ki], matmul_indices[vi] };
885
886 bool any_removed = false;
887 for (int idx : indices)
888 if (removed_ops.count(idx)) any_removed = true;
889 if (any_removed) continue;
890
891 bool shapes_ok = true;
892 for (int k = 0; k < 3; k++) {
893 const Mat& w = prog[indices[k]]->blobs[0];
894 if (w.dims != 2) { shapes_ok = false; break; }
895 }
896 if (!shapes_ok) continue;
897
898 int reshape_idx[3], num_heads[3], transpose_idx[3];
899 vector<int> perms[3];
900 std::set<int> extra_ops;
901 bool pattern_ok = true;
902 for (int k = 0; k < 3; k++) {
903 transpose_idx[k] = followProjChain(prog, indices[k],
904 &reshape_idx[k],
905 &num_heads[k], &perms[k],
906 &extra_ops);
907 if (transpose_idx[k] < 0) { pattern_ok = false; break; }
908 }
909 if (!pattern_ok) continue;
910 if (num_heads[0] != num_heads[1] || num_heads[1] != num_heads[2])
911 continue;
912
913 int k_slot = -1;
914 vector<int> perm_k = {0, 2, 3, 1};
915 for (int k = 0; k < 3; k++)
916 if (perms[k] == perm_k) k_slot = k;
917 if (k_slot < 0) continue;
918
919 int remaining[2];
920 { int j = 0; for (int k = 0; k < 3; k++) if (k != k_slot) remaining[j++] = k; }
921
922 Arg k_tr_out = prog[transpose_idx[k_slot]]->outputs[0];
923 // Tolerate a Shape consumer alongside the Mul/MatMul: the runtime-scale chain (Shape->Slice->Cast->Sqrt...) branches off the Q/K transpose.
924 int k_next = findMatchingConsumer(prog, k_tr_out,
925 [](LayerInfo* L) {
926 return dynamic_cast<NaryEltwiseLayer*>(L) != nullptr ||
927 dynamic_cast<MatMulLayer*>(L) != nullptr;
928 },
929 &extra_ops);
930 int k_mul_idx = -1;
931 float k_scale_val = 1.f;
932 int qk_matmul_idx = -1;
933 bool vit_style;
934 bool runtime_qk_scale = false;
935 std::set<int> qk_scale_chain_ops;
936 if (isScalarMul(prog, k_next, &k_scale_val)) {
937 vit_style = true;
938 k_mul_idx = k_next;
939 qk_matmul_idx = singleConsumer(prog[k_mul_idx]->outputs[0]);
940 } else if (isMatMul(prog, k_next)) {
941 vit_style = false;
942 qk_matmul_idx = k_next;
943 } else if (k_next >= 0 && k_next < (int)prog.size() && prog[k_next]) {
944 // K_Transpose -> Mul(K, runtime_scale) where scale = Sqrt(Cast(Div(1,Sqrt(Cast(Slice(Shape(...))))))) and the same scale feeds the Q-side Mul (verified later).
945 NaryEltwiseLayer* elt = dynamic_cast<NaryEltwiseLayer*>(prog[k_next].get());
946 if (!elt || elt->op != NaryEltwiseLayer::OPERATION::PROD ||
947 prog[k_next]->inputs.size() != 2) {
948 continue;
949 }
950 Arg k_scale_in;
951 bool tensor_input_seen = false;
952 for (Arg in : prog[k_next]->inputs) {
953 if (in.idx == k_tr_out.idx) tensor_input_seen = true;
954 else k_scale_in = in;
955 }
956 if (!tensor_input_seen) continue;
957 if (!isRuntimeQKScaleChain(prog, k_scale_in, qk_scale_chain_ops)) continue;
958
959 runtime_qk_scale = true;
960 vit_style = true;
961 k_mul_idx = k_next;
962 qk_matmul_idx = singleConsumer(prog[k_mul_idx]->outputs[0]);
963 } else {
964 continue;
965 }
966 if (!isMatMul(prog, qk_matmul_idx)) continue;
967
968 int q_slot = -1, v_slot = -1;
969 for (int attempt = 0; attempt < 2; attempt++) {
970 q_slot = remaining[attempt];
971 v_slot = remaining[1 - attempt];
972
973 int q_mul_idx = -1;
974 float q_scale_val = 1.f;
975 Arg q_tr_out = prog[transpose_idx[q_slot]]->outputs[0];
976
977 if (vit_style) {
978 int q_next = findMatchingConsumer(prog, q_tr_out,
979 [](LayerInfo* L) {
980 return dynamic_cast<NaryEltwiseLayer*>(L) != nullptr ||
981 dynamic_cast<MatMulLayer*>(L) != nullptr;
982 },
983 &extra_ops);
984 if (isScalarMul(prog, q_next, &q_scale_val)) {
985 // existing constant-scalar path
986 } else if (runtime_qk_scale && q_next >= 0 &&
987 q_next < (int)prog.size() && prog[q_next]) {
988 NaryEltwiseLayer* elt = dynamic_cast<NaryEltwiseLayer*>(prog[q_next].get());
989 if (!elt || elt->op != NaryEltwiseLayer::OPERATION::PROD ||
990 prog[q_next]->inputs.size() != 2) {
991 continue;
992 }
993 Arg q_scale_in;
994 bool tensor_input_seen = false;
995 for (Arg in : prog[q_next]->inputs) {
996 if (in.idx == q_tr_out.idx) tensor_input_seen = true;
997 else q_scale_in = in;
998 }
999 if (!tensor_input_seen) continue;
1000 if (!isRuntimeQKScaleChain(prog, q_scale_in, qk_scale_chain_ops)) continue;
1001 } else {
1002 continue;
1003 }
1004 q_mul_idx = q_next;
1005 Arg q_mul_out = prog[q_mul_idx]->outputs[0];
1006 int q_dest = singleConsumer(q_mul_out);
1007 if (q_dest != qk_matmul_idx) continue;
1008 } else {
1009 const auto& qk_inputs = prog[qk_matmul_idx]->inputs;
1010 bool q_connected = false;
1011 for (auto& ai : qk_inputs)
1012 if (ai.idx == q_tr_out.idx) q_connected = true;
1013 if (!q_connected) continue;
1014 }
1015
1016 // qk_matmul → [Div(√d_k)] → [Add(mask)] → Softmax (BERT)
1017 // OR → Softmax (ViT)
1018 Arg cur_out = prog[qk_matmul_idx]->outputs[0];
1019 int qk_div_idx = -1;
1020 int qk_add_idx = -1;
1021 Arg mask_arg;
1022 bool has_mask = false;
1023 float post_scale_val = 1.f;
1024
1025 int cur_consumer = singleConsumer(cur_out);
1026 if (!vit_style) {
1027 if (!isScalarDiv(prog, cur_consumer, &post_scale_val)) continue;
1028 qk_div_idx = cur_consumer;
1029 cur_out = prog[qk_div_idx]->outputs[0];
1030 cur_consumer = singleConsumer(cur_out);
1031 Arg maybe_mask;
1032 if (isMaskAdd(prog, cur_consumer, &maybe_mask)) {
1033 qk_add_idx = cur_consumer;
1034 mask_arg = maybe_mask;
1035 has_mask = true;
1036 cur_out = prog[qk_add_idx]->outputs[0];
1037 cur_consumer = singleConsumer(cur_out);
1038 }
1039 }
1040 int softmax_idx = cur_consumer;
1041 if (!isSoftmax(prog, softmax_idx)) continue;
1042
1043 Arg softmax_out = prog[softmax_idx]->outputs[0];
1044 int attnv_matmul_idx = singleConsumer(softmax_out);
1045 if (!isMatMul(prog, attnv_matmul_idx)) continue;
1046
1047 Arg v_tr_out = prog[transpose_idx[v_slot]]->outputs[0];
1048 const auto& av_inputs = prog[attnv_matmul_idx]->inputs;
1049 bool v_connected = false;
1050 for (auto& ai : av_inputs)
1051 if (ai.idx == v_tr_out.idx) v_connected = true;
1052 if (!v_connected) continue;
1053
1054 Arg attnv_out = prog[attnv_matmul_idx]->outputs[0];
1055 int out_transpose_idx = singleConsumer(attnv_out);
1056 if (!isTranspose(prog, out_transpose_idx)) continue;
1057
1058 Arg out_tr_out = prog[out_transpose_idx]->outputs[0];
1059 int out_reshape_idx = findMatchingConsumer(prog, out_tr_out,
1060 [](LayerInfo* L){ return dynamic_cast<Reshape2Layer*>(L) != nullptr; },
1061 &extra_ops);
1062 if (!isReshape(prog, out_reshape_idx)) continue;
1063
1064 const auto& or_inputs = prog[out_reshape_idx]->inputs;
1065 if (or_inputs.size() >= 2 && !netimpl->isConstArg(or_inputs[1])) {
1066 auto it = producer_.find(or_inputs[1].idx);
1067 if (it != producer_.end())
1068 collectShapeChain(prog, it->second, extra_ops);
1069 }

1072 // whether the source op is MatMul or Gemm-with-trans_b.
1073 Mat Wq = getProjWeight(prog[indices[q_slot]]);
1074 Mat Wk = getProjWeight(prog[indices[k_slot]]);
1075 Mat Wv = getProjWeight(prog[indices[v_slot]]);

1077 int q_hidden = Wq.size[1];
1078 int k_hidden = Wk.size[1];
1079 int v_hidden = Wv.size[1];
1080 int total_hidden = q_hidden + k_hidden + v_hidden;
1081
1082 int wshape[] = {input_hidden, total_hidden};
1083 Mat W_qkv(2, wshape, CV_32F);
1084 for (int r = 0; r < input_hidden; r++) {
1085 float* dst = W_qkv.ptr<float>(r);
1086 memcpy(dst, Wq.ptr<float>(r), q_hidden * sizeof(float));
1087 memcpy(dst + q_hidden, Wk.ptr<float>(r), k_hidden * sizeof(float));
1088 memcpy(dst + q_hidden + k_hidden, Wv.ptr<float>(r), v_hidden * sizeof(float));
1089 }
1090
1091 Mat bias_qkv;
1092 bool has_bias = prog[indices[q_slot]]->blobs.size() >= 2;
1093 if (has_bias) {
1094 const Mat& bq = prog[indices[q_slot]]->blobs[1];
1095 const Mat& bk = prog[indices[k_slot]]->blobs[1];
1096 const Mat& bv = prog[indices[v_slot]]->blobs[1];
1097 int bias_total = (int)(bq.total() + bk.total() + bv.total());
1098 bias_qkv.create(1, &bias_total, CV_32F);
1099 float* dst = bias_qkv.ptr<float>();
1100 memcpy(dst, bq.ptr<float>(), bq.total() * sizeof(float));
1101 memcpy(dst + bq.total(), bk.ptr<float>(), bk.total() * sizeof(float));
1102 memcpy(dst + bq.total() + bk.total(), bv.ptr<float>(), bv.total() * sizeof(float));
1103 }
1104
1105 float param_scale;
1106 if (runtime_qk_scale) {
1107 // Both Q and K were scaled by head_dim^-0.25, so
1108 // (Q'*K'^T) = (1/sqrt(d)) * (Q*K^T). The Attention
1109 // layer's `scale` param is the divisor applied
1110 // before softmax — i.e., sqrt(d).
1111 if (num_heads[0] <= 0 || q_hidden % num_heads[0] != 0) continue;
1112 int head_dim = q_hidden / num_heads[0];
1113 if (head_dim <= 0) continue;
1114 param_scale = std::sqrt((float)head_dim);
1115 } else {
1116 param_scale = vit_style ? (1.f / (q_scale_val * k_scale_val))
1117 : post_scale_val;
1118 }
1119
1120 LayerParams attn_params;
1121 attn_params.name = prog[indices[q_slot]]->name + "_fused_attention";
1122 attn_params.type = "Attention";
1123 attn_params.set("num_heads", num_heads[0]);
1124 DictValue qkv_sizes_param = DictValue::arrayInt(
1125 std::vector<int>{q_hidden, k_hidden, v_hidden}.data(), 3);
1126 attn_params.set("qkv_hidden_sizes", qkv_sizes_param);
1127 attn_params.set("scale", param_scale);
1128 attn_params.set("output_ndims", 3);
1129
1130 attn_params.blobs.push_back(W_qkv);
1131 if (has_bias)
1132 attn_params.blobs.push_back(bias_qkv);
1133
1134 Ptr<LayerInfo> attn_layer = LayerFactory::createLayerInstance(
1135 attn_params.type, attn_params);
1136 CV_Assert(attn_layer);
1137
1138 Arg shared_input = prog[indices[0]]->inputs[0];
1139 if (has_mask)
1140 attn_layer->inputs = { shared_input, mask_arg };
1141 else
1142 attn_layer->inputs = { shared_input };
1143 attn_layer->outputs = prog[out_reshape_idx]->outputs;
1144 attn_layer->netimpl = netimpl;
1145
1146 std::set<int> to_remove = {
1147 indices[0], indices[1], indices[2],
1148 reshape_idx[0], reshape_idx[1], reshape_idx[2],
1149 transpose_idx[0], transpose_idx[1], transpose_idx[2],
1150 qk_matmul_idx, softmax_idx, attnv_matmul_idx,
1151 out_transpose_idx, out_reshape_idx
1152 };
1153 if (k_mul_idx >= 0) to_remove.insert(k_mul_idx);
1154 if (q_mul_idx >= 0) to_remove.insert(q_mul_idx);
1155 if (qk_div_idx >= 0) to_remove.insert(qk_div_idx);
1156 if (qk_add_idx >= 0) to_remove.insert(qk_add_idx);
1157 for (int op : extra_ops) to_remove.insert(op);
1158
1159 // Drop chain ops (Sqrt/Cast/Div/Slice/Shape) only if every consumer is also being removed — by this fusion or another chain op already in the set; the upstream is shared between Q and K branches so the chain is treated as a unit.
1160 for (int op_idx : qk_scale_chain_ops) {
1161 if (op_idx < 0 || op_idx >= (int)prog.size() || !prog[op_idx]) continue;
1162 bool all_consumers_removed = true;
1163 for (Arg out : prog[op_idx]->outputs) {
1164 auto cit = consumers_.find(out.idx);
1165 if (cit == consumers_.end()) continue;
1166 for (int c : cit->second) {
1167 if (to_remove.count(c) || qk_scale_chain_ops.count(c)) continue;
1168 all_consumers_removed = false;
1169 break;
1170 }
1171 if (!all_consumers_removed) break;
1172 }
1173 if (all_consumers_removed) to_remove.insert(op_idx);
1174 }
1175
1176 std::set<int> attn_out_args;
1177 for (const Arg& o : attn_layer->outputs) attn_out_args.insert(o.idx);
1178 if (!removalIsSelfContained(prog, to_remove, attn_out_args))
1179 continue;
1180
1181 for (int op : to_remove)
1182 removed_ops.insert(op);
1183
1184 int insert_pos = *std::min_element(to_remove.begin(), to_remove.end());
1185 attention_replacements_.push_back({insert_pos, attn_layer});
1186 modified = true;
1187 break;
1188 }
1189 }
1190 }
1191
1192 // Pass 3: anchored on each Softmax.
1193 for (size_t i = 0; i < nops; i++) {
1194 if (!prog[i] || removed_ops.count((int)i)) continue;
1195 if (prog[i]->type != "Softmax") continue;
1196 if (tryFuseClipAttention(prog, (int)i, removed_ops,
1197 attention_replacements_))
1198 modified = true;
1199 }
1200
1201 if (modified) {
1202 vector<Ptr<LayerInfo>> newprog;
1203 std::sort(attention_replacements_.begin(), attention_replacements_.end(),
1204 [](auto& a, auto& b) { return a.first < b.first; });
1205
1206 size_t repl_idx = 0;
1207 for (size_t i = 0; i < nops; i++) {
1208 while (repl_idx < attention_replacements_.size() &&
1209 attention_replacements_[repl_idx].first == (int)i) {
1210 newprog.push_back(attention_replacements_[repl_idx].second);
1211 repl_idx++;
1212 }
1213 if (removed_ops.count((int)i))
1214 continue;
1215 newprog.push_back(prog[i]);
1216 }
1217 graph->setProg(newprog);
1218 }
1219
1220 return modified;
1221 }
