35// Detect a "swap innermost two dims" permutation, collapsing leading dims.
36// Returns true and fills outer/rows/cols when applicable.
37static bool isSwapLastTwo(const MatShape& inpShape,
38 const std::vector<int>& perm,
39 int esz,
40 int64_t& outer, int64_t& rows, int64_t& cols)
41{
42 if (esz != 4) return false;
43 int nd = inpShape.dims;
44 if (nd < 2 || (int)perm.size() != nd) return false;
45 for (int i = 0; i < nd - 2; i++)
46 if (perm[i] != i) return false;
47 if (perm[nd - 2] != nd - 1 || perm[nd - 1] != nd - 2) return false;
48 outer = 1;
49 for (int i = 0; i < nd - 2; i++) outer *= inpShape[i];
50 rows = inpShape[nd - 1]; // output rows = input inner dim
51 cols = inpShape[nd - 2]; // output cols = input outer-of-pair dim
52 return true;
53}

55static void transpose(const Mat& inp, const std::vector<int>& perm, Mat& out)
56{
57 enum {TRANSPOSE_MAX_DIMS=7};
58 MatShape inpShape = inp.shape();
59 MatShape outShape = out.shape();
60 int ndims = inpShape.dims;
61 size_t esz = inp.elemSize();
62 CV_Assert(esz == 1 || esz == 2 || esz == 4 || esz == 8);
63
64 int64_t outer = 1, rows = 0, cols = 0;
65 if (isSwapLastTwo(inpShape, perm, (int)esz, outer, rows, cols)
66 && inp.isContinuous() && out.isContinuous()) {
67 CV_CPU_DISPATCH(transpose2D_f32_,
68 (inp.ptr<float>(), out.ptr<float>(), outer, rows, cols),
69 NEON, AVX2, AVX, BASELINE);
70 return;
71 }
72
73 if (inp.isContinuous() && out.isContinuous() &&
74 ndims >= 2 && (int)perm.size() == ndims &&
75 perm[ndims - 1] == ndims - 1)
76 {
77 std::vector<int64_t> inStride(ndims, 0);
78 inStride[ndims - 1] = 1;
79 for (int i = ndims - 2; i >= 0; i--)
80 inStride[i] = inStride[i + 1] * (int64_t)inpShape[i + 1];
81 const int64_t inner = inpShape[ndims - 1];
82
83 std::vector<int> outOuterShape(ndims - 1);
84 for (int i = 0; i < ndims - 1; i++) outOuterShape[i] = outShape[i];
85 int64_t outerTotal = 1;
86 for (int i = 0; i < ndims - 1; i++) outerTotal *= outOuterShape[i];
87
88 if (outerTotal > 0 && inner > 0) {
89 const size_t innerBytes = (size_t)inner * esz;
90 const char* in_base = (const char*)inp.data;
91 char* out_base = (char*)out.data;
92
93 parallel_for_(Range(0, (int)outerTotal), [&](const Range& r) {
94 std::vector<int> outIdx(ndims - 1, 0);
95 // initialize outIdx to r.start in row-major
96 int64_t rem = r.start;
97 for (int k = ndims - 2; k >= 0; k--) {
98 outIdx[k] = (int)(rem % outOuterShape[k]);
99 rem /= outOuterShape[k];
100 }
101 for (int64_t idx = r.start; idx < r.end; idx++) {
102 // Compute input outer offset using perm.
103 int64_t inOff = 0;
104 for (int i = 0; i < ndims - 1; i++) {
105 // perm[i] is the input axis providing this output axis.
106 inOff += (int64_t)outIdx[i] * inStride[perm[i]];
107 }
108 // Output position: linear idx * inner.
109 std::memcpy(out_base + (size_t)idx * innerBytes,
110 in_base + (size_t)inOff * esz,
111 innerBytes);
112 // Advance outIdx (row-major over outOuterShape).
113 for (int k = ndims - 2; k >= 0; k--) {
114 if (++outIdx[k] < outOuterShape[k]) break;
115 outIdx[k] = 0;
116 }
117 }
118 }, std::max(1.0, (double)outerTotal * (double)innerBytes / 16384.0));
119 return;
120 }
121 }

123 int perm_[TRANSPOSE_MAX_DIMS];
124 int inpShape_[TRANSPOSE_MAX_DIMS];
125 int outShape_[TRANSPOSE_MAX_DIMS];
126 size_t inpStep_[TRANSPOSE_MAX_DIMS];
127 int delta = TRANSPOSE_MAX_DIMS - ndims;
128
129 CV_Assert(ndims <= TRANSPOSE_MAX_DIMS);
130 CV_Assert(inp.isContinuous());
131 CV_Assert(out.isContinuous());
132
133 for (int i = 0; i < TRANSPOSE_MAX_DIMS; i++) {
134 perm_[i] = i;
135 inpShape_[i] = outShape_[i] = 1;
136 inpStep_[i] = 0;
137 }
138 inpStep_[TRANSPOSE_MAX_DIMS-1] = 1; // step's are measured in elements, not bytes
139
140 for(int i = 0; i < ndims; i++) {
141 int j = perm.empty() ? ndims - i - 1 : perm[i];
142 if (j < 0)
143 j += ndims;
144 CV_Assert(0 <= j && j < ndims);
145 perm_[i + delta] = j + delta;
146 int inpsz = inpShape[j];
147 int outsz = outShape[i];
148 CV_Assert(inpsz == outsz);
149 inpShape_[i + delta] = inpShape[i];
150 outShape_[i + delta] = outShape[i];
151 }
152
153 for (int i = TRANSPOSE_MAX_DIMS-2; i >= 0; i--)
154 inpStep_[i] = inpStep_[i+1]*inpShape_[i+1];
155
156 int sz6 = outShape_[0], sz5 = outShape_[1];
157 int sz4 = outShape_[2], sz3 = outShape_[3];
158 int sz2 = outShape_[4], sz1 = outShape_[5], sz0 = outShape_[6];
159 size_t p6 = inpStep_[perm_[0]], p5 = inpStep_[perm_[1]];
160 size_t p4 = inpStep_[perm_[2]], p3 = inpStep_[perm_[3]];
161 size_t p2 = inpStep_[perm_[4]], p1 = inpStep_[perm_[5]], p0 = inpStep_[perm_[6]];
162
163 int64_t outerTotal = (int64_t)sz6 * sz5 * sz4 * sz3 * sz2;
164
165#undef CV_IMPLEMENT_TRANSPOSE
166#define CV_IMPLEMENT_TRANSPOSE(typ) \
167 parallel_for_(Range(0, (int)outerTotal), [&](const Range& r) { \
168 const typ* inptr0 = (const typ*)inp.data; \
169 typ* outptr0 = (typ*)out.data; \
170 for (int64_t idx = r.start; idx < r.end; idx++) { \
171 int64_t q = idx; \
172 int i2 = (int)(q % sz2); q /= sz2; \
173 int i3 = (int)(q % sz3); q /= sz3; \
174 int i4 = (int)(q % sz4); q /= sz4; \
175 int i5 = (int)(q % sz5); q /= sz5; \
176 int i6 = (int)q; \
177 const typ* inptrBase = inptr0 + i6*p6 + i5*p5 + i4*p4 + i3*p3 + i2*p2; \
178 typ* outptr = outptr0 + idx * ((int64_t)sz1 * sz0); \
179 for (int i1 = 0; i1 < sz1; i1++, outptr += sz0) { \
180 const typ* inptr = inptrBase + i1*p1; \
181 int i0 = 0; \
182 for (; i0 <= sz0 - 3; i0 += 3) { \
183 size_t ip0 = i0*p0; \
184 typ t0 = inptr[ip0]; \
185 typ t1 = inptr[ip0+p0]; \
186 typ t2 = inptr[ip0+p0*2]; \
187 outptr[i0] = t0; \
188 outptr[i0+1] = t1; \
189 outptr[i0+2] = t2; \
190 } \
191 for (; i0 < sz0; i0++) \
192 outptr[i0] = inptr[i0*p0]; \
193 } \
194 } \
195 });
196
197 if (esz == 4) {
198 CV_IMPLEMENT_TRANSPOSE(int)
199 } else if (esz == 2) {
200 CV_IMPLEMENT_TRANSPOSE(short)
201 } else if (esz == 1) {
202 CV_IMPLEMENT_TRANSPOSE(char)
203 } else if (esz == 8) {
204 CV_IMPLEMENT_TRANSPOSE(int64_t)

222 MatShape getOutShape(const MatShape& inpShape) const
223 {
224 MatShape outShape(inpShape.dims);
225 CV_Assert(perm.empty() || perm.size() == (size_t)inpShape.dims);
226
227 for (int i = 0; i < inpShape.dims; i++) {
228 int j = perm.empty() ? inpShape.dims - i - 1 : perm[i];
229 CV_Assert(0 <= j && j < inpShape.dims);
230 outShape[i] = inpShape[j];
231 }
232
233 return outShape;
234 }

269 CV_TRACE_FUNCTION();
270 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
271
272 Size size = inputs_arr.size();
273 int ninputs = size.area();
274 CV_Assert(ninputs == 1);
275
276 MatShape inpShape = inputs_arr.shape(0);
277 MatShape outShape = getOutShape(inpShape);
278 int inpType = inputs_arr.type(0);
279 int outKind = outputs_arr.kind();
280
281 CV_Assert(outKind == _InputArray::STD_VECTOR_MAT ||
282 outKind == _InputArray::STD_VECTOR_UMAT);
283
284 if (outKind == _InputArray::STD_VECTOR_MAT) {
285 Mat inp = inputs_arr.getMat(0);
286 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
287 outs.resize(1);
288 outs[0].fit(outShape, inpType);
289 runOp(inp, outs[0]);
290 } else {
291 // [TODO] more efficient OpenCL implementation
292 Mat inp = inputs_arr.getMat(0);
293 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
294 outs.resize(1);
295 outs[0].fit(outShape, inpType);
296 Mat temp(outShape, inpType);
297 runOp(inp, temp);
298 temp.copyTo(outs[0]);
299 }
300 }
