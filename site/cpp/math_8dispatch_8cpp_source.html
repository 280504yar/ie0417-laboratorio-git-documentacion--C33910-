17// ---- pluggable-HAL bridge for exp/log --------------------------------------------------------
18// The raw cv_hal_* entry points return int for a reason: without an installed HAL they are stubs
19// returning CV_HAL_ERROR_NOT_IMPLEMENTED. getMathFunc PROBES each one once (a 1-element call on
20// the safe input 1.0 - fine for both exp and log): implemented -> wrap it as an engine kernel
21// (the function pointer rides in TKernel::userdata, castKernel-style) and the engine adds tiling
22// and parallelism on top of the vendor code; not implemented -> the engine's own v_exp/v_log
23// kernels. Uniform over ANY HAL (an external vendor one, the IPP HAL module, ...) - the get is
24// called once per program build, the probe cost is nothing next to the kernel calls that follow.
25template<typename T>
26static int halUnaryKernel(const void* src0_, size_t s0y, size_t s0x,
27 const void*, size_t, size_t, const void*, size_t, size_t,
28 void* dst_, size_t dsty, int width, int height,
29 const double*, int, void* userdata)
30{
31 typedef int (*HalFunc)(const T*, T*, int);
32 const HalFunc fn = (HalFunc)userdata;
33 s0y /= sizeof(T);
34 dsty /= sizeof(T);
35 CV_Assert(s0x <= 1u);
36 const T* src0 = (const T*)src0_;
37 T* dst = (T*)dst_;
38 if (height > 1 && dsty == (size_t)width && s0y == s0x*(size_t)width) { width *= height; height = 1; }
39 const int urows = (s0y == 0 && height > 1) ? 1 : height; // vertical broadcast: 1 row + copies
40 for (int y = 0; y < urows; y++, src0 += s0y, dst += dsty)
41 {
42 int code;
43 if (s0x == 0) // broadcast-scalar source: one value covers the row
44 {
45 T v;
46 code = fn(src0, &v, 1);
47 for (int x = 0; x < width; x++) dst[x] = v;
48 }
49 else
50 code = fn(src0, dst, width);
51 if (code != CV_HAL_ERROR_OK)
52 return code; // shouldn't happen (probed at get time) - let exec assert
53 }
54 dst = (T*)dst_;
55 for (int y = urows; y < height; y++)
56 memcpy(dst + (size_t)y*dsty, dst, (size_t)width*sizeof(T));
57 return 0;
58}

60template<typename T, typename HalFunc>
61static TKernel probeHalUnary(HalFunc fn)
62{
63 T one = (T)1, r = (T)0;
64 if (fn(&one, &r, 1) == CV_HAL_ERROR_OK)
65 return {halUnaryKernel<T>, (void*)fn, 0};
66 return {};
67}

69// The engine's OWN kernel for (op, T) - v_exp/v_log & co, no HAL/IPP tiers. The final fallback of
70// getMathFunc, and what hal::exp32f & co use as THEIR built-in implementation (the former table
71// kernels are gone), via mathSpanEngine below.
72static TKernel getEngineMathFunc(TOp op, int T)
73{
74 CV_CPU_DISPATCH(getMathFunc_, (op, T), CV_CPU_DISPATCH_MODES_ALL);
75}

77// run the engine's own math kernel over one contiguous span (the shape hal::exp32f & co need)
78void mathSpanEngine(TOp op, int depth, const void* src, void* dst, int n)
79{
80 TKernel k = getEngineMathFunc(op, depth);
81 CV_Assert(k.fptr);
82 const double noparams[4] = {};
83 k.fptr(src, 0, 1, nullptr, 0, 0, nullptr, 0, 0, dst, 0, n, 1, noparams, k.flags, k.userdata);
84}

86TKernel getMathFunc(TOp op, int T)
87{
88 if ((op == OP_EXP || op == OP_LOG) && (T == CV_32F || T == CV_64F))
89 {
90 // IPP now rides the cv_hal_* hooks too (hal/ipp), so the single probe below picks it up
91 // together with any external vendor HAL - no separate IPP tier needed here.
92 // probe results are process-lifetime stable; cache them (thread-safe magic statics)
93 static const TKernel exp32 = probeHalUnary<float >(cv_hal_exp32f);
94 static const TKernel exp64 = probeHalUnary<double>(cv_hal_exp64f);
95 static const TKernel log32 = probeHalUnary<float >(cv_hal_log32f);
96 static const TKernel log64 = probeHalUnary<double>(cv_hal_log64f);
97 const TKernel* k = op == OP_EXP ? (T == CV_32F ? &exp32 : &exp64)
98 : (T == CV_32F ? &log32 : &log64);
99 if (k->fptr)
100 return *k;
101 }
102 return getEngineMathFunc(op, T);
103}
