14static constexpr std::uint32_t RANK_MAX = std::numeric_limits<std::uint32_t>::max();
15static constexpr std::size_t SZ_MAX = std::numeric_limits<std::size_t>::max();

17static std::uint32_t maybeGetRank(const ByteVecRankMap& ranks, const std::vector<std::uint8_t>& key) {
18 auto it = ranks.find(key);
19 return it == ranks.end() ? RANK_MAX : it->second;
20}
21/*
22 * This function takes a sequence of bytes and a map of
23 * mergeable byte pairs (with associated ranks/merge priority), and repeatedly merges the
24 * lowest-ranked adjacent pairs until no further merges are possible. The result is a vector
25 * describing the split points and ranks of the final token boundaries.
26 *
27 * This function is closely modeled after the original Rust implementation in OpenAI's tiktoken
28 * library, which can be found here:
29 * https://github.com/openai/tiktoken/blob/4560a8896f5fb1d35c6f8fd6eee0399f9a1a27ca/src/lib.rs#L17-L73
30 *
31 */
32std::vector<std::pair<std::size_t, std::uint32_t>> bytePairMerge(const ByteVecRankMap& ranks,
33 const std::vector<std::uint8_t>& piece) {
34 std::vector<std::pair<std::size_t, std::uint32_t>> parts;
35 parts.reserve(piece.size() + 1);
36
37 std::pair<std::uint32_t, std::size_t> minRank{RANK_MAX, SZ_MAX};
38
39 for (std::size_t i = 0; i+1 < piece.size(); ++i) {
40 std::vector<std::uint8_t> key(piece.begin() + i, piece.begin() + i + 2);
41 std::uint32_t r = maybeGetRank(ranks, key);
42 if (r < minRank.first) minRank = {r, i};
43 parts.emplace_back(i, r);
44 }
45 parts.emplace_back(piece.size() - 1, RANK_MAX);
46 parts.emplace_back(piece.size(), RANK_MAX);
47
48 auto getRank = [&](const std::vector<std::pair<std::size_t, std::uint32_t>>& p,
49 std::size_t idx) -> std::uint32_t {
50 if (idx + 3 < p.size()) {
51 std::size_t s = p[idx].first;
52 std::size_t e = p[idx+3].first;
53 std::vector<std::uint8_t> key(piece.begin()+s, piece.begin()+e);
54 return maybeGetRank(ranks, key);
55 }
56 return RANK_MAX;
57 };
58
59 while (minRank.first != RANK_MAX) {
60 std::size_t i = minRank.second;
61 if (i > 0) parts[i-1].second = getRank(parts, i-1);
62 if (i < parts.size()-2) parts[i].second = getRank(parts, i);
63 parts.erase(parts.begin() + static_cast<long>(i+1));
64
65 minRank = {RANK_MAX, SZ_MAX};
66 for (std::size_t j = 0; j + 1 < parts.size(); ++j) {
67 std::uint32_t r = parts[j].second;
68 if (r < minRank.first) minRank = {r, j};
69 }
70 }
71 return parts;
72}

74std::vector<std::uint32_t> bytePairEncode(const std::vector<std::uint8_t>& piece,
75 const ByteVecRankMap& ranks) {
76
77 if (piece.size() == 1u) {
78 auto it = ranks.find(piece);
79 return it == ranks.end() ? std::vector<std::uint32_t>{} : std::vector<std::uint32_t>{it->second};
80 }
81 auto merged = bytePairMerge(ranks, piece);
82 std::vector<std::uint32_t> out;
83 out.reserve(merged.size()-1);
84
85 for (std::size_t i = 0; i+1 < merged.size(); ++i) {
86 std::size_t s = merged[i].first, e = merged[i+1].first;
87 out.push_back(ranks.at(std::vector<std::uint8_t>(piece.begin()+s, piece.begin()+e)));
88 }
89 return out;
90}

92std::string CoreBPE::makeSpecialPattern(const std::unordered_map<std::string, std::uint32_t>& special) {
93 static const std::string meta = R"([.^$|()\‍[\‍]{}*+?\\‍])";
94 std::string pat;
95 pat.reserve(special.size() * 10);
96 bool first = true;
97 for (auto const& kv : special) {
98 if (!first) pat.push_back('|');
99 first = false;
100 // Escape each character in the token
101 for (char c : kv.first) {
102 if (meta.find(c) != std::string::npos)
103 pat.push_back('\\');
104 pat.push_back(c);
105 }
106 }
107 return pat;
108}

145CoreBPE::decodeBytes(const std::vector<std::uint32_t>& tokens) const {
146 std::vector<std::uint8_t> out;
147 out.reserve(tokens.size() * 2);
148
149 for (std::uint32_t t : tokens) {
150 const std::vector<std::uint8_t>* tokenBytes = nullptr;
151
152 auto it = decoder_.find(t);
153 if (it != decoder_.end()) {
154 tokenBytes = &it->second;
155 } else {
156 auto sit = specialDecoder_.find(t);
157 if (sit != specialDecoder_.end()) {
158 tokenBytes = &sit->second;
159 } else {
160 return std::vector<std::uint8_t>();
161 }
162 }
163 out.insert(out.end(), tokenBytes->begin(), tokenBytes->end());
164 }
165
166 return out;
167}

169/*
170 * This function tokenizes input text by handling special tokens and applying Byte Pair Encoding (BPE)
171 * to ordinary text segments. It searches for allowed special tokens, processes ordinary text with BPE,
172 * and emits a sequence of token IDs along with the count of tokens in the final processed segment.
173 *
174 * The logic and structure of this function are closely modeled after the original Rust implementation
175 * in OpenAI's tiktoken library, which can be found here:
176 * https://github.com/openai/tiktoken/blob/4560a8896f5fb1d35c6f8fd6eee0399f9a1a27ca/src/lib.rs#L234-L288
177 *
178 */
179std::pair<std::vector<std::uint32_t>, std::size_t>
180CoreBPE::encode(const std::string& text,
181 const std::unordered_set<std::string>& allowedSpecial) const
182{
183 std::vector<std::uint32_t> ret;
184 std::size_t last_piece_token_len = 0;
185 size_t start = 0;
186
187 // Use unicode_regex_split to find all special token matches in the text
188 std::vector<std::string> special_regexes = { specialPattern_ };
189 std::vector<std::pair<size_t, size_t>> special_matches;
190
191 // Find all special token matches and their positions
192 {
193 std::regex special_re(specialPattern_);
194 auto words_begin = std::sregex_iterator(text.begin(), text.end(), special_re);
195 auto words_end = std::sregex_iterator();
196 for (auto it = words_begin; it != words_end; ++it) {
197 std::string match_str = it->str();
198 if (allowedSpecial.count(match_str)) {
199 special_matches.emplace_back(it->position(), it->position() + match_str.size());
200 }
201 }
202 }
203
204 size_t match_idx = 0;
205 while (start < text.size()) {
206 // Find the next allowed special token
207 size_t next_special_start = std::string::npos;
208 size_t next_special_end = std::string::npos;
209 std::string matched_special;
210 while (match_idx < special_matches.size()) {
211 size_t s = special_matches[match_idx].first;
212 size_t e = special_matches[match_idx].second;
213 std::string candidate = text.substr(s, e - s);
214 if (allowedSpecial.count(candidate) && s >= start) {
215 next_special_start = s;
216 next_special_end = e;
217 matched_special = candidate;
218 break;
219 }
220 ++match_idx;
221 }
222
223 size_t end = (next_special_start != std::string::npos) ? next_special_start : text.size();
224
225 // Tokenize the ordinary segment [start, end)
226 if (end > start) {
227 std::string segment = text.substr(start, end - start);
228 std::vector<std::string> regexes = { pattern_ };
229 auto splits = unicode_regex_split(segment, regexes);
230
231 for (auto& subUtf8 : splits) {
232 std::vector<std::uint8_t> piece(subUtf8.begin(), subUtf8.end());
233 auto it = encoder_.find(piece);
234 if (it != encoder_.end()) {
235 last_piece_token_len = 1;
236 ret.push_back(it->second);
237 } else {
238 auto tokens = bytePairEncode(piece, encoder_);
239 last_piece_token_len = tokens.size();
240 ret.insert(ret.end(), tokens.begin(), tokens.end());
241 }
242 }
243 }
244
245 // If we found a special token, add it and advance
246 if (next_special_start != std::string::npos) {
247 std::uint32_t token = specialEncoder_.at(matched_special);
248 ret.push_back(token);
249 start = next_special_end;
250 last_piece_token_len = 0;
251 ++match_idx;
252 } else {
253 break;
254 }
255 }
256
257 return { ret, last_piece_token_len };
258}
