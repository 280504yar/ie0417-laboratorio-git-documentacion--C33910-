161void NetImplOpenVINO::forwardLayer(LayerData& ld)
162{
163 CV_TRACE_FUNCTION();
164
165 Ptr<Layer> layer = ld.layerInstance;
166
167 if (!ld.skip)
168 {
169 auto it = ld.backendNodes.find(preferableBackend);
170 if (ld.id == 0 || // input layer
171 it == ld.backendNodes.end() // non-supported layer or its mode
172 )
173 {
174 return Base::forwardLayer(ld);
175 }
176
177 CV_Assert(it != ld.backendNodes.end());
178 const Ptr<BackendNode>& node = it->second;
179 CV_Assert(!node.empty());
180 Ptr<InfEngineNgraphNode> ieNode = node.dynamicCast<InfEngineNgraphNode>();
181 CV_Assert(!ieNode.empty());
182 CV_Assert(ieNode->net);
183
184 TickMeter tm;
185 tm.start();
186
187 ieNode->net->forward(ld.outputBlobsWrappers, isAsync);
188
189 tm.stop();
190 int64 t = tm.getTimeTicks();
191 layersTimings[ld.id] = (t > 0) ? t : 1; // zero for skipped layers only
192 }
193 else
194 {
195 layersTimings[ld.id] = 0;
196 }
197
198 ld.flag = 1;
199}

201AsyncArray NetImplOpenVINO::getBlobAsync(const LayerPin& pin)
202{
203 CV_TRACE_FUNCTION();
204 if (!pin.valid())
205 CV_Error(Error::StsObjectNotFound, "Requested blob not found");
206
207 LayerData& ld = layers[pin.lid];
208 if ((size_t)pin.oid >= ld.outputBlobs.size())
209 {
210 CV_Error(Error::StsOutOfRange, format("Layer \"%s\" produce only %d outputs, "
211 "the #%d was requested",
212 ld.name.c_str(), (int)ld.outputBlobs.size(), (int)pin.oid));
213 }
214 if (preferableTarget != DNN_TARGET_CPU)
215 {
216 CV_Assert(!ld.outputBlobsWrappers.empty() && !ld.outputBlobsWrappers[pin.oid].empty());
217 // Transfer data to CPU if it's require.
218 ld.outputBlobsWrappers[pin.oid]->copyToHost();
219 }
220 CV_Assert(preferableBackend == DNN_BACKEND_INFERENCE_ENGINE_NGRAPH);
221
222 Ptr<NgraphBackendWrapper> wrapper = ld.outputBlobsWrappers[pin.oid].dynamicCast<NgraphBackendWrapper>();
223 return std::move(wrapper->futureMat);
224}

230void NetImplOpenVINO::addNgraphOutputs(LayerData& ld)
231{
232 CV_TRACE_FUNCTION();
233
234 CV_LOG_DEBUG(NULL, "DNN/IE: layer of new subnet: " << ld.name << "@" << ld.type);
235
236 Ptr<InfEngineNgraphNet> layerNet;
237 auto it = ld.backendNodes.find(preferableBackend);
238 if (it != ld.backendNodes.end())
239 {
240 Ptr<BackendNode> node = it->second;
241 if (!node.empty())
242 {
243 Ptr<InfEngineNgraphNode> ieNode = node.dynamicCast<InfEngineNgraphNode>();
244 CV_Assert(!ieNode.empty());
245 CV_Assert(!ieNode->net.empty());
246 layerNet = ieNode->net;
247 }
248 }
249
250 for (int i = 0; i < ld.inputBlobsId.size(); ++i)
251 {
252 LayerData& inpLd = layers[ld.inputBlobsId[i].lid];
253 Ptr<BackendNode> inpNode = inpLd.backendNodes[preferableBackend];
254 if (!inpNode.empty())
255 {
256 Ptr<InfEngineNgraphNode> ieInpNode = inpNode.dynamicCast<InfEngineNgraphNode>();
257 CV_Assert(!ieInpNode.empty());
258 CV_Assert(!ieInpNode->net.empty());
259 if (layerNet != ieInpNode->net)
260 {
261 CV_LOG_DEBUG(NULL, "DNN/IE: pin output between subnets: " << ieInpNode->node.get_node()->get_friendly_name());
262 ieInpNode->net->addOutput(ieInpNode);
263 }
264 }
265 }

268void NetImplOpenVINO::initBackend(const std::vector<LayerPin>& blobsToKeep_)
269{
270 CV_TRACE_FUNCTION();
271 CV_CheckEQ(preferableBackend, DNN_BACKEND_INFERENCE_ENGINE_NGRAPH, "");
272
273 Ptr<InfEngineNgraphNet> net;
274
275 for (MapIdToLayerData::const_iterator it = layers.begin(); it != layers.end(); ++it)
276 {
277 const LayerData& ld = it->second;
278 if (ld.id == 0)
279 {
280 CV_Assert((netInputLayer->outNames.empty() && ld.outputBlobsWrappers.size() == 1) ||
281 (netInputLayer->outNames.size() == ld.outputBlobsWrappers.size()));
282 for (int i = 0; i < ld.outputBlobsWrappers.size(); ++i)
283 {
284 std::string outputName = netInputLayer->outNames.empty() ? ld.name : netInputLayer->outNames[i];
285 outputName = ld.outputBlobsWrappers.size() > 1 ? (outputName + "." + std::to_string(i)) : outputName;
286 ld.outputBlobsWrappers[i].dynamicCast<NgraphBackendWrapper>()->name = outputName;
287 }
288 }
289 else
290 {
291 for (int i = 0; i < ld.outputBlobsWrappers.size(); ++i)
292 {
293 std::string outputName = ld.outputBlobsWrappers.size() > 1 ? (ld.name + "." + std::to_string(i)) : ld.name;
294 ld.outputBlobsWrappers[i].dynamicCast<NgraphBackendWrapper>()->name = outputName;
295 }
296 }
297 }
298
299 if (!basePtr_) // model is loaded by OpenVINO
300 {
301 Ptr<BackendNode> node = layers[lastLayerId].backendNodes[preferableBackend];
302 CV_Assert(!node.empty());
303
304 Ptr<InfEngineNgraphNode> ieNode = node.dynamicCast<InfEngineNgraphNode>();
305 CV_Assert(!ieNode.empty());
306
307 CV_Assert(ieNode->net);
308 InfEngineNgraphNet& ienet = *ieNode->net;
309 ienet.reset();
310
311 for (MapIdToLayerData::iterator it = layers.begin(); it != layers.end(); ++it)
312 {
313 LayerData& ld = it->second;
314 if (ld.id == 0)
315 {
316 for (int i = 0; i < ld.inputBlobsWrappers.size(); ++i)
317 {
318 ld.inputBlobsWrappers[i].dynamicCast<NgraphBackendWrapper>()->name = netInputLayer->outNames[i];
319 }
320 }
321 ienet.addBlobs(ld.inputBlobsWrappers);
322 ienet.addBlobs(ld.outputBlobsWrappers);
323 ld.skip = true;
324 }
325 layers[lastLayerId].skip = false;
326 ienet.init((Target)preferableTarget);
327 return;
328 }
329
330 // Build Inference Engine networks from sets of layers that support this
331 // backend. Split a whole model on several Inference Engine networks if
332 // some of layers are not implemented.
333 for (MapIdToLayerData::iterator it = layers.begin(); it != layers.end(); ++it)
334 {
335 LayerData& ld = it->second;
336
337 CV_LOG_DEBUG(NULL, "DNN/IE: processing layer " << ld.name << "@" << ld.type << " (" << ld.id << ") ...");
338
339 if (ld.id == 0 && ld.skip)
340 {
341 CV_LOG_DEBUG(NULL, "DNN/IE: SKIP!");
342 continue;
343 }
344
345 bool fused = ld.skip;
346 Ptr<Layer> layer = ld.layerInstance;
347 if (ld.id == 0)
348 continue;
349 ld.skip = true; // Initially skip all Inference Engine supported layers.
350
351 // Create a new network if one of inputs from different Inference Engine graph.
352 std::vector<Ptr<BackendNode>> inputNodes;
353 for (int i = 0; i < ld.inputBlobsId.size(); ++i)
354 {
355 // Layer_Test_ROIPooling.Accuracy has 2 inputs inpLD = 0, 0 -> has 4 inputNodes (input, rois, input, rois)
356 if (inputNodes.size() == ld.inputBlobsId.size())
357 {
358 break;
359 }
360 LayerData& inpLd = layers[ld.inputBlobsId[i].lid];
361 Ptr<BackendNode> inpNode = inpLd.backendNodes[preferableBackend];
362 if (!inpNode.empty())
363 {
364 Ptr<InfEngineNgraphNode> ieInpNode = inpNode.dynamicCast<InfEngineNgraphNode>();
365 CV_Assert(!ieInpNode.empty());
366 CV_Assert(!ieInpNode->net.empty());
367 if (ieInpNode->net == net && !fused)
368 {
369 inputNodes.push_back(inpNode);
370 continue;
371 }
372 }
373
374 if (net.empty())
375 {
376 net = Ptr<InfEngineNgraphNet>(new InfEngineNgraphNet(*this));
377 }
378
379 if (!fused)
380 {
381 std::vector<std::string> inputNames;
382 std::vector<cv::Mat> inputs;
383
384 auto curr_pos = inpLd.consumers.begin();
385 auto compare = [&ld](const LayerPin& lp) { return lp.lid == ld.id; };
386 auto cons = curr_pos;
387 while ((cons = std::find_if(curr_pos, inpLd.consumers.end(), compare)) !=
388 inpLd.consumers.end()) {
389 int cons_inp = cons->oid;
390 Ptr<NgraphBackendWrapper> inpWrapper = inpLd.outputBlobsWrappers[cons_inp].
391 dynamicCast<NgraphBackendWrapper>();
392 CV_Assert(!inpWrapper.empty());
393 auto iter = std::find(inputNames.begin(), inputNames.end(),
394 inpWrapper->name);
395 if (iter == inputNames.end())
396 {
397 inputNames.push_back(inpWrapper->name);
398 inputs.push_back(inpLd.outputBlobs[cons_inp]);
399 }
400 curr_pos = cons + 1;
401 }
402
403 auto inps = net->setInputs(inputs, inputNames);
404 for (auto& inp : inps)
405 {
406 inputNodes.emplace_back(Ptr<BackendNode>(new InfEngineNgraphNode(inp)));
407 }
408 }
409 }
410
411 Ptr<BackendNode> node;
412 if (!net.empty())
413 {
414 if (fused)
415 {
416 bool inPlace = ld.inputBlobsId.size() == 1 && ld.outputBlobs.size() == 1 &&
417 ld.inputBlobs[0]->data == ld.outputBlobs[0].data;
418 CV_Assert(inPlace);
419 node = layers[ld.inputBlobsId[0].lid].backendNodes[preferableBackend];
420 ld.inputBlobsWrappers = layers[ld.inputBlobsId[0].lid].inputBlobsWrappers;
421 }
422 }
423 else
424 {
425 net = Ptr<InfEngineNgraphNet>(new InfEngineNgraphNet(*this));
426 }
427
428 if (!fused)
429 {
430 CV_Assert(ld.inputBlobsId.size() == inputNodes.size());
431 for (int i = 0; i < ld.inputBlobsId.size(); ++i)
432 {
433 int lid = ld.inputBlobsId[i].lid;
434 int oid = ld.inputBlobsId[i].oid;
435
436 auto ieInpNode = inputNodes[i].dynamicCast<InfEngineNgraphNode>();
437 const auto& ngraph_input_node = ieInpNode->node.get_node_shared_ptr();
438 CV_LOG_DEBUG(NULL, "DNN/IE: bind output port " << lid << ":" << oid << " (" << ngraph_input_node->get_friendly_name() << ":" << ngraph_input_node->get_type_info().name << ")");
439
440 if ((oid == 0 && ngraph_input_node->get_output_size() == 1) || lid == 0)
441 continue;
442
443 // Handle parameters from other subnets. Output port is not used in this case
444 if ((ov::op::util::is_parameter(ngraph_input_node) || ov::op::util::is_constant(ngraph_input_node)) &&
445 ngraph_input_node->get_output_size() == 1)
446 {
447 inputNodes[i] = Ptr<BackendNode>(new InfEngineNgraphNode(ngraph_input_node));
448 continue;
449 }
450 CV_CheckLT((size_t)oid, ngraph_input_node->get_output_size(), "");
451 inputNodes[i] = new InfEngineNgraphNode(ngraph_input_node->output(oid));
452 }
453
454 if (layer->supportBackend(preferableBackend))
455 {
456 CV_LOG_DEBUG(NULL, "DNN/IE: wrap layer " << ld.name << "@" << ld.type << " - outputs: " << ld.outputBlobsWrappers.size());
457 node = layer->initNgraph(ld.inputBlobsWrappers, inputNodes);
458#if 0 // FIXIT doesn't work with multiple outputs (set name is applied to the same node)
459 for (int i = 0; i < ld.outputBlobsWrappers.size(); ++i)
460 {
461 InferenceEngine::DataPtr dataPtr = ngraphDataNode(ld.outputBlobsWrappers[i]);
462 node.dynamicCast<InfEngineNgraphNode>()->setName(dataPtr->getName());
463 }
464#else
465 node.dynamicCast<InfEngineNgraphNode>()->setName(layer->name);
466#endif
467 }
468 else
469 {
470 CV_LOG_DEBUG(NULL, "DNN/IE: layer is not supported: " << ld.name << "@" << ld.type);
471 node = Ptr<BackendNode>(new InfEngineNgraphNode(inputNodes,
472 ld.layerInstance, ld.inputBlobs, ld.outputBlobs, ld.internals));
473 }
474 }
475 else if (node.empty())
476 {
477 CV_LOG_DEBUG(NULL, "DNN/IE: node.empty() bypass...");
478 continue;
479 }
480
481 ld.backendNodes[preferableBackend] = node;
482
483 Ptr<InfEngineNgraphNode> ieNode = node.dynamicCast<InfEngineNgraphNode>();
484 CV_Assert(!ieNode.empty());
485 ieNode->net = net;
486
487 for (const auto& pin : blobsToKeep_)
488 {
489 if (pin.lid == ld.id)
490 {
491 ieNode->net->addOutput(ieNode);
492 break;
493 }
494 }
495
496 net->addBlobs(ld.inputBlobsWrappers);
497 net->addBlobs(ld.outputBlobsWrappers);
498 addNgraphOutputs(ld);
499 }
500
501 // User may choose to return only intermediate blobs but not network's result (see Test_TFLite.max_unpooling)
502 // Such layers should not be skipped when forwardLayer is called.
503 // Also, perform a sanity check that there is no double inferred networks (a single skip=false per unique net instance)
504 std::set<Ptr<InfEngineNgraphNet>> uniqueNets;
505 if (!blobsToKeep_.empty())
506 {
507 LayerPin latestLayerPin = getLatestLayerPin(blobsToKeep_);
508 for (MapIdToLayerData::iterator it = layers.begin(); it != layers.end(); ++it)
509 {
510 LayerData& ld = it->second;
511 auto iter = ld.backendNodes.find(preferableBackend);
512 if (iter == ld.backendNodes.end())
513 continue;
514
515 Ptr<BackendNode>& node = iter->second;
516 if (node.empty())
517 continue;
518
519 Ptr<InfEngineNgraphNode> ieNode = node.dynamicCast<InfEngineNgraphNode>();
520 if (ieNode.empty())
521 continue;
522
523 if (ld.id == latestLayerPin.lid) {
524 ld.skip = false;
525 uniqueNets.insert(ieNode->net);
526 break;
527 }
528 }
529 }
530
531 // Initialize all networks.
532 for (MapIdToLayerData::reverse_iterator it = layers.rbegin(); it != layers.rend(); ++it)
533 {
534 LayerData& ld = it->second;
535 auto iter = ld.backendNodes.find(preferableBackend);
536 if (iter == ld.backendNodes.end())
537 continue;
538
539 Ptr<BackendNode>& node = iter->second;
540 if (node.empty())
541 continue;
542
543 Ptr<InfEngineNgraphNode> ieNode = node.dynamicCast<InfEngineNgraphNode>();
544 if (ieNode.empty())
545 continue;
546
547 CV_Assert(!ieNode->net.empty());
548
549 if (!ieNode->net->isInitialized())
550 {
551 ieNode->net->addOutput(ieNode);
552 ieNode->net->createNet((Target)preferableTarget);
553 if (uniqueNets.find(ieNode->net) == uniqueNets.end()) {
554 ld.skip = false;
555 uniqueNets.insert(ieNode->net);
556 }
557 }
558 }
559 CV_Assert(uniqueNets.size() == 1);
560}

569void NetImplOpenVINO::fuseLayers(const std::vector<LayerPin>& blobsToKeep_)
570{
571 CV_TRACE_FUNCTION();
572
573 if(!fusion)
574 return;
575
576 CV_Check((int)preferableBackend, preferableBackend == DNN_BACKEND_INFERENCE_ENGINE_NGRAPH, "");
577
578#if 0 // FIXIT mode without fusion is broken due to unsupported layers and handling of "custom" nodes
579 return;
580#endif
581
582 // scan through all the layers. If there is convolution layer followed by the activation layer,
583 // we try to embed this activation into the convolution and disable separate execution of the activation
584
585 // FIXIT replace by layersToKeep to avoid hacks like "LayerPin(lid, 0)"
586 std::set<LayerPin> pinsToKeep(blobsToKeep_.begin(),
587 blobsToKeep_.end());
588 for (MapIdToLayerData::const_iterator it = layers.begin(); it != layers.end(); it++)
589 {
590 int lid = it->first;
591 LayerData& ld = layers[lid];
592 if (ld.skip)
593 {
594 printf_(("skipped %s: %s\n", ld.layerInstance->name.c_str(), ld.layerInstance->type.c_str()));
595 continue;
596 }
597 printf_(("analyzing %s: %s\n", ld.layerInstance->name.c_str(), ld.layerInstance->type.c_str()));
598
599 // the optimization #1. try to fuse batch norm, scaling and/or activation layers
600 // with the current layer if they follow it. Normally, the are fused with the convolution layer,
601 // but some of them (like activation) may be fused with fully-connected, elemwise (+) and
602 // some other layers.
603 Ptr<Layer>& currLayer = ld.layerInstance;
604 if (ld.consumers.size() == 1 && pinsToKeep.count(LayerPin(lid, 0)) == 0)
605 {
606 LayerData* nextData = &layers[ld.consumers[0].lid];
607 LayerPin lpNext(ld.consumers[0].lid, 0);
608 while (nextData)
609 {
610 if (preferableBackend == DNN_BACKEND_INFERENCE_ENGINE_NGRAPH && pinsToKeep.count(lpNext) != 0)
611 {
612 CV_LOG_DEBUG(NULL, "DNN/IE: skip fusing with 'output' node: " << nextData->name << "@" << nextData->type);
613 break;
614 }
615
616 /* we use `tryFuse` member of convolution layer to fuse eltwise later
617 * it's not intended to be fused here; hence, we stop when we encounter eltwise
618 */
619 Ptr<Layer> nextLayer = nextData->layerInstance;
620 if (currLayer->tryFuse(nextLayer))
621 {
622 printf_(("\tfused with %s\n", nextLayer->name.c_str()));
623 nextData->skip = true;
624 ld.outputBlobs = layers[lpNext.lid].outputBlobs;
625 ld.outputBlobsWrappers = layers[lpNext.lid].outputBlobsWrappers;
626 if (nextData->consumers.size() == 1)
627 {
628 int nextLayerId = nextData->consumers[0].lid;
629 nextData = &layers[nextLayerId];
630 lpNext = LayerPin(nextLayerId, 0);
631 }
632 else
633 {
634 nextData = 0;
635 break;
636 }
637 }
638 else
639 break;
640 }
641 }
642 }

658/*static*/
659Net NetImplOpenVINO::createNetworkFromModelOptimizer(std::shared_ptr<ov::Model>& ieNet)
660{
661 CV_TRACE_FUNCTION();
662
663 CV_TRACE_REGION("register_inputs");
664
665 std::vector<String> inputsNames;
666 std::vector<MatShape> inp_shapes;
667 for (auto& it : ieNet->get_parameters())
668 {
669 inputsNames.push_back(it->get_friendly_name());
670 std::vector<size_t> dims = it->get_shape();
671 inp_shapes.push_back(MatShape(dims.begin(), dims.end()));
672 }
673 // nGraph models produce output "Result" layers which have "/sink_port" suffix in their names.
674 // Their inputs are actual model outputs and we change friendly name to it.
675 // By this workaround, we produce similar outputs names comparing to ieNet.getOutputsInfo()
676 for (int i = 0; i < ieNet->get_output_size(); ++i) {
677 auto res = ieNet->output(i);
678 const std::string& name = res.get_any_name();
679 if (res.get_node()->get_friendly_name() != name)
680 res.get_node()->set_friendly_name(name);
681 }
682
683 Net cvNet;
684 Ptr<NetImplOpenVINO> openvino_impl_ptr = makePtr<NetImplOpenVINO>();
685 NetImplOpenVINO& openvino_impl = *openvino_impl_ptr;
686 accessor::DnnNetAccessor::getImplPtrRef(cvNet) = openvino_impl_ptr;
687
688 cvNet.setInputsNames(inputsNames);
689
690 // set empty input to determine input shapes
691 for (int inp_id = 0; inp_id < inputsNames.size(); ++inp_id)
692 {
693 cvNet.setInputShape(inputsNames[inp_id], inp_shapes[inp_id]);
694 }
695
696 CV_TRACE_REGION_NEXT("backendNode");
697
698 Ptr<BackendNode> backendNode;
699 {
700 auto fake_node = std::make_shared<ov::op::v0::Parameter>(ov::element::f32, ov::Shape {});
701 Ptr<InfEngineNgraphNode> backendNodeNGraph(new InfEngineNgraphNode(fake_node));
702 backendNodeNGraph->net = Ptr<InfEngineNgraphNet>(new InfEngineNgraphNet(openvino_impl, ieNet));
703 backendNode = backendNodeNGraph;
704 }
705
706 CV_TRACE_REGION_NEXT("register_outputs");
707
708 std::vector<std::shared_ptr<ov::Node>> ngraphOperations = ieNet->get_ops();
709
710 for (auto& it : ieNet->get_results())
711 {
712 CV_TRACE_REGION("output");
713 const auto& outputName = it->get_friendly_name();
714
715 LayerParams lp;
716 int lid = cvNet.addLayer(outputName, "", lp);
717
718 LayerData& ld = openvino_impl.layers[lid];
719
720 {
721 Ptr<Layer> cvLayer(new NgraphBackendLayer(ieNet));
722 cvLayer->name = outputName;
723 cvLayer->type = "_unknown_";
724
725 auto process_layer = [&](const std::string& name) -> bool
726 {
727 CV_TRACE_REGION("ngraph_function");
728 for (const auto& op : ngraphOperations)
729 {
730 CV_Assert(op);
731 if (op->get_friendly_name() == name)
732 {
733 const std::string typeName = op->get_type_info().name;
734 cvLayer->type = typeName;
735 return true;
736 }
737 }
738 return false;
739 };
740
741 bool found = process_layer(outputName);
742 if (!found)
743 {
744 auto pos = outputName.rfind('.'); // cut port number: ".0"
745 if (pos != std::string::npos)
746 {
747 std::string layerName = outputName.substr(0, pos);
748 found = process_layer(layerName);
749 }
750 }
751 if (!found)
752 CV_LOG_WARNING(NULL, "DNN/IE: Can't determine output layer type: '" << outputName << "'");
753
754 ld.layerInstance = cvLayer;
755 ld.backendNodes[DNN_BACKEND_INFERENCE_ENGINE_NGRAPH] = backendNode;
756 }
757
758 for (int i = 0; i < inputsNames.size(); ++i)
759 cvNet.connect(0, i, lid, i);
760 }
761
762 CV_TRACE_REGION_NEXT("finalize");
763
764 cvNet.setPreferableBackend(DNN_BACKEND_INFERENCE_ENGINE_NGRAPH);
765
766 return cvNet;
767}
