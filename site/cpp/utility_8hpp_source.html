724template<typename _Tp, typename Functor> inline
725void Mat::forEach_impl(const Functor& operation) {
726 if (false) {
727 operation(*reinterpret_cast<_Tp*>(0), reinterpret_cast<int*>(0));
728 // If your compiler fails in this line.
729 // Please check that your functor signature is
730 // (_Tp&, const int*) <- multi-dimensional
731 // or (_Tp&, void*) <- in case you don't need current idx.
732 }
733
734 CV_Assert(!empty());
735 CV_Assert(this->total() / this->size[this->dims - 1] <= INT_MAX);
736 const int LINES = static_cast<int>(this->total() / this->size[this->dims - 1]);
737
738 class PixelOperationWrapper :public ParallelLoopBody
739 {
740 public:
741 PixelOperationWrapper(Mat_<_Tp>* const frame, const Functor& _operation)
742 : mat(frame), op(_operation) {}
743 virtual ~PixelOperationWrapper(){}
744 // ! Overloaded virtual operator
745 // convert range call to row call.
746 virtual void operator()(const Range &range) const CV_OVERRIDE
747 {
748 const int DIMS = mat->dims;
749 const int COLS = mat->size[DIMS - 1];
750 if (DIMS <= 2) {
751 for (int row = range.start; row < range.end; ++row) {
752 this->rowCall2(row, COLS);
753 }
754 } else {
755 std::vector<int> idx(DIMS);
756 idx[DIMS - 2] = range.start - 1;
757
758 for (int line_num = range.start; line_num < range.end; ++line_num) {
759 idx[DIMS - 2]++;
760 for (int i = DIMS - 2; i >= 0; --i) {
761 if (idx[i] >= mat->size[i]) {
762 idx[i - 1] += idx[i] / mat->size[i];
763 idx[i] %= mat->size[i];
764 continue; // carry-over;
765 }
766 else {
767 break;
768 }
769 }
770 this->rowCall(&idx[0], COLS, DIMS);
771 }
772 }
773 }
774 private:
775 Mat_<_Tp>* const mat;
776 const Functor op;
777 // ! Call operator for each elements in this row.
778 inline void rowCall(int* const idx, const int COLS, const int DIMS) const {
779 int &col = idx[DIMS - 1];
780 col = 0;
781 _Tp* pixel = &(mat->template at<_Tp>(idx));
782
783 while (col < COLS) {
784 op(*pixel, const_cast<const int*>(idx));
785 pixel++; col++;
786 }
787 col = 0;
788 }
789 // ! Call operator for each elements in this row. 2d mat special version.
790 inline void rowCall2(const int row, const int COLS) const {
791 union Index{
792 int body[2];
793 operator const int*() const {
794 return reinterpret_cast<const int*>(this);
795 }
796 int& operator[](const int i) {
797 return body[i];
798 }
799 } idx = {{row, 0}};
800 // Special union is needed to avoid
801 // "error: array subscript is above array bounds [-Werror=array-bounds]"
802 // when call the functor `op` such that access idx[3].
803
804 _Tp* pixel = &(mat->template at<_Tp>(idx));
805 const _Tp* const pixel_end = pixel + COLS;
806 while(pixel < pixel_end) {
807 op(*pixel++, static_cast<const int*>(idx));
808 idx[1]++;
809 }
810 }
811 PixelOperationWrapper& operator=(const PixelOperationWrapper &) {
812 CV_Assert(false);
813 // We can not remove this implementation because Visual Studio warning C4822.
814 return *this;
815 }
816 };
817
818 parallel_for_(cv::Range(0, LINES), PixelOperationWrapper(reinterpret_cast<Mat_<_Tp>*>(this), operation));
819}

1150template<typename _Tp, size_t fixed_size> inline void
1151AutoBuffer<_Tp, fixed_size>::reserve(size_t _cap)
1152{
1153 if( _cap <= cap ) // never shrink; _cap > cap implies _cap > fixed_size, so always heap
1154 return;
1155 _Tp* prevptr = ptr;
1156 ptr = new _Tp[_cap];
1157 // only the LIVE elements [0, sz) are copied - for trivial types the inline buf tail beyond sz
1158 // is intentionally raw (AutoBuffer is a scratch buffer), which gcc's -Wmaybe-uninitialized
1159 // cannot prove when it inlines a grow-from-inline-storage call chain; the annotation below
1160 // documents exactly that, it does not change behavior
1161#if defined(__GNUC__) && !defined(__clang__)
1162#pragma GCC diagnostic push
1163#pragma GCC diagnostic ignored "-Wmaybe-uninitialized"
1164#endif
1165 for( size_t i = 0; i < sz; i++ ) // preserve the live elements
1166 ptr[i] = prevptr[i];
1167#if defined(__GNUC__) && !defined(__clang__)
1168#pragma GCC diagnostic pop
1169#endif
1170 if( prevptr != buf )
1171 delete[] prevptr;
1172 cap = _cap;
1173}

1175template<typename _Tp, size_t fixed_size> inline void
1176AutoBuffer<_Tp, fixed_size>::resize(size_t _size)
1177{
1178 if(_size <= sz) // shrink: keep the capacity and the surviving content
1179 {
1180 sz = _size;
1181 return;
1182 }
1183 if(_size > cap) // grow with geometric slack (like push_back) so incremental
1184 reserve(cap + cap/2 > _size ? cap + cap/2 : _size); // resize(size()+delta) loops don't realloc every step
1185 sz = _size;
1186 // !!! DO NOT ADD ANY INITIALIZATION OF THE NEW TAIL HERE (e.g. `for(i=sz..) ptr[i]=_Tp();`) !!!
1187 // AutoBuffer IS A RAW SCRATCH BUFFER. Value-initializing the tail zero-fills it on EVERY grow, which
1188 // silently dominates the cost of small allocations (measured: ~1.2us per few-KB resize) and there is
1189 // NOTHING to init anyway - callers write before they read. Class-type elements are already
1190 // constructed by `new _Tp[]` / the inline array. If you truly need filled slots, call the two-arg
1191 // overload resize(size, value) EXPLICITLY.
1192}
1193
1194template<typename _Tp, size_t fixed_size> inline void
1195AutoBuffer<_Tp, fixed_size>::resize(size_t _size, const _Tp& value)
1196{
1197 const size_t old = sz;
1198 resize(_size);
1199 for( size_t i = old; i < _size; i++ ) // fill every newly exposed slot
1200 ptr[i] = value;
1201}
