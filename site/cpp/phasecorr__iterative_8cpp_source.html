7void calculateCrossPowerSpectrum(const cv::Mat& dft1, const cv::Mat& dft2, cv::Mat& cps)
8{
9 for (int row = 0; row < dft1.rows; ++row)
10 {
11 auto* cpsp = cps.ptr<cv::Vec<T, 2>>(row);
12 const auto* dft1p = dft1.ptr<cv::Vec<T, 2>>(row);
13 const auto* dft2p = dft2.ptr<cv::Vec<T, 2>>(row);
14 for (int col = 0; col < dft1.cols; ++col)
15 {
16 const T re = dft1p[col][0] * dft2p[col][0] + dft1p[col][1] * dft2p[col][1];
17 const T im = dft1p[col][0] * dft2p[col][1] - dft1p[col][1] * dft2p[col][0];
18 const T mag = std::sqrt(re * re + im * im);
19 cpsp[col][0] = re / mag;
20 cpsp[col][1] = im / mag;
21 }
22 }
23}

105 cv::phaseCorrelateIterative(InputArray _src1, InputArray _src2, int L2size, int maxIters)
106{
107 CV_INSTRUMENT_REGION();
108
109 Mat src1 = _src1.getMat();
110 Mat src2 = _src2.getMat();
111
112 CV_Assert(src1.type() == src2.type());
113 CV_Assert(src1.size() == src2.size());
114 CV_Assert(src1.type() == CV_32FC1 || src1.type() == CV_64FC1);
115
116 // apply DFT window to input images
117 Mat window;
118 createHanningWindow(window, src1.size(), _src1.type());
119 Mat image1, image2;
120 multiply(_src1, window, image1);
121 multiply(_src2, window, image2);
122
123 // compute the DFTs of input images
124 dft(image1, image1, DFT_COMPLEX_OUTPUT);
125 dft(image2, image2, DFT_COMPLEX_OUTPUT);
126
127 // compute the phase correlation landscape L3
128 Mat L3 = calculateCrossPowerSpectrum(image1, image2);
129 dft(L3, L3, DFT_INVERSE | DFT_SCALE | DFT_REAL_OUTPUT);
130 fftshift(L3);
131 Point2d L3mid(L3.cols / 2, L3.rows / 2);
132
133 // calculate the maximum correlation location
134 Point2i L3peak;
135 minMaxLoc(L3, nullptr, nullptr, nullptr, &L3peak);
136
137 // reduce the L2size as long as L2 is out of bounds of L3
138 while (isOutOfBounds(L3peak, L3, L2size))
139 if (!reduceL2size(L2size))
140 return Point2d(L3peak) - L3mid;
141
142 // extract the L2 maximum correlation neighborhood from L3
143 Mat L2 = L3(Rect(L3peak.x - L2size / 2, L3peak.y - L2size / 2, L2size, L2size));
144
145 // upsample L2 maximum correlation neighborhood to get L2U
146 Mat L2U;
147 const int L2Usize = 223; // empirically determined optimal constant
148 resize(L2, L2U, {L2Usize, L2Usize}, 0, 0, INTER_LINEAR);
149 const Point2d L2Umid(L2U.cols / 2, L2U.rows / 2);
150
151 // run the iterative refinement algorithm using the specified L1 ratio
152 // gradually decrease L1 ratio if convergence is not achieved
153 const double L1ratioBase = 0.45; // empirically determined optimal constant
154 const double L1ratioStep = 0.025;
155 for (double L1ratio = L1ratioBase; getL1size(L2U.cols, L1ratio) > 0; L1ratio -= L1ratioStep)
156 {
157 Point2d L2Upeak = L2Umid; // reset the accumulated L2U peak position
158 const int L1size = getL1size(L2U.cols, L1ratio); // calculate the current L1 size
159 const Point2d L1mid(L1size / 2, L1size / 2); // update the L1 mid position
160
161 // perform the iterative refinement algorithm
162 for (int iter = 0; iter < maxIters; ++iter)
163 {
164 // verify that the L1 region is within the L2U region
165 if (isOutOfBounds(L2Upeak, L2U, L1size))
166 break;
167
168 // extract the L1 region from L2U
169 const Mat L1 = L2U(Rect(static_cast<int>(L2Upeak.x - L1size / 2),
170 static_cast<int>(L2Upeak.y - L1size / 2),
171 L1size,
172 L1size));
173
174 // calculate the centroid location
175 const Point2d L1peak = getPeakSubpixel(L1);
176
177 // add the contribution of the current iteration to the accumulated L2U peak location
178 L2Upeak += Point2d(std::round(L1peak.x - L1mid.x), std::round(L1peak.y - L1mid.y));
179
180 // check for convergence
181 if (accuracyReached(L1peak, L1mid))
182 // return the refined subpixel image shift
183 return Point2d(L3peak) - L3mid +
184 (L2Upeak - L2Umid + L1peak - L1mid) /
185 (static_cast<double>(L2Usize) / L2size);
186 }
187 }
188
189 // iterative refinement failed to converge, return non-iterative subpixel shift
190 return getSubpixelShift(L3, Point2d(L3peak), L3mid, L2size);
191}
