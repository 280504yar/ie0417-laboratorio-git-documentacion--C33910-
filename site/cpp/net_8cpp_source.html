518void Net::reserveKVCache(int maxSequenceLength)
519{
520 CV_Assert(impl);
521 CV_CheckTrue(impl->useKVCache && impl->kvCacheManager.isInitialized,
522 "reserveKVCache() requires enableKVCache() to be called first");
523 CV_CheckGE(maxSequenceLength, 0, "maxSequenceLength must be non-negative");
524 CV_CheckLE(maxSequenceLength, KV_CACHE_MAX_RESERVED_TOKENS,
525 "maxSequenceLength is unreasonably large");
526 if (impl->kvCacheManager.empty())
527 {
528 CV_LOG_WARNING(NULL, "DNN: reserveKVCache() has no effect, the model has no paged attention "
529 "layers (no Attention/MultiHeadAttention/GroupQueryAttention op was imported). "
530 "Only the present.* -> past_key_values.* routing is active.");
531 return;
532 }
533 impl->kvCacheManager.reserve(maxSequenceLength);
534}
