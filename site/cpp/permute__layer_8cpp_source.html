85 PermuteLayerImpl(const LayerParams &params)
86 : _count(0), _needsPermute(false), _numAxes(0)
87 {
88 if (!params.has("order"))
89 {
90 return;
91 }
92
93 DictValue paramOrder = params.get("order");
94 _numAxes = paramOrder.size();
95
96 for (size_t i = 0; i < _numAxes; i++)
97 {
98 int currentOrder = paramOrder.get<int>(i);
99 if (currentOrder < 0 || currentOrder > _numAxes)
100 {
101 CV_Error(Error::StsBadArg,
102 format("Orders of dimensions in Permute layer parameter"
103 "must be in [0...%zu]", _numAxes - 1));
104 }
105 if (std::find(_order.begin(), _order.end(), currentOrder) != _order.end())
106 {
107 CV_Error(Error::StsBadArg,
108 "Permute layer parameter contains duplicated orders.");
109 }
110 _order.push_back(currentOrder);
111 }
112
113 zeropoint = params.get<int>("zeropoints", 0);
114 scale = params.get<float>("scales", 1.0f);
115
116 setParamsFrom(params);
117 checkNeedForPermutation();
118 }

201 void computeStrides(const MatShape &shapeBefore, const MatShape &shapeAfter)
202 {
203 _oldStride.resize(_numAxes);
204 _newStride.resize(_numAxes);
205
206 _oldStride[_numAxes - 1] = 1;
207 _newStride[_numAxes - 1] = 1;
208
209 for(int i = _numAxes - 2; i >= 0; i--)
210 {
211 _oldStride[i] = _oldStride[i + 1] * shapeBefore[i + 1];
212 _newStride[i] = _newStride[i + 1] * shapeAfter[i + 1];
213 }
214
215 _count = _oldStride[0] * shapeBefore[0];
216 }

241 template <class T>
242 class PermuteInvoker : public ParallelLoopBody
243 {
244 public:
245 const Mat* inp;
246 Mat* out;
247 const std::vector<size_t>* order;
248 int nstripes;
249
250 static void run(const Mat& inp, Mat& out, const std::vector<size_t>& order, int nstripes)
251 {
252 PermuteInvoker p;
253 p.inp = &inp;
254 p.out = &out;
255 p.order = &order;
256 p.nstripes = nstripes;
257
258 CV_Assert( out.size[0] == inp.size[order[0]] &&
259 out.size[1] == inp.size[order[1]] &&
260 out.size[2] == inp.size[order[2]] &&
261 out.size[3] == inp.size[order[3]]);
262
263 parallel_for_(Range(0, nstripes), p, nstripes);
264 }
265
266 PermuteInvoker() : inp(0), out(0), order(0), nstripes(0) {}
267
268 void operator()(const Range& r) const CV_OVERRIDE
269 {
270 int n0 = out->size[0], n1 = out->size[1], n2 = out->size[2], n3 = out->size[3];
271
272 size_t orows = (size_t)n0*n1*n2;
273 size_t stripeSize = (orows + nstripes - 1)/nstripes;
274 size_t stripeStart = r.start*stripeSize;
275 size_t stripeEnd = std::min(r.end*stripeSize, orows);
276
277 const size_t esz = sizeof(T);
278 size_t ostep0 = out->step[0]/esz, ostep1 = out->step[1]/esz, ostep2 = out->step[2]/esz;
279 const size_t* ord = &order->at(0);
280 size_t istep0 = inp->step[ord[0]]/esz, istep1 = inp->step[ord[1]]/esz,
281 istep2 = inp->step[ord[2]]/esz, istep3 = inp->step[ord[3]]/esz;
282
283 size_t val = stripeStart;
284 int i2 = (int)(val % n2);
285 val /= n2;
286 int i1 = (int)(val % n1);
287 int i0 = (int)(val / n1);
288
289 const T* inptr_orig = inp->ptr<T>();
290 T* outptr_orig = out->ptr<T>();
291
292 for( size_t ofs = stripeStart; ofs < stripeEnd; ofs++ )
293 {
294 const T* inptr = inptr_orig + i0*istep0 + i1*istep1 + i2*istep2;
295 T* outptr = outptr_orig + i0*ostep0 + i1*ostep1 + i2*ostep2;
296
297 for( int i3 = 0; i3 < n3; i3++ )
298 outptr[i3] = inptr[i3*istep3];
299
300 if( ++i2 >= n2 )
301 {
302 i2 = 0;
303 if( ++i1 >= n1 )
304 {
305 i1 = 0;
306 if( ++i0 >= n0 )
307 break;
308 }
309 }
310 }
311 }
312 };

315 bool forward_ocl(InputArrayOfArrays inps, OutputArrayOfArrays outs, OutputArrayOfArrays internals)
316 {
317 std::vector<UMat> inputs;
318 std::vector<UMat> outputs;
319
320 inps.getUMatVector(inputs);
321 outs.getUMatVector(outputs);
322
323 if (!_needsPermute)
324 return false;
325
326 if (uorder.empty())
327 {
328 std::vector<int> orderVec(_order.begin(), _order.end());;
329 Mat morder(1, orderVec.size(), CV_32SC1, &orderVec[0]);
330
331 std::vector<int> oldStrideVec(_oldStride.begin(), _oldStride.end());
332 Mat mold_stride(1, _oldStride.size(), CV_32SC1, &oldStrideVec[0]);
333
334 std::vector<int> newStrideVec(_newStride.begin(), _newStride.end());
335 Mat mnew_stride(1, newStrideVec.size(), CV_32SC1, &newStrideVec[0]);
336
337 morder.copyTo(uorder);
338 mold_stride.copyTo(uold_stride);
339 mnew_stride.copyTo(unew_stride);
340 }
341
342 for (size_t i = 0; i < inputs.size(); i++)
343 {
344 String matType = matTypeToOclType(inputs[0].type());
345 String opts = " -DDtype=" + matType;
346 String kname = "permute_" + matType;
347
348 ocl::Kernel kernel(kname.c_str(), ocl::dnn::permute_oclsrc, opts);
349
350 kernel.set(0, (int)_count);
351 kernel.set(1, ocl::KernelArg::PtrReadOnly(inputs[i]));
352 kernel.set(2, ocl::KernelArg::PtrReadOnly(uorder));
353 kernel.set(3, ocl::KernelArg::PtrReadOnly(uold_stride));
354 kernel.set(4, ocl::KernelArg::PtrReadOnly(unew_stride));
355 kernel.set(5, (int)_numAxes);
356 kernel.set(6, ocl::KernelArg::PtrWriteOnly(outputs[i]));
357
358 if (!kernel.run(1, &_count, NULL, false))
359 return false;
360 }
361
362 return true;
363 }

366 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
367 {
368 CV_TRACE_FUNCTION();
369 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
370
371 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
372 forward_ocl(inputs_arr, outputs_arr, internals_arr))
373
374 std::vector<Mat> inputs, outputs;
375 inputs_arr.getMatVector(inputs);
376 outputs_arr.getMatVector(outputs);
377
378 size_t k, ninputs = inputs.size();
379 if(!_needsPermute)
380 {
381 for (k = 0; k < ninputs; k++)
382 {
383 CV_Assert(outputs[k].total() == inputs[k].total());
384 if (outputs[k].data != inputs[k].data)
385 inputs[k].copyTo(outputs[k]);
386 }
387 }
388 else
389 {
390 for (k = 0; k < ninputs; k++)
391 {
392 CV_Assert(inputs[k].dims == _numAxes && inputs[k].size == inputs[0].size);
393 CV_Assert(outputs[k].dims == _numAxes && outputs[k].size == outputs[0].size);
394
395 switch (inputs[k].depth())
396 {
397 case CV_32F:
398 forward_impl<float>(inputs[k], outputs[k]);
399 break;
400 case CV_16F:
401 forward_impl<int16_t>(inputs[k], outputs[k]);
402 break;
403 case CV_32S:
404 forward_impl<int32_t>(inputs[k], outputs[k]);
405 break;
406 case CV_64S:
407 forward_impl<int64_t>(inputs[k], outputs[k]);
408 break;
409 case CV_8S:
410 forward_impl<int8_t>(inputs[k], outputs[k]);
411 break;
412 case CV_8U:
413 forward_impl<uint8_t>(inputs[k], outputs[k]);
414 break;
415 case CV_Bool:
416 forward_impl<bool>(inputs[k], outputs[k]);
417 break;
418 default:
419 CV_Error(Error::BadDepth, "unsupported mat type");
420 }
421 }
422 }

425 template <class T>
426 void forward_impl(const Mat& inp, Mat& out)
427 {
428 const size_t* newStride = &_newStride[0];
429 const size_t* oldStride = &_oldStride[0];
430 const size_t* order = &_order[0];
431
432 CV_Assert(inp.isContinuous() && out.isContinuous());
433
434 if( _numAxes == 4 )
435 {
436 int nstripes = getNumThreads();
437 PermuteInvoker<T>::run(inp, out, _order, nstripes);
438 }
439 else
440 {
441 const T *srcData = inp.ptr<T>();
442 T *dstData = out.ptr<T>();
443
444 for (size_t i = 0; i < _count; ++i)
445 {
446 size_t oldPosition = 0;
447 size_t newPosition = i;
448
449 for (size_t j = 0; j < _numAxes; ++j)
450 {
451 oldPosition += (newPosition / newStride[j]) * oldStride[order[j]];
452 newPosition %= newStride[j];
453 }
454 dstData[i] = srcData[oldPosition];
455 }
456 }
457 }

610 // convert OpenCV NCHW order to WHCN order.
611 bool getOrderWHCN(std::vector<uint32_t>& orderWHCN)
612 {
613 std::map<int, int> lookup;
614 int orderLen = _order.size();
615 if (orderLen <2)
616 return false;
617 orderWHCN.assign(_order.begin(), _order.end());
618
619 if (orderLen == 2)
620 {
621 return true;
622 }
623 else if (orderLen >= 3)
624 {
625 for (int i = 0; i < orderLen; i++)
626 {
627 lookup[i] = orderLen - i - 1;
628 }
629
630 for (int i = 0; i < orderLen; i++)
631 {
632 orderWHCN[i] = lookup[_order[i]];
633 }
634 std::reverse(orderWHCN.begin(), orderWHCN.end());
635 return true;
636 }
637 else
638 return false;
639 }
