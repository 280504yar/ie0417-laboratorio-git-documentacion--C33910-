16static void maxPoolImpl(const void* inp_, void* out_, const ConvState& cs)
17{
18 constexpr int MAX_POOL_DIMS = ConvState::MAX_CONV_DIMS;
19 int NC1 = cs.inpshape[0] * cs.inpshape[1];
20
21 CV_Assert(cs.inpshape.layout == DATA_LAYOUT_BLOCK);
22 CV_Assert(cs.outshape.layout == DATA_LAYOUT_BLOCK);
23 CV_Assert(cs.inpshape.dims == cs.outshape.dims);
24
25 parallel_for_(Range(0, NC1), [&](const Range& r) {
26 CV_Assert(cs.nspatialdims <= MAX_POOL_DIMS && MAX_POOL_DIMS == 3);
27
28 int sdims = cs.nspatialdims;
29 int nc0 = r.start, nc1 = r.end;
30 int C0 = cs.inpshape.back();
31 int Di = sdims > 2 ? cs.inpshape[sdims - 1] : 1;
32 int Hi = sdims > 1 ? cs.inpshape[sdims] : 1;
33 int Wi = cs.inpshape[sdims + 1];
34 int D = sdims > 2 ? cs.outshape[sdims - 1] : 1;
35 int H = sdims > 1 ? cs.outshape[sdims] : 1;
36 int W = cs.outshape[sdims + 1];
37 int iplanesize = Di * Hi * Wi * C0;
38 int planesize = D * H * W * C0;
39 int SZ = cs.strides[0], SY = cs.strides[1], SX = cs.strides[2];
40 int padZ0 = cs.pads[0], padY0 = cs.pads[1], padX0 = cs.pads[2];
41 int inner_z0 = cs.inner[0], inner_z1 = cs.inner[MAX_POOL_DIMS];
42 int inner_y0 = cs.inner[1], inner_y1 = cs.inner[MAX_POOL_DIMS + 1];
43 int inner_x0 = cs.inner[2], inner_x1 = cs.inner[MAX_POOL_DIMS + 2];
44 int ksize = (int)cs.ofstab.size();
45 const int* zyxtab = cs.coordtab.data();
46 const int* ofstab = cs.ofstab.data();
47
48 const T* inp = (const T*)inp_ + nc0 * iplanesize;
49 T* out = (T*)out_ + nc0 * planesize;
50 const T INITVAL = std::numeric_limits<T>::min();
51
52 for (int nc = nc0; nc < nc1; nc++, inp += iplanesize) {
53 for (int z0 = 0; z0 < D; z0++) {
54 int zi_ = z0 * SZ - padZ0;
55 for (int y0 = 0; y0 < H; y0++, out += W * C0) {
56 int x0 = 0;
57 int x1 = z0 >= inner_z0 && z0 < inner_z1 &&
58 y0 >= inner_y0 && y0 < inner_y1 ? inner_x0 : W;
59 int yi_ = y0 * SY - padY0;
60
61 for (;;) {
62 for (; x0 < x1; x0++) {
63 int xi_ = x0 * SX - padX0;
64 for (int c = 0; c < C0; c++)
65 out[x0 * C0 + c] = INITVAL;
66
67 for (int k = 0; k < ksize; k++) {
68 int zi = zi_ + zyxtab[k * MAX_POOL_DIMS];
69 int yi = yi_ + zyxtab[k * MAX_POOL_DIMS + 1];
70 int xi = xi_ + zyxtab[k * MAX_POOL_DIMS + 2];
71 if ((unsigned)zi >= (unsigned)Di ||
72 (unsigned)yi >= (unsigned)Hi ||
73 (unsigned)xi >= (unsigned)Wi)
74 continue;
75 const T* inptr = inp + ((zi * Hi + yi) * Wi + xi) * C0;
76 for (int c = 0; c < C0; c++)
77 out[x0 * C0 + c] = std::max(out[x0 * C0 + c], inptr[c]);
78 }
79 }
80
81 if (x0 == W)
82 break;
83 x1 = inner_x1;
84
85 for (; x0 < x1; x0++) {
86 int xi_ = x0 * SX - padX0;
87 const T* inp_xi = inp + ((Hi * zi_ + yi_) * Wi + xi_) * C0;
88
89 for (int c = 0; c < C0; c++) {
90 T s = inp_xi[ofstab[0] + c];
91 for (int k = 1; k < ksize; k++)
92 s = std::max(s, inp_xi[ofstab[k] + c]);
93 out[x0 * C0 + c] = s;
94 }
95 }
96
97 x1 = W;
98 }
99 }
100 }
101 }
102 });
103}

113static void avgPoolInt8(const void* inp_, void* out_, const ConvState& cs,
114 float inp_sc, int inp_zp, float out_sc, int out_zp,
115 bool count_include_pad_, bool isU8)
116{
117 constexpr int MAX_POOL_DIMS = ConvState::MAX_CONV_DIMS;
118 int NC1 = cs.inpshape[0] * cs.inpshape[1];
119
120 CV_Assert(cs.inpshape.layout == DATA_LAYOUT_BLOCK);
121 CV_Assert(cs.outshape.layout == DATA_LAYOUT_BLOCK);
122 CV_Assert(cs.inpshape.dims == cs.outshape.dims);
123
124 parallel_for_(Range(0, NC1), [&](const Range& r) {
125 CV_Assert(cs.nspatialdims <= MAX_POOL_DIMS && MAX_POOL_DIMS == 3);
126
127 bool count_include_pad = count_include_pad_;
128 int sdims = cs.nspatialdims;
129 int nc0 = r.start, nc1 = r.end;
130 int C0 = cs.inpshape.back();
131 int Di = sdims > 2 ? cs.inpshape[sdims - 1] : 1;
132 int Hi = sdims > 1 ? cs.inpshape[sdims] : 1;
133 int Wi = cs.inpshape[sdims + 1];
134 int D = sdims > 2 ? cs.outshape[sdims - 1] : 1;
135 int H = sdims > 1 ? cs.outshape[sdims] : 1;
136 int W = cs.outshape[sdims + 1];
137 int iplanesize = Di * Hi * Wi * C0;
138 int planesize = D * H * W * C0;
139 int SZ = cs.strides[0], SY = cs.strides[1], SX = cs.strides[2];
140 int padZ0 = cs.pads[0], padY0 = cs.pads[1], padX0 = cs.pads[2];
141 int ksize_total = (int)cs.ofstab.size();
142 const int* zyxtab = cs.coordtab.data();
143
144 const uint8_t* inp = (const uint8_t*)inp_ + nc0 * iplanesize;
145 uint8_t* out = (uint8_t*)out_ + nc0 * planesize;
146
147 float scale_ratio = inp_sc / out_sc;
148
149 std::vector<int> accum(C0);
150
151 for (int nc = nc0; nc < nc1; nc++, inp += iplanesize) {
152 for (int z0 = 0; z0 < D; z0++) {
153 int zi_ = z0 * SZ - padZ0;
154 for (int y0 = 0; y0 < H; y0++, out += W * C0) {
155 int yi_ = y0 * SY - padY0;
156
157 for (int x0 = 0; x0 < W; x0++) {
158 int xi_ = x0 * SX - padX0;
159
160 for (int c = 0; c < C0; c++)
161 accum[c] = 0;
162
163 int count = 0;
164 for (int k = 0; k < ksize_total; k++) {
165 int zi = zi_ + zyxtab[k * MAX_POOL_DIMS];
166 int yi = yi_ + zyxtab[k * MAX_POOL_DIMS + 1];
167 int xi = xi_ + zyxtab[k * MAX_POOL_DIMS + 2];
168
169 if ((unsigned)zi >= (unsigned)Di ||
170 (unsigned)yi >= (unsigned)Hi ||
171 (unsigned)xi >= (unsigned)Wi) {
172 if (count_include_pad)
173 count++;
174 continue;
175 }
176
177 count++;
178 const uint8_t* inptr = inp + ((zi * Hi + yi) * Wi + xi) * C0;
179 for (int c = 0; c < C0; c++) {
180 int v = isU8 ? (int)inptr[c]
181 : (int)(int8_t)inptr[c];
182 accum[c] += v - inp_zp;
183 }
184 }
185
186 if (count == 0) count = 1;
187 float inv_count = 1.f / count;
188
189 for (int c = 0; c < C0; c++) {
190 float val = (float)accum[c] * inv_count * scale_ratio + (float)out_zp;
191 int ival = cvRound(val);
192 if (isU8)
193 out[x0 * C0 + c] = (uint8_t)std::max(0, std::min(255, ival));
194 else
195 out[x0 * C0 + c] = (uint8_t)(int8_t)std::max(-128, std::min(127, ival));
196 }
197 }
198 }
199 }
200 }
201 });
202}

204static void globalAvgPoolInt8(const void* inp_, void* out_,
205 int N, int C1, int spatialSize, int C0,
206 float inp_sc, int inp_zp, float out_sc, int out_zp,
207 bool isU8)
208{
209 int NC1 = N * C1;
210 float scale_ratio = inp_sc / out_sc;
211
212 parallel_for_(Range(0, NC1), [&](const Range& r) {
213 for (int nc = r.start; nc < r.end; nc++) {
214 const uint8_t* inptr = (const uint8_t*)inp_ + (size_t)nc * spatialSize * C0;
215 uint8_t* outptr = (uint8_t*)out_ + (size_t)nc * C0;
216
217 for (int c = 0; c < C0; c++) {
218 int sum = 0;
219 for (int s = 0; s < spatialSize; s++) {
220 int v = isU8 ? (int)inptr[s * C0 + c]
221 : (int)(int8_t)inptr[s * C0 + c];
222 sum += v - inp_zp;
223 }
224
225 float val = (float)sum / spatialSize * scale_ratio + (float)out_zp;
226 int ival = cvRound(val);
227 if (isU8)
228 outptr[c] = (uint8_t)std::max(0, std::min(255, ival));
229 else
230 outptr[c] = (uint8_t)(int8_t)std::max(-128, std::min(127, ival));
231 }
232 }
233 });
234}

323 void forward(InputArrayOfArrays inputs_arr,
324 OutputArrayOfArrays outputs_arr,
325 OutputArrayOfArrays) CV_OVERRIDE
326 {
327 size_t ninputs = inputs_arr.total();
328 CV_Assert(ninputs == 1);
329
330 const Mat& inp = inputs_arr.getMat(0);
331 int inptype = inp.type();
332 MatShape inpshape = inp.shape();
333 CV_Assert(inpshape.layout == DATA_LAYOUT_BLOCK);
334 CV_Assert(inptype == CV_8SC1 || inptype == CV_8UC1);
335
336 MatShape outshape;
337 if (is_global_pooling) {
338 outshape = inpshape;
339 for (int d = 2; d < outshape.dims - 1; d++)
340 outshape[d] = 1;
341 } else {
342 outshape = convInferShape(inpshape, MatShape(),
343 kernel_shape, 0, strides, dilations,
344 pads, auto_pad, ceil_mode);
345 }
346
347 int outkind = outputs_arr.kind();
348 Mat out;
349
350 if (outkind == _InputArray::STD_VECTOR_MAT) {
351 std::vector<Mat>& outs = outputs_arr.getMatVecRef();
352 outs.resize(1);
353 outs[0].fit(outshape, inptype);
354 out = outs[0];
355 } else {
356 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
357 outs.resize(1);
358 outs[0].fit(outshape, inptype);
359 out.fit(outshape, inptype);
360 }
361
362 if (is_global_pooling) {
363 int N = inpshape[0];
364 int C1 = inpshape[1];
365 int C0 = inpshape.back();
366 int spatialSize = 1;
367 for (int d = 2; d < inpshape.dims - 1; d++)
368 spatialSize *= inpshape[d];
369
370 globalAvgPoolInt8(inp.data, out.data, N, C1, spatialSize, C0,
371 input_sc, input_zp, output_sc, output_zp,
372 inptype == CV_8UC1);
373 } else {
374 ConvState cs_local;
375 cs_local.initPooling(inpshape, outshape, kernel_shape, strides,
376 dilations, pads, auto_pad, ceil_mode);
377
378 if (is_max_pool) {
379 maxPoolInt8(inp.data, out.data, cs_local, inptype == CV_8UC1);
380 } else {
381 avgPoolInt8(inp.data, out.data, cs_local,
382 input_sc, input_zp, output_sc, output_zp,
383 count_include_pad, inptype == CV_8UC1);
384 }
385 }
386
387 if (outkind == _InputArray::STD_VECTOR_UMAT) {
388 std::vector<UMat>& outs = outputs_arr.getUMatVecRef();
389 out.copyTo(outs[0]);
390 }
391 }
