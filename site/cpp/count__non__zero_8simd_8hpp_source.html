17template<typename T>
18static int countNonZero_(const T* src, int len )
19{
20 int nz = 0;
21 for( int i = 0; i < len; i++ )
22 nz += src[i] != 0;
23 return nz;
24}

33// AVX-512 exposes wide per-lane mask registers, so counting zero lanes with
34// v_signmask()+popcount is cheaper than the batched add-based reduction.
35// Narrower SIMD (AVX2, NEON, LASX, ...) keeps the legacy batched kernel to
36// avoid regressions on smaller vector widths.
37#undef CV_COUNTNONZERO_AVX512
38#if defined(CV_CPU_COMPILE_AVX512_SKX) || defined(CV_CPU_COMPILE_AVX512_ICL)
39#define CV_COUNTNONZERO_AVX512 1
40#else
41#define CV_COUNTNONZERO_AVX512 0
42#endif

44#if CV_COUNTNONZERO_AVX512
45static inline int cnz_popcount32(unsigned x)
46{
47#if defined(__GNUC__) || defined(__clang__)
48 return __builtin_popcount(x);
49#else
50 int c = 0;
51 for (; x; x &= x - 1)
52 ++c;
53 return c;
54#endif
55}

57static inline int cnz_popcount64(uint64 x)
58{
59#if defined(__GNUC__) || defined(__clang__)
60 return (int)__builtin_popcountll((unsigned long long)x);
61#else
62 int c = 0;
63 for (; x; x &= x - 1)
64 ++c;
65 return c;
66#endif
67}

73// SIMD body of the countNonZero kernels. The AVX-512 variant counts zero lanes
74// via signmask+popcount; every other target keeps the legacy batched add kernel.
75#undef CNZ_SIMD_BODY
76#if CV_COUNTNONZERO_AVX512
77#define CNZ_SIMD_BODY(suffix, ssuffix, rsuffix, VT, ST, cmp_op, add_op, update_sum) \
78 const int vlanes = VTraits<VT>::vlanes(); \
79 VT v_zero = vx_setzero_##suffix(); \
80 int nzeros = 0; \
81 for (; i <= len - vlanes*2; i += vlanes*2) \
82 { \
83 nzeros += cnz_lane_pop(v_signmask(v_reinterpret_as_##rsuffix(cmp_op(vx_load(src + i), v_zero)))); \
84 nzeros += cnz_lane_pop(v_signmask(v_reinterpret_as_##rsuffix(cmp_op(vx_load(src + i + vlanes), v_zero)))); \
85 } \
86 for (; i <= len - vlanes; i += vlanes) \
87 { \
88 nzeros += cnz_lane_pop(v_signmask(v_reinterpret_as_##rsuffix(cmp_op(vx_load(src + i), v_zero)))); \
89 } \
90 nz += i - nzeros; \
91 v_cleanup();
92#else
93#define CNZ_SIMD_BODY(suffix, ssuffix, rsuffix, VT, ST, cmp_op, add_op, update_sum) \
94 const int vlanes = VTraits<VT>::vlanes(); \
95 VT v_zero = vx_setzero_##suffix(); \
96 VT v_1 = vx_setall_##suffix(1); \
97 VT v_8 = vx_setall_##suffix(8); \
98 ST v_sum0 = vx_setzero_##ssuffix(); \
99 ST v_sum1 = v_sum0; \
100 for (i = 0; i <= len - vlanes*8; i += vlanes*8) \
101 { \
102 VT x0 = vx_load(src + i); \
103 VT x1 = vx_load(src + i + vlanes); \
104 VT x2 = vx_load(src + i + vlanes*2); \
105 VT x3 = vx_load(src + i + vlanes*3); \
106 VT x4 = vx_load(src + i + vlanes*4); \
107 VT x5 = vx_load(src + i + vlanes*5); \
108 VT x6 = vx_load(src + i + vlanes*6); \
109 VT x7 = vx_load(src + i + vlanes*7); \
110 x0 = cmp_op(x0, v_zero); \
111 x1 = cmp_op(x1, v_zero); \
112 x2 = cmp_op(x2, v_zero); \
113 x3 = cmp_op(x3, v_zero); \
114 x4 = cmp_op(x4, v_zero); \
115 x5 = cmp_op(x5, v_zero); \
116 x6 = cmp_op(x6, v_zero); \
117 x7 = cmp_op(x7, v_zero); \
118 x0 = add_op(x0, x1); \
119 x2 = add_op(x2, x3); \
120 x4 = add_op(x4, x5); \
121 x6 = add_op(x6, x7); \
122 x0 = add_op(x0, x2); \
123 x4 = add_op(x4, x6); \
124 x0 = add_op(add_op(x0, x4), v_8); \
125 update_sum(v_sum0, v_sum1, x0); \
126 } \
127 for (; i <= len - vlanes; i += vlanes) \
128 { \
129 VT x0 = vx_load(src + i); \
130 x0 = add_op(cmp_op(x0, v_zero), v_1); \
131 update_sum(v_sum0, v_sum1, x0); \
132 } \
133 nz += (int)v_reduce_sum(v_add(v_sum0, v_sum1)); \
134 v_cleanup();
135#endif

196#if CV_COUNTNONZERO_AVX512 && (CV_SIMD_64F || CV_SIMD_SCALABLE_64F)
197// 64-bit floats: FP compare against zero treats -0.0 as zero, matching the
198// scalar `src[i] != 0` path.
199static int countNonZero64f( const void* src_ptr, int len )
200{
201 const double* src = static_cast<const double*>(src_ptr);
202 int i = 0, nz = 0;
203 const int vlanes = VTraits<v_float64>::vlanes();
204 v_float64 v_zero = vx_setzero_f64();
205 int nzeros = 0;
206 for (; i <= len - vlanes*2; i += vlanes*2)
207 {
208 nzeros += cnz_lane_pop(v_signmask(v_eq(vx_load(src + i), v_zero)));
209 nzeros += cnz_lane_pop(v_signmask(v_eq(vx_load(src + i + vlanes), v_zero)));
210 }
211 for (; i <= len - vlanes; i += vlanes)
212 nzeros += cnz_lane_pop(v_signmask(v_eq(vx_load(src + i), v_zero)));
213 nz += i - nzeros;
214 v_cleanup();
215 for (; i < len; i++)
216 nz += src[i] != 0;
217 return nz;
218}
