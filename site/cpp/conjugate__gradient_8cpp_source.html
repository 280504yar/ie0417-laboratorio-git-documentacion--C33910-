49 double MinProblemSolver::Function::getGradientEps() const { return 1e-3; }
50 void MinProblemSolver::Function::getGradient(const double* x, double* grad)
51 {
52 double eps = getGradientEps();
53 int i, n = getDims();
54 AutoBuffer<double> x_buf(n);
55 double* x_ = x_buf.data();
56 for( i = 0; i < n; i++ )
57 x_[i] = x[i];
58 for( i = 0; i < n; i++ )
59 {
60 x_[i] = x[i] + eps;
61 double y1 = calc(x_);
62 x_[i] = x[i] - eps;
63 double y0 = calc(x_);
64 grad[i] = (y1 - y0)/(2*eps);
65 x_[i] = x[i];
66 }
67 }
