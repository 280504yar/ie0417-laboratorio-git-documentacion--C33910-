26 public:

29 /* offsets is indexed by output number and each subvector is indexed by axis number */
30 SliceOp(csl::Stream stream_, std::vector<std::vector<std::size_t>> offsets)
31 : stream(std::move(stream_)), offsets(std::move(offsets))

33 }
34
35 void forward(
36 const std::vector<cv::Ptr<BackendWrapper>>& inputs,
37 const std::vector<cv::Ptr<BackendWrapper>>& outputs,
38 csl::Workspace& workspace) override
39 {
40 /* sometimes the output shape is passed in the form of a second input tensor
41 * it's only required for initialization and not here
42 */
43 CV_Assert(inputs.size() == 1 || inputs.size() == 2);
44
45 auto input_wrapper = inputs[0].dynamicCast<wrapper_type>();
46 auto input = input_wrapper->getView();
47
48 CV_Assert(offsets.size() == outputs.size());
49
50 for (int i = 0; i < outputs.size(); ++i)
51 {
52 auto output_wrapper = outputs[i].dynamicCast<wrapper_type>();
53 auto output = output_wrapper->getSpan();
54
55 kernels::slice<T>(stream, output, input, offsets[i]);
56 }
57 }
58
59 private:
60 csl::Stream stream;
61 std::vector<std::vector<std::size_t>> offsets;
62 };
