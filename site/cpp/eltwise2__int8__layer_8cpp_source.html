21static Mat broadcastExpand(const Mat& src, const MatShape& dstShape)
22{
23 MatShape srcShape = src.shape();
24 const int ndst = (int)dstShape.size();
25 while ((int)srcShape.size() < ndst)
26 srcShape.insert(srcShape.begin(), 1);
27 if (srcShape == dstShape)
28 return src;
29
30 MatShape outShape = dstShape;
31 Mat out(outShape, src.type());
32 const int total = (int)out.total();
33
34 std::vector<int> dstStrides(ndst);
35 { int s = 1; for (int d = ndst-1; d >= 0; d--) { dstStrides[d] = s; s *= dstShape[d]; } }
36
37 std::vector<int> srcStrides(ndst, 0);
38 { int s = 1; for (int d = ndst-1; d >= 0; d--) {
39 srcStrides[d] = (srcShape[d] > 1) ? s : 0; s *= srcShape[d]; } }
40
41 const uint8_t* sp = src.ptr<uint8_t>();
42 uint8_t* dp = out.ptr<uint8_t>();
43 for (int i = 0; i < total; i++) {
44 int rem = i, srcIdx = 0;
45 for (int d = 0; d < ndst; d++) {
46 int coord = rem / dstStrides[d]; rem %= dstStrides[d];
47 srcIdx += coord * srcStrides[d];
48 }
49 dp[i] = sp[srcIdx];
50 }
51 return out;
52}

115 void ensureCoeffs()
116 {
117 if (!coeffs.empty() || scales.empty())
118 return;
119
120 CV_CheckEQ(scales.size(), zeropoints.size(),
121 "Eltwise2Int8: scales and zeropoints sizes must match");
122 CV_Assert(output_sc > 0.f);
123
124 coeffs.resize(scales.size());
125 if (op == Op::MUL) {
126 CV_Assert(scales.size() == 2);
127 mulCoeff = (scales[0] * scales[1]) / output_sc;
128 coeffs[0] = (float)zeropoints[0];
129 coeffs[1] = (float)zeropoints[1];
130 offset = (float)output_zp;
131 } else {
132 offset = (float)output_zp;
133 for (size_t i = 0; i < scales.size(); i++)
134 {
135 coeffs[i] = scales[i] / output_sc;
136 offset -= coeffs[i] * zeropoints[i];
137 }
138 }

141 bool getMemoryShapes(const std::vector<MatShape>& inputs,
142 const int requiredOutputs,
143 std::vector<MatShape>& outputs,
144 std::vector<MatShape>& internals) const CV_OVERRIDE
145 {
146 CV_Assert(inputs.size() >= 2);
147 int ndims = 0;
148 DataLayout outLayout = DATA_LAYOUT_UNKNOWN;
149 int outC = 0;
150 for (const auto& s : inputs) {
151 ndims = std::max(ndims, (int)s.size());
152 if (s.layout != DATA_LAYOUT_UNKNOWN && outLayout == DATA_LAYOUT_UNKNOWN)
153 outLayout = s.layout;
154 if (s.C > outC) outC = s.C;
155 }
156 MatShape outShape(ndims, 1);
157 for (const auto& s : inputs) {
158 int off = ndims - (int)s.size();
159 for (int d = 0; d < (int)s.size(); d++)
160 outShape[d + off] = std::max(outShape[d + off], s[d]);
161 }
162 outShape.layout = outLayout;
163 outShape.C = outC;
164 outputs.assign(1, outShape);
165 internals.clear();
166 return true;
167 }

506 int32x4_t vmax = vdupq_n_s32(out_max);
507

510 int8x8_t b8 = vld1_s8(p1 + i);
511 int16x8_t a16 = vmovl_s8(a8);
512 int16x8_t b16 = vmovl_s8(b8);
513
514 float32x4_t af_lo = vcvtq_f32_s32(vmovl_s16(vget_low_s16(a16)));
515 float32x4_t bf_lo = vcvtq_f32_s32(vmovl_s16(vget_low_s16(b16)));
516 float32x4_t af_hi = vcvtq_f32_s32(vmovl_s16(vget_high_s16(a16)));
517 float32x4_t bf_hi = vcvtq_f32_s32(vmovl_s16(vget_high_s16(b16)));
518

520 if (isMul) {
521 val_lo = vaddq_f32(vmulq_f32(vmulq_f32(vsubq_f32(af_lo, vzp0), vsubq_f32(bf_lo, vzp1)), vmc), vozp);
522 val_hi = vaddq_f32(vmulq_f32(vmulq_f32(vsubq_f32(af_hi, vzp0), vsubq_f32(bf_hi, vzp1)), vmc), vozp);
523 } else {
524 val_lo = vaddq_f32(vaddq_f32(voff, vmulq_f32(vc0, af_lo)), vmulq_f32(vc1, bf_lo));
525 val_hi = vaddq_f32(vaddq_f32(voff, vmulq_f32(vc0, af_hi)), vmulq_f32(vc1, bf_hi));
526 }
527
528#if CV_NEON_AARCH64
529 int32x4_t ival_lo = vminq_s32(vmaxq_s32(vcvtnq_s32_f32(val_lo), vmin), vmax);
530 int32x4_t ival_hi = vminq_s32(vmaxq_s32(vcvtnq_s32_f32(val_hi), vmin), vmax);
531#else
532 float32x4_t half = vdupq_n_f32(0.5f);
533 int32x4_t ival_lo = vminq_s32(vmaxq_s32(vcvtq_s32_f32(vaddq_f32(val_lo,
534 vbslq_f32(vcgeq_f32(val_lo, vdupq_n_f32(0.f)), half, vnegq_f32(half)))), vmin), vmax);
535 int32x4_t ival_hi = vminq_s32(vmaxq_s32(vcvtq_s32_f32(vaddq_f32(val_hi,
536 vbslq_f32(vcgeq_f32(val_hi, vdupq_n_f32(0.f)), half, vnegq_f32(half)))), vmin), vmax);
537#endif
538
539 int16x8_t p16 = vcombine_s16(vqmovn_s32(ival_lo), vqmovn_s32(ival_hi));
540 int8x8_t p8 = vqmovn_s16(p16);
541 if (lutptr) {
542 int8_t tmp[8];
543 vst1_s8(tmp, p8);
544 for (int k = 0; k < 8; k++)
545 outptr[i + k] = (int8_t)lutptr[(int)tmp[k] + 128];
546 } else {
547 vst1_s8(outptr + i, p8);
548 }
549 }
550 }
551 #endif
552 for (; i < r.end; i++)
553 {
554 float val;
555 if (isMul) {
556 val = mc * ((float)p0[i] - zp0f) * ((float)p1[i] - zp1f) + ozpf;
557 } else {
558 val = c0 * (float)p0[i] + c1 * (float)p1[i] + off;
559 for (int k = 2; k < ninputs; k++)
560 val += cptr[k] * (float)inptrs[k][i];
561 }
562
563 int ival = cvRound(val);
564 ival = std::max(out_min, std::min(out_max, ival));
565
566 if (lutptr)
567 ival = (int)lutptr[ival + 128];
568
569 outptr[i] = (int8_t)ival;
570 }
571 }, nstripes);
572 }
573
574 if (uouts) {
575 out.copyTo(uouts->at(0));
576 }
577 }
