66// Multi-head attention via MLAS flash-attention. Computes
67// output[b, i, h, :] = softmax(scale * Q[b,h,i,:] @ K[b,h,:,:]^T) @ V[b,h,:,:]
68// fused into one tiled kernel without materializing the q_seq x kv_seq
69// attention matrix.
70//
71// Layouts (row-major contiguous, FP32):
72// query : [batch, num_heads, q_seq_len, qk_head_size]
73// key : [batch, num_heads, kv_seq_len, qk_head_size]
74// value : [batch, num_heads, kv_seq_len, v_head_size]
75// output: [batch, q_seq_len, num_heads, v_head_size] (heads *after* seq)
76//
77// scale - usually 1 / sqrt(qk_head_size).
78// q_block_size - tile size along the q sequence (e.g. 256).
79// kv_block_size - tile size along the kv sequence (e.g. 256).
80// scratch - caller-owned buffer of at least
81// thread_count * mlasFlashAttentionBufferBytesPerThread(...).
82// thread_count - number of MLAS workers to fan out across (typically
83// cv::getNumThreads()).
84//
85// Returns false if MLAS is unavailable or arguments are invalid.
86bool mlasFlashAttention(const float* query, const float* key, const float* value,
87 float* output,
88 int batch_size, int num_heads,
89 int q_seq_len, int kv_seq_len,
90 int qk_head_size, int v_head_size,
91 float scale,
92 int q_block_size, int kv_block_size,
93 void* scratch, int thread_count);
