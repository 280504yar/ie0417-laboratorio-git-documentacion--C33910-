49void TrackerMILImpl::init(InputArray image, const Rect& boundingBox)
50{
51 sampler = makePtr<TrackerSampler>();
52 featureSet = makePtr<TrackerFeatureSet>();
53
54 Mat intImage;
55 compute_integral(image.getMat(), intImage);
56 TrackerSamplerCSC::Params CSCparameters;
57 CSCparameters.initInRad = params.samplerInitInRadius;
58 CSCparameters.searchWinSize = params.samplerSearchWinSize;
59 CSCparameters.initMaxNegNum = params.samplerInitMaxNegNum;
60 CSCparameters.trackInPosRad = params.samplerTrackInRadius;
61 CSCparameters.trackMaxPosNum = params.samplerTrackMaxPosNum;
62 CSCparameters.trackMaxNegNum = params.samplerTrackMaxNegNum;
63
64 Ptr<TrackerSamplerAlgorithm> CSCSampler = makePtr<TrackerSamplerCSC>(CSCparameters);
65 CV_Assert(sampler->addTrackerSamplerAlgorithm(CSCSampler));
66
67 //or add CSC sampler with default parameters
68 //sampler->addTrackerSamplerAlgorithm( "CSC" );
69
70 //Positive sampling
71 CSCSampler.staticCast<TrackerSamplerCSC>()->setMode(TrackerSamplerCSC::MODE_INIT_POS);
72 sampler->sampling(intImage, boundingBox);
73 std::vector<Mat> posSamples = sampler->getSamples();
74
75 //Negative sampling
76 CSCSampler.staticCast<TrackerSamplerCSC>()->setMode(TrackerSamplerCSC::MODE_INIT_NEG);
77 sampler->sampling(intImage, boundingBox);
78 std::vector<Mat> negSamples = sampler->getSamples();
79
80 CV_Assert(!posSamples.empty());
81 CV_Assert(!negSamples.empty());
82
83 //compute HAAR features
84 TrackerFeatureHAAR::Params HAARparameters;
85 HAARparameters.numFeatures = params.featureSetNumFeatures;
86 HAARparameters.rectSize = Size((int)boundingBox.width, (int)boundingBox.height);
87 HAARparameters.isIntegral = true;
88 Ptr<TrackerFeature> trackerFeature = makePtr<TrackerFeatureHAAR>(HAARparameters);
89 featureSet->addTrackerFeature(trackerFeature);
90
91 featureSet->extraction(posSamples);
92 const std::vector<Mat> posResponse = featureSet->getResponses();
93
94 featureSet->extraction(negSamples);
95 const std::vector<Mat> negResponse = featureSet->getResponses();
96
97 model = makePtr<TrackerMILModel>(boundingBox);
98 Ptr<TrackerStateEstimatorMILBoosting> stateEstimator = makePtr<TrackerStateEstimatorMILBoosting>(params.featureSetNumFeatures);
99 model->setTrackerStateEstimator(stateEstimator);
100
101 //Run model estimation and update
102 model.staticCast<TrackerMILModel>()->setMode(TrackerMILModel::MODE_POSITIVE, posSamples);
103 model->modelEstimation(posResponse);
104 model.staticCast<TrackerMILModel>()->setMode(TrackerMILModel::MODE_NEGATIVE, negSamples);
105 model->modelEstimation(negResponse);
106 model->modelUpdate();
107}

109bool TrackerMILImpl::update(InputArray image, Rect& boundingBox)
110{
111 Mat intImage;
112 compute_integral(image.getMat(), intImage);
113
114 //get the last location [AAM] X(k-1)
115 Ptr<TrackerTargetState> lastLocation = model->getLastTargetState();
116 Rect lastBoundingBox((int)lastLocation->getTargetPosition().x, (int)lastLocation->getTargetPosition().y, lastLocation->getTargetWidth(),
117 lastLocation->getTargetHeight());
118
119 //sampling new frame based on last location
120 auto& samplers = sampler->getSamplers();
121 CV_Assert(!samplers.empty());
122 CV_Assert(samplers[0]);
123 samplers[0].staticCast<TrackerSamplerCSC>()->setMode(TrackerSamplerCSC::MODE_DETECT);
124 sampler->sampling(intImage, lastBoundingBox);
125 std::vector<Mat> detectSamples = sampler->getSamples();
126 if (detectSamples.empty())
127 return false;
128
129 /*//TODO debug samples
130 Mat f;
131 image.copyTo(f);
132
133 for( size_t i = 0; i < detectSamples.size(); i=i+10 )
134 {
135 Size sz;
136 Point off;
137 detectSamples.at(i).locateROI(sz, off);
138 rectangle(f, Rect(off.x,off.y,detectSamples.at(i).cols,detectSamples.at(i).rows), Scalar(255,0,0), 1);
139 }*/
140
141 //extract features from new samples
142 featureSet->extraction(detectSamples);
143 std::vector<Mat> response = featureSet->getResponses();
144
145 //predict new location
146 ConfidenceMap cmap;
147 model.staticCast<TrackerMILModel>()->setMode(TrackerMILModel::MODE_ESTIMATON, detectSamples);
148 model.staticCast<TrackerMILModel>()->responseToConfidenceMap(response, cmap);
149 model->getTrackerStateEstimator().staticCast<TrackerStateEstimatorMILBoosting>()->setCurrentConfidenceMap(cmap);
150
151 if (!model->runStateEstimator())
152 {
153 return false;
154 }
155
156 Ptr<TrackerTargetState> currentState = model->getLastTargetState();
157 boundingBox = Rect((int)currentState->getTargetPosition().x, (int)currentState->getTargetPosition().y, currentState->getTargetWidth(),
158 currentState->getTargetHeight());
159
160 /*//TODO debug
161 rectangle(f, lastBoundingBox, Scalar(0,255,0), 1);
162 rectangle(f, boundingBox, Scalar(0,0,255), 1);
163 imshow("f", f);
164 //waitKey( 0 );*/
165
166 //sampling new frame based on new location
167 //Positive sampling
168 samplers[0].staticCast<TrackerSamplerCSC>()->setMode(TrackerSamplerCSC::MODE_INIT_POS);
169 sampler->sampling(intImage, boundingBox);
170 std::vector<Mat> posSamples = sampler->getSamples();
171
172 //Negative sampling
173 samplers[0].staticCast<TrackerSamplerCSC>()->setMode(TrackerSamplerCSC::MODE_INIT_NEG);
174 sampler->sampling(intImage, boundingBox);
175 std::vector<Mat> negSamples = sampler->getSamples();
176
177 if (posSamples.empty() || negSamples.empty())
178 return false;
179
180 //extract features
181 featureSet->extraction(posSamples);
182 std::vector<Mat> posResponse = featureSet->getResponses();
183
184 featureSet->extraction(negSamples);
185 std::vector<Mat> negResponse = featureSet->getResponses();
186
187 //model estimate
188 model.staticCast<TrackerMILModel>()->setMode(TrackerMILModel::MODE_POSITIVE, posSamples);
189 model->modelEstimation(posResponse);
190 model.staticCast<TrackerMILModel>()->setMode(TrackerMILModel::MODE_NEGATIVE, negSamples);
191 model->modelEstimation(negResponse);
192
193 //model update
194 model->modelUpdate();
195
196 return true;
197}
