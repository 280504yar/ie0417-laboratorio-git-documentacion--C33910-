7// Fuses multiple Gemm layers that share the same input into one wider Gemm.
8//
9// Pattern:
10// X --+--> Gemm[W0, b0] -> Y0
11// +--> Gemm[W1, b1] -> Y1
12// +--> Gemm[W2, b2] -> Y2
13// ...
14//
15// Replacement:
16// X -> Gemm[concat(Wi), concat(bi)] -> Y_concat
17// Y_concat -> Slice2(0, N0) -> Y0
18// -> Slice2(N0, N0+N1) -> Y1
19// -> Slice2(N0+N1, ...) -> Y2
20// ...
21//
22// Triggers when:
23// - At least 2 Gemm layers share the same input arg.
24// - Each Gemm has a const weight blob and (optionally) const bias.
25// - All Gemms have identical (alpha, beta, trans_a, trans_b) and matching K.

38static bool readGemmWeight(const Ptr<LayerInfo>& l, bool trans_b, Mat& W_out)
39{
40 if (l->blobs.empty()) return false;
41 const Mat& W = l->blobs[0];
42 if (W.dims != 2 || W.type() != CV_32F) return false;
43 if (trans_b) {
44 cv::transpose(W, W_out);
45 } else {
46 W.copyTo(W_out);
47 }
48 return true;
49}
50
51static bool readGemmBias(const Ptr<LayerInfo>& l, Mat& b_out)
52{
53 if (l->blobs.size() < 2) { b_out.release(); return true; }
54 const Mat& b = l->blobs[1];
55 if (b.type() != CV_32F) return false;
56 b.copyTo(b_out);
57 return true;
58}

79 bool inspectGemm(const vector<Ptr<LayerInfo>>& prog, int idx, GemmInfo& info) const
80 {
81 if (idx < 0 || idx >= (int)prog.size() || !prog[idx]) return false;
82 const Ptr<LayerInfo>& l = prog[idx];
83 GemmLayer* g = dynamic_cast<GemmLayer*>(l.get());
84 if (!g) return false;
85
86 if (l->inputs.size() != 1) return false;
87 if (l->blobs.empty() || l->blobs.size() > 2) return false;
88
89 const Mat& W = l->blobs[0];
90 if (W.dims != 2 || W.type() != CV_32F) return false;
91
92 bool trans_a = g->trans_a;
93 bool trans_b = g->trans_b;
94 float alpha = g->alpha;
95 float beta = g->beta;
96
97 if (trans_a) return false;
98 if (alpha != 1.f) return false;
99 if (l->blobs.size() == 2 && beta != 1.f) return false;
100
101 int K = trans_b ? W.size[1] : W.size[0];
102 int N = trans_b ? W.size[0] : W.size[1];
103 if (K <= 0 || N <= 0) return false;
104
105 if (l->blobs.size() == 2) {
106 const Mat& b = l->blobs[1];
107 if (b.type() != CV_32F) return false;
108 if ((int)b.total() != N) return false;
109 }
110
111 info.layer_idx = idx;
112 info.input = l->inputs[0];
113 info.K = K;
114 info.N = N;
115 info.trans_a = trans_a;
116 info.trans_b = trans_b;
117 info.alpha = alpha;
118 info.beta = beta;
119 info.has_bias = (l->blobs.size() == 2);
120 info.flatten_a = g->flatten_a;
121 return true;
122 }

126 bool fuseGraph(Ptr<Graph>& graph)
127 {
128 const vector<Ptr<LayerInfo>>& prog = graph->prog();
129 size_t nops = prog.size();
130
131 struct Key { int input_idx; int trans_b; int K; };
132 auto keyLess = [](const Key& a, const Key& b) {
133 if (a.input_idx != b.input_idx) return a.input_idx < b.input_idx;
134 if (a.trans_b != b.trans_b) return a.trans_b < b.trans_b;
135 return a.K < b.K;
136 };
137 std::map<Key, vector<GemmInfo>, decltype(keyLess)> groups(keyLess);
138
139 for (size_t i = 0; i < nops; i++) {
140 GemmInfo info;
141 if (!inspectGemm(prog, (int)i, info)) continue;
142 Key k{info.input.idx, info.trans_b ? 1 : 0, info.K};
143 groups[k].push_back(info);
144 }
145
146 bool modified = false;
147 std::set<int> removed_ops;
148 vector<std::pair<int, vector<Ptr<LayerInfo>>>> insertions; // (insert_pos, fused-and-slice layers)
149
150 for (auto& group : groups) {
151 auto infos = group.second;
152 if (infos.size() < 2) continue;
153
154 bool all_have_bias = infos[0].has_bias;
155 bool uniform = true;
156 for (auto& info : infos)
157 if (info.has_bias != all_have_bias) { uniform = false; break; }
158 if (!uniform) continue;
159
160 // Require uniform flatten_a too — the fused Gemm has a single
161 // value, and mixing 2D-output and ND-output downstream consumers
162 // would need different post-fusion shape handling.
163 bool all_flatten_a = infos[0].flatten_a;
164 for (auto& info : infos)
165 if (info.flatten_a != all_flatten_a) { uniform = false; break; }
166 if (!uniform) continue;
167
168 int K = infos[0].K;
169 int total_N = 0;
170 for (auto& info : infos) total_N += info.N;
171
172 int wshape[] = { K, total_N };
173 Mat W_concat(2, wshape, CV_32F);
174 int col_offset = 0;
175 for (auto& info : infos) {
176 Mat W;
177 if (!readGemmWeight(prog[info.layer_idx], info.trans_b, W)) {
178 uniform = false; break;
179 }
180 CV_Assert(W.size[0] == K && W.size[1] == info.N);
181 for (int r = 0; r < K; r++) {
182 float* dst = W_concat.ptr<float>(r) + col_offset;
183 memcpy(dst, W.ptr<float>(r), info.N * sizeof(float));
184 }
185 col_offset += info.N;
186 }
187 if (!uniform) continue;
188
189 Mat b_concat;
190 if (all_have_bias) {
191 int bshape[] = { total_N };
192 b_concat.create(1, bshape, CV_32F);
193 int b_offset = 0;
194 for (auto& info : infos) {
195 Mat b;
196 if (!readGemmBias(prog[info.layer_idx], b)) {
197 uniform = false; break;
198 }
199 CV_Assert((int)b.total() == info.N);
200 memcpy(b_concat.ptr<float>() + b_offset, b.ptr<float>(),
201 info.N * sizeof(float));
202 b_offset += info.N;
203 }
204 if (!uniform) continue;
205 }
206
207 int insert_pos = (int)nops;
208 for (auto& info : infos) insert_pos = std::min(insert_pos, info.layer_idx);
209
210 LayerParams fp;
211 fp.name = prog[infos[0].layer_idx]->name + "_shared_input_fused";
212 fp.type = "Gemm";
213 fp.set("transA", false);
214 fp.set("transB", false);
215 fp.set("alpha", 1.f);
216 fp.set("beta", 1.f);
217 fp.set("flatten_a", all_flatten_a);
218 // Mirror the constB / const_C / have_bias signalling that the
219 // GemmLayerImpl's getOpMode() reads from LayerParams. We always
220 // ship the weights as a constant blob, and (when biased) the bias
221 // too — so this fused Gemm has only one runtime input.
222 fp.set("constB", true);
223 fp.set("have_bias", all_have_bias);
224 fp.set("const_C", all_have_bias);
225 if (all_have_bias) fp.set("real_ndims_C", 1);
226 fp.blobs.push_back(W_concat);
227 if (all_have_bias) fp.blobs.push_back(b_concat);
228
229 Ptr<LayerInfo> fused = LayerFactory::createLayerInstance("Gemm", fp);
230 if (!fused) continue;
231
232 string fused_out_name = fp.name + "_out";
233 Arg fused_out_arg = netimpl->getArg(fused_out_name);
234 fused->inputs = { infos[0].input };
235 fused->outputs = { fused_out_arg };
236 fused->netimpl = netimpl;
237
238 vector<Ptr<LayerInfo>> slices;
239 int col_cursor = 0;
240 for (size_t s = 0; s < infos.size(); s++) {
241 int N_s = infos[s].N;
242 int begin = col_cursor;
243 int end = col_cursor + N_s;
244 col_cursor += N_s;
245
246 LayerParams sp;
247 sp.name = prog[infos[s].layer_idx]->name + "_shared_input_slice";
248 sp.type = "Slice2";
249
250 Mat starts(1, 1, CV_32S); starts.at<int>(0) = begin;
251 Mat ends (1, 1, CV_32S); ends.at<int>(0) = end;
252 Mat axes (1, 1, CV_32S); axes.at<int>(0) = -1; // last axis
253 Mat steps (1, 1, CV_32S); steps.at<int>(0) = 1;
254
255 Arg starts_arg = netimpl->newConstArg(sp.name + "_starts", starts);
256 Arg ends_arg = netimpl->newConstArg(sp.name + "_ends", ends);
257 Arg axes_arg = netimpl->newConstArg(sp.name + "_axes", axes);
258 Arg steps_arg = netimpl->newConstArg(sp.name + "_steps", steps);
259
260 Ptr<LayerInfo> slice = LayerFactory::createLayerInstance("Slice2", sp);
261 if (!slice) { uniform = false; break; }
262 slice->inputs = { fused_out_arg, starts_arg, ends_arg, axes_arg, steps_arg };
263 slice->outputs = prog[infos[s].layer_idx]->outputs;
264 slice->netimpl = netimpl;
265 slices.push_back(slice);
266 }
267 if (!uniform) continue;
268
269 for (auto& info : infos) removed_ops.insert(info.layer_idx);
270 vector<Ptr<LayerInfo>> bundle;
271 bundle.push_back(fused);
272 for (auto& s : slices) bundle.push_back(s);
273 insertions.emplace_back(insert_pos, std::move(bundle));
274 modified = true;
275 }
276
277 if (!modified) return false;
278
279 std::sort(insertions.begin(), insertions.end(),
280 [](auto& a, auto& b) { return a.first < b.first; });
281
282 vector<Ptr<LayerInfo>> newprog;
283 size_t ins_idx = 0;
284 for (size_t i = 0; i < nops; i++) {
285 while (ins_idx < insertions.size() &&
286 insertions[ins_idx].first == (int)i) {
287 for (auto& l : insertions[ins_idx].second) newprog.push_back(l);
288 ins_idx++;
289 }
290 if (removed_ops.count((int)i)) continue;
291 newprog.push_back(prog[i]);
292 }
293 graph->setProg(newprog);
294 return true;
295 }

298};
