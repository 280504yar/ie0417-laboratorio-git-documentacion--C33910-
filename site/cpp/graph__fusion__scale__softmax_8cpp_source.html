7// Folds a scalar Mul / Div immediately preceding Softmax into the Softmax
8// layer's `scale` parameter. The fused softmax bakes the scale into the
9// per-element load before max/exp/sum/divide, removing one full pass over
10// the (often large) [B, H, S, S] attention-logits tensor.
11//
12// Pattern:
13// X -> Mul(X, c) -> Softmax => X -> Softmax(scale=c)
14// X -> Div(X, c) -> Softmax => X -> Softmax(scale=1/c)

27 explicit ModelFusionScaleSoftmax(Net::Impl* netimpl_) : netimpl(netimpl_) {}

29 bool extractScalar(Arg a, float& out) const
30 {
31 if (!netimpl->isConstArg(a)) return false;
32 Mat t = netimpl->argTensor(a);
33 if (t.total() != 1) return false;
34 if (t.type() == CV_32F) { out = t.ptr<float>()[0]; return true; }
35 if (t.type() == CV_64F) { out = (float)t.ptr<double>()[0]; return true; }
36 return false;
37 }
38
39 void fuse() { fuseGraph(netimpl->mainGraph); }
40
41 bool fuseGraph(Ptr<Graph>& graph)
42 {
43 const vector<Ptr<LayerInfo>>& prog = graph->prog();
44 size_t nops = prog.size();
45 bool modified = false;
46
47 for (size_t i = 0; i < nops; i++) {
48 if (!prog[i]) continue;
49 vector<Ptr<Graph>>* subgraphs = prog[i]->subgraphs();
50 if (subgraphs) {
51 for (Ptr<Graph>& g : *subgraphs)
52 if (fuseGraph(g)) modified = true;
53 }
54 }
55
56 vector<int> usecounts;
57 netimpl->useCounts(usecounts);
58
59 std::set<int> externalArgs;
60 for (Arg out : graph->outputs())
61 externalArgs.insert(out.idx);
62
63 std::map<int, int> producer;
64 for (size_t i = 0; i < nops; i++) {
65 if (!prog[i]) continue;
66 for (Arg out : prog[i]->outputs)
67 producer[out.idx] = (int)i;
68 }
69
70 vector<bool> dropped(nops, false);
71
72 for (size_t i = 0; i < nops; i++) {
73 const Ptr<LayerInfo>& layer = prog[i];
74 if (!layer || dropped[i]) continue;
75
76 SoftmaxLayer* sm = dynamic_cast<SoftmaxLayer*>(layer.get());
77 if (!sm || sm->logSoftMax) continue;
78 if (layer->inputs.size() != 1) continue;
79
80 Arg sm_in = layer->inputs[0];
81 auto it = producer.find(sm_in.idx);
82 if (it == producer.end()) continue;
83 int prod_idx = it->second;
84 if (prod_idx < 0 || dropped[prod_idx]) continue;
85
86 const Ptr<LayerInfo>& pl = prog[prod_idx];
87 NaryEltwiseLayer* elt = dynamic_cast<NaryEltwiseLayer*>(pl.get());
88 if (!elt) continue;
89 const auto op = elt->op;
90 const bool is_mul = (op == NaryEltwiseLayer::OPERATION::PROD);
91 const bool is_div = (op == NaryEltwiseLayer::OPERATION::DIV);
92 if (!is_mul && !is_div) continue;
93 if (pl->inputs.size() != 2 || pl->outputs.size() != 1) continue;
94
95 // Identify which operand is the scalar constant.
96 float scalar = 0.f;
97 int x_slot = -1;
98 if (extractScalar(pl->inputs[1], scalar)) {
99 x_slot = 0;
100 } else if (is_mul && extractScalar(pl->inputs[0], scalar)) {
101 // Mul is commutative; Div with the data on the right doesn't
102 // simplify (1 / scale*x ≠ softmax-fold) so only handle Mul here.
103 x_slot = 1;
104 } else {
105 continue;
106 }
107
108 if (is_div) {
109 if (scalar == 0.f) continue;
110 scalar = 1.f / scalar;
111 }
112 if (!std::isfinite(scalar)) continue;
113
114 bool single_consumer = usecounts[sm_in.idx] == 1
115 && externalArgs.count(sm_in.idx) == 0;
116 if (!single_consumer) continue;
117
118 sm->scale *= scalar;
119 layer->inputs[0] = pl->inputs[x_slot];
120 dropped[prod_idx] = true;
121 usecounts[sm_in.idx] = 0;
122 modified = true;
123 }
124
125 if (modified) {
126 vector<Ptr<LayerInfo>> newprog;
127 newprog.reserve(nops);
128 for (size_t i = 0; i < nops; i++) {
129 if (!dropped[i] && prog[i])
130 newprog.push_back(prog[i]);
131 }
132 graph->setProg(newprog);
133 }
134
135 return modified;
136 }

138 Net::Impl* netimpl;
139};
