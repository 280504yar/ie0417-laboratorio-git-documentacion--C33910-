102class PluginWriter : public cv::IVideoWriter
103{
104 const OpenCV_VideoIO_Plugin_API_preview* plugin_api_;
105 CvPluginWriter writer_;
106
107public:
108 static
109 Ptr<PluginWriter> create(const OpenCV_VideoIO_Plugin_API_preview* plugin_api,
110 const std::string& filename, int fourcc, double fps, const cv::Size& sz,
111 const VideoWriterParameters& params)
112 {
113 CV_Assert(plugin_api);
114 CvPluginWriter writer = NULL;
115 if (plugin_api->api_header.api_version >= 1 && plugin_api->v1.Writer_open_with_params)
116 {
117 CV_Assert(plugin_api->v0.Writer_release);
118 CV_Assert(!filename.empty());
119 std::vector<int> vint_params = params.getIntVector();
120 int* c_params = &vint_params[0];
121 unsigned n_params = (unsigned)(vint_params.size() / 2);
122
123 if (CV_ERROR_OK == plugin_api->v1.Writer_open_with_params(filename.c_str(), fourcc, fps, sz.width, sz.height, c_params, n_params, &writer))
124 {
125 CV_Assert(writer);
126 return makePtr<PluginWriter>(plugin_api, writer);
127 }
128 }
129 else if (plugin_api->v0.Writer_open)
130 {
131 CV_Assert(plugin_api->v0.Writer_release);
132 CV_Assert(!filename.empty());
133 const bool isColor = params.get(VIDEOWRITER_PROP_IS_COLOR, true);
134 const int depth = params.get(VIDEOWRITER_PROP_DEPTH, CV_8U);
135 if (depth != CV_8U)
136 {
137 CV_LOG_WARNING(NULL, "Video I/O plugin doesn't support (due to lower API level) creation of VideoWriter with depth != CV_8U");
138 return Ptr<PluginWriter>();
139 }
140 if (CV_ERROR_OK == plugin_api->v0.Writer_open(filename.c_str(), fourcc, fps, sz.width, sz.height, isColor, &writer))
141 {
142 CV_Assert(writer);
143 return makePtr<PluginWriter>(plugin_api, writer);
144 }
145 }
146 return Ptr<PluginWriter>();
147 }
148
149 PluginWriter(const OpenCV_VideoIO_Plugin_API_preview* plugin_api, CvPluginWriter writer)
150 : plugin_api_(plugin_api), writer_(writer)
151 {
152 CV_Assert(plugin_api_); CV_Assert(writer_);
153 }
154
155 ~PluginWriter()
156 {
157 CV_DbgAssert(plugin_api_->v0.Writer_release);
158 if (CV_ERROR_OK != plugin_api_->v0.Writer_release(writer_))
159 CV_LOG_ERROR(NULL, "Video I/O: Can't release writer by plugin '" << plugin_api_->api_header.api_description << "'");
160 writer_ = NULL;
161 }
162 double getProperty(int prop) const CV_OVERRIDE
163 {
164 double val = CAP_PROP_UNKNOWN;
165 if (plugin_api_->v0.Writer_getProperty)
166 if (CV_ERROR_OK != plugin_api_->v0.Writer_getProperty(writer_, prop, &val))
167 val = CAP_PROP_UNKNOWN;
168 return val;
169 }
170 bool setProperty(int prop, double val) CV_OVERRIDE
171 {
172 if (plugin_api_->v0.Writer_setProperty)
173 if (CV_ERROR_OK == plugin_api_->v0.Writer_setProperty(writer_, prop, val))
174 return true;
175 return false;
176 }
177 bool isOpened() const CV_OVERRIDE
178 {
179 return writer_ != NULL; // TODO always true
180 }
181 bool write(cv::InputArray arr) CV_OVERRIDE
182 {
183 cv::Mat img = arr.getMat();
184 CV_DbgAssert(writer_);
185 CV_Assert(plugin_api_->v0.Writer_write);
186 if (CV_ERROR_OK != plugin_api_->v0.Writer_write(writer_, img.data, (int)img.step[0], img.cols, img.rows, img.channels()))
187 {
188 CV_LOG_DEBUG(NULL, "Video I/O: Can't write frame by plugin '" << plugin_api_->api_header.api_description << "'");
189 return false;
190 }
191 return true;
192 }
193 int getCaptureDomain() const CV_OVERRIDE
194 {
195 return plugin_api_->v0.captureAPI;
196 }
