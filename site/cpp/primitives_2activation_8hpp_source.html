21 template <template<class> class Op, class T>
22 struct BaseOp : public CUDABackendNode
23 {
24 protected:
25 using wrapper_type = GetCUDABackendWrapperType<T>;
26
27 void forward(
28 const std::vector<cuda::GpuMatND>& inputs,
29 const std::vector<cuda::GpuMatND>& outputs,
30 csl::Workspace& workspace) override
31 {
32 for (int i = 0; i < inputs.size(); i++)
33 {
34 auto input = csl::viewOf<T>(inputs[i]);
35 auto output = csl::spanOf<T>(outputs[i]);
36
37 static_cast<const Op<T>*>(this)->calculate(output, input);
38 }
39 }
40 };

74 template <class T>
75 class ChannelwiseReLUOp final : public BaseOp<ChannelwiseReLUOp, T> {
76 public:
77 ChannelwiseReLUOp(csl::Stream stream_, const Mat& slope)
78 : stream(std::move(stream_))
79 {
80 CV_Assert(!slope.empty());
81 slopeTensor = csl::makeTensorHeader<T>(slope);
82 csl::copyMatToTensor<T>(slope, slopeTensor, stream);
83 }
84
85 void calculate(csl::TensorSpan<T> output, csl::TensorView<T> input) const
86 {
87 CV_Assert(input.get_axis_size(1) == slopeTensor.size());
88 std::size_t inner_size = input.size_range(2, input.rank());
89 kernels::axiswise_relu<T>(stream, output, input, inner_size, slopeTensor);
90 }
91
92 private:
93 csl::Stream stream;
94 csl::Tensor<T> slopeTensor;
95 };
