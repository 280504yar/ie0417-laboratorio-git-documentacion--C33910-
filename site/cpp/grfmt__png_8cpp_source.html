154{
155 _delayNum = delayNum;
156 _delayDen = delayDen;
157
158 if (!src.empty())
159 {
160 png_uint_32 rowbytes = src.cols * (uint32_t)src.elemSize();
161 _width = src.cols;
162 _height = src.rows;
163 _colorType = src.channels() == 1 ? PNG_COLOR_TYPE_GRAY : src.channels() == 3 ? PNG_COLOR_TYPE_RGB : PNG_COLOR_TYPE_RGB_ALPHA;
164 _pixels = src.data;
165 _rows.resize(_height);

168 _rows[i] = _pixels + i * rowbytes;
169 return true;
170 }
171 return false;

252bool PngDecoder::readHeader()
253{
254 if (m_frame_count > 1 /* if true, it means readHeader() was called before */)
255 {
256 return true;
257 }
258
259 // Declare dynamic variables before a potential longjmp.
260 Chunk chunk;
261
262 if (!InitPngPtr())
263 return false;
264
265 if (setjmp(png_jmpbuf(m_png_ptr)))
266 return false;
267
268 m_buf_pos = 0;
269 unsigned char sig[8];
270 uint32_t id = 0;
271
272 if( !m_buf.empty() )
273 png_set_read_fn(m_png_ptr, this, (png_rw_ptr)readDataFromBuf );
274 else
275 {
276 m_f = fopen(m_filename.c_str(), "rb");
277 if (!m_f)
278 {
279 return false;
280 }
281 png_init_io(m_png_ptr, m_f);
282 }
283
284 // Read PNG header: 137 80 78 71 13 10 26 10
285 if (!readFromStreamOrBuffer(&sig, 8))
286 return false;
287
288 // IHDR chunk shall be first. ( https://www.w3.org/TR/png-3/#5ChunkOrdering )
289 id = read_chunk(m_chunkIHDR);
290 if (id == id_CgBI)
291 {
292 CV_LOG_ERROR(NULL, "CgBI chunk (Apple private) found as the first chunk. IHDR is expected.");
293 return false;
294 }
295 if (id != id_IHDR)
296 {
297 CV_LOG_ERROR(NULL, "IHDR chunk shall be first. This data may be broken or malformed.");
298 return false;
299 }
300
301 m_is_fcTL_loaded = false;
302 while (true)
303 {
304 id = read_chunk(chunk);
305
306 if (!id || (m_f && feof(m_f)) || (!m_buf.empty() && m_buf_pos > m_buf.total()))
307 {
308 return false;
309 }
310
311 if (id == id_IDAT)
312 {
313 if (m_f)
314 fseek(m_f, 0, SEEK_SET);
315 else
316 m_buf_pos = 0;
317 break;
318 }
319
320 if (id == id_acTL)
321 {
322 // https://wiki.mozilla.org/APNG_Specification#%60acTL%60:_The_Animation_Control_Chunk
323 m_animation.loop_count = png_get_uint_32(&chunk.p[12]);
324
325 m_frame_count = png_get_uint_32(&chunk.p[8]);
326 if (m_frame_count == 0)
327 return false;
328 }
329
330 if (id == id_fcTL)
331 {
332 // https://wiki.mozilla.org/APNG_Specification#%60fcTL%60:_The_Frame_Control_Chunk
333 m_is_fcTL_loaded = true;
334 w0 = png_get_uint_32(&chunk.p[12]);
335 h0 = png_get_uint_32(&chunk.p[16]);
336 x0 = png_get_uint_32(&chunk.p[20]);
337 y0 = png_get_uint_32(&chunk.p[24]);
338 delay_num = png_get_uint_16(&chunk.p[28]);
339 delay_den = png_get_uint_16(&chunk.p[30]);
340 dop = chunk.p[32];
341 bop = chunk.p[33];
342 if (dop > 2 || bop > 1)
343 return false;
344 }
345
346 if (id == id_PLTE || id == id_tRNS)
347 m_chunksInfo.push_back(chunk);
348 }
349
350 png_uint_32 wdth, hght;
351 int bit_depth, color_type, num_trans=0;
352 png_bytep trans;
353 png_color_16p trans_values;
354
355 // Free chunk in case png_read_info uses longjmp.
356 chunk.p.clear();
357 chunk.p.shrink_to_fit();
358
359 png_read_info( m_png_ptr, m_info_ptr );
360 png_get_IHDR(m_png_ptr, m_info_ptr, &wdth, &hght,
361 &bit_depth, &color_type, 0, 0, 0);
362
363 m_width = (int)wdth;
364 m_height = (int)hght;
365 m_color_type = color_type;
366 m_bit_depth = bit_depth;
367
368 if (m_is_fcTL_loaded && ((int64_t)x0 + w0 > m_width || (int64_t)y0 + h0 > m_height || dop > 2 || bop > 1))
369 return false;
370
371 png_color_16p background_color;
372 if (png_get_bKGD(m_png_ptr, m_info_ptr, &background_color))
373 m_animation.bgcolor = Scalar(background_color->blue, background_color->green, background_color->red);
374
375 if (bit_depth <= 8 || bit_depth == 16)
376 {
377 switch (color_type)
378 {
379 case PNG_COLOR_TYPE_RGB:
380 case PNG_COLOR_TYPE_PALETTE:
381 png_get_tRNS(m_png_ptr, m_info_ptr, &trans, &num_trans, &trans_values);
382 if (num_trans > 0)
383 m_type = CV_8UC4;
384 else
385 m_type = CV_8UC3;
386 break;
387 case PNG_COLOR_TYPE_GRAY_ALPHA:
388 case PNG_COLOR_TYPE_RGB_ALPHA:
389 m_type = CV_8UC4;
390 break;
391 default:
392 m_type = CV_8UC1;
393 }
394 if (bit_depth == 16)
395 m_type = CV_MAKETYPE(CV_16U, CV_MAT_CN(m_type));
396 }
397
398 return true;
399}

401bool PngDecoder::readData( Mat& img )
402{
403 // Declare dynamic variables before a potential longjmp.
404 AutoBuffer<unsigned char*> _buffer(m_height);
405 unsigned char** buffer = _buffer.data();
406 Chunk chunk;
407
408 if (m_frame_count > 1)
409 {
410 Mat mat_cur = Mat::zeros(img.rows, img.cols, m_type);
411 uint32_t id = 0;
412 uint32_t j = 0;
413 uint32_t imagesize = m_width * m_height * (uint32_t)mat_cur.elemSize();
414 m_is_IDAT_loaded = false;
415
416 if (m_frame_no == 0)
417 {
418 if (m_mat_raw.empty())
419 {
420 if (m_f)
421 fseek(m_f, -8, SEEK_CUR);
422 else
423 m_buf_pos -= 8;
424 }
425 m_mat_raw = Mat(img.rows, img.cols, m_type);
426 m_mat_next = Mat(img.rows, img.cols, m_type);
427 frameRaw.setMat(m_mat_raw);
428 frameNext.setMat(m_mat_next);
429 }
430 else
431 m_mat_next.copyTo(mat_cur);
432
433 frameCur.setMat(mat_cur);
434
435 if (!processing_start((void*)&frameRaw, mat_cur))
436 return false;
437
438 // See https://github.com/opencv/opencv/issues/27744
439 if( setjmp( png_jmpbuf ( m_png_ptr ) ) == 0 )
440 {
441 while (true)
442 {
443 id = read_chunk(chunk);
444 if (!id)
445 return false;
446
447 if (id == id_fcTL && m_is_IDAT_loaded)
448 {
449 if (!m_is_fcTL_loaded)
450 {
451 m_mat_raw.copyTo(m_animation.still_image);
452 }
453 else
454 {
455 if (processing_finish())
456 {
457 if (dop == 2)
458 memcpy(frameNext.getPixels(), frameCur.getPixels(), imagesize);
459
460 if ((uint64_t)x0 + w0 > frameCur.getWidth() || (uint64_t)y0 + h0 > frameCur.getHeight())
461 return false;
462
463 compose_frame(frameCur.getRows(), frameRaw.getRows(), bop, x0, y0, w0, h0, mat_cur);
464 if (!delay_den)
465 delay_den = 100;
466 m_animation.durations.push_back(cvRound(1000. * delay_num / delay_den));
467
468 if (mat_cur.channels() == img.channels())
469 {
470 if (mat_cur.depth() == CV_16U && img.depth() == CV_8U)
471 mat_cur.convertTo(img, CV_8U, 1. / 255);
472 else
473 mat_cur.copyTo(img);
474 }
475 else
476 {
477 Mat mat_cur_scaled;
478 if (mat_cur.depth() == CV_16U && img.depth() == CV_8U)
479 mat_cur.convertTo(mat_cur_scaled, CV_8U, 1. / 255);
480 else
481 mat_cur_scaled = mat_cur;
482
483 if (img.channels() == 1)
484 cvtColor(mat_cur_scaled, img, COLOR_BGRA2GRAY);
485 else if (img.channels() == 3)
486 cvtColor(mat_cur_scaled, img, COLOR_BGRA2BGR);
487 }
488
489 if (dop != 2)
490 {
491 memcpy(frameNext.getPixels(), frameCur.getPixels(), imagesize);
492 if (dop == 1)
493 for (j = 0; j < h0; j++)
494 memset(frameNext.getRows()[y0 + j] + x0 * img.channels(), 0, w0 * img.channels());
495 }
496 }
497 else
498 {
499 return false;
500 }
501 }
502
503 w0 = png_get_uint_32(&chunk.p[12]);
504 h0 = png_get_uint_32(&chunk.p[16]);
505 x0 = png_get_uint_32(&chunk.p[20]);
506 y0 = png_get_uint_32(&chunk.p[24]);
507 delay_num = png_get_uint_16(&chunk.p[28]);
508 delay_den = png_get_uint_16(&chunk.p[30]);
509 dop = chunk.p[32];
510 bop = chunk.p[33];
511
512 if ((int64_t)x0 + w0 > img.cols || (int64_t)y0 + h0 > img.rows || dop > 2 || bop > 1)
513 {
514 return false;
515 }
516
517 memcpy(&m_chunkIHDR.p[8], &chunk.p[12], 8);
518
519 if (m_is_fcTL_loaded)
520 return true;
521 else
522 {
523 m_is_fcTL_loaded = true;
524 ClearPngPtr();
525 if (!processing_start((void*)&frameRaw, mat_cur))
526 return false;
527 }
528 }
529 else if (id == id_IDAT)
530 {
531 m_is_IDAT_loaded = true;
532 png_process_data(m_png_ptr, m_info_ptr, chunk.p.data(), chunk.p.size());
533 }
534 else if (id == id_fdAT && m_is_fcTL_loaded)
535 {
536 m_is_IDAT_loaded = true;
537 png_save_uint_32(&chunk.p[4], static_cast<uint32_t>(chunk.p.size() - 16));
538 memcpy(&chunk.p[8], "IDAT", 4);
539 png_process_data(m_png_ptr, m_info_ptr, &chunk.p[4], chunk.p.size() - 4);
540 }
541 else if (id == id_IEND)
542 {
543 if (processing_finish())
544 {
545 compose_frame(frameCur.getRows(), frameRaw.getRows(), bop, x0, y0, w0, h0, mat_cur);
546 if (!delay_den)
547 delay_den = 100;
548 m_animation.durations.push_back(cvRound(1000.*delay_num/delay_den));
549
550 if (mat_cur.depth() == CV_16U && img.depth() == CV_8U && mat_cur.channels() == img.channels())
551 mat_cur.convertTo(img, CV_8U, 1. / 255);
552 else
553 {
554 if (mat_cur.depth() == CV_16U && img.depth() == CV_8U)
555 mat_cur.convertTo(mat_cur, CV_8U, 1. / 255);
556 if (mat_cur.channels() == img.channels())
557 mat_cur.copyTo(img);
558 else if (img.channels() == 1)
559 cvtColor(mat_cur, img, COLOR_BGRA2GRAY);
560 else if (img.channels() == 3)
561 cvtColor(mat_cur, img, COLOR_BGRA2BGR);
562 }
563 }
564 else
565 return false;
566
567 return true;
568 }
569 else
570 png_process_data(m_png_ptr, m_info_ptr, chunk.p.data(), chunk.p.size());
571 }
572 return false;
573 }
574 else
575 {
576 // libpng internal error is detected.
577 return false;
578 }
579 }
580
581 volatile bool result = false;
582 bool color = img.channels() > 1;
583
584 if( m_png_ptr && m_info_ptr && m_width && m_height )
585 {
586 if( setjmp( png_jmpbuf ( m_png_ptr ) ) == 0 )
587 {
588 int y;
589
590 if( img.depth() == CV_8U && m_bit_depth == 16 )
591 png_set_strip_16( m_png_ptr );
592 else if( !isBigEndian() )
593 png_set_swap( m_png_ptr );
594
595 if(img.channels() < 4)
596 {
597 /* observation: png_read_image() writes 400 bytes beyond
598 * end of data when reading a 400x118 color png
599 * "mpplus_sand.png". OpenCV crashes even with demo
600 * programs. Looking at the loaded image I'd say we get 4
601 * bytes per pixel instead of 3 bytes per pixel. Test
602 * indicate that it is a good idea to always ask for
603 * stripping alpha.. 18.11.2004 Axel Walthelm
604 */
605 png_set_strip_alpha( m_png_ptr );
606 } else
607 png_set_tRNS_to_alpha( m_png_ptr );
608
609 if( m_color_type == PNG_COLOR_TYPE_PALETTE )
610 png_set_palette_to_rgb( m_png_ptr );
611
612 if( (m_color_type & PNG_COLOR_MASK_COLOR) == 0 && m_bit_depth < 8 )
613#if (PNG_LIBPNG_VER_MAJOR*10000 + PNG_LIBPNG_VER_MINOR*100 + PNG_LIBPNG_VER_RELEASE >= 10209) || \
614 (PNG_LIBPNG_VER_MAJOR == 1 && PNG_LIBPNG_VER_MINOR == 0 && PNG_LIBPNG_VER_RELEASE >= 18)
615 png_set_expand_gray_1_2_4_to_8( m_png_ptr );
616#else
617 png_set_gray_1_2_4_to_8( png_ptr );
618#endif
619
620 if( (m_color_type & PNG_COLOR_MASK_COLOR) && color && !m_use_rgb)
621 png_set_bgr( m_png_ptr ); // convert RGB to BGR
622 else if( color )
623 png_set_gray_to_rgb( m_png_ptr ); // Gray->RGB
624 else
625 png_set_rgb_to_gray( m_png_ptr, 1, 0.299, 0.587 ); // RGB->Gray
626
627 png_set_interlace_handling( m_png_ptr );
628 png_read_update_info( m_png_ptr, m_info_ptr );
629
630 for( y = 0; y < m_height; y++ )
631 buffer[y] = img.data + y*img.step;
632
633 png_read_image( m_png_ptr, buffer );
634 png_read_end( m_png_ptr, m_info_ptr);
635
636 if (m_read_options) {
637 // Get tEXt chunks
638 png_textp text_ptr;
639 int num_text = 0;
640
641 png_get_text(m_png_ptr, m_info_ptr, &text_ptr, &num_text);
642
643 for (size_t i = 0; i < static_cast<size_t>(num_text); ++i) {
644 const char* key = text_ptr[i].key;
645 const char* value = text_ptr[i].text;
646 size_t len = text_ptr[i].text_length;
647
648 if (key && (!std::strcmp(key, "Raw profile type exif") || !std::strcmp(key, "Raw profile type APP1"))) {
649 m_exif.processRawProfile(value, len);
650 }
651 else if (key && !std::strcmp(key, "XML:com.adobe.xmp")) {
652 auto& out = m_metadata[IMAGE_METADATA_XMP];
653 out.insert(out.end(),
654 reinterpret_cast<const unsigned char*>(value),
655 reinterpret_cast<const unsigned char*>(value) + std::strlen(value) + 1);
656 }
657 }
658
659 png_charp icc_name;
660 int compression_type;

713void PngDecoder::compose_frame(std::vector<png_bytep>& rows_dst, const std::vector<png_bytep>& rows_src, unsigned char _bop, uint32_t x, uint32_t y, uint32_t w, uint32_t h, Mat& img)
714{
715 const size_t elem_size = img.elemSize();
716 if (_bop == 0) {
717 // Overwrite mode: copy source row directly to destination
718 for (uint32_t j = 0; j < h; ++j) {
719 std::memcpy(rows_dst[j + y] + x * elem_size, rows_src[j], w * elem_size);
720 }
721 return;
722 }
723
724 int channels = img.channels();
725 if (img.depth() == CV_16U)
726 cv::parallel_for_(cv::Range(0, h), [&](const cv::Range& range) {
727 for (int j = range.start; j < range.end; j++) {
728 uint16_t* sp = reinterpret_cast<uint16_t*>(rows_src[j]);
729 uint16_t* dp = reinterpret_cast<uint16_t*>(rows_dst[j + y]) + x * channels;
730
731 // Blending mode
732 for (unsigned int i = 0; i < w; i++, sp += channels, dp += channels) {
733 uint16_t alpha = channels < 4 ? 0 : sp[3];
734
735 if (channels < 4 || alpha == 65535 || dp[3] == 0) {
736 // Fully opaque OR destination fully transparent: direct copy
737 memcpy(dp, sp, elem_size);
738 continue;
739 }
740
741 if (alpha != 0) {
742 // Alpha blending
743 uint64_t u = static_cast<uint64_t>(alpha) * 65535;
744 uint64_t v = static_cast<uint64_t>(65535 - alpha) * dp[3];
745 uint64_t al = u + v;
746
747 dp[0] = static_cast<uint16_t>((sp[0] * u + dp[0] * v) / al); // Red
748 dp[1] = static_cast<uint16_t>((sp[1] * u + dp[1] * v) / al); // Green
749 dp[2] = static_cast<uint16_t>((sp[2] * u + dp[2] * v) / al); // Blue
750 dp[3] = static_cast<uint16_t>(al / 65535); // Alpha
751 }
752 }
753 }
754 });
755 else
756 cv::parallel_for_(cv::Range(0, h), [&](const cv::Range& range) {
757 for (int j = range.start; j < range.end; j++) {
758 unsigned char* sp = rows_src[j];
759 unsigned char* dp = rows_dst[j + y] + x * channels;
760
761 // Blending mode
762 for (unsigned int i = 0; i < w; i++, sp += channels, dp += channels) {
763 uint8_t alpha = channels < 4 ? 0 : sp[3];
764
765 if (channels < 4 || alpha == 255 || dp[3] == 0) {
766 // Fully opaque OR destination fully transparent: direct copy
767 memcpy(dp, sp, elem_size);
768 continue;
769 }
770
771 if (alpha != 0) {
772 // Alpha blending
773 uint32_t u = alpha * 255;
774 uint32_t v = (255 - alpha) * dp[3];
775 uint32_t al = u + v;
776
777 dp[0] = static_cast<uint8_t>((sp[0] * u + dp[0] * v) / al); // Red
778 dp[1] = static_cast<uint8_t>((sp[1] * u + dp[1] * v) / al); // Green
779 dp[2] = static_cast<uint8_t>((sp[2] * u + dp[2] * v) / al); // Blue
780 dp[3] = static_cast<uint8_t>(al / 255); // Alpha
781 }
782 }
783 }
784 });
785}

802uint32_t PngDecoder::read_chunk(Chunk& chunk)
803{
804 unsigned char size_id[8];
805 if (!readFromStreamOrBuffer(&size_id, 8))
806 return 0;
807 const size_t size = static_cast<size_t>(png_get_uint_32(size_id)) + 12;
808
809 const uint32_t id = png_get_uint_32(size_id + 4);
810 if (id == id_IHDR) {
811 // 8=HDR+size, 13=size of IHDR chunk, 4=CRC
812 // http://www.libpng.org/pub/png/spec/1.2/PNG-Chunks.html#C.IHDR
813 if (size != 8 + 13 + 4)
814 return 0;
815 } else if (id == id_acTL) {
816 // 8=HDR+size, 8=size of acTL chunk, 4=CRC
817 // https://wiki.mozilla.org/APNG_Specification#%60acTL%60:_The_Animation_Control_Chunk
818 if (size != 8 + 8 + 4)
819 return 0;
820 } else if (id == id_fcTL) {
821 // 8=HDR+size, 26=size of fcTL chunk, 4=CRC
822 // https://wiki.mozilla.org/APNG_Specification#%60fcTL%60:_The_Frame_Control_Chunk
823 if (size != 8 + 26 + 4)
824 return 0;
825 } else if (id == id_bKGD) {
826 // 8=HDR+size, (1, 2 or 6)=size of bKGD chunk, 4=CRC
827 // The spec is actually more complex:
828 // http://www.libpng.org/pub/png/spec/1.2/PNG-Chunks.html#C.bKGD
829 if (size != 8 + 1 + 4 && size != 8 + 2 + 4 && size != 8 + 6 + 4)
830 return 0;
831 } else if (id != id_fdAT && id != id_IDAT && id != id_IEND && id != id_PLTE && id != id_tEXt && id != id_tRNS) {
832 if (size > PNG_USER_CHUNK_MALLOC_MAX)
833 {
834 CV_LOG_WARNING(NULL, "user chunk data is too large");
835 return 0;
836 }
837 }
838
839 chunk.p.resize(size);
840 memcpy(chunk.p.data(), size_id, 8);
841 if (readFromStreamOrBuffer(&chunk.p[8], chunk.p.size() - 8))
842 return id;
843 return 0;
844}

846bool PngDecoder::processing_start(void* frame_ptr, const Mat& img)
847{
848 if (!InitPngPtr())
849 return false;
850
851 if (setjmp(png_jmpbuf(m_png_ptr)))
852 return false;
853
854 static uint8_t header[8] = { 137, 80, 78, 71, 13, 10, 26, 10 };
855
856 png_set_crc_action(m_png_ptr, PNG_CRC_QUIET_USE, PNG_CRC_QUIET_USE);
857 png_set_progressive_read_fn(m_png_ptr, frame_ptr, (png_progressive_info_ptr)info_fn, row_fn, NULL);
858
859 if (img.channels() < 4)
860 png_set_strip_alpha(m_png_ptr);
861 else
862 png_set_tRNS_to_alpha(m_png_ptr);
863
864 png_process_data(m_png_ptr, m_info_ptr, header, 8);
865 png_process_data(m_png_ptr, m_info_ptr, m_chunkIHDR.p.data(), m_chunkIHDR.p.size());
866
867 if ((m_color_type & PNG_COLOR_MASK_COLOR) && img.channels() > 1 && !m_use_rgb)
868 png_set_bgr(m_png_ptr); // convert RGB to BGR
869 else if (img.channels() > 1)
870 png_set_gray_to_rgb(m_png_ptr); // Gray->RGB
871 else
872 png_set_rgb_to_gray(m_png_ptr, 1, 0.299, 0.587); // RGB->Gray
873
874 if (!isBigEndian() && m_bit_depth == 16)
875 png_set_swap(m_png_ptr);
876
877 for (size_t i = 0; i < m_chunksInfo.size(); i++)
878 png_process_data(m_png_ptr, m_info_ptr, m_chunksInfo[i].p.data(), m_chunksInfo[i].p.size());
879
880 return true;
881}

974bool PngEncoder::write( const Mat& img, const std::vector<int>& params )
975{
976 png_structp png_ptr = png_create_write_struct( PNG_LIBPNG_VER_STRING, 0, 0, 0 );
977 png_infop info_ptr = 0;
978 FILE * volatile f = 0;
979 int y, width = img.cols, height = img.rows;
980 int depth = img.depth(), channels = img.channels();
981 volatile bool result = false;
982 AutoBuffer<uchar*> buffer;
983
984 if( depth != CV_8U && depth != CV_16U )
985 return false;
986
987 if( png_ptr )
988 {
989 info_ptr = png_create_info_struct( png_ptr );
990
991 if( info_ptr )
992 {
993 if( setjmp( png_jmpbuf ( png_ptr ) ) == 0 )
994 {
995 bool set_compression_level = false;
996 bool set_filter = false;
997
998 if( m_buf )
999 {
1000 png_set_write_fn(png_ptr, this,
1001 (png_rw_ptr)writeDataToBuf, (png_flush_ptr)flushBuf);
1002 }
1003 else
1004 {
1005 f = fopen( m_filename.c_str(), "wb" );
1006 if( f )
1007 png_init_io( png_ptr, (png_FILE_p)f );
1008 }
1009
1010 for( size_t i = 0; i < params.size(); i += 2 )
1011 {
1012 const int value = params[i+1];
1013 switch (params[i])
1014 {
1015 case IMWRITE_PNG_COMPRESSION:
1016 m_compression_strategy = IMWRITE_PNG_STRATEGY_DEFAULT; // Default strategy
1017 m_compression_level = MIN(MAX(value, 0), Z_BEST_COMPRESSION);
1018 if(value != m_compression_level) {
1019 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_PNG_COMPRESSION must be between 0 to 9. It is fallbacked to %d", value, m_compression_level));
1020 }
1021 set_compression_level = true;
1022 break;
1023
1024 case IMWRITE_PNG_STRATEGY:
1025 {
1026 switch(value) {
1027 case IMWRITE_PNG_STRATEGY_DEFAULT:
1028 case IMWRITE_PNG_STRATEGY_FILTERED:
1029 case IMWRITE_PNG_STRATEGY_HUFFMAN_ONLY:
1030 case IMWRITE_PNG_STRATEGY_RLE:
1031 case IMWRITE_PNG_STRATEGY_FIXED:
1032 m_compression_strategy = value;
1033 break;
1034 default:
1035 m_compression_strategy = IMWRITE_PNG_STRATEGY_RLE;
1036 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_PNG_STRATEGY must be one of ImwritePNGFlags. It is fallbacked to IMWRITE_PNG_STRATEGY_RLE", value));
1037 break;
1038 }
1039 }
1040 break;
1041
1042 case IMWRITE_PNG_BILEVEL:
1043 m_isBilevel = value != 0;
1044 if((value != 0) && (value != 1)) {
1045 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_PNG_BILEVEL must be 0 or 1. It is fallbacked to 1", value ));
1046 }
1047 break;
1048
1049 case IMWRITE_PNG_FILTER:
1050 {
1051 switch(value) {
1052 case IMWRITE_PNG_FILTER_NONE:
1053 case IMWRITE_PNG_FILTER_SUB:
1054 case IMWRITE_PNG_FILTER_UP:
1055 case IMWRITE_PNG_FILTER_AVG:
1056 case IMWRITE_PNG_FILTER_PAETH:
1057 case IMWRITE_PNG_FAST_FILTERS:
1058 case IMWRITE_PNG_ALL_FILTERS:
1059 m_filter = value;
1060 break;
1061 default:
1062 m_filter = IMWRITE_PNG_FILTER_SUB;
1063 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_PNG_FILTER must be one of ImwritePNGFilterFlags. It is fallbacked to IMWRITE_PNG_FILTER_SUB", value ));
1064 break;
1065 }
1066 }
1067 set_filter = true;
1068 break;
1069
1070 case IMWRITE_PNG_ZLIBBUFFER_SIZE:
1071 // The default value is 8 KiB.
1072 // The minimum limit is 6, which is from from https://github.com/opencv/opencv/blob/4.12.0/3rdparty/libpng/pngset.c#L1600 .
1073 // The maximum limit is 1 MiB, which has been provisionally set. libpng limitation is 2 GiB(INT32_MAX), but it is too large.
1074 // For normal use, 128 or 256 KiB may be sufficient. See https://zlib.net/zlib_how.html .
1075 {
1076 const int zlen = MIN(MAX(value, 6), 1024*1024);
1077 if(value != zlen) {
1078 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_PNG_ZLIBBUFFER_SIZE must be between 6 to 1024*1024. It is fallbacked to %d", value , zlen));
1079 }
1080 png_set_compression_buffer_size(png_ptr, zlen);
1081 }
1082 break;
1083
1084 default:
1085 break;
1086 }
1087 }
1088
1089 if( m_buf || f )
1090 {
1091 if (!set_compression_level || set_filter)
1092 png_set_filter(png_ptr, PNG_FILTER_TYPE_BASE, m_filter);
1093 png_set_compression_level(png_ptr, m_compression_level);
1094 png_set_compression_strategy(png_ptr, m_compression_strategy);
1095
1096 png_set_IHDR( png_ptr, info_ptr, width, height, depth == CV_8U ? m_isBilevel ? 1 : 8 : 16,
1097 channels == 1 ? PNG_COLOR_TYPE_GRAY :
1098 channels == 3 ? PNG_COLOR_TYPE_RGB : PNG_COLOR_TYPE_RGBA,
1099 PNG_INTERLACE_NONE, PNG_COMPRESSION_TYPE_DEFAULT,
1100 PNG_FILTER_TYPE_DEFAULT );
1101
1102 if (!m_metadata.empty()) {
1103 std::vector<uchar>& exif = m_metadata[IMAGE_METADATA_EXIF];
1104 if (!exif.empty()) {
1105#ifdef PNG_eXIf_SUPPORTED
1106 png_set_eXIf_1(png_ptr, info_ptr, static_cast<png_uint_32>(exif.size()), exif.data());
1107#else
1108 CV_LOG_WARNING(NULL, "Libpng is too old and does not support EXIF.");
1109#endif
1110 }
1111
1112 std::vector<uchar>& xmp = m_metadata[IMAGE_METADATA_XMP];
1113 if (!xmp.empty()) {
1114 png_text text_chunk;
1115 text_chunk.compression = PNG_TEXT_COMPRESSION_NONE;
1116 text_chunk.key = const_cast<char*>("XML:com.adobe.xmp");
1117 text_chunk.text = reinterpret_cast<char*>(xmp.data());
1118 text_chunk.text_length = static_cast<png_size_t>(xmp.size());
1119
1120 png_set_text(png_ptr, info_ptr, &text_chunk, 1);
1121 }
1122
1123 std::vector<uchar> iccp = m_metadata[IMAGE_METADATA_ICCP];
1124 if (!iccp.empty()) {
1125 // PNG standard requires a profile name (null-terminated, max 79 characters, printable Latin-1)
1126 char iccp_profile_name[] = "ICC Profile";
1127
1128 // Compression type must be 0 (deflate) as per libpng docs
1129 int compression_type = PNG_COMPRESSION_TYPE_BASE;
1130
1131 // Some ICC profiles are already compressed (e.g., if saved from Photoshop),
1132 // but png_set_iCCP still expects uncompressed input, and it compresses it internally.
1133
1134 png_set_iCCP(png_ptr, info_ptr,
1135 iccp_profile_name,
1136 compression_type,
1137#if (PNG_LIBPNG_VER_MAJOR*10000 + PNG_LIBPNG_VER_MINOR*100 + PNG_LIBPNG_VER_RELEASE >= 10500)
1138 reinterpret_cast<png_const_bytep>(iccp.data()),
1139#else
1140 reinterpret_cast<png_charp>(iccp.data()),
1141#endif
1142 static_cast<png_uint_32>(iccp.size()));
1143 }
1144
1145 std::vector<uchar>& cicp = m_metadata[IMAGE_METADATA_CICP];
1146 if (!cicp.empty()) {
1147#ifdef PNG_cICP_SUPPORTED
1148 CV_CheckEQ((size_t)4, cicp.size(), "The cICP chunk consists of four 1-byte unsigned integers");
1149 png_set_cICP(png_ptr, info_ptr, cicp[0], cicp[1], cicp[2], cicp[3]);
1150#else
1151 CV_LOG_WARNING(NULL, "Libpng is too old and does not support cICP.");
1152#endif
1153 }
1154 }
1155
1156 png_write_info( png_ptr, info_ptr );
1157
1158 if (m_isBilevel)
1159 png_set_packing(png_ptr);
1160
1161 png_set_bgr( png_ptr );
1162 if( !isBigEndian() )
1163 png_set_swap( png_ptr );
1164
1165 buffer.allocate(height);
1166 for( y = 0; y < height; y++ )
1167 buffer[y] = img.data + y*img.step;
1168
1169 png_write_image( png_ptr, buffer.data() );
1170 png_write_end( png_ptr, info_ptr );
1171
1172 result = true;
1173 }
1174 }
1175 }
1176 }
1177
1178 png_destroy_write_struct( &png_ptr, &info_ptr );
1179 if(f) fclose( (FILE*)f );
1180
1181 return result;
1182}

1204void PngEncoder::writeChunk(FILE* f, const char* name, unsigned char* data, uint32_t length)
1205{
1206 unsigned char buf[4];
1207 uint32_t crc = crc32(0, Z_NULL, 0);
1208
1209 png_save_uint_32(buf, length);
1210 writeToStreamOrBuffer(buf, 4, f);
1211 writeToStreamOrBuffer(name, 4, f);
1212 crc = crc32(crc, (const Bytef*)name, 4);
1213
1214 if (memcmp(name, "fdAT", 4) == 0)
1215 {
1216 png_save_uint_32(buf, next_seq_num++);
1217 writeToStreamOrBuffer(buf, 4, f);
1218 crc = crc32(crc, buf, 4);
1219 length -= 4;
1220 }
1221
1222 if (data != NULL && length > 0)
1223 {
1224 writeToStreamOrBuffer(data, length, f);
1225 crc = crc32(crc, data, length);
1226 }
1227
1228 png_save_uint_32(buf, crc);
1229 writeToStreamOrBuffer(buf, 4, f);
1230}

1232void PngEncoder::writeIDATs(FILE* f, int frame, unsigned char* data, uint32_t length, uint32_t idat_size)
1233{
1234 uint32_t z_cmf = data[0];
1235 if ((z_cmf & 0x0f) == 8 && (z_cmf & 0xf0) <= 0x70)
1236 {
1237 if (length >= 2)
1238 {
1239 uint32_t z_cinfo = z_cmf >> 4;
1240 uint32_t half_z_window_size = 1 << (z_cinfo + 7);
1241 while (idat_size <= half_z_window_size && half_z_window_size >= 256)
1242 {
1243 z_cinfo--;
1244 half_z_window_size >>= 1;
1245 }
1246 z_cmf = (z_cmf & 0x0f) | (z_cinfo << 4);
1247 if (data[0] != (unsigned char)z_cmf)
1248 {
1249 data[0] = (unsigned char)z_cmf;
1250 data[1] &= 0xe0;
1251 data[1] += (unsigned char)(0x1f - ((z_cmf << 8) + data[1]) % 0x1f);
1252 }
1253 }
1254 }
1255
1256 while (length > 0)
1257 {
1258 uint32_t ds = length;
1259 if (ds > 32768)
1260 ds = 32768;
1261
1262 if (frame == 0)
1263 writeChunk(f, "IDAT", data, ds);
1264 else
1265 writeChunk(f, "fdAT", data, ds + 4);
1266
1267 data += ds;
1268 length -= ds;
1269 }

1272void PngEncoder::processRect(unsigned char* row, int rowbytes, int bpp, int stride, int h, unsigned char* rows)
1273{
1274 int i, j, v;
1275 int a, b, c, pa, pb, pc, p;
1276 unsigned char* prev = NULL;
1277 unsigned char* dp = rows;
1278 unsigned char* out;
1279
1280 for (j = 0; j < h; j++)
1281 {
1282 uint32_t sum = 0;
1283 unsigned char* best_row = row_buf.data();
1284 uint32_t mins = ((uint32_t)(-1)) >> 1;
1285
1286 out = row_buf.data() + 1;
1287 for (i = 0; i < rowbytes; i++)
1288 {
1289 v = out[i] = row[i];
1290 sum += (v < 128) ? v : 256 - v;
1291 }
1292 mins = sum;
1293
1294 sum = 0;
1295 out = sub_row.data() + 1;
1296 for (i = 0; i < bpp; i++)
1297 {
1298 v = out[i] = row[i];
1299 sum += (v < 128) ? v : 256 - v;
1300 }
1301 for (i = bpp; i < rowbytes; i++)
1302 {
1303 v = out[i] = row[i] - row[i - bpp];
1304 sum += (v < 128) ? v : 256 - v;
1305 if (sum > mins)
1306 break;
1307 }
1308 if (sum < mins)
1309 {
1310 mins = sum;
1311 best_row = sub_row.data();
1312 }
1313
1314 if (prev)
1315 {
1316 sum = 0;
1317 out = up_row.data() + 1;
1318 for (i = 0; i < rowbytes; i++)
1319 {
1320 v = out[i] = row[i] - prev[i];
1321 sum += (v < 128) ? v : 256 - v;
1322 if (sum > mins)
1323 break;
1324 }
1325 if (sum < mins)
1326 {
1327 mins = sum;
1328 best_row = up_row.data();
1329 }
1330
1331 sum = 0;
1332 out = avg_row.data() + 1;
1333 for (i = 0; i < bpp; i++)
1334 {
1335 v = out[i] = row[i] - prev[i] / 2;
1336 sum += (v < 128) ? v : 256 - v;
1337 }
1338 for (i = bpp; i < rowbytes; i++)
1339 {
1340 v = out[i] = row[i] - (prev[i] + row[i - bpp]) / 2;
1341 sum += (v < 128) ? v : 256 - v;
1342 if (sum > mins)
1343 break;
1344 }
1345 if (sum < mins)
1346 {
1347 mins = sum;
1348 best_row = avg_row.data();
1349 }
1350
1351 sum = 0;
1352 out = paeth_row.data() + 1;
1353 for (i = 0; i < bpp; i++)
1354 {
1355 v = out[i] = row[i] - prev[i];
1356 sum += (v < 128) ? v : 256 - v;
1357 }
1358 for (i = bpp; i < rowbytes; i++)
1359 {
1360 a = row[i - bpp];
1361 b = prev[i];
1362 c = prev[i - bpp];
1363 p = b - c;
1364 pc = a - c;
1365 pa = abs(p);
1366 pb = abs(pc);
1367 pc = abs(p + pc);
1368 p = (pa <= pb && pa <= pc) ? a : (pb <= pc) ? b
1369 : c;
1370 v = out[i] = row[i] - p;
1371 sum += (v < 128) ? v : 256 - v;
1372 if (sum > mins)
1373 break;
1374 }
1375 if (sum < mins)
1376 {
1377 best_row = paeth_row.data();
1378 }
1379 }
1380
1381 if (rows == NULL)
1382 {
1383 // deflate_rect_op()
1384 op_zstream1.next_in = row_buf.data();
1385 op_zstream1.avail_in = rowbytes + 1;
1386 deflate(&op_zstream1, Z_NO_FLUSH);
1387
1388 op_zstream2.next_in = best_row;
1389 op_zstream2.avail_in = rowbytes + 1;
1390 deflate(&op_zstream2, Z_NO_FLUSH);
1391 }
1392 else
1393 {
1394 // deflate_rect_fin()
1395 memcpy(dp, best_row, rowbytes + 1);
1396 dp += rowbytes + 1;
1397 }
1398
1399 prev = row;
1400 row += stride;
1401 }
1402}

1442bool PngEncoder::getRect(uint32_t w, uint32_t h, unsigned char* pimage1, unsigned char* pimage2, unsigned char* ptemp, uint32_t bpp, uint32_t stride, int zbuf_size, uint32_t has_tcolor, uint32_t tcolor, int n)
1443{
1444 uint32_t i, j, x0, y0, w0, h0;
1445 uint32_t x_min = w - 1;
1446 uint32_t y_min = h - 1;
1447 uint32_t x_max = 0;
1448 uint32_t y_max = 0;
1449 uint32_t diffnum = 0;
1450 uint32_t over_is_possible = 1;
1451
1452 if (!has_tcolor)
1453 over_is_possible = 0;
1454
1455 if (bpp == 1)
1456 {
1457 unsigned char* pa = pimage1;
1458 unsigned char* pb = pimage2;
1459 unsigned char* pc = ptemp;
1460
1461 for (j = 0; j < h; j++)
1462 for (i = 0; i < w; i++)
1463 {
1464 unsigned char c = *pb++;
1465 if (*pa++ != c)
1466 {
1467 diffnum++;
1468 if (has_tcolor && c == tcolor)
1469 over_is_possible = 0;
1470 if (i < x_min)
1471 x_min = i;
1472 if (i > x_max)
1473 x_max = i;
1474 if (j < y_min)
1475 y_min = j;
1476 if (j > y_max)
1477 y_max = j;
1478 }
1479 else
1480 c = tcolor;
1481
1482 *pc++ = c;
1483 }
1484 }
1485 else if (bpp == 2)
1486 {
1487 unsigned short* pa = (unsigned short*)pimage1;
1488 unsigned short* pb = (unsigned short*)pimage2;
1489 unsigned short* pc = (unsigned short*)ptemp;
1490
1491 for (j = 0; j < h; j++)
1492 for (i = 0; i < w; i++)
1493 {
1494 uint32_t c1 = *pa++;
1495 uint32_t c2 = *pb++;
1496 if ((c1 != c2) && ((c1 >> 8) || (c2 >> 8)))
1497 {
1498 diffnum++;
1499 if ((c2 >> 8) != 0xFF)
1500 over_is_possible = 0;
1501 if (i < x_min)
1502 x_min = i;
1503 if (i > x_max)
1504 x_max = i;
1505 if (j < y_min)
1506 y_min = j;
1507 if (j > y_max)

1515 }
1516 else if (bpp == 3)
1517 {
1518 unsigned char* pa = pimage1;
1519 unsigned char* pb = pimage2;
1520 unsigned char* pc = ptemp;
1521
1522 for (j = 0; j < h; j++)
1523 for (i = 0; i < w; i++)
1524 {
1525 uint32_t c1 = (pa[2] << 16) + (pa[1] << 8) + pa[0];
1526 uint32_t c2 = (pb[2] << 16) + (pb[1] << 8) + pb[0];
1527 if (c1 != c2)
1528 {
1529 diffnum++;
1530 if (has_tcolor && c2 == tcolor)
1531 over_is_possible = 0;
1532 if (i < x_min)
1533 x_min = i;
1534 if (i > x_max)
1535 x_max = i;
1536 if (j < y_min)
1537 y_min = j;
1538 if (j > y_max)
1539 y_max = j;
1540 }
1541 else
1542 c2 = tcolor;
1543
1544 memcpy(pc, &c2, 3);
1545 pa += 3;
1546 pb += 3;
1547 pc += 3;
1548 }
1549 }
1550 else if (bpp == 4)
1551 {
1552 uint32_t* pa = (uint32_t*)pimage1;
1553 uint32_t* pb = (uint32_t*)pimage2;
1554 uint32_t* pc = (uint32_t*)ptemp;
1555
1556 for (j = 0; j < h; j++)
1557 for (i = 0; i < w; i++)
1558 {
1559 uint32_t c1 = *pa++;
1560 uint32_t c2 = *pb++;
1561 if ((c1 != c2) && ((c1 >> 24) || (c2 >> 24)))
1562 {
1563 diffnum++;
1564 if ((c2 >> 24) != 0xFF)
1565 over_is_possible = 0;
1566 if (i < x_min)
1567 x_min = i;
1568 if (i > x_max)
1569 x_max = i;
1570 if (j < y_min)
1571 y_min = j;
1572 if (j > y_max)
1573 y_max = j;
1574 }
1575 else
1576 c2 = 0;
1577
1578 *pc++ = c2;
1579 }
1580 }
1581
1582 if (diffnum == 0)
1583 {
1584 return false;
1585 }
1586 else
1587 {
1588 x0 = x_min;
1589 y0 = y_min;
1590 w0 = x_max - x_min + 1;
1591 h0 = y_max - y_min + 1;
1592 }
1593
1594 if (n < 3)
1595 {
1596 deflateRectOp(pimage2, x0, y0, w0, h0, bpp, stride, zbuf_size, n * 2);
1597
1598 if (over_is_possible)
1599 deflateRectOp(ptemp, x0, y0, w0, h0, bpp, stride, zbuf_size, n * 2 + 1);
1600 }
1601
1602 return true;
1603}
