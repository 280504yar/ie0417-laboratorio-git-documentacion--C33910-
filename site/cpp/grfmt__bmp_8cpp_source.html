81bool BmpDecoder::readHeader()
82{
83 bool result = false;
84 bool iscolor = false;
85
86 if( !m_buf.empty() )
87 {
88 if( !m_strm.open( m_buf ) )
89 return false;
90 }
91 else if( !m_strm.open( m_filename ))
92 return false;
93
94 try
95 {
96 m_strm.skip( 10 );
97 m_offset = m_strm.getDWord();
98
99 int size = m_strm.getDWord();
100 CV_Assert(size > 0); // overflow, 2Gb limit
101
102 initMask();
103 if( size >= 36 )
104 {
105 m_width = m_strm.getDWord();
106 m_height = m_strm.getDWord();
107 m_bpp = m_strm.getDWord() >> 16;
108 int m_rle_code_ = m_strm.getDWord();
109 CV_Assert(m_rle_code_ >= 0 && m_rle_code_ <= BMP_BITFIELDS);
110 m_rle_code = (BmpCompression)m_rle_code_;
111 m_strm.skip(12);
112 int clrused = m_strm.getDWord();
113
114 if( m_bpp == 32 && m_rle_code == BMP_BITFIELDS && size >= 56 )
115 {
116 m_strm.skip(4); //important colors
117 //0 is Red channel bit mask, 1 is Green channel bit mask, 2 is Blue channel bit mask, 3 is Alpha channel bit mask
118 for( int index_rgba = 0; index_rgba < 4; ++index_rgba )
119 {
120 uint mask = m_strm.getDWord();
121 m_rgba_mask[index_rgba] = mask;
122 if(mask != 0)
123 {
124 int bit_count = 0;
125 while(!(mask & 1))
126 {
127 mask >>= 1;
128 ++bit_count;
129 }
130 m_rgba_bit_offset[index_rgba] = bit_count;
131 m_rgba_scale_factor[index_rgba] = 255.0f / mask;
132 }
133 }
134 m_strm.skip( size - 56 );
135 }
136 else
137 m_strm.skip( size - 36 );
138
139 if( m_width > 0 && m_height != 0 &&
140 (((m_bpp == 1 || m_bpp == 4 || m_bpp == 8 ||
141 m_bpp == 24 || m_bpp == 32 ) && m_rle_code == BMP_RGB) ||
142 ((m_bpp == 16 || m_bpp == 32) && (m_rle_code == BMP_RGB || m_rle_code == BMP_BITFIELDS)) ||
143 (m_bpp == 4 && m_rle_code == BMP_RLE4) ||
144 (m_bpp == 8 && m_rle_code == BMP_RLE8)))
145 {
146 iscolor = true;
147 result = true;
148
149 if( m_bpp <= 8 )
150 {
151 CV_Assert(clrused >= 0 && clrused <= 256);
152 memset(m_palette, 0, sizeof(m_palette));
153 m_strm.getBytes(m_palette, (clrused == 0? 1<<m_bpp : clrused)*4 );
154 iscolor = IsColorPalette( m_palette, m_bpp );
155 }
156 else if( m_bpp == 16 && m_rle_code == BMP_BITFIELDS )
157 {
158 int redmask = m_strm.getDWord();
159 int greenmask = m_strm.getDWord();
160 int bluemask = m_strm.getDWord();
161
162 if( bluemask == 0x1f && greenmask == 0x3e0 && redmask == 0x7c00 )
163 m_bpp = 15;
164 else if( bluemask == 0x1f && greenmask == 0x7e0 && redmask == 0xf800 )
165 ;
166 else
167 result = false;
168 }
169 else if (m_bpp == 32 && m_rle_code == BMP_BITFIELDS)
170 {
171 // 32bit BMP not require to check something - we can simply allow it to use
172 ;
173 }
174 else if( m_bpp == 16 && m_rle_code == BMP_RGB )
175 m_bpp = 15;
176 }
177 }
178 else if( size == 12 )
179 {
180 m_width = m_strm.getWord();
181 m_height = m_strm.getWord();
182 m_bpp = m_strm.getDWord() >> 16;
183 m_rle_code = BMP_RGB;
184
185 if( m_width > 0 && m_height != 0 &&
186 (m_bpp == 1 || m_bpp == 4 || m_bpp == 8 ||
187 m_bpp == 24 || m_bpp == 32 ))
188 {
189 if( m_bpp <= 8 )
190 {
191 uchar buffer[256*3];
192 int j, clrused = 1 << m_bpp;
193 m_strm.getBytes( buffer, clrused*3 );
194 for( j = 0; j < clrused; j++ )
195 {
196 m_palette[j].b = buffer[3*j+0];
197 m_palette[j].g = buffer[3*j+1];
198 m_palette[j].r = buffer[3*j+2];
199 }
200 }
201 result = true;
202 }
203 }
204 }
205 catch(...)
206 {
207 throw;
208 }
209 // in 32 bit case alpha channel is used - so require CV_8UC4 type
210 m_type = iscolor ? ((m_bpp == 32 && m_rle_code != BMP_RGB) ? CV_8UC4 : CV_8UC3 ) : CV_8UC1;
211 m_origin = m_height > 0 ? ORIGIN_BL : ORIGIN_TL;
212 if ( m_height == std::numeric_limits<int>::min() ) {
213 // abs(std::numeric_limits<int>::min()) is undefined behavior.
214 result = false;
215 } else {
216 m_height = std::abs(m_height);
217 }
218
219 if( !result )
220 {
221 m_offset = -1;
222 m_width = m_height = -1;
223 m_strm.close();
224 }
225 return result;
226}

229bool BmpDecoder::readData( Mat& img )
230{
231 uchar* data = img.ptr();
232 int step = validateToInt(img.step);
233 bool color = img.channels() > 1;
234 uchar gray_palette[256] = {0};
235 bool result = false;
236 int nch = color ? 3 : 1;
237
238 const int effective_bpp = (m_bpp != 15) ? m_bpp : 16;
239 const RowPitchParams pitch_params = calculateRowPitch(m_width, effective_bpp, 4, "BMP");
240 const int src_pitch = pitch_params.src_pitch;
241 const int width3 = calculateRowSize(m_width, nch, "BMP");
242 int y;
243
244 if( m_offset < 0 || !m_strm.isOpened())
245 return false;
246
247 if( m_origin == ORIGIN_BL )
248 {
249 data += (m_height - 1)*(size_t)step;
250 step = -step;
251 }
252
253 AutoBuffer<uchar> _src, _bgr;
254 _src.allocate(src_pitch + 32);
255
256 if( !color )
257 {
258 if( m_bpp <= 8 )
259 {
260 CvtPaletteToGray( m_palette, gray_palette, 1 << m_bpp );
261 }
262 const size_t bgr_size = static_cast<size_t>(m_width) * 3 + 32;
263 CV_CheckLT(bgr_size, MAX_IMAGE_ROW_SIZE, "BMP: bgr buffer size exceeds maximum allowed size");
264 _bgr.allocate(bgr_size);
265 }
266 uchar *src = _src.data(), *bgr = _bgr.data();
267
268 try
269 {
270 m_strm.setPos( m_offset );
271
272 switch( m_bpp )
273 {
274 /************************* 1 BPP ************************/
275 case 1:
276 for( y = 0; y < m_height; y++, data += step )
277 {
278 m_strm.getBytes( src, src_pitch );
279 FillColorRow1( color ? data : bgr, src, m_width, m_palette );
280 if( !color )
281 icvCvt_BGR2Gray_8u_C3C1R( bgr, 0, data, 0, Size(m_width,1) );
282 }
283 result = true;
284 break;
285
286 /************************* 4 BPP ************************/
287 case 4:
288 if( m_rle_code == BMP_RGB )
289 {
290 for( y = 0; y < m_height; y++, data += step )
291 {
292 m_strm.getBytes( src, src_pitch );
293 if( color )
294 FillColorRow4( data, src, m_width, m_palette );
295 else
296 FillGrayRow4( data, src, m_width, gray_palette );
297 }
298 result = true;
299 }
300 else if( m_rle_code == BMP_RLE4 ) // rle4 compression
301 {
302 uchar* line_end = data + width3;
303 y = 0;
304
305 for(;;)
306 {
307 int code = m_strm.getWord();
308 const int len = code & 255;
309 code >>= 8;
310 if( len != 0 ) // encoded mode
311 {
312 PaletteEntry clr[2];
313 uchar gray_clr[2];
314 int t = 0;
315
316 clr[0] = m_palette[code >> 4];
317 clr[1] = m_palette[code & 15];
318 gray_clr[0] = gray_palette[code >> 4];
319 gray_clr[1] = gray_palette[code & 15];
320
321 uchar* end = data + len*nch;
322 if( end > line_end ) goto decode_rle4_bad;
323 do
324 {
325 if( color )
326 WRITE_PIX( data, clr[t] );
327 else
328 *data = gray_clr[t];
329 t ^= 1;
330 }
331 while( (data += nch) < end );
332 }
333 else if( code > 2 ) // absolute mode
334 {
335 if( data + code*nch > line_end ) goto decode_rle4_bad;
336 int sz = (((code + 1)>>1) + 1) & (~1);
337 CV_Assert((size_t)sz < _src.size());
338 m_strm.getBytes(src, sz);
339 if( color )
340 data = FillColorRow4( data, src, code, m_palette );
341 else
342 data = FillGrayRow4( data, src, code, gray_palette );
343 }
344 else
345 {
346 ptrdiff_t x_shift3 = line_end - data;
347
348 if( code == 2 )
349 {
350 x_shift3 = m_strm.getByte()*nch;
351 m_strm.getByte();
352 }
353
354 if( color )
355 data = FillUniColor( data, line_end, step, width3,
356 y, m_height, x_shift3,
357 m_palette[0] );
358 else
359 data = FillUniGray( data, line_end, step, width3,
360 y, m_height, x_shift3,
361 gray_palette[0] );
362
363 if( y >= m_height )
364 break;
365 }
366 }
367
368 result = true;
369decode_rle4_bad: ;
370 }
371 break;
372
373 /************************* 8 BPP ************************/
374 case 8:
375 if( m_rle_code == BMP_RGB )
376 {
377 for( y = 0; y < m_height; y++, data += step )
378 {
379 m_strm.getBytes( src, src_pitch );
380 if( color )
381 FillColorRow8( data, src, m_width, m_palette );
382 else
383 FillGrayRow8( data, src, m_width, gray_palette );
384 }
385 result = true;
386 }
387 else if( m_rle_code == BMP_RLE8 ) // rle8 compression
388 {
389 uchar* line_end = data + width3;
390 int line_end_flag = 0;
391 y = 0;
392
393 for(;;)
394 {
395 int code = m_strm.getWord();
396 int len = code & 255;
397 code >>= 8;
398 if( len != 0 ) // encoded mode
399 {
400 int prev_y = y;
401 len *= nch;
402
403 if( data + len > line_end )
404 goto decode_rle8_bad;
405
406 if( color )
407 data = FillUniColor( data, line_end, step, width3,
408 y, m_height, len,
409 m_palette[code] );
410 else
411 data = FillUniGray( data, line_end, step, width3,
412 y, m_height, len,
413 gray_palette[code] );
414
415 line_end_flag = y - prev_y;
416
417 if( y >= m_height )
418 break;
419 }
420 else if( code > 2 ) // absolute mode
421 {
422 int prev_y = y;
423 int code3 = code*nch;
424
425 if( data + code3 > line_end )
426 goto decode_rle8_bad;
427 int sz = (code + 1) & (~1);
428 CV_Assert((size_t)sz < _src.size());
429 m_strm.getBytes(src, sz);
430 if( color )
431 data = FillColorRow8( data, src, code, m_palette );
432 else
433 data = FillGrayRow8( data, src, code, gray_palette );
434
435 line_end_flag = y - prev_y;
436 }
437 else
438 {
439 ptrdiff_t x_shift3 = line_end - data;
440 int y_shift = m_height - y;
441
442 if( code || !line_end_flag || x_shift3 < width3 )
443 {
444 if( code == 2 )
445 {
446 x_shift3 = m_strm.getByte()*nch;
447 y_shift = m_strm.getByte();
448 }
449
450 x_shift3 += ((ptrdiff_t)y_shift * width3) & ((code == 0) - 1);
451
452 if( y >= m_height )
453 break;
454
455 if( color )
456 data = FillUniColor( data, line_end, step, width3,
457 y, m_height, x_shift3,
458 m_palette[0] );
459 else
460 data = FillUniGray( data, line_end, step, width3,
461 y, m_height, x_shift3,
462 gray_palette[0] );
463
464 if( y >= m_height )
465 break;
466 }
467
468 line_end_flag = 0;
469 if( y >= m_height )
470 break;
471 }
472 }
473
474 result = true;
475decode_rle8_bad: ;
476 }
477 break;
478 /************************* 15 BPP ************************/
479 case 15:
480 for( y = 0; y < m_height; y++, data += step )
481 {
482 m_strm.getBytes( src, src_pitch );
483 if( !color )
484 icvCvt_BGR5552Gray_8u_C2C1R( src, 0, data, 0, Size(m_width,1) );
485 else
486 icvCvt_BGR5552BGR_8u_C2C3R( src, 0, data, 0, Size(m_width,1) );
487 }
488 result = true;
489 break;
490 /************************* 16 BPP ************************/
491 case 16:
492 for( y = 0; y < m_height; y++, data += step )
493 {
494 m_strm.getBytes( src, src_pitch );
495 if( !color )
496 icvCvt_BGR5652Gray_8u_C2C1R( src, 0, data, 0, Size(m_width,1) );
497 else
498 icvCvt_BGR5652BGR_8u_C2C3R( src, 0, data, 0, Size(m_width,1) );
499 }
500 result = true;
501 break;
502 /************************* 24 BPP ************************/
503 case 24:
504 for( y = 0; y < m_height; y++, data += step )
505 {
506 m_strm.getBytes( src, src_pitch );
507 if(!color)
508 icvCvt_BGR2Gray_8u_C3C1R( src, 0, data, 0, Size(m_width,1) );
509 else
510 memcpy( data, src, m_width*3 );
511 }
512 result = true;
513 break;
514 /************************* 32 BPP ************************/
515 case 32:
516 {
517 bool has_bit_mask = (m_rgba_bit_offset[0] >= 0) && (m_rgba_bit_offset[1] >= 0) && (m_rgba_bit_offset[2] >= 0);
518 for( y = 0; y < m_height; y++, data += step )
519 {
520 m_strm.getBytes( src, src_pitch );
521
522 if( !color )
523 {
524 if ( has_bit_mask )
525 maskBGRAtoGray(data, src, m_width);
526 else
527 icvCvt_BGRA2Gray_8u_C4C1R( src, 0, data, 0, Size(m_width,1) );
528 }
529 else if( img.channels() == 3 )
530 {
531 if ( has_bit_mask )
532 maskBGRA(data, src, m_width, false);
533 else
534 icvCvt_BGRA2BGR_8u_C4C3R(src, 0, data, 0, Size(m_width, 1));
535 }
536 else if ( img.channels() == 4 )
537 {
538 if ( has_bit_mask )
539 maskBGRA(data, src, m_width, true);
540 else
541 memcpy(data, src, m_width * 4);
542 }
543 }
544 }
545 result = true;
546 break;
547 default:
548 CV_Error(cv::Error::StsError, "Invalid/unsupported mode");
549 }
550 }
551 catch(...)
552 {
553 throw;
554 }
555
556 if (m_use_rgb && color && img.channels() == 3)
557 {
558 cv::cvtColor(img, img, cv::COLOR_BGR2RGB);
559 }
560
561 return result;
562}

564void BmpDecoder::initMask()
565{
566 memset(m_rgba_mask, 0, sizeof(m_rgba_mask));
567 memset(m_rgba_bit_offset, -1, sizeof(m_rgba_bit_offset));
568 for (size_t i = 0; i < 4; i++) {
569 m_rgba_scale_factor[i] = 1.0f;
570 }

573void BmpDecoder::maskBGRA(uchar* des, const uchar* src, int num, bool alpha_required)
574{
575 int dest_stride = alpha_required ? 4 : 3;
576 for( int i = 0; i < num; i++, des += dest_stride, src += 4 )
577 {
578 uint data = *((uint*)src);
579 des[0] = (uchar)(((m_rgba_mask[2] & data) >> m_rgba_bit_offset[2]) * m_rgba_scale_factor[2]);
580 des[1] = (uchar)(((m_rgba_mask[1] & data) >> m_rgba_bit_offset[1]) * m_rgba_scale_factor[1]);
581 des[2] = (uchar)(((m_rgba_mask[0] & data) >> m_rgba_bit_offset[0]) * m_rgba_scale_factor[0]);
582 if (alpha_required)
583 {
584 if (m_rgba_bit_offset[3] >= 0)
585 des[3] = (uchar)(((m_rgba_mask[3] & data) >> m_rgba_bit_offset[3]) * m_rgba_scale_factor[3]);
586 else
587 des[3] = 255;
588 }
589 }
590}
591
592void BmpDecoder::maskBGRAtoGray(uchar* des, const uchar* src, int num)
593{
594 for( int i = 0; i < num; i++, des++, src += 4 )
595 {
596 uint data = *((uint*)src);
597 int red = (uchar)(((m_rgba_mask[0] & data) >> m_rgba_bit_offset[0]) * m_rgba_scale_factor[0]);
598 int green = (uchar)(((m_rgba_mask[1] & data) >> m_rgba_bit_offset[1]) * m_rgba_scale_factor[1]);
599 int blue = (uchar)(((m_rgba_mask[2] & data) >> m_rgba_bit_offset[2]) * m_rgba_scale_factor[2]);
600
601 *des = (uchar)(0.299f * red + 0.587f * green + 0.114f * blue);
602 }
603}

623bool BmpEncoder::write( const Mat& img, const std::vector<int>& params )
624{
625 int width = img.cols, height = img.rows, channels = img.channels();
626 int fileStep = (width*channels + 3) & -4;
627 uchar zeropad[] = "\0\0\0\0";
628
629 WLByteStream strm;
630
631 if( m_buf )
632 {
633 if( !strm.open( *m_buf ) )
634 return false;
635 }
636 else if( !strm.open( m_filename ))
637 return false;
638
639 // sRGB colorspace requires BITMAPV5HEADER.
640 // See https://learn.microsoft.com/en-us/windows/win32/api/wingdi/ns-wingdi-bitmapv5header
641 bool useV5BitFields = true;
642 for(size_t i = 0; i < params.size(); i+=2)
643 {
644 const int value = params[i+1];
645 switch(params[i])
646 {
647 case IMWRITE_BMP_COMPRESSION:
648 {
649 switch(value) {
650 case IMWRITE_BMP_COMPRESSION_RGB:
651 useV5BitFields = false;
652 break;
653 case IMWRITE_BMP_COMPRESSION_BITFIELDS:
654 useV5BitFields = true;
655 break;
656 default:
657 useV5BitFields = true;
658 CV_LOG_WARNING(nullptr, cv::format("The value(%d) for IMWRITE_BMP_COMPRESSION must be one of ImwriteBMPCompressionFlags. It is fallbacked to true", value));
659 break;
660 }
661 }
662 break;
663 }
664 }
665 useV5BitFields &= (channels == 4); // BMP_BITFIELDS requires 32 bit per pixel.
666
667 int bitmapHeaderSize = useV5BitFields ? 124 : 40;
668 int paletteSize = channels > 1 ? 0 : 1024;
669 int headerSize = 14 /* fileheader */ + bitmapHeaderSize + paletteSize;
670 size_t fileSize = (size_t)fileStep*height + headerSize;
671 PaletteEntry palette[256];
672
673 if( m_buf )
674 m_buf->reserve( alignSize(fileSize + 16, 256) );
675
676 // write signature 'BM'
677 CHECK_WRITE(strm.putBytes( fmtSignBmp, (int)strlen(fmtSignBmp) ));
678
679 // write file header
680 CHECK_WRITE(strm.putDWord( validateToInt(fileSize) )); // file size
681 CHECK_WRITE(strm.putDWord( 0 ));
682 CHECK_WRITE(strm.putDWord( headerSize ));
683
684 // write bitmap header
685 CHECK_WRITE(strm.putDWord( bitmapHeaderSize ));
686 CHECK_WRITE(strm.putDWord( width ));
687 CHECK_WRITE(strm.putDWord( height ));
688 CHECK_WRITE(strm.putWord( 1 ));
689 CHECK_WRITE(strm.putWord( channels << 3 ));
690 CHECK_WRITE(strm.putDWord( useV5BitFields ? BMP_BITFIELDS : BMP_RGB ));
691 CHECK_WRITE(strm.putDWord( 0 ));
692 CHECK_WRITE(strm.putDWord( 0 ));
693 CHECK_WRITE(strm.putDWord( 0 ));
694 CHECK_WRITE(strm.putDWord( 0 ));
695 CHECK_WRITE(strm.putDWord( 0 ));
696
697 if( useV5BitFields )
698 {
699 CHECK_WRITE(strm.putDWord( 0x00FF0000 )); // bV5RedMask
700 CHECK_WRITE(strm.putDWord( 0x0000FF00 )); // bV5GreenMask
