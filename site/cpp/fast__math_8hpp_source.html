199inline int
200cvRound( double value )
201{
202#if defined CV_INLINE_ROUND_DBL
203 CV_INLINE_ROUND_DBL(value);
204#elif defined(_MSC_VER) && (defined(_M_ARM64) || defined(_M_ARM64EC))
205 float64x1_t v = vdup_n_f64(value);
206 int64x1_t r = vcvtn_s64_f64(v);
207 return static_cast<int>(vget_lane_s64(r, 0));
208#elif ((defined _MSC_VER && defined _M_X64) || (defined __GNUC__ && defined __SSE2__)) && !defined(__CUDACC__)
209 __m128d t = _mm_set_sd( value );
210 return _mm_cvtsd_si32(t);
211#elif defined _MSC_VER && defined _M_IX86
212 int t;
213 __asm
214 {
215 fld value;
216 fistp t;
217 }
218 return t;
219#elif defined CV__FASTMATH_ENABLE_GCC_MATH_BUILTINS || \
220 defined CV__FASTMATH_ENABLE_CLANG_MATH_BUILTINS
221 return (int)__builtin_lrint(value);
222#else
223 return (int)lrint(value);
224#endif
225}

235inline int cvFloor( double value )
236{
237#if defined CV__FASTMATH_ENABLE_GCC_MATH_BUILTINS || \
238 defined CV__FASTMATH_ENABLE_CLANG_MATH_BUILTINS
239 return (int)__builtin_floor(value);
240#elif defined(_MSC_VER) && (defined(_M_ARM64) || defined(_M_ARM64EC))
241 return (int)vcvtmd_s64_f64(value);
242#elif defined __loongarch64
243 int i;
244 double tmp;
245 __asm__ ("ftintrm.l.d %[tmp], %[in] \n\t"
246 "movfr2gr.d %[i], %[tmp] \n\t"
247 : [i] "=r" (i), [tmp] "=f" (tmp)
248 : [in] "f" (value)
249 :);
250 return i;
251#else
252 int i = (int)value;
253 return i - (i > value);
254#endif
255}

292inline int cvIsNaN( double value )
293{
294#if defined CV_INLINE_ISNAN_DBL
295 CV_INLINE_ISNAN_DBL(value);
296#else
297 Cv64suf ieee754;
298 ieee754.f = value;
299 return ((unsigned)(ieee754.u >> 32) & 0x7fffffff) +
300 ((unsigned)ieee754.u != 0) > 0x7ff00000;
301#endif
302}

310inline int cvIsInf( double value )
311{
312#if defined CV_INLINE_ISINF_DBL
313 CV_INLINE_ISINF_DBL(value);
314#elif defined(__x86_64__) || defined(_M_X64) || defined(__aarch64__) || defined(_M_ARM64) || defined(_M_ARM64EC) || defined(__PPC64__) || defined(__loongarch64)
315 Cv64suf ieee754;
316 ieee754.f = value;
317 return (ieee754.u & 0x7fffffffffffffff) ==
318 0x7ff0000000000000;
319#else
320 Cv64suf ieee754;
321 ieee754.f = value;
322 return ((unsigned)(ieee754.u >> 32) & 0x7fffffff) == 0x7ff00000 &&
323 (unsigned)ieee754.u == 0;
324#endif
325}
