75void CirclesGridClusterFinder::hierarchicalClustering(const std::vector<Point2f> &points, const Size &patternSz, std::vector<Point2f> &patternPoints)
76{
77 int j, n = (int)points.size();
78 size_t pn = static_cast<size_t>(patternSz.area());
79
80 patternPoints.clear();
81 if (pn >= points.size())
82 {
83 if (pn == points.size())
84 patternPoints = points;
85 return;
86 }
87
88 Mat dists(n, n, CV_32FC1, Scalar(0));
89 Mat distsMask(dists.size(), CV_8UC1, Scalar(0));
90 for(int i = 0; i < n; i++)
91 {
92 for(j = i+1; j < n; j++)
93 {
94 dists.at<float>(i, j) = (float)norm(points[i] - points[j]);
95 distsMask.at<uchar>(i, j) = 255;
96 //TODO: use symmetry
97 distsMask.at<uchar>(j, i) = 255;//distsMask.at<uchar>(i, j);
98 dists.at<float>(j, i) = dists.at<float>(i, j);
99 }
100 }
101
102 std::vector<std::list<size_t> > clusters(points.size());
103 for(size_t i=0; i<points.size(); i++)
104 {
105 clusters[i].push_back(i);
106 }
107
108 int patternClusterIdx = 0;
109 while(clusters[patternClusterIdx].size() < pn)
110 {
111 Point minLoc;
112 minMaxLoc(dists, 0, 0, &minLoc, 0, distsMask);
113 int minIdx = std::min(minLoc.x, minLoc.y);
114 int maxIdx = std::max(minLoc.x, minLoc.y);
115
116 distsMask.row(maxIdx).setTo(0);
117 distsMask.col(maxIdx).setTo(0);
118 Mat tmpRow = dists.row(minIdx);
119 Mat tmpCol = dists.col(minIdx);
120 cv::min(dists.row(minLoc.x), dists.row(minLoc.y), tmpRow);
121 tmpRow = tmpRow.t();
122 tmpRow.copyTo(tmpCol);
123
124 clusters[minIdx].splice(clusters[minIdx].end(), clusters[maxIdx]);
125 patternClusterIdx = minIdx;
126 }
127
128 //the largest cluster can have more than pn points -- we need to filter out such situations
129 if(clusters[patternClusterIdx].size() != static_cast<size_t>(patternSz.area()))
130 {
131 return;
132 }
133
134 patternPoints.reserve(clusters[patternClusterIdx].size());
135 for(std::list<size_t>::iterator it = clusters[patternClusterIdx].begin(); it != clusters[patternClusterIdx].end();++it)
136 {
137 patternPoints.push_back(points[*it]);
138 }
139}

141void CirclesGridClusterFinder::findGrid(const std::vector<cv::Point2f> &points, cv::Size _patternSize, std::vector<Point2f>& centers)
142{
143 patternSize = _patternSize;
144 centers.clear();
145 if(points.empty())
146 {
147 return;
148 }
149
150 std::vector<Point2f> patternPoints;
151 hierarchicalClustering(points, patternSize, patternPoints);
152 if(patternPoints.empty())
153 {
154 return;
155 }
156
157#ifdef DEBUG_CIRCLES
158 Mat patternPointsImage(1024, 1248, CV_8UC1, Scalar(0));
159 drawPoints(patternPoints, patternPointsImage);
160 imshow("pattern points", patternPointsImage);
161#endif
162
163 std::vector<Point2f> hull2f;
164 convexHull(patternPoints, hull2f, false);
165 const size_t cornersCount = isAsymmetricGrid ? 6 : 4;
166 if(hull2f.size() < cornersCount)
167 return;
168
169 std::vector<Point2f> corners;
170 findCorners(hull2f, corners);
171 if(corners.size() != cornersCount)
172 return;
173
174 std::vector<Point2f> outsideCorners, sortedCorners;
175 if(isAsymmetricGrid)
176 {
177 findOutsideCorners(corners, outsideCorners);
178 const size_t outsideCornersCount = 2;
179 if(outsideCorners.size() != outsideCornersCount)
180 return;
181 }
182 getSortedCorners(hull2f, patternPoints, corners, outsideCorners, sortedCorners);
183 if(sortedCorners.size() != cornersCount)
184 return;
185
186 std::vector<Point2f> rectifiedPatternPoints;
187 rectifyPatternPoints(patternPoints, sortedCorners, rectifiedPatternPoints);
188 if(patternPoints.size() != rectifiedPatternPoints.size())
189 return;
190
191 parsePatternPoints(patternPoints, rectifiedPatternPoints, centers);
192}

194void CirclesGridClusterFinder::findCorners(const std::vector<cv::Point2f> &hull2f, std::vector<cv::Point2f> &corners)
195{
196 //find angles (cosines) of vertices in convex hull
197 std::vector<float> angles;
198 for(size_t i=0; i<hull2f.size(); i++)
199 {
200 Point2f vec1 = hull2f[(i+1) % hull2f.size()] - hull2f[i % hull2f.size()];
201 Point2f vec2 = hull2f[(i-1 + static_cast<int>(hull2f.size())) % hull2f.size()] - hull2f[i % hull2f.size()];
202 float angle = (float)(vec1.ddot(vec2) / (norm(vec1) * norm(vec2)));
203 angles.push_back(angle);
204 }
205
206 //sort angles by cosine
207 //corners are the most sharp angles (6)
208 Mat anglesMat = Mat(angles);
209 Mat sortedIndices;
210 sortIdx(anglesMat, sortedIndices, SORT_EVERY_ROW + SORT_DESCENDING);
211 CV_Assert(sortedIndices.type() == CV_32SC1);
212 CV_Assert(sortedIndices.rows == 1);
213 const int cornersCount = isAsymmetricGrid ? 6 : 4;
214 Mat cornersIndices;
215 cv::sort(sortedIndices.colRange(0, cornersCount), cornersIndices, SORT_EVERY_ROW + SORT_ASCENDING);
216 corners.clear();
217 for(int i=0; i<cornersCount; i++)
218 {
219 corners.push_back(hull2f[cornersIndices.at<int>(i)]);
220 }

223void CirclesGridClusterFinder::findOutsideCorners(const std::vector<cv::Point2f> &corners, std::vector<cv::Point2f> &outsideCorners)
224{
225 CV_Assert(!corners.empty());
226 outsideCorners.clear();
227 //find two pairs of the most nearest corners
228 const size_t n = corners.size();
229
230#ifdef DEBUG_CIRCLES
231 Mat cornersImage(1024, 1248, CV_8UC1, Scalar(0));
232 drawPoints(corners, cornersImage);
233 imshow("corners", cornersImage);
234#endif
235
236 std::vector<Point2f> tangentVectors(n);
237 for(size_t k=0; k < n; k++)
238 {
239 Point2f diff = corners[(k + 1) % n] - corners[k];
240 tangentVectors[k] = diff * (1.0f / norm(diff));
241 }
242
243 //compute angles between all sides
244 Mat cosAngles((int)n, (int)n, CV_32FC1, 0.0f);
245 for(size_t i = 0; i < n; i++)
246 {
247 for(size_t j = i + 1; j < n; j++)
248 {
249 float val = fabs(tangentVectors[i].dot(tangentVectors[j]));
250 cosAngles.at<float>((int)i, (int)j) = val;
251 cosAngles.at<float>((int)j, (int)i) = val;
252 }
253 }
254
255 //find two parallel sides to which outside corners belong
256 Point maxLoc;
257 minMaxLoc(cosAngles, 0, 0, 0, &maxLoc);
258 const int diffBetweenFalseLines = 3;
259 if(abs(maxLoc.x - maxLoc.y) == diffBetweenFalseLines)
260 {
261 cosAngles.row(maxLoc.x).setTo(0.0f);
262 cosAngles.col(maxLoc.x).setTo(0.0f);
263 cosAngles.row(maxLoc.y).setTo(0.0f);
264 cosAngles.col(maxLoc.y).setTo(0.0f);
265 minMaxLoc(cosAngles, 0, 0, 0, &maxLoc);
266 }
267
268#ifdef DEBUG_CIRCLES
269 Mat linesImage(1024, 1248, CV_8UC1, Scalar(0));
270 line(linesImage, corners[maxLoc.y], corners[(maxLoc.y + 1) % n], Scalar(255));
271 line(linesImage, corners[maxLoc.x], corners[(maxLoc.x + 1) % n], Scalar(255));
272 imshow("lines", linesImage);
273#endif
274
275 int maxIdx = std::max(maxLoc.x, maxLoc.y);
276 int minIdx = std::min(maxLoc.x, maxLoc.y);
277 const int bigDiff = 4;
278 if(maxIdx - minIdx == bigDiff)
279 {
280 minIdx += (int)n;
281 std::swap(maxIdx, minIdx);
282 }
283 if(maxIdx - minIdx != (int)n - bigDiff)
284 {
285 return;
286 }
287
288 int outsidersSegmentIdx = (minIdx + maxIdx) / 2;
289
290 outsideCorners.push_back(corners[outsidersSegmentIdx % n]);
291 outsideCorners.push_back(corners[(outsidersSegmentIdx + 1) % n]);
292
293#ifdef DEBUG_CIRCLES
294 drawPoints(outsideCorners, cornersImage, 2, Scalar(128));
295 imshow("corners", cornersImage);
296#endif
297}

300double pointLineDistance(const cv::Point2f &p, const cv::Vec4f &line)
301{
302 Vec3f pa( line[0], line[1], 1 );
303 Vec3f pb( line[2], line[3], 1 );
304 Vec3f l = pa.cross(pb);
305 return std::abs((p.x * l[0] + p.y * l[1] + l[2])) * 1.0 /
306 std::sqrt(double(l[0] * l[0] + l[1] * l[1]));
307}

310void CirclesGridClusterFinder::getSortedCorners(const std::vector<cv::Point2f> &hull2f, const std::vector<cv::Point2f> &patternPoints, const std::vector<cv::Point2f> &corners, const std::vector<cv::Point2f> &outsideCorners, std::vector<cv::Point2f> &sortedCorners)
311{
312 Point2f firstCorner;
313 if(isAsymmetricGrid)
314 {
315 Point2f center = std::accumulate(corners.begin(), corners.end(), Point2f(0.0f, 0.0f));
316 center *= 1.0 / corners.size();
317
318 std::vector<Point2f> centerToCorners;
319 for(size_t i=0; i<outsideCorners.size(); i++)
320 {
321 centerToCorners.push_back(outsideCorners[i] - center);
322 }
323
324 //TODO: use CirclesGridFinder::getDirection
325 float crossProduct = centerToCorners[0].x * centerToCorners[1].y - centerToCorners[0].y * centerToCorners[1].x;
326 //y axis is inverted in computer vision so we check > 0
327 bool isClockwise = crossProduct > 0;
328 firstCorner = isClockwise ? outsideCorners[1] : outsideCorners[0];
329 }
330 else
331 {
332 firstCorner = corners[0];
333 }
334
335 std::vector<Point2f>::const_iterator firstCornerIterator = std::find(hull2f.begin(), hull2f.end(), firstCorner);
336 sortedCorners.clear();
337 for(std::vector<Point2f>::const_iterator it = firstCornerIterator; it != hull2f.end();++it)
338 {
339 std::vector<Point2f>::const_iterator itCorners = std::find(corners.begin(), corners.end(), *it);
340 if(itCorners != corners.end())
341 {
342 sortedCorners.push_back(*it);
343 }
344 }
345 for(std::vector<Point2f>::const_iterator it = hull2f.begin(); it != firstCornerIterator;++it)
346 {
347 std::vector<Point2f>::const_iterator itCorners = std::find(corners.begin(), corners.end(), *it);
348 if(itCorners != corners.end())
349 {
350 sortedCorners.push_back(*it);
351 }
352 }
353
354 if(!isAsymmetricGrid)
355 {
356 double dist01 = norm(sortedCorners[0] - sortedCorners[1]);
357 double dist12 = norm(sortedCorners[1] - sortedCorners[2]);
358 // Use half the average distance between circles on the shorter side as threshold for determining whether a point lies on an edge.
359 double thresh = min(dist01, dist12) / min(patternSize.width, patternSize.height) / 2;
360
361 size_t circleCount01 = 0;
362 size_t circleCount12 = 0;
363 Vec4f line01( sortedCorners[0].x, sortedCorners[0].y, sortedCorners[1].x, sortedCorners[1].y );
364 Vec4f line12( sortedCorners[1].x, sortedCorners[1].y, sortedCorners[2].x, sortedCorners[2].y );
365 // Count the circles along both edges.
366 for (size_t i = 0; i < patternPoints.size(); i++)
367 {
368 if (pointLineDistance(patternPoints[i], line01) < thresh)
369 circleCount01++;
370 if (pointLineDistance(patternPoints[i], line12) < thresh)
371 circleCount12++;
372 }
373
374 // Ensure that the edge from sortedCorners[0] to sortedCorners[1] is the one with more circles (i.e. it is interpreted as the pattern's width).
375 if ((circleCount01 > circleCount12 && patternSize.height > patternSize.width) || (circleCount01 < circleCount12 && patternSize.height < patternSize.width))
376 {
377 for(size_t i=0; i<sortedCorners.size()-1; i++)
378 {
379 sortedCorners[i] = sortedCorners[i+1];
380 }
381 sortedCorners[sortedCorners.size() - 1] = firstCorner;
382 }
383 }

386void CirclesGridClusterFinder::rectifyPatternPoints(const std::vector<cv::Point2f> &patternPoints, const std::vector<cv::Point2f> &sortedCorners, std::vector<cv::Point2f> &rectifiedPatternPoints)
387{
388 //indices of corner points in pattern
389 std::vector<Point> trueIndices;
390 trueIndices.emplace_back(0, 0);
391 trueIndices.emplace_back(patternSize.width - 1, 0);
392 if(isAsymmetricGrid)
393 {
394 trueIndices.emplace_back(patternSize.width - 1, 1);
395 trueIndices.emplace_back(patternSize.width - 1, patternSize.height - 2);
396 }
397 trueIndices.emplace_back(patternSize.width - 1, patternSize.height - 1);
398 trueIndices.emplace_back(0, patternSize.height - 1);
399
400 std::vector<Point2f> idealPoints;
401 for(size_t idx=0; idx<trueIndices.size(); idx++)
402 {
403 int i = trueIndices[idx].y;
404 int j = trueIndices[idx].x;
405 if(isAsymmetricGrid)
406 {
407 idealPoints.emplace_back((2*j + i % 2)*squareSize, i*squareSize);
408 }
409 else
410 {
411 idealPoints.emplace_back(j*squareSize, i*squareSize);
412 }
413 }
414
415 Mat homography = findHomography(sortedCorners, idealPoints, 0);
416 Mat rectifiedPointsMat;
417 transform(patternPoints, rectifiedPointsMat, homography);
418 rectifiedPatternPoints.clear();
419 convertPointsFromHomogeneous(rectifiedPointsMat, rectifiedPatternPoints);
420}

422void CirclesGridClusterFinder::parsePatternPoints(const std::vector<cv::Point2f> &patternPoints,
423 const std::vector<cv::Point2f> &rectifiedPatternPoints,
424 std::vector<cv::Point2f> &centers)
425{
426#ifndef HAVE_OPENCV_FLANN
427 CV_UNUSED(patternPoints);
428 CV_UNUSED(rectifiedPatternPoints);
429 CV_UNUSED(centers);
430 CV_Error(Error::StsNotImplemented, "The desired functionality requires flann module, which was disabled.");
431#else
432 flann::LinearIndexParams flannIndexParams;
433 flann::Index flannIndex(Mat(rectifiedPatternPoints).reshape(1,
434 (int)rectifiedPatternPoints.size()), flannIndexParams);
435
436 centers.clear();
437 for( int i = 0; i < patternSize.height; i++ )
438 {
439 for( int j = 0; j < patternSize.width; j++ )
440 {
441 Point2f idealPt;
442 if(isAsymmetricGrid)
443 idealPt = Point2f((2*j + i % 2)*squareSize, i*squareSize);
444 else
445 idealPt = Point2f(j*squareSize, i*squareSize);
446
447 Mat query(1, 2, CV_32F, &idealPt);
448 const int knn = 1;
449 int indicesbuf[knn] = {0};
450 float distsbuf[knn] = {0.f};
451 Mat indices(1, knn, CV_32S, &indicesbuf);
452 Mat dists(1, knn, CV_32F, &distsbuf);
453 flannIndex.knnSearch(query, indices, dists, knn, flann::SearchParams());
454 centers.push_back(patternPoints.at(indicesbuf[0]));
455
456 if(distsbuf[0] > maxRectifiedDistance)
457 {
458#ifdef DEBUG_CIRCLES
459 std::cout << "Pattern not detected: too large rectified distance" << std::endl;
460#endif
461 centers.clear();
462 return;
463 }
464 }
465 }
466#endif

527void Graph::floydWarshall(cv::Mat &distanceMatrix, int infinity) const
528{
529 const int edgeWeight = 1;
530
531 const int n = (int)getVerticesCount();
532 distanceMatrix.create(n, n, CV_32SC1);
533 distanceMatrix.setTo(infinity);
534 for (Vertices::const_iterator it1 = vertices.begin(); it1 != vertices.end();++it1)
535 {
536 distanceMatrix.at<int> ((int)it1->first, (int)it1->first) = 0;
537 for (Neighbors::const_iterator it2 = it1->second.neighbors.begin(); it2 != it1->second.neighbors.end();++it2)
538 {
539 CV_Assert( it1->first != *it2 );
540 distanceMatrix.at<int> ((int)it1->first, (int)*it2) = edgeWeight;
541 }
542 }
543
544 for (Vertices::const_iterator it1 = vertices.begin(); it1 != vertices.end();++it1)
545 {
546 for (Vertices::const_iterator it2 = vertices.begin(); it2 != vertices.end();++it2)
547 {
548 for (Vertices::const_iterator it3 = vertices.begin(); it3 != vertices.end();++it3)
549 {
550 int i1 = (int)it1->first, i2 = (int)it2->first, i3 = (int)it3->first;
551 int val1 = distanceMatrix.at<int> (i2, i3);
552 int val2;
553 if (distanceMatrix.at<int> (i2, i1) == infinity ||
554 distanceMatrix.at<int> (i1, i3) == infinity)
555 val2 = val1;
556 else
557 {
558 val2 = distanceMatrix.at<int> (i2, i1) + distanceMatrix.at<int> (i1, i3);
559 }
560 distanceMatrix.at<int> (i2, i3) = (val1 == infinity) ? val2 : std::min(val1, val2);
561 }
562 }
563 }

614bool CirclesGridFinder::findHoles()
615{
616 switch (parameters.gridType)
617 {
618 case CirclesGridFinderParameters::SYMMETRIC_GRID:
619 {
620 std::vector<Point2f> vectors, filteredVectors, basis;
621 Graph rng(0);
622 computeRNG(rng, vectors);
623 filterOutliersByDensity(vectors, filteredVectors);
624 std::vector<Graph> basisGraphs;
625 findBasis(filteredVectors, basis, basisGraphs);
626 findMCS(basis, basisGraphs);
627 break;
628 }
629
630 case CirclesGridFinderParameters::ASYMMETRIC_GRID:
631 {
632 std::vector<Point2f> vectors, tmpVectors, filteredVectors, basis;
633 Graph rng(0);
634 computeRNG(rng, tmpVectors);
635 rng2gridGraph(rng, vectors);
636 filterOutliersByDensity(vectors, filteredVectors);
637 std::vector<Graph> basisGraphs;
638 findBasis(filteredVectors, basis, basisGraphs);
639 findMCS(basis, basisGraphs);

645 }
646
647 default:
648 CV_Error(Error::StsBadArg, "Unknown pattern type");
649 }
650 return (isDetectionCorrect());

701bool CirclesGridFinder::isDetectionCorrect()
702{
703 switch (parameters.gridType)
704 {
705 case CirclesGridFinderParameters::SYMMETRIC_GRID:
706 {
707 rotatedGrid = holes.size() != patternSize.height && holes.size() == patternSize.width;
708 if (holes.size() != patternSize.height && holes.size() != patternSize.width)
709 return false;
710
711 size_t num_vertices = 0ull;
712 for (size_t i = 0; i < holes.size(); i++)
713 {
714 if (holes[i].size() != patternSize.width && rotatedGrid == false)
715 {
716 return false;
717 }
718 else if (holes[i].size() != patternSize.height && rotatedGrid == true)
719 {
720 rotatedGrid = false;
721 return false;
722 }
723
724 num_vertices += holes[i].size();
725 }
726 return num_vertices == patternSize.area();
727 }
728
729 case CirclesGridFinderParameters::ASYMMETRIC_GRID:
730 {
731 if (holes.size() < holes2.size() || holes[0].size() < holes2[0].size())
732 {
733 largeHoles = &holes2;
734 smallHoles = &holes;
735 }
736 else
737 {
738 largeHoles = &holes;
739 smallHoles = &holes2;
740 }
741
742 size_t largeWidth = patternSize.width;
743 size_t largeHeight = (size_t)ceil(patternSize.height / 2.);
744 size_t smallWidth = patternSize.width;
745 size_t smallHeight = (size_t)floor(patternSize.height / 2.);
746
747 size_t sw = smallWidth, sh = smallHeight, lw = largeWidth, lh = largeHeight;
748 if (largeHoles->size() != largeHeight)
749 {
750 std::swap(lh, lw);
751 }
752 if (smallHoles->size() != smallHeight)
753 {
754 std::swap(sh, sw);
755 }
756
757 if (largeHoles->size() != lh || smallHoles->size() != sh)
758 {
759 return false;
760 }
761
762 std::set<size_t> vertices;
763 for (size_t i = 0; i < largeHoles->size(); i++)
764 {
765 if (largeHoles->at(i).size() != lw)
766 {
767 return false;
768 }
769
770 for (size_t j = 0; j < largeHoles->at(i).size(); j++)
771 {
772 vertices.insert(largeHoles->at(i)[j]);
773 }
774
775 if (i < smallHoles->size())
776 {
777 if (smallHoles->at(i).size() != sw)
778 {
779 return false;
780 }
781
782 for (size_t j = 0; j < smallHoles->at(i).size(); j++)
783 {
784 vertices.insert(smallHoles->at(i)[j]);
785 }
786 }
787 }
788 return (vertices.size() == largeHeight * largeWidth + smallHeight * smallWidth);
789 }
790 }
791 CV_Error(Error::StsBadArg, "Unknown pattern type");
792}

853Mat CirclesGridFinder::rectifyGrid(Size detectedGridSize, const std::vector<Point2f>& centers,
854 const std::vector<Point2f> &keypoints, std::vector<Point2f> &warpedKeypoints)
855{
856 CV_Assert( !centers.empty() );
857 const float edgeLength = 30;
858 const Point2f offset(150, 150);
859
860 std::vector<Point2f> dstPoints;
861 bool isClockwiseBefore =
862 getDirection(centers[0], centers[detectedGridSize.width - 1], centers[centers.size() - 1]) < 0;
863
864 int iStart = isClockwiseBefore ? 0 : detectedGridSize.height - 1;
865 int iEnd = isClockwiseBefore ? detectedGridSize.height : -1;
866 int iStep = isClockwiseBefore ? 1 : -1;
867 for (int i = iStart; i != iEnd; i += iStep)
868 {
869 for (int j = 0; j < detectedGridSize.width; j++)
870 {
871 dstPoints.push_back(offset + Point2f(edgeLength * j, edgeLength * i));
872 }
873 }
874
875 Mat H = findHomography(centers, dstPoints, RANSAC);
876 //Mat H = findHomography(corners, dstPoints);
877
878 if (H.empty())
879 {
880 H = Mat::zeros(3, 3, CV_64FC1);
881 warpedKeypoints.clear();
882 return H;
883 }
884
885 std::vector<Point2f> srcKeypoints;
886 for (size_t i = 0; i < keypoints.size(); i++)
887 {
888 srcKeypoints.push_back(keypoints[i]);
889 }
890
891 Mat dstKeypointsMat;
892 transform(srcKeypoints, dstKeypointsMat, H);
893 std::vector<Point2f> dstKeypoints;
894 convertPointsFromHomogeneous(dstKeypointsMat, dstKeypoints);
895
896 warpedKeypoints.clear();
897 for (auto &pt:dstKeypoints)
898 {
899 warpedKeypoints.emplace_back(std::move(pt));
900 }
901
902 return H;
903}

1047 const std::vector<size_t> &points, const std::vector<size_t> &seeds)

1050 float confidence = 0;
1051 const size_t vCount = basisGraphs[0].getVerticesCount();
1052 CV_Assert( basisGraphs[0].getVerticesCount() == basisGraphs[1].getVerticesCount() );
1053
1054 for (size_t i = 0; i < seeds.size(); i++)
1055 {
1056 if (seeds[i] < vCount && points[i] < vCount)
1057 {
1058 if (!basisGraphs[addRow].areVerticesAdjacent(seeds[i], points[i]))
1059 {
1060 confidence += parameters.vertexPenalty;
1061 }
1062 else
1063 {
1064 confidence += parameters.vertexGain;
1065 }
1066 }
1067
1068 if (points[i] < vCount)
1069 {
1070 confidence += parameters.existingVertexGain;
1071 }
1072 }
1073
1074 for (size_t i = 1; i < points.size(); i++)
1075 {
1076 if (points[i - 1] < vCount && points[i] < vCount)
1077 {
1078 if (!basisGraphs[!addRow].areVerticesAdjacent(points[i - 1], points[i]))
1079 {
1080 confidence += parameters.edgePenalty;
1081 }
1082 else
1083 {
1084 confidence += parameters.edgeGain;
1085 }
1086 }
1087 }
1088 return confidence;

1127void CirclesGridFinder::findBasis(const std::vector<Point2f> &samples, std::vector<Point2f> &basis, std::vector<Graph> &basisGraphs)
1128{
1129 basis.clear();
1130 Mat bestLabels;
1131 TermCriteria termCriteria;
1132 Mat centers;
1133 const int clustersCount = 4;
1134 int nsamples = (int)samples.size();
1135 kmeans(Mat(samples).reshape(1, nsamples), clustersCount, bestLabels, termCriteria, parameters.kmeansAttempts,
1136 KMEANS_RANDOM_CENTERS, centers);
1137 CV_Assert( centers.type() == CV_32FC1 );
1138
1139 std::vector<int> basisIndices;
1140 //TODO: only remove duplicate
1141 for (int i = 0; i < clustersCount; i++)
1142 {
1143 int maxIdx = (fabs(centers.at<float> (i, 0)) < fabs(centers.at<float> (i, 1)));
1144 if (centers.at<float> (i, maxIdx) > 0)
1145 {
1146 Point2f vec(centers.at<float> (i, 0), centers.at<float> (i, 1));
1147 basis.push_back(vec);
1148 basisIndices.push_back(i);
1149 }
1150 }
1151 if (basis.size() != 2)
1152 CV_Error(cv::Error::StsError, "Basis size is not 2");
1153
1154 if (basis[1].x > basis[0].x)
1155 {
1156 std::swap(basis[0], basis[1]);
1157 std::swap(basisIndices[0], basisIndices[1]);
1158 }
1159
1160 const float minBasisDif = 2;
1161 if (norm(basis[0] - basis[1]) < minBasisDif)
1162 CV_Error(cv::Error::StsError, "degenerate basis" );
1163
1164 std::vector<std::vector<Point2f> > clusters(2), hulls(2);
1165 for (int k = 0; k < (int)samples.size(); k++)
1166 {
1167 int label = bestLabels.at<int> (k, 0);
1168 int idx = -1;
1169 if (label == basisIndices[0])
1170 idx = 0;
1171 if (label == basisIndices[1])
1172 idx = 1;
1173 if (idx >= 0)
1174 {
1175 clusters[idx].push_back(basis[idx] + parameters.convexHullFactor * (samples[k] - basis[idx]));
1176 }
1177 }
1178 for (size_t i = 0; i < basis.size(); i++)
1179 {
1180 convexHull(clusters[i], hulls[i]);
1181 }
1182
1183 basisGraphs.resize(basis.size(), Graph(keypoints.size()));
1184 for (size_t i = 0; i < keypoints.size(); i++)
1185 {
1186 for (size_t j = 0; j < keypoints.size(); j++)
1187 {
1188 if (i == j)
1189 continue;
1190
1191 Point2f vec = keypoints[i] - keypoints[j];
1192
1193 for (size_t k = 0; k < hulls.size(); k++)
1194 {
1195 if (pointPolygonTest(hulls[k], vec, false) >= 0)
1196 {
1197 basisGraphs[k].addEdge(i, j);
1198 }
1199 }
1200 }
1201 }
1202 if (basisGraphs.size() != 2)
1203 CV_Error(cv::Error::StsError, "Number of basis graphs is not 2");
1204}

1206void CirclesGridFinder::computeRNG(Graph &rng, std::vector<cv::Point2f> &vectors, Mat *drawImage) const
1207{
1208 rng = Graph(keypoints.size());
1209 vectors.clear();
1210
1211 const size_t n = keypoints.size();
1212 if (n < 2)
1213 return;
1214
1215 // RNG is a subgraph of the Delaunay triangulation, so we only need to test
1216 // Delaunay edges as candidates. This brings the complexity from O(N^3) down
1217 // to O(N^2) in the worst case, and much better in practice for regular grids
1218 // where Delaunay edges (~3N) are almost all RNG edges anyway.
1219
1220 float minX = keypoints[0].x, minY = keypoints[0].y;
1221 float maxX = minX, maxY = minY;
1222 for (size_t i = 1; i < n; i++)
1223 {
1224 minX = std::min(minX, keypoints[i].x);
1225 minY = std::min(minY, keypoints[i].y);
1226 maxX = std::max(maxX, keypoints[i].x);
1227 maxY = std::max(maxY, keypoints[i].y);
1228 }
1229
1230 // Subdiv2D requires a rect that strictly contains all points.
1231 const float margin = 1.f;
1232 Rect2f rect(minX - margin, minY - margin,
1233 (maxX - minX) + 2*margin,
1234 (maxY - minY) + 2*margin);
1235 Subdiv2D subdiv(rect);
1236 subdiv.insert(std::vector<Point2f>(keypoints.begin(), keypoints.end()));
1237
1238 // Map coordinates back to keypoint indices. Subdiv2D stores and returns the
1239 // exact float values we inserted, so direct comparison is safe here.
1240 std::map<std::pair<float, float>, size_t> ptToIdx;
1241 for (size_t i = 0; i < n; i++)
1242 ptToIdx[{keypoints[i].x, keypoints[i].y}] = i;
1243
1244 std::vector<Vec4f> edgeList;
1245 subdiv.getEdgeList(edgeList);
1246
1247 for (const Vec4f& e : edgeList)
1248 {
1249 auto it1 = ptToIdx.find({e[0], e[1]});
1250 auto it2 = ptToIdx.find({e[2], e[3]});
1251 // Edges involving the virtual bounding-rect vertices won't be in ptToIdx.
1252 if (it1 == ptToIdx.end() || it2 == ptToIdx.end())
1253 continue;
1254
1255 size_t i = it1->second;
1256 size_t j = it2->second;
1257 if (i == j)
1258 continue;
1259 if (i > j)
1260 std::swap(i, j);
1261
1262 Point2f vec = keypoints[i] - keypoints[j];
1263 double distSq = (double)vec.x*vec.x + (double)vec.y*vec.y;
1264
1265 bool isRNG = true;
1266 for (size_t k = 0; k < n; k++)
1267 {
1268 if (k == i || k == j)
1269 continue;
1270 Point2f d1 = keypoints[i] - keypoints[k];
1271 Point2f d2 = keypoints[j] - keypoints[k];
1272 double d1Sq = (double)d1.x*d1.x + (double)d1.y*d1.y;
1273 double d2Sq = (double)d2.x*d2.x + (double)d2.y*d2.y;
1274 if (d1Sq < distSq && d2Sq < distSq)
1275 {
1276 isRNG = false;
1277 break;
1278 }
1279 }
1280
1281 if (isRNG)
1282 {
1283 rng.addEdge(i, j);
1284 // Push both directions; findBasis needs the full set to cluster into
1285 // the 4 groups (two grid axes and their negatives) via k-means.
1286 vectors.push_back(keypoints[i] - keypoints[j]);
1287 vectors.push_back(keypoints[j] - keypoints[i]);
1288 if (drawImage != 0)
1289 {
1290 line(*drawImage, keypoints[i], keypoints[j], Scalar(255, 0, 0), 2);
1291 circle(*drawImage, keypoints[i], 3, Scalar(0, 0, 255), -1);
1292 circle(*drawImage, keypoints[j], 3, Scalar(0, 0, 255), -1);
1293 }
1294 }
1295 }
1296}

1298size_t CirclesGridFinder::findLongestPath(std::vector<Graph> &basisGraphs, Path &bestPath)
1299{
1300 std::vector<Path> longestPaths(1);
1301 std::vector<int> confidences;

1310 // basisGraphs are sparse -- each vertex connects only to grid neighbors in

1312 std::vector<int> dist(n);

1314
1315 int maxDist = 0;

1317

1344 }
1345 }

1350 confidences.clear();

1352 }
1353 if (longestPaths.empty() || (maxDist == longestPaths[0].length && graphIdx == bestGraphIdx))
1354 {
1355 Path path = Path(srcBest, dstBest, maxDist);
1356
1357 // BFS again from srcBest to reconstruct the path to dstBest
1358 std::vector<int> pred(n, -1);
1359 std::fill(dist.begin(), dist.end(), -1);
1360 dist[srcBest] = 0;
1361 q.push(srcBest);
1362 while (!q.empty())
1363 {

1365 for (size_t nb : g.getNeighbors((size_t)v))
1366 {
1367 int u = (int)nb;
1368 if (dist[u] < 0)
1369 {
1370 dist[u] = dist[v] + 1;
1371 pred[u] = v;
1372 q.push(u);
1373 }
1374 }
1375 }
1376 std::vector<size_t> pathVertices;
1377 for (int cur = dstBest; cur != srcBest; cur = pred[cur])
1378 pathVertices.push_back((size_t)cur);
1379 pathVertices.push_back((size_t)srcBest);
1380 std::reverse(pathVertices.begin(), pathVertices.end());
1381 path.vertices = pathVertices;
1382
1383 longestPaths.push_back(path);
1384
1385 int conf = 0;
1386 for (int v2 = 0; v2 < (int)path.vertices.size(); v2++)
1387 conf += (int)basisGraphs[1 - (int)graphIdx].getDegree(path.vertices[v2]);
1388 confidences.push_back(conf);
1389 }
1390 }
1391
1392 int maxConf = -1;
1393 int bestPathIdx = -1;
1394 for (int i = 0; i < (int)confidences.size(); i++)
1395 {
1396 if (confidences[i] > maxConf)
1397 {
1398 maxConf = confidences[i];
1399 bestPathIdx = i;
1400 }
1401 }
1402
1403 bestPath = longestPaths.at(bestPathIdx);
1404 bool needReverse = (bestGraphIdx == 0 && keypoints[bestPath.lastVertex].x < keypoints[bestPath.firstVertex].x)
1405 || (bestGraphIdx == 1 && keypoints[bestPath.lastVertex].y < keypoints[bestPath.firstVertex].y);
1406 if (needReverse)
1407 {
1408 std::swap(bestPath.lastVertex, bestPath.firstVertex);
1409 std::reverse(bestPath.vertices.begin(), bestPath.vertices.end());
1410 }
1411 return bestGraphIdx;
