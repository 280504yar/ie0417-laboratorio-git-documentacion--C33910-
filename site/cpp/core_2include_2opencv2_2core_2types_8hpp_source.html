1513template<> inline
1514int Point3_<int>::dot(const Point3_<int>& pt) const
1515{
1516 const int64_t xx = (int64_t)x * pt.x;
1517 const int64_t yy = (int64_t)y * pt.y;
1518 const int64_t zz = (int64_t)z * pt.z;
1519
1520 // Sort the three products so that lo <= mid <= hi.
1521 // We add in the order (lo + hi) first, then add mid.
1522 // This minimizes the absolute value of the intermediate sum,
1523 // reducing the chance of int64 overflow during addition.
1524 int64_t lo = xx, mid = yy, hi = zz;
1525 if (lo > mid) std::swap(lo, mid);
1526 if (mid > hi) std::swap(mid, hi);
1527 if (lo > mid) std::swap(lo, mid);
1528
1529 // Step 1: lo + hi
1530 // If lo + hi overflows int64, the final result must saturate to INT32_MAX/INT32_MIN.
1531 // The middle value (mid) cannot bring the result back into the int32 range,
1532 // so we can return immediately without considering mid.
1533 if (hi > 0 && lo > INT64_MAX - hi)
1534 {
1535 return INT32_MAX;
1536 }
1537 if (hi < 0 && lo < INT64_MIN - hi)
1538 {
1539 return INT32_MIN;
1540 }
1541 int64_t sum = lo + hi;
1542
1543 // Step 2: sum + mid
1544 if (mid > 0 && sum > INT64_MAX - mid)
1545 {
1546 return INT32_MAX;
1547 }
1548 if (mid < 0 && sum < INT64_MIN - mid)
1549 {
1550 return INT32_MIN;
1551 }
1552 sum += mid;
1553
1554 return cv::saturate_cast<int>(sum);
1555}

1994template<typename _Tp> inline
1995Rect_<_Tp>& operator &= ( Rect_<_Tp>& a, const Rect_<_Tp>& b )
1996{
1997 if (a.empty() || b.empty()) {
1998 a = Rect_<_Tp>();
1999 return a;
2000 }
2001 const Rect_<_Tp>& Rx_min = (a.x < b.x) ? a : b;
2002 const Rect_<_Tp>& Rx_max = (a.x < b.x) ? b : a;
2003 const Rect_<_Tp>& Ry_min = (a.y < b.y) ? a : b;
2004 const Rect_<_Tp>& Ry_max = (a.y < b.y) ? b : a;
2005 // Looking at the formula below, we will compute Rx_min.width - (Rx_max.x - Rx_min.x)
2006 // but we want to avoid overflows. Rx_min.width >= 0 and (Rx_max.x - Rx_min.x) >= 0
2007 // by definition so the difference does not overflow. The only thing that can overflow
2008 // is (Rx_max.x - Rx_min.x). And it can only overflow if Rx_min.x < 0.
2009 // Let us first deal with the following case.
2010 if ((Rx_min.x < 0 && Rx_min.x + Rx_min.width < Rx_max.x) ||
2011 (Ry_min.y < 0 && Ry_min.y + Ry_min.height < Ry_max.y)) {
2012 a = Rect_<_Tp>();
2013 return a;
2014 }
2015 // We now know that either Rx_min.x >= 0, or
2016 // Rx_min.x < 0 && Rx_min.x + Rx_min.width >= Rx_max.x and therefore
2017 // Rx_min.width >= (Rx_max.x - Rx_min.x) which means (Rx_max.x - Rx_min.x)
2018 // is inferior to a valid int and therefore does not overflow.
2019 a.width = std::min(Rx_min.width - (Rx_max.x - Rx_min.x), Rx_max.width);
2020 a.height = std::min(Ry_min.height - (Ry_max.y - Ry_min.y), Ry_max.height);
2021 a.x = Rx_max.x;
2022 a.y = Ry_max.y;
2023 if (a.empty())
2024 a = Rect_<_Tp>();
2025 return a;
2026}
