16// ONNX op: Scan — https://onnx.ai/onnx/operators/onnx__Scan.html
17// Supported opsets: 9 .. latest (opset-9 dataflow semantics).
18// Opset-8 Scan (leading batch dimension + optional sequence_lens input) has different
19// semantics and is NOT supported.
20//
21// Net::Impl::forwardGraph drives the loop; this layer only holds the body and its attributes.
22class ScanLayerImpl CV_FINAL : public ScanLayer
23{
24public:
25 explicit ScanLayerImpl(const LayerParams& params)
26 {
27 setParamsFrom(params);
28 num_scan_inputs = params.get<int>("num_scan_inputs");
29 readIntList(params, "scan_input_axes", input_axes);
30 readIntList(params, "scan_output_axes", output_axes);
31 readIntList(params, "scan_input_directions", input_dirs);
32 readIntList(params, "scan_output_directions", output_dirs);
33 readIntList(params, "scan_output_ranks", output_ranks);
34 }
35
36 std::vector<Ptr<Graph> >* subgraphs() const CV_OVERRIDE { return &body_; }
37 bool dynamicOutputShapes() const CV_OVERRIDE { return true; }
38
39 bool getMemoryShapes(const std::vector<MatShape>&,
40 const int requiredOutputs,
41 std::vector<MatShape>& outputs,
42 std::vector<MatShape>& internals) const CV_OVERRIDE
43 {
44 outputs.assign(std::max(1, requiredOutputs), MatShape());
45 internals.clear();
46 return false;
47 }
48
49 // Scan iterates by shape/axis alone; there's no value to gate on.
50 void getTypes(const std::vector<MatType>& inputs,
51 const int requiredOutputs,
52 const int requiredInternals,
53 std::vector<MatType>& outputs,
54 std::vector<MatType>& internals) const CV_OVERRIDE
55 {
56 CV_Assert(inputs.size());
57 for (auto input : inputs)
58 CV_CheckType(input, input == CV_32F || input == CV_64F || input == CV_8U || input == CV_8S ||
59 input == CV_16U || input == CV_16S || input == CV_32U || input == CV_32S ||
60 input == CV_64U || input == CV_64S || input == CV_Bool,
61 "Scan: unsupported input type");
62
63 outputs.assign(requiredOutputs, inputs[0]);
64 internals.assign(requiredInternals, inputs[0]);
65 }
66
67 int numScanInputs() const CV_OVERRIDE { return num_scan_inputs; }
68 const std::vector<int>& scanInputAxes() const CV_OVERRIDE { return input_axes; }
69 const std::vector<int>& scanOutputAxes() const CV_OVERRIDE { return output_axes; }
70 const std::vector<int>& scanInputDirections() const CV_OVERRIDE { return input_dirs; }
71 const std::vector<int>& scanOutputDirections() const CV_OVERRIDE { return output_dirs; }
72 const std::vector<int>& scanOutputRanks() const CV_OVERRIDE { return output_ranks; }
73
74private:
75 static void readIntList(const LayerParams& params, const std::string& name, std::vector<int>& out)
76 {
77 out.clear();
78 if (!params.has(name)) return;
79 const DictValue& v = params.get(name);
80 out.resize(v.size());
81 for (int i = 0; i < v.size(); ++i) out[i] = v.get<int>(i);
82 }
83
84 int num_scan_inputs = 0;
85 std::vector<int> input_axes, output_axes, input_dirs, output_dirs, output_ranks;
86 mutable std::vector<Ptr<Graph> > body_;
87};

89Ptr<ScanLayer> ScanLayer::create(const LayerParams& params)

92}
