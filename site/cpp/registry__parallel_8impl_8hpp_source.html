56class ParallelBackendRegistry
57{
58protected:
59 std::vector<ParallelBackendInfo> enabledBackends;
60 ParallelBackendRegistry()
61 {
62 enabledBackends = getBuiltinParallelBackendsInfo();
63 int N = (int)enabledBackends.size();
64 for (int i = 0; i < N; i++)
65 {
66 ParallelBackendInfo& info = enabledBackends[i];
67 info.priority = 1000 - i * 10;
68 }
69 CV_LOG_DEBUG(NULL, "core(parallel): Builtin backends(" << N << "): " << dumpBackends());
70 if (readPrioritySettings())
71 {
72 CV_LOG_INFO(NULL, "core(parallel): Updated backends priorities: " << dumpBackends());
73 N = (int)enabledBackends.size();
74 }
75 int enabled = 0;
76 for (int i = 0; i < N; i++)
77 {
78 ParallelBackendInfo& info = enabledBackends[enabled];
79 if (enabled != i)
80 info = enabledBackends[i];
81 size_t param_priority = utils::getConfigurationParameterSizeT(cv::format("OPENCV_PARALLEL_PRIORITY_%s", info.name.c_str()).c_str(), (size_t)info.priority);
82 CV_Assert(param_priority == (size_t)(int)param_priority); // overflow check
83 if (param_priority > 0)
84 {
85 info.priority = (int)param_priority;
86 enabled++;
87 }
88 else
89 {
90 CV_LOG_INFO(NULL, "core(parallel): Disable backend: " << info.name);
91 }
92 }
93 enabledBackends.resize(enabled);
94 CV_LOG_DEBUG(NULL, "core(parallel): Available backends(" << enabled << "): " << dumpBackends());
95 std::sort(enabledBackends.begin(), enabledBackends.end(), sortByPriority);
96 CV_LOG_INFO(NULL, "core(parallel): Enabled backends(" << enabled << ", sorted by priority): " << (enabledBackends.empty() ? std::string("N/A") : dumpBackends()));
97 }

99 static std::vector<std::string> tokenize_string(const std::string& input, char token)
100 {
101 std::vector<std::string> result;
102 std::string::size_type prev_pos = 0, pos = 0;
103 while((pos = input.find(token, pos)) != std::string::npos)
104 {
105 result.push_back(input.substr(prev_pos, pos-prev_pos));
106 prev_pos = ++pos;
107 }
108 result.push_back(input.substr(prev_pos));
109 return result;
110 }
111 bool readPrioritySettings()
112 {
113 bool hasChanges = false;
114 cv::String prioritized_backends = utils::getConfigurationParameterString("OPENCV_PARALLEL_PRIORITY_LIST");
115 if (prioritized_backends.empty())
116 return hasChanges;
117 CV_LOG_INFO(NULL, "core(parallel): Configured priority list (OPENCV_PARALLEL_PRIORITY_LIST): " << prioritized_backends);
118 const std::vector<std::string> names = tokenize_string(prioritized_backends, ',');
119 for (size_t i = 0; i < names.size(); i++)
120 {
121 const std::string& name = names[i];
122 int priority = (int)(100000 + (names.size() - i) * 1000);
123 bool found = false;
124 for (size_t k = 0; k < enabledBackends.size(); k++)
125 {
126 ParallelBackendInfo& info = enabledBackends[k];
127 if (name == info.name)
128 {
129 info.priority = priority;
130 CV_LOG_DEBUG(NULL, "core(parallel): New backend priority: '" << name << "' => " << info.priority);
131 found = true;
132 hasChanges = true;
133 break;
134 }
135 }
136 if (!found)
137 {
138 CV_LOG_INFO(NULL, "core(parallel): Adding parallel backend (plugin): '" << name << "'");
139 enabledBackends.push_back(ParallelBackendInfo{priority, name, createPluginParallelBackendFactory(name)});
140 hasChanges = true;
141 }
142 }
143 return hasChanges;
144 }
145public:
146 std::string dumpBackends() const
147 {
148 std::ostringstream os;
149 for (size_t i = 0; i < enabledBackends.size(); i++)
150 {
151 if (i > 0) os << "; ";
152 const ParallelBackendInfo& info = enabledBackends[i];
153 os << info.name << '(' << info.priority << ')';
154 }
155 return os.str();
156 }
