15inline Point2d projectThrough(const Mat& P, double X, double Y, double Z)
16{
17 double u = P.at<double>(0,0)*X + P.at<double>(0,1)*Y + P.at<double>(0,2)*Z + P.at<double>(0,3);
18 double v = P.at<double>(1,0)*X + P.at<double>(1,1)*Y + P.at<double>(1,2)*Z + P.at<double>(1,3);
19 double w = P.at<double>(2,0)*X + P.at<double>(2,1)*Y + P.at<double>(2,2)*Z + P.at<double>(2,3);
20 if (std::abs(w) < 1e-12) return Point2d(0, 0);
21 return Point2d(u / w, v / w);
22}

31void VisualOdometryImpl::promoteKeyframeAndGrowMap(Frame& cur)
32{
33 KeyFrame* newKf = new KeyFrame();
34 newKf->poseCw = cur.poseCw;
35 newKf->keypoints = cur.keypoints;
36 newKf->descriptors = cur.descriptors.clone();
37 newKf->undistKpts = cur.undistKpts;
38 newKf->imageSize = cur.imageSize;
39 newKf->mapPoints.assign(cur.keypoints.size(), nullptr);
40 newKf->parent = lastKf;
41
42 map.addKeyframe(newKf);
43
44 for (size_t i = 0; i < cur.mapPoints.size(); ++i)
45 {
46 MapPoint* mp = cur.mapPoints[i];
47 if (!mp || mp->bad || cur.outliers[i]) continue;
48 map.addObservation(newKf, i, mp);
49 }
50
51 std::vector<DMatch> kfToCur;
52 matchFrames(lastKf->keypoints, lastKf->descriptors, lastKf->imageSize,
53 cur.keypoints, cur.descriptors, cur.imageSize, kfToCur);
54
55 Mat Rt1(3, 4, CV_64F), Rt2(3, 4, CV_64F);
56 for (int i = 0; i < 3; ++i)
57 for (int j = 0; j < 4; ++j)
58 {
59 Rt1.at<double>(i,j) = lastKf->poseCw(i,j);
60 Rt2.at<double>(i,j) = cur.poseCw(i,j);
61 }
62 Mat P1 = K * Rt1;
63 Mat P2 = K * Rt2;
64
65 std::vector<Point2f> pts1, pts2;
66 std::vector<int> triMatchIdx;
67 for (size_t i = 0; i < kfToCur.size(); ++i)
68 {
69 const DMatch& m = kfToCur[i];
70 if ((size_t)m.queryIdx >= lastKf->mapPoints.size()) continue;
71 if (lastKf->mapPoints[m.queryIdx] != nullptr) continue;
72 if ((size_t)m.trainIdx >= newKf->mapPoints.size()) continue;
73 if (newKf->mapPoints[m.trainIdx] != nullptr) continue;
74 pts1.push_back(lastKf->undistKpts[m.queryIdx]);
75 pts2.push_back(cur.undistKpts[m.trainIdx]);
76 triMatchIdx.push_back((int)i);
77 }
78
79 if (pts1.empty())
80 {
81 detail::updateCovisibility(newKf);
82 map.setCurrentKeyframe(newKf);
83 lastKf = newKf;
84 lastKfInliers = 0;
85 framesSinceKf = 0;
86 return;
87 }
88
89 Mat pts4D;
90 triangulatePoints(P1, P2, pts1, pts2, pts4D);
91
92 int nNew = 0;
93 for (int i = 0; i < pts4D.cols; ++i)
94 {
95 double w = pts4D.at<float>(3, i);
96 if (std::abs(w) < 1e-9) continue;
97 double X = pts4D.at<float>(0, i) / w;
98 double Y = pts4D.at<float>(1, i) / w;
99 double Z = pts4D.at<float>(2, i) / w;
100
101 if (cameraDepth(lastKf->poseCw, X, Y, Z) <= 0) continue;
102 if (cameraDepth(cur.poseCw, X, Y, Z) <= 0) continue;
103
104 Point2d p1p = projectThrough(P1, X, Y, Z);
105 Point2d p2p = projectThrough(P2, X, Y, Z);
106 double e1 = std::hypot(p1p.x - pts1[i].x, p1p.y - pts1[i].y);
107 double e2 = std::hypot(p2p.x - pts2[i].x, p2p.y - pts2[i].y);
108 if (e1 > params.pnpReprojThresh || e2 > params.pnpReprojThresh) continue;
109
110 Point3d Xw(X, Y, Z);
111 if (detail::parallaxDeg(Xw, lastKf->poseCw, cur.poseCw)
112 < params.minGrowthParallaxDeg) continue;
113
114 MapPoint* mp = new MapPoint();
115 mp->pos = Xw;
116 const DMatch& dm = kfToCur[triMatchIdx[i]];
117 mp->refDesc = cur.descriptors.row(dm.trainIdx).clone();
118
119 map.addMapPoint(mp);
120 map.addObservation(lastKf, (size_t)dm.queryIdx, mp);
121 map.addObservation(newKf, (size_t)dm.trainIdx, mp);
122 ++nNew;
123 }
124
125 detail::updateCovisibility(newKf);
126
127 map.setCurrentKeyframe(newKf);
128 lastKf = newKf;
129 lastKfInliers = 0;
130 framesSinceKf = 0;
131
132 (void)nNew;
133}
