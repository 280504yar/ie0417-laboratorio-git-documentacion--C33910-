14CGImageRef MatToCGImage(const cv::Mat& image) {
15 NSData *data = [NSData dataWithBytes:image.data
16 length:image.step.p[0] * image.rows];
17
18 CGColorSpaceRef colorSpace;
19
20 if (image.elemSize() == 1) {
21 colorSpace = CGColorSpaceCreateDeviceGray();
22 } else {
23 colorSpace = CGColorSpaceCreateDeviceRGB();
24 }
25
26 CGDataProviderRef provider =
27 CGDataProviderCreateWithCFData((__bridge CFDataRef)data);
28
29 // Preserve alpha transparency, if exists
30 bool alpha = image.channels() == 4;
31 CGBitmapInfo bitmapInfo = (CGBitmapInfo)(alpha ? kCGImageAlphaLast : kCGImageAlphaNone) | (CGBitmapInfo)kCGBitmapByteOrderDefault;
32
33 // Creating CGImage from cv::Mat
34 CGImageRef imageRef = CGImageCreate(image.cols,
35 image.rows,
36 8 * image.elemSize1(),
37 8 * image.elemSize(),
38 image.step.p[0],
39 colorSpace,
40 bitmapInfo,
41 provider,
42 NULL,
43 false,
44 kCGRenderingIntentDefault
45 );
46
47 CGDataProviderRelease(provider);
48 CGColorSpaceRelease(colorSpace);
49
50 return imageRef;
51}

53void CGImageToMat(const CGImageRef image, cv::Mat& m, bool alphaExist) {
54 CGColorSpaceRef colorSpace = CGImageGetColorSpace(image);
55 CGFloat cols = CGImageGetWidth(image), rows = CGImageGetHeight(image);
56 CGContextRef contextRef;
57 CGBitmapInfo bitmapInfo = kCGImageAlphaPremultipliedLast;
58 if (CGColorSpaceGetModel(colorSpace) == kCGColorSpaceModelMonochrome)
59 {
60 m.create(rows, cols, CV_8UC1); // 8 bits per component, 1 channel
61 bitmapInfo = kCGImageAlphaNone;
62 if (!alphaExist)
63 bitmapInfo = kCGImageAlphaNone;
64 else
65 m = cv::Scalar(0);
66 contextRef = CGBitmapContextCreate(m.data, m.cols, m.rows, 8,
67 m.step[0], colorSpace,
68 bitmapInfo);
69 }
70 else if (CGColorSpaceGetModel(colorSpace) == kCGColorSpaceModelIndexed)
71 {
72 // CGBitmapContextCreate() does not support indexed color spaces.
73 colorSpace = CGColorSpaceCreateDeviceRGB();
74 m.create(rows, cols, CV_8UC4); // 8 bits per component, 4 channels
75 if (!alphaExist)
76 bitmapInfo = (CGBitmapInfo)kCGImageAlphaNoneSkipLast |
77 (CGBitmapInfo)kCGBitmapByteOrderDefault;
78 else
79 m = cv::Scalar(0);
80 contextRef = CGBitmapContextCreate(m.data, m.cols, m.rows, 8,
81 m.step[0], colorSpace,
82 bitmapInfo);
83 CGColorSpaceRelease(colorSpace);
84 }
85 else
86 {
87 m.create(rows, cols, CV_8UC4); // 8 bits per component, 4 channels
88 if (!alphaExist)
89 bitmapInfo = (CGBitmapInfo)kCGImageAlphaNoneSkipLast |
90 (CGBitmapInfo)kCGBitmapByteOrderDefault;
91 else
92 m = cv::Scalar(0);
93 contextRef = CGBitmapContextCreate(m.data, m.cols, m.rows, 8,
94 m.step[0], colorSpace,
95 bitmapInfo);
96 }
97 CGContextDrawImage(contextRef, CGRectMake(0, 0, cols, rows),
98 image);
99 CGContextRelease(contextRef);
100}
