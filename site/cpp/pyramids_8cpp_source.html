880template<class CastOp> void
881pyrDown_( const Mat& _src, Mat& _dst, int borderType )
882{
883 const int PD_SZ = 5;
884 CV_Assert( !_src.empty() );
885 Size ssize = _src.size(), dsize = _dst.size();
886 int cn = _src.channels();
887
888 AutoBuffer<int> _tabM(dsize.width * cn), _tabL(cn * (PD_SZ + 2)),
889 _tabR(cn * (PD_SZ + 2));
890 int *tabM = _tabM.data(), *tabL = _tabL.data(), *tabR = _tabR.data();
891
892 CV_Assert( ssize.width > 0 && ssize.height > 0 &&
893 std::abs(dsize.width*2 - ssize.width) <= 2 &&
894 std::abs(dsize.height*2 - ssize.height) <= 2 );
895 int width0 = std::min((ssize.width-PD_SZ/2-1)/2 + 1, dsize.width);
896
897 for (int x = 0; x <= PD_SZ+1; x++)
898 {
899 int sx0 = borderInterpolate(x - PD_SZ/2, ssize.width, borderType)*cn;
900 int sx1 = borderInterpolate(x + width0*2 - PD_SZ/2, ssize.width, borderType)*cn;
901 for (int k = 0; k < cn; k++)
902 {
903 tabL[x*cn + k] = sx0 + k;
904 tabR[x*cn + k] = sx1 + k;
905 }
906 }
907
908 for (int x = 0; x < dsize.width*cn; x++)
909 tabM[x] = (x/cn)*2*cn + x % cn;
910
911 int *tabLPtr = tabL;
912 int *tabRPtr = tabR;
913
914 cv::parallel_for_(Range(0,dsize.height), cv::PyrDownInvoker<CastOp>(_src, _dst, borderType, &tabRPtr, &tabM, &tabLPtr), cv::getNumThreads());
915}

917template<class CastOp>
918void PyrDownInvoker<CastOp>::operator()(const Range& range) const
919{
920 const int PD_SZ = 5;
921 typedef typename CastOp::type1 WT;
922 typedef typename CastOp::rtype T;
923 Size ssize = _src->size(), dsize = _dst->size();
924 int cn = _src->channels();
925 int bufstep = (int)alignSize(dsize.width*cn, 16);
926 AutoBuffer<WT> _buf(bufstep*PD_SZ + 16);
927 WT* buf = alignPtr((WT*)_buf.data(), 16);
928 WT* rows[PD_SZ];
929 CastOp castOp;
930
931 int sy0 = -PD_SZ/2, sy = range.start * 2 + sy0, width0 = std::min((ssize.width-PD_SZ/2-1)/2 + 1, dsize.width);
932
933 ssize.width *= cn;
934 dsize.width *= cn;
935 width0 *= cn;
936
937 for (int y = range.start; y < range.end; y++)
938 {
939 T* dst = (T*)_dst->ptr<T>(y);
940 WT *row0, *row1, *row2, *row3, *row4;
941
942 // fill the ring buffer (horizontal convolution and decimation)
943 int sy_limit = y*2 + 2;
944 for( ; sy <= sy_limit; sy++ )
945 {
946 WT* row = buf + ((sy - sy0) % PD_SZ)*bufstep;
947 int _sy = borderInterpolate(sy, ssize.height, _borderType);
948 const T* src = _src->ptr<T>(_sy);
949
950 do {
951 int x = 0;
952 const int* tabL = *_tabL;
953 for( ; x < cn; x++ )
954 {
955 row[x] = src[tabL[x+cn*2]]*6 + (src[tabL[x+cn]] + src[tabL[x+cn*3]])*4 +
956 src[tabL[x]] + src[tabL[x+cn*4]];
957 }
958
959 if( x == dsize.width )
960 break;
961
962 if( cn == 1 )
963 {
964 x += PyrDownVecH<T, WT, 1>(src + x * 2 - 2, row + x, width0 - x);
965 for( ; x < width0; x++ )
966 row[x] = src[x*2]*6 + (src[x*2 - 1] + src[x*2 + 1])*4 +
967 src[x*2 - 2] + src[x*2 + 2];
968 }
969 else if( cn == 2 )
970 {
971 x += PyrDownVecH<T, WT, 2>(src + x * 2 - 4, row + x, width0 - x);
972 for( ; x < width0; x += 2 )
973 {
974 const T* s = src + x*2;
975 WT t0 = s[0] * 6 + (s[-2] + s[2]) * 4 + s[-4] + s[4];
976 WT t1 = s[1] * 6 + (s[-1] + s[3]) * 4 + s[-3] + s[5];
977 row[x] = t0; row[x + 1] = t1;
978 }
979 }
980 else if( cn == 3 )
981 {
982 x += PyrDownVecH<T, WT, 3>(src + x * 2 - 6, row + x, width0 - x);
983 for( ; x < width0; x += 3 )
984 {
985 const T* s = src + x*2;
986 WT t0 = s[0]*6 + (s[-3] + s[3])*4 + s[-6] + s[6];
987 WT t1 = s[1]*6 + (s[-2] + s[4])*4 + s[-5] + s[7];
988 WT t2 = s[2]*6 + (s[-1] + s[5])*4 + s[-4] + s[8];
989 row[x] = t0; row[x+1] = t1; row[x+2] = t2;
990 }
991 }
992 else if( cn == 4 )
993 {
994 x += PyrDownVecH<T, WT, 4>(src + x * 2 - 8, row + x, width0 - x);
995 for( ; x < width0; x += 4 )
996 {
997 const T* s = src + x*2;
998 WT t0 = s[0]*6 + (s[-4] + s[4])*4 + s[-8] + s[8];
999 WT t1 = s[1]*6 + (s[-3] + s[5])*4 + s[-7] + s[9];
1000 row[x] = t0; row[x+1] = t1;
1001 t0 = s[2]*6 + (s[-2] + s[6])*4 + s[-6] + s[10];
1002 t1 = s[3]*6 + (s[-1] + s[7])*4 + s[-5] + s[11];
1003 row[x+2] = t0; row[x+3] = t1;
1004 }
1005 }
1006 else
1007 {
1008 for( ; x < width0; x++ )
1009 {
1010 int sx = (*_tabM)[x];
1011 row[x] = src[sx]*6 + (src[sx - cn] + src[sx + cn])*4 +
1012 src[sx - cn*2] + src[sx + cn*2];
1013 }
1014 }
1015
1016 // tabR
1017 const int* tabR = *_tabR;
1018 for (int x_ = 0; x < dsize.width; x++, x_++)
1019 {
1020 row[x] = src[tabR[x_+cn*2]]*6 + (src[tabR[x_+cn]] + src[tabR[x_+cn*3]])*4 +
1021 src[tabR[x_]] + src[tabR[x_+cn*4]];
1022 }
1023 } while (0);
1024 }
1025
1026 // do vertical convolution and decimation and write the result to the destination image
1027 for (int k = 0; k < PD_SZ; k++)
1028 rows[k] = buf + ((y*2 - PD_SZ/2 + k - sy0) % PD_SZ)*bufstep;
1029 row0 = rows[0]; row1 = rows[1]; row2 = rows[2]; row3 = rows[3]; row4 = rows[4];
1030
1031 int x = PyrDownVecV<WT, T>(rows, dst, dsize.width);
1032 for (; x < dsize.width; x++ )
1033 dst[x] = castOp(row2[x]*6 + (row1[x] + row3[x])*4 + row0[x] + row4[x]);
1034 }
1035}

1038template<class CastOp> void
1039pyrUp_( const Mat& _src, Mat& _dst, int)
1040{
1041 const int PU_SZ = 3;
1042 typedef typename CastOp::type1 WT;
1043 typedef typename CastOp::rtype T;
1044
1045 Size ssize = _src.size(), dsize = _dst.size();
1046 int cn = _src.channels();
1047 int bufstep = (int)alignSize((dsize.width+1)*cn, 16);
1048 AutoBuffer<WT> _buf(bufstep*PU_SZ + 16);
1049 WT* buf = alignPtr((WT*)_buf.data(), 16);
1050 AutoBuffer<int> _dtab(ssize.width*cn);
1051 int* dtab = _dtab.data();
1052 WT* rows[PU_SZ];
1053 T* dsts[2];
1054 CastOp castOp;
1055 //PyrUpVecH<T, WT> vecOpH;
1056
1057 CV_Assert( std::abs(dsize.width - ssize.width*2) == dsize.width % 2 &&
1058 std::abs(dsize.height - ssize.height*2) == dsize.height % 2);
1059 int k, x, sy0 = -PU_SZ/2, sy = sy0;
1060
1061 ssize.width *= cn;
1062 dsize.width *= cn;
1063
1064 for( x = 0; x < ssize.width; x++ )
1065 dtab[x] = (x/cn)*2*cn + x % cn;
1066
1067 for( int y = 0; y < ssize.height; y++ )
1068 {
1069 T* dst0 = _dst.ptr<T>(y*2);
1070 T* dst1 = _dst.ptr<T>(std::min(y*2+1, dsize.height-1));
1071 WT *row0, *row1, *row2;
1072
1073 // fill the ring buffer (horizontal convolution and decimation)
1074 for( ; sy <= y + 1; sy++ )
1075 {
1076 WT* row = buf + ((sy - sy0) % PU_SZ)*bufstep;
1077 int _sy = borderInterpolate(sy*2, ssize.height*2, BORDER_REFLECT_101)/2;
1078 const T* src = _src.ptr<T>(_sy);
1079
1080 if( ssize.width == cn )
1081 {
1082 for( x = 0; x < cn; x++ )
1083 row[x] = row[x + cn] = src[x]*8;
1084 continue;
1085 }
1086
1087 for( x = 0; x < cn; x++ )
1088 {
1089 int dx = dtab[x];
1090 WT t0 = src[x]*6 + src[x + cn]*2;
1091 WT t1 = (src[x] + src[x + cn])*4;
1092 row[dx] = t0; row[dx + cn] = t1;
1093 dx = dtab[ssize.width - cn + x];
1094 int sx = ssize.width - cn + x;
1095 t0 = src[sx - cn] + src[sx]*7;
1096 t1 = src[sx]*8;
1097 row[dx] = t0; row[dx + cn] = t1;
1098
1099 if (dsize.width > ssize.width*2)
1100 {
1101 row[(_dst.cols-1) * cn + x] = row[dx + cn];
1102 }
1103 }
1104
1105 for( x = cn; x < ssize.width - cn; x++ )
1106 {
1107 int dx = dtab[x];
1108 WT t0 = src[x-cn] + src[x]*6 + src[x+cn];
1109 WT t1 = (src[x] + src[x+cn])*4;
1110 row[dx] = t0;
1111 row[dx+cn] = t1;
1112 }
1113 }
1114
1115 // do vertical convolution and decimation and write the result to the destination image
1116 for( k = 0; k < PU_SZ; k++ )
1117 rows[k] = buf + ((y - PU_SZ/2 + k - sy0) % PU_SZ)*bufstep;
1118 row0 = rows[0]; row1 = rows[1]; row2 = rows[2];
1119 dsts[0] = dst0; dsts[1] = dst1;
1120
1121 if (dst0 != dst1)
1122 {
1123 x = PyrUpVecV<WT, T>(rows, dsts, dsize.width);
1124 for( ; x < dsize.width; x++ )
1125 {
1126 T t1 = castOp((row1[x] + row2[x])*4);
1127 T t0 = castOp(row0[x] + row1[x]*6 + row2[x]);
1128 dst1[x] = t1; dst0[x] = t0;
1129 }
1130 }
1131 else
1132 {
1133 x = PyrUpVecVOneRow<WT, T>(rows, dst0, dsize.width);
1134 for( ; x < dsize.width; x++ )
1135 {
1136 T t0 = castOp(row0[x] + row1[x]*6 + row2[x]);
1137 dst0[x] = t0;
1138 }
1139 }
1140
1141 }
1142
1143 if (dsize.height > ssize.height*2)
1144 {
1145 T* dst0 = _dst.ptr<T>(ssize.height*2-2);
1146 T* dst2 = _dst.ptr<T>(ssize.height*2);
1147
1148 for(x = 0; x < dsize.width ; x++ )
1149 {
1150 dst2[x] = dst0[x];
1151 }
1152 }
1153}
