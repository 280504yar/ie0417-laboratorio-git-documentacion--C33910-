144 virtual inline double fastRandom(void){
145 uint64_t x = prng.s[0];
146 uint64_t y = prng.s[1];
147 x ^= x << 23; // a
148 x ^= x >> 17; // b
149 x ^= y ^ (y >> 26); // c
150 prng.s[0] = y;
151 prng.s[1] = x;
152 uint64_t s = x + y;
153
154 return s * 5.421010862427522e-20;/* 2^-64 */
155 }

692inline int RHO_HEST_REFC::ensureCapacity(unsigned N, double beta){
693 if(N == 0){
694 /* Clear. */
695 nr.tbl.clear();
696 nr.size = 0;
697 }else if(nr.beta != beta){
698 /* Beta changed. Redo all the work. */
699 nr.tbl.resize(N);
700 nr.beta = beta;
701 sacInitNonRand(nr.beta, 0, N, &nr.tbl[0]);
702 nr.size = N;
703 }else if(N > nr.size){
704 /* Work is partially done. Do rest of it. */
705 nr.tbl.resize(N);
706 sacInitNonRand(nr.beta, nr.size, N, &nr.tbl[nr.size]);
707 nr.size = N;
708 }else{
709 /* Work is already done. Do nothing. */
710 }
711
712 return 1;
713}

2298 const float* Jte,
2299 const float S,
2300 const float newS,
2301 const float lambda){

2307 for(i=0;i<8;i++){
2308 dL += dH[i]*dH[i];

2313 for(i=0;i<8;i++){

2342 float lambda,
2343 float (*L)[8]){

2345 int i, j, k;
2346 float lambdap1 = lambda + 1.0f;

2348
2349 for(i=0;i<N;i++){/* Row */
2350 /* Pre-diagonal elements */
2351 for(j=0;j<i;j++){
2352 x = A[i][j]; /* Aij */
2353 for(k=0;k<j;k++){
2354 x -= L[i][k] * L[j][k];/* - Sum_{k=0..j-1} Lik*Ljk */

2357 }

2360 {j = i;
2361 x = A[j][j] * lambdap1; /* Ajj */
2362 for(k=0;k<j;k++){
2363 x -= L[j][k] * L[j][k];/* - Sum_{k=0..j-1} Ljk^2 */
2364 }
2365 if(x<0){
2366 return 0;
2367 }

2369 }
2370 }
