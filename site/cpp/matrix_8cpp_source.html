218static void finalizeBlockLayout(MatShape& size, int C=0)
219{
220 if (size.layout == DATA_LAYOUT_BLOCK) {
221 CV_Assert(size.dims >= 4);
222 int C0 = size.p[size.dims-1];
223 CV_Assert(C0 > 1 && (C0 & (C0-1)) == 0);
224 size.C = C > 0 ? C : size.p[1]*size.p[size.dims-1];

226 size.C = 0;
227 }
228 for (int i = std::max(size.dims, 0); i < MatShape::MAX_DIMS; i++)
229 size.p[i] = 0;
230 if (size.dims == 0)
231 size.p[0] = 1;
232}

344MatShape MatShape::toLayout(DataLayout newLayout, int C0) const
345{
346 CV_Assert(layout == DATA_LAYOUT_BLOCK || layout == DATA_LAYOUT_NCHW || layout == DATA_LAYOUT_NHWC);
347 CV_Assert(newLayout == DATA_LAYOUT_BLOCK || newLayout == DATA_LAYOUT_NCHW || newLayout == DATA_LAYOUT_NHWC);
348
349 MatShape newsize = *this;
350 newsize.layout = newLayout;
351
352 if (newLayout == DATA_LAYOUT_BLOCK) {
353 // any => BLOCK
354 CV_Assert_N(C0 > 1, (C0 & (C0-1)) == 0);
355 int Corig = channels();
356 newsize.C = Corig;
357
358 if (layout == DATA_LAYOUT_NHWC) {
359 for (int i = 2; i < dims; i++) {
360 newsize.p[i] = p[i-1];
361 }
362 }
363
364 newsize.dims += layout != DATA_LAYOUT_BLOCK;
365 newsize.p[1] = (Corig + C0 - 1)/C0;
366 newsize.p[newsize.dims-1] = C0;
367 } else if (layout == DATA_LAYOUT_BLOCK) {
368 // BLOCK => any (except for BLOCK, which is handled above)
369 newsize.C = 0;
370 if (newLayout == DATA_LAYOUT_NHWC) {
371 for (int i = 2; i < dims; i++) {
372 newsize.p[i-1] = p[i];
373 }
374 }
375 newsize.p[newLayout == DATA_LAYOUT_NCHW ? 1 : dims-2] = C;
376 newsize.dims--;
377 } else {
378 CV_Assert_N(C0 <= 1);
379
380 if (newLayout == layout)
381 return newsize;
382
383 // NHWC => NCHW
384 if (newLayout == DATA_LAYOUT_NCHW) {
385 for (int i = 2; i < dims; i++)
386 newsize.p[i] = p[i-1];
387 newsize.p[1] = p[dims-1];
388 } else {
389 // NCHW => NHWC
390 for (int i = 2; i < dims; i++)
391 newsize.p[i-1] = p[i];
392 newsize.p[dims-1] = p[1];
393 }
394 }
395
396 return newsize;
397}

399MatShape MatShape::expand(const MatShape& another) const
400{
401 if (dims == 0)
402 return another;
403 if (another.dims == 0)
404 return *this;
405
406 if ((layout == DATA_LAYOUT_NCHW || layout == DATA_LAYOUT_NHWC) &&
407 (another.layout == DATA_LAYOUT_NCHW || another.layout == DATA_LAYOUT_NHWC)) {
408 CV_Assert(layout == another.layout);
409 }
410 // [TODO] support block layout
411 CV_Assert(layout != DATA_LAYOUT_BLOCK && another.layout != DATA_LAYOUT_BLOCK);
412
413 MatShape result;
414
415 if (dims < 0 || another.dims < 0)
416 return result;
417
418 result = *this;
419 result.dims = std::max(dims, another.dims);
420 result.layout = layout == DATA_LAYOUT_UNKNOWN ? another.layout :
421 layout == DATA_LAYOUT_ND && (another.layout == DATA_LAYOUT_NCHW ||
422 another.layout == DATA_LAYOUT_NHWC) ? another.layout : layout;
423 for (int i = result.dims-1; i >= 0; i--) {
424 int i1 = i - (result.dims - dims);
425 int i2 = i - (result.dims - another.dims);
426 int sz1 = i1 < 0 ? 1 : p[i1];
427 int sz2 = i2 < 0 ? 1 : another.p[i2];
428 CV_Assert(sz1 == sz2 || sz1 == 1 || sz2 == 1);
429 // [TODO] handle symbolic shapes
430 result.p[i] = sz1 != 1 ? sz1 : sz2;
431 }
432 return result;
433}

630void setSize( Mat& m, int _dims, const int* _sz, const size_t* _steps, bool autoSteps)
631{
632 CV_Assert( 0 <= _dims && _dims <= CV_MAX_DIM && _dims <= MatShape::MAX_DIMS);
633
634 m.dims = _dims;
635 m.size = MatShape(_dims, _sz);
636 m.step[std::max(_dims-1, 0)] = CV_ELEM_SIZE(m.flags);
637 for (int i = _dims-2; i >= 0; i--) {
638 size_t autostep = m.size[i+1]*m.step[i+1];
639 if (_steps) {
640 m.step[i] = _steps[i];
641 //CV_Assert(m.step[i] >= autostep);
642 } else if (autoSteps) {
643 m.step[i] = autostep;
644 } else {
645 m.step[i] = 0;
646 }
647 }
648
649 if( _dims <= 2 )
650 {
651 m.cols = _dims == 0 ? 1 : _sz ? _sz[_dims > 1] : 0;
652 m.rows = _dims < 2 ? 1 : _sz ? _sz[0] : 0;
653 } else {
654 m.cols = m.rows = -1;
655 }

658int updateContinuityFlag(int flags, int dims, const int* size, const size_t* step)
659{
660 int i, j;
661 if (dims <= 1)
662 return flags | Mat::CONTINUOUS_FLAG;
663 for( i = 0; i < dims; i++ )
664 {
665 if( size[i] > 1 )
666 break;
667 }
668
669 uint64 t = (uint64)size[std::min(i, dims-1)]*CV_MAT_CN(flags);
670 for( j = dims-1; j > i; j-- )
671 {
672 t *= size[j];
673 if( step[j]*size[j] < step[j-1] )
674 break;
675 }
676
677 if( j <= i && t == (uint64)(int)t )
678 return flags | Mat::CONTINUOUS_FLAG;
679 return flags & ~Mat::CONTINUOUS_FLAG;
680}

687void finalizeHdr(Mat& m)
688{
689 m.updateContinuityFlag();
690 int d = m.dims;
691 if( d > 2 )
692 m.rows = m.cols = -1;
693 if(m.u)
694 m.datastart = m.data = m.u->data;
695 if( m.data && d > 0 )
696 {
697 m.datalimit = m.datastart + m.size[0]*m.step[0];
698 if( m.size[0] > 0 )
699 {
700 int lastdim = d > 0 ? d - 1 : 0;
701 m.dataend = m.ptr() + m.size[lastdim]*m.step[lastdim];
702 for( int i = 0; i < d-1; i++ )
703 m.dataend += (m.size[i] - 1)*m.step[i];
704 }
705 else
706 m.dataend = m.datalimit;
707 }
708 else
709 m.dataend = m.datalimit = m.data;
710}

1086{
1087 int sz1 = 1, d = d0;
1088 int i;
1089 if (d == 0) {
1090 d = 1;
1091 _sizes = (const int*)&sz1;
1092 }
1093 CV_Assert(0 <= d && d <= CV_MAX_DIM && _sizes);
1094 _type = CV_MAT_TYPE(_type);
1095
1096 if( data && d == dims && _type == type() )
1097 {
1098 for( i = 0; i < d; i++ )
1099 if( size[i] != _sizes[i] )
1100 break;
1101 if( i == d )
1102 return;
1103 }
1104
1105 int _sizes_backup[CV_MAX_DIM]; // #5991
1106 if (_sizes == (this->size.p))
1107 {
1108 for(i = 0; i < d; i++ )
1109 _sizes_backup[i] = _sizes[i];
1110 _sizes = _sizes_backup;
1111 }
1112
1113 release();
1114 if( d == 0 )
1115 return;
1116 flags = (_type & CV_MAT_TYPE_MASK) | MAGIC_VAL;
1117 setSize(*this, d, _sizes, 0, true);
1118
1119 if( total() > 0 )
1120 {
1121 MatAllocator *a = allocator, *a0 = getDefaultAllocator();
1122 if(!a)
1123 a = a0;
1124 try
1125 {
1126 u = a->allocate(dims, size.p, _type, 0, step.p, ACCESS_RW /* ignored */, USAGE_DEFAULT);
1127 CV_Assert(u != 0);
1128 allocator = a;
1129 }
1130 catch (...)
1131 {
1132 if (a == a0)
1133 throw;
1134 u = a0->allocate(dims, size.p, _type, 0, step.p, ACCESS_RW /* ignored */, USAGE_DEFAULT);
1135 CV_Assert(u != 0);

1137 }

1139 }
1140

1142 finalizeHdr(*this);
1143 size.dims = dims = d0;
