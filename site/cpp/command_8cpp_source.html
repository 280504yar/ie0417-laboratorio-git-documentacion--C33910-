49void CommandBuffer::barrierSource(VkBuffer source, size_t start, size_t size, BarrierType type) const
50{
51 VkBufferMemoryBarrier barrier;
52 barrier.sType = VK_STRUCTURE_TYPE_BUFFER_MEMORY_BARRIER;
53 barrier.buffer = source;
54 barrier.dstQueueFamilyIndex = VK_QUEUE_FAMILY_IGNORED;
55 barrier.srcQueueFamilyIndex = VK_QUEUE_FAMILY_IGNORED;
56 barrier.offset = start;
57 barrier.pNext = nullptr;
58 barrier.size = size;
59 switch (type) {
60 case READ_WRITE:
61 barrier.srcAccessMask = VK_ACCESS_SHADER_WRITE_BIT | VK_ACCESS_TRANSFER_WRITE_BIT;
62 barrier.dstAccessMask = VK_ACCESS_SHADER_READ_BIT | VK_ACCESS_TRANSFER_READ_BIT;
63 break;
64 case WRITE_WRITE:
65 barrier.srcAccessMask = VK_ACCESS_SHADER_WRITE_BIT | VK_ACCESS_TRANSFER_WRITE_BIT;
66 barrier.dstAccessMask = VK_ACCESS_SHADER_WRITE_BIT | VK_ACCESS_SHADER_WRITE_BIT;
67 break;
68 default:
69 break;
70 }
71 vkCmdPipelineBarrier(cmdBuffer, VK_PIPELINE_STAGE_COMPUTE_SHADER_BIT | VK_PIPELINE_STAGE_TRANSFER_BIT,
72 VK_PIPELINE_STAGE_COMPUTE_SHADER_BIT | VK_PIPELINE_STAGE_TRANSFER_BIT, 0, 0, nullptr, 1,
73 &barrier, 0, nullptr);
74}

129void CommandPool::reset()
130{
131 // reset all bufferQueue.
132 while (!bufferQueue.empty())
133 {
134 auto cmdBuffer = bufferQueue.front();
135 bufferQueue.pop();
136
137 vkFreeCommandBuffers(kDevice, cmdPool, 1, &cmdBuffer);

160void CommandPool::submitAndWait(VkCommandBuffer& _buffer) const
161{
162 auto buffer = _buffer;
163 Fence fence = Fence();
164 VkFence fenceVk = fence.get();
165 VkSubmitInfo submit_info = {
166 /* .sType = */ VK_STRUCTURE_TYPE_SUBMIT_INFO,
167 /* .pNext = */ nullptr,
168 /* .waitSemaphoreCount = */ 0,
169 /* .pWaitSemaphores = */ nullptr,
170 /* .pWaitDstStageMask = */ nullptr,
171 /* .commandBufferCount = */ 1,
172 /* .pCommandBuffers = */ &buffer,
173 /* .signalSemaphoreCount = */ 0,
174 /* .pSignalSemaphores = */ nullptr};
175 // need the queue class.
176 VK_CHECK_RESULT(vkQueueSubmit(queue, 1, &submit_info, fenceVk));
177 VK_CHECK_RESULT(fence.wait());
178}
