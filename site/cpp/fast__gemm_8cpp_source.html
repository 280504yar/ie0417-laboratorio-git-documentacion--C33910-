313static inline int fast_gemm_thin_lanes() {
314#if (CV_SIMD || CV_SIMD_SCALABLE)

318#endif

321#if CV_SIMD_SCALABLE
322// Apply the alpha/beta epilogue to one accumulator row.
323static inline void fast_gemm_thin_store_row(float* c, v_float32 acc,
324 v_float32 v_alpha, float beta) {
325 if (beta == 0.f)
326 vx_store(c, v_mul(acc, v_alpha));
327 else if (beta == 1.f)
328 vx_store(c, v_fma(acc, v_alpha, vx_load(c)));
329 else
330 vx_store(c, v_fma(acc, v_alpha, v_mul(vx_load(c), vx_setall_f32(beta))));
331}
332
333// Multiply MR (1..4) rows of A by one packed-B strip. Sizeless vector types
334// cannot form the acc[] array the CV_SIMD path uses, so the accumulators are
335// named variables selected by the compile-time MR (the `if (MR > n)` branches
336// fold away) and stay in registers for the whole K loop. Accumulation remains
337// in increasing-k order, bit-exact with the previous scratch-buffer code.
338template<int MR>
339static inline void fast_gemm_thin_block(int K, const float* A, int lda0, int lda1,
340 const float* b_strip, int NR,
341 v_float32 v_alpha, float beta,
342 float* c_strip, int ldc) {
343 v_float32 c0 = vx_setzero_f32(), c1 = vx_setzero_f32();
344 v_float32 c2 = vx_setzero_f32(), c3 = vx_setzero_f32();
345 for (int k = 0; k < K; k++) {
346 v_float32 bv = vx_load(b_strip + k * NR);
347 const float* a = A + k * lda1;
348 c0 = v_fma(bv, vx_setall_f32(a[0]), c0);
349 if (MR > 1) c1 = v_fma(bv, vx_setall_f32(a[lda0]), c1);
350 if (MR > 2) c2 = v_fma(bv, vx_setall_f32(a[2 * lda0]), c2);
351 if (MR > 3) c3 = v_fma(bv, vx_setall_f32(a[3 * lda0]), c3);
352 }
353 fast_gemm_thin_store_row(c_strip, c0, v_alpha, beta);
354 if (MR > 1) fast_gemm_thin_store_row(c_strip + ldc, c1, v_alpha, beta);
355 if (MR > 2) fast_gemm_thin_store_row(c_strip + 2 * ldc, c2, v_alpha, beta);
356 if (MR > 3) fast_gemm_thin_store_row(c_strip + 3 * ldc, c3, v_alpha, beta);
357}
358#endif // CV_SIMD_SCALABLE

360static inline void fast_gemm_thin_strip(int M, int K, float alpha,
361 const float* A, int lda0, int lda1,
362 const float* b_strip,
363 float beta, float* c_strip, int ldc) {
364#if CV_SIMD
365 const int NR = VTraits<v_float32>::vlanes();
366 v_float32 acc[FAST_GEMM_THIN_MAX_M];
367 for (int m = 0; m < M; m++) acc[m] = vx_setzero_f32();
368
369 for (int k = 0; k < K; k++) {
370 v_float32 bv = vx_load(b_strip + k * NR);
371 for (int m = 0; m < M; m++) {
372 v_float32 am = vx_setall_f32(A[m * lda0 + k * lda1]);
373 acc[m] = v_fma(bv, am, acc[m]);
374 }
375 }
376
377 const v_float32 v_alpha = vx_setall_f32(alpha);
378 if (beta == 0.f) {
379 for (int m = 0; m < M; m++)
380 vx_store(c_strip + m * ldc, v_mul(acc[m], v_alpha));
381 } else if (beta == 1.f) {
382 for (int m = 0; m < M; m++) {
383 v_float32 cv = vx_load(c_strip + m * ldc);
384 vx_store(c_strip + m * ldc, v_fma(acc[m], v_alpha, cv));
385 }
386 } else {
387 const v_float32 v_beta = vx_setall_f32(beta);
388 for (int m = 0; m < M; m++) {
389 v_float32 cv = vx_load(c_strip + m * ldc);
390 cv = v_mul(cv, v_beta);
391 vx_store(c_strip + m * ldc, v_fma(acc[m], v_alpha, cv));
392 }
393 }
394#elif CV_SIMD_SCALABLE
395 // Process the strip in register-resident row blocks; the common thin case
396 // M <= 4 (transformer token generation) is a single pass.
397 const int NR = VTraits<v_float32>::vlanes();
398 const v_float32 v_alpha = vx_setall_f32(alpha);
399 int m = 0;
400 for (; m + 4 <= M; m += 4)
401 fast_gemm_thin_block<4>(K, A + m * lda0, lda0, lda1, b_strip, NR, v_alpha, beta, c_strip + m * ldc, ldc);
402 switch (M - m) {
403 case 1: fast_gemm_thin_block<1>(K, A + m * lda0, lda0, lda1, b_strip, NR, v_alpha, beta, c_strip + m * ldc, ldc); break;
404 case 2: fast_gemm_thin_block<2>(K, A + m * lda0, lda0, lda1, b_strip, NR, v_alpha, beta, c_strip + m * ldc, ldc); break;
405 case 3: fast_gemm_thin_block<3>(K, A + m * lda0, lda0, lda1, b_strip, NR, v_alpha, beta, c_strip + m * ldc, ldc); break;
406 }
407#else
408 const int NR = FAST_GEMM_THIN_SCALAR_NR;
409 float acc[FAST_GEMM_THIN_MAX_M * FAST_GEMM_THIN_SCALAR_NR] = {0};
410 for (int k = 0; k < K; k++) {
411 for (int m = 0; m < M; m++) {
412 float a_mk = A[m * lda0 + k * lda1];
413 for (int c = 0; c < NR; c++)
414 acc[m * NR + c] += a_mk * b_strip[k * NR + c];
415 }
416 }
417 if (beta == 0.f) {
418 for (int m = 0; m < M; m++)
419 for (int c = 0; c < NR; c++)
420 c_strip[m * ldc + c] = alpha * acc[m * NR + c];
421 } else {
422 for (int m = 0; m < M; m++)
423 for (int c = 0; c < NR; c++)
424 c_strip[m * ldc + c] = beta * c_strip[m * ldc + c] + alpha * acc[m * NR + c];
425 }
426#endif
427}

429bool fastGemmThinEligible(int M, int N, int K) {
430 if (M <= 0 || N <= 0 || K <= 0) return false;
431 if (M > FAST_GEMM_THIN_MAX_M) return false;
432 const int NR = fast_gemm_thin_lanes();
433 if (N < 2 * NR) return false;
434 return true;
435}

443void fastGemmThinPackB(int N, int K, const float* B, size_t ldb_K, size_t ldb_N, float* packed_B) {
444 const int NR = fast_gemm_thin_lanes();
445 const int n_strips = (N + NR - 1) / NR;
446 for (int s = 0; s < n_strips; s++) {
447 const int j0 = s * NR;
448 const int nr = std::min(NR, N - j0);
449 float* strip = packed_B + (size_t)s * NR * K;
450 for (int k = 0; k < K; k++) {
451 float* out = strip + k * NR;
452 int c = 0;
453 for (; c < nr; c++) out[c] = B[k * ldb_K + (j0 + c) * ldb_N];
454 for (; c < NR; c++) out[c] = 0.f;
455 }
456 }
457}

459void fastGemmThin(int M, int N, int K, float alpha,
460 const float* A, int lda0, int lda1,
461 const float* packed_B, float beta,
462 float* C, int ldc, bool multi_thread) {
463 const int NR = fast_gemm_thin_lanes();
464 const int n_full_strips = N / NR;
465 const int n_tail = N - n_full_strips * NR;
466
467 auto fn = [&](const Range& r) {
468 for (int s = r.start; s < r.end; s++) {
469 const float* b_strip = packed_B + (size_t)s * NR * K;
470 float* c_strip = C + s * NR;
471 fast_gemm_thin_strip(M, K, alpha, A, lda0, lda1, b_strip, beta, c_strip, ldc);
472 }
473 };
474 if (multi_thread && n_full_strips > 1) {
475 parallel_for_(Range(0, n_full_strips), fn, (double)n_full_strips * M * K * NR * (1.0 / 1024.0));
476 } else {
477 fn(Range(0, n_full_strips));
478 }
479
480 if (n_tail > 0) {
481 const int j0 = n_full_strips * NR;
482 const float* b_strip = packed_B + (size_t)n_full_strips * NR * K;
483 for (int m = 0; m < M; m++) {
484 for (int c = 0; c < n_tail; c++) {
485 float acc_s = 0.f;
486 for (int k = 0; k < K; k++)
487 acc_s += A[m * lda0 + k * lda1] * b_strip[k * NR + c];
488 const float mul = alpha * acc_s;
489 float* p = C + m * ldc + j0 + c;
490 *p = (beta == 0.f) ? mul : beta * (*p) + mul;
491 }
492 }
493 }
494}

496static void fast_gemm_thin(float alpha, float beta, int M, int N, int K,
497 const char *a_, int lda0, int lda1,
498 const char *b_, int ldb,
499 char *c_, int ldc, bool multi_thread) {
500 const float* A = (const float*)a_;
501 const float* B = (const float*)b_;
502 float* C = (float*)c_;
503
504 if (fastGemmThinEligible(M, N, K)) {
505 AutoBuffer<float> packed(fastGemmThinPackBSize(N, K));
506 fastGemmThinPackB(N, K, B, (size_t)ldb, 1, packed.data());
507 fastGemmThin(M, N, K, alpha, A, lda0, lda1, packed.data(), beta, C, ldc, multi_thread);
508 return;
509 }
510
511 auto fn = [&](const Range &r) {
512 for (int i = r.start; i < r.end; i++) {
513 float* c_i = C + i * ldc;
514 if (beta == 0.f)
515 for (int j = 0; j < N; j++) c_i[j] = 0.f;
516 else if (beta != 1.f)
517 for (int j = 0; j < N; j++) c_i[j] *= beta;
518 for (int k = 0; k < K; k++) {
519 const float* b_k = B + k * ldb;
520 float aval = alpha * A[i * lda0 + k * lda1];
521 for (int j = 0; j < N; j++)
522 c_i[j] += aval * b_k[j];
523 }
524 }
525 };
526 if (multi_thread) {
527 parallel_for_(Range(0, M), fn, (double)M * K * N * (1.0 / 1024.0));
528 } else {
529 fn(Range(0, M));
530 }
531}
