41 bool getMemoryShapes(const std::vector<MatShape> &inputs,
42 const int requiredOutputs,
43 std::vector<MatShape> &outputs,
44 std::vector<MatShape> &internals) const CV_OVERRIDE {
45 const auto &input = inputs[0];
46 const auto &scale = inputs[1];
47 const auto &bias = inputs[2];
48 CV_CheckGE(input.size(), static_cast<size_t>(3), "DNN/GroupNorm: input dimension >= 3 is required");
49
50 int C = input.layout == DATA_LAYOUT_BLOCK ? input.C : input[1];
51 int scale_dim = std::accumulate(scale.begin(), scale.end(), 1, std::multiplies<int>());
52 CV_CheckEQ(scale_dim, C, "DNN/InstanceNorm: scale must be a 1d tensor and match the channel of input");
53 int bias_dim = std::accumulate(bias.begin(), bias.end(), 1, std::multiplies<int>());
54 CV_CheckEQ(bias_dim, C, "DNN/InstanceNorm: bias must be a 1d tensor and match the channel of input");
55
56 outputs.assign(1, inputs[0]);
57 return false;
58 }

90#ifdef HAVE_OPENCL
91 bool forward_ocl(InputArrayOfArrays inputs_, OutputArrayOfArrays outputs_, OutputArrayOfArrays internals_) {
92 std::vector<UMat> inputs;
93 std::vector<UMat> outputs;
94
95 inputs_.getUMatVector(inputs);
96 outputs_.getUMatVector(outputs);
97
98 const auto &input = inputs[0], &scale = inputs[1], &bias = inputs[2];
99 auto &output = outputs[0];
100
101 const auto input_shape = shape(input);
102 size_t N = input_shape[0], C = input_shape[1];
103 size_t num_groups = this->num_groups;
104 size_t channels_per_group = C / num_groups;
105 size_t loops = N * num_groups, norm_size = static_cast<size_t>(total(input_shape, 2)) * channels_per_group;
106 float inv_norm_size = 1.f / norm_size;
107
108 // no fp16 support
109 if (input.depth() == CV_16F) {
110 return false;
111 }
112
113 String base_opts = format(" -DT=float -DT4=float4 -Dconvert_T=convert_float4");
114
115 // Calculate mean
116 UMat one = UMat::ones(norm_size, 1, CV_32F);
117 UMat mean = UMat(loops, 1, CV_32F);
118 UMat mean_square = UMat(loops, 1, CV_32F);
119 UMat tmp = UMat(loops, norm_size, CV_32F);
120 bool ret = ocl4dnn::ocl4dnnGEMV<float>(ocl4dnn::CblasNoTrans, loops, norm_size, inv_norm_size,
121 input, 0, one, 0, 0.f, mean, 0);
122 if (!ret) {
123 return false;
124 }
125 // Calculate mean_square
126 int num_vector = (norm_size % 8 == 0) ? 8 : ((norm_size % 4 == 0) ? 4 : 1);
127 size_t global[] = {loops, static_cast<size_t>(norm_size / num_vector)};
128 String build_opt = format(" -DNUM=%d", num_vector) + base_opts;
129 String mean_square_kernel_name = format("calc_mean%d", num_vector);
130 ocl::Kernel mean_square_kernel(mean_square_kernel_name.c_str(), ocl::dnn::mvn_oclsrc, build_opt + " -DKERNEL_MEAN");
131 if (mean_square_kernel.empty()) {
132 return false;
133 }
134 mean_square_kernel.set(0, ocl::KernelArg::PtrReadOnly(input));
135 mean_square_kernel.set(1, (int)loops);
136 mean_square_kernel.set(2, (int)norm_size);
137 mean_square_kernel.set(3, ocl::KernelArg::PtrReadOnly(mean));
138 mean_square_kernel.set(4, ocl::KernelArg::PtrWriteOnly(tmp));
139 ret = mean_square_kernel.run(2, global, NULL, false);
140 if (!ret) {
141 return false;
142 }
143 ret = ocl4dnn::ocl4dnnGEMV<float>(ocl4dnn::CblasNoTrans, loops, norm_size, inv_norm_size,
144 tmp, 0, one, 0, 0.f, mean_square, 0);
145 if (!ret) {
146 return false;
147 }
148 // Calculate group norm: output = scale * (x - mean) / sqrt(var + eps) + bias
149 String mvn_group_kernel_name = format("mvn_group%d", num_vector);
150 build_opt += " -DNORM_VARIANCE -DKERNEL_MVN_GROUP";
151 ocl::Kernel mvn_group_kernel(mvn_group_kernel_name.c_str(), ocl::dnn::mvn_oclsrc, build_opt);
152 if (mvn_group_kernel.empty()) {
153 return false;
154 }
155 mvn_group_kernel.set(0, ocl::KernelArg::PtrReadOnly(input));
156 mvn_group_kernel.set(1, (int)loops);
157 mvn_group_kernel.set(2, (int)norm_size);
158 mvn_group_kernel.set(3, (float)epsilon);
159 mvn_group_kernel.set(4, ocl::KernelArg::PtrReadOnly(mean));
160 mvn_group_kernel.set(5, ocl::KernelArg::PtrReadOnly(mean_square));
161 mvn_group_kernel.set(6, ocl::KernelArg::PtrReadOnly(scale));
162 mvn_group_kernel.set(7, ocl::KernelArg::PtrReadOnly(bias));
163 mvn_group_kernel.set(8, (int)C);
164 mvn_group_kernel.set(9, (int)num_groups);
165 mvn_group_kernel.set(10, (float)0.f);
166 mvn_group_kernel.set(11, ocl::KernelArg::PtrWriteOnly(output));
167 ret = mvn_group_kernel.run(2, global, NULL, false);
168 if (!ret) {
169 return false;
170 }
171
172 return true;
173 }
