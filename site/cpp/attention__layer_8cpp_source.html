14static void packWeight(size_t num_heads, size_t head_size, size_t input_hidden_size,
15 const float *weight_data, size_t hidden_size, std::vector<float> &packed_weight, const FastGemmOpt &opt) {
16 // num_heads * pack(head_size, input_hidden_size)
17 size_t pack_size = fastGemmPackBSize(head_size, input_hidden_size, opt);
18 size_t packed_weight_size = num_heads * pack_size;
19 packed_weight.resize(packed_weight_size, 0.f);
20 auto *packed_weight_data = packed_weight.data();
21 for (size_t i = 0; i < num_heads; i++) {
22 fastGemmPackB(false, head_size, input_hidden_size, weight_data, hidden_size, packed_weight_data, opt);
23 packed_weight_data += pack_size;
24 weight_data += head_size;
25 }
26}

29static void rotationKernel(
30 float* data, const float* rotation_table,
31 size_t seq_len, size_t d
32)
33{
34 CV_Assert(d % 2 == 0);
35 const size_t d_half = d / 2;
36
37 double nstripes = double(seq_len) * d_half * (1.0/1024.0);
38
39 auto fn = [&](const cv::Range& range)
40 {
41 for (int t = range.start; t < range.end; ++t)
42 {
43 float* out_ptr = data + size_t(t) * d;
44 const float* table_ptr = rotation_table + size_t(t) * d;
45 size_t i = 0;
46
47#if (CV_SIMD || CV_SIMD_SCALABLE)
48 const size_t w = VTraits<v_float32>::vlanes();
49 for (; i + w <= d_half; i += w)
50 {
51 v_float32 sin_v, cos_v, x_even, x_odd;
52 v_load_deinterleave(table_ptr + 2*i, sin_v, cos_v);
53 v_load_deinterleave(out_ptr + 2*i, x_even, x_odd);
54
55 v_float32 out_even = v_sub(v_mul(cos_v, x_even), v_mul(sin_v, x_odd));
56 v_float32 out_odd = v_add(v_mul(sin_v, x_even), v_mul(cos_v, x_odd));
57
58 v_store_interleave(out_ptr + 2*i, out_even, out_odd);
59 }
60#endif
61 // scalar tail
62 for (; i < d_half; ++i)
63 {
64 float s = table_ptr[2*i ];
65 float c = table_ptr[2*i+1];
66 float xe = out_ptr[2*i];
67 float xo = out_ptr[2*i+1];
68 out_ptr[2*i] = xe * c - xo * s;
69 out_ptr[2*i+1] = xo * c + xe * s;
70 }
71 }
72 };
73 // This will spin up threads and run fn over [0, seq_len)
74 parallel_for_(cv::Range(0, int(seq_len)), fn, nstripes);
75}

77// Precomputes RoPE sin/cos table of shape [seq_len, d] (https://arxiv.org/pdf/2104.09864).
78static void precompRotationTable(float *data,
79 size_t seq_len,
80 size_t d) {
81 // RoPE precomputation
82 // RoPE is a positional encoding method used in transformer models.
83 // It uses sine and cosine functions to encode the position of tokens in a sequence
84 // initially introduced for NLP in https://arxiv.org/pdf/2104.09864
85
86 // assume data is of shape [seq_ken,d]
87 const float logBase = std::log(10000.0f);
88 const float inv_d = 1.0f / float(d);
89 const size_t d_half = d / 2;
90 for (size_t pos = 0; pos < seq_len; ++pos) {
91
92 size_t i = 0;
93 float* data_ptr = data + pos * d;
94
95#if (CV_SIMD || CV_SIMD_SCALABLE)
96 const size_t w = VTraits<v_float32>::vlanes();
97 const v_float32 v_logBase = vx_setall_f32(logBase);
98 const v_float32 v_inv_d = vx_setall_f32(inv_d);
99 const v_float32 v_neg2 = vx_setall_f32(-2.0f);
100
101 for (; i + w <= d_half; i+=w) {
102 int idx_buf[VTraits<v_float32>::max_nlanes];
103 for (int k = 0; k < int(w); ++k)
104 idx_buf[k] = int(i + k);
105 // [i, i+1, …, i+w-1]
106 v_float32 v_idx = v_cvt_f32(vx_load(idx_buf));
107 // [10_000^(-i/d), 10_000^(-(i+1)/d), …, 10_000^(-(i+w-1)/d)]
108 v_float32 v_theta = v_exp(v_mul(v_mul(v_neg2, v_mul(v_idx, v_inv_d)), v_logBase));
109 v_theta = v_mul(vx_setall_f32(float(pos)), v_theta);
110 v_float32 sin_v, cos_v;
111 v_sincos(v_theta, sin_v, cos_v);
112 // store back with interleave
113 v_store_interleave(data_ptr + 2*i, sin_v, cos_v);
114 }
115#endif
116 // scalar tail
117 for (; i < d_half; i+=1)
118 {
119 float theta = pos * std::exp(-2.f * i * inv_d * logBase);
120 data_ptr[2*i ] = std::sin(theta);
121 data_ptr[2*i + 1] = std::cos(theta);
122 }
123 }
124}

127// Operator spec: https://github.com/microsoft/onnxruntime/blob/v1.16.1/docs/ContribOperators.md#com.microsoft.Attention
128class AttentionLayerImpl CV_FINAL : public AttentionLayer {
129 public:
130 AttentionLayerImpl(const LayerParams &params) {
131 setParamsFrom(params);
132
133 CV_CheckTrue(params.has("num_heads"), "DNN/Attention: num_heads is required but missing");
134 num_heads = params.get<int>("num_heads"); // required, no default value
135
136 CV_CheckTrue(params.has("qkv_hidden_sizes"), "DNN/Attention: qkv_hidden_sizes is required but missing");
137 auto param_qkv_hidden_sizes = params.get("qkv_hidden_sizes");
138 CV_CheckEQ(param_qkv_hidden_sizes.size(), 3, "DNN/Attention: qkv_hidden_sizes must and only have three elements");
139
140 qkv_hidden_sizes.clear();
141 qkv_hidden_sizes.resize(3);
142 qkv_hidden_sizes[0] = static_cast<size_t>(param_qkv_hidden_sizes.get<int>(0));
143 qkv_hidden_sizes[1] = static_cast<size_t>(param_qkv_hidden_sizes.get<int>(1));
144 /* v_hidden_size needs to be initialized in finalize in case v_slice_end=INT_MAX */
145
146 qkv_head_sizes.clear();
147 qkv_head_sizes.resize(3);
148 qkv_head_sizes[0] = static_cast<size_t>(qkv_hidden_sizes[0] / num_heads);
149 qkv_head_sizes[1] = static_cast<size_t>(qkv_hidden_sizes[1] / num_heads);
150
151 scale = 1.f / params.get<float>("scale", sqrt(qkv_head_sizes[0]));
152
153 output_ndims = params.get<int>("output_ndims", 3);
154
155 do_rotary = params.get<bool>("do_rotary", false);
156 }

162 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
163 const int requiredOutputs,
164 std::vector<MatShape> &outputs,
165 std::vector<MatShape> &internals) const CV_OVERRIDE {
166 int num_inputs = inputs.size() + blobs.size();
167 CV_CheckGE(num_inputs, 3, "DNN/Attention: at least three inputs are required (data, weight, bias)");
168 CV_CheckLE(num_inputs, 4, "DNN/Attention: at most four inputs are supported (data, weight, bias, mask)");
169 const auto &input_shape = inputs[0];
170 const auto &weight_shape = blobs.empty() ? inputs[1] : shape(blobs.front());
171 const auto &bias_shape = blobs.empty() ? inputs[2] : shape(blobs.back());
172
173 CV_CheckEQ(input_shape.size(), static_cast<size_t>(3), "DNN/Attention: invalid input dimension");
174 CV_CheckEQ(weight_shape.size(), static_cast<size_t>(2), "DNN/Attention: invalid weight dimension");
175
176 CV_CheckEQ(input_shape[2], weight_shape[0], "DNN/Attention: invalid input shape");
177 CV_CheckEQ(weight_shape[1], bias_shape[0], "DNN/Attention: invalid weight or bias shape");
178
179 if (output_ndims == 3) {
180 outputs.assign(1, inputs[0]);
181 } else if (output_ndims == 2) {
182 int batch = input_shape[0], seq_len = input_shape[1], input_hidden_size = input_shape[2];
183 MatShape output_shape{batch * seq_len, input_hidden_size};
184 outputs.assign(1, output_shape);
185 } else {
186 CV_Error(Error::StsBadArg, format("DNN/Attention: invalid output dimension %zu, valid value is 2 or 3", output_ndims));
187 }
188
189 const int batch_size_ = input_shape[0], seq_len_ = input_shape[1],
190 hidden_size_ = weight_shape.back(),
191 num_heads_ = static_cast<int>(num_heads),
192 v_head_size_ = static_cast<int>((hidden_size_ - qkv_hidden_sizes[0] - qkv_hidden_sizes[1]) / num_heads);
193
194 MatShape gemm_buffer_shape{batch_size_, seq_len_, hidden_size_},
195 attention_prob_shape{batch_size_ * num_heads_, seq_len_, seq_len_},
196 output_buffer_shape{batch_size_ * num_heads_, seq_len_, v_head_size_};
197 internals.assign(1, gemm_buffer_shape);
198 internals.push_back(attention_prob_shape);
199 internals.push_back(output_buffer_shape);
200
201 if (do_rotary)
202 {
203 CV_Assert(qkv_head_sizes[0] == qkv_head_sizes[1]);
204 const int d = qkv_head_sizes[0];
205 CV_Assert(d % 2 == 0);
206 // pick maximum of q and k head dim
207
208 MatShape rotation_table_shape{seq_len_, d};
209 internals.push_back(rotation_table_shape);
210 }
211
212 return false;
213 }

215 virtual int64 getFLOPS(const std::vector<MatShape> &inputs,
216 const std::vector<MatShape> &outputs) const CV_OVERRIDE
217 {
218 const auto &input_shape = inputs[0];
219 int64 B = input_shape[0];
220 int64 S = input_shape[1];
221 int64 D = input_shape[2];
222
223 const auto &weight_shape = blobs.empty() ? inputs[1] : shape(blobs.front());
224 int64 hidden = weight_shape[1];
225
226 int64 q_size = (int64)qkv_hidden_sizes[0];
227 int64 k_size = (int64)qkv_hidden_sizes[1];
228 int64 v_size = hidden - q_size - k_size;
229 int64 q_head = (int64)qkv_head_sizes[0];
230 int64 v_head = v_size / (int64)num_heads;
231
232 // Input projection: Q, K, V = input * W + b
233 int64 flops = B * S * (CV_BIG_INT(2) * D * hidden);
234 // QK^T: (B*num_heads) * S * S * q_head_size
235 flops += B * (int64)num_heads * CV_BIG_INT(2) * S * S * q_head;
236 // Softmax: ~4 ops per element
237 flops += B * (int64)num_heads * 4 * S * S;
238 // Attention * V: (B*num_heads) * S * v_head_size * S
239 flops += B * (int64)num_heads * CV_BIG_INT(2) * S * v_head * S;
240 return flops;
241 }

243 // Geometry derives from the weight shape alone, so this is safe before finalize().
244 void packQKV(const Mat& weight) {
245 opt.init();
246 const auto weight_shape = shape(weight);
247 input_hidden_size = static_cast<size_t>(weight_shape[0]);
248 hidden_size = weight_shape[1];
249 qkv_hidden_sizes[2] = hidden_size - qkv_hidden_sizes[0] - qkv_hidden_sizes[1];
250 qkv_head_sizes[2] = static_cast<size_t>(qkv_hidden_sizes[2] / num_heads);
251
252 const auto *weight_data = weight.ptr<const float>();
253 packWeight(num_heads, qkv_head_sizes[0], input_hidden_size, weight_data, hidden_size, packed_weight_q, opt);
254 packWeight(num_heads, qkv_head_sizes[1], input_hidden_size, weight_data + qkv_hidden_sizes[0], hidden_size, packed_weight_k, opt);
255 packWeight(num_heads, qkv_head_sizes[2], input_hidden_size, weight_data + qkv_hidden_sizes[0] + qkv_hidden_sizes[1], hidden_size, packed_weight_v, opt);
256 }

264 virtual void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr) CV_OVERRIDE {
265 opt.init();
266
267 std::vector<Mat> inputs;
268 inputs_arr.getMatVector(inputs);
269 const auto input_shape = shape(inputs[0]);
270 batch_size = static_cast<size_t>(input_shape[0]);
271 seq_len = static_cast<size_t>(input_shape[1]);
272 input_hidden_size = static_cast<size_t>(input_shape[2]);
273
274 const auto &weight = blobs.empty() ? inputs[1] : blobs.front();
275 const auto weight_shape = shape(weight);
276 hidden_size = weight_shape[1];
277 qkv_hidden_sizes[2] = hidden_size - qkv_hidden_sizes[0] - qkv_hidden_sizes[1];
278 qkv_head_sizes[2] = static_cast<size_t>(qkv_hidden_sizes[2] / num_heads);

281 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE {
282 CV_TRACE_FUNCTION();
283 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
284
285 if (inputs_arr.depth() == CV_16F)
286 {
287 forward_fallback(inputs_arr, outputs_arr, internals_arr);
288 return;
289 }
290
291 std::vector<Mat> inputs, outputs, internals;
292 inputs_arr.getMatVector(inputs);
293 outputs_arr.getMatVector(outputs);
294 internals_arr.getMatVector(internals);
295
296 // Dynamic weight (blobs empty) may change each call, so repack it.
297 if (blobs.empty())
298 packQKV(inputs[1]);
299 // Const weight: pack once as fallback when prepackWeights() wasn't called.
300 else if (packed_weight_q.empty())
301 packQKV(blobs.front());
302
303 float *packed_weights[3] = {packed_weight_q.data(), packed_weight_k.data(), packed_weight_v.data()};
304 size_t packed_weights_size[3] = {packed_weight_q.size() / num_heads, packed_weight_k.size() / num_heads, packed_weight_v.size() / num_heads};
305 CV_Assert(internals.size() == 3 + (do_rotary ? 1 : 0));
306
307 if (do_rotary)
308 {
309 auto &rope_table = internals.back();
310 auto *rope_table_data = rope_table.ptr<float>();
311 // RoPE currently requires q and k head sizes to match.
312 CV_Assert(qkv_head_sizes[0] == qkv_head_sizes[1]);
313 precompRotationTable(rope_table_data, seq_len, qkv_head_sizes[0]);
314 }
315
316 auto &gemm_buffer = internals[0];
317 auto *Q = gemm_buffer.ptr<float>();
318 auto *K = Q + batch_size * seq_len * qkv_hidden_sizes[0];
319 auto *V = K + batch_size * seq_len * qkv_hidden_sizes[1];
320 float *QKV[3] = {Q, K, V}; // [B, N, S, H] per tensor
321 {
322 const auto &input = inputs[0];
323 const auto &bias = blobs.empty() ? inputs[2] : blobs.back();
324 const auto *input_data = input.ptr<const float>();
325 const auto *bias_data = bias.ptr<const float>();
326
327 // When do_rotary is false, internals.back() aliases output_buffer; harmless because rope_table is then unused.
328 const auto &rope_table = internals.back();
329
330 opt.multi_thread = false;
331 auto fn = [&](const Range &r) {
332 for (int i = r.start; i < r.end; i++) {
333 const int batch_index = static_cast<int>((i / 3) / num_heads);
334 const int head_index = static_cast<int>((i / 3) % num_heads);
335 const int qkv_index = static_cast<int>(i % 3);
336
337 auto *dst = QKV[qkv_index];
338 size_t head_size = qkv_head_sizes[qkv_index];
339
340 int input_offset = batch_index * seq_len * input_hidden_size;
341 int bias_offset = qkv_index * qkv_hidden_sizes[0] + head_index * head_size;
342 int dst_offset = (batch_index * num_heads + head_index) * (seq_len * head_size);
343
344 // Broadcast bias [NH] -> [BN, SH] into dst before gemm so beta=1 adds it implicitly.
345 const auto *bias_data_src = bias_data + bias_offset;
346 auto *dst_data = dst + dst_offset;
347 for (size_t seq_len_idx = 0; seq_len_idx < seq_len; seq_len_idx++) {
348 std::memcpy(dst_data, bias_data_src, head_size * sizeof(float));
349 dst_data += head_size;
350 }
351
352 auto *packed_weight = packed_weights[qkv_index] + packed_weights_size[qkv_index] * head_index;
353 // Single-threaded gemm; outer parallel_for_ already partitions over (batch, head, qkv).
354 fastGemm(false, seq_len, head_size, input_hidden_size,
355 1.f, input_data + input_offset, input_hidden_size,
356 packed_weight, 1.f, dst + dst_offset, head_size, opt);
357
358 if(qkv_index < 2 && do_rotary) {
359 // Apply RoPE to Q/K in place (V is left untouched).
360 const auto *rope_table_data = rope_table.ptr<const float>();
361 rotationKernel(
362 dst + dst_offset,
363 rope_table_data,
364 seq_len,
365 qkv_head_sizes[qkv_index]
366 );
367 }
368 }
369 };
370
371 size_t loops = 3 * batch_size * num_heads;
372 double nstripes = loops * seq_len * qkv_head_sizes[0] * input_hidden_size * (1 / 1024.0);
373 parallel_for_(Range(0, loops), fn, nstripes);
374 }
375
376 const int num_non_blob_inputs = (int)inputs.size();
377 const bool has_mask = (!blobs.empty() && num_non_blob_inputs >= 2) ||
378 ( blobs.empty() && num_non_blob_inputs >= 4);
379
380 if (mlasAvailable() && !has_mask &&
381 batch_size > 0 && num_heads > 0 && seq_len > 0 &&
382 qkv_head_sizes[0] > 0 && qkv_head_sizes[2] > 0)
383 {
384 const int B = (int)batch_size;
385 const int H = (int)num_heads;
386 const int S = (int)seq_len;
387 const int Dqk = (int)qkv_head_sizes[0];
388 const int Dv = (int)qkv_head_sizes[2];
389 // ORT-style default tiling, clamped to the actual sequence length.
390 const int q_block = std::min(256, S);
391 const int kv_block = std::min(256, S);
392 const int threads = std::max(1, cv::getNumThreads());
393
394 const size_t per_thread =
395 mlasFlashAttentionBufferBytesPerThread(q_block, kv_block, Dv);
396 flash_scratch.resize((size_t)threads * per_thread);
397
398 if (mlasFlashAttention(Q, K, V,
399 outputs[0].ptr<float>(),
400 B, H, S, S, Dqk, Dv, scale,
401 q_block, kv_block,
402 flash_scratch.data(), threads))
403 {
404 return; // fast path done; outputs[0] is fully written
405 }
406 }
407
408 // Compute Softmax(scale * MatMul(Q, K))
409 auto &attention_prob = internals[1];
410 {
411 auto *output = attention_prob.ptr<float>();
412
413 auto loops = batch_size * num_heads;
414 auto seq_len_square = seq_len * seq_len;
415 auto qk_head_size = qkv_head_sizes[0];
416 auto qk_inner_size = seq_len * qk_head_size;
417
418 // One batched fastGemmBatch call over all (batch, head) pairs — wrapping per-head fastGemm in parallel_for_ caused nested parallelism (inner MLAS gemm issued its own parallel_for_, serializing on most threads).
419 std::vector<size_t> qk_a_offs(loops), qk_b_offs(loops), qk_c_offs(loops);
420 for (int i = 0; i < (int)loops; i++) {
421 qk_a_offs[i] = (size_t)i * qk_inner_size;
422 qk_b_offs[i] = (size_t)i * qk_inner_size;
423 qk_c_offs[i] = (size_t)i * seq_len_square;
424 }
425 opt.multi_thread = true;
426 // ldb0=1, ldb1=qk_head_size signals trans_b for K (stored as [seq_len, qk_head_size]) to fastGemmBatch.
427 fastGemmBatch(loops, qk_a_offs.data(), qk_b_offs.data(), qk_c_offs.data(),
428 (int)seq_len, (int)seq_len, (int)qk_head_size,
429 scale, Q, (int)qk_head_size, 1,
430 K, 1, (int)qk_head_size, 0.f,
431 output, (int)seq_len, opt);
432
433 // Additive mask broadcast-aligned right to [B,H,S,S] (size-1 dims broadcast); only present when the fusion pass attached an external mask input.
434 int num_non_blob_inputs = (int)inputs.size();
435 bool has_mask = (!blobs.empty() && num_non_blob_inputs >= 2) ||
436 (blobs.empty() && num_non_blob_inputs >= 4);
437 if (has_mask) {
438 const Mat &mask_mat = blobs.empty() ? inputs[3] : inputs[1];
439 CV_CheckTypeEQ(mask_mat.type(), CV_32F,
440 "DNN/Attention: mask must be float32");
441
442 const auto mask_shape = shape(mask_mat);
443 const int mask_ndim = mask_shape.dims;
444 auto fmt_shape = [&]() {
445 std::ostringstream ss;
446 for (int d = 0; d < mask_ndim; d++) { if (d) ss << ","; ss << mask_shape[d]; }
447 return ss.str();
448 };
449 // Right-align mask dims to (B, H, Q=S, K=S); missing leading dims default to 1.
450 auto get_dim = [&](int t) -> int {
451 int md = mask_ndim - 4 + t;
452 return (md >= 0) ? mask_shape[md] : 1;
453 };
454 // Extra leading dims beyond rank 4 must all be 1 (no broadcasting beyond [B,H,S,S]).
455 for (int d = 0; d < mask_ndim - 4; d++) {
456 if (mask_shape[d] != 1)
457 CV_Error(Error::StsNotImplemented,
458 cv::format("DNN/Attention: unsupported mask shape [%s] (B=%d, H=%d, S=%d)",
459 fmt_shape().c_str(), (int)batch_size, (int)num_heads, (int)seq_len));
460 }
461 const int dim_b = get_dim(0);
462 const int dim_h = get_dim(1);
463 const int dim_q = get_dim(2);
464 const int dim_k = get_dim(3);
465 auto check_bcast = [&](int md, int td, const char* axis) {
466 if (md != 1 && md != td)
467 CV_Error(Error::StsNotImplemented,
468 cv::format("DNN/Attention: mask shape [%s] not broadcastable to [%d,%d,%d,%d] (axis %s)",
469 fmt_shape().c_str(), (int)batch_size, (int)num_heads,
470 (int)seq_len, (int)seq_len, axis));
471 };
472 check_bcast(dim_b, (int)batch_size, "batch");
473 check_bcast(dim_h, (int)num_heads, "head");
474 check_bcast(dim_q, (int)seq_len, "query");
475 check_bcast(dim_k, (int)seq_len, "key");
476
477 const size_t mask_b_stride = (dim_b == 1) ? 0 : (size_t)dim_h * dim_q * dim_k;
478 const size_t mask_h_stride = (dim_h == 1) ? 0 : (size_t)dim_q * dim_k;
479 const size_t mask_q_stride = (dim_q == 1) ? 0 : (size_t)dim_k;
480 const bool mask_k_bcast = (dim_k == 1);
481
482 const float *mask_data = mask_mat.ptr<const float>();
483 parallel_for_(Range(0, (int)loops), [&](const Range &r) {
484 for (int i = r.start; i < r.end; i++) {
485 const int b = i / (int)num_heads;
486 const int h = i % (int)num_heads;
487 const float *m_bh = mask_data + b * mask_b_stride + h * mask_h_stride;
488 float *prob = output + i * seq_len_square;
489 for (size_t row = 0; row < seq_len; row++) {
490 const float *m = m_bh + row * mask_q_stride;
491 float *p = prob + row * seq_len;
492 if (mask_k_bcast) {
493 float v = m[0];
494 for (size_t j = 0; j < seq_len; j++) p[j] += v;
495 } else {
496 for (size_t j = 0; j < seq_len; j++) p[j] += m[j];
497 }
498 }
499 }
500 }, loops * seq_len * (1 / 1024.0));
501 }
502
503 softmax(attention_prob, attention_prob, shape(attention_prob).size() - 1);
504 }
505
506 // output = attention_prob @ V, then transpose back to [B, S, H*D]
507 auto &output_buffer = internals[2];
508 {
509 auto *output = outputs[0].ptr<float>();
510 auto *output_buff = output_buffer.ptr<float>();
511 const auto *prob = attention_prob.ptr<const float>();
512
513 auto loops = batch_size * num_heads;
514 auto prob_inner_size = seq_len * seq_len;
515 auto v_head_size = qkv_head_sizes[2];
516 auto v_inner_size = seq_len * v_head_size;
517
518 // Batched fastGemmBatch over (batch, head) — same nested-parallelism rationale as QK^T above.
519 std::vector<size_t> av_a_offs(loops), av_b_offs(loops), av_c_offs(loops);
520 for (int i = 0; i < (int)loops; i++) {
521 av_a_offs[i] = (size_t)i * prob_inner_size;
522 av_b_offs[i] = (size_t)i * v_inner_size;
523 av_c_offs[i] = (size_t)i * v_inner_size;
524 }
525 opt.multi_thread = true;
526 fastGemmBatch(loops, av_a_offs.data(), av_b_offs.data(), av_c_offs.data(),
527 (int)seq_len, (int)v_head_size, (int)seq_len,
528 1.f, prob, (int)seq_len, 1,
529 V, (int)v_head_size, 1, 0.f,
530 output_buff, (int)v_head_size, opt);
531
532 // Transpose [B*H, S, D] -> [B, S, H*D] in place via per-(batch, head) memcpy strips.
533 parallel_for_(Range(0, (int)loops), [&] (const Range &r) {
534 for (int i = r.start; i < r.end; i++) {
535 const int output_offset = i * (int)v_inner_size;
536 const int batch_index = static_cast<int>(i / num_heads);
537 const int head_index = static_cast<int>(i % num_heads);
538 const float *src = output_buff + output_offset;
539 float *dst = output + (batch_index * (int)seq_len * (int)num_heads + head_index) * (int)v_head_size;
540 for (int j = 0; j < (int)seq_len; j++) {
541 std::memcpy(dst, src, v_head_size * sizeof(float));
542 src += v_head_size;
543 dst += qkv_hidden_sizes[2];
544 }
545 }
546 }, loops * seq_len * v_head_size * (1 / 1024.0));
547 }
548 }
549
550 private:
551 size_t num_heads;
552 std::vector<size_t> qkv_hidden_sizes; // order: {qk_hidden_size, qk_hidden_size, v_hidden_size}
553 float scale;
554 size_t output_ndims;
555
556 std::vector<size_t> qkv_head_sizes; // order: {qk_head_size, qk_head_size, v_head_size}
557
558 size_t batch_size;
559 size_t seq_len;
560 size_t input_hidden_size;
561 size_t hidden_size;
562
563 bool do_rotary;
564 std::vector<float> packed_weight_q;
565 std::vector<float> packed_weight_k;
566 std::vector<float> packed_weight_v;
567 std::vector<unsigned char> flash_scratch;
568
569 FastGemmOpt opt;
570};
