47void LayerFactory::registerLayer(const String& type, Constructor constructor)
48{
49 CV_TRACE_FUNCTION();
50 CV_TRACE_ARG_VALUE(type, "type", type.c_str());
51
52 cv::AutoLock lock(getLayerFactoryMutex());
53 LayerFactory_Impl::iterator it = getLayerFactoryImpl().find(type);
54
55 if (it != getLayerFactoryImpl().end())
56 {
57 if (it->second.back() == constructor)
58 CV_Error(cv::Error::StsBadArg, "Layer \"" + type + "\" already was registered");
59 it->second.push_back(constructor);
60 }
61 getLayerFactoryImpl().insert(std::make_pair(type, std::vector<Constructor>(1, constructor)));
62}

152Ptr<Layer> LayerFactory::createExec(const String& type, int backendId,
153 const Ptr<LayerInfo>& data, void* backendCtx)
154{
155 CV_TRACE_FUNCTION();
156 CV_TRACE_ARG_VALUE(type, "type", type.c_str());
157 ExecConstructor ctor = nullptr;
158 {
159 cv::AutoLock lock(getLayerFactoryMutex());
160 ExecFactory_Impl& impl = getExecFactoryImpl();
161 ExecFactory_Impl::const_iterator it = impl.find(type);
162 if (it != impl.end()) {
163 auto bit = it->second.find(backendId);
164 if (bit != it->second.end())
165 ctor = bit->second;
166 }
167 }
168 if (ctor)
169 return ctor(data, backendCtx);
170 return Ptr<Layer>();
171}
