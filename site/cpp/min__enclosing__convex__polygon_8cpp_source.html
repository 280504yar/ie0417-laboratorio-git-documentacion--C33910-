242IntersectionPoint::IntersectionPoint(const cv::Point2f& a, const cv::Point2f& b,
243 const cv::Point2f& c, const cv::Point2f& d)
244{
245 const cv::Point2f ab = b - a, cd = d - c, ac = c - a;
246 const double det = ab.cross(-cd);
247 if(std::abs (det) < EPSILON )
248 return;
249
250 const double loc = ac.cross(-cd) / det;
251 if(loc <= 0)
252 return;
253
254 point = a + loc * ab;
255 position = true;
256}

334const FlushIntersect& FlushIntersections::lineIntersect(int i, int j)
335{
336 FlushIntersect& itr = intersections[i][j];
337 if(itr.done)
338 return itr;
339
340 const int n = (int)ngon.size();
341 if((i + 1) % n == j)
342 {
343 itr.intersection.point = ngon[j];
344 itr.intersection.position = true;
345 itr.extra_area = 0.0;
346 itr.done = true;
347 return itr;
348 }
349 itr.intersection = IntersectionPoint(i, j, ngon);
350 if(itr.intersection.position)
351 {
352 area_edges.resize(0);
353 for(int t = (i + 1) % n; t != (j + 1) % n; t = (t + 1) % n)
354 {
355 area_edges.push_back(ngon[t]);
356 }
357 area_edges.push_back(itr.intersection.point);
358 itr.extra_area = cv::contourArea(area_edges);
359 itr.done = true;
360 }
361 else
362 {
363 itr.extra_area = std::numeric_limits<double>::max();
364 itr.intersection.position = false;
365 itr.done = true;
366 }
367 return itr;
368}

398BalancedIntersect BalancedIntersections::flush(int i, int j, int e)
399{
400 CV_Assert(j != e);
401
402 const int n = (int)ngon.size();
403 const int before = (e - 1 + n) % n;
404 BalancedIntersect bi = balanced_intersections[i][j];
405
406 const IntersectionPoint left_e(i, e, ngon);
407 const IntersectionPoint right_e(e, j, ngon);
408
409 if(left_e.position == true && right_e.position == true)
410 {
411 double extra_area = extraArea((i + 1) % n, e, ngon[e], left_e.point);
412 if(extra_area < bi.extra_area)
413 {
414 extra_area += extraArea((e + 1) % n, j, ngon[j], right_e.point);
415 if(extra_area < bi.extra_area)
416 {
417 bi.extra_area = extra_area;
418 bi.pi = left_e.point;
419 bi.pj = right_e.point;
420 bi.position = true;
421 bi.flush = e;
422 }
423 }
424 }
425
426 if(before != i)
427 {
428 const IntersectionPoint left_before(i, before, ngon);
429 const IntersectionPoint right_before(before, j, ngon);
430 if(left_before.position == true && right_before.position == true)
431 {
432 double extra_area =
433 extraArea((i + 1) % n, before, ngon[before], left_before.point);
434
435 if(extra_area < bi.extra_area)
436 {
437 extra_area += extraArea(e, j, ngon[j], right_before.point);
438 if(extra_area < bi.extra_area)
439 {
440 bi.extra_area = extra_area;
441 bi.pi = left_before.point;
442 bi.pj = right_before.point;
443 bi.position = true;
444 bi.flush = before;
445 }
446 }
447 }
448 }
449 return bi;
450}

458BalancedIntersect BalancedIntersections::balancedIntersect(int i, int j, int e)
459{
460 if(balanced_intersections[i][j].done)
461 return balanced_intersections[i][j];
462
463 const int n = (int)ngon.size();
464 if((i + 2) % n == j)
465 {
466 BalancedIntersect& bi = balanced_intersections[i][j];
467 bi.pi = ngon[(i + 1) % n];
468 bi.pj = ngon[j];
469 bi.flush = (i + 1) % n;
470 bi.position = true;
471 bi.extra_area = 0.0;
472 bi.done = true;
473 return bi;
474 }
475
476 const cv::Point2f p1 = ngon[i], p2 = ngon[(i + 1) % n], p3 = ngon[e],
477 p4 = ngon[j], p5 = ngon[(j + 1) % n];
478 const cv::Point2f dir12 = p2 - p1, dir45 = p5 - p4;
479 const double det = dir12.cross(dir45);
480 if(std::abs (det) < EPSILON )
481 {
482 flush(i, j, e);
483 return balanced_intersections[i][j];
484 }
485
486 BalancedIntersect bi;
487 cv::Point2f temp = 2 * p3 - p2 - p4;
488 const double s = temp.cross(dir45) / det;
489 const double t = temp.cross(dir12) / det;
490 if(s >= 0 && t >= 0)
491 {
492 bi.pi = p2 + dir12 * s;
493 bi.pj = p4 - dir45 * t;
494 bi.position = true;
495
496 const cv::Point2f dir_balanced = bi.pj - bi.pi,
497 dir_left = p3 - ngon[(e - 1 + n) % n],
498 dir_right = ngon[(e + 1) % n] - p3;
499
500 const double cross_left = dir_balanced.cross(dir_left),
501 cross_right = dir_balanced.cross(dir_right);
502 if((cross_left < 0 && cross_right < 0) || (cross_left > 0 && cross_right > 0))
503 {
504 BalancedIntersect reset;
505 bi = reset;
506 bi = flush(i, j, e);
507 }
508 else if(std::abs (cross_left) < EPSILON )
509 {
510 bi.flush = (e - 1 + n) % n;
511 bi.extra_area =
512 extraArea((i + 1) % n, (e - 1 + n) % n, ngon[(e - 1 + n) % n], bi.pi)
513 + extraArea(e, j, ngon[j], bi.pj);
514 }
515 else if(std::abs (cross_right) < EPSILON )
516 {
517 bi.flush = e;
518 bi.extra_area = extraArea((i + 1) % n, e, ngon[e], bi.pi)
519 + extraArea((e + 1) % n, j, ngon[j], bi.pj);
520 }
521 else
522 {
523 bi.extra_area = extraArea((i + 1) % n, e, ngon[e], bi.pi)
524 + extraArea(e, j, ngon[j], bi.pj);
525 }
526 }
527 else
528 {
529 flush(i, j, e);
530 }
531
532 if(bi.extra_area < balanced_intersections[i][j].extra_area)
533 {
534 balanced_intersections[i][j] = bi;
535 }
536 return bi;
537}

566void Chains::findSingleE(int i, int j, int l, int r)
567{
568 const size_t n = ngon.size();
569 Segment& one = single_sides[i][j];
570 if (one.done)
571 return;
572
573 double min_area = std::numeric_limits<double>::max();
574 for (int e = l; e != r + 1 && e != j; e = (e + 1) %n)
575 {
576 BalancedIntersect candidate = balanced_inters.balancedIntersect(i, j, e);
577 if(candidate.extra_area < min_area)
578 {
579 min_area = candidate.extra_area;
580 one.side = e;
581 one.extra_area = candidate.extra_area;
582 one.flush = false;
583 one.exists = true;
584 }
585 }
586 one.done = true;
587 balanced_inters.markAsDone(i, j);
588}

596void Chains::singleSideImpl(int i, int j1, int j2)
597{
598 const int n = (int)ngon.size();
599 if((j1 + 1) %n == j2)
600 {
601 return;
602 }
603
604 int mid = (j1 < j2 ? ((j1 + j2) / 2) : ((j1 + n + j2) / 2)) % n;
605 int l = single_sides[i][j1].side < 0 ? (j1 + 1) %n : single_sides[i][j1].side;
606 int r = single_sides[i][j2].side < 0 ? (j2 - 1 + n) %n : single_sides[i][j2].side;
607
608 findSingleE(i, mid, l, r);
609 singleSideImpl(i, j1, mid);
610 singleSideImpl(i, mid, j2);
611
612 return;
613}

622void Chains::findMiddleE1(int i, int j, int l, int r)
623{
624 const size_t n = ngon.size();
625 Segment& one = middle_sides[1][i][j];
626 if (one.done)
627 return;
628
629 double min_area = std::numeric_limits<double>::max();
630 for (int e = l; e != r + 1 && e != j; e = (e + 1) %n)
631 {
632 const FlushIntersect& before = intersections.lineIntersect(i, e);
633 if(!before.intersection.position)
634 continue;
635 const FlushIntersect& after = intersections.lineIntersect(e, j);
636 if(!after.intersection.position)
637 continue;
638
639 double tmp_area = before.extra_area + after.extra_area;
640 if(tmp_area < min_area)
641 {
642 min_area = tmp_area;
643 one.side = e;
644 one.extra_area = tmp_area;
645 one.exists = true;
646 one.flush = true;
647 }
648 }
649 one.done = true;
650}

685void Chains::findMiddleE(int h, int i, int j, int l, int r)
686{
687 const int n = (int)ngon.size();
688 Segment& one = middle_sides[h][i][j];
689 if (one.done)
690 return;
691
692 const int dist = (i <= j ? (j - i) : (j + n - i));
693 const int h_floor = (h - 1) / 2;
694 const int h_ceil = h - 1 - h_floor;
695
696 CV_Assert(dist != 0);
697
698 if(dist - 1 < h)
699 {
700 one.done = true;
701 return;
702 }
703 if(dist - 1 == h)
704 {
705 one.side = (i + h_floor + 1) % n;
706 one.extra_area = 0.0;
707 one.exists = true;
708 one.flush = true;
709 one.done = true;
710 return;
711 }
712
713 double min_area = std::numeric_limits<double>::max();
714 for (int e = l; e != r + 1 && e != j; e = (e + 1) %n)
715 {
716 const Segment& before = middle_sides[h_floor][i][e];
717 if (before.extra_area == std::numeric_limits<double>::max())
718 continue;
719 const Segment& after = middle_sides[h_ceil][e][j];
720 if(after.extra_area == std::numeric_limits<double>::max())
721 continue;
722
723 double tmp_area = before.extra_area + after.extra_area;
724 if(tmp_area < min_area)
725 {
726 min_area = tmp_area;
727 one.side = e;
728 one.extra_area = tmp_area;
729 one.exists = true;
730 one.flush = true;
731 }
732 }
733 one.done = true;
734}

766std::set<int> Chains::relevantChainLengths(int h)
767{
768 if(h <= 1)
769 return {h};
770
771 std::set<int> hs = {h};
772 const int h_floor = (h - 1) / 2;
773 const int h_ceil = h - 1 - h_floor;
774 std::set<int> h1 = relevantChainLengths(h_floor);
775 for(const int& hNew : h1)
776 hs.insert(hNew);
777 if (h_ceil != h_floor)
778 {
779 std::set<int> h2 = relevantChainLengths(h_ceil);
780 for(const int& hNew : h2)
781 hs.insert(hNew);
782 }
783 return hs;
784}

806void Chains::calcMiddleChains(int h)
807{
808 const int n = (int)ngon.size();
809 if (h == 0)
810 {
811 for (int i = 0; i < n; i++)
812 {
813 for (int j = 0; j < n; j++)
814 {
815 Segment& one = middle_sides[h][i][j];
816 const FlushIntersect itrs = intersections.lineIntersect(i, j);
817
818 one.side = -1;
819 one.extra_area = itrs.extra_area;
820 one.exists = false;
821 one.flush = false;
822 one.done = true;
823 }
824 }
825 return;
826 }
827 if (h == 1)
828 {
829 for (int i = 0; i < n; i++)
830 {
831 int j1 = (i + 2) %n, j2 = (i - 2 + n) %n;
832
833 findMiddleE1(i, j1, (i + 1) %n, (j1 - 1 + n) %n);
834 findMiddleE1(i, j2, (i + 1) %n, (j2 - 1 + n) %n);
835 middleSideImpl1(i, j1, j2);
836 }
837 return;
838 }
839
840 for (int i = 0; i < n; i++)
841 {
842 int j1 = (i + 2) %n, j2 = (i - 2 + n) %n;
843
844 findMiddleE(h, i, j1, (i + 1) %n, (j1 - 1 + n) %n);
845 findMiddleE(h, i, j2, (i + 1) %n, (j2 - 1 + n) %n);
846 middleSideImpl(h, i, j1, j2);
847 }
848}

855Minimum Chains::minimumArea(int n, int k)
856{
857 Minimum min{};
858 for(int i = 0; i < n; i++)
859 {
860 for(int j = 0; j < n; j++)
861 {
862 if(!single_sides[i][j].exists || !middle_sides[k - 3][j][i].exists)
863 continue;
864
865 double tmp_area =
866 single_sides[i][j].extra_area + middle_sides[k - 3][j][i].extra_area;
867 if(tmp_area < min.area)
868 {
869 min.area = tmp_area;
870 min.i = i;
871 min.j = j;
872 }
873 }
874 }
875 return min;
876}

884std::vector<Side> Chains::reconstructHSidedChain(int h, int i, int j)
885{
886 CV_Assert(h != 0);
887
888 if(h == 1)
889 {
890 return std::vector<Side>{{middle_sides[h][i][j].side, true}};
891 }
892
893 std::vector<Side> before, after;
894 const int h_floor = (h - 1) / 2;
895 const int h_ceil = h - 1 - h_floor;
896 if(h_floor > 0)
897 before = reconstructHSidedChain(h_floor, i, middle_sides[h][i][j].side);
898 if(h_ceil > 0)
899 after = reconstructHSidedChain(h_ceil, middle_sides[h][i][j].side, j);
900
901 std::vector<Side> sides{{middle_sides[h][i][j].side, true}};

913std::vector<Side> Chains::findKSides(int k, int i, int j)
914{
915 std::vector<Side> sides;
916 sides.reserve(k); // Optimization and avoiding GCC's false positive warning (-Wstringop-overflow)
917 sides.push_back({i, true});
918 sides.push_back({j, true});
919
920 if(single_sides[i][j].flush)
921 sides.push_back({single_sides[i][j].side, true});
922 else
923 sides.push_back({single_sides[i][j].side, false});
924
925 std::vector<Side> flush_chain = reconstructHSidedChain(k - 3, j, i);
926 sides.insert(sides.end(), flush_chain.begin(), flush_chain.end());
927 std::sort(sides.begin(), sides.end(),
928 [](const Side& lhs, const Side& rhs) { return lhs.side < rhs.side; });
929

937std::vector<cv::Point2f> Chains::findKVertices(std::vector<Side>& sides)
938{
939 const int k = (int)sides.size();
940 std::vector<cv::Point2f> vertices(k);
941
942 for(int u = 0; u < k; u++)
943 {
944 const int next = (u + 1) % k;
945 if(sides[u].flush && sides[next].flush)
946 {
947 vertices[u] = intersections.lineIntersect(sides[u].side,
948 sides[next].side).intersection.point;
949 }
950 else if(sides[u].flush && !sides[next].flush)
951 {
952 vertices[u] = balanced_inters[sides[u].side][sides[(u + 2) % k].side].pi;
953 }
954 else if(!sides[u].flush && sides[next].flush)
955 {
956 vertices[u] = balanced_inters[sides[(u - 1 + k) %k].side][sides[next].side].pj;
957 }
958 else
959 {
960 CV_Error(cv::Error::StsInternal, "findKVertices logic error!");
961 }
962 }
963 return vertices;
964}

981 std::vector<cv::Point2f> &minPolygon, int k)
982{
983 const int n = (int)ngon.size();
984
985 Chains chains(ngon, k);
986 chains.calcOneSidedChains();
987 std::set<int> hs = chains.relevantChainLengths(k - 3);
988 for(auto& h : hs)
989 {
990 chains.calcMiddleChains(h);
991 }
992
993 Kgon kgon{};
994
995 const Minimum min = chains.minimumArea(n, k);
996
997 kgon.i = min.i;
998 kgon.j = min.j;
999 kgon.extra_area = min.area;
1000 kgon.sides = chains.findKSides(k, min.i, min.j);
1001 kgon.vertices = chains.findKVertices(kgon.sides);
1002
1003 minPolygon = kgon.vertices;
1004 return cv::contourArea(minPolygon);
1005}

1029
1030 n = (int)ngon.size();
1031
1032 if (n < k)
1033 {
1034 CV_LOG_WARNING(NULL, "convex hull of size " << n << " must have equal or more different points than k = " << k);
1035 polygon.release();
1036 return 0.;
1037 }
1038
1039 if (n == k)
1040 {
1041 cv::Mat(ngon).copyTo(polygon);

1043 }

1046 {
1047 CV_LOG_WARNING(NULL, "Singular input poligon");
1048 polygon.release();
1049 return 0.;
1050 }
1051
