81static void* AppleCLGetProcAddress(const char* name)
82{
83 static bool initialized = false;
84 static void* handle = NULL;
85 if (!handle && !initialized)
86 {
87 cv::AutoLock lock(cv::getInitializationMutex());
88 if (!initialized)
89 {
90 const char* defaultPath = "/System/Library/Frameworks/OpenCL.framework/Versions/Current/OpenCL";
91 std::string path = getRuntimePath(defaultPath);
92 if (!path.empty())
93 handle = dlopen(path.c_str(), RTLD_LAZY | RTLD_GLOBAL);
94 if (handle == NULL)
95 {
96 if (!path.empty() && path != defaultPath)
97 fprintf(stderr, ERROR_MSG_CANT_LOAD);
98 }
99 else if (dlsym(handle, OPENCL_FUNC_TO_CHECK_1_1) == NULL)
100 {
101 fprintf(stderr, ERROR_MSG_INVALID_VERSION);
102 handle = NULL;
103 }
104 initialized = true;
105 }
106 }
107 if (!handle)
108 return NULL;
109 return dlsym(handle, name);
110}

117static void* WinGetProcAddress(const char* name)
118{
119 static bool initialized = false;
120 static HMODULE handle = NULL;
121 if (!handle && !initialized)
122 {
123 cv::AutoLock lock(cv::getInitializationMutex());
124 if (!initialized)
125 {
126 handle = GetModuleHandleA("OpenCL.dll");
127 if (!handle)
128 {
129 const std::string defaultPath = "OpenCL.dll";
130 const std::string path = getRuntimePath(defaultPath);
131 if (!path.empty())
132 handle = LoadLibraryA(path.c_str());
133 if (!handle)
134 {
135 if (!path.empty() && path != defaultPath)
136 fprintf(stderr, ERROR_MSG_CANT_LOAD);
137 }
138 else if (GetProcAddress(handle, OPENCL_FUNC_TO_CHECK_1_1) == NULL)
139 {
140 fprintf(stderr, ERROR_MSG_INVALID_VERSION);
141 FreeLibrary(handle);
142 handle = NULL;
143 }
144 }
145 initialized = true;
146 }
147 }
148 if (!handle)
149 return NULL;
150 return (void*)GetProcAddress(handle, name);
151}

193static void* GetProcAddress(const char* name)
194{
195 static bool initialized = false;
196 static void* handle = NULL;
197 if (!handle && !initialized)
198 {
199 cv::AutoLock lock(cv::getInitializationMutex());
200 if (!initialized)
201 {
202 bool foundOpenCL = false;
203 for (unsigned int i = 0; i < (sizeof(defaultAndroidPaths)/sizeof(char*)); i++)
204 {
205 const std::string path = (i==0) ? getRuntimePath(defaultAndroidPaths[i]) : defaultAndroidPaths[i];
206 if (!path.empty()) {
207 handle = GetHandle(path.c_str());
208 if (handle) {
209 foundOpenCL = true;
210 break;
211 }
212 }
213 }
214 initialized = true;
215 if (!foundOpenCL)
216 fprintf(stderr, ERROR_MSG_CANT_LOAD);
217 }
218 }
219 if (!handle)
220 return NULL;
221 return dlsym(handle, name);
222}

350static void* opencl_check_fn(int ID)
351{
352 const struct DynamicFnEntry* e = NULL;
353 if (ID < CUSTOM_FUNCTION_ID)
354 {
355 CV_Assert(ID >= 0 && ID < (int)(sizeof(opencl_fn_list)/sizeof(opencl_fn_list[0])));
356 e = opencl_fn_list[ID];
357 }
358#ifdef HAVE_OPENCL_SVM
359 else if (ID >= SVM_FUNCTION_ID_START && ID < SVM_FUNCTION_ID_END)
360 {
361 ID = ID - SVM_FUNCTION_ID_START;
362 CV_Assert(ID >= 0 && ID < (int)(sizeof(opencl_svm_fn_list)/sizeof(opencl_svm_fn_list[0])));
363 e = opencl_svm_fn_list[ID];
364 }
365#endif
366 else
367 {
368 CV_Error(cv::Error::StsBadArg, "Invalid function ID");
369 }
370 void* func = CV_CL_GET_PROC_ADDRESS(e->fnName);
371 if (!func)
372 {
373 throw cv::Exception(cv::Error::OpenCLApiCallError,
374 cv::format("OpenCL function is not available: [%s]", e->fnName),
375 CV_Func, __FILE__, __LINE__);
376 }
377 *(e->ppFn) = func;
378 return func;
379}
