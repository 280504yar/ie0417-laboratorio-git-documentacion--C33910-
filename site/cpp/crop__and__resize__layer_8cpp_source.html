53 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
54 {
55 CV_TRACE_FUNCTION();
56 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
57
58 if (inputs_arr.depth() == CV_16F)
59 {
60 forward_fallback(inputs_arr, outputs_arr, internals_arr);
61 return;
62 }
63
64 std::vector<Mat> inputs, outputs;
65 inputs_arr.getMatVector(inputs);
66 outputs_arr.getMatVector(outputs);
67
68 Mat& inp = inputs[0];
69 Mat& out = outputs[0];
70 Mat boxes = inputs[1].reshape(1, inputs[1].total() / 7);
71 const int numChannels = inp.size[1];
72 const int inpHeight = inp.size[2];
73 const int inpWidth = inp.size[3];
74 const int inpSpatialSize = inpHeight * inpWidth;
75 const int outSpatialSize = outHeight * outWidth;
76 CV_Assert_N(inp.isContinuous(), out.isContinuous());
77
78 for (int b = 0; b < boxes.rows; ++b)
79 {
80 float* outDataBox = out.ptr<float>(b);
81 float left = boxes.at<float>(b, 3);
82 float top = boxes.at<float>(b, 4);
83 float right = boxes.at<float>(b, 5);
84 float bottom = boxes.at<float>(b, 6);
85 float boxWidth = right - left;
86 float boxHeight = bottom - top;
87
88 float heightScale = boxHeight * static_cast<float>(inpHeight - 1) / (outHeight - 1);
89 float widthScale = boxWidth * static_cast<float>(inpWidth - 1) / (outWidth - 1);
90 for (int y = 0; y < outHeight; ++y)
91 {
92 float input_y = top * (inpHeight - 1) + y * heightScale;
93 int y0 = static_cast<int>(input_y);
94 const float* inpData_row0 = inp.ptr<float>(0, 0, y0);
95 const float* inpData_row1 = (y0 + 1 < inpHeight) ? (inpData_row0 + inpWidth) : inpData_row0;
96 for (int x = 0; x < outWidth; ++x)
97 {
98 float input_x = left * (inpWidth - 1) + x * widthScale;
99 int x0 = static_cast<int>(input_x);
100 int x1 = std::min(x0 + 1, inpWidth - 1);
101
102 float* outData = outDataBox + y * outWidth + x;
103 const float* inpData_row0_c = inpData_row0;
104 const float* inpData_row1_c = inpData_row1;
105 for (int c = 0; c < numChannels; ++c)
106 {
107 *outData = inpData_row0_c[x0] +
108 (input_y - y0) * (inpData_row1_c[x0] - inpData_row0_c[x0]) +
109 (input_x - x0) * (inpData_row0_c[x1] - inpData_row0_c[x0] +
110 (input_y - y0) * (inpData_row1_c[x1] - inpData_row0_c[x1] - inpData_row1_c[x0] + inpData_row0_c[x0]));
111
112 inpData_row0_c += inpSpatialSize;
113 inpData_row1_c += inpSpatialSize;
114 outData += outSpatialSize;
115 }
116 }
117 }
118 }
119 if (boxes.rows < out.size[0])
120 {
121 // left = top = right = bottom = 0
122 std::vector<cv::Range> dstRanges(4, Range::all());
123 dstRanges[0] = Range(boxes.rows, out.size[0]);
124 out(dstRanges).setTo(inp.ptr<float>(0, 0, 0)[0]);
125 }
126 }

128#ifdef HAVE_DNN_NGRAPH
129 virtual Ptr<BackendNode> initNgraph(const std::vector<Ptr<BackendWrapper> >& inputs,
130 const std::vector<Ptr<BackendNode> >& nodes) CV_OVERRIDE
131 {
132 // Slice second input: from 1x1xNx7 to 1x1xNx5
133 auto input = nodes[0].dynamicCast<InfEngineNgraphNode>()->node;
134 auto rois = nodes[1].dynamicCast<InfEngineNgraphNode>()->node;
135
136 auto rois_shape = rois.get_shape();
137 std::vector<int64_t> dims(rois_shape.begin(), rois_shape.end()), offsets(4, 0);
138 offsets[3] = 2;
139 dims[3] = 7;
140
141 auto lower_bounds = std::make_shared<ov::op::v0::Constant>(ov::element::i64,
142 ov::Shape{offsets.size()}, offsets.data());
143 auto upper_bounds = std::make_shared<ov::op::v0::Constant>(ov::element::i64,
144 ov::Shape{dims.size()}, dims.data());
145 auto strides = std::make_shared<ov::op::v0::Constant>(ov::element::i64,
146 ov::Shape{dims.size()}, std::vector<int64_t>((int64_t)dims.size(), 1));
147 auto slice = std::make_shared<ov::op::v1::StridedSlice>(rois,
148 lower_bounds, upper_bounds, strides, std::vector<int64_t>{}, std::vector<int64_t>{});
149
150 // Reshape rois from 4D to 2D
151 std::vector<int64_t> shapeData = {dims[2], 5};
152 auto shape = std::make_shared<ov::op::v0::Constant>(ov::element::i64, ov::Shape{2}, shapeData.data());
153 auto reshape = std::make_shared<ov::op::v1::Reshape>(slice, shape, true);
154
155 auto roiPooling =
156 std::make_shared<ov::op::v0::ROIPooling>(input, reshape,
157 ov::Shape{(size_t)outHeight, (size_t)outWidth},
158 1.0f, "bilinear");
159
160 return Ptr<BackendNode>(new InfEngineNgraphNode(roiPooling));
161 }
162#endif // HAVE_DNN_NGRAPH
