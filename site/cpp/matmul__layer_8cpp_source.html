55 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
56 const int requiredOutputs,
57 std::vector<MatShape> &outputs,
58 std::vector<MatShape> &internals) const CV_OVERRIDE {
59 int num_inputs = inputs.size() + blobs.size();
60 CV_CheckGE(num_inputs, 2, "DNN/MatMul: two inputs at least");
61 CV_CheckLE(num_inputs, 3, "DNN/MatMul: three inputs at most");
62
63 const auto shape_A = inputs[0], shape_B = blobs.empty() ? inputs[1] : shape(blobs[0]);
64 CV_CheckGE(shape_A.size(), static_cast<size_t>(2), "DNN/MatMul: invalid shape of input A");
65 CV_CheckGE(shape_B.size(), static_cast<size_t>(2), "DNN/MatMul: invalid shape of input B");
66
67 // Check legal matrix multiplication
68 int mA = shape_A[shape_A.size() - 2], nA = shape_A.back();
69 int mB = shape_B[shape_B.size() - 2], nB = shape_B.back();
70 int M = trans_a ? nA : mA;
71 int N = trans_b ? mB : nB;
72 int K_A = trans_a ? mA : nA;
73 int K_B = trans_b ? nB : mB;
74 CV_CheckEQ(K_A, K_B, "DNN/MatMul: invalid dimension K");
75
76 // Check if inputs are broadcastable.
77 MatShape common_shape;
78 if (shape_A.size() != 2 || shape_B.size() != 2) {
79 const auto &shape_more_dims = shape_A.size() > shape_B.size() ? shape_A : shape_B;
80 const auto &shape_less_dims = shape_A.size() > shape_B.size() ? shape_B : shape_A;
81 size_t diff_dims = shape_more_dims.size() - shape_less_dims.size();
82 common_shape = shape_more_dims;
83 for (size_t i = 0; i < shape_less_dims.size() - 2; i++) {
84 const auto dl = shape_less_dims[i], dm = shape_more_dims[i + diff_dims];
85 if (dl != 1 && dm != 1 && dl != dm) {
86 CV_Error(Error::StsBadSize, format("DNN/MatMul: invalid shape for broadcasting, shape_A[%zu]=%d, shape_B[%zu]=%d\n", i, shape_less_dims[i], i, shape_more_dims[i + diff_dims]));
87 }
88
89 if (dm == 1) {
90 common_shape[i + diff_dims] = dl;
91 }
92 }
93 common_shape[common_shape.size() - 2] = M;
94 common_shape[common_shape.size() - 1] = N;
95 } else {
96 common_shape.resize(2);
97 common_shape[0] = M;
98 common_shape[1] = N;
99 }
100
101 // Check if bias is broadcastable
102 if (num_inputs == 3) {
103 const auto shape_C = blobs.empty() ? inputs.back() : shape(blobs.back());
104 if (real_ndims_C == 1) { // (1) or (N)
105 CV_Check(shape_C[0], shape_C[0] == 1 || shape_C[0] == N, "DNN/MatMul: invalid dimension of C");
106 } else if (real_ndims_C >= 2) {
107 const auto &shape_large = common_shape.size() > shape_C.size() ? common_shape : shape_C;
108 const auto &shape_small = common_shape.size() > shape_C.size() ? shape_C : common_shape;
109 size_t diff_dims = shape_large.size() - shape_small.size();
110 for (size_t i = 0; i < shape_small.size(); i++) {
111 const auto dl = shape_small[i], dm = shape_large[i + diff_dims];
112 if (dl != 1 && dm != 1 && dl != dm) {
113 CV_Error(Error::StsBadSize, "DNN/MatMul: invalid shape of C");
114 }
115 }
116 }
117 }
118
119 outputs.assign(1, common_shape);
120 return false;
121 }

139 virtual int64 getFLOPS(const std::vector<MatShape> &inputs,
140 const std::vector<MatShape> &outputs) const CV_OVERRIDE
141 {
142 CV_Assert(!inputs.empty());
143 const auto shape_A = inputs[0], shape_B = blobs.empty() ? inputs[1] : shape(blobs[0]);
144 int mA = shape_A[shape_A.size() - 2], nA = shape_A.back();
145 int mB = shape_B[shape_B.size() - 2], nB = shape_B.back();
146 int M = trans_a ? nA : mA;
147 int N = trans_b ? mB : nB;
148 int K = trans_a ? mA : nA;
149
150 int64 batch = 1;
151 for (size_t i = 0; i + 2 < outputs[0].size(); i++)
152 batch *= outputs[0][i];
153
154 // 2*M*N*K multiply-adds per batch element, +M*N for bias if present
155 int64 flops = batch * (CV_BIG_INT(2) * M * N * K);
156 int num_inputs = (int)inputs.size() + (int)blobs.size();
157 if (num_inputs == 3)
158 flops += batch * M * N;
159 return flops;
160 }

162 virtual void finalize(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr) CV_OVERRIDE {
163 opt.init();
164
165 std::vector<Mat> inputs, outputs;
166 inputs_arr.getMatVector(inputs);
167 outputs_arr.getMatVector(outputs);
168
169 const auto A_shape = shape(inputs[0]),
170 B_shape = blobs.empty() ? shape(inputs[1]) : shape(blobs[0]),
171 C_shape = shape(outputs[0]);
172 helper.compute(trans_a, trans_b, A_shape, B_shape, C_shape);
173
174 // These five types skip the float-only packed-B/MLAS caching below.
175 {
176 int depth0 = inputs[0].depth();
177 if (depth0 == CV_64F || depth0 == CV_32S || depth0 == CV_64S || depth0 == CV_32U || depth0 == CV_64U)
178 return;
179 }
180
181 // Pack only 2D weight matrices; skip higher-dim tensors (e.g. Q@K^T in attention).
182 const Mat* B_mat = !blobs.empty() ? &blobs[0] :
183 (inputs.size() >= 2 && inputs[1].dims == 2 ? &inputs[1] : nullptr);
184 if (B_mat && B_mat->data != last_packed_input_B_data) {
185 packed_input_B.clear();
186 packed_input_B.shrink_to_fit();
187 thin_packed_B.clear();
188
189 if (helper.batch == 1 && B_mat->type() == CV_32F &&
190 fastGemmThinEligible(helper.M, helper.N, helper.K)) {
191 thin_packed_B.resize(fastGemmThinPackBSize(helper.N, helper.K));
192 fastGemmThinPackB(helper.N, helper.K,
193 B_mat->ptr<const float>(),
194 (size_t)helper.ldb0, (size_t)helper.ldb1,
195 thin_packed_B.data());
196 } else {
197 fastGemmPackB(*B_mat, packed_input_B, trans_b, opt);
198 helper.updatePackedBOffsets(packed_input_B.size());
199 }
200 last_packed_input_B_data = B_mat->data;
201 }
202
203 // broadcast bias if needed
204 if ((inputs.size() + blobs.size()) >= 3 && blobs.size() >= 2) {
205 const auto bias_mat = blobs.back();
206 const auto bias_shape = shape(bias_mat);
207 bool is_broadcast_needed = real_ndims_C == 0 || real_ndims_C == 1 || (total(bias_shape) != total(C_shape) || bias_shape.size() != C_shape.size());
208
209 if (is_broadcast_needed) {
210 broadcast_bias = Mat(C_shape, CV_32F);
211 auto *broadcast_bias_ptr = broadcast_bias.ptr<float>();
212
213 const auto *bias = bias_mat.ptr<const float>();
214 if (bias_mat.total() == 1) { // [], [1], [1, ...]
215 float b = (*bias) * beta;
216 for (size_t i = 0; i < broadcast_bias.total(); i++) {
217 broadcast_bias_ptr[i] = b;
218 }
219 } else if (real_ndims_C == 1) { // [n]
220 size_t inner_size = C_shape.back(),
221 loops = total(C_shape) / inner_size;
222 for (size_t i = 0; i < loops; i++) {
223 size_t step = i * inner_size;
224 for (size_t j = 0; j < inner_size; j++) {
225 broadcast_bias_ptr[step + j] = beta * bias[j];
226 }
227 }
228 } else {
229 broadcast(bias_mat, C_shape, broadcast_bias);
230 }
231 } else {
232 broadcast_bias = blobs.back();
233 }
234 }
235
236#ifdef HAVE_OPENCL
237 weight_umat.release();
238 bias_umat.release();
239#endif
240 }

242 // works like Y = numpy.matmul(A, B)
243 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE {
244 CV_TRACE_FUNCTION();
245 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
246
247 CV_OCL_RUN(IS_DNN_OPENCL_TARGET(preferableTarget),
248 forward_ocl(inputs_arr, outputs_arr, internals_arr))
249
250 if (inputs_arr.depth() == CV_16F)
251 {
252 forward_fallback(inputs_arr, outputs_arr, internals_arr);
253 return;
254 }
255
256 std::vector<Mat> inputs, outputs;
257 inputs_arr.getMatVector(inputs);
258 outputs_arr.getMatVector(outputs);
259
260 switch (inputs[0].depth()) {
261 case CV_64F: forwardDouble(inputs, outputs); return;
262 case CV_32S: forwardInt<int32_t>(inputs, outputs); return;
263 case CV_64S: forwardInt<int64_t>(inputs, outputs); return;
264 case CV_32U: forwardInt<uint32_t>(inputs, outputs); return;
265 case CV_64U: forwardInt<uint64_t>(inputs, outputs); return;
266 default: break;
267 }
268
269 const auto &A = inputs[0];
270 auto &Y = outputs[0];
271
272 const auto *a = A.ptr<const float>();
273 auto *y = Y.ptr<float>();
274 // add bias if existed
275 if ((inputs.size() + blobs.size()) >= 3) {
276 const auto &shape_Y = shape(Y);
277 if (blobs.empty()) { // bias from input
278 const auto &bias_mat = inputs.back();
279 const auto *bias = bias_mat.ptr<const float>();
280 if (bias_mat.total() == 1) { // [], [1], [1, ...]
281 float b = (*bias) * beta;
282 for (size_t i = 0; i < Y.total(); i++) {
283 y[i] = b;
284 }
285 } else if (real_ndims_C == 1) { // [n]
286 const size_t inner_size = shape_Y.back(),
287 batches = total(Y) / inner_size;
288 parallel_for_(Range(0, batches), [&] (const Range &r) {
289 for (int i = r.start; i < r.end; i++) {
290 const size_t output_offset = i * inner_size;
291 for (size_t j = 0; j < inner_size; j++) {
292 y[output_offset + j] = beta * bias[j];
293 }
294 }
295 }, double(batches * inner_size * (1 / 1024.0)));
296 } else {
297 broadcast(bias_mat, shape_Y, Y);
298 }
299 } else { // bias from constant
300 const auto *bias = broadcast_bias.ptr<const float>();
301 std::memcpy(y, bias, total(shape_Y) * sizeof(float));
302 }
303 } else {
304 std::memset(y, 0, Y.total() * sizeof(float));
305 }
306
307 if (blobs.empty()) {
308 const auto &B = inputs[1];
309 const auto *b = B.ptr<const float>();
310 bool done = false;
311 if (mlasAvailable() && !opt.use_rvv && helper.M > 0 && helper.N > 0 && helper.K > 0) {
312 const auto A_shape = shape(A);
313 const auto B_shape = shape(B);
314 const int lda_mem = A_shape.back();
315 const int ldb_mem = B_shape.back();
316 done = mlasSgemmBatch(helper.batch,
317 helper.A_offsets.data(), helper.B_offsets.data(), helper.C_offsets.data(),
318 trans_a, trans_b, helper.M, helper.N, helper.K,
319 alpha, a, lda_mem, b, ldb_mem, beta, y, helper.ldc);
320 }
321 if (!done) {
322 fastGemmBatch(helper.batch, helper.A_offsets.data(), helper.B_offsets.data(), helper.C_offsets.data(),
323 helper.M, helper.N, helper.K, alpha, a, helper.lda0, helper.lda1,
324 b, helper.ldb0, helper.ldb1, beta, y, helper.ldc, opt);
325 }
326 } else if (!thin_packed_B.empty()) {
327 fastGemmThin(helper.M, helper.N, helper.K, alpha,
328 a, helper.lda0, helper.lda1,
329 thin_packed_B.data(), beta,
330 y, helper.ldc, opt.multi_thread);
331 } else if (!packed_input_B.empty()) {
332 fastGemmBatch(helper.batch, helper.A_offsets.data(), helper.packed_B_offsets.data(), helper.C_offsets.data(),
333 helper.M, helper.N, helper.K, alpha, a, helper.lda0, helper.lda1,
334 packed_input_B.data(), beta, y, helper.ldc, opt);
335 } else {
336 // truly dynamic B (changes every call — no packing cache available)
337 const auto &B = inputs[1];
338 const auto *b = B.ptr<const float>();
339 fastGemmBatch(helper.batch, helper.A_offsets.data(), helper.B_offsets.data(), helper.C_offsets.data(),
340 helper.M, helper.N, helper.K, alpha, a, helper.lda0, helper.lda1,
341 b, helper.ldb0, helper.ldb1, beta, y, helper.ldc, opt);
342 }
343 }

345 // CV_64F: one cv::gemm call per batch slice (batches don't collapse like Gemm's).
346 void forwardDouble(const std::vector<Mat>& inputs, std::vector<Mat>& outputs)
347 {
348 const Mat &A = inputs[0];
349 Mat &Y = outputs[0];
350 const Mat &B = blobs.empty() ? inputs[1] : blobs[0];
351
352 CV_CheckTypeEQ(B.depth(), CV_64F, "DNN/MatMul: B must be CV_64F to match A");
353 CV_Assert(A.isContinuous() && B.isContinuous() && Y.isContinuous());
354
355 const auto shape_A = shape(A), shape_B = shape(B), shape_Y = shape(Y);
356 const int ma = shape_A[shape_A.size() - 2], na = shape_A.back();
357 const int mb = shape_B[shape_B.size() - 2], nb = shape_B.back();
358 const int M = helper.M, N = helper.N;
359
360 double* yptr0 = Y.ptr<double>();
361 const bool haveBias = (inputs.size() + blobs.size()) >= 3;
362 if (haveBias) {
363 const Mat& bias_mat = blobs.size() >= 2 ? blobs.back() : inputs.back();
364 CV_CheckTypeEQ(bias_mat.depth(), CV_64F, "DNN/MatMul: bias must be CV_64F to match A");
365 if (bias_mat.total() == 1) {
366 std::fill_n(yptr0, Y.total(), (double)beta * bias_mat.ptr<double>()[0]);
367 } else {
368 Mat biasBroadcast(shape_Y, CV_64F);
369 cv::broadcast(bias_mat, shape_Y, biasBroadcast);
370 const double* bb = biasBroadcast.ptr<double>();
371 double b = (double)beta;
372 for (size_t i = 0; i < Y.total(); i++)
373 yptr0[i] = b * bb[i];
374 }
375 } else {
376 std::fill_n(yptr0, Y.total(), 0.0);
377 }
378
379 const int flags = (trans_a ? GEMM_1_T : 0) | (trans_b ? GEMM_2_T : 0);
380 const double* aptr0 = A.ptr<double>();
381 const double* bptr0 = B.ptr<double>();
382
383 for (size_t i = 0; i < helper.batch; i++) {
384 Mat Aview(ma, na, CV_64F, (void*)(aptr0 + helper.A_offsets[i]));
385 Mat Bview(mb, nb, CV_64F, (void*)(bptr0 + helper.B_offsets[i]));
386 Mat Yview(M, N, CV_64F, (void*)(yptr0 + helper.C_offsets[i]));
387 cv::gemm(Aview, Bview, (double)alpha, Yview, 1.0, Yview, flags);
388 }
389 }

391 // No BLAS routine handles integers; accumulates in 64-bit Acc and wraps on cast.
392 template<typename T>
393 void forwardInt(const std::vector<Mat>& inputs, std::vector<Mat>& outputs)
394 {
395 typedef typename std::conditional<std::is_unsigned<T>::value, uint64_t, int64_t>::type Acc;
396
397 const Mat &A = inputs[0];
398 Mat &Y = outputs[0];
399 const Mat &B = blobs.empty() ? inputs[1] : blobs[0];
400
401 CV_CheckTypeEQ(B.depth(), A.depth(), "DNN/MatMul: B must match A's type");
402 CV_Assert(A.isContinuous() && B.isContinuous() && Y.isContinuous());
403
404 const auto shape_Y = shape(Y);
405 const int M = helper.M, N = helper.N, K = helper.K;
406 const int lda0 = helper.lda0, lda1 = helper.lda1;
407 const int ldb0 = helper.ldb0, ldb1 = helper.ldb1;
408 const int ldc = helper.ldc;
409
410 const T* aptr0 = A.ptr<T>();
411 const T* bptr0 = B.ptr<T>();
412 T* yptr0 = Y.ptr<T>();
413
414 // Reject non-integer alpha/beta rather than round: rounding 0.4/0.2 to 0 would silently zero the result.
415 const bool haveBias = (inputs.size() + blobs.size()) >= 3;
416 double alpha_d = (double)alpha, beta_d = haveBias ? (double)beta : 0.0;
417 CV_CheckTrue(std::floor(alpha_d) == alpha_d, "DNN/MatMul: alpha must be an integer value for integer types");
418 CV_CheckTrue(std::floor(beta_d) == beta_d, "DNN/MatMul: beta must be an integer value for integer types");
419 const Acc alphaScale = (Acc)alpha_d;
420 const Acc betaScale = (Acc)beta_d;
421
422 Mat biasBroadcast;
423 if (haveBias) {
424 const Mat& bias_mat = blobs.size() >= 2 ? blobs.back() : inputs.back();
425 CV_CheckTypeEQ(bias_mat.depth(), A.depth(), "DNN/MatMul: bias must match A's type");
426 if ((size_t)bias_mat.total() != Y.total() || shape(bias_mat).size() != shape_Y.size()) {
427 biasBroadcast = Mat(shape_Y, A.depth());
428 cv::broadcast(bias_mat, shape_Y, biasBroadcast);
429 } else {
430 biasBroadcast = bias_mat;
431 }
432 }
433 const T* biasPtr0 = haveBias ? biasBroadcast.ptr<T>() : nullptr;
434
435 parallel_for_(Range(0, (int)(helper.batch * (size_t)M)), [&](const Range& r) {
436 for (int idx = r.start; idx < r.end; idx++) {
437 int bi = idx / M;
438 int m = idx % M;
439 const T* aBase = aptr0 + helper.A_offsets[bi];
440 const T* bBase = bptr0 + helper.B_offsets[bi];
441 T* yRow = yptr0 + helper.C_offsets[bi] + (size_t)m * ldc;
442 const T* biasRow = haveBias ? biasPtr0 + helper.C_offsets[bi] + (size_t)m * ldc : nullptr;
443
444 for (int n = 0; n < N; n++) {
445 Acc sum = 0;
446 for (int k = 0; k < K; k++) {
447 Acc av = (Acc)aBase[(size_t)m * lda0 + (size_t)k * lda1];
448 Acc bv = (Acc)bBase[(size_t)k * ldb0 + (size_t)n * ldb1];
449 sum += av * bv;
450 }
451 Acc result = sum * alphaScale;
452 if (haveBias)
453 result += betaScale * (Acc)biasRow[n];
454 yRow[n] = (T)result;
455 }
456 }
457 }, (double)helper.batch * M * N * (1 / 1024.0));
458 }

460#ifdef HAVE_OPENCL
461 bool forward_ocl(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, InputArrayOfArrays internals) {
462 std::vector<UMat> inputs;
463 std::vector<UMat> outputs;
464
465 bool use_half = (inputs_arr.depth() == CV_16F);
466 inputs_arr.getUMatVector(inputs);
467 outputs_arr.getUMatVector(outputs);
468
469 // does not support bias as input
470 if (inputs.size() >= 3) {
471 return false;
472 }
473
474 const auto &input_A = inputs[0];
475 auto &output = outputs[0];
476 const auto output_shape = shape(output);
477
478 if (blobs.empty()) {
479 weight_umat = inputs[1];
480 if ((inputs.size() + blobs.size() >= 3)) {
481 bias_umat = UMat::zeros(output_shape.size(), output_shape.data(), CV_32F);
482 }
483 } else {
484 if (weight_umat.empty()) {
485 blobs.front().copyTo(weight_umat);
486 }
487 if ((inputs.size() + blobs.size() >= 3)) {
488 if (bias_umat.empty()) {
489 broadcast_bias.copyTo(bias_umat);
490 }
491 } else {
492 if (bias_umat.empty()) {
493 bias_umat = UMat::zeros(output_shape.size(), output_shape.data(), CV_32F);
494 }
495 }
496 }
497
498 auto &input_B = weight_umat;
499
500 int M = static_cast<int>(helper.M),
501 N = static_cast<int>(helper.N),
502 K = static_cast<int>(helper.K),
503 batch = static_cast<int>(helper.batch);
504 int batch_A = total(shape(input_A)) / (M * K),
505 batch_B = total(shape(input_B)) / (N * K);
506 MatShape new_shape_A{batch_A, M * K}, new_shape_B{batch_B, N * K}, new_shape_output{batch, M * N};
507
508 const auto input_A_2d = input_A.reshape(1, new_shape_A.size(), &new_shape_A[0]),
509 input_B_2d = input_B.reshape(1, new_shape_B.size(), &new_shape_B[0]);
510 auto output_2d = output.reshape(1, new_shape_output.size(), &new_shape_output[0]);
511 UMat A, B, C, A_fp32, B_fp32, C_fp32;
512 for (int i = 0; i < batch; i++) {
513 A = input_A_2d.row(helper.A_rows[i]).reshape(1, trans_a ? K : M);
514 B = input_B_2d.row(helper.B_rows[i]).reshape(1, trans_b ? N : K);
515 C = output_2d.row(helper.C_rows[i]).reshape(1, M);
516
517 if (trans_a) {
518 A = A.t();
519 }
520 if (trans_b) {
521 B = B.t();
522 }
523
524 if (use_half) {
525 A.convertTo(A_fp32, CV_32F);
526 B.convertTo(B_fp32, CV_32F);
527 C.convertTo(C_fp32, CV_32F);
528 } else {

538 }

544 }
