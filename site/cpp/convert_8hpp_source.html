32static inline void vx_load_as(const unsigned* ptr, v_float32& a)
33{
34 v_uint32 delta = vx_setall_u32(0x80000000U);
35 v_uint32 ua = vx_load(ptr);
36 v_uint32 mask_a = v_and(v_ge(ua, delta), delta);
37 v_float32 fmask_a = v_cvt_f32(v_reinterpret_as_s32(mask_a)); // 0.f or (float)(-(1 << 31))
38 a = v_cvt_f32(v_reinterpret_as_s32(v_sub(ua, mask_a)));
39 // restore the original values
40 a = v_sub(a, fmask_a); // subtract 0 or a large negative number

334static inline void vx_load_pair_as(const unsigned* ptr, v_float32& a, v_float32& b)
335{
336 v_uint32 delta = vx_setall_u32(0x80000000U);
337 v_uint32 ua = vx_load(ptr);
338 v_uint32 ub = vx_load(ptr + VTraits<v_uint32>::vlanes());
339 v_uint32 mask_a = v_and(v_ge(ua, delta), delta), mask_b = v_and(v_ge(ub, delta), delta);
340 v_float32 fmask_a = v_cvt_f32(v_reinterpret_as_s32(mask_a)); // 0.f or (float)(-(1 << 31))
341 v_float32 fmask_b = v_cvt_f32(v_reinterpret_as_s32(mask_b)); // 0.f or (float)(-(1 << 31))
342 a = v_cvt_f32(v_reinterpret_as_s32(v_sub(ua, mask_a)));
343 b = v_cvt_f32(v_reinterpret_as_s32(v_sub(ub, mask_b)));
344 // restore the original values
345 a = v_sub(a, fmask_a); // subtract 0 or a large negative number
346 b = v_sub(b, fmask_b); // subtract 0 or a large negative number
347}

434static inline void v_store_pair_as(unsigned* ptr, const v_float32& a, const v_float32& b)
435{
436 // v_round(f32) narrows to v_int32, so values in [2^31, 2^32) saturate to INT32_MAX. Scalar for now
437 // (same class as the f64->{u64,s64,u32} stores above); a proper f32->u32 intrinsic can replace it.
438 const int n = VTraits<v_float32>::vlanes();
439 float buf[VTraits<v_float32>::max_nlanes*2];
440 v_store(buf, a); v_store(buf + n, b);
441 for (int i = 0; i < 2*n; i++) ptr[i] = saturate_cast<unsigned>(buf[i]);
442}
