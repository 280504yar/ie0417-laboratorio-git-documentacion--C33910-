89void TrackerStateEstimatorMILBoosting::prepareData(const ConfidenceMap& confidenceMap, Mat& positive, Mat& negative)
90{
91
92 int posCounter = 0;
93 int negCounter = 0;
94
95 for (size_t i = 0; i < confidenceMap.size(); i++)
96 {
97 Ptr<TrackerMILTargetState> currentTargetState = confidenceMap.at(i).first.staticCast<TrackerMILTargetState>();
98 CV_DbgAssert(currentTargetState);
99 if (currentTargetState->isTargetFg())
100 posCounter++;
101 else
102 negCounter++;
103 }
104
105 positive.create(posCounter, numFeatures, CV_32FC1);
106 negative.create(negCounter, numFeatures, CV_32FC1);
107
108 //TODO change with mat fast access
109 //initialize trainData (positive and negative)
110
111 int pc = 0;
112 int nc = 0;
113 for (size_t i = 0; i < confidenceMap.size(); i++)
114 {
115 Ptr<TrackerMILTargetState> currentTargetState = confidenceMap.at(i).first.staticCast<TrackerMILTargetState>();
116 Mat stateFeatures = currentTargetState->getFeatures();
117
118 if (currentTargetState->isTargetFg())
119 {
120 for (int j = 0; j < stateFeatures.rows; j++)
121 {
122 //fill the positive trainData with the value of the feature j for sample i
123 positive.at<float>(pc, j) = stateFeatures.at<float>(j, 0);
124 }
125 pc++;
126 }
127 else
128 {
129 for (int j = 0; j < stateFeatures.rows; j++)
130 {
131 //fill the negative trainData with the value of the feature j for sample i
132 negative.at<float>(nc, j) = stateFeatures.at<float>(j, 0);
133 }
134 nc++;
135 }
136 }
137}
