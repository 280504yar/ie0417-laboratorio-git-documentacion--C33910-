17template<typename T> static void
18mixChannels_( const T** src, const int* sdelta,
19 T** dst, const int* ddelta,
20 int len, int npairs )
21{
22 int i, k;
23 for( k = 0; k < npairs; k++ )
24 {
25 const T* s = src[k];
26 T* d = dst[k];
27 int ds = sdelta[k], dd = ddelta[k];
28 if( s )
29 {
30 for( i = 0; i <= len - 2; i += 2, s += ds*2, d += dd*2 )
31 {
32 T t0 = s[0], t1 = s[ds];
33 d[0] = t0; d[dd] = t1;
34 }
35 if( i < len )
36 d[0] = s[0];
37 }
38 else
39 {
40 for( i = 0; i <= len - 2; i += 2, d += dd*2 )
41 d[0] = d[dd] = 0;
42 if( i < len )
43 d[0] = 0;
44 }
45 }
46}

97void cv::mixChannels( const Mat* src, size_t nsrcs, Mat* dst, size_t ndsts, const int* fromTo, size_t npairs )
98{
99 CV_INSTRUMENT_REGION();
100
101 if( npairs == 0 )
102 return;
103 CV_Assert( src && nsrcs > 0 && dst && ndsts > 0 && fromTo && npairs > 0 );
104
105 size_t i, j, k, esz1 = dst[0].elemSize1();
106 int depth = dst[0].depth();
107
108 AutoBuffer<uchar> buf((nsrcs + ndsts + 1)*(sizeof(Mat*) + sizeof(uchar*)) + npairs*(sizeof(uchar*)*2 + sizeof(int)*6));
109 const Mat** arrays = (const Mat**)(uchar*)buf.data();
110 uchar** ptrs = (uchar**)(arrays + nsrcs + ndsts);
111 const uchar** srcs = (const uchar**)(ptrs + nsrcs + ndsts + 1);
112 uchar** dsts = (uchar**)(srcs + npairs);
113 int* tab = (int*)(dsts + npairs);
114 int *sdelta = (int*)(tab + npairs*4), *ddelta = sdelta + npairs;
115
116 for( i = 0; i < nsrcs; i++ )
117 arrays[i] = &src[i];
118 for( i = 0; i < ndsts; i++ )
119 arrays[i + nsrcs] = &dst[i];
120 ptrs[nsrcs + ndsts] = 0;
121
122 for( i = 0; i < npairs; i++ )
123 {
124 int i0 = fromTo[i*2], i1 = fromTo[i*2+1];
125 if( i0 >= 0 )
126 {
127 for( j = 0; j < nsrcs; i0 -= src[j].channels(), j++ )
128 if( i0 < src[j].channels() )
129 break;
130 CV_Assert(j < nsrcs && src[j].depth() == depth);
131 tab[i*4] = (int)j; tab[i*4+1] = (int)(i0*esz1);
132 sdelta[i] = src[j].channels();
133 }
134 else
135 {
136 tab[i*4] = (int)(nsrcs + ndsts); tab[i*4+1] = 0;
137 sdelta[i] = 0;
138 }
139
140 for( j = 0; j < ndsts; i1 -= dst[j].channels(), j++ )
141 if( i1 < dst[j].channels() )
142 break;
143 CV_Assert(i1 >= 0 && j < ndsts && dst[j].depth() == depth);
144 tab[i*4+2] = (int)(j + nsrcs); tab[i*4+3] = (int)(i1*esz1);
145 ddelta[i] = dst[j].channels();
146 }
147
148 NAryMatIterator it(arrays, ptrs, (int)(nsrcs + ndsts));
149 int total = (int)it.size, blocksize = std::min(total, (int)((BLOCK_SIZE + esz1-1)/esz1));
150 MixChannelsFunc func = getMixchFunc(depth);
151 CV_Assert(func);
152
153 for( i = 0; i < it.nplanes; i++, ++it )
154 {
155 for( k = 0; k < npairs; k++ )
156 {
157 srcs[k] = ptrs[tab[k*4]] + tab[k*4+1];
158 dsts[k] = ptrs[tab[k*4+2]] + tab[k*4+3];
159 }
160
161 for( int t = 0; t < total; t += blocksize )
162 {
163 int bsz = std::min(total - t, blocksize);
164 func( (const void**)srcs, sdelta, (void **)dsts, ddelta, bsz, (int)npairs );
165
166 if( t + blocksize < total )
167 for( k = 0; k < npairs; k++ )
168 {
169 srcs[k] += blocksize*sdelta[k]*esz1;
170 dsts[k] += blocksize*ddelta[k]*esz1;
171 }
172 }
173 }
174}

180static void getUMatIndex(const std::vector<UMat> & um, int cn, int & idx, int & cnidx)
181{
182 int totalChannels = 0;
183 for (size_t i = 0, size = um.size(); i < size; ++i)
184 {
185 int ccn = um[i].channels();
186 totalChannels += ccn;
187
188 if (totalChannels == cn)
189 {
190 idx = (int)(i + 1);
191 cnidx = 0;
192 return;
193 }
194 else if (totalChannels > cn)
195 {
196 idx = (int)i;
197 cnidx = i == 0 ? cn : (cn - totalChannels + ccn);
198 return;
199 }
200 }
201
202 idx = cnidx = -1;
203}

205static bool ocl_mixChannels(InputArrayOfArrays _src, InputOutputArrayOfArrays _dst,
206 const int* fromTo, size_t npairs)
207{
208 std::vector<UMat> src, dst;
209 _src.getUMatVector(src);
210 _dst.getUMatVector(dst);
211
212 size_t nsrc = src.size(), ndst = dst.size();
213 CV_Assert(nsrc > 0 && ndst > 0);
214
215 Size size = src[0].size();
216 int depth = src[0].depth(), esz = CV_ELEM_SIZE(depth),
217 rowsPerWI = ocl::Device::getDefault().isIntel() ? 4 : 1;
218
219 for (size_t i = 1, ssize = src.size(); i < ssize; ++i)
220 CV_Assert(src[i].size() == size && src[i].depth() == depth);
221 for (size_t i = 0, dsize = dst.size(); i < dsize; ++i)
222 CV_Assert(dst[i].size() == size && dst[i].depth() == depth);
223
224 String declsrc, decldst, declproc, declcn, indexdecl;
225 std::vector<UMat> srcargs(npairs), dstargs(npairs);
226
227 for (size_t i = 0; i < npairs; ++i)
228 {
229 int scn = fromTo[i<<1], dcn = fromTo[(i<<1) + 1];
230 int src_idx, src_cnidx, dst_idx, dst_cnidx;
231
232 getUMatIndex(src, scn, src_idx, src_cnidx);
233 getUMatIndex(dst, dcn, dst_idx, dst_cnidx);
234
235 CV_Assert(dst_idx >= 0 && src_idx >= 0);
236
237 srcargs[i] = src[src_idx];
238 srcargs[i].offset += src_cnidx * esz;
239
240 dstargs[i] = dst[dst_idx];
241 dstargs[i].offset += dst_cnidx * esz;
242
243 declsrc += format("DECLARE_INPUT_MAT(%zu)", i);
244 decldst += format("DECLARE_OUTPUT_MAT(%zu)", i);
245 indexdecl += format("DECLARE_INDEX(%zu)", i);
246 declproc += format("PROCESS_ELEM(%zu)", i);
247 declcn += format(" -D scn%zu=%d -D dcn%zu=%d", i, src[src_idx].channels(), i, dst[dst_idx].channels());
248 }
249
250 ocl::Kernel k("mixChannels", ocl::core::mixchannels_oclsrc,
251 format("-D T=%s -D DECLARE_INPUT_MAT_N=%s -D DECLARE_OUTPUT_MAT_N=%s"
252 " -D PROCESS_ELEM_N=%s -D DECLARE_INDEX_N=%s%s",
253 ocl::memopTypeToStr(depth), declsrc.c_str(), decldst.c_str(),
254 declproc.c_str(), indexdecl.c_str(), declcn.c_str()));
255 if (k.empty())
256 return false;
257
258 int argindex = 0;
259 for (size_t i = 0; i < npairs; ++i)
260 argindex = k.set(argindex, ocl::KernelArg::ReadOnlyNoSize(srcargs[i]));
261 for (size_t i = 0; i < npairs; ++i)
262 argindex = k.set(argindex, ocl::KernelArg::WriteOnlyNoSize(dstargs[i]));
263 argindex = k.set(argindex, size.height);
264 argindex = k.set(argindex, size.width);
265 k.set(argindex, rowsPerWI);
266
267 size_t globalsize[2] = { (size_t)size.width, ((size_t)size.height + rowsPerWI - 1) / rowsPerWI };
268 return k.run(2, globalsize, NULL, false);
269}

275void cv::mixChannels(InputArrayOfArrays src, InputOutputArrayOfArrays dst,
276 const int* fromTo, size_t npairs)
277{
278 CV_INSTRUMENT_REGION();
279
280 if (npairs == 0 || fromTo == NULL)
281 return;
282
283 CV_OCL_RUN(dst.isUMatVector(),
284 ocl_mixChannels(src, dst, fromTo, npairs))
285
286 bool src_is_mat = src.kind() != _InputArray::STD_VECTOR_MAT &&
287 src.kind() != _InputArray::STD_ARRAY_MAT &&
288 src.kind() != _InputArray::STD_VECTOR_VECTOR &&
289 src.kind() != _InputArray::STD_VECTOR_UMAT;
290 bool dst_is_mat = dst.kind() != _InputArray::STD_VECTOR_MAT &&
291 dst.kind() != _InputArray::STD_ARRAY_MAT &&
292 dst.kind() != _InputArray::STD_VECTOR_VECTOR &&
293 dst.kind() != _InputArray::STD_VECTOR_UMAT;
294 int i;
295 int nsrc = src_is_mat ? 1 : (int)src.total();
296 int ndst = dst_is_mat ? 1 : (int)dst.total();
297
298 CV_Assert(nsrc > 0 && ndst > 0);
299 cv::AutoBuffer<Mat> _buf(nsrc + ndst);
300 Mat* buf = _buf.data();
301 for( i = 0; i < nsrc; i++ )
302 buf[i] = src.getMat(src_is_mat ? -1 : i);
303 for( i = 0; i < ndst; i++ )
304 buf[nsrc + i] = dst.getMat(dst_is_mat ? -1 : i);
305 mixChannels(&buf[0], nsrc, &buf[nsrc], ndst, fromTo, npairs);
306}

423void cv::extractChannel(InputArray _src, OutputArray _dst, int coi)
424{
425 CV_INSTRUMENT_REGION();
426
427 int type = _src.type(), depth = CV_MAT_DEPTH(type), cn = CV_MAT_CN(type);
428 CV_Assert( 0 <= coi && coi < cn );
429 int ch[] = { coi, 0 };
430
431#ifdef HAVE_OPENCL
432 if (ocl::isOpenCLActivated() && _src.dims() <= 2 && _dst.isUMat())
433 {
434 UMat src = _src.getUMat();
435 // NB: fetch the source before (re)allocating the destination so that
436 // in-place operation (_src and _dst referencing the same array) keeps
437 // the multi-channel source alive while dst is reshaped to 1 channel.
438 _dst.createSameSize(_src, depth);
439 UMat dst = _dst.getUMat();
440 mixChannels(std::vector<UMat>(1, src), std::vector<UMat>(1, dst), ch, 1);
441 return;
442 }
443#endif
444
445 Mat src = _src.getMat();
446 _dst.createSameSize(_src, depth);
447 Mat dst = _dst.getMat();
448
449 CV_IPP_RUN_FAST(ipp_extractChannel(src, dst, coi))
450
451 mixChannels(&src, 1, &dst, 1, ch, 1);
452}
