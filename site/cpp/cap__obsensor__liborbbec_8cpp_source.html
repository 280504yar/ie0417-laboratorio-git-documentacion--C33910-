36VideoCapture_obsensor::VideoCapture_obsensor(int, const cv::VideoCaptureParameters& params)
37{
38 ob::Context::setLoggerToFile(OB_LOG_SEVERITY_OFF, "");
39 config = std::make_shared<ob::Config>();
40 pipe = std::make_shared<ob::Pipeline>();
41#if ORBBEC_SDK_VERSION_MAJOR != 1
42 alignFilter = std::make_shared<ob::Align>(OB_STREAM_COLOR);
43#endif
44
45 int color_width = params.get<int>(CAP_PROP_FRAME_WIDTH, OB_WIDTH_ANY);
46 int color_height = params.get<int>(CAP_PROP_FRAME_HEIGHT, OB_HEIGHT_ANY);
47 int color_fps = params.get<int>(CAP_PROP_FPS, OB_FPS_ANY);
48
49 auto colorProfiles = pipe->getStreamProfileList(OB_SENSOR_COLOR);
50 if (color_width == OB_WIDTH_ANY && color_height == OB_HEIGHT_ANY && color_fps == OB_FPS_ANY)
51 {
52 CV_LOG_INFO(NULL, "Use default color stream profile");
53 auto colorProfile = colorProfiles->getProfile(OB_PROFILE_DEFAULT);
54 config->enableStream(colorProfile->as<ob::VideoStreamProfile>());
55 }
56 else
57 {
58 CV_LOG_INFO(NULL, "Looking for custom color profile " << color_width << "x" << color_height << "@" << color_fps << " fps");
59 auto colorProfile = colorProfiles->getVideoStreamProfile(color_width, color_height, OB_FORMAT_MJPG, color_fps);
60 config->enableStream(colorProfile->as<ob::VideoStreamProfile>());
61 }
62
63 int depth_width = params.get<int>(CAP_PROP_OBSENSOR_DEPTH_WIDTH, OB_WIDTH_ANY);
64 int depth_height = params.get<int>(CAP_PROP_OBSENSOR_DEPTH_HEIGHT, OB_HEIGHT_ANY);
65 int depth_fps = params.get<int>(CAP_PROP_OBSENSOR_DEPTH_FPS, OB_FPS_ANY);
66
67 auto depthProfiles = pipe->getStreamProfileList(OB_SENSOR_DEPTH);
68 if (depth_width == OB_WIDTH_ANY && depth_height == OB_HEIGHT_ANY && depth_fps == OB_FPS_ANY)
69 {
70 CV_LOG_INFO(NULL, "Use default depth stream profile");
71 auto depthProfile = depthProfiles->getProfile(OB_PROFILE_DEFAULT);
72 config->enableStream(depthProfile->as<ob::VideoStreamProfile>());
73 }
74 else
75 {
76 CV_LOG_INFO(NULL, "Looking for custom color profile " << depth_width << "x" << depth_height << "@" << depth_fps << " fps");
77 auto depthProfile = depthProfiles->getVideoStreamProfile(depth_width, depth_height, OB_FORMAT_Y14, depth_fps);
78 config->enableStream(depthProfile->as<ob::VideoStreamProfile>());
79 }
80
81#if ORBBEC_SDK_VERSION_MAJOR == 1
82 config->setAlignMode(ALIGN_D2C_SW_MODE);
83#else
84 config->setFrameAggregateOutputMode(OB_FRAME_AGGREGATE_OUTPUT_ALL_TYPE_FRAME_REQUIRE);
85 pipe->enableFrameSync();
86#endif
87
88 pipe->start(config, [&](std::shared_ptr<ob::FrameSet> frameset) {
89 std::unique_lock<std::mutex> lk(videoFrameMutex);
90#if ORBBEC_SDK_VERSION_MAJOR == 1
91 colorFrame = frameset->colorFrame();
92 depthFrame = frameset->depthFrame();
93#else
94 auto alignFrameSet = alignFilter->process(frameset);
95 if (alignFrameSet) {
96 colorFrame = alignFrameSet->as<ob::FrameSet>()->colorFrame();
97 depthFrame = alignFrameSet->as<ob::FrameSet>()->depthFrame();
98 }
99#endif
100 });
101
102 auto param = pipe->getCameraParam();
103 camParam.intrinsicColor[0] = param.rgbIntrinsic.fx;
104 camParam.intrinsicColor[1] = param.rgbIntrinsic.fy;
105 camParam.intrinsicColor[2] = param.rgbIntrinsic.cx;
106 camParam.intrinsicColor[3] = param.rgbIntrinsic.cy;
107
108 camParam.distortionColor[0] = param.depthDistortion.k1;
109 camParam.distortionColor[1] = param.depthDistortion.k2;
110 camParam.distortionColor[2] = param.depthDistortion.k3;
111 camParam.distortionColor[3] = param.depthDistortion.k4;
112 camParam.distortionColor[4] = param.depthDistortion.k5;
113 camParam.distortionColor[5] = param.depthDistortion.k6;
114 camParam.distortionColor[6] = param.depthDistortion.p1;
115 camParam.distortionColor[7] = param.depthDistortion.p2;
116}

122double VideoCapture_obsensor::getProperty(int propIdx) const
123{
124 double rst = CAP_PROP_UNKNOWN;
125 propIdx = propIdx & (~CAP_OBSENSOR_GENERATORS_MASK);
126 switch (propIdx)
127 {
128 case CAP_PROP_OBSENSOR_INTRINSIC_FX:
129 rst = camParam.intrinsicColor[0];
130 break;
131 case CAP_PROP_OBSENSOR_INTRINSIC_FY:
132 rst = camParam.intrinsicColor[1];
133 break;
134 case CAP_PROP_OBSENSOR_INTRINSIC_CX:
135 rst = camParam.intrinsicColor[2];
136 break;
137 case CAP_PROP_OBSENSOR_INTRINSIC_CY:
138 rst = camParam.intrinsicColor[3];
139 break;
140 case CAP_PROP_OBSENSOR_COLOR_DISTORTION_K1:
141 rst = camParam.distortionColor[0];
142 break;
143 case CAP_PROP_OBSENSOR_COLOR_DISTORTION_K2:
144 rst = camParam.distortionColor[1];
145 break;
146 case CAP_PROP_OBSENSOR_COLOR_DISTORTION_K3:
147 rst = camParam.distortionColor[2];
148 break;
149 case CAP_PROP_OBSENSOR_COLOR_DISTORTION_K4:
150 rst = camParam.distortionColor[3];
151 break;
152 case CAP_PROP_OBSENSOR_COLOR_DISTORTION_K5:
153 rst = camParam.distortionColor[4];
154 break;
155 case CAP_PROP_OBSENSOR_COLOR_DISTORTION_K6:
156 rst = camParam.distortionColor[5];
157 break;
158 case CAP_PROP_OBSENSOR_COLOR_DISTORTION_P1:
159 rst = camParam.distortionColor[6];
160 break;
161 case CAP_PROP_OBSENSOR_COLOR_DISTORTION_P2:
162 rst = camParam.distortionColor[7];
163 break;
164
165 case CAP_PROP_POS_MSEC:
166 case CAP_PROP_OBSENSOR_RGB_POS_MSEC:
167 if (grabbedColorFrame)
168 {
169 rst = grabbedColorFrame->globalTimeStampUs();
170 if (rst == 0.0)
171 {
172 CV_LOG_ONCE_WARNING(NULL, "Camera reports zero global timestamp. System timestamp is used instead.");
173 rst = grabbedColorFrame->systemTimeStamp();
174 }
175 }
176 break;
177 case CAP_PROP_OBSENSOR_DEPTH_POS_MSEC:
178 if (grabbedDepthFrame)
179 {
180 rst = grabbedDepthFrame->systemTimeStamp();
181 if (rst == 0.0)
182 {
183 CV_LOG_ONCE_WARNING(NULL, "Camera reports zero global timestamp. System timestamp is used instead.");
184 rst = grabbedDepthFrame->systemTimeStamp();
185 }
186 }
187 break;
188 }
189
190 return rst;
191}

221bool VideoCapture_obsensor::retrieveFrame(int outputType, cv::OutputArray frame)
222{
223 switch (outputType)
224 {
225 case CAP_OBSENSOR_BGR_IMAGE:
226 if(grabbedColorFrame != nullptr){
227 auto format = grabbedColorFrame->format();
228 if(format != OB_FORMAT_MJPEG){
229 CV_LOG_WARNING(NULL, "Unsupported color frame format");
230 return false;
231 }
232 auto mjpgMat = Mat(1, grabbedColorFrame->dataSize() , CV_8UC1, grabbedColorFrame->data()).clone();
233 auto bgrMat = imdecode(mjpgMat, IMREAD_COLOR);
234 if(bgrMat.empty()){
235 CV_LOG_WARNING(NULL, "Failed to decode color frame");
236 return false;
237 }
238 bgrMat.copyTo(frame);
239 return true;
240 }
241 break;
242 case CAP_OBSENSOR_DEPTH_MAP:
243 if(grabbedDepthFrame != nullptr){
244 auto format = grabbedDepthFrame->format();
245 if(format != OB_FORMAT_Y16){
246 CV_LOG_WARNING(NULL, "Unsupported depth frame format");
247 return false;
248 }
249 Mat(grabbedDepthFrame->height(), grabbedDepthFrame->width(), CV_16UC1, grabbedDepthFrame->data()).copyTo(frame);
250 return true;
251 }
252 break;
253 default:
254 return false;
255 }
256
257 return false;
258}
