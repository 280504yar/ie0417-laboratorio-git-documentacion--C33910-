29 void forward(
30 const std::vector<cv::Ptr<BackendWrapper>>& inputs,
31 const std::vector<cv::Ptr<BackendWrapper>>& outputs,
32 csl::Workspace& workspace) override
33 {
34 CV_Assert(outputs.size() == 1);
35
36 auto output_wrapper = outputs[0].dynamicCast<wrapper_type>();
37 auto output = output_wrapper->getSpan();
38
39 auto input_wrapper = inputs[0].dynamicCast<wrapper_type>();
40 auto input = input_wrapper->getView();
41
42 /* output shape is determined by the input shape */
43 CV_Assert(is_shape_same(output, input));
44
45 for (int i = 1; i < inputs.size(); i++)
46 {
47 auto from_wrapper = inputs[i].dynamicCast<wrapper_type>();
48 auto from = from_wrapper->getView();
49
50 CV_Assert(output.rank() == from.rank());
51 for (int i = 0; i < output.rank(); i++) {
52 if (i != 1) {
53 CV_Assert(from.get_axis_size(i) == output.get_axis_size(i));
54 }
55 }
56
57 if (i == 1)
58 {
59 /* optimized path for first two inputs */
60 kernels::input_shortcut<T>(stream, output, input, from);
61 }
62 else
63 {
64 kernels::input_shortcut<T>(stream, output, output, from);
65 }
66 }
67
68 }
