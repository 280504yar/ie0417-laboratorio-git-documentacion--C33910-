145struct RegionStatisticsStatus
146{
147 int _skipDepth;
148#ifdef HAVE_IPP
149 int ignoreDepthImplIPP;
150#endif
151#ifdef HAVE_OPENCL
152 int ignoreDepthImplOpenCL;
153#endif
154
155 RegionStatisticsStatus() { reset(); }
156
157 void reset()
158 {
159 _skipDepth = -1;
160#ifdef HAVE_IPP
161 ignoreDepthImplIPP = 0;
162#endif
163#ifdef HAVE_OPENCL
164 ignoreDepthImplOpenCL = 0;
165#endif
166 }
167
168 void propagateFrom(const RegionStatisticsStatus& src)
169 {
170 _skipDepth = -1;
171 if (src._skipDepth >= 0)
172 enableSkipMode(0);
173#ifdef HAVE_IPP
174 ignoreDepthImplIPP = src.ignoreDepthImplIPP ? 1 : 0;
175#endif
176#ifdef HAVE_OPENCL
177 ignoreDepthImplOpenCL = src.ignoreDepthImplOpenCL ? 1 : 0;
178#endif
179 }
180
181 void enableSkipMode(int depth);
182 void checkResetSkipMode(int leaveDepth);
183};
