22 struct MaxPoolingConfiguration {
23 /* the size of the following vectors must be equal to the pooling order */
24 std::vector<std::size_t> window_size;
25 std::vector<std::size_t> strides;
26
27 enum class PaddingMode {
28 MANUAL, /* uses explicit padding values provided in `pads_begin` and `pads_end` */
29 VALID, /* no padding is added */
30 SAME /* TensorFlow logic is used for same padding */
31 };
32
33 PaddingMode padMode;
34
35 /* explicit paddings are used if and only if padMode is set to manual */
36 std::vector<std::size_t> pads_begin;
37
38 /* full shape inclusive of channel and batch axis */
39 std::vector<std::size_t> input_shape;
40 };

42 template <class T, class T_INDEX>
43 class MaxPoolingOp final : public CUDABackendNode {
44 public:
45 using wrapper_type = GetCUDABackendWrapperType<T>;
46
47 MaxPoolingOp(csl::Stream stream_, const MaxPoolingConfiguration& config)
48 : stream(std::move(stream_))
49 {
50 window_size = config.window_size;
51
52 const auto pooling_order = window_size.size();
53
54 strides = config.strides;
55 CV_Assert(pooling_order == strides.size());
56
57 if (pooling_order < 1 || pooling_order > 3)
58 CV_Error(Error::StsNotImplemented, "Only 1D/2D/3D max-pooling are supported.");
59
60 padding_left.resize(pooling_order);
61 if (config.padMode == MaxPoolingConfiguration::PaddingMode::MANUAL)
62 {
63 const auto& pads_begin = config.pads_begin;
64 CV_Assert(pooling_order == pads_begin.size());
65
66 padding_left.assign(std::begin(pads_begin), std::end(pads_begin));
67 }
68 else if (config.padMode == MaxPoolingConfiguration::PaddingMode::VALID)
69 {
70 /* nothing to do as the paddings are already preset to zero */
71 }
72 else if (config.padMode == MaxPoolingConfiguration::PaddingMode::SAME)
73 {
74 /* TensorFlow Logic:
75 * total_padding[i] = (o[i] - 1) * s[i] + effective_k[i] - i[i]
76 *
77 * if total padding is odd, the extra is added towards the end
78 */
79 const auto& input_shape = config.input_shape;
80 CV_Assert(input_shape.size() == pooling_order + 2);
81
82 for (int i = 0; i < pooling_order; i++)
83 {
84 const auto output_dim = (input_shape[i + 2] - 1 + strides[i]) / strides[i];
85 const auto required_total_padding =
86 std::max<std::int64_t>(0, (output_dim - 1) * strides[i] + window_size[i] - input_shape[i + 2]);
87
88 padding_left[i] = required_total_padding / 2;
89 }
90 }
91 }

136 struct MaxUnpoolingConfiguration {
137 /* the size of the following vectors must be equal to the unpooling order */
138 std::vector<std::size_t> window_size;
139 std::vector<std::size_t> strides;
140 std::vector<std::size_t> pads_begin;

143 template <class T, class T_INDEX>
144 class MaxUnpoolingOp final : public CUDABackendNode {
145 public:
146 using wrapper_type = GetCUDABackendWrapperType<T>;
147
148 MaxUnpoolingOp(csl::Stream stream_, const MaxUnpoolingConfiguration& config)
149 : stream(std::move(stream_))
150 {
151 window_size = config.window_size;
152
153 const auto pooling_order = window_size.size();
154 CV_Assert(pooling_order >= 1);
155
156 strides = config.strides;
157 padding_left = config.pads_begin;
158 CV_Assert(strides.size() == pooling_order);
159 CV_Assert(padding_left.size() == pooling_order);
160
161 if (pooling_order != 2 && pooling_order != 3)
162 CV_Error(Error::StsNotImplemented, "Only 2D/3D max-unpooling are supported.");
163 }

165 void forward(
166 const std::vector<cv::Ptr<BackendWrapper>>& inputs,
167 const std::vector<cv::Ptr<BackendWrapper>>& outputs,
168 csl::Workspace& workspace) override
169 {
170 /* sometimes a third input is passed to provide the output shape; we won't need it */
171 CV_Assert(inputs.size() == 2 || inputs.size() == 3);
172 CV_Assert(outputs.size() >= 1);
173
174 for(int i = 0; i < outputs.size(); i++)
175 {
176 auto input_wrapper = inputs[0].dynamicCast<wrapper_type>();
177 auto input_data = input_wrapper->getView();
178
179 auto indices_wrapper = inputs[1].dynamicCast<GetCUDABackendWrapperType<T_INDEX>>();
180 auto input_indices = indices_wrapper->getView();
181
182 auto output_wrapper = outputs[i].dynamicCast<wrapper_type>();
183 auto output_data = output_wrapper->getSpan();
184
185 kernels::max_unpooling<T, T_INDEX>(stream, output_data, input_data, input_indices, window_size, strides, padding_left);
186 }
187 }
188
189 private:
190 csl::Stream stream;
191
192 std::vector<std::size_t> window_size, strides, padding_left;
193 };
