14inline bool hasExtension(const String &filename, const String &ext)
15{
16 if (filename.size() <= ext.size())
17 return false;
18 const size_t diff = filename.size() - ext.size();
19 const size_t found_at = filename.rfind(ext);
20 return found_at == diff;
21}

23inline mfxU32 determineCodecId(const String &filename)
24{
25 if (hasExtension(filename, ".h264") || hasExtension(filename, ".264"))
26 return MFX_CODEC_AVC;
27 else if (hasExtension(filename, ".mp2") || hasExtension(filename, ".mpeg2"))
28 return MFX_CODEC_MPEG2;
29 else if (hasExtension(filename, ".265") || hasExtension(filename, ".hevc"))
30 return MFX_CODEC_HEVC;
31 else
32 return (mfxU32)-1;
33}

37VideoCapture_IntelMFX::VideoCapture_IntelMFX(const cv::String &filename)
38 : session(0), plugin(0), deviceHandler(0), bs(0), decoder(0), pool(0), outSurface(0), good(false)
39{
40 mfxStatus res = MFX_ERR_NONE;
41
42 // Init device and session
43 deviceHandler = createDeviceHandler();
44 session = new MFXVideoSession_WRAP();
45 if (!deviceHandler->init(*session))
46 {
47 MSG(cerr << "MFX: Can't initialize session" << endl);
48 return;
49 }
50
51 // Load appropriate plugin
52
53 mfxU32 codecId = determineCodecId(filename);
54 if (codecId == (mfxU32)-1)
55 {
56 MSG(cerr << "MFX: Unsupported extension: " << filename << endl);
57 return;
58 }
59 plugin = Plugin::loadDecoderPlugin(*session, codecId);
60 if (plugin && !plugin->isGood())
61 {
62 MSG(cerr << "MFX: LoadPlugin failed for codec: " << codecId << " (" << filename << ")" << endl);
63 return;
64 }
65
66 // Read some content from file
67
68 bs = new ReadBitstream(filename.c_str());
69 if (!bs->read())
70 {
71 MSG(cerr << "MFX: Failed to read bitstream" << endl);
72 return;
73 }
74
75 // Create decoder and decode stream header
76
77 decoder = new MFXVideoDECODE(*session);
78 mfxVideoParam params;
79 memset(&params, 0, sizeof(params));
80 params.mfx.CodecId = codecId;
81 params.IOPattern = MFX_IOPATTERN_OUT_SYSTEM_MEMORY;
82 res = decoder->DecodeHeader(&bs->stream, &params);
83 DBG(cout << "DecodeHeader: " << res << endl << params.mfx << params.mfx.FrameInfo << endl);
84 if (res < MFX_ERR_NONE)
85 {
86 MSG(cerr << "MFX: Failed to decode stream header: " << res << endl);
87 return;
88 }
89
90 // Adjust parameters - COMMENTED: h265 decoder resets crop size to 0 (oneVPL/Win)
91
92 //res = decoder->Query(&params, &params);
93 //DBG(cout << "MFX Query: " << res << endl << params.mfx << params.mfx.FrameInfo);
94 //CV_Assert(res >= MFX_ERR_NONE);
95
96 // Init surface pool
97
98 pool = SurfacePool::create(decoder, params);
99 if (!pool)
100 {
101 MSG(cerr << "MFX: Failed to create surface pool" << endl);
102 return;
103 }
104
105 // Init decoder
106
107 res = decoder->Init(&params);
108 DBG(cout << "MFX decoder Init: " << res << endl << params.mfx.FrameInfo);
109 if (res < MFX_ERR_NONE)
110 {
111 MSG(cerr << "MFX: Failed to init decoder: " << res << endl);
112 return;
113 }
114
115 frameSize = Size(params.mfx.FrameInfo.CropW, params.mfx.FrameInfo.CropH);
116 if (frameSize == Size(0, 0)) // sometimes Crop size is 0
117 {
118 frameSize = Size(params.mfx.FrameInfo.Width, params.mfx.FrameInfo.Height);
119 }
120 good = true;
121}

160bool VideoCapture_IntelMFX::grabFrame()
161{
162 mfxStatus res;
163 mfxFrameSurface1 *workSurface = 0;
164 mfxSyncPoint sync;
165
166 workSurface = pool->getFreeSurface();
167
168 while (true)
169 {
170 if (!workSurface)
171 {
172 // not enough surfaces
173 MSG(cerr << "MFX: Failed to get free surface" << endl);
174 return false;
175 }
176
177 outSurface = 0;
178 res = decoder->DecodeFrameAsync(bs->drain ? 0 : &bs->stream, workSurface, (mfxFrameSurface1**)&outSurface, &sync);
179 if (res == MFX_ERR_NONE)
180 {
181 res = session->SyncOperation(sync, 1000); // 1 sec, TODO: provide interface to modify timeout
182 if (res == MFX_ERR_NONE)
183 {
184 // ready to retrieve
185 DBG(cout << "Frame ready to retrieve" << endl);
186 return true;
187 }
188 else
189 {
190 MSG(cerr << "MFX: Sync error: " << res << endl);
191 return false;
192 }
193 }
194 else if (res == MFX_ERR_MORE_DATA)
195 {
196 if (bs->isDone())
197 {
198 if (bs->drain)
199 {
200 // finish
201 DBG(cout << "Drain finished" << endl);
202 return false;
203 }
204 else
205 {
206 DBG(cout << "Bitstream finished - Drain started" << endl);
207 bs->drain = true;
208 continue;
209 }
210 }
211 else
212 {
213 bool read_res = bs->read();
214 if (!read_res)
215 {
216 // failed to read
217 MSG(cerr << "MFX: Bitstream read failure" << endl);
218 return false;
219 }
220 else
221 {
222 DBG(cout << "Bitstream read success" << endl);
223 continue;
224 }
225 }
226 }
227 else if (res == MFX_ERR_MORE_SURFACE)
228 {
229 DBG(cout << "Getting another surface" << endl);
230 workSurface = pool->getFreeSurface();
231 continue;
232 }
233 else if (res == MFX_WRN_DEVICE_BUSY)
234 {
235 DBG(cout << "Waiting for device" << endl);
236 sleep_ms(1000);
237 continue;
238 }
239 else if (res == MFX_WRN_VIDEO_PARAM_CHANGED)
240 {
241 DBG(cout << "Video param changed" << endl);
242 continue;
243 }
244 else
245 {
246 MSG(cerr << "MFX: Bad status: " << res << endl);
247 return false;
248 }
249 }
250}

253bool VideoCapture_IntelMFX::retrieveFrame(int, OutputArray out)
254{
255 if (!outSurface)
256 {
257 MSG(cerr << "MFX: No frame ready to retrieve" << endl);
258 return false;
259 }
260 mfxFrameSurface1 * s = (mfxFrameSurface1*)outSurface;
261 mfxFrameInfo &info = s->Info;
262 mfxFrameData &data = s->Data;
263
264 const int cols = info.CropW;
265 const int rows = info.CropH;
266
267 out.create(rows, cols, CV_8UC3);
268 Mat res = out.getMat();
269
270 hal::cvtTwoPlaneYUVtoBGR(data.Y, data.UV, data.Pitch, res.data, res.step, cols, rows, 3, false, 0);
271
272 return true;
273}
