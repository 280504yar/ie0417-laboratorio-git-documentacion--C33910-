49static void init_maskbuf(ST* maskbuf, int cn, int simd_width)
50{
51 memset(maskbuf, 0, simd_width*9*sizeof(maskbuf[0]));
52 if (cn == 1)
53 ;
54 else if (cn == 2)
55 for (int i = 0; i < simd_width; i += 2) {
56 maskbuf[i] = (ST)-1;

229template<typename T, typename ST, typename WT=T>
230static int sum_(const T* src0, const uchar* mask, ST* dst, int len, int cn )
231{
232 const T* src = src0;
233 if( !mask )
234 {
235 Sum_SIMD<T, ST> vop(cn);
236 int i0 = vop(src0, mask, dst, len, cn), i = i0, k = cn % 4;
237 src += i0 * cn;
238
239 if( k == 1 )
240 {
241 ST s0 = dst[0];
242
243 #if CV_ENABLE_UNROLLED
244 for(; i <= len - 4; i += 4, src += cn*4 )
245 s0 += (WT)src[0] + (WT)src[cn] + (WT)src[cn*2] + (WT)src[cn*3];
246 #endif
247 for( ; i < len; i++, src += cn )
248 s0 += (WT)src[0];
249 dst[0] = s0;
250 }
251 else if( k == 2 )
252 {
253 ST s0 = dst[0], s1 = dst[1];
254 for( ; i < len; i++, src += cn )
255 {
256 s0 += (WT)src[0];
257 s1 += (WT)src[1];
258 }
259 dst[0] = s0;
260 dst[1] = s1;
261 }
262 else if( k == 3 )
263 {
264 ST s0 = dst[0], s1 = dst[1], s2 = dst[2];
265 for( ; i < len; i++, src += cn )
266 {
267 s0 += (WT)src[0];
268 s1 += (WT)src[1];
269 s2 += (WT)src[2];
270 }
271 dst[0] = s0;
272 dst[1] = s1;
273 dst[2] = s2;
274 }
275
276 for( ; k < cn; k += 4 )
277 {
278 src = src0 + i0*cn + k;
279 ST s0 = dst[k], s1 = dst[k+1], s2 = dst[k+2], s3 = dst[k+3];
280 for( i = i0; i < len; i++, src += cn )
281 {
282 s0 += (WT)src[0]; s1 += (WT)src[1];
283 s2 += (WT)src[2]; s3 += (WT)src[3];
284 }
285 dst[k] = s0;
286 dst[k+1] = s1;
287 dst[k+2] = s2;
288 dst[k+3] = s3;
289 }
290 return len;
291 }
292
293 int i, nzm = 0;
294 if( cn == 1 )
295 {
296 ST s = dst[0];
297 for( i = 0; i < len; i++ )
298 if( mask[i] )
299 {
300 s += (WT)src[i];
301 nzm++;
302 }
303 dst[0] = s;
304 }
305 else if( cn == 3 )
306 {
307 ST s0 = dst[0], s1 = dst[1], s2 = dst[2];
308 for( i = 0; i < len; i++, src += 3 )
309 if( mask[i] )
310 {
311 s0 += (WT)src[0];
312 s1 += (WT)src[1];
313 s2 += (WT)src[2];
314 nzm++;
315 }
316 dst[0] = s0;
317 dst[1] = s1;
318 dst[2] = s2;
319 }
320 else
321 {
322 for( i = 0; i < len; i++, src += cn )
323 if( mask[i] )
324 {
325 int k = 0;
326 #if CV_ENABLE_UNROLLED
327 for( ; k <= cn - 4; k += 4 )
328 {
329 ST s0, s1;
330 s0 = dst[k] + (WT)src[k];
331 s1 = dst[k+1] + (WT)src[k+1];
332 dst[k] = s0; dst[k+1] = s1;
333 s0 = dst[k+2] + (WT)src[k+2];
334 s1 = dst[k+3] + (WT)src[k+3];
335 dst[k+2] = s0; dst[k+3] = s1;
336 }
337 #endif
338 for( ; k < cn; k++ )
339 dst[k] += (WT)src[k];
340 nzm++;
341 }
342 }
343 return nzm;
344}
