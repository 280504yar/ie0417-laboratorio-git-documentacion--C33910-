91void IndexParams::getAll(std::vector<String>& names,
92 std::vector<FlannIndexType>& types,
93 std::vector<String>& strValues,
94 std::vector<double>& numValues) const
95{
96 names.clear();
97 types.clear();
98 strValues.clear();
99 numValues.clear();
100
101 ::cvflann::IndexParams& p = get_params(*this);
102 ::cvflann::IndexParams::const_iterator it = p.begin(), it_end = p.end();
103
104 for( ; it != it_end; ++it )
105 {
106 names.push_back(it->first);
107 try
108 {
109 String val = it->second.cast<String>();
110 types.push_back(FLANN_INDEX_TYPE_STRING);
111 strValues.push_back(val);
112 numValues.push_back(-1);
113 continue;
114 }
115 catch (...) {}
116
117 strValues.push_back(it->second.type().name());
118
119 try
120 {
121 double val = it->second.cast<double>();
122 types.push_back(FLANN_INDEX_TYPE_64F);
123 numValues.push_back(val);
124 continue;
125 }
126 catch (...) {}
127 try
128 {
129 float val = it->second.cast<float>();
130 types.push_back(FLANN_INDEX_TYPE_32F);
131 numValues.push_back(val);
132 continue;
133 }
134 catch (...) {}
135 try
136 {
137 int val = it->second.cast<int>();
138 types.push_back(FLANN_INDEX_TYPE_32S);
139 numValues.push_back(val);
140 continue;
141 }
142 catch (...) {}
143 try
144 {
145 short val = it->second.cast<short>();
146 types.push_back(FLANN_INDEX_TYPE_16S);
147 numValues.push_back(val);
148 continue;
149 }
150 catch (...) {}
151 try
152 {
153 ushort val = it->second.cast<ushort>();
154 types.push_back(FLANN_INDEX_TYPE_16U);
155 numValues.push_back(val);
156 continue;
157 }
158 catch (...) {}
159 try
160 {
161 char val = it->second.cast<char>();
162 types.push_back(FLANN_INDEX_TYPE_8S);
163 numValues.push_back(val);
164 continue;
165 }
166 catch (...) {}
167 try
168 {
169 uchar val = it->second.cast<uchar>();
170 types.push_back(FLANN_INDEX_TYPE_8U);
171 numValues.push_back(val);
172 continue;
173 }
174 catch (...) {}
175 try
176 {
177 bool val = it->second.cast<bool>();
178 types.push_back(FLANN_INDEX_TYPE_BOOL);
179 numValues.push_back(val);
180 continue;
181 }
182 catch (...) {}
183 try
184 {
185 cvflann::flann_algorithm_t val = it->second.cast<cvflann::flann_algorithm_t>();
186 types.push_back(FLANN_INDEX_TYPE_ALGORITHM);
187 numValues.push_back(val);
188 continue;
189 }
190 catch (...) {}
191
192
193 types.push_back((FlannIndexType)-1); // unknown type
194 numValues.push_back(-1);
195 }

297SearchParams::SearchParams( int checks, float eps, bool sorted, bool explore_all_trees )
298{
299 ::cvflann::IndexParams& p = get_params(*this);
300
301 // how many leafs to visit when searching for neighbours (-1 for unlimited)
302 p["checks"] = checks;
303 // search for eps-approximate neighbours (default: 0)

305 // only for radius search, require neighbours sorted by distance (default: true)
306 p["sorted"] = sorted;

333{

335 if(DataType<ElementType>::type != data.type())

339

348 {
349 delete _index;
350 _index = NULL;

388void Index::build(InputArray _data, const IndexParams& params, flann_distance_t _distType)
389{
390 CV_INSTRUMENT_REGION();
391
392 release();
393
394 // Index may reuse 'data' during search, need to keep it alive
395 features_clone = _data.getMat().clone();
396 Mat data = features_clone;
397
398 algo = getParam<flann_algorithm_t>(params, "algorithm", FLANN_INDEX_LINEAR);
399 if( algo == FLANN_INDEX_SAVED )
400 {
401 load_(getParam<String>(params, "filename", String()));
402 return;
403 }
404
405 index = 0;
406 featureType = data.type();
407 distType = _distType;
408
409 if ( algo == FLANN_INDEX_LSH)
410 {
411 distType = FLANN_DIST_HAMMING;
412 }
413
414 switch( distType )
415 {
416 case FLANN_DIST_HAMMING:
417 buildIndex< HammingDistance >(index, data, params);
418 break;
419 case FLANN_DIST_L2:
420 buildIndex< ::cvflann::L2<float> >(index, data, params);
421 break;
422 case FLANN_DIST_L1:
423 buildIndex< ::cvflann::L1<float> >(index, data, params);
424 break;
425#if MINIFLANN_SUPPORT_EXOTIC_DISTANCE_TYPES
426 case FLANN_DIST_DNAMMING:
427 buildIndex< DNAmmingDistance >(index, data, params);
428 break;
429 case FLANN_DIST_MAX:
430 buildIndex< ::cvflann::MaxDistance<float> >(index, data, params);
431 break;
432 case FLANN_DIST_HIST_INTERSECT:
433 buildIndex< ::cvflann::HistIntersectionDistance<float> >(index, data, params);
434 break;
435 case FLANN_DIST_HELLINGER:
436 buildIndex< ::cvflann::HellingerDistance<float> >(index, data, params);
437 break;
438 case FLANN_DIST_CHI_SQUARE:
439 buildIndex< ::cvflann::ChiSquareDistance<float> >(index, data, params);
440 break;
441 case FLANN_DIST_KL:
442 buildIndex< ::cvflann::KL_Divergence<float> >(index, data, params);
443 break;
444#endif
445 default:
446 CV_Error(Error::StsBadArg, "Unknown/unsupported distance type");
447 }

511template<typename Distance, typename IndexType>
512void runKnnSearch_(void* index, const Mat& query, Mat& indices, Mat& dists,
513 int knn, const SearchParams& params)
514{
515 typedef typename Distance::ElementType ElementType;
516 typedef typename Distance::ResultType DistanceType;
517 int type = DataType<ElementType>::type;
518 int dtype = DataType<DistanceType>::type;
519 IndexType* index_ = (IndexType*)index;
520
521 CV_Assert((size_t)knn <= index_->size());
522 CV_Assert(query.type() == type && indices.type() == CV_32S && dists.type() == dtype);
523 CV_Assert(query.isContinuous() && indices.isContinuous() && dists.isContinuous());
524
525 ::cvflann::Matrix<ElementType> _query((ElementType*)query.data, query.rows, query.cols);
526 ::cvflann::Matrix<int> _indices(indices.ptr<int>(), indices.rows, indices.cols);
527 ::cvflann::Matrix<DistanceType> _dists(dists.ptr<DistanceType>(), dists.rows, dists.cols);
528
529 index_->knnSearch(_query, _indices, _dists, knn,
530 (const ::cvflann::SearchParams&)get_params(params));
531}

568static void createIndicesDists(OutputArray _indices, OutputArray _dists,
569 Mat& indices, Mat& dists, int rows,
570 int minCols, int maxCols, int dtype)
571{
572 if( _indices.needed() )
573 {
574 indices = _indices.getMat();
575 if( !indices.isContinuous() || indices.type() != CV_32S ||
576 indices.rows != rows || indices.cols < minCols || indices.cols > maxCols )
577 {
578 if( !indices.isContinuous() )
579 _indices.release();
580 _indices.create( rows, minCols, CV_32S );
581 indices = _indices.getMat();
582 }
583 }
584 else
585 indices.create( rows, minCols, CV_32S );
586
587 if( _dists.needed() )
588 {
589 dists = _dists.getMat();
590 if( !dists.isContinuous() || dists.type() != dtype ||
591 dists.rows != rows || dists.cols < minCols || dists.cols > maxCols )
592 {
593 if( !_dists.isContinuous() )
594 _dists.release();
595 _dists.create( rows, minCols, dtype );
596 dists = _dists.getMat();
597 }
598 }
599 else
600 dists.create( rows, minCols, dtype );
601}

606{

608
609 Mat query = _query.getMat(), indices, dists;
610 int dtype = (distType == FLANN_DIST_HAMMING)

613 createIndicesDists( _indices, _dists, indices, dists, query.rows, knn, knn, dtype );
614
615 switch( distType )

619 break;
620 case FLANN_DIST_L2:

622 break;

625 break;

629 break;
630 case FLANN_DIST_MAX:
631 runKnnSearch< ::cvflann::MaxDistance<float> >(index, query, indices, dists, knn, params);
632 break;
633 case FLANN_DIST_HIST_INTERSECT:
634 runKnnSearch< ::cvflann::HistIntersectionDistance<float> >(index, query, indices, dists, knn, params);
635 break;
636 case FLANN_DIST_HELLINGER:

638 break;
639 case FLANN_DIST_CHI_SQUARE:

651int Index::radiusSearch(InputArray _query, OutputArray _indices,
652 OutputArray _dists, double radius, int maxResults,
653 const SearchParams& params)
654{
655 CV_INSTRUMENT_REGION();
656
657 Mat query = _query.getMat(), indices, dists;
658 int dtype = (distType == FLANN_DIST_HAMMING)
659 || (distType == FLANN_DIST_DNAMMING) ? CV_32S : CV_32F;
660 CV_Assert( maxResults > 0 );
661 createIndicesDists( _indices, _dists, indices, dists, query.rows, maxResults, INT_MAX, dtype );
662
663 if( algo == FLANN_INDEX_LSH )
664 CV_Error( Error::StsNotImplemented, "LSH index does not support radiusSearch operation" );
665
666 int rsearch_ret;
667 switch( distType )
668 {
669 case FLANN_DIST_HAMMING:
670 rsearch_ret = runRadiusSearch< HammingDistance >(index, query, indices, dists, radius, params);
671 break;
672 case FLANN_DIST_L2:
673 rsearch_ret = runRadiusSearch< ::cvflann::L2<float> >(index, query, indices, dists, radius, params);
674 break;
675 case FLANN_DIST_L1:
676 rsearch_ret = runRadiusSearch< ::cvflann::L1<float> >(index, query, indices, dists, radius, params);
677 break;
678#if MINIFLANN_SUPPORT_EXOTIC_DISTANCE_TYPES
679 case FLANN_DIST_DNAMMING:
680 rsearch_ret = runRadiusSearch< DNAmmingDistance >(index, query, indices, dists, radius, params);
681 break;
682 case FLANN_DIST_MAX:
683 rsearch_ret = runRadiusSearch< ::cvflann::MaxDistance<float> >(index, query, indices, dists, radius, params);
684 break;
685 case FLANN_DIST_HIST_INTERSECT:
686 rsearch_ret = runRadiusSearch< ::cvflann::HistIntersectionDistance<float> >(index, query, indices, dists, radius, params);
687 break;
688 case FLANN_DIST_HELLINGER:
689 rsearch_ret = runRadiusSearch< ::cvflann::HellingerDistance<float> >(index, query, indices, dists, radius, params);
690 break;
691 case FLANN_DIST_CHI_SQUARE:
692 rsearch_ret = runRadiusSearch< ::cvflann::ChiSquareDistance<float> >(index, query, indices, dists, radius, params);
693 break;
694 case FLANN_DIST_KL:
695 rsearch_ret = runRadiusSearch< ::cvflann::KL_Divergence<float> >(index, query, indices, dists, radius, params);
696 break;
697#endif
698 default:
699 CV_Error(Error::StsBadArg, "Unknown/unsupported distance type");
700 return -1;
701 }
702 if (rsearch_ret > maxResults)
703 rsearch_ret = maxResults;
704 if (rsearch_ret < maxResults) {
705 if (_indices.needed())

821bool Index::load_(const String& filename)
822{
823 Mat data = features_clone;
824 bool ok = true;
825
826 FILE* fin = fopen(filename.c_str(), "rb");
827 if (fin == NULL) {
828 return false;
829 }
830 FILEScopeGuard fscgd(fin);
831
832 ::cvflann::IndexHeader header = ::cvflann::load_header(fin);
833
834 algo = header.index_type;
835 featureType = header.data_type == FLANN_UINT8 ? CV_8U :
836 header.data_type == FLANN_INT8 ? CV_8S :
837 header.data_type == FLANN_UINT16 ? CV_16U :
838 header.data_type == FLANN_INT16 ? CV_16S :
839 header.data_type == FLANN_INT32 ? CV_32S :
840 header.data_type == FLANN_FLOAT32 ? CV_32F :
841 header.data_type == FLANN_FLOAT64 ? CV_64F : -1;
842
843 if( (int)header.rows != data.rows || (int)header.cols != data.cols ||
844 featureType != data.type() )
845 {
846 fprintf(stderr, "Reading FLANN index error: the saved data size (%d, %d) or type (%d) is different from the passed one (%d, %d), %d\n",
847 (int)header.rows, (int)header.cols, featureType, data.rows, data.cols, data.type());
848 return false;
849 }
850
851 int idistType = 0;
852 ::cvflann::load_value(fin, idistType);
853 distType = (flann_distance_t)idistType;
854
855 if( !((distType == FLANN_DIST_HAMMING && featureType == CV_8U) ||
856 (distType == FLANN_DIST_DNAMMING && featureType == CV_8U) ||
857 (distType != FLANN_DIST_HAMMING && featureType == CV_32F)) )
858 {
859 fprintf(stderr, "Reading FLANN index error: unsupported feature type %d for the index type %d\n", featureType, algo);
860 return false;
861 }
862
863 switch( distType )
864 {
865 case FLANN_DIST_HAMMING:
866 loadIndex< HammingDistance >(this, index, data, fin);
867 break;
868 case FLANN_DIST_L2:
869 loadIndex< ::cvflann::L2<float> >(this, index, data, fin);
870 break;
871 case FLANN_DIST_L1:
872 loadIndex< ::cvflann::L1<float> >(this, index, data, fin);
873 break;
874#if MINIFLANN_SUPPORT_EXOTIC_DISTANCE_TYPES
875 case FLANN_DIST_DNAMMING:
876 loadIndex< DNAmmingDistance >(this, index, data, fin);
877 break;
878 case FLANN_DIST_MAX:
879 loadIndex< ::cvflann::MaxDistance<float> >(this, index, data, fin);
880 break;
881 case FLANN_DIST_HIST_INTERSECT:
882 loadIndex< ::cvflann::HistIntersectionDistance<float> >(this, index, data, fin);
883 break;
884 case FLANN_DIST_HELLINGER:
885 loadIndex< ::cvflann::HellingerDistance<float> >(this, index, data, fin);
886 break;
887 case FLANN_DIST_CHI_SQUARE:
888 loadIndex< ::cvflann::ChiSquareDistance<float> >(this, index, data, fin);
889 break;
890 case FLANN_DIST_KL:
891 loadIndex< ::cvflann::KL_Divergence<float> >(this, index, data, fin);
892 break;
893#endif
894 default:
895 fprintf(stderr, "Reading FLANN index error: unsupported distance type %d\n", distType);
896 ok = false;
897 }

900}
