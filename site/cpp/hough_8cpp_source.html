61struct hough_cmp_gt
62{
63 hough_cmp_gt(const int* _aux) : aux(_aux) {}
64 inline bool operator()(int l1, int l2) const
65 {
66 return aux[l1] > aux[l2] || (aux[l1] == aux[l2] && l1 < l2);
67 }
68 const int* aux;

71static inline int
72computeNumangle( double min_theta, double max_theta, double theta_step )
73{
74 int numangle = cvFloor((max_theta - min_theta) / theta_step) + 1;
75 // If the distance between the first angle and the last angle is
76 // approximately equal to pi, then the last angle will be removed
77 // in order to prevent a line to be detected twice.
78 if ( numangle > 1 && fabs(CV_PI - (numangle-1)*theta_step) < theta_step/2 )
79 --numangle;
80 return numangle;
81}

83static void
84createTrigTable( int numangle, double min_theta, double theta_step,
85 float irho, float *tabSin, float *tabCos )
86{
87 float ang = static_cast<float>(min_theta);
88 for(int n = 0; n < numangle; ang += (float)theta_step, n++ )
89 {
90 tabSin[n] = (float)(sin((double)ang) * irho);
91 tabCos[n] = (float)(cos((double)ang) * irho);
92 }
93}

95static void
96findLocalMaximums( int numrho, int numangle, int threshold,
97 const int *accum, std::vector<int>& sort_buf )
98{
99 for(int r = 0; r < numrho; r++ )
100 for(int n = 0; n < numangle; n++ )
101 {
102 int base = (n+1) * (numrho+2) + r+1;
103 if( accum[base] > threshold &&
104 accum[base] > accum[base - 1] && accum[base] >= accum[base + 1] &&
105 accum[base] > accum[base - numrho - 2] && accum[base] >= accum[base + numrho + 2] )
106 sort_buf.push_back(base);
107 }
108}

111Here image is an input raster;
112step is it's step; size characterizes it's ROI;
113rho and theta are discretization steps (in pixels and radians correspondingly).
114threshold is the minimum number of pixels in the feature for it

119static void
120HoughLinesStandard( InputArray src, OutputArray lines, int type,
121 float rho, float theta,
122 int threshold, int linesMax,
123 double min_theta, double max_theta, bool use_edgeval = false )
124{
125 CV_CheckType(type, type == CV_32FC2 || type == CV_32FC3, "Internal error");
126
127 Mat img = src.getMat();
128
129 int i, j;
130 float irho = 1 / rho;
131
132 CV_Assert( img.type() == CV_8UC1 );
133 CV_Assert( linesMax > 0 );
134
135 const uchar* image = img.ptr();
136 int step = (int)img.step;
137 int width = img.cols;
138 int height = img.rows;
139
140 int max_rho = width + height;
141 int min_rho = -max_rho;
142
143 CV_CheckGE(max_theta, min_theta, "max_theta must be greater than min_theta");
144
145 int numangle = computeNumangle(min_theta, max_theta, theta);
146 int numrho = cvRound(((max_rho - min_rho) + 1) / rho);
147
148#if defined HAVE_IPP && IPP_VERSION_X100 >= 810 && !IPP_DISABLE_HOUGH
149 if (type == CV_32FC2 && CV_IPP_CHECK_COND)
150 {
151 IppiSize srcSize = { width, height };
152 IppPointPolar delta = { rho, theta };
153 IppPointPolar dstRoi[2] = {{(Ipp32f) min_rho, (Ipp32f) min_theta},{(Ipp32f) max_rho, (Ipp32f) max_theta}};
154 int bufferSize;
155 int nz = countNonZero(img);
156 int ipp_linesMax = std::min(linesMax, nz*numangle/threshold);
157 int linesCount = 0;
158 std::vector<Vec2f> _lines(ipp_linesMax);
159 IppStatus ok = ippiHoughLineGetSize_8u_C1R(srcSize, delta, ipp_linesMax, &bufferSize);
160 Ipp8u* buffer = ippsMalloc_8u_L(bufferSize);
161 if (ok >= 0) {ok = CV_INSTRUMENT_FUN_IPP(ippiHoughLine_Region_8u32f_C1R, image, step, srcSize, (IppPointPolar*) &_lines[0], dstRoi, ipp_linesMax, &linesCount, delta, threshold, buffer);};
162 ippsFree(buffer);
163 if (ok >= 0)
164 {
165 lines.create(linesCount, 1, CV_32FC2);
166 Mat(linesCount, 1, CV_32FC2, &_lines[0]).copyTo(lines);
167 CV_IMPL_ADD(CV_IMPL_IPP);
168 return;
169 }
170 setIppErrorStatus();
171 }
172#endif
173
174
175 Mat _accum = Mat::zeros( (numangle+2), (numrho+2), CV_32SC1 );
176 std::vector<int> _sort_buf;
177 AutoBuffer<float> _tabSin(numangle);
178 AutoBuffer<float> _tabCos(numangle);
179 int *accum = _accum.ptr<int>();
180 float *tabSin = _tabSin.data(), *tabCos = _tabCos.data();
181
182 // create sin and cos table
183 createTrigTable( numangle, min_theta, theta,
184 irho, tabSin, tabCos);
185
186 // stage 1. fill accumulator
187 // Use the serial implementation for small numangle values to avoid parallel overhead.
188 constexpr int kParallelAngleThreshold = 100;
189 if (numangle < kParallelAngleThreshold) {
190 if (use_edgeval) {
191 for( i = 0; i < height; i++ )
192 for( j = 0; j < width; j++ )
193 {
194 if( image[i * step + j] != 0 )
195 for(int n = 0; n < numangle; n++ )
196 {
197 int r = cvRound( j * tabCos[n] + i * tabSin[n] );
198 r += (numrho - 1) / 2;
199 accum[(n + 1) * (numrho + 2) + r + 1] += image[i * step + j];
200 }
201 }
202 } else {
203 for( i = 0; i < height; i++ )
204 for( j = 0; j < width; j++ )
205 {
206 if( image[i * step + j] != 0 )
207 for(int n = 0; n < numangle; n++ )
208 {
209 int r = cvRound( j * tabCos[n] + i * tabSin[n] );
210 r += (numrho - 1) / 2;
211 accum[(n + 1) * (numrho + 2) + r + 1]++;
212 }
213 }
214 }
215 } else {
216 // Extract the coordinates of all edge points
217 std::vector<int> x_coords, y_coords, edge_vals;
218 size_t estimated_edges = (size_t)width * (size_t)height / 10;
219 x_coords.reserve(estimated_edges);
220 y_coords.reserve(estimated_edges);
221 if (use_edgeval) {
222 edge_vals.reserve(estimated_edges);
223 }
224
225 for (int y = 0; y < height; y++) {
226 const uchar* row_ptr = image + y * step;
227 for (int x = 0; x < width; x++) {
228 int val = row_ptr[x];
229 if (val != 0) {
230 x_coords.push_back(x);
231 y_coords.push_back(y);
232 if (use_edgeval) edge_vals.push_back(val);
233 }
234 }
235 }
236 int num_edges = (int)x_coords.size();
237
238 // Perform multi-threaded segmentation according to the numangle
239 // Since accum is divided into blocks according to angles, the accum areas written by different threads will not overlap
240 auto process_hough_by_angle = [&](const cv::Range& range) {
241 for (int n = range.start; n < range.end; n++) {
242 float cos_n = tabCos[n];
243 float sin_n = tabSin[n];
244
245 int* accum_n = accum + (n + 1) * (numrho + 2) + 1 + (numrho - 1) / 2;
246
247 if (use_edgeval) {
248 for (int k = 0; k < num_edges; k++) {
249 int r = cvRound(x_coords[k] * cos_n + y_coords[k] * sin_n);
250 accum_n[r] += edge_vals[k];
251 }
252 } else {
253 for (int k = 0; k < num_edges; k++) {
254 int r = cvRound(x_coords[k] * cos_n + y_coords[k] * sin_n);
255 accum_n[r]++;
256 }
257 }
258 }
259 };
260
261 cv::parallel_for_(cv::Range(0, numangle), process_hough_by_angle);
262 }
263
264 // stage 2. find local maximums
265 findLocalMaximums( numrho, numangle, threshold, accum, _sort_buf );
266
267 // stage 3. sort the detected lines by accumulator value
268 std::sort(_sort_buf.begin(), _sort_buf.end(), hough_cmp_gt(accum));
269
270 // stage 4. store the first min(total,linesMax) lines to the output buffer
271 linesMax = std::min(linesMax, (int)_sort_buf.size());
272 double scale = 1./(numrho+2);
273
274 lines.create(linesMax, 1, type);
275 Mat _lines = lines.getMat();
276 for( i = 0; i < linesMax; i++ )
277 {
278 LinePolar line;
279 int idx = _sort_buf[i];
280 int n = cvFloor(idx*scale) - 1;
281 int r = idx - (n+1)*(numrho+2) - 1;
282 line.rho = (r - (numrho - 1)/2) * rho;
283 line.angle = static_cast<float>(min_theta) + n * theta;
284 if (type == CV_32FC2)
285 {
286 _lines.at<Vec2f>(i) = Vec2f(line.rho, line.angle);
287 }
288 else
289 {
290 CV_DbgAssert(type == CV_32FC3);
291 _lines.at<Vec3f>(i) = Vec3f(line.rho, line.angle, (float)accum[idx]);
292 }
293 }
294}

310static void
311HoughLinesSDiv( InputArray image, OutputArray lines, int type,
312 float rho, float theta, int threshold,
313 int srn, int stn, int linesMax,
314 double min_theta, double max_theta )
315{
316 CV_CheckType(type, type == CV_32FC2 || type == CV_32FC3, "Internal error");
317
318 #define _POINT(row, column)\
319 (image_src[(row)*step+(column)])
320
321 Mat img = image.getMat();
322 int index, i;
323 int ri, ti, ti1, ti0;
324 int row, col;
325 float r, t; /* Current rho and theta */
326 float rv; /* Some temporary rho value */
327
328 int fn = 0;
329 float xc, yc;
330
331 const float d2r = (float)(CV_PI / 180);
332 int sfn = srn * stn;
333 int fi;
334 int count;
335 int cmax = 0;
336
337 std::vector<hough_index> lst;
338
339 CV_Assert( img.type() == CV_8UC1 );
340 CV_Assert( linesMax > 0 );
341
342 threshold = MIN( threshold, 255 );
343
344 const uchar* image_src = img.ptr();
345 int step = (int)img.step;
346 int w = img.cols;
347 int h = img.rows;
348
349 float irho = 1 / rho;
350 float itheta = 1 / theta;
351 float srho = rho / srn;
352 float stheta = theta / stn;
353 float isrho = 1 / srho;
354 float istheta = 1 / stheta;
355
356 int rn = cvFloor( std::sqrt( (double)w * w + (double)h * h ) * irho );
357 int tn = cvFloor( 2 * CV_PI * itheta );
358
359 lst.push_back(hough_index(threshold, -1.f, 0.f));
360
361 // Precalculate sin table
362 AutoBuffer<float> _sinTable( 5 * tn * stn );
363 float* sinTable = &_sinTable[0];
364
365 for( index = 0; index < 5 * tn * stn; index++ )
366 sinTable[index] = (float)cos( stheta * index * 0.2f );
367
368 AutoBuffer<uchar> _caccum(rn * tn, (uchar)0);
369 uchar* caccum = &_caccum[0];
370
371 // Counting all feature pixels
372 for( row = 0; row < h; row++ )
373 for( col = 0; col < w; col++ )
374 fn += _POINT( row, col ) != 0;
375
376 AutoBuffer<int> _x(fn), _y(fn);
377 int* x = &_x[0], *y = &_y[0];
378
379 // Full Hough Transform (it's accumulator update part)
380 fi = 0;
381 for( row = 0; row < h; row++ )
382 {
383 for( col = 0; col < w; col++ )
384 {
385 if( _POINT( row, col ))
386 {
387 int halftn;
388 float r0;
389 float scale_factor;
390 int iprev = -1;
391 float phi, phi1;
392 float theta_it; // Value of theta for iterating
393
394 // Remember the feature point
395 x[fi] = col;
396 y[fi] = row;
397 fi++;
398
399 yc = (float) row + 0.5f;
400 xc = (float) col + 0.5f;
401
402 /* Update the accumulator */
403 t = (float) fabs( fastAtan2( yc, xc ) * d2r );
404 r = (float) std::sqrt( (double)xc * xc + (double)yc * yc );
405 r0 = r * irho;
406 ti0 = cvFloor( (t + CV_PI*0.5) * itheta );
407
408 caccum[ti0]++;
409
410 theta_it = rho / r;
411 theta_it = theta_it < theta ? theta_it : theta;
412 scale_factor = theta_it * itheta;
413 halftn = cvFloor( CV_PI / theta_it );
414 for( ti1 = 1, phi = theta_it - (float)(CV_PI*0.5), phi1 = (theta_it + t) * itheta;
415 ti1 < halftn; ti1++, phi += theta_it, phi1 += scale_factor )
416 {
417 rv = r0 * std::cos( phi );
418 i = (int)rv * tn;
419 i += cvFloor( phi1 );
420 CV_Assert( i >= 0 );
421 CV_Assert( i < rn * tn );
422 caccum[i] = (uchar) (caccum[i] + ((i ^ iprev) != 0));
423 iprev = i;
424 if( cmax < caccum[i] )
425 cmax = caccum[i];
426 }
427 }
428 }
429 }
430
431 // Starting additional analysis
432 count = 0;
433 for( ri = 0; ri < rn; ri++ )
434 {
435 for( ti = 0; ti < tn; ti++ )
436 {
437 if( caccum[ri * tn + ti] > threshold )
438 count++;
439 }
440 }
441
442 if( count * 100 > rn * tn )
443 {
444 HoughLinesStandard( image, lines, type, rho, theta, threshold, linesMax, min_theta, max_theta );
445 return;
446 }
447
448 AutoBuffer<uchar> _buffer(srn * stn + 2);
449 uchar* buffer = &_buffer[0];
450 uchar* mcaccum = buffer + 1;
451
452 count = 0;
453 for( ri = 0; ri < rn; ri++ )
454 {
455 for( ti = 0; ti < tn; ti++ )
456 {
457 if( caccum[ri * tn + ti] > threshold )
458 {
459 count++;
460 memset( mcaccum, 0, sfn * sizeof( uchar ));
461
462 for( index = 0; index < fn; index++ )
463 {
464 int ti2;
465 float r0;
466
467 yc = (float) y[index] + 0.5f;
468 xc = (float) x[index] + 0.5f;
469
470 // Update the accumulator
471 t = (float) fabs( fastAtan2( yc, xc ) * d2r );
472 r = (float) std::sqrt( (double)xc * xc + (double)yc * yc ) * isrho;
473 ti0 = cvFloor( (t + CV_PI * 0.5) * istheta );
474 ti2 = (ti * stn - ti0) * 5;
475 r0 = (float) ri *srn;
476
477 for( ti1 = 0; ti1 < stn; ti1++, ti2 += 5 )
478 {
479 rv = r * sinTable[(int) (std::abs( ti2 ))] - r0;
480 i = cvFloor( rv ) * stn + ti1;
481
482 i = CV_IMAX( i, -1 );
483 i = CV_IMIN( i, sfn );
484 mcaccum[i]++;
485 CV_Assert( i >= -1 );
486 CV_Assert( i <= sfn );
487 }
488 }
489
490 // Find peaks in maccum...
491 for( index = 0; index < sfn; index++ )
492 {
493 int pos = (int)(lst.size() - 1);
494 if( pos < 0 || lst[pos].value < mcaccum[index] )
495 {
496 hough_index vi(mcaccum[index],
497 index / stn * srho + ri * rho,
498 index % stn * stheta + ti * theta - (float)(CV_PI*0.5));
499 lst.push_back(vi);
500 for( ; pos >= 0; pos-- )
501 {
502 if( lst[pos].value > vi.value )
503 break;
504 lst[pos+1] = lst[pos];
505 }
506 lst[pos+1] = vi;
507 if( (int)lst.size() > linesMax )
508 lst.pop_back();
509 }
510 }
511 }
512 }
513 }
514
515 int pos = (int)(lst.size() - 1);
516 if( pos >= 0 && lst[pos].rho < 0 )
517 lst.pop_back();
518
519 lines.create((int)lst.size(), 1, type);
520 Mat _lines = lines.getMat();
521 for( size_t idx = 0; idx < lst.size(); idx++ )
522 {
523 if (type == CV_32FC2)
524 {
525 _lines.at<Vec2f>((int)idx) = Vec2f(lst[idx].rho, lst[idx].theta);
526 }
527 else
528 {
529 CV_DbgAssert(type == CV_32FC3);
530 _lines.at<Vec3f>((int)idx) = Vec3f(lst[idx].rho, lst[idx].theta, (float)lst[idx].value);
531 }
532 }
533}

540static void
541HoughLinesProbabilistic( Mat& image,
542 float rho, float theta, int threshold,
543 int lineLength, int lineGap,
544 std::vector<Vec4i>& lines, int linesMax )
545{
546 Point pt;
547 float irho = 1 / rho;
548 RNG rng((uint64)-1);
549
550 CV_Assert( image.type() == CV_8UC1 );
551
552 int width = image.cols;
553 int height = image.rows;
554
555 int numangle = computeNumangle(0.0, CV_PI, theta);
556 int numrho = cvRound(((width + height) * 2 + 1) / rho);
557
558#if defined HAVE_IPP && IPP_VERSION_X100 >= 810 && !IPP_DISABLE_HOUGH
559 CV_IPP_CHECK()
560 {
561 IppiSize srcSize = { width, height };
562 IppPointPolar delta = { rho, theta };
563 IppiHoughProbSpec* pSpec;
564 int bufferSize, specSize;
565 int ipp_linesMax = std::min(linesMax, numangle*numrho);
566 int linesCount = 0;
567 lines.resize(ipp_linesMax);
568 IppStatus ok = ippiHoughProbLineGetSize_8u_C1R(srcSize, delta, &specSize, &bufferSize);
569 Ipp8u* buffer = ippsMalloc_8u_L(bufferSize);
570 pSpec = (IppiHoughProbSpec*) ippsMalloc_8u_L(specSize);
571 if (ok >= 0) ok = ippiHoughProbLineInit_8u32f_C1R(srcSize, delta, ippAlgHintNone, pSpec);
572 if (ok >= 0) {ok = CV_INSTRUMENT_FUN_IPP(ippiHoughProbLine_8u32f_C1R, image.data, (int)image.step, srcSize, threshold, lineLength, lineGap, (IppiPoint*) &lines[0], ipp_linesMax, &linesCount, buffer, pSpec);};
573
574 ippsFree(pSpec);
575 ippsFree(buffer);
576 if (ok >= 0)
577 {
578 lines.resize(linesCount);
579 CV_IMPL_ADD(CV_IMPL_IPP);
580 return;
581 }
582 lines.clear();
583 setIppErrorStatus();
584 }
585#endif
586
587 Mat accum = Mat::zeros( numangle, numrho, CV_32SC1 );
588 Mat mask( height, width, CV_8UC1 );
589 AutoBuffer<float> trigtab(numangle*2);
590
591 for( int n = 0; n < numangle; n++ )
592 {
593 trigtab[n*2] = (float)(cos((double)n*theta) * irho);
594 trigtab[n*2+1] = (float)(sin((double)n*theta) * irho);
595 }
596 const float* ttab = &trigtab[0];
597 uchar* mdata0 = mask.ptr();
598 std::vector<Point> nzloc;
599
600 // stage 1. collect non-zero image points
601 for( pt.y = 0; pt.y < height; pt.y++ )
602 {
603 const uchar* data = image.ptr(pt.y);
604 uchar* mdata = mask.ptr(pt.y);
605 for( pt.x = 0; pt.x < width; pt.x++ )
606 {
607 if( data[pt.x] )
608 {
609 mdata[pt.x] = (uchar)1;
610 nzloc.push_back(pt);
611 }
612 else
613 mdata[pt.x] = 0;
614 }
615 }
616
617 int count = (int)nzloc.size();
618
619 // stage 2. process all the points in random order
620 for( ; count > 0; count-- )
621 {
622 // choose random point out of the remaining ones
623 int idx = rng.uniform(0, count);
624 int max_val = threshold-1, max_n = 0;
625 Point point = nzloc[idx];
626 Point line_end[2];
627 float a, b;
628 int* adata = accum.ptr<int>();
629 int i = point.y, j = point.x, k, xflag;
630 int64_t x0, y0, dx0, dy0;
631 int good_line;
632 const int shift = 16;
633
634 // "remove" it by overriding it with the last element
635 nzloc[idx] = nzloc[count-1];
636
637 // check if it has been excluded already (i.e. belongs to some other line)
638 if( !mdata0[i*width + j] )
639 continue;
640
641 // update accumulator, find the most probable line
642 for( int n = 0; n < numangle; n++, adata += numrho )
643 {
644 int r = cvRound( j * ttab[n*2] + i * ttab[n*2+1] );
645 r += (numrho - 1) / 2;
646 int val = ++adata[r];
647 if( max_val < val )
648 {
649 max_val = val;
650 max_n = n;
651 }
652 }
653
654 // if it is too "weak" candidate, continue with another point
655 if( max_val < threshold )
656 continue;
657
658 // from the current point walk in each direction
659 // along the found line and extract the line segment
660 a = -ttab[max_n*2+1];
661 b = ttab[max_n*2];
662 x0 = j;
663 y0 = i;
664 if( fabs(a) > fabs(b) )
665 {
666 xflag = 1;
667 dx0 = a > 0 ? 1 : -1;
668 dy0 = cvRound( b*(1 << shift)/fabs(a) );
669 y0 = (y0 << shift) + (1 << (shift-1));
670 }
671 else
672 {
673 xflag = 0;
674 dy0 = b > 0 ? 1 : -1;
675 dx0 = cvRound( a*(1 << shift)/fabs(b) );
676 x0 = (x0 << shift) + (1 << (shift-1));
677 }
678
679 for( k = 0; k < 2; k++ )
680 {
681 int gap = 0;
682 int64_t x = x0, y = y0, dx = dx0, dy = dy0 ;
683 if( k > 0 )
684 dx = -dx, dy = -dy;
685
686 // walk along the line using fixed-point arithmetic,
687 // stop at the image border or in case of too big gap
688 for( ;; x += dx, y += dy )
689 {
690 uchar* mdata;
691 int64_t i1, j1;
692
693 if( xflag )
694 {
695 j1 = x;
696 i1 = y >> shift;
697 }
698 else
699 {
700 j1 = x >> shift;
701 i1 = y;
702 }
703
704 if( j1 < 0 || j1 >= width || i1 < 0 || i1 >= height )
705 break;
706
707 mdata = mdata0 + i1*width + j1;
708
709 // for each non-zero point:
710 // update line end,
711 // clear the mask element
712 // reset the gap
713 if( *mdata )
714 {
715 gap = 0;
716 line_end[k].y = static_cast<int>(i1);
717 line_end[k].x = static_cast<int>(j1);
718 }
719 else if( ++gap > lineGap )
720 break;
721 }
722 }
723
724 good_line = std::abs(line_end[1].x - line_end[0].x) >= lineLength ||
725 std::abs(line_end[1].y - line_end[0].y) >= lineLength;
726
727 for( k = 0; k < 2; k++ )
728 {
729 int64_t x = x0, y = y0, dx = dx0, dy = dy0;
730
731 if( k > 0 )
732 dx = -dx, dy = -dy;
733
734 // walk along the line using fixed-point arithmetic,
735 // stop at the image border or in case of too big gap
736 for( ;; x += dx, y += dy )
737 {
738 uchar* mdata;
739 int64_t i1, j1;
740
741 if( xflag )
742 {
743 j1 = x;
744 i1 = y >> shift;
745 }
746 else
747 {
748 j1 = x >> shift;
749 i1 = y;
750 }
751
752 mdata = mdata0 + i1*width + j1;
753
754 // for each non-zero point:
755 // update line end,
756 // clear the mask element
757 // reset the gap
758 if( *mdata )
759 {
760 if( good_line )
761 {
762 adata = accum.ptr<int>();
763 for( int n = 0; n < numangle; n++, adata += numrho )
764 {
765 int r = cvRound( j1 * ttab[n*2] + i1 * ttab[n*2+1] );
766 r += (numrho - 1) / 2;
767 adata[r]--;
768 }
769 }
770 *mdata = 0;
771 }
772
773 if( i1 == line_end[k].y && j1 == line_end[k].x )
774 break;
775 }
776 }
777
778 if( good_line )
779 {
780 Vec4i lr(line_end[0].x, line_end[0].y, line_end[1].x, line_end[1].y);
781 lines.push_back(lr);
782 if( (int)lines.size() >= linesMax )
783 return;
784 }
785 }
786}

1012void HoughLinesPointSet( InputArray _point, OutputArray _lines, int lines_max, int threshold,
1013 double min_rho, double max_rho, double rho_step,
1014 double min_theta, double max_theta, double theta_step )
1015{
1016 std::vector<Vec3d> lines;
1017 std::vector<Point2f> point;
1018 _point.copyTo(point);
1019
1020 CV_Assert( _point.type() == CV_32FC2 || _point.type() == CV_32SC2 );
1021 if( lines_max <= 0 ) {
1022 CV_Error( Error::StsBadArg, "lines_max must be greater than 0" );
1023 }
1024 if( threshold < 0) {
1025 CV_Error( Error::StsBadArg, "threshold must be greater than 0" );
1026 }
1027 if( ((max_rho - min_rho) <= 0) || ((max_theta - min_theta) <= 0) ) {
1028 CV_Error( Error::StsBadArg, "max must be greater than min" );
1029 }
1030 if( ((rho_step <= 0)) || ((theta_step <= 0)) ) {
1031 CV_Error( Error::StsBadArg, "step must be greater than 0" );
1032 }
1033
1034 int i;
1035 float irho = 1 / (float)rho_step;
1036 float irho_min = ((float)min_rho * irho);
1037 int numangle = computeNumangle(min_theta, max_theta, theta_step);
1038 int numrho = cvRound((max_rho - min_rho + 1) / rho_step);
1039
1040 Mat _accum = Mat::zeros( (numangle+2), (numrho+2), CV_32SC1 );
1041 std::vector<int> _sort_buf;
1042 AutoBuffer<float> _tabSin(numangle);
1043 AutoBuffer<float> _tabCos(numangle);
1044 int *accum = _accum.ptr<int>();
1045 float *tabSin = _tabSin.data(), *tabCos = _tabCos.data();
1046
1047 // create sin and cos table
1048 createTrigTable( numangle, min_theta, theta_step,
1049 irho, tabSin, tabCos );
1050
1051 // stage 1. fill accumulator
1052 for( i = 0; i < (int)point.size(); i++ )
1053 for(int n = 0; n < numangle; n++ )
1054 {
1055 int r = cvRound( point.at(i).x * tabCos[n] + point.at(i).y * tabSin[n] - irho_min);
1056 if ( r >= 0 && r <= numrho) {
1057 accum[(n+1) * (numrho+2) + r+1]++;
1058 }
1059 }
1060
1061 // stage 2. find local maximums
1062 findLocalMaximums( numrho, numangle, threshold, accum, _sort_buf );
1063
1064 // stage 3. sort the detected lines by accumulator value
1065 std::sort(_sort_buf.begin(), _sort_buf.end(), hough_cmp_gt(accum));
1066
1067 // stage 4. store the first min(total,linesMax) lines to the output buffer
1068 lines_max = std::min(lines_max, (int)_sort_buf.size());
1069 double scale = 1./(numrho+2);
1070 for( i = 0; i < lines_max; i++ )
1071 {
1072 LinePolar line;
1073 int idx = _sort_buf[i];
1074 int n = cvFloor(idx*scale) - 1;
1075 int r = idx - (n+1)*(numrho+2) - 1;
1076 line.rho = static_cast<float>(min_rho) + r * (float)rho_step;
1077 line.angle = static_cast<float>(min_theta) + n * (float)theta_step;
1078 lines.emplace_back((double)accum[idx], (double)line.rho, (double)line.angle);
1079 }
1080
1081 Mat(lines).copyTo(_lines);
1082}

1098{
1099 // Compare everything so the order is completely deterministic
1100 // Larger accum first
1101 if (left.accum > right.accum)
1102 return true;
1103 else if (left.accum < right.accum)
1104 return false;
1105 // Larger radius first
1106 else if (left.c[2] > right.c[2])
1107 return true;
1108 else if (left.c[2] < right.c[2])
1109 return false;
1110 // Smaller X
1111 else if (left.c[0] < right.c[0])
1112 return true;
1113 else if (left.c[0] > right.c[0])
1114 return false;
1115 // Smaller Y
1116 else if (left.c[1] < right.c[1])
1117 return true;
1118 else if (left.c[1] > right.c[1])
1119 return false;
1120 // Identical - neither object is less than the other
1121 else
1122 return false;
1123}

1186class HoughCirclesAccumInvoker : public ParallelLoopBody
1187{
1188public:
1189 HoughCirclesAccumInvoker(const Mat &_edges, const Mat &_dx, const Mat &_dy, int _minRadius, int _maxRadius, float _idp,
1190 std::vector<Mat>& _accumVec, NZPointSet& _nz, Mutex& _mtx) :
1191 edges(_edges), dx(_dx), dy(_dy), minRadius(_minRadius), maxRadius(_maxRadius), idp(_idp),
1192 accumVec(_accumVec), nz(_nz), mutex(_mtx)
1193 {
1194 acols = cvCeil(edges.cols * idp), arows = cvCeil(edges.rows * idp);
1195 astep = acols + 2;
1196 }
1197
1198 ~HoughCirclesAccumInvoker() { }
1199
1200 void operator()(const Range &boundaries) const CV_OVERRIDE
1201 {
1202 Mat accumLocal = Mat(arows + 2, acols + 2, CV_32SC1, Scalar::all(0));
1203 int *adataLocal = accumLocal.ptr<int>();
1204 NZPointSet nzLocal(nz.positions.rows, nz.positions.cols);
1205 int startRow = boundaries.start;
1206 int endRow = boundaries.end;
1207 int numCols = edges.cols;
1208
1209 if(edges.isContinuous() && dx.isContinuous() && dy.isContinuous())
1210 {
1211 numCols *= (boundaries.end - boundaries.start);
1212 endRow = boundaries.start + 1;
1213 }
1214
1215 // Accumulate circle evidence for each edge pixel
1216 for(int y = startRow; y < endRow; ++y )
1217 {
1218 const uchar* edgeData = edges.ptr<const uchar>(y);
1219 const short* dxData = dx.ptr<const short>(y);
1220 const short* dyData = dy.ptr<const short>(y);
1221 int x = 0;
1222
1223 for(; x < numCols; ++x )
1224 {
1225#if (CV_SIMD || CV_SIMD_SCALABLE)
1226 {
1227 v_uint8 v_zero = vx_setzero_u8();
1228
1229 for(; x <= numCols - 2*VTraits<v_uint8>::vlanes(); x += 2*VTraits<v_uint8>::vlanes()) {
1230 v_uint8 v_edge1 = (v_ne(vx_load(edgeData + x), v_zero));
1231 v_uint8 v_edge2 = (v_ne(vx_load(edgeData + x + VTraits<v_uint8>::vlanes()), v_zero));
1232
1233 if(v_check_any(v_edge1))
1234 {
1235 x += v_scan_forward(v_edge1);
1236 goto _next_step;
1237 }
1238
1239 if(v_check_any(v_edge2))
1240 {
1241 x += VTraits<v_uint8>::vlanes() + v_scan_forward(v_edge2);
1242 goto _next_step;
1243 }
1244 }
1245 }
1246#endif
1247 for(; x < numCols && !edgeData[x]; ++x)
1248 ;
1249
1250 if(x == numCols)
1251 continue;
1252#if (CV_SIMD || CV_SIMD_SCALABLE)
1253_next_step:
1254#endif
1255 float vx, vy;
1256 int sx, sy, x0, y0, x1, y1;
1257
1258 vx = dxData[x];
1259 vy = dyData[x];
1260
1261 if(vx == 0 && vy == 0)
1262 continue;
1263
1264 float mag = std::sqrt(vx*vx+vy*vy);
1265
1266 if(mag < 1.0f)
1267 continue;
1268
1269 Point pt = Point(x % edges.cols, y + x / edges.cols);
1270 nzLocal.insert(pt);
1271
1272 sx = cvRound((vx * idp) * 1024 / mag);
1273 sy = cvRound((vy * idp) * 1024 / mag);
1274
1275 x0 = cvRound((pt.x * idp) * 1024);
1276 y0 = cvRound((pt.y * idp) * 1024);
1277
1278 // Step from min_radius to max_radius in both directions of the gradient
1279 for(int k1 = 0; k1 < 2; k1++ )
1280 {
1281 x1 = x0 + minRadius * sx;
1282 y1 = y0 + minRadius * sy;
1283
1284 for(int r = minRadius; r <= maxRadius; x1 += sx, y1 += sy, r++ )
1285 {
1286 int x2 = x1 >> 10, y2 = y1 >> 10;
1287 if( (unsigned)x2 >= (unsigned)acols ||
1288 (unsigned)y2 >= (unsigned)arows )
1289 break;
1290
1291 adataLocal[y2*astep + x2]++;
1292 }
1293
1294 sx = -sx; sy = -sy;
1295 }
1296 }
1297 }
1298
1299 { // TODO Try using TLSContainers
1300 AutoLock lock(mutex);
1301 accumVec.push_back(accumLocal);
1302 nz.insert(nzLocal);
1303 }
1304 }

1700template <typename CircleType>
1701static void HoughCirclesGradient(InputArray _image, OutputArray _circles,
1702 float dp, float minDist,
1703 int minRadius, int maxRadius, int cannyThreshold,
1704 int accThreshold, int maxCircles, int kernelSize, bool centersOnly)
1705{
1706 CV_Assert(kernelSize == -1 || kernelSize == 3 || kernelSize == 5 || kernelSize == 7);
1707
1708 dp = max(dp, 1.f);
1709 float idp = 1.f/dp;
1710
1711 Mat edges, dx, dy;
1712
1713 Sobel(_image, dx, CV_16S, 1, 0, kernelSize, 1, 0, BORDER_REPLICATE);
1714 Sobel(_image, dy, CV_16S, 0, 1, kernelSize, 1, 0, BORDER_REPLICATE);
1715 Canny(dx, dy, edges, std::max(1, cannyThreshold / 2), cannyThreshold, false);
1716
1717 Mutex mtx;
1718 int numThreads = std::max(1, getNumThreads());
1719 std::vector<Mat> accumVec;
1720 NZPointSet nz(_image.rows(), _image.cols());
1721 parallel_for_(Range(0, edges.rows),
1722 HoughCirclesAccumInvoker(edges, dx, dy, minRadius, maxRadius, idp, accumVec, nz, mtx),
1723 numThreads);
1724 int nzSz = cv::countNonZero(nz.positions);
1725 if(nzSz <= 0)
1726 return;
1727
1728 Mat accum = accumVec[0];
1729 for(size_t i = 1; i < accumVec.size(); i++)
1730 {
1731 accum += accumVec[i];
1732 }
1733 accumVec.clear();
1734
1735 std::vector<int> centers;
1736
1737 // 4 rows when multithreaded because there is a bit overhead
1738 // and on the other side there are some row ranges where centers are concentrated
1739 parallel_for_(Range(1, accum.rows - 1),
1740 HoughCirclesFindCentersInvoker(accum, centers, accThreshold, mtx),
1741 (numThreads > 1) ? ((accum.rows - 2) / 4) : 1);
1742
1743 int centerCnt = (int)centers.size();
1744 if(centerCnt == 0)
1745 return;
1746
1747 std::sort(centers.begin(), centers.end(), hough_cmp_gt(accum.ptr<int>()));
1748
1749 std::vector<CircleType> circles;
1750 circles.reserve(256);
1751 if (centersOnly)
1752 {
1753 // Just get the circle centers
1754 GetCircleCenters(centers, circles, accum.cols, minDist, dp);
1755 }
1756 else
1757 {

1760 {
1761 // Faster to use a list

1767 accThreshold, minRadius, maxRadius, dp, mtx),

1769 }
1770 else

1776 accThreshold, minRadius, maxRadius, dp, mtx),

1793 }
