281void checkTestTags()
282{
283 std::string skipTag;
284 const std::vector<std::string>& testTags = currentDirectTestTags;
285 {
286 const std::vector<std::string>& tags = getTestTagsRequiredList();
287 if (!tags.empty())
288 {
289 size_t found = 0;
290 for (size_t i = 0; i < tags.size(); ++i)
291 {
292 const std::string& tag = tags[i];
293 for (size_t j = 0; j < testTags.size(); ++j)
294 {
295 const std::string& testTag = testTags[i];
296 if (testTag == tag
297 || (testTag.size() >= tag.size()
298 && tag[tag.size() - 1] == '*'
299 && tag.substr(0, tag.size() - 1) == testTag.substr(0, tag.size() - 1)
300 )
301 )
302 {
303 found++;
304 break;
305 }
306 }
307 }
308 if (found != tags.size())
309 {
310 skipped_tests.push_back(::testing::UnitTest::GetInstance()->current_test_info());
311 throw details::SkipTestExceptionBase("Test tags don't pass required tags list (--test_tag parameter)", true);
312 }
313 }
314 }
315 for (size_t i = 0; i < testTags.size(); ++i)
316 {
317 const std::string& testTag = testTags[i];
318 if (isTestTagForced(testTag))
319 return;
320 }
321 std::string skip_message;
322 for (size_t i = 0; i < testTags.size(); ++i)
323 {
324 const std::string& testTag = testTags[i];
325 if (isTestTagSkipped(testTag, skipTag))
326 {
327 testTagIncreaseSkipCount(skipTag, skip_message.empty());
328 if (skip_message.empty()) skip_message = "Test with tag '" + testTag + "' is skipped ('" + skipTag + "' is in skip list)";
329 }
330 }
331 const std::vector<std::string>& testTagsImplied = currentImpliedTestTags;
332 for (size_t i = 0; i < testTagsImplied.size(); ++i)
333 {
334 const std::string& testTag = testTagsImplied[i];
335 if (isTestTagSkipped(testTag, skipTag))
336 {
337 testTagIncreaseSkipCount(skipTag, skip_message.empty());
338 if (skip_message.empty()) skip_message = "Test with tag '" + testTag + "' is skipped (implied '" + skipTag + "' is in skip list)";
339 }
340 }
341
342 if (!skip_message.empty())
343 {
344 skipped_tests.push_back(::testing::UnitTest::GetInstance()->current_test_info());
345 throw details::SkipTestExceptionBase(skip_message, true);
346 }

349static bool applyTestTagImpl(const std::string& tag, bool direct = false)
350{
351 CV_Assert(!tag.empty());
352 std::vector<std::string>& testTags = direct ? currentDirectTestTags : currentImpliedTestTags;
353 for (size_t i = 0; i < testTags.size(); ++i)
354 {
355 const std::string& testTag = testTags[i];
356 if (tag == testTag)
357 {
358 return false; // already exists, skip
359 }
360 }
361 testTags.push_back(tag);
362
363 // Tags implies logic
364 if (tag == CV_TEST_TAG_MEMORY_14GB)
365 applyTestTagImpl(CV_TEST_TAG_MEMORY_6GB);
366 if (tag == CV_TEST_TAG_MEMORY_6GB)
367 applyTestTagImpl(CV_TEST_TAG_MEMORY_2GB);
368 if (tag == CV_TEST_TAG_MEMORY_2GB)
369 applyTestTagImpl(CV_TEST_TAG_MEMORY_1GB);
370 if (tag == CV_TEST_TAG_MEMORY_1GB)
371 applyTestTagImpl(CV_TEST_TAG_MEMORY_512MB);
372 if (tag == CV_TEST_TAG_VERYLONG)
373 {
374 applyTestTagImpl(CV_TEST_TAG_DEBUG_VERYLONG);
375 applyTestTagImpl(CV_TEST_TAG_LONG);
376 }
377 else if (tag == CV_TEST_TAG_DEBUG_VERYLONG)
378 {
379 applyTestTagImpl(CV_TEST_TAG_DEBUG_LONG);
380 }
381 else if (tag == CV_TEST_TAG_LONG)
382 {
383 applyTestTagImpl(CV_TEST_TAG_DEBUG_LONG);
384 }
385
386 if (tag == CV_TEST_TAG_SIZE_4K)
387 applyTestTagImpl(CV_TEST_TAG_SIZE_FULLHD);
388 if (tag == CV_TEST_TAG_SIZE_FULLHD)
389 applyTestTagImpl(CV_TEST_TAG_SIZE_HD);
390
391 return true;
392}

409static std::vector<std::string> parseStringList(const std::string& s)
410{
411 std::vector<std::string> result;
412 size_t start_pos = 0;
413 while (start_pos != std::string::npos)
414 {
415 while (start_pos < s.size() && s[start_pos] == ' ')
416 start_pos++;
417 const size_t pos_ = s.find(',', start_pos);
418 size_t pos = (pos_ == std::string::npos ? s.size() : pos_);
419 while (pos > start_pos && s[pos - 1] == ' ')
420 pos--;
421 if (pos > start_pos)
422 {
423 const std::string one_piece(s, start_pos, pos - start_pos);
424 result.push_back(one_piece);
425 }
426 start_pos = (pos_ == std::string::npos ? pos_ : pos_ + 1);
427 }
428 return result;

447 if (tag == skipTags[i])

449 found = true;

452 }
