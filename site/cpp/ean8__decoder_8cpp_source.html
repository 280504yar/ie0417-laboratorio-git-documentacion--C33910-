14Result Ean8Decoder::decode(const vector<uchar> &data) const
15{
16 std::string result;
17 char decode_result[EAN8DIGIT_NUM + 1]{'\0'};
18 if (data.size() < EAN8BITS_NUM)
19 {
20 return Result("Wrong Size", Result::BARCODE_NONE);
21 }
22 pair<uint, uint> pattern;
23 if (!findStartGuardPatterns(data, pattern))
24 {
25 return Result("Begin Pattern Not Found", Result::BARCODE_NONE);
26 }
27 uint start = pattern.second;
28 Counter counter(vector<int>{0, 0, 0, 0});
29 size_t end = data.size();
30 for (int i = 0; i < 4 && start < end; ++i)
31 {
32 int bestMatch = decodeDigit(data, counter, start, get_A_or_C_Patterns());
33 if (bestMatch == -1)
34 {
35 return Result("Decode Error", Result::BARCODE_NONE);
36 }
37 decode_result[i] = static_cast<char>('0' + bestMatch % 10);
38 start = counter.sum + start;
39 }
40
41 Counter middle_counter(vector<int>(MIDDLE_PATTERN().size()));
42
43 if (!findGuardPatterns(data, start, true, MIDDLE_PATTERN(), middle_counter, pattern))
44 {
45 return Result("Middle Pattern Not Found", Result::BARCODE_NONE);
46 }
47
48 start = pattern.second;
49 for (int i = 0; i < 4 && start < end; ++i)
50 {
51 int bestMatch = decodeDigit(data, counter, start, get_A_or_C_Patterns());
52 if (bestMatch == -1)
53 {
54 return Result("Decode Error", Result::BARCODE_NONE);
55 }
56 decode_result[i + 4] = static_cast<char>('0' + bestMatch);
57 start = counter.sum + start;
58 }
59 Counter end_counter(vector<int>(BEGIN_PATTERN().size()));
60 if (!findGuardPatterns(data, start, false, BEGIN_PATTERN(), end_counter, pattern))
61 {
62 return Result("End Pattern Not Found", Result::BARCODE_NONE);
63 }
64 result = string(decode_result);
65 if (!isValid(result))
66 {
67 return Result("Wrong: " + result.append(string(EAN8DIGIT_NUM - result.size(), ' ')), Result::BARCODE_NONE);
68 }
69 return Result(result, Result::BARCODE_EAN_8);
70}
