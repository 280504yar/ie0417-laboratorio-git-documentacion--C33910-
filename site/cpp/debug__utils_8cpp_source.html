54void detail::LayerHandler::printMissing() const
55{
56 if (layers.empty())
57 {
58 return;
59 }
60
61 std::ostringstream ss;
62 ss << "DNN: Not supported types:\n";
63 for (const auto& type_names : layers)
64 {
65 const auto& type = type_names.first;
66 ss << "Type='" << type << "', affected nodes:\n[";
67 for (const auto& name : type_names.second)
68 {
69 ss << "'" << name << "', ";
70 }
71 ss.seekp(-2, std::ios_base::end);
72 ss << "]\n";
73 }
74 CV_LOG_ERROR(NULL, ss.str());
