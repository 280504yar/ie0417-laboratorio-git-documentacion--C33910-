17 CorrelationLayerImpl(const LayerParams& params)
18 {
19 setParamsFrom(params);
20 pad = params.get<int>("pad", 0);
21 CV_Assert_N(params.has("kernel_size"), params.has("max_displacement"));
22 max_displacement = params.get<int>("max_displacement");
23 kernel = params.get<int>("kernel_size");
24 if (kernel % 2 == 0)
25 CV_Error(Error::StsNotImplemented, "Odd kernel size required.");
26
27 stride_1 = params.get<int>("stride_1", 1);
28 stride_2 = params.get<int>("stride_2", 1);

31 virtual bool getMemoryShapes(const std::vector<MatShape> &inputs,
32 const int requiredOutputs,
33 std::vector<MatShape> &outputs,
34 std::vector<MatShape> &internals) const CV_OVERRIDE
35 {
36 CV_Assert_N(inputs.size() == 2, inputs[0].size() == 4, inputs[1].size() == 4);
37
38 int padded_height = inputs[0][2] + 2 * pad;
39 int padded_width = inputs[0][3] + 2 * pad;
40
41 int kernel_radius = (kernel - 1) / 2;
42 int border_size = max_displacement + kernel_radius;
43
44 int neighborhood_grid_radius = max_displacement / stride_2;
45 int neighborhood_grid_width = neighborhood_grid_radius * 2 + 1;
46
47 MatShape outShape;
48
49 int num = inputs[0][0];
50 outShape.push_back(num);
51
52 int out_c = neighborhood_grid_width * neighborhood_grid_width;
53 outShape.push_back(out_c);
54
55 int out_h = ceil(static_cast<float>(padded_height - border_size * 2) / stride_1);
56 int out_w = ceil(static_cast<float>(padded_width - border_size * 2) / stride_1);
57 CV_Assert_N(out_h >= 1, out_w >= 1);
58
59 outShape.push_back(out_h);
60 outShape.push_back(out_w);
61 outputs.assign(1, outShape);
62 return false;
63 }

78 void blobRearrangeKernel2(const Mat& input, Mat& output)
79 {
80 const int num = input.size[0];
81 const int channels = input.size[1];
82 const int height = input.size[2];
83 const int width = input.size[3];
84 const int area = height * width;
85 const int pad_area = (width + 2 * pad) * (height + 2 * pad);
86
87 const float* in = input.ptr<float>();
88 float* out = output.ptr<float>();
89 for (int n = 0; n < num; n++)
90 {
91 for (int ch = 0; ch < channels; ch++)
92 {
93 for (int xy = 0; xy < area; xy++)
94 {
95 float value = in[(n * channels + ch) * area + xy];
96 int xpad = (xy % width + pad);
97 int ypad = (xy / width + pad);
98 int xypad = ypad * (width + 2 * pad) + xpad;
99 out[(n * pad_area + xypad) * channels + ch] = value;
100 }
101 }
102 }

105 void correlationKernelSubtraction(const Mat& input0, const Mat& input1, Mat& output, int item)
106 {
107 const int inp_h = input0.size[1];
108 const int inp_w = input0.size[2];
109 const int inp_c = input0.size[3];
110
111 const int out_c = output.size[1];
112 const int out_h = output.size[2];
113 const int out_w = output.size[3];
114
115 int topcount = output.total(1);
116 int neighborhood_grid_radius = max_displacement / stride_2;
117 int neighborhood_grid_width = neighborhood_grid_radius * 2 + 1;
118
119 const float* inp0_data = input0.ptr<float>();
120 const float* inp1_data = input1.ptr<float>();
121 float* out_data = output.ptr<float>();
122 int sumelems = kernel * kernel * inp_c;
123 std::vector<float> patch_data(sumelems, 0);
124 for (int y = 0; y < out_h; y++)
125 {
126 for (int x = 0; x < out_w; x++)
127 {
128 int x1 = x * stride_1 + max_displacement;
129 int y1 = y * stride_1 + max_displacement;
130
131 for (int j = 0; j < kernel; j++)
132 {
133 for (int i = 0; i < kernel; i++)
134 {
135 int ji_off = ((j * kernel) + i) * inp_c;
136 for (int ch = 0; ch < inp_c; ch++)
137 {
138 int idx1 = ((item * inp_h + y1 + j) * inp_w + x1 + i) * inp_c + ch;
139 int idxPatchData = ji_off + ch;
140 patch_data[idxPatchData] = inp0_data[idx1];
141 }
142 }
143 }
144
145 for (int out_ch = 0; out_ch < out_c; out_ch++)
146 {
147 float sum = 0;
148 int s2o = (out_ch % neighborhood_grid_width - neighborhood_grid_radius) * stride_2;
149 int s2p = (out_ch / neighborhood_grid_width - neighborhood_grid_radius) * stride_2;
150
151 int x2 = x1 + s2o;
152 int y2 = y1 + s2p;
153 for (int j = 0; j < kernel; j++)
154 {
155 for (int i = 0; i < kernel; i++)
156 {
157 int ji_off = ((j * kernel) + i) * inp_c;
158 for (int ch = 0; ch < inp_c; ch++)
159 {
160 int idxPatchData = ji_off + ch;
161 int idx2 = ((item * inp_h + y2 + j) * inp_w + x2 + i) * inp_c + ch;
162 sum += patch_data[idxPatchData] * inp1_data[idx2];
163 }
164 }
165 }
166 int index = ((out_ch * out_h + y) * out_w) + x;
167 out_data[index + item * topcount] = static_cast<float>(sum) / sumelems;
168 }
169 }
170 }
171 }

174 void forward(InputArrayOfArrays inputs_arr, OutputArrayOfArrays outputs_arr, OutputArrayOfArrays internals_arr) CV_OVERRIDE
175 {
176 CV_TRACE_FUNCTION();
177 CV_TRACE_ARG_VALUE(name, "name", name.c_str());
178
179 std::vector<Mat> inputs, outputs, internals;
180 inputs_arr.getMatVector(inputs);
181 outputs_arr.getMatVector(outputs);
182 internals_arr.getMatVector(internals);
183
184 blobRearrangeKernel2(inputs[0], rbot0);
185 blobRearrangeKernel2(inputs[1], rbot1);
186 for (int i = 0; i < inputs[0].size[0]; i++)
187 {
188 correlationKernelSubtraction(rbot0, rbot1, outputs[0], i);
189 }
190 }
