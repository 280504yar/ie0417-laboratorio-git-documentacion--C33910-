101#define CV__CHECK(id, op, type, v1, v2, v1_str, v2_str, msg_str) do { \
102 if(CV__TEST_##op((v1), (v2))) ; else { \
103 CV__DEFINE_CHECK_CONTEXT(id, msg_str, cv::detail::TEST_ ## op, v1_str, v2_str); \
104 cv::detail::check_failed_ ## type((v1), (v2), CV__CHECK_LOCATION_VARNAME(id)); \
105 } \
106} while (0)

108#define CV__CHECK_CUSTOM_TEST(id, type, v, test_expr, v_str, test_expr_str, msg_str) do { \

110 CV__DEFINE_CHECK_CONTEXT(id, msg_str, cv::detail::TEST_CUSTOM, v_str, test_expr_str); \
111 cv::detail::check_failed_ ## type((v), CV__CHECK_LOCATION_VARNAME(id)); \

113} while (0)
