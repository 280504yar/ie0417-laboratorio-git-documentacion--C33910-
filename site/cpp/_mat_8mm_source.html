21static int idx2Offset(cv::Mat* mat, std::vector<int>& indices) {
22 int offset = indices[0];
23 for (int dim=1; dim < mat->dims; dim++) {
24 offset = offset*mat->size[dim] + indices[dim];
25 }
26 return offset;
27}

29static void offset2Idx(cv::Mat* mat, size_t offset, std::vector<int>& indices) {
30 for (int dim=mat->dims-1; dim>=0; dim--) {
31 indices[dim] = offset % mat->size[dim];
32 offset = (offset - indices[dim]) / mat->size[dim];
33 }
34}

36// returns true if final index was reached
37static bool updateIdx(cv::Mat* mat, std::vector<int>& indices, size_t inc) {
38 size_t currentOffset = idx2Offset(mat, indices);
39 size_t newOffset = currentOffset + inc;
40 bool reachedEnd = newOffset>=(size_t)mat->total();
41 offset2Idx(mat, reachedEnd?0:newOffset, indices);
42 return reachedEnd;
43}

590- (int)put:(NSArray<NSNumber*>*)indices data:(NSArray<NSNumber*>*)data {
591 cv::Mat* mat = _nativePtr;
592 int type = mat->type();
593 int rawValueSize = (int)(mat->elemSize() / mat->channels());
594 if (data == nil || data.count % [CvType channels:type] != 0) {
595 NSException* exception = [NSException
596 exceptionWithName:@"UnsupportedOperationException"
597 reason:[NSString stringWithFormat:@"Provided data element number (%lu) should be multiple of the Mat channels count (%d)", (unsigned long)(data == nil ? 0 : data.count), [CvType channels:type]]
598 userInfo:nil];
599 @throw exception;
600 }
601 std::vector<int> tempIndices;
602 for (NSNumber* index in indices) {
603 tempIndices.push_back(index.intValue);
604 }
605 for (int index = 0; index < mat->dims; index++) {
606 if (mat->size[index]<=tempIndices[index]) {
607 return 0; // indexes out of range
608 }
609 }
610
611 int arrayAvailable = (int)data.count;
612 int matAvailable = getMatAvailable(mat, tempIndices);
613 int available = MIN(arrayAvailable, matAvailable);
614 int copyOffset = 0;
615 int copyCount = MIN((mat->size[mat->dims - 1] - tempIndices[mat->dims - 1]) * mat->channels(), available);
616 int result = (int)(available * rawValueSize);
617
618 while (available > 0) {
619 [self put:mat->ptr(tempIndices.data()) data:data offset:(int)copyOffset count:copyCount];
620 if (updateIdx(mat, tempIndices, copyCount / mat->channels())) {
621 break;
622 }
623 available -= copyCount;
624 copyOffset += copyCount;
625 copyCount = MIN(mat->size[mat->dims-1] * mat->channels(), available);
626 }
627 return result;
628}

661- (int)get:(NSArray<NSNumber*>*)indices data:(NSMutableArray<NSNumber*>*)data {
662 cv::Mat* mat = _nativePtr;
663 int type = mat->type();
664 if (data == nil || data.count % [CvType channels:type] != 0) {
665 NSException* exception = [NSException
666 exceptionWithName:@"UnsupportedOperationException"
667 reason:[NSString stringWithFormat:@"Provided data element number (%lu) should be multiple of the Mat channels count (%d)", (unsigned long)(data == nil ? 0 : data.count), [CvType channels:type]]
668 userInfo:nil];
669 @throw exception;
670 }
671 std::vector<int> tempIndices;
672 for (NSNumber* index in indices) {
673 tempIndices.push_back(index.intValue);
674 }
675 for (int index = 0; index < mat->dims; index++) {
676 if (mat->size[index]<=tempIndices[index]) {
677 return 0; // indexes out of range
678 }
679 }
680
681 int arrayAvailable = (int)data.count;
682 int copyOffset = 0;
683 int matAvailable = getMatAvailable(mat, tempIndices);
684 int available = MIN(arrayAvailable, matAvailable);
685 int copyCount = MIN((mat->size[mat->dims - 1] - tempIndices[mat->dims - 1]) * mat->channels(), available);
686 int result = (int)(available * mat->elemSize() / mat->channels());
687
688 while (available > 0) {
689 [self get:mat->ptr(tempIndices.data()) data:data offset:(int)copyOffset count:copyCount];
690 if (updateIdx(mat, tempIndices, copyCount / mat->channels())) {
691 break;
692 }
693 available -= copyCount;
694 copyOffset += copyCount;
695 copyCount = MIN(mat->size[mat->dims-1] * mat->channels(), available);
696 }
697 return result;
698}

730int getMatAvailable(cv::Mat* mat, std::vector<int>& indices) {
731 int blockSize = 1;
732 int unavailableCount = 0;
733 for (int index = mat->dims - 1; index >= 0; index--) {
734 unavailableCount += blockSize * indices[index];
735 blockSize *= mat->size[index];
736 }
737 return (int)(mat->channels() * (mat->total() - unavailableCount));
738}

740template<typename T> int getData(NSArray<NSNumber*>* indices, cv::Mat* mat, int count, T* tBuffer) {
741 std::vector<int> tempIndices;
742 for (NSNumber* index in indices) {
743 tempIndices.push_back(index.intValue);
744 }
745 for (int index = 0; index < mat->dims; index++) {
746 if (mat->size[index]<=tempIndices[index]) {
747 return 0; // indexes out of range
748 }
749 }
750
751 int arrayAvailable = count;
752 size_t countBytes = count * sizeof(T);
753 size_t remainingBytes = (size_t)(mat->total() - idx2Offset(mat, tempIndices))*mat->elemSize();
754 countBytes = (countBytes>remainingBytes)?remainingBytes:countBytes;
755 int result = (int)countBytes;
756 int matAvailable = getMatAvailable(mat, tempIndices);
757 int available = MIN(arrayAvailable, matAvailable);
758 if (mat->isContinuous()) {
759 memcpy(tBuffer, mat->ptr(tempIndices.data()), available * sizeof(T));
760 } else {
761 char* buff = (char*)tBuffer;
762 size_t blockSize = mat->size[mat->dims-1] * mat->elemSize();
763 size_t firstPartialBlockSize = (mat->size[mat->dims-1] - tempIndices[mat->dims-1]) * mat->step[mat->dims-1];
764 for (int dim=mat->dims-2; dim>=0 && blockSize == mat->step[dim]; dim--) {
765 blockSize *= mat->size[dim];
766 firstPartialBlockSize += (mat->size[dim] - (tempIndices[dim]+1)) * mat->step[dim];
767 }
768 size_t copyCount = (countBytes<firstPartialBlockSize)?countBytes:firstPartialBlockSize;
769 uchar* data = mat->ptr(tempIndices.data());
770 while(countBytes>0) {
771 memcpy(buff, data, copyCount);
772 updateIdx(mat, tempIndices, copyCount / mat->elemSize());
773 countBytes -= copyCount;
774 buff += copyCount;
775 copyCount = countBytes<blockSize?countBytes:blockSize;
776 data = mat->ptr(tempIndices.data());
777 }
778 }
779 return result;
780}
