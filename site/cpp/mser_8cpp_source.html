173 int getGray(const Pixel* ptr0, const uchar* imgptr0, int mask) const
174 {
175 return imgptr0[this - ptr0] ^ mask;
176 }
177 int getNext() const { return (val & NEXT_MASK); }
178 void setNext(int next) { val = (val & ~NEXT_MASK) | next; }

197 // the history of region grown
198 struct CompHistory
199 {
200 CompHistory()
201 {
202 parent_ = child_ = next_ = 0;
203 val = size = 0;
204 var = -1.f;
205 head = 0;
206 checked = false;
207 }
208 void updateTree( WParams& wp, CompHistory** _h0, CompHistory** _h1, bool final )
209 {
210 if( var >= 0.f )
211 return;
212 int delta = wp.p.delta;
213
214 CompHistory* h0_ = 0, *h1_ = 0;
215 CompHistory* c = child_;
216 if( size >= wp.p.minArea )
217 {
218 for( ; c != 0; c = c->next_ )
219 {
220 if( c->var < 0.f )
221 c->updateTree(wp, c == child_ ? &h0_ : 0, c == child_ ? &h1_ : 0, final);
222 if( c->var < 0.f )
223 return;
224 }
225 }
226
227 // find h0 and h1 such that:
228 // h0->val >= h->val - delta and (h0->parent == 0 or h0->parent->val < h->val - delta)
229 // h1->val <= h->val + delta and (h1->child == 0 or h1->child->val < h->val + delta)
230 // then we will adjust h0 and h1 as h moves towards latest
231 CompHistory* h0 = this, *h1 = h1_ && h1_->size > size ? h1_ : this;
232 if( h0_ )
233 {
234 for( h0 = h0_; h0 != this && h0->val < val - delta; h0 = h0->parent_ )
235 ;
236 }
237 else
238 {
239 for( ; h0->child_ && h0->child_->val >= val - delta; h0 = h0->child_ )
240 ;
241 }
242
243 for( ; h1->parent_ && h1->parent_->val <= val + delta; h1 = h1->parent_ )
244 ;
245
246 if( _h0 ) *_h0 = h0;
247 if( _h1 ) *_h1 = h1;
248
249 // when we do not well-defined ER(h->val + delta), we stop
250 // the process of computing variances unless we are at the final step
251 if( !final && !h1->parent_ && h1->val < val + delta )
252 return;
253
254 var = (float)(h1->size - h0->size)/size;
255 c = child_;
256 for( ; c != 0; c = c->next_ )
257 c->checkAndCapture(wp);
258 if( final && !parent_ )
259 checkAndCapture(wp);
260 }
261
262 void checkAndCapture( WParams& wp )
263 {
264 if( checked )
265 return;
266 checked = true;
267 if( size < wp.p.minArea || size > wp.p.maxArea || var < 0.f || var > wp.p.maxVariation )
268 return;
269 if( child_ )
270 {
271 CompHistory* c = child_;
272 for( ; c != 0; c = c->next_ )
273 {
274 if( c->var >= 0.f && var > c->var )
275 return;
276 }
277 }
278 if( var > 0.f && parent_ && parent_->var >= 0.f && var >= parent_->var )
279 return;
280 int xmin = INT_MAX, ymin = INT_MAX, xmax = INT_MIN, ymax = INT_MIN, j = 0;
281 wp.msers->push_back(vector<Point>());
282 vector<Point>& region = wp.msers->back();
283 region.resize(size);
284 const Pixel* pix0 = wp.pix0;
285 int step = wp.step;
286
287 for( PPixel pix = head; j < size; j++, pix = pix0[pix].getNext() )
288 {
289 int y = pix/step;
290 int x = pix - y*step;
291
292 xmin = std::min(xmin, x);
293 xmax = std::max(xmax, x);
294 ymin = std::min(ymin, y);
295 ymax = std::max(ymax, y);
296
297 region[j] = Point(x, y);
298 }
299
300 wp.bboxvec->push_back(Rect(xmin, ymin, xmax - xmin + 1, ymax - ymin + 1));
301 }
302
303 CompHistory* child_;
304 CompHistory* parent_;
305 CompHistory* next_;
306 int val;
307 int size;
308 float var;
309 PPixel head;
310 bool checked;
311 };

322 head = tail = 0;

328 // add history chunk to a connected component
329 void growHistory(CompHistory*& hptr, WParams& wp, int new_gray_level, bool final)
330 {
331 if (new_gray_level < gray_level)
332 new_gray_level = gray_level;
333
334 CompHistory *h;
335 if (history && history->val == gray_level)
336 {
337 h = history;
338 }
339 else
340 {
341 h = hptr++;
342 h->parent_ = 0;
343 h->child_ = history;
344 h->next_ = 0;
345
346 if (history)
347 {
348 history->parent_ = h;
349 }
350 }
351 CV_Assert(h != NULL);
352 h->val = gray_level;
353 h->size = size;
354 h->head = head;
355 h->var = FLT_MAX;
356 h->checked = true;
357 if (h->size >= wp.p.minArea)
358 {
359 h->var = -1.f;
360 h->checked = false;
361 }
362
363 gray_level = new_gray_level;
364 history = h;
365 if (history && history->val != gray_level)
366 {
367 history->updateTree(wp, 0, 0, final);
368 }
369 }
370
371 // merging two connected components
372 void merge( ConnectedComp* comp1, ConnectedComp* comp2,
373 CompHistory*& hptr, WParams& wp )
374 {
375 if (comp1->gray_level < comp2->gray_level)
376 std::swap(comp1, comp2);
377
378 gray_level = comp1->gray_level;
379 comp1->growHistory(hptr, wp, gray_level, false);
380 comp2->growHistory(hptr, wp, gray_level, false);
381
382 if (comp1->size == 0)
383 {
384 head = comp2->head;
385 tail = comp2->tail;
386 }
387 else
388 {
389 head = comp1->head;
390 wp.pix0[comp1->tail].setNext(comp2->head);
391 tail = comp2->tail;
392 }
393
394 size = comp1->size + comp2->size;
395 history = comp1->history;
396
397 CompHistory *h1 = history->child_;
398 CompHistory *h2 = comp2->history;
399 // the child_'s size should be the large one
400 if (h1 && h1->size > h2->size)
401 {
402 // add h2 as a child only if its size is large enough
403 if(h2->size >= wp.p.minArea)
404 {
405 h2->next_ = h1->next_;
406 h1->next_ = h2;
407 h2->parent_ = history;
408 }
409 }
410 else
411 {
412 history->child_ = h2;
413 h2->parent_ = history;
414 // reserve h1 as a child only if its size is large enough
415 if (h1 && h1->size >= wp.p.minArea)
416 {
417 h2->next_ = h1;
418 }
419 }
420 }
421
422 PPixel head;
423 PPixel tail;
424 CompHistory* history;
425 int gray_level;
426 int size;
427 };

434 void preprocess1( const Mat& img, int* level_size )
435 {
436 memset(level_size, 0, 256*sizeof(level_size[0]));
437
438 int i, j, cols = img.cols, rows = img.rows;
439 int step = cols;
440 pixbuf.resize(step*rows);
441 heapbuf.resize(cols*rows + 256);
442 histbuf.resize(cols*rows);
443 Pixel borderpix;
444 borderpix.setDir(5);
445
446 for( j = 0; j < step; j++ )
447 {
448 pixbuf[j] = pixbuf[j + (rows-1)*step] = borderpix;
449 }
450
451 for( i = 1; i < rows-1; i++ )
452 {
453 const uchar* imgptr = img.ptr(i);
454 Pixel* pptr = &pixbuf[i*step];
455 pptr[0] = pptr[cols-1] = borderpix;
456 for( j = 1; j < cols-1; j++ )
457 {
458 int val = imgptr[j];
459 level_size[val]++;
460 pptr[j].val = 0;
461 }
462 }
463 }

487 void pass( const Mat& img, vector<vector<Point> >& msers, vector<Rect>& bboxvec,
488 Size size, const int* level_size, int mask )
489 {
490 CompHistory* histptr = &histbuf[0];
491 int step = size.width;
492 Pixel *ptr0 = &pixbuf[0], *ptr = &ptr0[step+1];
493 const uchar* imgptr0 = img.ptr();
494 Pixel** heap[256];
495 ConnectedComp comp[257];
496 ConnectedComp* comptr = &comp[0];
497 WParams wp;
498 wp.p = params;
499 wp.msers = &msers;
500 wp.bboxvec = &bboxvec;
501 wp.pix0 = ptr0;
502 wp.step = step;
503
504 heap[0] = &heapbuf[0];
505 heap[0][0] = 0;
506
507 for( int i = 1; i < 256; i++ )
508 {
509 heap[i] = heap[i-1] + level_size[i-1] + 1;
510 heap[i][0] = 0;
511 }
512
513 comptr->gray_level = 256;
514 comptr++;
515 comptr->gray_level = ptr->getGray(ptr0, imgptr0, mask);
516 ptr->setDir(1);
517 int dir[] = { 0, 1, step, -1, -step };
518 for( ;; )
519 {
520 int curr_gray = ptr->getGray(ptr0, imgptr0, mask);
521 int nbr_idx = ptr->getDir();
522 // take tour of all the 4 directions
523 for( ; nbr_idx <= 4; nbr_idx++ )
524 {
525 // get the neighbor
526 Pixel* ptr_nbr = ptr + dir[nbr_idx];
527 if( !ptr_nbr->isVisited() )
528 {
529 // set dir=1, next=0
530 ptr_nbr->val = 1 << DIR_SHIFT;
531 int nbr_gray = ptr_nbr->getGray(ptr0, imgptr0, mask);
532 if( nbr_gray < curr_gray )
533 {
534 // when the value of neighbor smaller than current
535 // push current to boundary heap and make the neighbor to be the current one
536 // create an empty comp
537 *(++heap[curr_gray]) = ptr;
538 ptr->val = (nbr_idx+1) << DIR_SHIFT;
539 ptr = ptr_nbr;
540 comptr++;
541 comptr->init(nbr_gray);
542 curr_gray = nbr_gray;
543 nbr_idx = 0;
544 continue;
545 }
546 // otherwise, push the neighbor to boundary heap
547 *(++heap[nbr_gray]) = ptr_nbr;
548 }
549 }
550
551 // set dir = nbr_idx, next = 0
552 ptr->val = nbr_idx << DIR_SHIFT;
553 int ptrofs = (int)(ptr - ptr0);
554 CV_Assert(ptrofs != 0);
555
556 // add a pixel to the pixel list
557 if( comptr->tail )
558 ptr0[comptr->tail].setNext(ptrofs);
559 else
560 comptr->head = ptrofs;
561 comptr->tail = ptrofs;
562 comptr->size++;
563 // get the next pixel from boundary heap
564 if( *heap[curr_gray] )
565 {
566 ptr = *heap[curr_gray];
567 heap[curr_gray]--;
568 }
569 else
570 {
571 for( curr_gray++; curr_gray < 256; curr_gray++ )
572 {
573 if( *heap[curr_gray] )
574 break;
575 }
576 if( curr_gray >= 256 )
577 break;
578
579 ptr = *heap[curr_gray];
580 heap[curr_gray]--;
581
582 if (curr_gray < comptr[-1].gray_level)
583 {
584 comptr->growHistory(histptr, wp, curr_gray, false);
585 CV_DbgAssert(comptr->size == comptr->history->size);
586 }
587 else
588 {
589 // there must one pixel with the second component's gray level in the heap,
590 // so curr_gray is not large than the second component's gray level
591 comptr--;
592 CV_DbgAssert(curr_gray == comptr->gray_level);
593 comptr->merge(comptr, comptr + 1, histptr, wp);
594 CV_DbgAssert(curr_gray == comptr->gray_level);
595 }
596 }
597 }
598
599 for( ; comptr->gray_level != 256; comptr-- )
600 {
601 comptr->growHistory(histptr, wp, 256, true);
602 }
603 }

657static double ChiSquaredDistance( const uchar* x, const uchar* y )
658{
659 return (double)((x[0]-y[0])*(x[0]-y[0]))/(double)(x[0]+y[0]+1e-10)+
660 (double)((x[1]-y[1])*(x[1]-y[1]))/(double)(x[1]+y[1]+1e-10)+
661 (double)((x[2]-y[2])*(x[2]-y[2]))/(double)(x[2]+y[2]+1e-10);
662}

665{
666 node->gmsr = node->tmsr = NULL;
667 node->reinit = 0xffff;
668 node->rank = 0;
669 node->sizei = node->size = 1;
670 node->prev = node->next = node->shortcut = node;

673// the preprocess to get the edge list with proper gaussian blur
674static int preprocessMSER_8uC3( MSCRNode* node,
675 MSCREdge* edge,
676 double* total,
677 const Mat& src,
678 Mat& dx,
679 Mat& dy,
680 int Ne,
681 int edgeBlurSize )
682{
683 int srccpt = (int)(src.step-src.cols*3);
684 const uchar* srcptr = src.ptr();
685 const uchar* lastptr = srcptr+3;
686 double* dxptr = dx.ptr<double>();
687 for ( int i = 0; i < src.rows; i++ )
688 {
689 for ( int j = 0; j < src.cols-1; j++ )
690 {
691 *dxptr = ChiSquaredDistance( srcptr, lastptr );
692 dxptr++;
693 srcptr += 3;
694 lastptr += 3;
695 }
696 srcptr += srccpt+3;
697 lastptr += srccpt+3;
698 }
699 srcptr = src.ptr();
700 lastptr = srcptr+src.step;
701 double* dyptr = dy.ptr<double>();
702 for ( int i = 0; i < src.rows-1; i++ )
703 {
704 for ( int j = 0; j < src.cols; j++ )
705 {
706 *dyptr = ChiSquaredDistance( srcptr, lastptr );
707 dyptr++;
708 srcptr += 3;
709 lastptr += 3;
710 }
711 srcptr += srccpt;
712 lastptr += srccpt;
713 }
714 // get dx and dy and blur it
715 if ( edgeBlurSize >= 1 )
716 {
717 GaussianBlur( dx, dx, Size(edgeBlurSize, edgeBlurSize), 0 );
718 GaussianBlur( dy, dy, Size(edgeBlurSize, edgeBlurSize), 0 );
719 }
720 dxptr = dx.ptr<double>();
721 dyptr = dy.ptr<double>();
722 // assian dx, dy to proper edge list and initialize mscr node
723 // the nasty code here intended to avoid extra loops
724 MSCRNode* nodeptr = node;
725 initMSCRNode( nodeptr );
726 nodeptr->index = 0;
727 *total += edge->chi = *dxptr;
728 dxptr++;
729 edge->left = nodeptr;
730 edge->right = nodeptr+1;
731 edge++;

825class LessThanEdge
826{
827public:
828 bool operator()(const MSCREdge& a, const MSCREdge& b) const { return a.chi < b.chi; }
829};

831// to find the root of one region
832static MSCRNode* findMSCR( MSCRNode* x )
833{
834 MSCRNode* prev = x;
835 MSCRNode* next;
836 for ( ; ; )
837 {
838 next = x->shortcut;
839 x->shortcut = prev;
840 if ( next == x ) break;
841 prev= x;
842 x = next;
843 }
844 MSCRNode* root = x;
845 for ( ; ; )
846 {
847 prev = x->shortcut;
848 x->shortcut = root;
849 if ( prev == x ) break;
850 x = prev;
851 }
852 return root;
853}

855// the stable mscr should be:
856// bigger than minArea and smaller than maxArea
857// differ from its ancestor more than minDiversity
858static bool MSCRStableCheck( MSCRNode* x, const MSER_Impl::Params& params )
859{
860 if ( x->size <= params.minArea || x->size >= params.maxArea )
861 return false;
862 if ( x->gmsr == NULL )
863 return true;
864 double div = (double)(x->size-x->gmsr->size)/(double)x->size;
865 return div > params.minDiversity;
866}

868static void
869extractMSER_8uC3( const Mat& src,
870 vector<vector<Point> >& msers,
871 vector<Rect>& bboxvec,
872 const MSER_Impl::Params& params )
873{
874 bboxvec.clear();
875 AutoBuffer<MSCRNode> mapBuf(src.cols*src.rows);
876 MSCRNode* map = mapBuf.data();
877 int Ne = src.cols*src.rows*2-src.cols-src.rows;
878 AutoBuffer<MSCREdge> edgeBuf(Ne);
879 MSCREdge* edge = edgeBuf.data();
880 AutoBuffer<TempMSCR> mscrBuf(src.cols*src.rows);
881 TempMSCR* mscr = mscrBuf.data();
882 double emean = 0;
883 Mat dx( src.rows, src.cols-1, CV_64FC1 );
884 Mat dy( src.rows-1, src.cols, CV_64FC1 );
885 Ne = preprocessMSER_8uC3( map, edge, &emean, src, dx, dy, Ne, params.edgeBlurSize );
886 emean = emean / (double)Ne;
887 std::sort(edge, edge + Ne, LessThanEdge());
888 MSCREdge* edge_ub = edge+Ne;
889 MSCREdge* edgeptr = edge;
890 TempMSCR* mscrptr = mscr;
891 // the evolution process
892 for ( int i = 0; i < params.maxEvolution; i++ )
893 {
894 double k = (double)i/(double)params.maxEvolution*(TABLE_SIZE-1);
895 int ti = cvFloor(k);
896 double reminder = k-ti;
897 double thres = emean*(chitab3[ti]*(1-reminder)+chitab3[ti+1]*reminder);
898 // to process all the edges in the list that chi < thres
899 while ( edgeptr < edge_ub && edgeptr->chi < thres )
900 {
901 MSCRNode* lr = findMSCR( edgeptr->left );
902 MSCRNode* rr = findMSCR( edgeptr->right );
903 // get the region root (who is responsible)
904 if ( lr != rr )
905 {
906 // rank idea take from: N-tree Disjoint-Set Forests for Maximally Stable Extremal Regions
907 if ( rr->rank > lr->rank )
908 {
909 MSCRNode* tmp;
910 CV_SWAP( lr, rr, tmp );
911 } else if ( lr->rank == rr->rank ) {
912 // at the same rank, we will compare the size
913 if ( lr->size > rr->size )
914 {
915 MSCRNode* tmp;
916 CV_SWAP( lr, rr, tmp );
917 }
918 lr->rank++;
919 }
920 rr->shortcut = lr;
921 lr->size += rr->size;
922 // join rr to the end of list lr (lr is a endless double-linked list)
923 lr->prev->next = rr;
924 lr->prev = rr->prev;
925 rr->prev->next = lr;
926 rr->prev = lr;
927 // area threshold force to reinitialize
928 if ( lr->size > (lr->size-rr->size)*params.areaThreshold )
929 {
930 lr->sizei = lr->size;
931 lr->reinit = i;
932 if ( lr->tmsr != NULL )
933 {
934 lr->tmsr->m = lr->dt-lr->di;
935 lr->tmsr = NULL;
936 }
937 lr->di = edgeptr->chi;
938 lr->s = 1e10;
939 }
940 lr->dt = edgeptr->chi;
941 if ( i > lr->reinit )
942 {
943 double s = (double)(lr->size-lr->sizei)/(lr->dt-lr->di);
944 if ( s < lr->s )
945 {
946 // skip the first one and check stability
947 if ( i > lr->reinit+1 && MSCRStableCheck( lr, params ) )
948 {
949 if ( lr->tmsr == NULL )
950 {
951 lr->gmsr = lr->tmsr = mscrptr;
952 mscrptr++;
953 }
954 lr->tmsr->size = lr->size;
955 lr->tmsr->head = lr;
956 lr->tmsr->tail = lr->prev;
957 lr->tmsr->m = 0;
958 }
959 lr->s = s;
960 }
961 }
962 }
963 edgeptr++;
964 }
965 if ( edgeptr >= edge_ub )
966 break;
967 }
968 for ( TempMSCR* ptr = mscr; ptr < mscrptr; ptr++ )
969 // to prune area with margin less than minMargin
970 if ( ptr->m > params.minMargin )
971 {
972 MSCRNode* lpt = ptr->head;
973 int xmin = INT_MAX, ymin = INT_MAX, xmax = INT_MIN, ymax = INT_MIN;
974 msers.push_back(vector<Point>());
975 vector<Point>& mser = msers.back();
976
977 for ( int i = 0; i < ptr->size; i++ )
978 {
979 Point pt;
980 pt.x = (lpt->index)&0xffff;
981 pt.y = (lpt->index)>>16;
982 xmin = std::min(xmin, pt.x);
983 xmax = std::max(xmax, pt.x);
984 ymin = std::min(ymin, pt.y);
985 ymax = std::max(ymax, pt.y);
986
987 lpt = lpt->next;
988 mser.push_back(pt);
989 }
990 bboxvec.push_back(Rect(xmin, ymin, xmax - xmin + 1, ymax - ymin + 1));
991 }
992}

994void MSER_Impl::detectRegions( InputArray _src, vector<vector<Point> >& msers, vector<Rect>& bboxes )
995{
996 CV_INSTRUMENT_REGION();
997
998 Mat src = _src.getMat();
999
1000 msers.clear();
1001 bboxes.clear();
1002
1003 if( src.rows < 3 || src.cols < 3 )
1004 CV_Error(Error::StsBadArg, "Input image is too small. Expected at least 3x3");
1005
1006 Size size = src.size();
1007
1008 if( src.type() == CV_8U )
1009 {
1010 int level_size[256];
1011 if( !src.isContinuous() )
1012 {
1013 src.copyTo(tempsrc);
1014 src = tempsrc;
1015 }
1016
1017 // darker to brighter (MSER+)
1018 preprocess1( src, level_size );
1019 if( !params.pass2Only )
1020 pass( src, msers, bboxes, size, level_size, 0 );
1021 // brighter to darker (MSER-)
1022 preprocess2( src, level_size );
1023 pass( src, msers, bboxes, size, level_size, 255 );
1024 }
1025 else
1026 {
1027 CV_Assert( src.type() == CV_8UC3 || src.type() == CV_8UC4 );
1028 extractMSER_8uC3( src, msers, bboxes, params );
1029 }

1032void MSER_Impl::detect( InputArray _image, vector<KeyPoint>& keypoints, InputArray _mask )
1033{
1034 CV_INSTRUMENT_REGION();
1035
1036 // Empty input yields no keypoints, matching the base Feature2D::detect() this overrides.
1037 if( _image.empty() )
1038 {
1039 keypoints.clear();
1040 return;
1041 }
1042
1043 if( _image.type() != CV_8UC1 && _image.type() != CV_8UC3 && _image.type() != CV_8UC4 )
1044 CV_Error( Error::StsBadArg, "image has incorrect type (must be CV_8UC1, CV_8UC3 or CV_8UC4)" );
1045
1046 if( !_mask.empty() && (_mask.type() != CV_8UC1 || _mask.size() != _image.size()) )
1047 CV_Error( Error::StsBadArg, "mask has incorrect type (!=CV_8UC1) or size (!=image size)" );
1048
1049 keypoints.clear();
1050
1051 vector<Rect> bboxes;
1052 vector<vector<Point> > msers;
1053 Mat mask = _mask.getMat();
1054
1055 detectRegions(_image, msers, bboxes);
1056 int i, ncomps = (int)msers.size();
1057
1058 for( i = 0; i < ncomps; i++ )
1059 {
1060 Rect r = bboxes[i];
1061 // TODO check transformation from MSER region to KeyPoint
1062 RotatedRect rect = fitEllipse(Mat(msers[i]));
1063 float diam = std::sqrt(rect.size.height*rect.size.width);
1064
1065 if( diam > std::numeric_limits<float>::epsilon() && r.contains(rect.center) &&
1066 (mask.empty() || mask.at<uchar>(cvRound(rect.center.y), cvRound(rect.center.x)) != 0) )
1067 keypoints.push_back( KeyPoint(rect.center, diam) );
1068 }
1069}
