253 protected int getFrameRotation(boolean cameraFacingFront, int cameraSensorOrientation) {
254 WindowManager windowManager = (WindowManager) getContext().getSystemService(Context.WINDOW_SERVICE);
255 int screenOrientation = windowManager.getDefaultDisplay().getRotation();
256 int screenRotation = 0;
257 switch (screenOrientation) {
258 case Surface.ROTATION_0:
259 screenRotation = 0;
260 break;
261 case Surface.ROTATION_90:
262 screenRotation = 90;
263 break;
264 case Surface.ROTATION_180:
265 screenRotation = 180;
266 break;
267 case Surface.ROTATION_270:
268 screenRotation = 270;
269 break;
270 }
271
272 int frameRotation;
273 if (cameraFacingFront) {
274 frameRotation = (cameraSensorOrientation + screenRotation) % 360;
275 } else {
276 frameRotation = (cameraSensorOrientation - screenRotation + 360) % 360;
277 }
278
279 return frameRotation;

403 private void checkCurrentState() {
404 Log.d(TAG, "call checkCurrentState");
405 int targetState;
406
407 if (mEnabled && mCameraPermissionGranted && mSurfaceExist && getVisibility() == VISIBLE) {
408 targetState = STARTED;
409 } else {
410 targetState = STOPPED;
411 }
412
413 if (targetState != mState) {
414 /* The state change detected. Need to exit the current state and enter target state */
415 processExitState(mState);
416 mState = targetState;
417 processEnterState(mState);
418 }

492 protected void deliverAndDrawFrame(CvCameraViewFrame frame) {
493 Mat modified;
494
495 if (mListener != null) {
496 modified = mListener.onCameraFrame(frame);
497 } else {
498 modified = frame.rgba();
499 }
500
501 boolean bmpValid = true;
502 if (modified != null) {
503 try {
504 Utils.matToBitmap(modified, mCacheBitmap);
505 } catch(Exception e) {
506 Log.e(TAG, "Mat type: " + modified);
507 Log.e(TAG, "Bitmap type: " + mCacheBitmap.getWidth() + "*" + mCacheBitmap.getHeight());
508 Log.e(TAG, "Utils.matToBitmap() throws an exception: " + e.getMessage());
509 bmpValid = false;
510 }
511 }
512
513 if (bmpValid && mCacheBitmap != null) {
514 Canvas canvas = getHolder().lockCanvas();
515 if (canvas != null) {
516 canvas.drawColor(0, android.graphics.PorterDuff.Mode.CLEAR);
517 if (BuildConfig.DEBUG)
518 Log.d(TAG, "mStretch value: " + mScale);
519
520 if (mScale != 0) {
521 canvas.drawBitmap(mCacheBitmap, new Rect(0,0,mCacheBitmap.getWidth(), mCacheBitmap.getHeight()),
522 new Rect((int)((canvas.getWidth() - mScale*mCacheBitmap.getWidth()) / 2),
523 (int)((canvas.getHeight() - mScale*mCacheBitmap.getHeight()) / 2),
524 (int)((canvas.getWidth() - mScale*mCacheBitmap.getWidth()) / 2 + mScale*mCacheBitmap.getWidth()),
525 (int)((canvas.getHeight() - mScale*mCacheBitmap.getHeight()) / 2 + mScale*mCacheBitmap.getHeight())), null);
526 } else {
527 canvas.drawBitmap(mCacheBitmap, new Rect(0,0,mCacheBitmap.getWidth(), mCacheBitmap.getHeight()),
528 new Rect((canvas.getWidth() - mCacheBitmap.getWidth()) / 2,
529 (canvas.getHeight() - mCacheBitmap.getHeight()) / 2,
530 (canvas.getWidth() - mCacheBitmap.getWidth()) / 2 + mCacheBitmap.getWidth(),
531 (canvas.getHeight() - mCacheBitmap.getHeight()) / 2 + mCacheBitmap.getHeight()), null);
532 }
533
534 if (mFpsMeter != null) {
535 mFpsMeter.measure();
536 mFpsMeter.draw(canvas, 20, 30);
537 }
538 getHolder().unlockCanvasAndPost(canvas);
539 }

578 protected Size calculateCameraFrameSize(List<?> supportedSizes, ListItemAccessor accessor, int surfaceWidth, int surfaceHeight) {
579 int calcWidth = 0;
580 int calcHeight = 0;
581
582 int maxAllowedWidth = (mMaxWidth != MAX_UNSPECIFIED && mMaxWidth < surfaceWidth)? mMaxWidth : surfaceWidth;
583 int maxAllowedHeight = (mMaxHeight != MAX_UNSPECIFIED && mMaxHeight < surfaceHeight)? mMaxHeight : surfaceHeight;
584
585 for (Object size : supportedSizes) {
586 int width = accessor.getWidth(size);
587 int height = accessor.getHeight(size);
588 Log.d(TAG, "trying size: " + width + "x" + height);
589
590 if (width <= maxAllowedWidth && height <= maxAllowedHeight) {
591 if (width >= calcWidth && height >= calcHeight) {
592 calcWidth = (int) width;
593 calcHeight = (int) height;
594 }
595 }
596 }
597 if ((calcWidth == 0 || calcHeight == 0) && supportedSizes.size() > 0)
598 {
599 Log.i(TAG, "fallback to the first frame size");
600 Object size = supportedSizes.get(0);
601 calcWidth = accessor.getWidth(size);
602 calcHeight = accessor.getHeight(size);
603 }
604
605 return new Size(calcWidth, calcHeight);
606 }
