9/*
10* Uniform Sampler:
11* Choose uniformly m (sample size) points from N (points size).
12* Uses Fisher-Yates shuffle.
13*/
14class UniformSamplerImpl : public UniformSampler {
15private:
16 std::vector<int> points_random_pool;
17 int sample_size, points_size = 0;
18 RNG rng;
19public:
20
21 UniformSamplerImpl (int state, int sample_size_, int points_size_)
22 : rng(state)
23 {
24 sample_size = sample_size_;
25 setPointsSize (points_size_);
26 }
27 void setNewPointsSize (int points_size_) override {
28 setPointsSize(points_size_);
29 }
30 void generateSample (std::vector<int> &sample) override {
31 int random_pool_size = points_size; // random points of entire range
32 for (int i = 0; i < sample_size; i++) {
33 // get random point index
34 const int array_random_index = rng.uniform(0, random_pool_size);
35 // get point by random index
36 // store sample
37 sample[i] = points_random_pool[array_random_index];
38 // swap random point with the end of random pool
39 std::swap(points_random_pool[array_random_index],
40 points_random_pool[--random_pool_size]);
41 }
42 }
43private:
44 void setPointsSize (int points_size_) {
45 CV_Assert (sample_size <= points_size_);
46
47 if (points_size_ > points_size)
48 points_random_pool = std::vector<int>(points_size_);
49
50 if (points_size != points_size_) {
51 points_size = points_size_;
52
53 for (int i = 0; i < points_size; i++)
54 points_random_pool[i] = i;
55 }
56 }
57};

63/*
64* PROSAC (simple) sampler does not use array of precalculated T_n (n is subset size) samples, but computes T_n for
65* specific n directly in generateSample() function.
66* Also, the stopping length (or maximum subset size n*) by default is set to points_size (N) and does not update
67* during computation.
68*/
69class ProsacSimpleSamplerImpl : public ProsacSimpleSampler {
70protected:
71 int points_size, subset_size, t_n_prime, kth_sample_number,
72 max_prosac_samples_count, largest_sample_size, sample_size;
73 double t_n;
74 Ptr<UniformRandomGenerator> random_gen;
75public:
76 ProsacSimpleSamplerImpl (int state, int points_size_, int sample_size_,
77 int max_prosac_samples_count_) : random_gen(UniformRandomGenerator::create(state)) {
78
79 CV_Assert(sample_size_ <= points_size_);
80 sample_size = sample_size_;
81 points_size = points_size_;
82 max_prosac_samples_count = max_prosac_samples_count_;
83 initialize ();
84 }
85
86 void generateSample (std::vector<int> &sample) override {
87 if (kth_sample_number > max_prosac_samples_count) {
88 // do uniform sampling, if prosac has not found solution
89 random_gen->generateUniqueRandomSet(sample, sample_size, points_size);
90 return;
91 }
92
93 kth_sample_number++; // t := t + 1
94
95 // Choice of the hypothesis generation set
96 if (kth_sample_number >= t_n_prime && subset_size < largest_sample_size) {
97 // do not use array of growth sample, calculate it directly
98 double t_n_plus1 = (subset_size + 1) * t_n / (subset_size + 1 - sample_size);
99 t_n_prime += static_cast<int>(ceil(t_n_plus1 - t_n));
100 t_n = t_n_plus1;
101 subset_size++;
102 }
103
104 // Semi-random sample Mt of size m
105 if (t_n_prime < kth_sample_number) {
106 random_gen->generateUniqueRandomSet(sample, sample_size, subset_size);
107 } else {
108 random_gen->generateUniqueRandomSet(sample, sample_size-1, subset_size-1);
109 sample[sample_size-1] = subset_size-1; // Make the last point from the nth position.
110 }
111 }
112
113 // Set the sample such that you are sampling the kth prosac sample (Eq. 6).
114 void setSampleNumber (int k) {
115 kth_sample_number = k;
116
117 // If the method should act exactly like RANSAC
118 if (kth_sample_number > max_prosac_samples_count)
119 return;
120 else { // Increment the size of the sampling pool while required
121 t_n = max_prosac_samples_count;
122 t_n_prime = 1; // reset growth function
123 subset_size = sample_size; // reset subset size as from the beginning
124 for (int i = 0; i < sample_size; i++)
125 t_n *= static_cast<double>(subset_size - i) / (points_size - i);
126
127 while (kth_sample_number > t_n_prime) { // t_n_prime == growth_function
128 double t_n_plus1 = static_cast<double>(subset_size + 1) * t_n / (subset_size + 1 - sample_size);
129 t_n_prime += static_cast<int>(ceil(t_n_plus1 - t_n));
130 t_n = t_n_plus1;
131 subset_size++;
132 }
133 if (subset_size > points_size)
134 subset_size = points_size;
135 }
136 }
137
138 void setNewPointsSize (int points_size_) override {
139 CV_Assert(sample_size <= points_size_);
140 points_size = points_size_;
141 initialize ();
142 }
143private:
144 void initialize () {
145 largest_sample_size = points_size; // termination length, n*
146 subset_size = sample_size; // n
147 t_n = max_prosac_samples_count;
148 t_n_prime = 1;
149
150 // From Equations leading up to Eq 3 in Chum et al.
151 // t_n samples containing only data points from U_n and
152 // t_n+1 samples containing only data points from U_n+1
153 for (int i = 0; i < sample_size; i++)
154 t_n *= static_cast<double>(subset_size - i) / (points_size - i);
155
156 kth_sample_number = 0;
157 }

166class ProsacSamplerImpl : public ProsacSampler {
167protected:
168 std::vector<int> growth_function;
169
170 // subset_size = size of sampling range (subset of good sorted points)
171 // termination_length = n*, maximum sampling range (largest subset size)
172 int points_size, sample_size, subset_size, termination_length;
173
174 // it is T_N
175 // Imagine standard RANSAC drawing T_N samples of size m out of N data points
176 // In our experiments, the parameter was set to T_N = 200000
177 int growth_max_samples;
178
179 // how many times PROSAC generateSample() was called
180 int kth_sample_number;
181 Ptr<UniformRandomGenerator> random_gen;
182public:
183 void setTerminationLength (int termination_length_) override {
184 termination_length = termination_length_;
185 }
186
187 // return constant reference to prosac termination criteria
188 int getKthSample () const override {
189 return kth_sample_number;
190 }
191
192 // return constant points of growth function to prosac termination criteria
193 const std::vector<int> & getGrowthFunction () const override {
194 return growth_function;
195 }
196
197 ProsacSamplerImpl (int state, int points_size_, int sample_size_,
198 int growth_max_samples_) : random_gen(UniformRandomGenerator::create(state)) {
199 CV_Assert(sample_size_ <= points_size_);
200
201 sample_size = sample_size_;
202 points_size = points_size_;
203
204 growth_max_samples = growth_max_samples_;
205 growth_function = std::vector<int>(points_size);
206
207 kth_sample_number = 0;
208
209 // The data points in U_N are sorted in descending order w.r.t. the quality function q.
210 // Let {Mi}i = 1...T_N denote the sequence of samples Mi c U_N that are uniformly drawn by Ransac.
211
212 // Let T_n be an average number of samples from {Mi}i=1...T_N that contain data points from U_n only.
213 // compute initial value for T_n
214 // n - i
215 // T_n = T_N * Product i = 0...m-1 -------, n >= sample size, N = points size
216 // N - i
217 double T_n = growth_max_samples;
218 for (int i = 0; i < sample_size; i++)
219 T_n *= static_cast<double> (sample_size-i) / (points_size-i);
220
221 int T_n_prime = 1;
222
223 // fill growth function with T'_n until sample_size
224 for (int n = 0; n < sample_size; n++)
225 growth_function[n] = T_n_prime;
226
227 // compute values using recurrent relation
228 // n + 1
229 // T(n+1) = --------- T(n), m is sample size.
230 // n + 1 - m
231
232 // growth function is defined as
233 // g(t) = min {n, T'_(n) >= t}
234 // T'_(n+1) = T'_(n) + (T_(n+1) - T_(n))
235 // T'_m = 1
236 for (int n = sample_size; n < points_size; n++) {
237 double Tn_plus1 = static_cast<double>(n + 1) * T_n / (n + 1 - sample_size);
238 growth_function[n] = T_n_prime + (int) ceil(Tn_plus1 - T_n); // T'_{n+1}
239
240 // update
241 T_n = Tn_plus1;
242 T_n_prime = growth_function[n]; // T'_{n+1}
243 }
244
245 // other initializations
246 termination_length = points_size; // n* = N, largest set sampled in PROSAC (termination length)
247 subset_size = sample_size; // n, size of the current sampling pool
248 kth_sample_number = 0; // t (iteration)
249 }
250
251 void generateSample (std::vector<int> &sample) override {
252 if (kth_sample_number > growth_max_samples) {
253 // if PROSAC has not converged to solution then do uniform sampling.
254 random_gen->generateUniqueRandomSet(sample, sample_size, points_size);
255 return;
256 }
257
258 kth_sample_number++; // t := t + 1
259
260 // Choice of the hypothesis generation set
261 // if (t = T'_n) & (n < n*) then n = n + 1 (eqn. 4)
262 if (kth_sample_number >= growth_function[subset_size-1] && subset_size < termination_length)
263 subset_size++;
264
265 // Semi-random sample M_t of size m
266 // if T'n < t then
267 if (growth_function[subset_size-1] < kth_sample_number) {
268 if (subset_size >= termination_length) {
269 random_gen->generateUniqueRandomSet(sample, sample_size, subset_size);
270 } else {
271 // The sample contains m-1 points selected from U_(n-1) at random and u_n
272 random_gen->generateUniqueRandomSet(sample, sample_size-1, subset_size-1);
273 sample[sample_size-1] = subset_size-1;
274 }
275 } else {
276 // Select m points from U_n at random.
277 random_gen->generateUniqueRandomSet(sample, sample_size, subset_size);
278 }
279 }
280
281 // Set the sample such that you are sampling the kth prosac sample (Eq. 6).
282 void setSampleNumber (int k) {
283 kth_sample_number = k;
284
285 // If the method should act exactly like RANSAC
286 if (kth_sample_number > growth_max_samples)
287 return;
288 else { // Increment the size of the sampling pool while required
289 subset_size = sample_size; // reset subset size as from the beginning
290 while (kth_sample_number > growth_function[subset_size-1]) {
291 subset_size++;
292 if (subset_size >= points_size){
293 subset_size = points_size;
294 break;
295 }
296 }
297 if (termination_length < subset_size)
298 termination_length = subset_size;
299 }
300 }
301
302 void setNewPointsSize (int /*points_size_*/) override {
303 CV_Error(cv::Error::StsError, "Changing points size in PROSAC requires to change also "
304 "termination criteria! Use PROSAC simpler version");
305 }

314class ProgressiveNapsacImpl : public ProgressiveNapsac {
315private:
316 int max_progressive_napsac_iterations, points_size;
317 // how many times generateSample() was called.
318 int kth_sample_number, grid_layers_number, sample_size, sampler_length;
319
320 const Ptr<UniformRandomGenerator> random_generator;
321 ProsacSamplerImpl one_point_prosac, prosac_sampler;
322
323 // The overlapping neighborhood layers
324 const std::vector<Ptr<NeighborhoodGraph>> * layers;
325
326 std::vector<int> growth_function;
327 std::vector<int> hits_per_point; // number of iterations, t
328 std::vector<int> subset_size_per_point; // k
329 std::vector<int> current_layer_per_point; // layer of grid neighborhood graph
330public:
331
332 // points must be sorted
333 ProgressiveNapsacImpl (int state,int points_size_, int sample_size_,
334 const std::vector<Ptr<NeighborhoodGraph>> &layers_, int sampler_length_) :
335 // initialize one-point prosac sampler and global prosac sampler
336 random_generator (UniformRandomGenerator::create(state)),
337 one_point_prosac (random_generator->getRandomNumber(INT_MAX), points_size_,
338 1 /* sample_size*/,points_size_),
339 prosac_sampler (random_generator->getRandomNumber(INT_MAX), points_size_,
340 sample_size_, 200000), layers(&layers_) {
341 CV_Assert(sample_size_ <= points_size_);
342 sample_size = sample_size_;
343 points_size = points_size_;
344 sampler_length = sampler_length_;
345 grid_layers_number = static_cast<int>(layers_.size());
346
347 // Create growth function for P-NAPSAC
348 growth_function = std::vector<int>(points_size);
349
350 // 20 is sampler_length = The length of fully blending to global sampling
351 max_progressive_napsac_iterations = sampler_length * points_size;
352
353 const int local_sample_size = sample_size - 1; // not including initial point
354 double T_n = max_progressive_napsac_iterations;
355 for (int i = 0; i < local_sample_size; i++)
356 T_n *= static_cast<double> (local_sample_size - i) / (points_size - i);
357
358 // calculate growth function by recurrent relation (see PROSAC)
359 int T_n_prime = 1;
360 for (int n = 0; n < points_size; n++) {
361 if (n + 1 <= local_sample_size) {
362 growth_function[n] = T_n_prime;
363 continue;
364 }
365 double Tn_plus1 = (n+1) * T_n / (n + 1 - local_sample_size);
366 growth_function[n] = T_n_prime + static_cast<int>(ceil(Tn_plus1 - T_n));
367 T_n = Tn_plus1;
368 T_n_prime = growth_function[n];
369 }
370
371 subset_size_per_point = std::vector<int>(points_size, sample_size); // subset size
372 hits_per_point = std::vector<int>(points_size, 0); // 0 hits
373 current_layer_per_point = std::vector<int>(points_size, 0); // 0-th layer
374
375 kth_sample_number = 0; // iteration
376 }
377
378 void generateSample (std::vector<int> &sample) override {
379 // Do completely global sampling (PROSAC is used now), instead of Progressive NAPSAC,
380 // if the maximum iterations has been done without finding the sought model.
381 if (kth_sample_number > max_progressive_napsac_iterations) {
382 prosac_sampler.generateSample(sample);
383 return;
384 }
385
386 kth_sample_number++;
387
388 // get PROSAC one-point sample (initial point)
389 one_point_prosac.generateSample(sample);
390 const int initial_point = sample[0];
391
392 // get hits number and subset size (i.e., the size of the neighborhood sphere)
393 // of initial point (note, get by reference)
394 int &iters_of_init_pt = ++hits_per_point[initial_point]; // t := t + 1, increase iteration
395 int &subset_size_of_init_pt = subset_size_per_point[initial_point];
396
397 while (iters_of_init_pt > growth_function[subset_size_of_init_pt - 1] && subset_size_of_init_pt < points_size)
398 subset_size_of_init_pt++;
399
400 // Get layer of initial point (note, get by reference)
401 int &current_layer = current_layer_per_point[initial_point];
402
403 bool is_last_layer = false;
404 do {// Try to find the grid which contains enough points
405 // In the case when the grid with a single cell is used,
406 // apply PROSAC.
407 if (current_layer >= grid_layers_number) {
408 is_last_layer = true;
409 break;
410 }
411
412 // If there are not enough points in the cell, start using a
413 // less fine grid.
414 if ((int)layers->at(current_layer)->getNeighbors(initial_point).size() < subset_size_of_init_pt) {
415 ++current_layer; // Jump to the next layer with bigger cells.
416 continue;
417 }
418 // If the procedure got to this point, there is no reason to choose a different layer of grids
419 // since the current one has enough points.
420 break;
421 } while (true);
422
423 // If not the last layer has been chosen, sample from the neighbors of the initially selected point.
424 if (!is_last_layer) {
425 // The indices of the points which are in the same cell as the
426 // initially selected one.
427 const std::vector<int> &neighbors = layers->at(current_layer)->getNeighbors(initial_point);
428
429 // Put the selected point to the end of the sample array to avoid
430 // being overwritten when sampling the remaining points.
431 sample[sample_size - 1] = initial_point;
432
433 // The next point should be the farthest one from the initial point. Note that the points in the grid cell are
434 // not ordered w.r.t. to their distances from the initial point. However, they are ordered as in PROSAC.
435 sample[sample_size - 2] = neighbors[subset_size_of_init_pt - 1];
436
437 // Select n - 2 points randomly
438 random_generator->generateUniqueRandomSet(sample, sample_size - 2, subset_size_of_init_pt - 1);
439
440 for (int i = 0; i < sample_size - 2; i++) {
441 sample[i] = neighbors[sample[i]]; // Replace the neighbor index by the index of the point
442 ++hits_per_point[sample[i]]; // Increase the hit number of each selected point
443 }
444 ++hits_per_point[sample[sample_size - 2]]; // Increase the hit number of each selected point
445 }
446 // If the last layer (i.e., the layer with a single cell) has been chosen, do global sampling
447 // by PROSAC sampler.
448 else {
449 // last layer, all points are neighbors
450 // If local sampling
451 prosac_sampler.setSampleNumber(kth_sample_number);
452 prosac_sampler.generateSample (sample);
453 sample[sample_size - 1] = initial_point;
454 }
455 }

458 CV_Error(cv::Error::StsError, "Changing points size requires changing neighborhood graph! "
459 "You must reinitialize P-NAPSAC!");
460 }

469class NapsacSamplerImpl : public NapsacSampler {
470private:
471 const Ptr<NeighborhoodGraph> neighborhood_graph;
472 const Ptr<UniformRandomGenerator> random_generator;
473 bool do_uniform = false;
474 std::vector<int> points_large_neighborhood;
475 int points_large_neighborhood_size, points_size, sample_size;
476public:
477
478 NapsacSamplerImpl (int state, int points_size_, int sample_size_,
479 const Ptr<NeighborhoodGraph> &neighborhood_graph_) :
480 neighborhood_graph (neighborhood_graph_),
481 random_generator(UniformRandomGenerator::create(state, points_size_, sample_size_)) {
482
483 CV_Assert(points_size_ >= sample_size_);
484
485 points_size = points_size_;
486 sample_size = sample_size_;
487 points_large_neighborhood = std::vector<int>(points_size);
488
489 points_large_neighborhood_size = 0;
490
491 // find indices of points that have sufficient neighborhood (at least sample_size-1)
492 for (int pt_idx = 0; pt_idx < points_size; pt_idx++)
493 if ((int)neighborhood_graph->getNeighbors(pt_idx).size() >= sample_size-1)
494 points_large_neighborhood[points_large_neighborhood_size++] = pt_idx;
495
496 // if no points with sufficient neighborhood then do only uniform sampling
497 if (points_large_neighborhood_size == 0)
498 do_uniform = true;
499
500 // set random generator to generate random points of sample_size-1
501 random_generator->setSubsetSize(sample_size-1);
502 }
503
504 void generateSample (std::vector<int> &sample) override {
505 if (do_uniform)
506 // uniform sampling
507 random_generator->generateUniqueRandomSet(sample, points_size);
508 else {
509 // Take uniformly one initial point from points with sufficient neighborhood
510 int initial_point = points_large_neighborhood
511 [random_generator->getRandomNumber(points_large_neighborhood_size)];
512
513 const std::vector<int> &neighbors = neighborhood_graph->getNeighbors(initial_point);
514
515 // select random neighbors of initial point
516 random_generator->generateUniqueRandomSet(sample, (int)neighbors.size());
517 for (int i = 0; i < sample_size-1; i++)
518 sample[i] = neighbors[sample[i]];
519
520 // sample includes initial point too.
521 sample[sample_size-1] = initial_point;
522 }
523 }
524
525 void setNewPointsSize (int /*points_size_*/) override {
526 CV_Error(cv::Error::StsError, "Changing points size requires changing neighborhood graph!"
527 " You must reinitialize NAPSAC!");
528 }
