41KeyFrame* Map::addKeyframe(KeyFrame* kf)
42{
43 CV_Assert(kf);
44 if (kf->id < 0)
45 kf->id = impl->nextKfId++;
46 else if (kf->id >= impl->nextKfId)
47 impl->nextKfId = kf->id + 1;
48 impl->keyframes.insert(kf);
49 impl->kfIndex[kf->id] = kf;
50 return kf;
51}

87void Map::addObservation(KeyFrame* kf, size_t kpIdx, MapPoint* mp)
88{
89 CV_Assert(kf && mp);
90 CV_Assert(kpIdx < kf->mapPoints.size());
91 if (kf->mapPoints[kpIdx] != nullptr) return;
92 kf->mapPoints[kpIdx] = mp;
93 mp->observations[kf] = kpIdx;
94}

96void Map::removeObservation(KeyFrame* kf, MapPoint* mp)
97{
98 if (!kf || !mp) return;
99 auto it = mp->observations.find(kf);
100 if (it == mp->observations.end()) return;
101 size_t kpIdx = it->second;
102 if (kpIdx < kf->mapPoints.size())
103 kf->mapPoints[kpIdx] = nullptr;
104 mp->observations.erase(it);
105}

107void Map::removeMapPoint(MapPoint* mp)
108{
109 if (!mp) return;
110 for (auto& [kf, kpIdx] : mp->observations)
111 if (kpIdx < kf->mapPoints.size())
112 kf->mapPoints[kpIdx] = nullptr;
113 impl->mapPoints.erase(mp);
114 impl->mpIndex.erase(mp->id);
115 delete mp;
116}
