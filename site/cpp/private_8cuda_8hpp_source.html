133// NppStreamContext is introduced in NPP version 10100 included in CUDA toolkit 10.1 (CUDA_VERSION == 10010) however not all of the NPP functions called internally by OpenCV
134// - have an NppStreamContext argument (e.g. nppiHistogramEvenGetBufferSize_8u_C1R_Ctx in CUDA 12.3) and/or
135// - have a corresponding function in the supplied library (e.g. nppiEvenLevelsHost_32s_Ctx is not present in nppist.lib or libnppist.so as of CUDA 12.6)
136// Because support for these functions has gradually been introduced without being mentioned in the release notes this flag is set to a version of NPP (version 12205 included in CUDA toolkit 12.4) which is known to work.
137#define USE_NPP_STREAM_CTX NPP_VERSION >= 12205
138#if USE_NPP_STREAM_CTX
139 class NppStreamHandler
140 {
141 public:
142 inline explicit NppStreamHandler(cudaStream_t newStream)
143 {
144 nppStreamContext = {};
145 #if CUDA_VERSION < 12090
146 nppSafeCall(nppGetStreamContext(&nppStreamContext));
147 #else
148 int device = 0;
149 cudaSafeCall(cudaGetDevice(&device));
150
151 cudaDeviceProp prop{};
152 cudaSafeCall(cudaGetDeviceProperties(&prop, device));
153
154 nppStreamContext.nCudaDeviceId = device;
155 nppStreamContext.nMultiProcessorCount = prop.multiProcessorCount;
156 nppStreamContext.nMaxThreadsPerMultiProcessor = prop.maxThreadsPerMultiProcessor;
157 nppStreamContext.nMaxThreadsPerBlock = prop.maxThreadsPerBlock;
158 nppStreamContext.nSharedMemPerBlock = prop.sharedMemPerBlock;
159 nppStreamContext.nCudaDevAttrComputeCapabilityMajor = prop.major;
160 nppStreamContext.nCudaDevAttrComputeCapabilityMinor = prop.minor;
161 #endif
162 nppStreamContext.hStream = newStream;
163 cudaSafeCall(cudaStreamGetFlags(nppStreamContext.hStream, &nppStreamContext.nStreamFlags));
164 }
165
166 inline explicit NppStreamHandler(Stream& newStream) : NppStreamHandler(StreamAccessor::getStream(newStream)) {}
167
168 inline operator NppStreamContext() const {
169 return nppStreamContext;
170 }
171
172 inline NppStreamContext get() { return nppStreamContext; }
173
174 private:
175 NppStreamContext nppStreamContext;
176 };
177#else
178 class NppStreamHandler
179 {
180 public:
181 inline explicit NppStreamHandler(Stream& newStream)
182 {
183 oldStream = nppGetStream();
184 nppSafeSetStream(oldStream, StreamAccessor::getStream(newStream));
