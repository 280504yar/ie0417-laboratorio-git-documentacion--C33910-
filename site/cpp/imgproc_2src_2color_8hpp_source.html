274 bool createKernel(cv::String name, ocl::ProgramSource& source, cv::String options)
275 {
276 ocl::Device dev = ocl::Device::getDefault();
277 int pxPerWIy = dev.isIntel() && (dev.type() & ocl::Device::TYPE_GPU) ? 4 : 1;
278 int pxPerWIx = 1;
279
280 cv::String baseOptions = format("-D SRC_DEPTH=%d -D SCN=%d -D PIX_PER_WI_Y=%d ",
281 src.depth(), src.channels(), pxPerWIy);
282
283 switch (sizePolicy)
284 {
285 case TO_YUV:
286 if (dev.isIntel() &&
287 src.cols % 4 == 0 && src.step % 4 == 0 && src.offset % 4 == 0 &&
288 dst.step % 4 == 0 && dst.offset % 4 == 0)
289 {
290 pxPerWIx = 2;
291 }
292 globalSize[0] = (size_t)dst.cols/(2*pxPerWIx);
293 globalSize[1] = ((size_t)dst.rows/3 + pxPerWIy - 1) / pxPerWIy;
294 baseOptions += format("-D PIX_PER_WI_X=%d ", pxPerWIx);
295 break;
296 case FROM_YUV:
297 globalSize[0] = (size_t)dst.cols/2;
298 globalSize[1] = ((size_t)dst.rows/2 + pxPerWIy - 1) / pxPerWIy;
299 break;
300 case NONE:
301 default:
302 globalSize[0] = (size_t)src.cols;
303 globalSize[1] = ((size_t)src.rows + pxPerWIy - 1) / pxPerWIy;
304 break;
305 }
306
307 k.create(name.c_str(), source, baseOptions + options);
308
309 if(k.empty())
310 return false;
311
312 nArgs = k.set(0, ocl::KernelArg::ReadOnlyNoSize(src));
313 nArgs = k.set(nArgs, ocl::KernelArg::WriteOnly(dst));
314 return true;
315 }

317 bool run()
318 {
319 return k.run(2, globalSize, NULL, false);
320 }
