52void KalmanFilter::init(int DP, int MP, int CP, int type)
53{
54 CV_Assert( DP > 0 && MP > 0 );
55 CV_Assert( type == CV_32F || type == CV_64F );
56 CP = std::max(CP, 0);
57
58 statePre = Mat::zeros(DP, 1, type);
59 statePost = Mat::zeros(DP, 1, type);
60 transitionMatrix = Mat::eye(DP, DP, type);
61
62 processNoiseCov = Mat::eye(DP, DP, type);
63 measurementMatrix = Mat::zeros(MP, DP, type);
64 measurementNoiseCov = Mat::eye(MP, MP, type);
65
66 errorCovPre = Mat::zeros(DP, DP, type);
67 errorCovPost = Mat::zeros(DP, DP, type);
68 gain = Mat::zeros(DP, MP, type);
69
70 if( CP > 0 )
71 controlMatrix = Mat::zeros(DP, CP, type);
72 else
73 controlMatrix.release();
74
75 temp1.create(DP, DP, type);
76 temp2.create(MP, DP, type);
77 temp3.create(MP, MP, type);
78 temp4.create(MP, DP, type);
79 temp5.create(MP, 1, type);
80}

82const Mat& KalmanFilter::predict(const Mat& control)
83{
84 CV_INSTRUMENT_REGION();
85
86 // update the state: x'(k) = A*x(k)
87 statePre = transitionMatrix*statePost;
88
89 if( !control.empty() )
90 // x'(k) = x'(k) + B*u(k)
91 statePre += controlMatrix*control;
92
93 // update error covariance matrices: temp1 = A*P(k)
94 temp1 = transitionMatrix*errorCovPost;
95
96 // P'(k) = temp1*At + Q
97 gemm(temp1, transitionMatrix, 1, processNoiseCov, 1, errorCovPre, GEMM_2_T);
98
99 // handle the case when there will be no measurement before the next predict.
100 statePre.copyTo(statePost);
101 errorCovPre.copyTo(errorCovPost);
102
103 return statePre;
104}

106const Mat& KalmanFilter::correct(const Mat& measurement)
107{
108 CV_INSTRUMENT_REGION();
109
110 // temp2 = H*P'(k)
111 temp2 = measurementMatrix * errorCovPre;
112
113 // temp3 = temp2*Ht + R
114 gemm(temp2, measurementMatrix, 1, measurementNoiseCov, 1, temp3, GEMM_2_T);
115
116 // temp4 = inv(temp3)*temp2 = Kt(k)
117 solve(temp3, temp2, temp4, DECOMP_SVD);
118
119 // K(k)
120 gain = temp4.t();
121
122 // temp5 = z(k) - H*x'(k)
123 temp5 = measurement - measurementMatrix*statePre;
124
125 // x(k) = x'(k) + K(k)*temp5
126 statePost = statePre + gain*temp5;
127
128 // P(k) = P'(k) - K(k)*temp2
129 errorCovPost = errorCovPre - gain*temp2;
130
131 return statePost;
132}
