263Volume::Volume(VolumeType vtype, const VolumeSettings& settings)
264{
265 switch (vtype)
266 {
267 case VolumeType::TSDF:
268 this->impl = makePtr<TsdfVolume>(settings);
269 break;
270 case VolumeType::HashTSDF:
271 this->impl = makePtr<HashTsdfVolume>(settings);
272 break;
273 case VolumeType::ColorTSDF:
274 this->impl = makePtr<ColorTsdfVolume>(settings);
275 break;
276 case VolumeType::ColorHashTSDF:
277 this->impl = makePtr<ColorHashTsdfVolume>(settings);
278 break;
279 default:
280 CV_Error(Error::StsInternal, "Incorrect VolumeType");
281 break;
282 }
283}
