60void cv::matMulDeriv( InputArray A_, InputArray B_, OutputArray dABdA_, OutputArray dABdB_ )
61{
62 CV_INSTRUMENT_REGION();
63
64 Mat A = A_.getMat(), B = B_.getMat();
65 int type = A.type();
66 CV_Assert(type == B.type());

68 CV_Assert(A.cols == B.rows);

70 dABdA_.create(A.rows*B.cols, A.rows*A.cols, type);

74 int M = A.rows, L = A.cols, N = B.cols;

77 if( type == CV_32F )

79 for( int i = 0; i < M*N; i++ )
80 {
81 int j, i1 = i / N, i2 = i % N;

89 memset(dcdb, 0, L*N*sizeof(dcdb[0]));

91 for( j = 0; j < L; j++ )
92 {
93 dcda[i1*L + j] = b[j*bstep];

96 }
97 }

99 {

121void cv::Rodrigues(InputArray _src, OutputArray _dst, OutputArray _jacobian)
122{
123 CV_INSTRUMENT_REGION();
124
125 Mat src = _src.getMat();
126 const Size srcSz = src.size();
127 int srccn = src.channels();
128 int depth = src.depth();
129 CV_Check(srcSz, ((srcSz == Size(3, 1) || srcSz == Size(1, 3)) && srccn == 1) ||
130 (srcSz == Size(1, 1) && srccn == 3) ||
131 (srcSz == Size(3, 3) && srccn == 1),
132 "Input matrix must be 1x3 or 3x1 for a rotation vector, or 3x3 for a rotation matrix");
133
134 bool v2m = src.cols == 1 || src.rows == 1;
135 _dst.create(3, v2m ? 3 : 1, depth);
136 Mat dst = _dst.getMat(), jacobian;
137 if( _jacobian.needed() )
138 {
139 _jacobian.create(v2m ? Size(9, 3) : Size(3, 9), src.depth());
140 jacobian = _jacobian.getMat();
141 }
142
143 double J[27] = {0};
144 Mat matJ( 3, 9, CV_64F, J);
145
146 dst.setZero();
147
148 if( depth != CV_32F && depth != CV_64F )
149 CV_Error( cv::Error::StsUnsupportedFormat, "The matrices must have 32f or 64f data type" );
150
151 if( v2m )
152 {
153 int sstep = src.rows > 1 ? (int)src.step1() : 1;
154
155 Point3d r;
156 if( depth == CV_32F )
157 {
158 const float* sptr = src.ptr<float>();
159 r.x = sptr[0];
160 r.y = sptr[sstep];
161 r.z = sptr[sstep*2];
162 }
163 else
164 {
165 const double* sptr = src.ptr<double>();
166 r.x = sptr[0];
167 r.y = sptr[sstep];
168 r.z = sptr[sstep*2];
169 }
170
171 double theta = norm(r);
172 if( theta < DBL_EPSILON )
173 {
174 dst = Mat::eye(3, 3, depth);
175 if( jacobian.data )
176 {
177 memset( J, 0, sizeof(J) );
178 J[5] = J[15] = J[19] = -1;
179 J[7] = J[11] = J[21] = 1;
180 }
181 }
182 else
183 {
184 double c = std::cos(theta);
185 double s = std::sin(theta);
186 double c1 = 1. - c;
187 double itheta = theta ? 1./theta : 0.;
188
189 r *= itheta;
190
191 Matx33d rrt( r.x*r.x, r.x*r.y, r.x*r.z, r.x*r.y, r.y*r.y, r.y*r.z, r.x*r.z, r.y*r.z, r.z*r.z );
192 Matx33d r_x( 0, -r.z, r.y,
193 r.z, 0, -r.x,
194 -r.y, r.x, 0 );
195
196 // R = cos(theta)*I + (1 - cos(theta))*r*rT + sin(theta)*[r_x]
197 Matx33d R = c*Matx33d::eye() + c1*rrt + s*r_x;
198 R.convertTo(dst, depth);
199
200 if( jacobian.data )
201 {
202 const double I[] = { 1, 0, 0, 0, 1, 0, 0, 0, 1 };
203 double drrt[] = { r.x+r.x, r.y, r.z, r.y, 0, 0, r.z, 0, 0,
204 0, r.x, 0, r.x, r.y+r.y, r.z, 0, r.z, 0,
205 0, 0, r.x, 0, 0, r.y, r.x, r.y, r.z+r.z };
206 double d_r_x_[] = { 0, 0, 0, 0, 0, -1, 0, 1, 0,
207 0, 0, 1, 0, 0, 0, -1, 0, 0,
208 0, -1, 0, 1, 0, 0, 0, 0, 0 };
209 for( int i = 0; i < 3; i++ )
210 {
211 double ri = i == 0 ? r.x : i == 1 ? r.y : r.z;
212 double a0 = -s*ri, a1 = (s - 2*c1*itheta)*ri, a2 = c1*itheta;
213 double a3 = (c - s*itheta)*ri, a4 = s*itheta;
214 for( int k = 0; k < 9; k++ )
215 J[i*9+k] = a0*I[k] + a1*rrt.val[k] + a2*drrt[i*9+k] +
216 a3*r_x.val[k] + a4*d_r_x_[i*9+k];
217 }
218 }
219 }
220 }
221 else
222 {
223 Matx33d U, Vt;
224 Vec3d W;
225 double theta, s, c;
226 int dstep = dst.rows > 1 ? (int)dst.step1() : 1;
227
228 Matx33d R;
229 src.convertTo(R, CV_64F);
230
231 if( !checkRange(R, true, NULL, -100, 100) )
232 {
233 dst.setZero();
234 if (jacobian.data)
235 jacobian.setZero();
236 return;
237 }
238
239 SVD::compute(R, W, U, Vt);
240 R = U*Vt;
241
242 Point3d r(R(2, 1) - R(1, 2), R(0, 2) - R(2, 0), R(1, 0) - R(0, 1));
243
244 s = std::sqrt((r.x*r.x + r.y*r.y + r.z*r.z)*0.25);
245 c = (R(0, 0) + R(1, 1) + R(2, 2) - 1)*0.5;
246 c = c > 1. ? 1. : c < -1. ? -1. : c;
247 theta = std::acos(c);
248
249 if( s < 1e-5 )
250 {
251 double t;
252
253 if( c > 0 )
254 r = Point3d(0, 0, 0);
255 else
256 {
257 t = (R(0, 0) + 1)*0.5;
258 r.x = std::sqrt(MAX(t,0.));
259 t = (R(1, 1) + 1)*0.5;
260 r.y = std::sqrt(MAX(t,0.))*(R(0, 1) < 0 ? -1. : 1.);
261 t = (R(2, 2) + 1)*0.5;
262 r.z = std::sqrt(MAX(t,0.))*(R(0, 2) < 0 ? -1. : 1.);
263 if( fabs(r.x) < fabs(r.y) && fabs(r.x) < fabs(r.z) && (R(1, 2) > 0) != (r.y*r.z > 0) )
264 r.z = -r.z;
265 theta /= norm(r);
266 r *= theta;
267 }
268
269 if( jacobian.data )
270 {
271 memset( J, 0, sizeof(J) );
272 if( c > 0 )
273 {
274 J[5] = J[15] = J[19] = -0.5;
275 J[7] = J[11] = J[21] = 0.5;
276 }
277 }
278 }
279 else
280 {
281 double vth = 1/(2*s);
282
283 if( jacobian.data )
284 {
285 double t, dtheta_dtr = -1./s;
286 // var1 = [vth;theta]
287 // var = [om1;var1] = [om1;vth;theta]
288 double dvth_dtheta = -vth*c/s;
289 double d1 = 0.5*dvth_dtheta*dtheta_dtr;
290 double d2 = 0.5*dtheta_dtr;
291 // dvar1/dR = dvar1/dtheta*dtheta/dR = [dvth/dtheta; 1] * dtheta/dtr * dtr/dR
292 double dvardR[5*9] =
293 {
294 0, 0, 0, 0, 0, 1, 0, -1, 0,
295 0, 0, -1, 0, 0, 0, 1, 0, 0,
296 0, 1, 0, -1, 0, 0, 0, 0, 0,
297 d1, 0, 0, 0, d1, 0, 0, 0, d1,
298 d2, 0, 0, 0, d2, 0, 0, 0, d2
299 };
300 // var2 = [om;theta]
301 double dvar2dvar[] =
302 {
303 vth, 0, 0, r.x, 0,
304 0, vth, 0, r.y, 0,
305 0, 0, vth, r.z, 0,
306 0, 0, 0, 0, 1
307 };
308 double domegadvar2[] =
309 {
310 theta, 0, 0, r.x*vth,
311 0, theta, 0, r.y*vth,
312 0, 0, theta, r.z*vth
313 };
314
315 Mat _dvardR( 5, 9, CV_64FC1, dvardR );
316 Mat _dvar2dvar( 4, 5, CV_64FC1, dvar2dvar );
317 Mat _domegadvar2( 3, 4, CV_64FC1, domegadvar2 );
318 double t0[3*5];
319 Mat _t0( 3, 5, CV_64FC1, t0 );
320
321 gemm(_domegadvar2, _dvar2dvar, 1, noArray(), 0, _t0);
322 gemm(_t0, _dvardR, 1, noArray(), 0, matJ);
323 CV_Assert(matJ.ptr<double>() == J);
324
325 // transpose every row of matJ (treat the rows as 3x3 matrices)
326 CV_SWAP(J[1], J[3], t); CV_SWAP(J[2], J[6], t); CV_SWAP(J[5], J[7], t);
327 CV_SWAP(J[10], J[12], t); CV_SWAP(J[11], J[15], t); CV_SWAP(J[14], J[16], t);
328 CV_SWAP(J[19], J[21], t); CV_SWAP(J[20], J[24], t); CV_SWAP(J[23], J[25], t);
329 }
330
331 vth *= theta;
332 r *= vth;
333 }
334
335 if( depth == CV_32F )
336 {
337 float* dptr = dst.ptr<float>();
338 dptr[0] = (float)r.x;
339 dptr[dstep] = (float)r.y;
340 dptr[dstep*2] = (float)r.z;
341 }
342 else
343 {
344 double* dptr = dst.ptr<double>();
345 dptr[0] = r.x;
346 dptr[dstep] = r.y;
347 dptr[dstep*2] = r.z;
348 }
349 }
350
351 if( jacobian.data )
352 {
353 if( depth == CV_32F )
354 {
355 if( jacobian.rows == matJ.rows )
356 matJ.convertTo(jacobian, CV_32F);
357 else
358 {
359 float Jf[3*9];
360 Mat _Jf( matJ.rows, matJ.cols, CV_32FC1, Jf );
361 matJ.convertTo(_Jf, CV_32F);
362 transpose(_Jf, jacobian);
363 }
364 }
365 else if( jacobian.rows == matJ.rows )
366 matJ.copyTo(jacobian);
367 else
368 transpose(matJ, jacobian);
369 }
370}

1031
1032cv::Vec3d cv::RQDecomp3x3( InputArray _Marr,
1033 OutputArray _Rarr,
1034 OutputArray _Qarr,
1035 OutputArray _Qx,
1036 OutputArray _Qy,
1037 OutputArray _Qz )
1038{
1039 CV_INSTRUMENT_REGION();
1040
1041 Matx33d M, Q;
1042 double z, c, s;
1043 Mat Mmat = _Marr.getMat();
1044 int depth = Mmat.depth();
1045 Mmat.convertTo(M, CV_64F);
1046
1047 /* Find Givens rotation Q_x for x axis (left multiplication). */
1048 /*
1049 ( 1 0 0 )
1050 Qx = ( 0 c s ), c = m33/sqrt(m32^2 + m33^2), s = m32/sqrt(m32^2 + m33^2)
1051 ( 0 -s c )
1052 */
1053 s = std::abs(M(2, 1)) > DBL_EPSILON ? M(2, 1): 0.;
1054 c = std::abs(M(2, 1)) > DBL_EPSILON ? M(2, 2): 1.;
1055 z = 1./std::sqrt(c * c + s * s);
1056 c *= z;
1057 s *= z;
1058
1059 Matx33d Qx(1, 0, 0, 0, c, s, 0, -s, c);
1060 Matx33d R = M*Qx;
1061
1062 assert(fabs(R(2, 1)) < FLT_EPSILON);
1063 R(2, 1) = 0;
1064
1065 /* Find Givens rotation for y axis. */
1066 /*
1067 ( c 0 -s )
1068 Qy = ( 0 1 0 ), c = m33/sqrt(m31^2 + m33^2), s = -m31/sqrt(m31^2 + m33^2)
1069 ( s 0 c )
1070 */
1071 s = std::abs(R(2, 0)) > DBL_EPSILON ? -R(2, 0): 0.;
1072 c = std::abs(R(2, 0)) > DBL_EPSILON ? R(2, 2): 1.;
1073 z = 1./std::sqrt(c * c + s * s);
1074 c *= z;
1075 s *= z;
1076
1077 Matx33d Qy(c, 0, -s, 0, 1, 0, s, 0, c);
1078 M = R*Qy;
1079
1080 CV_Assert(fabs(M(2, 0)) < FLT_EPSILON);
1081 M(2, 0) = 0;
1082
1083 /* Find Givens rotation for z axis. */
1084 /*
1085 ( c s 0 )
1086 Qz = (-s c 0 ), c = m22/sqrt(m21^2 + m22^2), s = m21/sqrt(m21^2 + m22^2)
1087 ( 0 0 1 )
1088 */
1089
1090 s = std::abs(M(1, 0)) > DBL_EPSILON ? M(1, 0): 0.;
1091 c = std::abs(M(1, 0)) > DBL_EPSILON ? M(1, 1): 1.;
1092 z = 1./std::sqrt(c * c + s * s);
1093 c *= z;
1094 s *= z;
1095
1096 Matx33d Qz(c, s, 0, -s, c, 0, 0, 0, 1);
1097 R = M*Qz;
1098
1099 CV_Assert(fabs(R(1, 0)) < FLT_EPSILON);
1100 R(1, 0) = 0;
1101
1102 // Solve the decomposition ambiguity.
1103 // Diagonal entries of R, except the last one, shall be positive.
1104 // Further rotate R by 180 degree if necessary
1105 if( R(0, 0) < 0 )
1106 {
1107 if( R(1, 1) < 0 )
1108 {
1109 // rotate around z for 180 degree, i.e. a rotation matrix of
1110 // [-1, 0, 0],
1111 // [ 0, -1, 0],
1112 // [ 0, 0, 1]
1113 R(0, 0) *= -1;
1114 R(0, 1) *= -1;
1115 R(1, 1) *= -1;
1116
1117 Qz(0, 0) *= -1;
1118 Qz(0, 1) *= -1;
1119 Qz(1, 0) *= -1;
1120 Qz(1, 1) *= -1;
1121 }
1122 else
1123 {
1124 // rotate around y for 180 degree, i.e. a rotation matrix of
1125 // [-1, 0, 0],
1126 // [ 0, 1, 0],
1127 // [ 0, 0, -1]
1128 R(0, 0) *= -1;
1129 R(0, 2) *= -1;
1130 R(1, 2) *= -1;
1131 R(2, 2) *= -1;
1132
1133 Qz = Qz.t();
1134
1135 Qy(0, 0) *= -1;
1136 Qy(0, 2) *= -1;
1137 Qy(2, 0) *= -1;
1138 Qy(2, 2) *= -1;
1139 }
1140 }
1141 else if( R(1, 1) < 0 )
1142 {
1143 // ??? for some reason, we never get here ???
1144
1145 // rotate around x for 180 degree, i.e. a rotation matrix of
1146 // [ 1, 0, 0],
1147 // [ 0, -1, 0],
1148 // [ 0, 0, -1]
1149 R(0, 1) *= -1;
1150 R(0, 2) *= -1;
1151 R(1, 1) *= -1;
1152 R(1, 2) *= -1;
1153 R(2, 2) *= -1;
1154
1155 Qz = Qz.t();
1156 Qy = Qy.t();
1157
1158 Qx(1, 1) *= -1;
1159 Qx(1, 2) *= -1;
1160 Qx(2, 1) *= -1;
1161 Qx(2, 2) *= -1;
1162 }
1163
1164 // calculate the euler angle
1165 Vec3d eulerAngles(
1166 std::acos(Qx(1, 1)) * (Qx(1, 2) >= 0 ? 1 : -1) * (180.0 / CV_PI),
1167 std::acos(Qy(0, 0)) * (Qy(2, 0) >= 0 ? 1 : -1) * (180.0 / CV_PI),
1168 std::acos(Qz(0, 0)) * (Qz(0, 1) >= 0 ? 1 : -1) * (180.0 / CV_PI));
1169
1170 /* Calculate orthogonal matrix. */
1171 /*
1172 Q = QzT * QyT * QxT
1173 */
1174 M = Qz.t()*Qy.t();
1175 Q = M*Qx.t();
1176
1177 /* Save R and Q matrices. */
1178 R.convertTo(_Rarr, depth);
1179 Q.convertTo(_Qarr, depth);
1180
1181 if(_Qx.needed())
1182 Qx.convertTo(_Qx, depth);
1183 if(_Qy.needed())
1184 Qy.convertTo(_Qy, depth);
1185 if(_Qz.needed())
1186 Qz.convertTo(_Qz, depth);
1187 return eulerAngles;
1188}

1218void cv::findExtrinsicCameraParams2( const Mat& objectPoints,
1219 const Mat& imagePoints, const Mat& A,
1220 const Mat& distCoeffs, Mat& rvec, Mat& tvec,
1221 int useExtrinsicGuess )
1222{
1223 const int max_iter = 20;
1224 Mat matM, _m, _mn;
1225
1226 int i, count;
1227 double a[9], ar[9]={1,0,0,0,1,0,0,0,1}, R[9];
1228 double MM[9] = { 0 }, U[9] = { 0 }, V[9] = { 0 }, W[3] = { 0 };
1229 double param[6] = { 0 };
1230 Mat matA( 3, 3, CV_64F, a );
1231 Mat _Ar( 3, 3, CV_64F, ar );
1232 Mat matR( 3, 3, CV_64F, R );
1233 Mat _r( 3, 1, CV_64F, param );
1234 Mat _t( 3, 1, CV_64F, param + 3 );
1235 Mat _MM( 3, 3, CV_64F, MM );
1236 Mat matU( 3, 3, CV_64F, U );
1237 Mat matV( 3, 3, CV_64F, V );
1238 Mat matW( 3, 1, CV_64F, W );
1239 Mat _param( 6, 1, CV_64F, param );
1240 Mat _dpdr, _dpdt;
1241
1242 count = MAX(objectPoints.cols, objectPoints.rows);
1243 if (objectPoints.checkVector(3) > 0)
1244 objectPoints.convertTo(matM, CV_64F);
1245 else
1246 convertPointsFromHomogeneous(objectPoints, matM, CV_64F);
1247 if (imagePoints.checkVector(2) > 0)
1248 imagePoints.convertTo(_m, CV_64F);
1249 else
1250 convertPointsFromHomogeneous(imagePoints, _m, CV_64F);
1251 A.convertTo(matA, CV_64F);
1252
1253 CV_Assert((count >= 4) || (count == 3 && useExtrinsicGuess)); // it is unsafe to call LM optimisation without an extrinsic guess in the case of 3 points. This is because there is no guarantee that it will converge on the correct solution.
1254
1255 // normalize image points
1256 // (unapply the intrinsic matrix transformation and distortion)
1257 undistortPoints(_m, _mn, matA, distCoeffs, Mat(), _Ar);
1258
1259 if( useExtrinsicGuess )
1260 {
1261 CV_Assert((rvec.rows == 1 || rvec.cols == 1) && rvec.total()*rvec.channels() == 3);
1262 CV_Assert((tvec.rows == 1 || tvec.cols == 1) && tvec.total()*tvec.channels() == 3);
1263 Mat _r_temp(rvec.rows, rvec.cols, CV_MAKETYPE(CV_64F,rvec.channels()), param);
1264 Mat _t_temp(tvec.rows, tvec.cols, CV_MAKETYPE(CV_64F,tvec.channels()), param + 3);
1265 rvec.convertTo(_r_temp, CV_64F);
1266 tvec.convertTo(_t_temp, CV_64F);
1267 }
1268 else
1269 {
1270 Scalar Mc = mean(matM);
1271 Mat _Mc( 1, 3, CV_64F, Mc.val );
1272
1273 matM = matM.reshape(1, count);
1274 mulTransposed(matM, _MM, true, _Mc);
1275 SVDecomp(_MM, matW, noArray(), matV, SVD::MODIFY_A);
1276 CV_Assert(matW.ptr<double>() == W);
1277 CV_Assert(matV.ptr<double>() == V);
1278
1279 // initialize extrinsic parameters
1280 if( W[2]/W[1] < 1e-3)
1281 {
1282 // a planar structure case (all M's lie in the same plane)
1283 double tt[3];
1284 Mat R_transform = matV;
1285 Mat T_transform( 3, 1, CV_64F, tt );
1286
1287 if( V[2]*V[2] + V[5]*V[5] < 1e-10 )
1288 R_transform = Mat::eye(3, 3, CV_64F);
1289
1290 if( determinant(R_transform) < 0 )
1291 R_transform *= -1.;
1292
1293 gemm( R_transform, _Mc, -1, Mat(), 0, T_transform, GEMM_2_T );
1294
1295 const double* Rp = R_transform.ptr<double>();
1296 const double* Tp = T_transform.ptr<double>();
1297
1298 Mat _Mxy(count, 1, CV_64FC2);
1299 const double* src = (double*)matM.data;
1300 double* dst = (double*)_Mxy.data;
1301
1302 for( i = 0; i < count; i++, src += 3, dst += 2 )
1303 {
1304 dst[0] = Rp[0]*src[0] + Rp[1]*src[1] + Rp[2]*src[2] + Tp[0];
1305 dst[1] = Rp[3]*src[0] + Rp[4]*src[1] + Rp[5]*src[2] + Tp[1];
1306 }
1307
1308 Mat matH = findHomography(_Mxy, _mn);
1309
1310 if( checkRange(matH, true))
1311 {
1312 Mat _h1 = matH.col(0);
1313 Mat _h2 = matH.col(1);
1314 Mat _h3 = matH.col(2);
1315 double* h = matH.ptr<double>();
1316 CV_Assert(matH.isContinuous());
1317 double h1_norm = std::sqrt(h[0]*h[0] + h[3]*h[3] + h[6]*h[6]);
1318 double h2_norm = std::sqrt(h[1]*h[1] + h[4]*h[4] + h[7]*h[7]);
1319
1320 _h1 *= 1./MAX(h1_norm, DBL_EPSILON);
1321 _h2 *= 1./MAX(h2_norm, DBL_EPSILON);
1322 _t = _h3 * (2./MAX(h1_norm + h2_norm, DBL_EPSILON));
1323 _h1.cross(_h2).copyTo(_h3);
1324
1325 Rodrigues( matH, _r );
1326 Rodrigues( _r, matH );
1327 _t += matH*T_transform;
1328 matR = matH * R_transform;
1329 }
1330 else
1331 {
1332 setIdentity(matR);
1333 _t.setZero();
1334 }
1335
1336 Rodrigues( matR, _r );
1337 }
1338 else
1339 {
1340 // non-planar structure. Use DLT method
1341 CV_CheckGE(count, 6, "DLT algorithm needs at least 6 points for pose estimation from 3D-2D point correspondences.");
1342 double LL[12*12], LW[12], LV[12*12], sc;
1343 Mat _LL( 12, 12, CV_64F, LL );
1344 Mat _LW( 12, 1, CV_64F, LW );
1345 Mat _LV( 12, 12, CV_64F, LV );
1346 Point3d* M = (Point3d*)matM.data;
1347 Point2d* mn = (Point2d*)_mn.data;
1348
1349 Mat matL(2*count, 12, CV_64F);
1350 double* L = matL.ptr<double>();
1351
1352 for( i = 0; i < count; i++, L += 24 )
1353 {
1354 double x = -mn[i].x, y = -mn[i].y;
1355 L[0] = L[16] = M[i].x;
1356 L[1] = L[17] = M[i].y;
1357 L[2] = L[18] = M[i].z;
1358 L[3] = L[19] = 1.;
1359 L[4] = L[5] = L[6] = L[7] = 0.;
1360 L[12] = L[13] = L[14] = L[15] = 0.;
1361 L[8] = x*M[i].x;
1362 L[9] = x*M[i].y;
1363 L[10] = x*M[i].z;
1364 L[11] = x;
1365 L[20] = y*M[i].x;
1366 L[21] = y*M[i].y;
1367 L[22] = y*M[i].z;
1368 L[23] = y;
1369 }
1370
1371 mulTransposed( matL, _LL, true );
1372 SVDecomp( _LL, _LW, noArray(), _LV, SVD::MODIFY_A );
1373 Mat _RRt( 3, 4, CV_64F, LV + 11*12 );
1374 Mat _RR = _RRt.colRange(0, 3);
1375 Mat _tt = _RRt.col(3);
1376 if( determinant(_RR) < 0 )
1377 _RRt *= -1.0;
1378 sc = norm(_RR, NORM_L2);
1379 CV_Assert(fabs(sc) > DBL_EPSILON);
1380 SVDecomp(_RR, matW, matU, matV, SVD::MODIFY_A);
1381 matR = matU*matV;
1382 _tt.convertTo(_t, CV_64F, norm(matR, NORM_L2)/sc);
1383 Rodrigues(matR, _r);
1384 }
1385 }
1386
1387 matM = matM.reshape(3, 1);
1388 _mn = _mn.reshape(2, 1);
1389
1390 // refine extrinsic parameters using iterative algorithm
1391 auto callback = [matM, _m, matA, distCoeffs]
1392 (InputOutputArray param_, OutputArray _err, OutputArray _Jac) -> bool
1393 {
1394 const Mat& objpt = matM;
1395 const Mat& imgpt = _m;
1396 const Mat& cameraMatrix = matA;
1397 Mat x = param_.getMat();
1398 CV_Assert((x.cols == 1 || x.rows == 1) && x.total() == 6 && x.type() == CV_64F);
1399 double* pdata = x.ptr<double>();
1400 Mat rv(3, 1, CV_64F, pdata);
1401 Mat tv(3, 1, CV_64F, pdata + 3);
1402 int errCount = objpt.rows + objpt.cols - 1;
1403 _err.create(errCount * 2, 1, CV_64F);
1404 Mat err = _err.getMat();
1405 err = err.reshape(2, errCount);
1406 if (_Jac.needed())
1407 {
1408 _Jac.create(errCount * 2, 6, CV_64F);
1409 Mat Jac = _Jac.getMat();
1410 Mat dpdr = Jac.colRange(0, 3);
1411 Mat dpdt = Jac.colRange(3, 6);
1412 projectPoints(objpt, rv, tv, cameraMatrix, distCoeffs,
1413 err, dpdr, dpdt, noArray(), noArray(), noArray(), noArray());
1414 }
1415 else
1416 {
1417 projectPoints(objpt, rv, tv, cameraMatrix, distCoeffs, err);
1418 }
1419 err = err - imgpt;
1420 err = err.reshape(1, 2 * errCount);
1421 return true;
1422 };
1423
1424 LevMarq solver(_param, callback, LevMarq::Settings().setMaxIterations((unsigned int)max_iter).setGeodesic(true));
1425 solver.optimize();
1426
1427 _param.rowRange(0, 3).copyTo(rvec);
1428 _param.rowRange(3, 6).copyTo(tvec);
1429}

1480void cv::getUndistortRectangles(InputArray _cameraMatrix, InputArray _distCoeffs,
1481 InputArray R, InputArray newCameraMatrix, Size imgSize,
1482 Rect_<double>& inner, Rect_<double>& outer )
1483{
1484 const int N = 9;
1485 int x, y, k;
1486 Mat _pts(1, 4*(N-1), CV_64FC2);
1487 Point2d* pts = _pts.ptr<Point2d>();
1488
1489 // generate a grid of points across the image to estimate the distortion deformation
1490 double stepX = (imgSize.width - 1) / static_cast<double>(N - 1);
1491 double stepY = (imgSize.height - 1) / static_cast<double>(N - 1);
1492 for( y = k = 0; y < N; y++ )
1493 {
1494 for( x = 0; x < N; x++ )
1495 {
1496 if (x != 0 && x != N - 1 && y != 0 && y != N - 1)
1497 {
1498 // skip all points except those on the image border, because inner grid points
1499 // have no influence on the two deformation rectangles that are calculated below
1500 continue;
1501 }
1502 pts[k++] = Point2d(x * stepX, y * stepY);
1503 }
1504 }
1505
1506 undistortPoints(_pts, _pts, _cameraMatrix, _distCoeffs, R, newCameraMatrix);
1507
1508 double iX0=-FLT_MAX, iX1=FLT_MAX, iY0=-FLT_MAX, iY1=FLT_MAX;
1509 double oX0=FLT_MAX, oX1=-FLT_MAX, oY0=FLT_MAX, oY1=-FLT_MAX;
1510 // find the inscribed rectangle.
1511 // the code will likely not work with extreme rotation matrices (R) (>45%)
1512 for( y = k = 0; y < N; y++ )
1513 {
1514 for( x = 0; x < N; x++ )
1515 {
1516 if (x != 0 && x != N - 1 && y != 0 && y != N - 1)
1517 {
1518 continue;
1519 }
1520
1521 Point2d p = pts[k++];
1522 oX0 = MIN(oX0, p.x);
1523 oX1 = MAX(oX1, p.x);
1524 oY0 = MIN(oY0, p.y);
1525 oY1 = MAX(oY1, p.y);
1526
1527 if( x == 0 )
1528 iX0 = MAX(iX0, p.x);
1529 if( x == N-1 )
1530 iX1 = MIN(iX1, p.x);
1531 if( y == 0 )
1532 iY0 = MAX(iY0, p.y);
1533 if( y == N-1 )
1534 iY1 = MIN(iY1, p.y);
1535 }
1536 }
1537 inner = Rect_<double>(iX0, iY0, iX1-iX0, iY1-iY0);
1538 outer = Rect_<double>(oX0, oY0, oX1-oX0, oY1-oY0);
1539}

1541cv::Mat cv::getOptimalNewCameraMatrix( InputArray _cameraMatrix, InputArray _distCoeffs,
1542 Size imgSize, double alpha, Size newImgSize,
1543 Rect* validPixROI, bool centerPrincipalPoint )
1544{
1545 Rect_<double> inner, outer;
1546 newImgSize = newImgSize.width*newImgSize.height != 0 ? newImgSize : imgSize;
1547
1548 Mat cameraMatrix = _cameraMatrix.getMat(), M;
1549 cameraMatrix.convertTo(M, CV_64F);
1550 CV_Assert(M.isContinuous());
1551
1552 if( centerPrincipalPoint )
1553 {
1554 double cx0 = M.at<double>(0, 2);
1555 double cy0 = M.at<double>(1, 2);
1556 double cx = (newImgSize.width-1)*0.5;
1557 double cy = (newImgSize.height-1)*0.5;
1558
1559 getUndistortRectangles( _cameraMatrix, _distCoeffs, Mat(), cameraMatrix, imgSize, inner, outer );
1560 double s0 = std::max(std::max(std::max((double)cx/(cx0 - inner.x), (double)cy/(cy0 - inner.y)),
1561 (double)cx/(inner.x + inner.width - cx0)),
1562 (double)cy/(inner.y + inner.height - cy0));
1563 double s1 = std::min(std::min(std::min((double)cx/(cx0 - outer.x), (double)cy/(cy0 - outer.y)),
1564 (double)cx/(outer.x + outer.width - cx0)),
1565 (double)cy/(outer.y + outer.height - cy0));
1566 double s = s0*(1 - alpha) + s1*alpha;
