30double VideoCapture_LibRealsense::getProperty(int propIdx) const
31{
32 double propValue = CAP_PROP_UNKNOWN;
33
34 const int purePropIdx = propIdx & ~CAP_INTELPERC_GENERATORS_MASK;
35 if((propIdx & CAP_INTELPERC_GENERATORS_MASK) == CAP_INTELPERC_IMAGE_GENERATOR)
36 {
37 propValue = getImageGeneratorProperty(purePropIdx);
38 }
39 else if((propIdx & CAP_INTELPERC_GENERATORS_MASK) == CAP_INTELPERC_DEPTH_GENERATOR)
40 {
41 propValue = getDepthGeneratorProperty(purePropIdx);
42 }
43 else if((propIdx & CAP_INTELPERC_GENERATORS_MASK) == CAP_INTELPERC_IR_GENERATOR)
44 {
45 propValue = getIrGeneratorProperty(purePropIdx);
46 }
47 else
48 {
49 propValue = getCommonProperty(purePropIdx);
50 }
51
52 return propValue;
53}

193bool VideoCapture_LibRealsense::retrieveFrame(int outputType, cv::OutputArray frame)
194{
195 rs2::video_frame _frame(nullptr);
196 int type;
197 switch (outputType)
198 {
199 case CAP_INTELPERC_DEPTH_MAP:
200 _frame = mData.get_depth_frame().as<rs2::video_frame>();
201 type = CV_16UC1;
202 break;
203 case CAP_INTELPERC_IR_MAP:
204 _frame = mData.get_infrared_frame();
205 type = CV_8UC1;
206 break;
207 case CAP_INTELPERC_IMAGE:
208 _frame = mData.get_color_frame();
209 type = CV_8UC3;
210 break;
211 default:
212 return false;
213 }
214
215 try
216 {
217 // we copy the data straight away, so const_cast should be fine
218 void* data = const_cast<void*>(_frame.get_data());
219 Mat(_frame.get_height(), _frame.get_width(), type, data, _frame.get_stride_in_bytes()).copyTo(frame);
220
221 if(_frame.get_profile().format() == RS2_FORMAT_RGB8)
222 cvtColor(frame, frame, COLOR_RGB2BGR);
223 }
224 catch (const rs2::error&)
225 {
226 return false;
227 }
228
229 return true;
230}
