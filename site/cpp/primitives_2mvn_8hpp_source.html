26 struct MVNConfiguration {
27 std::vector<std::vector<std::size_t>> input_shapes;
28
29 /*
30 * [0, split_axis) = outer range
31 * [split_axis, -1] = inner range
32 *
33 * for each location in the outer range, all the values in the inner range are normalized as a group
34 */
35 std::size_t split_axis;
36
37 /* The group (described above) is centered always. The following parameter controls whether the variance
38 * is also normalized.
39 */
40 bool normalize_variance;
41 float epsilon;
42 };

44 template <class T>
45 class MVNOp final : public CUDABackendNode {
46 public:
47 using wrapper_type = GetCUDABackendWrapperType<T>;
48
49 MVNOp(csl::Stream stream_, const MVNConfiguration& config)
50 : stream(std::move(stream_))
51 {
52 split_axis = config.split_axis;
53 normalize_variance = config.normalize_variance;
54 epsilon = config.epsilon;
55
56 std::size_t max_outer_size = 0;
57 const auto& input_shapes = config.input_shapes;
58 for (int i = 0; i < input_shapes.size(); i++)
59 {
60 std::size_t outer_size = 1;
61 for (int j = 0; j < split_axis; j++)
62 outer_size *= input_shapes[i][j];
63 max_outer_size = std::max(max_outer_size, outer_size);
64 }
65
66 csl::WorkspaceBuilder builder;
67 builder.require<float>(max_outer_size);
68 if (normalize_variance)
69 builder.require<float>(max_outer_size);
70 scratch_mem_in_bytes = builder.required_workspace_size();
71 }
72
73 void forward(
74 const std::vector<cv::Ptr<BackendWrapper>>& inputs,
75 const std::vector<cv::Ptr<BackendWrapper>>& outputs,
76 csl::Workspace& workspace) override
77 {
78 CV_Assert(inputs.size() == outputs.size());
79
80 for (int i = 0; i < inputs.size(); i++)
81 {
82 auto input_wrapper = inputs[i].dynamicCast<wrapper_type>();
83 auto input = input_wrapper->getView();
84
85 auto output_wrapper = outputs[i].dynamicCast<wrapper_type>();
86 auto output = output_wrapper->getSpan();
87
88 auto outer_size = input.size_range(0, split_axis);
89 auto inner_size = input.size_range(split_axis, input.rank());
90 if (inner_size == 1)
91 {
92 kernels::fill<T>(stream, output, 0.0f);
93 return;
94 }
95 else
96 {
97 auto ws_allocator = csl::WorkspaceAllocator(workspace);
98
99 auto means = ws_allocator.get_span<float>(outer_size);
100 kernels::fill<float>(stream, means, 0);
101
102 if (normalize_variance)
103 {
104 auto scales = ws_allocator.get_span<float>(outer_size);
105 kernels::fill<float>(stream, scales, 0);
106
107 kernels::reduce_mean_sqr_sum<T>(stream, means, scales, input, inner_size);
108 kernels::compute_normalization_scale(stream, scales, means, scales, inner_size, epsilon);
109 kernels::normalize_mean_variance<T>(stream, output, input, means, scales, inner_size);
110 }
111 else
112 {
113 kernels::reduce_mean<T>(stream, means, input, inner_size);
114 kernels::normalize_mean<T>(stream, output, input, means, inner_size);
115 }
116 }
117 }
118 }
119
120 std::size_t get_workspace_memory_in_bytes() const noexcept override { return scratch_mem_in_bytes; }
121
122 private:
123 csl::Stream stream;
124
125 bool normalize_variance;
126 float epsilon;
127 std::size_t split_axis;
128
129 std::size_t scratch_mem_in_bytes;
130 };
