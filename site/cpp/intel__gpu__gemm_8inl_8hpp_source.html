30static bool intel_gpu_gemm(
31 UMat A, Size sizeA,
32 UMat B, Size sizeB,
33 UMat D, Size sizeD,
34 double alpha, double beta,
35 bool atrans, bool btrans,
36 bool& isPropagatedC2D
37)
38{
39 CV_UNUSED(sizeB);
40
41 int M = sizeD.height, N = sizeD.width, K = ((atrans)? sizeA.height : sizeA.width);
42
43 if (M < 4 || N < 4 || K < 4) // vload4
44 return false;
45
46 CV_LOG_VERBOSE(NULL, 0, "M=" << M << " N=" << N << " K=" << K);
47
48 std::string kernelName;
49
50 unsigned int lx = 8, ly = 4;
51 unsigned int dx = 4, dy = 8;
52
53 if(!atrans && !btrans)
54 {
55 if (M % 32 == 0 && N % 32 == 0 && K % 16 == 0)
56 {
57 kernelName = "intelblas_gemm_buffer_NN_sp";
58 }
59 else
60 {
61 if (M % 2 != 0)
62 return false;
63 // vload4(0, dst_write0) - 4 cols
64 // multiply by lx: 8
65 if (N % (4*8) != 0)
66 return false;
67 kernelName = "intelblas_gemm_buffer_NN";
68 }
69 }
70 else if(atrans && !btrans)
71 {
72 if (M % 32 != 0)
73 return false;
74 if (N % 32 != 0)
75 return false;
76 kernelName = "intelblas_gemm_buffer_TN";
77 }
78 else if(!atrans && btrans)
79 {
80 if (K % 4 != 0)
81 return false;
82 kernelName = "intelblas_gemm_buffer_NT";
83 ly = 16;
84 dx = 1;
85 }
86 else
87 {
88 if (M % 32 != 0)
89 return false;
90 if (N % 32 != 0)
91 return false;
92 if (K % 16 != 0)
93 return false;
94 kernelName = "intelblas_gemm_buffer_TT";
95 }
96
97 CV_LOG_DEBUG(NULL, "kernel: " << kernelName << " (M=" << M << " N=" << N << " K=" << K << ")");
98
99 const size_t gx = divUp((size_t)N, dx);
100 const size_t gy = divUp((size_t)M, dy);
101
102 size_t local[] = {lx, ly, 1};
103 size_t global[] = {roundUp(gx, lx), roundUp(gy, ly), 1};
104
105 ocl::Queue q;
106 String errmsg;
107 const ocl::Program program = ocl::Context::getDefault().getProg(ocl::core::intel_gemm_oclsrc, "", errmsg);
108
109 if(!atrans && btrans)
110 {
111 ocl::Kernel k(kernelName.c_str(), program);
112 if (k.empty())
113 {
114 return false;
115 }
116
117 k.args(ocl::KernelArg::PtrReadOnly(A),
118 (int) (A.offset / sizeof(float)),
119 ocl::KernelArg::PtrReadOnly(B),
120 (int) (B.offset / sizeof(float)),
121 ocl::KernelArg::PtrWriteOnly(D),
122 (int) (D.offset / sizeof(float)),
123 M, N, K,
124 (float)alpha,
125 (float)beta,
126 (int)(A.step / sizeof(float)),
127 (int)(B.step / sizeof(float)),
128 (int)(D.step / sizeof(float))
129 );
130
131 bool ret = k.run(2, global, local, false, q);
132 return ret;
133 }
134 else
135 {
136 int stride = (M * N < 1024 * 1024) ? 10000000 : 256;
137
138 for(int start_index = 0; start_index < K; start_index += stride)
139 {
140 ocl::Kernel k(kernelName.c_str(), program);
141 k.args(ocl::KernelArg::PtrReadOnly(A),
142 (int) (A.offset / sizeof(float)),
143 ocl::KernelArg::PtrReadOnly(B),
144 (int) (B.offset / sizeof(float)),
145 ocl::KernelArg::PtrWriteOnly(D),
146 (int) (D.offset / sizeof(float)),
147 M, N, K,
148 (float)alpha,
149 (float)beta,
150 (int)(A.step / sizeof(float)),
151 (int)(B.step / sizeof(float)),
152 (int)(D.step / sizeof(float)),
153 (int) start_index, // 14 start_index
154 stride);
155
156 bool ret = k.run(2, global, local, false, q);
157 if (!ret)
158 {
159 if (start_index != 0)
160 isPropagatedC2D = false; // D array content is changed, need to rewrite
161 return false;
162 }
163 }
164 return true;
165 }
166}
