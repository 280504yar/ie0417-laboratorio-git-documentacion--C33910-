37int normHamming(const uchar* a, int n)
38{
39 CV_AVX_GUARD;
40
41 int i = 0;
42 int result = 0;
43
44#if (CV_SIMD || CV_SIMD_SCALABLE)
45 {
46 v_uint64 t = vx_setzero_u64();
47 for (; i <= n - VTraits<v_uint8>::vlanes(); i += VTraits<v_uint8>::vlanes())
48 t = v_add(t, v_popcount(v_reinterpret_as_u64(vx_load(a + i))));
49 result = (int)v_reduce_sum(t);
50 vx_cleanup();
51 }
52#endif
53
54#if CV_POPCNT
55 {
56# if defined CV_POPCNT_U64
57 for(; i <= n - 8; i += 8)
58 {
59 uint64_t val;
60 std::memcpy(&val, a + i, sizeof(val));
61 result += (int)CV_POPCNT_U64(val);
62 }
63# endif
64 for(; i <= n - 4; i += 4)
65 {
66 uint32_t val;
67 std::memcpy(&val, a + i, sizeof(val));
68 result += CV_POPCNT_U32(val);
69 }
70 }
71#endif
72#if CV_ENABLE_UNROLLED
73 for(; i <= n - 4; i += 4)
74 {
75 result += popCountTable[a[i]] + popCountTable[a[i+1]] +
76 popCountTable[a[i+2]] + popCountTable[a[i+3]];
77 }
78#endif
79 for(; i < n; i++)
80 {
81 result += popCountTable[a[i]];
82 }
83 return result;
84}

86int normHamming(const uchar* a, const uchar* b, int n)
87{
88 CV_AVX_GUARD;
89
90 int i = 0;
91 int result = 0;
92
93#if (CV_SIMD || CV_SIMD_SCALABLE)
94 {
95 v_uint64 t = vx_setzero_u64();
96 for (; i <= n - VTraits<v_uint8>::vlanes(); i += VTraits<v_uint8>::vlanes())
97 t = v_add(t, v_popcount(v_reinterpret_as_u64(v_xor(vx_load(a + i), vx_load(b + i)))));
98 result += (int)v_reduce_sum(t);
99 }
100#endif
101
102#if CV_POPCNT
103 {
104# if defined CV_POPCNT_U64
105 for(; i <= n - 8; i += 8)
106 {
107 uint64_t val_a, val_b;
108 std::memcpy(&val_a, a + i, sizeof(val_a));
109 std::memcpy(&val_b, b + i, sizeof(val_b));
110 result += (int)CV_POPCNT_U64(val_a ^ val_b);
111 }
112# endif
113 for(; i <= n - 4; i += 4)
114 {
115 uint32_t val_a, val_b;
116 std::memcpy(&val_a, a + i, sizeof(val_a));
117 std::memcpy(&val_b, b + i, sizeof(val_b));
118 result += (int)CV_POPCNT_U32(val_a ^ val_b);
119 }
120 }
121#endif
122#if CV_ENABLE_UNROLLED
123 for(; i <= n - 4; i += 4)
124 {
125 result += popCountTable[a[i] ^ b[i]] + popCountTable[a[i+1] ^ b[i+1]] +
126 popCountTable[a[i+2] ^ b[i+2]] + popCountTable[a[i+3] ^ b[i+3]];
127 }
128#endif
129 for(; i < n; i++)
130 {
131 result += popCountTable[a[i] ^ b[i]];
132 }
133 return result;
134}
