52typedef struct {
53 int valid; /* indicate which fields are valid */
54 char programtype[16]; /* listed at beginning of file to identify it
55 * after "#?". defaults to "RGBE" */
56 float gamma; /* image has already been gamma corrected with
57 * given gamma. defaults to 1.0 (no correction) */
58 float exposure; /* a value of 1.0 in an image corresponds to
59 * <exposure> watts/steradian/m^2.
60 * defaults to 1.0 */
61} rgbe_header_info;
62
63/* flags indicating which fields in an rgbe_header_info are valid */
64#define RGBE_VALID_PROGRAMTYPE 0x01
65#define RGBE_VALID_GAMMA 0x02
66#define RGBE_VALID_EXPOSURE 0x04
67
68/* return codes for rgbe routines */
69#define RGBE_RETURN_SUCCESS 0
70#define RGBE_RETURN_FAILURE -1
