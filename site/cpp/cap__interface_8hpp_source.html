35public:
36 struct VideoParameter {
37 VideoParameter() = default;
38
39 VideoParameter(int key_, int value_) : key(key_), value(value_) {}
40
41 int key{-1};
42 int value{-1};
43 mutable bool isConsumed{false};
44 };
45
46 VideoParameters() = default;
47
48 explicit VideoParameters(const std::vector<int>& params)
49 {
50 const auto count = params.size();
51 if (count % 2 != 0)
52 {
53 CV_Error_(Error::StsVecLengthErr,
54 ("Vector of VideoWriter parameters should have even length"));
55 }
56 params_.reserve(count / 2);
57 for (std::size_t i = 0; i < count; i += 2)
58 {
59 add(params[i], params[i + 1]);
60 }
61 }
62
63 VideoParameters(int* params, unsigned n_params)
64 {
65 params_.reserve(n_params);
66 for (unsigned i = 0; i < n_params; ++i)
67 {
68 add(params[2*i], params[2*i + 1]);
69 }
70 }
71
72 void add(int key, int value)
73 {
74 params_.emplace_back(key, value);
75 }
76
77 bool has(int key) const
78 {
79 auto it = std::find_if(params_.begin(), params_.end(),
80 [key](const VideoParameter &param)
81 {
82 return param.key == key;
83 }
84 );
85 return it != params_.end();
86 }
87
88 template <class ValueType>
89 ValueType get(int key) const
90 {
91 auto it = std::find_if(params_.begin(), params_.end(),
92 [key](const VideoParameter &param)
93 {
94 return param.key == key;
95 }
96 );
97 if (it != params_.end())
98 {
99 it->isConsumed = true;
100 return castParameterTo<ValueType>(it->value);
101 }
102 else
103 {
104 CV_Error_(Error::StsBadArg, ("Missing value for parameter: [%d]", key));
105 }
106 }
107
108 template <class ValueType>
109 ValueType get(int key, ValueType defaultValue) const
110 {
111 auto it = std::find_if(params_.begin(), params_.end(),
112 [key](const VideoParameter &param)
113 {
114 return param.key == key;
115 }
116 );
117 if (it != params_.end())
118 {
119 it->isConsumed = true;
120 return castParameterTo<ValueType>(it->value);
121 }
122 else
123 {
124 return defaultValue;
125 }
126 }
127
128 std::vector<int> getUnused() const
129 {
130 std::vector<int> unusedParams;
131 for (const auto &param : params_)
132 {
133 if (!param.isConsumed)
134 {
135 unusedParams.push_back(param.key);
136 }
137 }
138 return unusedParams;
139 }
140
141 std::vector<int> getIntVector() const
142 {
143 std::vector<int> vint_params;
144 for (const auto& param : params_)
145 {
146 vint_params.push_back(param.key);
147 vint_params.push_back(param.value);
148 }
149 return vint_params;
150 }
151
152 bool empty() const
153 {
154 return params_.empty();
155 }
156
157 bool warnUnusedParameters() const
158 {
159 bool found = false;
160 for (const auto &param : params_)
161 {
162 if (!param.isConsumed)
163 {
164 found = true;
165 CV_LOG_INFO(NULL, "VIDEOIO: unused parameter: [" << param.key << "]=" <<
166 cv::format("%lld / 0x%016llx", (long long)param.value, (long long)param.value));
167 }
168 }
169 return found;
170 }

222// Advanced base class for VideoCapture backends providing some extra functionality
223class VideoCaptureBase : public IVideoCapture
224{
225public:
226 VideoCaptureBase() : autorotate(true) {}
227 double getProperty(int propId) const CV_OVERRIDE
228 {
229 switch(propId)
230 {
231 case cv::CAP_PROP_ORIENTATION_AUTO:
232 return static_cast<double>(autorotate);
233
234 case cv::CAP_PROP_FRAME_WIDTH:
235 return shouldSwapWidthHeight() ? getProperty_(cv::CAP_PROP_FRAME_HEIGHT) : getProperty_(cv::CAP_PROP_FRAME_WIDTH);
236
237 case cv::CAP_PROP_FRAME_HEIGHT:
238 return shouldSwapWidthHeight() ? getProperty_(cv::CAP_PROP_FRAME_WIDTH) : getProperty_(cv::CAP_PROP_FRAME_HEIGHT);
239
240 default:
241 return getProperty_(propId);
242 }
243 }
244 bool setProperty(int propId, double value) CV_OVERRIDE
245 {
246 switch(propId)
247 {
248 case cv::CAP_PROP_ORIENTATION_AUTO:
249 autorotate = (value != 0);
250 return true;
251
252 default:
253 return setProperty_(propId, value);
254 }
255 }
256 bool retrieveFrame(int channel, OutputArray image) CV_OVERRIDE
257 {
258 const bool res = retrieveFrame_(channel, image);
259 if (res)
260 applyMetadataRotation(image);
261 return res;
262 }
263
264protected:
265 virtual double getProperty_(int) const = 0;
266 virtual bool setProperty_(int, double) = 0;
267 virtual bool retrieveFrame_(int, OutputArray) = 0;
268
269protected:
270 bool shouldSwapWidthHeight() const
271 {
272 if (!autorotate)
273 return false;
274 int rotation = static_cast<int>(getProperty(cv::CAP_PROP_ORIENTATION_META));
275 return std::abs(rotation % 180) == 90;
276 }
277 void applyMetadataRotation(OutputArray mat) const
278 {
279 bool rotation_auto = 0 != getProperty(CAP_PROP_ORIENTATION_AUTO);
280 int rotation_angle = static_cast<int>(getProperty(CAP_PROP_ORIENTATION_META));
281 if(!rotation_auto || rotation_angle%360 == 0)
282 {
283 return;
284 }
285 cv::RotateFlags flag;
286 if(rotation_angle == 90 || rotation_angle == -270) { // Rotate clockwise 90 degrees
287 flag = cv::ROTATE_90_CLOCKWISE;
288 } else if(rotation_angle == 270 || rotation_angle == -90) { // Rotate clockwise 270 degrees
289 flag = cv::ROTATE_90_COUNTERCLOCKWISE;
290 } else if(rotation_angle == 180 || rotation_angle == -180) { // Rotate clockwise 180 degrees
291 flag = cv::ROTATE_180;
292 } else { // Unsupported rotation
293 return;
294 }
295 cv::rotate(mat, mat, flag);
296 }
297
298protected:
299 bool autorotate;
300};
