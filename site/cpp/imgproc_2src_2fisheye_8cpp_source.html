11void initUndistortRectifyMap( InputArray K, InputArray D, InputArray R, InputArray P,
12 const cv::Size& size, int m1type, OutputArray map1, OutputArray map2 )
13{
14 CV_INSTRUMENT_REGION();
15
16 CV_Assert( m1type == CV_16SC2 || m1type == CV_32F || m1type <=0 );
17 map1.create( size, m1type <= 0 ? CV_16SC2 : m1type );
18 map2.create( size, map1.type() == CV_16SC2 ? CV_16UC1 : CV_32F );
19
20 CV_Assert((K.depth() == CV_32F || K.depth() == CV_64F) && (D.depth() == CV_32F || D.depth() == CV_64F));
21 CV_Assert((P.empty() || P.depth() == CV_32F || P.depth() == CV_64F) && (R.empty() || R.depth() == CV_32F || R.depth() == CV_64F));
22 CV_Assert(K.size() == Size(3, 3) && (D.empty() || D.total() == 4));
23 CV_Assert(R.empty() || R.size() == Size(3, 3) || R.total() * R.channels() == 3);
24 CV_Assert(P.empty() || P.size() == Size(3, 3) || P.size() == Size(4, 3));
25
26 Vec2d f, c;
27 if (K.depth() == CV_32F)
28 {
29 Matx33f camMat = K.getMat();
30 f = Vec2f(camMat(0, 0), camMat(1, 1));
31 c = Vec2f(camMat(0, 2), camMat(1, 2));
32 }
33 else
34 {
35 Matx33d camMat = K.getMat();
36 f = Vec2d(camMat(0, 0), camMat(1, 1));
37 c = Vec2d(camMat(0, 2), camMat(1, 2));
38 }
39
40 Vec4d k = Vec4d::all(0);
41 if (!D.empty())
42 k = D.depth() == CV_32F ? (Vec4d)*D.getMat().ptr<Vec4f>(): *D.getMat().ptr<Vec4d>();
43
44 Matx33d RR = Matx33d::eye();
45 if (!R.empty() && R.total() * R.channels() == 3)
46 {
47 Vec3d rvec;
48 R.getMat().convertTo(rvec, CV_64F);
49 RR = Affine3d(rvec).rotation();
50 }
51 else if (!R.empty() && R.size() == Size(3, 3))
52 R.getMat().convertTo(RR, CV_64F);
53
54 Matx33d PP = Matx33d::eye();
55 if (!P.empty())
56 P.getMat().colRange(0, 3).convertTo(PP, CV_64F);
57
58 Matx33d iR = (PP * RR).inv(cv::DECOMP_SVD);
59
60 for( int i = 0; i < size.height; ++i)
61 {
62 float* m1f = map1.getMat().ptr<float>(i);
63 float* m2f = map2.getMat().ptr<float>(i);
64 short* m1 = (short*)m1f;
65 ushort* m2 = (ushort*)m2f;
66
67 double _x = i*iR(0, 1) + iR(0, 2),
68 _y = i*iR(1, 1) + iR(1, 2),
69 _w = i*iR(2, 1) + iR(2, 2);
70
71 for( int j = 0; j < size.width; ++j)
72 {
73 double u, v;
74 if( _w <= 0)
75 {
76 u = (_x > 0) ? -std::numeric_limits<double>::infinity() : std::numeric_limits<double>::infinity();
77 v = (_y > 0) ? -std::numeric_limits<double>::infinity() : std::numeric_limits<double>::infinity();
78 }
79 else
80 {
81 double x = _x/_w, y = _y/_w;
82
83 double r = sqrt(x*x + y*y);
84 double theta = std::atan(r);
85
86 double theta2 = theta*theta, theta4 = theta2*theta2, theta6 = theta4*theta2, theta8 = theta4*theta4;
87 double theta_d = theta * (1 + k[0]*theta2 + k[1]*theta4 + k[2]*theta6 + k[3]*theta8);
88
89 double scale = (r == 0) ? 1.0 : theta_d / r;
90 u = f[0]*x*scale + c[0];
91 v = f[1]*y*scale + c[1];
92 }
93
94 if( m1type == CV_16SC2 )
95 {
96 int iu = cv::saturate_cast<int>(u*static_cast<double>(cv::INTER_TAB_SIZE));
97 int iv = cv::saturate_cast<int>(v*static_cast<double>(cv::INTER_TAB_SIZE));
98 m1[j*2+0] = (short)(iu >> cv::INTER_BITS);
99 m1[j*2+1] = (short)(iv >> cv::INTER_BITS);
100 m2[j] = (ushort)((iv & (cv::INTER_TAB_SIZE-1))*cv::INTER_TAB_SIZE + (iu & (cv::INTER_TAB_SIZE-1)));
101 }
102 else if( m1type == CV_32FC1 )
103 {
104 m1f[j] = (float)u;
105 m2f[j] = (float)v;
106 }
107
108 _x += iR(0, 0);
109 _y += iR(1, 0);
110 _w += iR(2, 0);
111 }
112 }
113}

115void undistortImage(InputArray distorted, OutputArray undistorted,
116 InputArray K, InputArray D, InputArray Knew, const Size& new_size)
117{
118 CV_INSTRUMENT_REGION();

120 Size size = !new_size.empty() ? new_size : distorted.size();

122 Mat map1, map2;
123 fisheye::initUndistortRectifyMap(K, D, Matx33d::eye(), Knew, size, CV_16SC2, map1, map2 );

125}
