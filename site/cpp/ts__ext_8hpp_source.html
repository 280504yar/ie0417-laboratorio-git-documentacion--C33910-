214#define CV_TEST_EXPECT_EXCEPTION_MESSAGE(statement, msg) \
215 GTEST_AMBIGUOUS_ELSE_BLOCKER_ \
216 if (::testing::internal::AlwaysTrue()) { \
217 const char* msg_ = msg; \
218 bool hasException = false; \
219 try { \
220 GTEST_SUPPRESS_UNREACHABLE_CODE_WARNING_BELOW_(statement); \
221 } \
222 catch (const cv::Exception& e) { \
223 if (NULL == strstr(e.what(), msg_)) \
224 ADD_FAILURE() << "Unexpected cv::Exception is raised: " << #statement << "\n Expected message substring: '" << msg_ << "'. Actual message:\n" << e.what(); \
225 hasException = true; \
226 } \
227 catch (const std::exception& e) { \
228 ADD_FAILURE() << "Unexpected std::exception is raised: " << #statement << "\n" << e.what(); \
229 hasException = true; \
230 } \
231 catch (...) { \
232 ADD_FAILURE() << "Unexpected C++ exception is raised: " << #statement; \
233 hasException = true; \
234 } \
235 if (!hasException) { \
236 goto GTEST_CONCAT_TOKEN_(gtest_label_test_, __LINE__); \
237 } \
238 } else \
239 GTEST_CONCAT_TOKEN_(gtest_label_test_, __LINE__): \
240 ADD_FAILURE() << "Failed: Expected: " #statement " throws an '" << msg << "' exception.\n" \
241 " Actual: it doesn't."
