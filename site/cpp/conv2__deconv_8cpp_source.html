14static void deconvBlock32f(const void* inp__, const void* /*residual*/,
15 void* out__, const ConvState& cs,
16 const void* weights__, const float* /*scale*/,
17 const float* bias__)
18{
19 CV_Assert(cs.inpshape.layout == DATA_LAYOUT_BLOCK);
20 CV_Assert(cs.outshape.layout == DATA_LAYOUT_BLOCK);
21 CV_Assert(cs.wshape.dims == 5);
22
23 enum { MAX_DIMS = ConvState::MAX_CONV_DIMS };
24 const int sdims = cs.nspatialdims;
25 const int C0 = cs.inpshape.back();
26 const int K0 = C0;
27
28 const int N = cs.inpshape[0];
29 const int C1 = cs.inpshape[1];
30 const int K1 = cs.outshape[1];
31 const int C = cs.inpshape.C;
32 const int K = cs.outshape.C;
33 const int ngroups = cs.ngroups;
34 const int Kg = K / ngroups;
35 const int Cg = C / ngroups;
36 const int Kblk = cs.wshape[1];
37 const int ksize = cs.wshape[2];
38 const int C1Max = cs.wshape[3];
39
40 int ispatial = 1, ospatial = 1;
41 for (int i = 0; i < sdims; i++) {
42 ispatial *= cs.inpshape[2 + i];
43 ospatial *= cs.outshape[2 + i];
44 }
45
46 int oDims[MAX_DIMS], iDims[MAX_DIMS];
47 for (int i = 0; i < MAX_DIMS; i++) oDims[i] = iDims[i] = 1;
48 for (int i = 0; i < sdims; i++) {
49 oDims[MAX_DIMS - sdims + i] = cs.outshape[2 + i];
50 iDims[MAX_DIMS - sdims + i] = cs.inpshape[2 + i];
51 }
52
53 std::vector<std::array<int, MAX_DIMS>> kcoords_tab(ksize);
54 for (int ks = 0; ks < ksize; ks++) {
55 int ktmp = ks;
56 for (int i = sdims - 1; i >= 0; i--) {
57 int di = MAX_DIMS - sdims + i;
58 kcoords_tab[ks][di] = ktmp % cs.kshape[di];
59 ktmp /= cs.kshape[di];
60 }
61 }
62
63 const float* wdata = (const float*)weights__;
64 const float* bias = bias__;
65
66#if (CV_SIMD || CV_SIMD_SCALABLE)
67 // SIMD path is safe when ngroups==1 or Kg%C0==0; repackDeconvWeights()
68 // zero-fills padded lanes.
69 const bool simd_ok =
70 ((ngroups == 1) || (Kg % C0 == 0)) && (C0 % VTraits<v_float32>::vlanes() == 0);
71#endif
72
73 const int NK1 = N * K1;
74 parallel_for_(Range(0, NK1), [&](const Range& range) {
75 for (int nk1 = range.start; nk1 < range.end; nk1++) {
76 const int n = nk1 / K1;
77 const int k1 = nk1 % K1;
78
79 const int k_base = k1 * C0;
80 const int currK0 = std::min(C0, K - k_base);
81 if (currK0 <= 0) continue;
82
83 float* out_k1 = (float*)out__ + ((int64_t)n * K1 + k1) * ospatial * C0;
84 const float* inp_n = (const float*)inp__ + (int64_t)n * C1 * ispatial * C0;
85
86#if (CV_SIMD || CV_SIMD_SCALABLE)
87 // Precompute the shared (g, kblk, c1_abs_base) for the SIMD path.
88 int simd_g = 0, simd_kblk = 0, simd_c1_abs_base = 0;
89 if (simd_ok) {
90 simd_g = k_base / Kg;
91 simd_kblk = (k_base - simd_g * Kg) / K0;
92 const int c_start = simd_g * Cg;
93 simd_c1_abs_base = c_start / C0;
94 }
95#endif
96
97 for (int opos_flat = 0; opos_flat < ospatial; opos_flat++) {
98 float* out_ptr = out_k1 + opos_flat * C0;
99
100 // Bias init: write currK0 valid lanes + zero-pad the rest.
101 if (bias) {
102 for (int k0 = 0; k0 < currK0; k0++) out_ptr[k0] = bias[k_base + k0];
103 } else {
104 for (int k0 = 0; k0 < currK0; k0++) out_ptr[k0] = 0.f;
105 }
106 for (int k0 = currK0; k0 < C0; k0++) out_ptr[k0] = 0.f;
107
108 int ocoords[MAX_DIMS];
109 {
110 int tmp = opos_flat;
111 for (int i = sdims - 1; i >= 0; i--) {
112 int di = MAX_DIMS - sdims + i;
113 ocoords[di] = tmp % oDims[di];
114 tmp /= oDims[di];
115 }
116 }
117
118#if (CV_SIMD || CV_SIMD_SCALABLE)
119 if (simd_ok) {
120 const int VLANES = VTraits<v_float32>::vlanes();
121 const int v_per_block = C0 / VLANES;
122 const bool two_vecs = (v_per_block == 2);
123 if (v_per_block == 1 || two_vecs) {
124 v_float32 acc0 = vx_load(out_ptr);
125 v_float32 acc1 = two_vecs ? vx_load(out_ptr + VLANES)
126 : vx_setzero_f32();
127
128 for (int ks = 0; ks < ksize; ks++) {
129 bool valid = true;
130 int ipos_flat = 0;
131 for (int i = 0; i < sdims; i++) {
132 int di = MAX_DIMS - sdims + i;
133 int raw = ocoords[di] + cs.pads[di]
134 - kcoords_tab[ks][di] * cs.dilations[di];
135 if (raw < 0 || raw % cs.strides[di] != 0) {
136 valid = false; break;
137 }
138 int ic = raw / cs.strides[di];
139 if (ic >= iDims[di]) { valid = false; break; }
140 ipos_flat = ipos_flat * iDims[di] + ic;
141 }
142 if (!valid) continue;
143
144 const float* w_base = wdata +
145 ((int64_t)(simd_g * Kblk + simd_kblk) * ksize + ks)
146 * C1Max * C0 * K0;
147
148 for (int c1p = 0; c1p < C1Max; c1p++) {
149 const int c1_abs = simd_c1_abs_base + c1p;
150 if (c1_abs >= C1) break;
151
152 const float* inp_ptr = inp_n +
153 (int64_t)(c1_abs * ispatial + ipos_flat) * C0;
154 const float* w_c1p = w_base + (int64_t)c1p * C0 * K0;
155
156 for (int c0 = 0; c0 < C0; c0++) {
157 v_float32 inp_v = vx_setall_f32(inp_ptr[c0]);
158 v_float32 w0 = vx_load(w_c1p + c0 * K0);
159 acc0 = v_fma(inp_v, w0, acc0);
160 if (two_vecs) {
161 v_float32 w1 = vx_load(w_c1p + c0 * K0 + VLANES);
162 acc1 = v_fma(inp_v, w1, acc1);
163 }
164 }
165 }
166 }
167
168 vx_store(out_ptr, acc0);
169 if (two_vecs) vx_store(out_ptr + VLANES, acc1);
170 continue;
171 }
172 }
173#endif
174
175 // Scalar fallback.
176 for (int ks = 0; ks < ksize; ks++) {
177 bool valid = true;
178 int ipos_flat = 0;
179 for (int i = 0; i < sdims; i++) {
180 int di = MAX_DIMS - sdims + i;
181 int raw = ocoords[di] + cs.pads[di]
182 - kcoords_tab[ks][di] * cs.dilations[di];
183 if (raw < 0 || raw % cs.strides[di] != 0) {
184 valid = false; break;
185 }
186 int ic = raw / cs.strides[di];
187 if (ic >= iDims[di]) { valid = false; break; }
188 ipos_flat = ipos_flat * iDims[di] + ic;
189 }
190 if (!valid) continue;
191
192 for (int k0 = 0; k0 < currK0; k0++) {
193 const int k = k_base + k0;
194 const int g = k / Kg;
195 const int kin = k - g * Kg;
196 const int kblk = kin / K0;
197 const int k0l = kin & (K0 - 1);
198
199 const float* w_base = wdata +
200 ((int64_t)(g * Kblk + kblk) * ksize + ks) * C1Max * C0 * K0;
201
202 const int c_start = g * Cg;
203 const int c1_abs_base = c_start / C0;
204
205 float sum = 0.f;
206 for (int c1p = 0; c1p < C1Max; c1p++) {
207 const int c1_abs = c1_abs_base + c1p;
208 if (c1_abs >= C1) break;
209
210 const float* inp_ptr = inp_n +
211 (int64_t)(c1_abs * ispatial + ipos_flat) * C0;
212 const float* w_c1p = w_base + (int64_t)c1p * C0 * K0;
213
214 for (int c0 = 0; c0 < C0; c0++)
215 sum += w_c1p[c0 * K0 + k0l] * inp_ptr[c0];
216 }
217 out_ptr[k0] += sum;
218 }
219 }
220 }
221 }
222 });
223}
