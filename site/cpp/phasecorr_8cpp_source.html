46static void magSpectrums( InputArray _src, OutputArray _dst)
47{
48 Mat src = _src.getMat();
49 int depth = src.depth(), cn = src.channels(), type = src.type();
50 int rows = src.rows, cols = src.cols;
51 int j, k;
52
53 CV_Assert( type == CV_32FC1 || type == CV_32FC2 || type == CV_64FC1 || type == CV_64FC2 );
54
55 if(src.depth() == CV_32F)
56 _dst.create( src.rows, src.cols, CV_32FC1 );
57 else
58 _dst.create( src.rows, src.cols, CV_64FC1 );
59
60 Mat dst = _dst.getMat();
61 dst.setTo(0);//Mat elements are not equal to zero by default!
62
63 bool is_1d = (rows == 1 || (cols == 1 && src.isContinuous() && dst.isContinuous()));
64
65 if( is_1d )
66 cols = cols + rows - 1, rows = 1;
67
68 int ncols = cols*cn;
69 int j0 = cn == 1;
70 int j1 = ncols - (cols % 2 == 0 && cn == 1);
71
72 if( depth == CV_32F )
73 {
74 const float* dataSrc = src.ptr<float>();
75 float* dataDst = dst.ptr<float>();
76
77 size_t stepSrc = src.step/sizeof(dataSrc[0]);
78 size_t stepDst = dst.step/sizeof(dataDst[0]);
79
80 if( !is_1d && cn == 1 )
81 {
82 for( k = 0; k < (cols % 2 ? 1 : 2); k++ )
83 {
84 if( k == 1 )
85 dataSrc += cols - 1, dataDst += cols - 1;
86 dataDst[0] = (float)std::abs(dataSrc[0]);
87 if( rows % 2 == 0 )
88 dataDst[(rows-1)*stepDst] = (float)std::abs(dataSrc[(rows-1)*stepSrc]);
89
90 for( j = 1; j <= rows - 2; j += 2 )
91 {
92 dataDst[j*stepDst] = (float)std::sqrt((double)dataSrc[j*stepSrc]*dataSrc[j*stepSrc] +
93 (double)dataSrc[(j+1)*stepSrc]*dataSrc[(j+1)*stepSrc]);
94 }
95
96 if( k == 1 )
97 dataSrc -= cols - 1, dataDst -= cols - 1;
98 }
99 }
100
101 for( ; rows--; dataSrc += stepSrc, dataDst += stepDst )
102 {
103 if( is_1d && cn == 1 )
104 {
105 dataDst[0] = (float)std::abs(dataSrc[0]);
106 if( cols % 2 == 0 )
107 dataDst[j1] = (float)std::abs(dataSrc[j1]);
108 }
109
110 for( j = j0; j < j1; j += 2 )
111 {
112 dataDst[j] = (float)std::sqrt((double)dataSrc[j]*dataSrc[j] + (double)dataSrc[j+1]*dataSrc[j+1]);
113 }
114 }
115 }
116 else
117 {
118 const double* dataSrc = src.ptr<double>();
119 double* dataDst = dst.ptr<double>();
120
121 size_t stepSrc = src.step/sizeof(dataSrc[0]);
122 size_t stepDst = dst.step/sizeof(dataDst[0]);
123
124 if( !is_1d && cn == 1 )
125 {
126 for( k = 0; k < (cols % 2 ? 1 : 2); k++ )
127 {
128 if( k == 1 )
129 dataSrc += cols - 1, dataDst += cols - 1;
130 dataDst[0] = std::abs(dataSrc[0]);
131 if( rows % 2 == 0 )
132 dataDst[(rows-1)*stepDst] = std::abs(dataSrc[(rows-1)*stepSrc]);
133
134 for( j = 1; j <= rows - 2; j += 2 )
135 {
136 dataDst[j*stepDst] = std::sqrt(dataSrc[j*stepSrc]*dataSrc[j*stepSrc] +
137 dataSrc[(j+1)*stepSrc]*dataSrc[(j+1)*stepSrc]);
138 }
139
140 if( k == 1 )
141 dataSrc -= cols - 1, dataDst -= cols - 1;
142 }
143 }
144
145 for( ; rows--; dataSrc += stepSrc, dataDst += stepDst )
146 {
147 if( is_1d && cn == 1 )
148 {
149 dataDst[0] = std::abs(dataSrc[0]);
150 if( cols % 2 == 0 )
151 dataDst[j1] = std::abs(dataSrc[j1]);
152 }
153
154 for( j = j0; j < j1; j += 2 )
155 {
156 dataDst[j] = std::sqrt(dataSrc[j]*dataSrc[j] + dataSrc[j+1]*dataSrc[j+1]);
157 }
158 }
159 }
160}

162static void fftShift(InputOutputArray _out)
163{
164 Mat out = _out.getMat();
165
166 if(out.rows == 1 && out.cols == 1)
167 {
168 // trivially shifted.
169 return;
170 }
171
172 std::vector<Mat> planes;
173 split(out, planes);
174
175 int xMid = out.cols >> 1;
176 int yMid = out.rows >> 1;
177
178 bool is_1d = xMid == 0 || yMid == 0;
179
180 if(is_1d)
181 {
182 int is_odd = (xMid > 0 && out.cols % 2 == 1) || (yMid > 0 && out.rows % 2 == 1);
183 xMid = xMid + yMid;
184
185 for(size_t i = 0; i < planes.size(); i++)
186 {
187 Mat tmp;
188 Mat half0(planes[i], Rect(0, 0, xMid + is_odd, 1));
189 Mat half1(planes[i], Rect(xMid + is_odd, 0, xMid, 1));
190
191 half0.copyTo(tmp);
192 half1.copyTo(planes[i](Rect(0, 0, xMid, 1)));
193 tmp.copyTo(planes[i](Rect(xMid, 0, xMid + is_odd, 1)));
194 }
195 }
196 else
197 {
198 int isXodd = out.cols % 2 == 1;
199 int isYodd = out.rows % 2 == 1;
200 for(size_t i = 0; i < planes.size(); i++)
201 {
202 // perform quadrant swaps...
203 Mat q0(planes[i], Rect(0, 0, xMid + isXodd, yMid + isYodd));
204 Mat q1(planes[i], Rect(xMid + isXodd, 0, xMid, yMid + isYodd));
205 Mat q2(planes[i], Rect(0, yMid + isYodd, xMid + isXodd, yMid));
206 Mat q3(planes[i], Rect(xMid + isXodd, yMid + isYodd, xMid, yMid));
207
208 if(!(isXodd || isYodd))
209 {
210 Mat tmp;
211 q0.copyTo(tmp);
212 q3.copyTo(q0);
213 tmp.copyTo(q3);
214
215 q1.copyTo(tmp);
216 q2.copyTo(q1);
217 tmp.copyTo(q2);
218 }
219 else
220 {
221 Mat tmp0, tmp1, tmp2 ,tmp3;
222 q0.copyTo(tmp0);
223 q1.copyTo(tmp1);
224 q2.copyTo(tmp2);
225 q3.copyTo(tmp3);
226
227 tmp0.copyTo(planes[i](Rect(xMid, yMid, xMid + isXodd, yMid + isYodd)));
228 tmp3.copyTo(planes[i](Rect(0, 0, xMid, yMid)));
229
230 tmp1.copyTo(planes[i](Rect(0, yMid, xMid, yMid + isYodd)));
231 tmp2.copyTo(planes[i](Rect(xMid, 0, xMid + isXodd, yMid)));
232 }
233 }
234 }
235
236 merge(planes, out);
237}

239static Point2d weightedCentroid(InputArray _src, cv::Point peakLocation, cv::Size weightBoxSize, double* response)
240{
241 Mat src = _src.getMat();
242
243 int type = src.type();
244 CV_Assert( type == CV_32FC1 || type == CV_64FC1 );
245
246 int minr = peakLocation.y - (weightBoxSize.height >> 1);
247 int maxr = peakLocation.y + (weightBoxSize.height >> 1);
248 int minc = peakLocation.x - (weightBoxSize.width >> 1);
249 int maxc = peakLocation.x + (weightBoxSize.width >> 1);
250
251 Point2d centroid;
252 double sumIntensity = 0.0;
253
254 // clamp the values to min and max if needed.
255 if(minr < 0)
256 {
257 minr = 0;
258 }
259
260 if(minc < 0)
261 {
262 minc = 0;
263 }
264
265 if(maxr > src.rows - 1)
266 {
267 maxr = src.rows - 1;
268 }
269
270 if(maxc > src.cols - 1)
271 {
272 maxc = src.cols - 1;
273 }
274
275 if(type == CV_32FC1)
276 {
277 const float* dataIn = src.ptr<float>();
278 dataIn += minr*src.cols;
279 for(int y = minr; y <= maxr; y++)
280 {
281 for(int x = minc; x <= maxc; x++)
282 {
283 centroid.x += (double)x*dataIn[x];
284 centroid.y += (double)y*dataIn[x];
285 sumIntensity += (double)dataIn[x];
286 }
287
288 dataIn += src.cols;
289 }
290 }
291 else
292 {
293 const double* dataIn = src.ptr<double>();
294 dataIn += minr*src.cols;
295 for(int y = minr; y <= maxr; y++)
296 {
297 for(int x = minc; x <= maxc; x++)
298 {
299 centroid.x += (double)x*dataIn[x];
300 centroid.y += (double)y*dataIn[x];
301 sumIntensity += dataIn[x];
302 }
303
304 dataIn += src.cols;
305 }
306 }
307
308 if(response)
309 *response = sumIntensity;
310
311 sumIntensity += DBL_EPSILON; // prevent div0 problems...
312
313 centroid.x /= sumIntensity;
314 centroid.y /= sumIntensity;
315
316 return centroid;
317}

321cv::Point2d cv::phaseCorrelate(InputArray _src1, InputArray _src2, InputArray _window, double* response)
322{
323 CV_INSTRUMENT_REGION();
324
325 Mat src1 = _src1.getMat();
326 Mat src2 = _src2.getMat();
327 Mat window = _window.getMat();
328
329 CV_Assert( src1.type() == src2.type());
330 CV_Assert( src1.type() == CV_32FC1 || src1.type() == CV_64FC1 );
331 CV_Assert( src1.size == src2.size);
332
333 if(!window.empty())
334 {
335 CV_Assert( src1.type() == window.type());
336 CV_Assert( src1.size == window.size);
337 }
338
339 int M = getOptimalDFTSize(src1.rows);
340 int N = getOptimalDFTSize(src1.cols);
341
342 Mat padded1, padded2, paddedWin;
343
344 if(M != src1.rows || N != src1.cols)
345 {
346 copyMakeBorder(src1, padded1, 0, M - src1.rows, 0, N - src1.cols, BORDER_CONSTANT, Scalar::all(0));
347 copyMakeBorder(src2, padded2, 0, M - src2.rows, 0, N - src2.cols, BORDER_CONSTANT, Scalar::all(0));
348
349 if(!window.empty())
350 {
351 copyMakeBorder(window, paddedWin, 0, M - window.rows, 0, N - window.cols, BORDER_CONSTANT, Scalar::all(0));
352 }
353 }
354 else
355 {
356 padded1 = src1;
357 padded2 = src2;
358 paddedWin = window;
359 }
360
361 Mat FFT1, FFT2, P, Pm, C;
362
363 // perform window multiplication if available
364 if(!paddedWin.empty())
365 {
366 // apply window to both images before proceeding...
367 multiply(paddedWin, padded1, padded1);
368 multiply(paddedWin, padded2, padded2);
369 }
370
371 // execute phase correlation equation
372 // Reference: http://en.wikipedia.org/wiki/Phase_correlation
373 dft(padded1, FFT1, DFT_REAL_OUTPUT);
374 dft(padded2, FFT2, DFT_REAL_OUTPUT);
375
376 mulSpectrums(FFT1, FFT2, P, 0, true);
377
378 magSpectrums(P, Pm);
379 divSpectrums(P, Pm, C, 0, false); // FF* / |FF*| (phase correlation equation completed here...)
380
381 idft(C, C); // gives us the nice peak shift location...
382
383 fftShift(C); // shift the energy to the center of the frame.
384
385 // locate the highest peak
386 Point peakLoc;
387 minMaxLoc(C, NULL, NULL, NULL, &peakLoc);
388
389 // get the phase shift with sub-pixel accuracy, 5x5 window seems about right here...
390 Point2d t;
391 t = weightedCentroid(C, peakLoc, Size(5, 5), response);
392
393 // max response is M*N (not exactly, might be slightly larger due to rounding errors)
394 if(response)
395 *response /= M*N;
396
397 // adjust shift relative to image center...
398 Point2d center((double)padded1.cols / 2.0, (double)padded1.rows / 2.0);
399
400 return (center - t);
401}

404void cv::createHanningWindow(OutputArray _dst, cv::Size winSize, int type)
405{
406 CV_INSTRUMENT_REGION();
407
408 CV_Assert( type == CV_32FC1 || type == CV_64FC1 );
409 CV_Assert( winSize.width > 1 && winSize.height > 1 );
410
411 _dst.create(winSize, type);
412 Mat dst = _dst.getMat();
413
414 int rows = dst.rows, cols = dst.cols;
415
416 AutoBuffer<double> _wc(cols);
417 double* const wc = _wc.data();
418
419 double coeff0 = 2.0 * CV_PI / (double)(cols - 1), coeff1 = 2.0 * CV_PI / (double)(rows - 1);
420 int c = 0;
421#if CV_SIMD_64F || CV_SIMD_SCALABLE_64F
422 const int nlanes32 = VTraits<v_float32>::vlanes();
423 const int nlanes64 = VTraits<v_float64>::vlanes();
424 const int max_nlanes = VTraits<v_float64>::max_nlanes;
425 std::array<double, max_nlanes> index;
426 std::iota(index.data(), index.data()+max_nlanes, 0.f);
427 v_float64 vindex = vx_load(index.data());
428 v_float64 delta = vx_setall_f64(VTraits<v_float64>::vlanes());
429 v_float64 vcoeff0 = vx_setall_f64(coeff0);
430 v_float64 one = vx_setall_f64(1.f);
431 v_float64 half = vx_setall_f64(0.5f);
432 for (; c <= cols - nlanes64; c += nlanes64)
433 {
434 v_float64 v = v_mul(half, v_sub(one, v_cos(v_mul(vcoeff0, vindex))));
435 vx_store(wc + c, v);
436 vindex = v_add(vindex, delta);
437 }
438#endif
439 for(; c < cols; c++)
440 wc[c] = 0.5 * (1.0 - cos(coeff0 * c));
441
442 if(dst.depth() == CV_32F)
443 {
444 for(int i = 0; i < rows; i++)
445 {
446 float* dstData = dst.ptr<float>(i);
447 double wr = 0.5 * (1.0 - cos(coeff1 * i));
448 int j = 0;
449#if CV_SIMD_64F || CV_SIMD_SCALABLE_64F
450 v_float64 vwr = vx_setall_f64(wr);
451 for (; j <= cols - nlanes32; j += nlanes32)
452 {
453 v_float64 v0 = v_mul(vwr, vx_load(wc + j));
454 v_float64 v1 = v_mul(vwr, vx_load(wc + j + nlanes64));
455 vx_store(dstData + j, v_cvt_f32(v0, v1));
456 }
457#endif
458 for(; j < cols; j++)
459 dstData[j] = (float)(wr * wc[j]);
460 }
461 }
462 else
463 {
464 for(int i = 0; i < rows; i++)
465 {
466 double* dstData = dst.ptr<double>(i);
467 double wr = 0.5 * (1.0 - cos(coeff1 * i));
468 int j = 0;
469#if CV_SIMD_64F || CV_SIMD_SCALABLE_64F
470 v_float64 vwr = vx_setall_f64(wr);
471 for (; j <= cols - nlanes64; j += nlanes64)
472 {
473 v_float64 v = v_mul(vwr, vx_load(wc + j));
474 vx_store(dstData + j, v);
475 }
476#endif
477 for(; j < cols; j++)
478 dstData[j] = wr * wc[j];
479 }
480 }
481
482 // perform batch sqrt for SSE performance gains
483 cv::sqrt(dst, dst);
484}
