44double cv::matchShapes(InputArray contour1, InputArray contour2, int method, double)
45{
46 CV_INSTRUMENT_REGION();
47
48 double ma[7], mb[7];
49 int i, sma, smb;
50 double eps = 1.e-5;
51 double mmm;
52 double result = 0;
53 bool anyA = false, anyB = false;
54
55 HuMoments( moments(contour1), ma );
56 HuMoments( moments(contour2), mb );
57
58 switch (method)
59 {
60 case 1:
61 for( i = 0; i < 7; i++ )
62 {
63 double ama = fabs( ma[i] );
64 double amb = fabs( mb[i] );
65
66 if (ama > 0)
67 anyA = true;
68 if (amb > 0)
69 anyB = true;
70
71 if( ma[i] > 0 )
72 sma = 1;
73 else if( ma[i] < 0 )
74 sma = -1;
75 else
76 sma = 0;
77 if( mb[i] > 0 )
78 smb = 1;
79 else if( mb[i] < 0 )
80 smb = -1;
81 else
82 smb = 0;
83
84 if( ama > eps && amb > eps )
85 {
86 ama = 1. / (sma * log10( ama ));
87 amb = 1. / (smb * log10( amb ));
88 result += fabs( -ama + amb );
89 }
90 }
91 break;
92
93 case 2:
94 for( i = 0; i < 7; i++ )
95 {
96 double ama = fabs( ma[i] );
97 double amb = fabs( mb[i] );
98
99 if (ama > 0)
100 anyA = true;
101 if (amb > 0)
102 anyB = true;
103
104 if( ma[i] > 0 )
105 sma = 1;
106 else if( ma[i] < 0 )
107 sma = -1;
108 else
109 sma = 0;
110 if( mb[i] > 0 )
111 smb = 1;
112 else if( mb[i] < 0 )
113 smb = -1;
114 else
115 smb = 0;
116
117 if( ama > eps && amb > eps )
118 {
119 ama = sma * log10( ama );
120 amb = smb * log10( amb );
121 result += fabs( -ama + amb );
122 }
123 }
124 break;
125
126 case 3:
127 for( i = 0; i < 7; i++ )
128 {
129 double ama = fabs( ma[i] );
130 double amb = fabs( mb[i] );
131
132 if (ama > 0)
133 anyA = true;
134 if (amb > 0)
135 anyB = true;
136
137 if( ma[i] > 0 )
138 sma = 1;
139 else if( ma[i] < 0 )
140 sma = -1;
141 else
142 sma = 0;
143 if( mb[i] > 0 )
144 smb = 1;
145 else if( mb[i] < 0 )
146 smb = -1;
147 else
148 smb = 0;
149
150 if( ama > eps && amb > eps )
151 {
152 ama = sma * log10( ama );
153 amb = smb * log10( amb );
154 mmm = fabs( (ama - amb) / ama );
155 if( result < mmm )
156 result = mmm;
157 }
158 }
159 break;
160 default:
161 CV_Error( cv::Error::StsBadArg, "Unknown comparison method" );
162 }
163
164 //If anyA and anyB are both true, the result is correct.
165 //If anyA and anyB are both false, the distance is 0, perfect match.
166 //If only one is true, then it's a false 0 and return large error.
167 if (anyA != anyB)
168 result = DBL_MAX;
169
170 return result;
171}
