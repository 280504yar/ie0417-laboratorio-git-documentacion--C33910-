37 void processGraph(Ptr<Graph>& graph)
38 {
39 const std::vector<Ptr<LayerInfo> >& prog = graph->prog();
40 size_t i, nops = prog.size();
41 std::vector<Arg> removed_args;
42 std::vector<Arg> saved_tail_inputs;
43
44 for (i = 0; i < nops; i++) {
45 const Ptr<LayerInfo>& layer = prog[i];
46 LayerInfo* layer_ptr = const_cast<LayerInfo*>(layer.get());
47 std::vector<Ptr<Graph> >* subgraphs = layer->subgraphs();
48 if (subgraphs) {
49 for (Ptr<Graph>& g: *subgraphs) {
50 processGraph(g);
51 }
52 }
53 const std::vector<Arg>& inputs = layer->inputs;
54 size_t j, ninputs = inputs.size();
55 if (ninputs == 1) {
56 continue;
57 }
58 bool tail_const = true, unuse_tail = false;
59 saved_tail_inputs.clear();
60 for (j = 1; j < ninputs; j++) {
61 Arg inp = inputs[j];
62 bool const_arg = netimpl->isConstArg(inp);
63 if (!const_arg)
64 tail_const = false;
65 saved_tail_inputs.push_back(inp);
66 }
67
68 Conv2Layer* conv = dynamic_cast<Conv2Layer*>(layer_ptr);
69 ConvTranspose2Layer* deconv = dynamic_cast<ConvTranspose2Layer*>(layer_ptr);
70 BatchNorm2Layer* bn = dynamic_cast<BatchNorm2Layer*>(layer_ptr);
71 ChannelsPReLULayer* prelu = dynamic_cast<ChannelsPReLULayer*>(layer_ptr);
72 //ActivationLayer* activ = dynamic_cast<ActivationLayer*>(layer_ptr);
73
74 if (tail_const) {
75 if (conv) {
76 // convolution with constant weights and bias
77 conv->setWeights(netimpl->__tensors__[inputs[1]],
78 ninputs > 2 ? netimpl->__tensors__[inputs[2]] : Mat(),
79 netimpl->defaultC0, netimpl->accuracy);
80 conv->inputs.resize(1);
81 unuse_tail = true;
82 } else if (deconv) {
83 // deconvolution with constant weights and bias
84 deconv->setWeights(netimpl->__tensors__[inputs[1]],
85 ninputs > 2 ? netimpl->__tensors__[inputs[2]] : Mat(),
86 netimpl->defaultC0, netimpl->accuracy);
87 deconv->inputs.resize(1);
88 unuse_tail = true;
89 } else if (bn && bn->freezeScaleBias()) {
90 // batch norm with constant parameters
91 unuse_tail = true;
92 } else if (prelu && ninputs == 2) {
93 prelu->setSlope(netimpl->__tensors__[inputs[1].idx]);
94 prelu->inputs.resize(1);
95 unuse_tail = true;
96 }/* else if (activ && dynamic_cast<ReLU6Layer>(activ)) {
97 // [TODO] ...
98 unuse_tail = true;
99 }*/
100 }
101
102 if (unuse_tail) {
103 for (Arg inp: saved_tail_inputs)
104 unuse(inp);
105 }
106 }
107 }
