71// from Ceres, equation energy change:
72// eq. energy = 1/2 * (residuals + J * step)^2 =
73// 1/2 * ( residuals^2 + 2 * residuals^T * J * step + (J*step)^T * J * step)
74// eq. energy change = 1/2 * residuals^2 - eq. energy =
75// 1/2 * residuals^2 - 1/2 * ( residuals^2 + 2 * residuals^T * J * step + (J*step)^T * J * step) =
76// 1/2 * ( residuals^2 - residuals^2 - 2 * residuals^T * J * step - (J*step)^T * J * step) =
77// - 1/2 * ( 2 * residuals^T * J * step + (J*step)^T * J * step) =
78// - 1/2 * ( 2 * residuals^T * J + (J*step)^T * J ) * step =
79// - 1/2 * ( 2 * residuals^T * J + step^T * J^T * J ) * step =
80// - 1/2 * step^T * ( 2 * J^t * residuals + J^T * J * step ) =
81// - 1/2 * step^T * ( 2 * J^t * residuals + (J^T * J + LMDiag - LMDiag) * step ) =
82// - 1/2 * step^T * ( 2 * J^t * residuals + (J^T * J + LMDiag) * step - LMDiag * step ) =
83// - 1/2 * step^T * ( 2 * J^t * residuals - J^T * residuals - LMDiag * step ) =
84// - 1/2 * step^T * ( J^t * residuals - LMDiag * step ) =
85// - 1/2 * x^T * ( jtb - LMDiag * x )
86static double calcJacCostChangeLm(const cv::Mat_<double>& jtb, const cv::Mat_<double>& x, const cv::Mat_<double>& lmDiag)
87{
88 return -0.5 * cv::sum(x.mul(jtb - lmDiag.mul(x)))[0];
89}

92// calculates J^T*rvv where rvv is second directional derivative of the function in direction v
93// rvv = (f(x0 + v*h) - f(x0))/h - J*v)/h
94// where v is a LevMarq equation solution
95// J^T*rvv = J^T*((f(x0 + v*h) - f(x0))/h - J*v)/h =
96// J^T*(f(x0 + v*h) - f(x0) - J*v*h)/h^2 =
97// (J^T*f(x0 + v*h) - J^T*f(x0) - J^T*J*v*h)/h^2 =
98// < using (J^T*J + lmdiag) * v = J^T*b, also f(x0 + v*h) = b_v, f(x0) = b >
99// (J^T*b_v - J^T*b - (J^T*J + lmdiag - lmdiag)*v*h)/h^2 =
100// (J^T*b_v - J^T*b + J^t*b*h + lmdiag*v*h)/h^2 =
101// (J^T*b_v - J^t*b*(1 - h) + lmdiag*v*h)/h^2
102static void calcJtrvv(const Mat_<double>& jtbv, const Mat_<double>& jtb, const Mat_<double>& lmdiag,
103 const Mat_<double>& v, const double hGeo,
104 Mat_<double>& jtrvv)
105{
106 jtrvv = (jtbv + jtb * (hGeo - 1.0) + lmdiag.mul(v, hGeo)) / (hGeo * hGeo);
107}

110LevMarq::Report detail::LevMarqBase::optimize()
111{
112 if (settings.geodesic && !backend->enableGeo())
113 {
114 CV_LOG_INFO(NULL, "The backend does not support geodesic acceleration, please turn off the corresponding option");
115 return LevMarq::Report(false, 0, 0); // not found
116 }
117
118 // this function sets probe vars to current X
119 backend->prepareVars();
120
121 double energy = 0.0;
122 if (!backend->calcFunc(energy, /*calcEnergy*/ true, /*calcJacobian*/ true) || energy < 0)
123 {
124 CV_LOG_INFO(NULL, "Error while calculating energy function");
125 return LevMarq::Report(false, 0, 0); // not found
126 }
127
128 double oldEnergy = energy;
129
130 CV_LOG_DEBUG(NULL, "#s" << " energy: " << energy);
131
132 // diagonal clamping options
133 const double minDiag = 1e-6;
134 const double maxDiag = 1e32;
135 // limit lambda inflation
136 const double maxLambda = 1e32;
137
138 // finish reasons
139 bool tooLong = false; // => not found
140 bool bigLambda = false; // => not found
141 bool smallGradient = false; // => found
142 bool smallStep = false; // => found
143 bool smallEnergyDelta = false; // => found
144 bool smallEnergy = false; // => found
145
146 // column scale inverted, for jacobian scaling
147 Mat_<double> di;
148
149 // do the jacobian conditioning improvement used in Ceres
150 if (settings.jacobiScaling)
151 {
152 // L2-normalize each jacobian column
153 // vec d = {d_j = sum(J_ij^2) for each column j of J} = get_diag{ J^T * J }
154 // di = { 1/(1+sqrt(d_j)) }, extra +1 to avoid div by zero
155 Mat_<double> ds;
156 const Mat_<double> diag = backend->getDiag();
157 cv::sqrt(diag, ds);
158 di = 1.0 / (ds + 1.0);
159 }
160
161 double lmUpFactor = settings.initialLmUpFactor;
162 double lambdaLevMarq = settings.initialLambda;
163
164 unsigned int iter = 0;
165 bool done = false;
166 while (!done)
167 {
168 // At this point we should have jtj and jtb built
169
170 CV_LOG_DEBUG(NULL, "#LM#s" << " energy: " << energy);
171
172 // do the jacobian conditioning improvement used in Ceres
173 if (settings.jacobiScaling)
174 {
175 backend->doJacobiScaling(di);
176 }
177
178 const Mat_<double> jtb = backend->getJtb();
179
180 double gradientMax = cv::norm(jtb, NORM_INF);
181
182 // Save original diagonal of jtj matrix for LevMarq
183 const Mat_<double> diag = backend->getDiag();
184
185 // Solve using LevMarq and get delta transform
186 bool enoughLm = false;
187
188 while (!enoughLm && !done)
189 {
190 // form LevMarq matrix
191 Mat_<double> lmDiag, jtjDiag;
192 lmDiag = diag * lambdaLevMarq;
193 if (settings.clampDiagonal)
194 lmDiag = cv::min(cv::max(lmDiag, minDiag), maxDiag);
195 jtjDiag = lmDiag + diag;
196 backend->setDiag(jtjDiag);
197
198 CV_LOG_DEBUG(NULL, "linear decompose...");
199
200 bool decomposed = backend->decompose();
201
202 CV_LOG_DEBUG(NULL, (decomposed ? "OK" : "FAIL"));
203
204 CV_LOG_DEBUG(NULL, "linear solve...");
205 // use double or convert everything to float
206 Mat_<double> x((int)jtb.rows, 1);
207 bool solved = decomposed && backend->solveDecomposed(jtb, x);
208
209 CV_LOG_DEBUG(NULL, (solved ? "OK" : "FAIL"));
210
211 double costChange = 0.0;
212 double jacCostChange = 0.0;
213 double stepQuality = 0.0;
214 double xNorm = 0.0;
215 if (solved)
216 {
217 // what energy drop should be according to local model estimation
218 jacCostChange = calcJacCostChangeLm(jtb, x, lmDiag);
219
220 // x norm
221 xNorm = cv::norm(x, settings.stepNormInf ? NORM_INF : NORM_L2SQR);
222
223 // undo jacobi scaling
224 if (settings.jacobiScaling)
225 {
226 x = x.mul(di);
227 }
228
229 if (settings.geodesic)
230 {
231 backend->currentOplusX(x * settings.hGeo, /*geo*/ true);
232
233 Mat_<double> jtbv(jtb.rows, 1);
234 if (backend->calcJtbv(jtbv))
235 {
236 Mat_<double> jtrvv(jtb.rows, 1);
237 calcJtrvv(jtbv, jtb, lmDiag, x, settings.hGeo, jtrvv);
238
239 Mat_<double> xgeo((int)jtb.rows, 1);
240 bool geoSolved = backend->solveDecomposed(jtrvv, xgeo);
241
242 if (geoSolved)
243 {
244 double truncerr = sqrt(xgeo.dot(xgeo) / x.dot(x));
245 bool geoIsGood = (truncerr < 1.0);
246 if (geoIsGood)
247 x += xgeo * settings.geoScale;
248
249 CV_LOG_DEBUG(NULL, "Geo truncerr: " << truncerr << (geoIsGood ? ", use it" : ", skip it") );
250 }
251 else
252 {
253 CV_LOG_DEBUG(NULL, "Geo: failed to solve");
254 }
255 }
256 }
257
258 // calc energy with current delta x
259 backend->currentOplusX(x, /*geo*/ false);
260
261 bool success = backend->calcFunc(energy, /*calcEnergy*/ true, /*calcJacobian*/ false);
262 if (!success || energy < 0 || std::isnan(energy))
263 {
264 CV_LOG_INFO(NULL, "Error while calculating energy function");
265 return LevMarq::Report(false, iter, oldEnergy); // not found
266 }
267
268 costChange = oldEnergy - energy;
269
270 stepQuality = costChange / jacCostChange;
271
272 CV_LOG_DEBUG(NULL, "#LM#" << iter
273 << " energy: " << energy
274 << " deltaEnergy: " << costChange
275 << " deltaEqEnergy: " << jacCostChange
276 << " max(J^T*b): " << gradientMax
277 << (settings.stepNormInf ? " normInf(x): " : " norm2(x): ") << xNorm
278 << " deltaEnergy/energy: " << costChange / energy);
279 }
280
281 // zero cost change is treated like an algorithm failure if checkRelEnergyChange is off
282 if (!solved || costChange < 0 || (!settings.checkRelEnergyChange && abs(costChange) < DBL_EPSILON))
283 {
284 // failed to optimize, increase lambda and repeat
285
286 lambdaLevMarq *= lmUpFactor;
287 if (settings.upDouble)
288 lmUpFactor *= 2.0;
289
290 CV_LOG_DEBUG(NULL, "LM goes up, lambda: " << lambdaLevMarq << ", old energy: " << oldEnergy);
291 }
292 else
293 {
294 // optimized successfully, decrease lambda and set variables for next iteration
295 enoughLm = true;
296
297 if (settings.useStepQuality)
298 lambdaLevMarq *= std::max(1.0 / settings.initialLmDownFactor, 1.0 - pow(2.0 * stepQuality - 1.0, 3));
299 else
300 lambdaLevMarq *= 1.0 / settings.initialLmDownFactor;
301 lmUpFactor = settings.initialLmUpFactor;
302
303 // Once set, these flags will be activated until next successful LM iteration - this is not a bug
304 smallGradient = (gradientMax < settings.minGradientTolerance);
305 smallStep = (xNorm < settings.stepNormTolerance);
306 smallEnergyDelta = (costChange / energy < settings.relEnergyDeltaTolerance);
307 smallEnergy = (energy < settings.smallEnergyTolerance);
308
309 backend->acceptProbe();
310
311 CV_LOG_DEBUG(NULL, "#" << iter << " energy: " << energy);
312
313 oldEnergy = energy;
314
315 CV_LOG_DEBUG(NULL, "LM goes down, lambda: " << lambdaLevMarq << " step quality: " << stepQuality);
316 }
317
318 iter++;
319
320 tooLong = (iter >= settings.maxIterations);
321 bigLambda = (lambdaLevMarq >= maxLambda);
322
323 done = tooLong || bigLambda;
324 done = done || (settings.checkMinGradient && smallGradient);
325 done = done || (settings.checkStepNorm && smallStep);
326 done = done || (settings.checkRelEnergyChange && smallEnergyDelta);
327 done = done || (smallEnergy);
328 }
329
330 // calc jacobian for next iteration
331 if (!done)
332 {
333 double dummy;
334 if (!backend->calcFunc(dummy, /*calcEnergy*/ false, /*calcJacobian*/ true))
335 {
336 CV_LOG_INFO(NULL, "Error while calculating jacobian");
337 return LevMarq::Report(false, iter, oldEnergy); // not found
338 }
339 }
340 }
341
342 bool found = (smallGradient || smallStep || smallEnergyDelta || smallEnergy);
343
344 CV_LOG_DEBUG(NULL, "Finished: " << (found ? "" : "not ") << "found");
345 std::string fr = "Finish reason: ";
346 if (settings.checkMinGradient && smallGradient)
347 {
348 CV_LOG_DEBUG(NULL, fr + "gradient max val dropped below threshold");
349 }
350 if (settings.checkStepNorm && smallStep)
351 {
352 CV_LOG_DEBUG(NULL, fr + "step size dropped below threshold");
353 }
354 if (settings.checkRelEnergyChange && smallEnergyDelta)
355 {
356 CV_LOG_DEBUG(NULL, fr + "relative energy change between iterations dropped below threshold");
357 }
358 if (smallEnergy)
359 {
360 CV_LOG_DEBUG(NULL, fr + "energy dropped below threshold");
361 }
362 if (tooLong)
363 {
364 CV_LOG_DEBUG(NULL, fr + "max number of iterations reached");
365 }
366 if (bigLambda)
367 {
368 CV_LOG_DEBUG(NULL, fr + "lambda has grown above the threshold, the trust region is too small");
369 }
370
371 return LevMarq::Report(found, iter, oldEnergy);
372}

559 static void subMatrix(const Mat_<double>& src, Mat_<double>& dst, const Mat_<uchar>& mask)
560 {
561 CV_Assert(src.type() == CV_64F && dst.type() == CV_64F);
562 int m = src.rows, n = src.cols;
563 int i1 = 0, j1 = 0;
564 for (int i = 0; i < m; i++)
565 {
566 if (mask(i))
567 {
568 const double* srcptr = src[i];
569 double* dstptr = dst[i1++];
570
571 for (int j = j1 = 0; j < n; j++)
572 {
573 if (n < m || mask(j))
574 dstptr[j1++] = srcptr[j];
575 }
576 }
577 }

580 virtual bool calcFunc(double& energy, bool calcEnergy = true, bool calcJacobian = false) CV_OVERRIDE
581 {
582 Mat_<double> xd = probeX;
583
584 double sd = 0.0;
585 if (calcJacobian)
586 {
587 jtbFull.setZero();
588 jtjFull.setZero();
589
590 if (!cb_alt)
591 {
592 jLong.setZero();
593 }
594 }
595
596 if (cb_alt)
597 {
598 bool r = calcJacobian ? cb_alt(xd, jtbFull, jtjFull, sd) : cb_alt(xd, noArray(), noArray(), sd);
599 if (!r)
600 return false;
601 }
602 else
603 {
604 bLong.setZero();
605 bool r = calcJacobian ? cb(xd, bLong, jLong) : cb(xd, bLong, noArray());
606 if (!r)
607 return false;
608 }
609
610 if (calcJacobian)
611 {
612 if (cb_alt)
613 {
614 completeSymm(jtjFull, LtoR);
615 }
616 else
617 {
618 mulTransposed(jLong, jtjFull, true);
619 gemm(jLong, bLong, 1, noArray(), 0, jtbFull, GEMM_1_T);
620 }
621 }
622
623 if (calcEnergy)
624 {
625 if (cb_alt)
626 {
627 energy = sd;
628 }
629 else
630 {
631 energy = norm(bLong, NORM_L2SQR);
632 }
633 }
634
635 if (!mask.empty())
636 {
637 subMatrix(jtjFull, jtj, mask);
638 subMatrix(jtbFull, jtb, mask);
639 }
640 else
641 {
642 jtj = jtjFull;
643 jtb = jtbFull;
644 }
645
646 return true;
647 }

664 virtual void doJacobiScaling(const Mat_<double>& di) CV_OVERRIDE
665 {
666 // J := J * d_inv, d_inv = make_diag(di)
667 // J^T*J := (J * d_inv)^T * J * d_inv = diag(di) * (J^T * J) * diag(di) = eltwise_mul(J^T*J, di*di^T)
668 // J^T*b := (J * d_inv)^T * b = d_inv^T * J^T*b = eltwise_mul(J^T*b, di)
669 // scaling J^T*J
670 for (int i = 0; i < (int)nVars; i++)
671 {
672 double* jtjrow = jtj.ptr<double>(i);
673 for (int j = 0; j < (int)nVars; j++)
674 {
675 jtjrow[j] *= di(i) * di(j);
676 }
677 }
678 // scaling J^T*b

712 for (int i = 0; i < (int)allVars; i++)
713 {
714 if (mask.empty() || mask(i))

716 jLong.col(i).copyTo(jLongFiltered.col(ctr));

718 }
719 }

721 jtbv = jLongFiltered.t() * b_v;
722 return true;
723 }
