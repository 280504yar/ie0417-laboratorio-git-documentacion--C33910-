36Polyfit::Polyfit(Mat x, Mat y, int deg_)
37 : deg(deg_)
38{
39 int n = x.cols * x.rows * x.channels();
40 x = x.reshape(1, n);
41 y = y.reshape(1, n);
42 Mat_<double> A = Mat_<double>::ones(n, deg + 1);
43 for (int i = 0; i < n; ++i)
44 {
45 for (int j = 1; j < A.cols; ++j)
46 {
47 A.at<double>(i, j) = x.at<double>(i) * A.at<double>(i, j - 1);
48 }
49 }
50 Mat y_(y);
51 cv::solve(A, y_, p, DECOMP_SVD);
52}

59double Polyfit::fromEW(double x)
60{
61 double res = 0;
62 for (int d = 0; d <= deg; ++d)
63 {
64 res += pow(x, d) * p.at<double>(d, 0);
65 }
66 return res;
67};

92LogPolyfit::LogPolyfit(Mat x, Mat y, int deg_)
93 : deg(deg_)
94{
95 Mat mask_ = (x > 0) & (y > 0);
96 Mat src_, dst_, s_, d_;
97 src_ = maskCopyTo(x, mask_);
98 dst_ = maskCopyTo(y, mask_);
99 log(src_, s_);
100 log(dst_, d_);
101 p = Polyfit(s_, d_, deg);
102}

104Mat LogPolyfit::operator()(const Mat& inp)
105{
106 Mat mask_ = inp >= 0;
107 Mat y, y_, res;
108 log(inp, y);
109 y = p(y);
110 exp(y, y_);
111 y_.copyTo(res, mask_);
112 return res;
113};

253std::shared_ptr<Linear> getLinear(double gamma, int deg, Mat src, Color dst, Mat mask, RGBBase_ cs, LinearizationType linearizationType)
254{
255 std::shared_ptr<Linear> p = std::make_shared<Linear>();
256 switch (linearizationType)
257 {
258 case cv::ccm::LINEARIZATION_IDENTITY:
259 p = std::make_shared<LinearIdentity>();
260 break;
261 case cv::ccm::LINEARIZATION_GAMMA:
262 p = std::make_shared<LinearGamma>(gamma);
263 break;
264 case cv::ccm::LINEARIZATION_COLORPOLYFIT:
265 p = std::make_shared<LinearColor<Polyfit>>(deg, src, dst, mask, cs);
266 break;
267 case cv::ccm::LINEARIZATION_COLORLOGPOLYFIT:
268 p = std::make_shared<LinearColor<LogPolyfit>>(deg, src, dst, mask, cs);
269 break;
270 case cv::ccm::LINEARIZATION_GRAYPOLYFIT:
271 p = std::make_shared<LinearGray<Polyfit>>(deg, src, dst, mask, cs);
272 break;
273 case cv::ccm::LINEARIZATION_GRAYLOGPOLYFIT:
274 p = std::make_shared<LinearGray<LogPolyfit>>(deg, src, dst, mask, cs);
275 break;
276 default:
277 CV_Error(Error::StsBadArg, "Wrong linearizationType!" );
278 break;
279 }
280 return p;
281};
