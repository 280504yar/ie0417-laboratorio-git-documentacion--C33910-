113 virtual void setWeights(InputArray weights_arr, InputArray bias_arr,
114 int C0, int accuracy) CV_OVERRIDE
115 {
116 Mat weights_ = weights_arr.getMat();
117 Mat bias_ = bias_arr.getMat();
118 CV_Assert(!weights_.empty());
119 int wtype0 = weights_.type();
120 CV_Assert(wtype0 == CV_32F || wtype0 == CV_16F || wtype0 == CV_16BF);
121 CV_Assert(accuracy == -1 || accuracy == CV_32F);
122 int wtype = accuracy < 0 ? CV_32F : accuracy;
123
124 wshape0 = weights_.shape();
125#ifdef HAVE_CUDA
126 // Retain the original NCHW filter for the CUDA (cuDNN) path.
127 weights_.convertTo(origWeights, CV_32F);
128#endif
129 bool depthwise = ngroups == wshape0[0] && wshape0[1] == 1;
130
131 if (depthwise) {
132 repackDepthwiseConvWeights(weights_, weights, wtype, C0);
133 } else {
134 repackConvWeights(weights_, weights, wtype, ngroups, C0);
135 }
136
137 if (!bias_.empty()) {
138 CV_Assert(bias_.isContinuous() && bias_.total() == wshape0[0]);
139 bias_.convertTo(bias, CV_32F);
140 }
141
142 // Pre-pack for MLAS 1x1 SGEMM (1x1 dense, stride-1, FP32, Cout*Cin
143 // >= 256*256 so the reorder cost is amortized).
144 mlas_packed_B_.release();
145 mlas_packed_M_ = mlas_packed_K_ = 0;
146 if (!depthwise && ngroups == 1 && wtype0 == CV_32F &&
147 wtype == CV_32F && mlasAvailable())
148 {
149 bool ksize_all_one = wshape0.dims >= 3;
150 for (int k = 2; k < wshape0.dims; k++)
151 if (wshape0[k] != 1) { ksize_all_one = false; break; }
152 bool strides_all_one = !strides.empty();
153 for (int s : strides) if (s != 1) { strides_all_one = false; break; }
154 const int Cout = wshape0[0];
155 const int Cin = wshape0[1];
156 const bool big_enough = (int64_t)Cout * (int64_t)Cin >= (int64_t)(256 * 256);
157 if (ksize_all_one && strides_all_one && big_enough) {
158 size_t pack_bytes = mlasSgemmPackBSize(false, true, Cout, Cin);
159 if (pack_bytes > 0 && pack_bytes <= (size_t)INT_MAX) {
160 mlas_packed_B_.create(1, (int)pack_bytes, CV_8U);
161 // Weight is (Cout, Cin, 1, ..., 1) contiguous; reshape to
162 // (Cout, Cin) and PackB with trans_b=true.
163 Mat W2D = weights_.reshape(1, Cout);
164 if (!mlasSgemmPackB(false, true, Cout, Cin,
165 W2D.ptr<float>(), Cin,
166 mlas_packed_B_.data))
167 {
168 mlas_packed_B_.release();
169 } else {
170 mlas_packed_M_ = Cout;
171 mlas_packed_K_ = Cin;
172 }
173 }
174 }
175 }

178 void fuseBatchNormWeights(const BatchNorm2Layer* bn)
179 {
180 Mat bn_scale, bn_bias;
181 bn->getScaleBias(bn_scale, bn_bias);
182
183 CV_Assert(bn_scale.isContinuous() && bn_bias.isContinuous());
184 CV_Assert(bn_scale.type() == CV_32F && bn_bias.type() == CV_32F);
185 CV_Assert(bn_scale.total() == bn_bias.total());
186 int K = (int)bn_scale.total();
187 CV_Assert(bias.empty() || (bias.type() == CV_32F && bias.total() == (size_t)K));
188 const float* bias_data = bias.data ? bias.ptr<float>() : nullptr;
189
190 fusedScale.fit(1, &K, CV_32F);
191 fusedBias.fit(1, &K, CV_32F);
192
193 const float* bn_scale_data = bn_scale.ptr<float>();
194 const float* bn_bias_data = bn_bias.ptr<float>();
195 float* fused_scale_data = fusedScale.ptr<float>();
196 float* fused_bias_data = fusedBias.ptr<float>();
197
198 // (sum(x*w) + bias)*bn_scale + bn_bias => sum(x*w)*fused_scale + fused_bias,
199 // where fused_scale = bn_scale and fused_bias = bias*bn_scale + bn_bias.
200 for (size_t i = 0; i < K; i++) {
201 fused_scale_data[i] = bn_scale_data[i];
202 fused_bias_data[i] = (bias_data ? bn_scale_data[i]*bias_data[i] : 0.f) + bn_bias_data[i];
203 }
204 }

206 virtual bool fuseBatchNorm(const Ptr<Layer>& bnlayer) override
207 {
208 BatchNorm2Layer* bn = dynamic_cast<BatchNorm2Layer*>(bnlayer.get());
209 // addResidual means the graph order is conv->add->bn: the residual is added
210 // before BN, but the conv applies the fused BN scale only to conv(x), not to
211 // the residual. Refuse so BN stays separate and runs on (conv + residual).
212 if (fusedBatchNorm || !bn || bn->inputs.size() > 1 ||
213 fastActivation != FAST_ACTIV_NONE || !activ.empty() || addResidual)
214 return false;
215 fuseBatchNormWeights(bn);
216 fusedBatchNorm = true;
217 return true;
218 }

220 virtual bool fuseAddBias(InputArray arr) CV_OVERRIDE
221 {
222 if (inputs.size() > 1 || fusedBatchNorm || addResidual)
223 return false;
224 Mat new_bias = arr.getMat();
225 CV_Assert(new_bias.isContinuous() && new_bias.dims == 1);
226 if (new_bias.type() != CV_32F) {
227 Mat temp;
228 new_bias.convertTo(temp, CV_32F);
229 new_bias = temp;
230 CV_Assert(new_bias.type() == CV_32F);
231 }
232 if (!bias.empty()) {
233 CV_Assert(bias.shape() == new_bias.shape());
234 add(bias, new_bias, bias);
235 } else {
236 new_bias.copyTo(bias);
237 }
238 return true;

241 virtual bool fuseActivation(const Ptr<Layer>& activlayer) override
242 {
243 ActivationLayer* activ_ptr = dynamic_cast<ActivationLayer*>(activlayer.get());
244 if (!activ_ptr || fastActivation != FAST_ACTIV_NONE ||
245 activationFunc != nullptr || !activ.empty())
246 return false;
247
248 ReLULayer* activRelu = dynamic_cast<ReLULayer*>(activ_ptr);
249 ReLU6Layer* activClip = dynamic_cast<ReLU6Layer*>(activ_ptr);
250 ChannelsPReLULayer* activPRelu = dynamic_cast<ChannelsPReLULayer*>(activ_ptr);
251 if (activRelu) {
252 float alpha = activRelu->negativeSlope;
253 if (alpha == 0.f) {
254 fastActivation = FAST_ACTIV_RELU;
255 } else {
256 fastActivation = FAST_ACTIV_LEAKY_RELU;
257 activParams = {alpha};
258 }
259 } else if (activClip && activClip->minValue == 0.f) {
260 fastActivation = FAST_ACTIV_CLIP;
261 activParams = {activClip->minValue, activClip->maxValue};
262 } else if (activPRelu && activPRelu->blobs.size() == 1) {
263 fastActivation = FAST_ACTIV_PRELU;
264 const Mat& slopes = activPRelu->blobs[0];
265 int slopesType = slopes.type();
266 CV_Assert_N((slopesType == CV_32F || slopesType == CV_16F || slopesType == CV_16BF),
267 slopes.isContinuous());
268 int nslopes = int(slopes.total());
269 Mat(1, &nslopes, slopesType, (void*)slopes.data).convertTo(activParams, CV_32F);
270 } else {
271 activationFunc = activ_ptr->getActivationFunc(CV_32F, activParams);
272 if (!activationFunc)
273 return false;
274 }
275 return true;
276 }

352 void forward(InputArrayOfArrays input_arrs,
353 OutputArrayOfArrays output_arrs,
354 OutputArrayOfArrays temp_arrs) CV_OVERRIDE
355 {
356 auto* netimpl_ = getNetImpl(this);
357 std::vector<Mat>* temp_mats = &temp_arrs.getMatVecRef();
358 temp_mats->resize(2);
359 int ninputs = (int)input_arrs.total();
360 CV_Assert(ninputs >= 1);
361 const Mat& inp = input_arrs.getMat(0);
362 Mat residual;
363 const void* resptr = nullptr;
364 int inptype = inp.type();
365 MatShape inpshape = inp.shape();
366 CV_Assert(inpshape.layout == DATA_LAYOUT_BLOCK);
367 CV_Assert(inp.isContinuous());
368
369 if (addResidual) {
370 residual = input_arrs.getMat(ninputs-1);
371 resptr = residual.data;
372 ninputs--;
373 }
374
375 bool dynamicWeights = false;
376 for (int i = 1; i < ninputs; i++) {
377 if (!netimpl_->isConstArg(inputs[i]))
378 dynamicWeights = true;
379 }
380 if (dynamicWeights || weights.empty()) {
381 setWeights(input_arrs.getMat(1), ninputs > 2 ? input_arrs.getMat(2) : Mat(),
382 inpshape.back(), netimpl_->accuracy);
383 }
384
385 MatShape outshape = convInferShape(inpshape, wshape0, emptyKernelShape,
386 ngroups, strides, dilations,
387 pads, auto_pad, ceil_mode);
388 int outtype = inferType(inptype);
389 int C0 = inpshape.back();
390 int outkind = output_arrs.kind();
391 CV_Assert(outkind == _InputArray::STD_VECTOR_MAT ||
392 outkind == _InputArray::STD_VECTOR_UMAT);
393
394 if (addResidual && (residual.size != outshape || residual.type() != outtype))
395 {
396 CV_Error(Error::StsBadArg,
397 "residual added after convolution must have the same shape and the "
398 "same type as the convolution output. If this error occurs, the only "
399 "solution for now is to edit the model and add 'Expand' and/or 'Cast' "
400 "operators to make the residual tensor match the convolution shape and type");
401 }
402
403 int nspatialdims = inpshape.dims - 3;
404 CV_Assert(wshape0.dims == nspatialdims+2);
405
406 if (inpshape != prevInpshape) {
407 cs.initConv(inpshape, wshape0, outshape, ngroups,
408 strides, dilations, pads, auto_pad, ceil_mode,
409 fastActivation, activationFunc, activParams);
410 prevInpshape = inpshape;
411 }
412
413 const float* scale_data = nullptr;
414 const float* bias_data = bias.ptr<float>();
415
416 if (fusedBatchNorm) {
417 scale_data = fusedScale.ptr<float>();
418 bias_data = fusedBias.ptr<float>();
419 }
420
421 std::vector<Mat>* outs = nullptr;
422 std::vector<UMat>* uouts = nullptr;
423 Mat out;
424
425 if (outkind == _InputArray::STD_VECTOR_MAT) {
426 outs = &output_arrs.getMatVecRef();
427 outs->resize(1);
428 outs->at(0).fit(outshape, outtype);
429 out = outs->at(0);
430 } else {
431 uouts = &output_arrs.getUMatVecRef();
432 uouts->resize(1);
433 uouts->at(0).fit(outshape, outtype);
434 out.fit(outshape, outtype);
435 }
436
437 const void* inptr = inp.data;
438 void* outptr = out.data;
439 const void* wptr = weights.data;
440
441 // MLAS 1x1 SGEMM path. Skipped for small spatial: the gather/scatter
442 // tax exceeds MLAS's SGEMM speedup over the in-place NCHWc8 kernel.
443 if (mlas1x1Enabled() && inptype == CV_32F &&
444 !activationFunc && !addResidual && inpshape.back() == 8)
445 {
446 const int ndims = inpshape.dims;
447 int HW = 1;
448 for (int i = 2; i < ndims - 1; i++) HW *= inpshape[i];
449 constexpr int MLAS_MIN_SPATIAL = 256;
450 if (HW >= MLAS_MIN_SPATIAL) {
451 forwardMlas1x1(inp, out);
452 if (uouts) out.copyTo(uouts->at(0));
453 if (dynamicWeights) weights.release();
454 return;
455 }
456 }
457
458 ConvFunc func = cs.depthwise ? getDepthwiseConvFunc(inptype) : getConvFunc(inptype, C0);
459 CV_Assert(func != nullptr);
460 func(inptr, resptr, outptr, cs, wptr, scale_data, bias_data);
461
462 if (uouts) {
463 out.copyTo(uouts->at(0));
464 }
465
466 if (dynamicWeights) {
467 // to keep memory footprint low in the case of
468 // very rare situation of dynamic convolution weights,
469 // we release temporarily allocated and reordered copy of the weights
470 weights.release();
471 }
472 }

475 bool mlas1x1Enabled() const {
476 return !mlas_packed_B_.empty() && !addResidual && mlasAvailable();
477 }

479 // NCHWc8 -> row-major (M_chunk, Cin); (c1-outer, mi-inner) so each
480 // thread streams one C-block.
481 static void gatherNCHWcToNHWC(const float* inp, int C1, int HW,
482 int p_start, int p_end,
483 float* dst, int Cin)
484 {
485 const int M = p_end - p_start;
486 cv::parallel_for_(cv::Range(0, C1), [&](const cv::Range& r) {
487 for (int c1 = r.start; c1 < r.end; c1++) {
488 const int c_base = c1 * 8;
489 const int n_valid_in = std::min(8, Cin - c_base);
490 const float* src = inp + (size_t)c1 * (size_t)HW * 8
491 + (size_t)p_start * 8;
492 float* dst_col = dst + c_base;
493 int mi = 0;
494 if (n_valid_in == 8) {
495#if CV_SIMD256 && defined(__AVX2__)
496 for (; mi + 8 <= M; mi += 8) {
497 __m256 r0 = _mm256_loadu_ps(src + 0 * 8);
498 __m256 r1 = _mm256_loadu_ps(src + 1 * 8);
499 __m256 r2 = _mm256_loadu_ps(src + 2 * 8);
500 __m256 r3 = _mm256_loadu_ps(src + 3 * 8);
501 __m256 r4 = _mm256_loadu_ps(src + 4 * 8);
502 __m256 r5 = _mm256_loadu_ps(src + 5 * 8);
503 __m256 r6 = _mm256_loadu_ps(src + 6 * 8);
504 __m256 r7 = _mm256_loadu_ps(src + 7 * 8);
505 _mm256_storeu_ps(dst_col + (size_t)(mi + 0) * Cin, r0);
506 _mm256_storeu_ps(dst_col + (size_t)(mi + 1) * Cin, r1);
507 _mm256_storeu_ps(dst_col + (size_t)(mi + 2) * Cin, r2);
508 _mm256_storeu_ps(dst_col + (size_t)(mi + 3) * Cin, r3);
509 _mm256_storeu_ps(dst_col + (size_t)(mi + 4) * Cin, r4);
510 _mm256_storeu_ps(dst_col + (size_t)(mi + 5) * Cin, r5);
511 _mm256_storeu_ps(dst_col + (size_t)(mi + 6) * Cin, r6);
512 _mm256_storeu_ps(dst_col + (size_t)(mi + 7) * Cin, r7);
513 src += 8 * 8;
514 }
515#endif
516 for (; mi < M; mi++) {
517 std::memcpy(dst_col + (size_t)mi * Cin, src, 8 * sizeof(float));
518 src += 8;
519 }
520 } else {
521 const size_t copy_bytes = (size_t)n_valid_in * sizeof(float);
522 for (; mi < M; mi++) {
523 std::memcpy(dst_col + (size_t)mi * Cin, src, copy_bytes);
524 src += 8;
525 }
526 }
527 }
528 });
529 }

531 // Row-major (M_chunk, Cout) -> NCHWc8 with fused BN + bias + activation
532 // in the same pass; writes stream within one C-block.
533 void scatterAndActivate(const float* src, int M, int p_start,
534 int K1, int HW, float* out) const
535 {
536 const int Cout = mlas_packed_M_;
537 const float* biasptr = bias.empty() ? nullptr : bias.ptr<float>();
538 const float* scaleptr = fusedBatchNorm ? fusedScale.ptr<float>() : nullptr;
539 const float* fbiasptr = fusedBatchNorm ? fusedBias.ptr<float>() : biasptr;
540 const FastActivation act = fastActivation;
541 // PReLU has per-channel slopes; LEAKY_RELU broadcasts a single one.
542 const float* alphaptr = (act == FAST_ACTIV_PRELU) ? activParams.data() : nullptr;
543 const float leaky_alpha = (act == FAST_ACTIV_LEAKY_RELU) ? activParams[0] : 0.f;
544 const float clip_min = (act == FAST_ACTIV_CLIP) ? activParams[0] : -FLT_MAX;
545 const float clip_max = (act == FAST_ACTIV_CLIP) ? activParams[1] : FLT_MAX;
546
547 cv::parallel_for_(cv::Range(0, K1), [&](const cv::Range& r) {
548 for (int c1 = r.start; c1 < r.end; c1++) {
549 const int c_base = c1 * 8;
550 const int n_valid = std::min(8, Cout - c_base);
551 float* dst = out + (size_t)c1 * (size_t)HW * 8
552 + (size_t)p_start * 8;
553 const float* row = src + c_base;
554
555 CV_DECL_ALIGNED(32) float scalebuf[8] = {0};
556 CV_DECL_ALIGNED(32) float biasbuf[8] = {0};
557 CV_DECL_ALIGNED(32) float alphabuf[8] = {0};
558 for (int c0 = 0; c0 < n_valid; c0++) {
559 scalebuf[c0] = scaleptr ? scaleptr[c_base + c0] : 1.f;
560 biasbuf[c0] = fbiasptr ? fbiasptr[c_base + c0] : 0.f;
561 alphabuf[c0] = alphaptr ? alphaptr[c_base + c0] : leaky_alpha;
562 }
563
564#if CV_SIMD256 && defined(__AVX2__)
565 if (n_valid == 8) {
566 const __m256 vscale = _mm256_loadu_ps(scalebuf);
567 const __m256 vbias = _mm256_loadu_ps(biasbuf);
568 const __m256 valpha = _mm256_loadu_ps(alphabuf);
569 const __m256 vzero = _mm256_setzero_ps();
570 const __m256 vlo = _mm256_set1_ps(clip_min);
571 const __m256 vhi = _mm256_set1_ps(clip_max);
572 for (int mi = 0; mi < M; mi++) {
573 __m256 v = _mm256_loadu_ps(row + (size_t)mi * Cout);
574 if (scaleptr) {
575 v = _mm256_fmadd_ps(v, vscale, vbias);
576 } else if (fbiasptr) {
577 v = _mm256_add_ps(v, vbias);
578 }
579 if (act == FAST_ACTIV_RELU) {
580 v = _mm256_max_ps(v, vzero);
581 } else if (act == FAST_ACTIV_LEAKY_RELU || act == FAST_ACTIV_PRELU) {
582 __m256 neg = _mm256_mul_ps(v, valpha);
583 __m256 mask = _mm256_cmp_ps(v, vzero, _CMP_GE_OQ);
584 v = _mm256_blendv_ps(neg, v, mask);
585 } else if (act == FAST_ACTIV_CLIP) {
586 v = _mm256_min_ps(_mm256_max_ps(v, vlo), vhi);
587 }
588 _mm256_storeu_ps(dst + (size_t)mi * 8, v);
589 }
590 } else
591#endif
592 {
593 for (int mi = 0; mi < M; mi++) {
594 for (int c0 = 0; c0 < 8; c0++) {
595 float v = (c0 < n_valid) ? row[(size_t)mi * Cout + c0] : 0.f;
596 if (scaleptr) v = v * scalebuf[c0] + biasbuf[c0];
597 else if (fbiasptr) v = v + biasbuf[c0];
598 if (act == FAST_ACTIV_RELU) v = v > 0.f ? v : 0.f;
599 else if (act == FAST_ACTIV_LEAKY_RELU || act == FAST_ACTIV_PRELU)
600 v = v > 0.f ? v : v * alphabuf[c0];
601 else if (act == FAST_ACTIV_CLIP)
602 v = std::min(std::max(v, clip_min), clip_max);
603 dst[(size_t)mi * 8 + c0] = v;
604 }
605 }
606 }
607 }
608 });
609 if (activationFunc) {
610 float* dst = out;
611 int total = K1 * HW * 8;
612 activationFunc(dst, dst, total, activParams.data());
613 }
614 }

616 // Chunked 1x1 SGEMM-as-conv: gather NCHWc8 -> SGEMM -> scatter+activate
617 // per spatial chunk. Scratch is ~CHUNK*(Cin+Cout)*4 bytes.
618 void forwardMlas1x1(const Mat& inp, Mat& out)
619 {
620 const MatShape& inpshape = inp.shape();
621 const MatShape& outshape = out.shape();
622 const int Cout = mlas_packed_M_;
623 const int Cin = mlas_packed_K_;
624 const int ndims = inpshape.dims;
625 const int N = inpshape[0];
626 const int H = ndims >= 5 ? inpshape[ndims - 3] : 1;
627 const int W = inpshape[ndims - 2];
628 const int HW = H * W;
629 const int C1_in = (Cin + 7) / 8;
630 const int C1_out = (Cout + 7) / 8;
631
632 CV_Assert(inpshape.back() == 8 && outshape.back() == 8);
633 CV_Assert(C1_in == inpshape[1]);
634
635 const int CHUNK = std::min(HW, 1024);
636 const float* inpdata = inp.ptr<float>();
637 float* outdata = out.ptr<float>();
638
639 // Per-layer scratch, grown lazily.
640 if ((int)scratch_A_.total() < CHUNK * Cin)
641 scratch_A_.create(1, CHUNK * Cin, CV_32F);
642 if ((int)scratch_C_.total() < CHUNK * Cout)
643 scratch_C_.create(1, CHUNK * Cout, CV_32F);
644
645 const size_t in_batch_stride = (size_t)C1_in * (size_t)HW * 8;
646 const size_t out_batch_stride = (size_t)C1_out * (size_t)HW * 8;
647
648 for (int n = 0; n < N; n++) {
649 const float* inp_n = inpdata + (size_t)n * in_batch_stride;
650 float* out_n = outdata + (size_t)n * out_batch_stride;
651
652 for (int p_start = 0; p_start < HW; p_start += CHUNK) {
653 int p_end = std::min(p_start + CHUNK, HW);
654 int M = p_end - p_start;
655
656 gatherNCHWcToNHWC(inp_n, C1_in, HW, p_start, p_end,
657 scratch_A_.ptr<float>(), Cin);
658
659 // C = A @ W^T (W is (Cout, Cin), packed with trans_b=true).
660 bool ok = mlasSgemmPacked(false, true, M, Cout, Cin,
661 1.0f,
662 scratch_A_.ptr<float>(), Cin,
663 mlas_packed_B_.data,
664 0.0f,
665 scratch_C_.ptr<float>(), Cout);
666 CV_Assert(ok);
667
668 scatterAndActivate(scratch_C_.ptr<float>(), M, p_start,
669 C1_out, HW, out_n);
670 }
671 }
672 }

689 Ptr<BackendNode> initCudaConvNode(void* context_, const MatShape& inpShape,
690 const MatShape& outShape, int targetId)
691 {
692 csl::CSLContext context = *reinterpret_cast<csl::CSLContext*>(context_);
693 const int nspatial = wshape0.dims - 2;
694
695 ConvolutionConfiguration config;
696 for (int i = 0; i < nspatial; i++) {
697 config.kernel_size.push_back((size_t)wshape0[2 + i]);
698 config.strides.push_back(strides.empty() ? 1 : (size_t)strides[i]);
699 config.dilations.push_back(dilations.empty() ? 1 : (size_t)dilations[i]);
700 }
701 if (auto_pad == AUTO_PAD_VALID) {
702 config.padMode = ConvolutionConfiguration::PaddingMode::VALID;
703 } else {
704 config.padMode = ConvolutionConfiguration::PaddingMode::MANUAL;
705 for (int i = 0; i < nspatial; i++) {
706 config.pads_begin.push_back(pads.empty() ? 0 : (size_t)pads[i]);
707 config.pads_end.push_back(pads.empty() ? 0 : (size_t)pads[i + nspatial]);
708 }
709 }
710 config.input_shape.assign(inpShape.begin(), inpShape.end());
711 config.output_shape.assign(outShape.begin(), outShape.end());
712 config.groups = (size_t)ngroups;
713
714 Mat filters = origWeights, biasMat = bias;
715 if (fusedBatchNorm) {
716 filters = origWeights.clone();
717 const int Cout = wshape0[0];
718 const size_t inner = filters.total() / (size_t)Cout;
719 const float* sc = fusedScale.ptr<float>();
720 float* wp = filters.ptr<float>();
721 for (int co = 0; co < Cout; co++) {
722 float s = sc[co];
723 for (size_t k = 0; k < inner; k++)
724 wp[co * inner + k] *= s;
725 }
726 biasMat = fusedBias; // already b*scale + bn_bias
727 }
728
729 config.activation_type = ConvolutionConfiguration::ActivationType::IDENTITY;
730 config.relu_negative_slope = 0.f;
731 config.crelu_floor = 0.f; config.crelu_ceil = 0.f;
732 config.power_exp = 1.f; config.power_scale = 1.f; config.power_shift = 0.f;
733 bool hasAct = fastActivation != FAST_ACTIV_NONE;
734 if (fastActivation == FAST_ACTIV_RELU) {
735 config.activation_type = ConvolutionConfiguration::ActivationType::RELU;
736 } else if (fastActivation == FAST_ACTIV_LEAKY_RELU) {
737 config.activation_type = ConvolutionConfiguration::ActivationType::RELU;
738 config.relu_negative_slope = activParams.empty() ? 0.f : activParams[0];
739 } else if (fastActivation == FAST_ACTIV_CLIP) {
740 config.activation_type = ConvolutionConfiguration::ActivationType::CLIPPED_RELU;
741 config.crelu_floor = activParams.size() > 0 ? activParams[0] : 0.f;
742 config.crelu_ceil = activParams.size() > 1 ? activParams[1] : 6.f;
743 }
744
745 if (addResidual && hasAct)
746 config.fusion_mode = ConvolutionConfiguration::FusionMode::ELTWISE_SUM_THEN_ACTIVATION;
747 else if (addResidual)
748 config.fusion_mode = ConvolutionConfiguration::FusionMode::ELTWISE_SUM;
749 else if (hasAct)
750 config.fusion_mode = ConvolutionConfiguration::FusionMode::ACTIVATION;
751 else
752 config.fusion_mode = ConvolutionConfiguration::FusionMode::NONE;
753
754 return make_cuda_node<cuda4dnn::ConvolutionOp>(
755 targetId, std::move(context.stream), std::move(context.cudnn_handle),
756 config, filters, biasMat);
757 }
