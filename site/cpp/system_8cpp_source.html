339void Exception::formatMessage()
340{
341 size_t pos = err.find('\n');
342 bool multiline = pos != cv::String::npos;
343 if (multiline)
344 {
345 std::stringstream ss;
346 size_t prev_pos = 0;
347 while (pos != cv::String::npos)
348 {
349 ss << "> " << err.substr(prev_pos, pos - prev_pos) << std::endl;
350 prev_pos = pos + 1;
351 pos = err.find('\n', prev_pos);
352 }
353 ss << "> " << err.substr(prev_pos);
354 if (err[err.size() - 1] != '\n')
355 ss << std::endl;
356 err = ss.str();
357 }
358 if (func.size() > 0)
359 {
360 if (multiline)
361 msg = format("OpenCV(%s) %s:%d: error: (%d:%s) in function '%s'\n%s", CV_VERSION, file.c_str(), line, code, codeMessage(), func.c_str(), err.c_str());
362 else
363 msg = format("OpenCV(%s) %s:%d: error: (%d:%s) %s in function '%s'\n", CV_VERSION, file.c_str(), line, code, codeMessage(), err.c_str(), func.c_str());
364 }
365 else
366 {
367 msg = format("OpenCV(%s) %s:%d: error: (%d:%s) %s%s", CV_VERSION, file.c_str(), line, code, codeMessage(), err.c_str(), multiline ? "" : "\n");
368 }

850 void readSettings(const int* baseline_features, int baseline_count)
851 {
852 bool dump = true;
853 std::string disabled_features = utils::getConfigurationParameterString("OPENCV_CPU_DISABLE");
854 if (!disabled_features.empty())
855 {
856 const char* start = disabled_features.c_str();
857 for (;;)
858 {
859 while (start[0] != 0 && isSymbolSeparator(start[0]))
860 {
861 start++;
862 }
863 if (start[0] == 0)
864 break;
865 const char* end = start;
866 while (end[0] != 0 && !isSymbolSeparator(end[0]))
867 {
868 end++;
869 }
870 if (end == start)
871 continue;
872 cv::String feature(start, end);
873 start = end;
874
875 CV_Assert(feature.size() > 0);
876
877 bool found = false;
878 for (int i = 0; i < CV_HARDWARE_MAX_FEATURE; i++)
879 {
880 if (!g_hwFeatureNames[i]) continue;
881 size_t len = strlen(g_hwFeatureNames[i]);
882 if (len != feature.size()) continue;
883 if (feature.compare(g_hwFeatureNames[i]) == 0)
884 {
885 bool isBaseline = false;
886 for (int k = 0; k < baseline_count; k++)
887 {
888 if (baseline_features[k] == i)
889 {
890 isBaseline = true;
891 break;
892 }
893 }
894 if (isBaseline)
895 {
896 if (dump) fprintf(stderr, "OPENCV: Trying to disable baseline CPU feature: '%s'."
897 "This has very limited effect, because code optimizations for this feature are executed unconditionally "
898 "in the most cases.\n", getHWFeatureNameSafe(i));
899 }
900 if (!have[i])
901 {
902 if (dump) fprintf(stderr, "OPENCV: Trying to disable unavailable CPU feature on the current platform: '%s'.\n",
903 getHWFeatureNameSafe(i));
904 }
905 have[i] = false;
906
907 found = true;
908 break;
909 }
910 }
911 if (!found)
912 {
913 if (dump) fprintf(stderr, "OPENCV: Trying to disable unknown CPU feature: '%s'.\n", feature.c_str());
914 }
915 }
916 }
917 }

1616 void releaseThread(void* tlsValue = NULL)
1617 {
1618 TlsAbstraction* tls = getTlsAbstraction();
1619 if (NULL == tls)
1620 return; // TLS singleton is not available (terminated)
1621 ThreadData *pTD = tlsValue == NULL ? (ThreadData*)tls->getData() : (ThreadData*)tlsValue;
1622 if (pTD == NULL)
1623 return; // no OpenCV TLS data for this thread
1624 AutoLock guard(mtxGlobalAccess);
1625 for (size_t i = 0; i < threads.size(); i++)
1626 {
1627 if (pTD == threads[i])
1628 {
1629 threads[i] = NULL;
1630 if (tlsValue == NULL)
1631 tls->setData(0);
1632 std::vector<void*>& thread_slots = pTD->slots;
1633 for (size_t slotIdx = 0; slotIdx < thread_slots.size(); slotIdx++)
1634 {
1635 void* pData = thread_slots[slotIdx];
1636 thread_slots[slotIdx] = NULL;
1637 if (!pData)
1638 continue;
1639 TLSDataContainer* container = tlsSlots[slotIdx].container;
1640 if (container)
1641 container->deleteDataInstance(pData);
1642 else
1643 {
1644 fprintf(stderr, "OpenCV ERROR: TLS: container for slotIdx=%d is NULL. Can't release thread data\n", (int)slotIdx);
1645 fflush(stderr);
1646 }
1647 }
1648 delete pTD;
1649 return;
1650 }
1651 }
1652 fprintf(stderr, "OpenCV WARNING: TLS: Can't release thread TLS data (unknown pointer or data race): %p\n", (void*)pTD); fflush(stderr);
1653 }

1746 if (NULL == tls)
