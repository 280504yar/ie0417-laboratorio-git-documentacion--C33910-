70 // return minimal sample size required for non-minimal estimation.
71 int getMinimumRequiredSampleSize() const override { return 3; }
72 // return maximum number of possible solutions.
73 int getMaxNumberOfSolutions () const override { return 27; }
74#if defined(HAVE_LAPACK) || defined(HAVE_EIGEN)
75 int estimate(const std::vector<int> &sample, int sample_number,
76 std::vector<Mat> &models_, const std::vector<double> &/*weights_*/) const override {
77 if (sample_number < getMinimumRequiredSampleSize())
78 return 0;
79
80 // Estimate the model parameters from the given point sample
81 // using weighted fitting if possible.
82
83 // Holds the normalized feature positions cross multiplied with itself
84 // i.e. n * n^t. This value is used multiple times so it is efficient to
85 // pre-compute it.
86 std::vector<Matx33d> normalized_feature_cross(sample_number);
87 std::vector<Vec3d> world_points(sample_number);
88 const Matx33d eye = Matx33d::eye();
89
90 // The bottom-right symmetric block matrix of inverse(A^T * A). Matrix H from
91 // Eq. 25 in the Appendix of the DLS paper.
92 Matx33d h_inverse = sample_number * eye;
93
94 // Compute V*W*b with the rotation parameters factored out. This is the
95 // translation parameterized by the 9 entries of the rotation matrix.
96 Matx<double, 3, 9> translation_factor = Matx<double, 3, 9>::zeros();
97 const float * points = points_mat.ptr<float>();
98 const float * calib_norm_points = calib_norm_points_mat.ptr<float>();
99 for (int i = 0; i < sample_number; i++) {
100 const int idx_world = 5 * sample[i], idx_calib = 3 * sample[i];
101 Vec3d normalized_feature_pos(calib_norm_points[idx_calib],
102 calib_norm_points[idx_calib+1],
103 calib_norm_points[idx_calib+2]);
104 normalized_feature_cross[i] = normalized_feature_pos * normalized_feature_pos.t();
105 world_points[i] = Vec3d(points[idx_world + 2], points[idx_world + 3], points[idx_world + 4]);
106
107 h_inverse -= normalized_feature_cross[i];
108 translation_factor += (normalized_feature_cross[i] - eye) * leftMultiplyMatrix(world_points[i]);
109 }
110
111 const Matx33d h_matrix = h_inverse.inv();
112 translation_factor = h_matrix * translation_factor;
113
114 // Compute the cost function J' of Eq. 17 in DLS paper. This is a factorized
115 // version where the rotation matrix parameters have been pulled out. The
116 // entries to this equation are the coefficients to the cost function which is
117 // a quartic in the rotation parameters.
118 Matx<double, 9, 9> ls_cost_coefficients = Matx<double, 9, 9>::zeros();
119 for (int i = 0; i < sample_number; i++)
120 ls_cost_coefficients +=
121 (leftMultiplyMatrix(world_points[i]) + translation_factor).t() *
122 (eye - normalized_feature_cross[i]) *
123 (leftMultiplyMatrix(world_points[i]) + translation_factor);
124
125 // Extract the coefficients of the jacobian (Eq. 18) from the
126 // ls_cost_coefficients matrix. The jacobian represent 3 monomials in the
127 // rotation parameters. Each entry of the jacobian will be 0 at the roots of
128 // the polynomial, so we can arrange a system of polynomials from these
129 // equations.
130 double f1_coeff[20], f2_coeff[20], f3_coeff[20];
131 extractJacobianCoefficients(ls_cost_coefficients.val, f1_coeff, f2_coeff, f3_coeff);
132
133 // We create one equation with random terms that is generally non-zero at the
134 // roots of our system.
135 RNG rng;
136 const double macaulay_term[4] = { 100 * rng.uniform(-1.,1.), 100 * rng.uniform(-1.,1.),
137 100 * rng.uniform(-1.,1.), 100 * rng.uniform(-1.,1.) };
138
139 // Create Macaulay matrix that will be used to solve our polynonomial system.
140 Mat macaulay_matrix = Mat_<double>::zeros(120, 120);
141 createMacaulayMatrix(f1_coeff, f2_coeff, f3_coeff, macaulay_term, (double*)macaulay_matrix.data);
142
143 // Via the Schur complement trick, the top-left of the Macaulay matrix
144 // contains a multiplication matrix whose eigenvectors correspond to solutions
145 // to our system of equations.
146 Mat sol;
147 if (!solve(macaulay_matrix.colRange(27, 120).rowRange(27, 120),
148 macaulay_matrix.colRange(0 , 27).rowRange(27, 120), sol, DECOMP_LU))
149 return 0;
150
151 const Mat solution_polynomial = macaulay_matrix.colRange(0,27).rowRange(0,27) -
152 (macaulay_matrix.colRange(27,120).rowRange(0,27) * sol);
153
154 // Extract eigenvectors of the solution polynomial to obtain the roots which
155 // are contained in the entries of the eigenvectors.
156#ifdef HAVE_EIGEN
157 Eigen::Map<Eigen::Matrix<double, 27, 27>> sol_poly((double*)solution_polynomial.data);
158 const Eigen::EigenSolver<Eigen::MatrixXd> eigen_solver(sol_poly);
159 const auto &eigen_vectors = eigen_solver.eigenvectors();
160 const auto &eigen_values = eigen_solver.eigenvalues();
161#else
162
163#if defined (ACCELERATE_NEW_LAPACK) && defined (ACCELERATE_LAPACK_ILP64)
164 long mat_order = 27, info, lda = 27, ldvl = 1, ldvr = 27, lwork = 500;
165#else
166 int mat_order = 27, info, lda = 27, ldvl = 1, ldvr = 27, lwork = 500;
167#endif
168 double wr[27], wi[27] = {0}; // 27 = mat_order
169 std::vector<double> work(lwork), eig_vecs(729);
170 char jobvl = 'N', jobvr = 'V'; // only left eigen vectors are computed
171 OCV_LAPACK_FUNC(dgeev)(&jobvl, &jobvr, &mat_order, (double*)solution_polynomial.data, &lda, wr, wi, nullptr, &ldvl,
172 &eig_vecs[0], &ldvr, &work[0], &lwork, &info);
173 if (info != 0) return 0;
174#endif
175 models_ = std::vector<Mat>(); models_.reserve(3);
176 const int max_pts_to_eval = std::min(sample_number, 100);
177 std::vector<int> pts_random_shuffle(sample_number);
178 for (int i = 0; i < sample_number; i++)
179 pts_random_shuffle[i] = i;
180 randShuffle(pts_random_shuffle);
181 for (int i = 0; i < 27; i++) {
182 // If the rotation solutions are real, treat this as a valid candidate
183 // rotation.
184 // The first entry of the eigenvector should equal 1 according to our
185 // polynomial, so we must divide each solution by the first entry.
186
187#ifdef HAVE_EIGEN
188 if (eigen_values(i).imag() != 0)
189 continue;
190 const double eigen_vec_1i = 1 / eigen_vectors(0, i).real();
191 const double s1 = eigen_vectors(9, i).real() * eigen_vec_1i,
192 s2 = eigen_vectors(3, i).real() * eigen_vec_1i,
193 s3 = eigen_vectors(1, i).real() * eigen_vec_1i;
194#else
195 if (wi[i] != 0)
196 continue;
197 const double eigen_vec_1i = 1 / eig_vecs[mat_order*i];
198 const double s1 = eig_vecs[mat_order*i+9] * eigen_vec_1i,
199 s2 = eig_vecs[mat_order*i+3] * eigen_vec_1i,
200 s3 = eig_vecs[mat_order*i+1] * eigen_vec_1i;
201#endif
202 // Compute the rotation (which is the transpose rotation of our solution)
203 // and translation.
204 const double qi = s1, qi2 = qi*qi, qj = s2, qj2 = qj*qj, qk = s3, qk2 = qk*qk;
205 const double s = 1 / (1 + qi2 + qj2 + qk2);
206 Matx33d rot_mat (1-2*s*(qj2+qk2), 2*s*(qi*qj+qk), 2*s*(qi*qk-qj),
207 2*s*(qi*qj-qk), 1-2*s*(qi2+qk2), 2*s*(qj*qk+qi),
208 2*s*(qi*qk+qj), 2*s*(qj*qk-qi), 1-2*s*(qi2+qj2));
209 const Matx31d soln_translation = translation_factor * rot_mat.reshape<9,1>();
210
211 // Check that all points are in front of the camera. Discard the solution
212 // if this is not the case.
213 bool all_points_in_front_of_camera = true;
214 const Vec3d r3 (rot_mat(2,0),rot_mat(2,1),rot_mat(2,2));
215 const double z = soln_translation(2);
216 for (int pt = 0; pt < max_pts_to_eval; pt++) {
217 if (r3.dot(world_points[pts_random_shuffle[pt]]) + z < 0) {
218 all_points_in_front_of_camera = false;
219 break;
220 }
221 }
222
223 if (all_points_in_front_of_camera) {
224 // https://github.com/opencv/opencv/blob/2ba688f23c4e20754f32179d9396ba9b54b3b064/modules/calib3d/src/usac/pnp_solver.cpp#L395
225 // Use directly cv::Rodrigues
226 Matx31d rvec;
227 Rodrigues(rot_mat, rvec);
228 Rodrigues(rvec, rot_mat);
229
230 Mat model;
231 hconcat(rot_mat, soln_translation, model);
232 models_.emplace_back(K * model);
233 }
234 }
235 return static_cast<int>(models_.size());
236#else
