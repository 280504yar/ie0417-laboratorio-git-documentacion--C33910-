70int xioctl(int fd, int req, void* arg)
71{
72 int rst;
73 int retry = 5;
74 do
75 {
76 rst = ioctl(fd, req, arg);
77 retry--;
78 } while (rst == -1 && (errno == EAGAIN || (errno == EBUSY && retry > 0)));
79
80 if (rst < 0)
81 {
82 CV_LOG_WARNING(NULL, "ioctl: fd=" << fd << ", req=" << req);
83 }
84 return rst;
85}

93std::vector<UvcDeviceInfo> V4L2Context::queryUvcDeviceInfoList()
94{
95 std::vector<UvcDeviceInfo> uvcDevList;
96 std::map<std::string, UvcDeviceInfo> uvcDevMap;
97 const cv::String videosDir = "/sys/class/video4linux";
98 cv::utils::Paths videos;
99 if (cv::utils::fs::isDirectory(videosDir))
100 {
101 cv::utils::fs::glob(videosDir, "*", videos, false, true);
102 for (const auto& video : videos)
103 {
104 UvcDeviceInfo uvcDev{};
105 cv::String videoName = video.substr(video.find_last_of("/") + 1);
106 char buf[PATH_MAX];
107 if (realpath(video.c_str(), buf) == nullptr || cv::String(buf).find("virtual") != std::string::npos)
108 {
109 continue;
110 }
111 cv::String videoRealPath = buf;
112 cv::String interfaceRealPath = videoRealPath.substr(0, videoRealPath.find_last_of("/"));
113
114 std::string busNum, devNum, devPath;
115 while (videoRealPath.find_last_of("/") != std::string::npos)
116 {
117 videoRealPath = videoRealPath.substr(0, videoRealPath.find_last_of("/"));
118 if (!(std::ifstream(videoRealPath + "/busnum") >> busNum))
119 {
120 continue;
121 }
122 if (!(std::ifstream(videoRealPath + "/devpath") >> devPath))
123 {
124 continue;
125 }
126 if (!(std::ifstream(videoRealPath + "/devnum") >> devNum))
127 {
128 continue;
129 }
130 uvcDev.uid = busNum + "-" + devPath + "-" + devNum;
131 break;
132 /* code */
133 }
134
135 uvcDev.id = cv::String("/dev/") + videoName;
136 v4l2_capability caps = {};
137 int videoFd = open(uvcDev.id.c_str(), O_RDONLY);
138 IOCTL_FAILED_EXEC(xioctl(videoFd, VIDIOC_QUERYCAP, &caps), {
139 close(videoFd);
140 continue;
141 });
142 close(videoFd);
143
144 if (caps.capabilities & V4L2_CAP_VIDEO_CAPTURE)
145 {
146 cv::String modalias;
147 if (!(std::ifstream(video + "/device/modalias") >> modalias) ||
148 modalias.size() < 14 ||
149 modalias.substr(0, 5) != "usb:v" ||
150 modalias[9] != 'p')
151 {
152 continue;
153 }
154 std::istringstream(modalias.substr(5, 4)) >> std::hex >> uvcDev.vid;
155 std::istringstream(modalias.substr(10, 4)) >> std::hex >> uvcDev.pid;
156 std::ifstream iface(video + "/device/interface");
157 std::getline(iface, uvcDev.name);
158 std::ifstream(video + "/device/bInterfaceNumber") >> uvcDev.mi;
159 uvcDevMap.insert({ interfaceRealPath, uvcDev });
160 }
161 }
162 }
163 for (const auto& item : uvcDevMap)
164 {
165 const auto uvcDev = item.second; // alias
166 CV_LOG_INFO(NULL, "UVC device found: name=" << uvcDev.name << ", vid=" << uvcDev.vid << ", pid=" << uvcDev.pid << ", mi=" << uvcDev.mi << ", uid=" << uvcDev.uid << ", id=" << uvcDev.id);
167 uvcDevList.push_back(uvcDev);
168 }
169 return uvcDevList;
170}

204void V4L2StreamChannel::start(const StreamProfile& profile, FrameCallback frameCallback)
205{
206 if (streamState_ != STREAM_STOPPED)
207 {
208 CV_LOG_ERROR(NULL, devInfo_.id << ": repetitive operation!")
209 return;
210 }
211 frameCallback_ = frameCallback;
212 currentProfile_ = profile;
213
214 struct v4l2_format fmt = {};
215 fmt.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
216 fmt.fmt.pix.width = profile.width;
217 fmt.fmt.pix.height = profile.height;
218 fmt.fmt.pix.pixelformat = frameFormatToFourcc(profile.format);
219 IOCTL_FAILED_RETURN(xioctl(devFd_, VIDIOC_S_FMT, &fmt));
220 IOCTL_FAILED_RETURN(xioctl(devFd_, VIDIOC_G_FMT, &fmt));
221
222 struct v4l2_streamparm streamParm = {};
223 streamParm.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
224 streamParm.parm.capture.timeperframe.numerator = 1;
225 streamParm.parm.capture.timeperframe.denominator = profile.fps;
226 IOCTL_FAILED_RETURN(xioctl(devFd_, VIDIOC_S_PARM, &streamParm));
227 IOCTL_FAILED_RETURN(xioctl(devFd_, VIDIOC_G_PARM, &streamParm));
228
229 struct v4l2_requestbuffers req = {};
230 req.count = MAX_FRAME_BUFFER_NUM;
231 req.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
232 req.memory = V4L2_MEMORY_MMAP;
233 IOCTL_FAILED_RETURN(xioctl(devFd_, VIDIOC_REQBUFS, &req));
234
235 for (uint32_t i = 0; i < req.count && i < MAX_FRAME_BUFFER_NUM; i++)
236 {
237 struct v4l2_buffer buf = {};
238 buf.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
239 buf.memory = V4L2_MEMORY_MMAP;
240 buf.index = i; // only one buffer
241 IOCTL_FAILED_RETURN(xioctl(devFd_, VIDIOC_QUERYBUF, &buf));
242 frameBuffList[i].ptr = (uint8_t*)mmap(NULL, buf.length, PROT_READ | PROT_WRITE, MAP_SHARED, devFd_, buf.m.offset);
243 frameBuffList[i].length = buf.length;
244 }
245
246 // stream on
247 std::unique_lock<std::mutex> lk(streamStateMutex_);
248 streamState_ = STREAM_STARTING;
249 uint32_t type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
250 IOCTL_FAILED_EXEC(xioctl(devFd_, VIDIOC_STREAMON, &type), {
251 streamState_ = STREAM_STOPPED;
252 for (uint32_t i = 0; i < MAX_FRAME_BUFFER_NUM; i++)
253 {
254 if (frameBuffList[i].ptr)
255 {
256 munmap(frameBuffList[i].ptr, frameBuffList[i].length);
257 frameBuffList[i].ptr = nullptr;
258 frameBuffList[i].length = 0;
259 }
260 }
261 return;
262 });
263 grabFrameThread_ = std::thread(&V4L2StreamChannel::grabFrame, this);
264}

266void V4L2StreamChannel::grabFrame()
267{
268 fd_set fds;
269 FD_ZERO(&fds);
270 FD_SET(devFd_, &fds);
271 struct timeval tv = {};
272 tv.tv_sec = 0;
273 tv.tv_usec = 100000; // 100ms
274
275 struct v4l2_buffer buf = {};
276 buf.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
277 buf.memory = V4L2_MEMORY_MMAP;
278 // buf.index = 0;
279
280 IOCTL_FAILED_EXEC(xioctl(devFd_, VIDIOC_QBUF, &buf), {
281 std::unique_lock<std::mutex> lk(streamStateMutex_);
282 streamState_ = STREAM_STOPPED;
283 streamStateCv_.notify_all();
284 return;
285 });
286
287 while (streamState_ == STREAM_STARTING || streamState_ == STREAM_STARTED)
288 {
289 IOCTL_FAILED_CONTINUE(select(devFd_ + 1, &fds, NULL, NULL, &tv));
290 IOCTL_FAILED_CONTINUE(xioctl(devFd_, VIDIOC_DQBUF, &buf));
291 if (streamState_ == STREAM_STARTING)
292 {
293 std::unique_lock<std::mutex> lk(streamStateMutex_);
294 streamState_ = STREAM_STARTED;
295 streamStateCv_.notify_all();
296 }
297 Frame fo = { currentProfile_.format, currentProfile_.width, currentProfile_.height, buf.length, frameBuffList[buf.index].ptr };
298 if (depthFrameProcessor_)
299 {
300 depthFrameProcessor_->process(&fo);
301 }
302 frameCallback_(&fo);
303 IOCTL_FAILED_CONTINUE(xioctl(devFd_, VIDIOC_QBUF, &buf));
304 }
305 std::unique_lock<std::mutex> lk(streamStateMutex_);
306 streamState_ = STREAM_STOPPED;
307 streamStateCv_.notify_all();
308}

310bool V4L2StreamChannel::setXu(uint8_t ctrl, const uint8_t* data, uint32_t len)
311{
312 if (xuSendBuf_.size() < XU_MAX_DATA_LENGTH) {
313 xuSendBuf_.resize(XU_MAX_DATA_LENGTH);
314 }
315 memcpy(xuSendBuf_.data(), data, len);
316 struct uvc_xu_control_query xu_ctrl_query = {
317 .unit = xuUnit_.unit,
318 .selector = ctrl,
319 .query = UVC_SET_CUR,
320 .size = (__u16)(ctrl == 1 ? 512 : (ctrl == 2 ? 64 : 1024)),
321 .data = xuSendBuf_.data()
322 };
323 if (devFd_ > 0)
324 {
325 IOCTL_FAILED_EXEC(xioctl(devFd_, UVCIOC_CTRL_QUERY, &xu_ctrl_query), { return false; });
326 }
327 return true;
328}

353void V4L2StreamChannel::stop()
354{
355 if (streamState_ == STREAM_STARTING || streamState_ == STREAM_STARTED)
356 {
357 streamState_ = STREAM_STOPPING;
358 std::unique_lock<std::mutex> lk(streamStateMutex_);
359 streamStateCv_.wait_for(lk, std::chrono::milliseconds(1000), [&](){
360 return streamState_ == STREAM_STOPPED;
361 });
362 uint32_t type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
363 IOCTL_FAILED_LOG(xioctl(devFd_, VIDIOC_STREAMOFF, &type));
364 }
365 if (grabFrameThread_.joinable())
366 {
367 grabFrameThread_.join();
368 }
369 for (uint32_t i = 0; i < MAX_FRAME_BUFFER_NUM; i++)
370 {
371 if (frameBuffList[i].ptr)
372 {
373 munmap(frameBuffList[i].ptr, frameBuffList[i].length);
374 frameBuffList[i].ptr = nullptr;
375 frameBuffList[i].length = 0;
376 }
377 }
