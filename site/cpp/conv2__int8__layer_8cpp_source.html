18static MatShape getWpackShapeInt8(const MatShape& wshape, int ngroups, int C0)
19{
20 CV_Assert(wshape.dims >= 3);
21 int K = wshape[0], Cg = wshape[1];
22 int ksize = (int)(wshape.total()) / (K * Cg);
23 CV_Assert(K % ngroups == 0);
24 int Kg = K / ngroups, K0 = C0;
25 int Kblk = (Kg + K0 - 1) / K0;
26 int C1Max = 0;
27 for (int g = 0; g < ngroups; ++g) {
28 int c_start = g * Cg;
29 int c00 = c_start & (C0 - 1);
30 int cblocks = (c00 + Cg + C0 - 1) / C0;
31 C1Max = std::max(C1Max, cblocks);
32 }
33 return MatShape({ngroups, Kblk, ksize, C1Max, C0 * K0}, DATA_LAYOUT_UNKNOWN);
34}

36static void repackConvWeightsInt8(const Mat& weights, Mat& Wpack, int ngroups, int C0_)
37{
38 CV_Assert(weights.isContinuous());
39 CV_Assert(weights.type() == CV_8SC1);
40 CV_Assert(ngroups > 0);
41 CV_Assert((C0_ & (C0_ - 1)) == 0 && C0_ >= 4);
42
43 MatShape wshape = weights.shape();
44 CV_Assert(wshape.dims >= 3);
45
46 int K = wshape[0];
47 CV_Assert(K % ngroups == 0);
48
49 if (!Wpack.isContinuous())
50 Wpack.release();
51
52 MatShape wpackShape = getWpackShapeInt8(wshape, ngroups, C0_);
53 Wpack.create(wpackShape, CV_8SC1);
54 Wpack.setZero();
55
56 parallel_for_(Range(0, K), [&](const Range& range) {
57 int Cg = wshape[1], Kg = K / ngroups;
58 int ksize = wpackShape[2], C1Max = wpackShape[3];
59 int C0 = C0_, K0 = C0;
60 const int8_t* wdata = weights.ptr<int8_t>();
61 int8_t* Wpackdata = Wpack.ptr<int8_t>();
62
63 for (int k = range.start; k < range.end; ++k) {
64 int g = k / Kg;
65 int kin = k - g * Kg;
66 int kblk = kin / K0;
67 int k0 = kin & (K0 - 1);
68
69 int c_start = g * Cg;
70 int c00 = c_start & (C0 - 1);
71
72 for (int c = 0; c < Cg; ++c) {
73 int ch = c00 + c;
74 int c1 = ch / C0;
75 int c0 = ch & (C0 - 1);
76
77 const int8_t* wptr = wdata + ((k * Cg + c) * ksize);
78 int8_t* wpackptr = Wpackdata +
79 (((g * (int64_t)wpackShape[1] + kblk) * ksize * C1Max + c1) * C0 + c0) * K0 + k0;
80 for (int i = 0; i < ksize; ++i) {
81 wpackptr[i * (C1Max * C0 * K0)] = wptr[i];
82 }
83 }
84 }
85 });
86}

88static void repackWeightsForVNNI(const Mat& wpack, int ngroups, int Kg, int Cg,
89 const Mat& biasInt32,
90 Mat& wpackVNNI, Mat& biasVNNI)
91{
92 constexpr int C0 = 8, K0 = 8;
93 MatShape ws = wpack.shape();
94 int Kblk = ws[1], ksize = ws[2], C1Max = ws[3];
95
96 wpackVNNI.create(ws, CV_8SC1);
97 const int8_t* src = wpack.ptr<int8_t>();
98 int8_t* dst = wpackVNNI.ptr<int8_t>();
99
100 int blockSize = C0 * K0;
101 int totalBlocks = (int)(ws.total() / blockSize);
102
103 parallel_for_(Range(0, totalBlocks), [&](const Range& range) {
104 for (int b = range.start; b < range.end; b++) {
105 const int8_t* s = src + (size_t)b * blockSize;
106 int8_t* d = dst + (size_t)b * blockSize;
107 for (int k = 0; k < K0; k++) {
108 d[k*4 + 0] = s[0*K0 + k];
109 d[k*4 + 1] = s[1*K0 + k];
110 d[k*4 + 2] = s[2*K0 + k];
111 d[k*4 + 3] = s[3*K0 + k];
112 d[32 + k*4 + 0] = s[4*K0 + k];
113 d[32 + k*4 + 1] = s[5*K0 + k];
114 d[32 + k*4 + 2] = s[6*K0 + k];
115 d[32 + k*4 + 3] = s[7*K0 + k];
116 }
117 }
118 });
119
120 int K = ngroups * Kg;
121 biasVNNI.create({K}, CV_32SC1);
122 const int32_t* bsrc = biasInt32.ptr<int32_t>();
123 int32_t* bdst = biasVNNI.ptr<int32_t>();
124
125 parallel_for_(Range(0, K), [&](const Range& range) {
126 for (int k = range.start; k < range.end; k++) {
127 int g = k / Kg;
128 int kin = k - g * Kg;
129 int kblk = kin / K0;
130 int k0 = kin & (K0 - 1);
131 int c_start = g * Cg;
132 int c00 = c_start & (C0 - 1);
133 int cblocks = (c00 + Cg + C0 - 1) / C0;
134
135 const int8_t* wbase = src + (size_t)((g * Kblk + kblk) * ksize * C1Max) * C0 * K0;
136 int32_t wsum = 0;
137 for (int ki = 0; ki < ksize; ki++)
138 for (int c1 = 0; c1 < cblocks; c1++)
139 for (int c0 = 0; c0 < C0; c0++)
140 wsum += (int)wbase[((size_t)ki * C1Max + c1) * C0 * K0 + c0 * K0 + k0];
141 bdst[k] = bsrc[k] - 128 * wsum;
142 }
143 });
144}

334 void quantizeInterleaveBlock(const Mat& fp32_nchw, Mat& int8_block, int C0) const
335 {
336 MatShape nchw = fp32_nchw.shape();
337 if (nchw.layout == DATA_LAYOUT_UNKNOWN)
338 nchw.layout = DATA_LAYOUT_NCHW;
339 CV_Assert(nchw.dims >= 2);
340 int N = nchw[0], C = nchw[1];
341 int planesize = 1;
342 for (int d = 2; d < nchw.dims; d++) planesize *= nchw[d];
343 int C1 = (C + C0 - 1) / C0;
344
345 MatShape blkshape = nchw.toLayout(DATA_LAYOUT_BLOCK, C0);
346 int8_block.fit(blkshape, inputIsU8 ? CV_8UC1 : CV_8SC1);
347
348 float inv_sc = input_sc > 0.f ? 1.f / input_sc : 0.f;
349 int zp = input_zp;
350 const bool is_u8 = inputIsU8;
351 const float* src = fp32_nchw.ptr<float>();
352 uint8_t* dst = int8_block.ptr<uint8_t>();
353
354 parallel_for_(Range(0, N * C1), [&](const Range& range) {
355 for (int nc = range.start; nc < range.end; nc++) {
356 int n = nc / C1, c1 = nc % C1;
357 int c_base = c1 * C0;
358 int nzc = std::min(C0, C - c_base);
359 const float* src_block = src + (n * C + c_base) * planesize;
360 uint8_t* dst_block = dst + (size_t)(n * C1 + c1) * planesize * C0;
361 for (int i = 0; i < planesize; i++) {
362 for (int c = 0; c < nzc; c++) {
363 float x = src_block[c * planesize + i];
364 int q = cvRound(x * inv_sc) + zp;
365 if (is_u8)
366 dst_block[i * C0 + c] = saturate_cast<uchar>(q);
367 else
368 dst_block[i * C0 + c] = (uint8_t)(int8_t)saturate_cast<schar>(q);
369 }
370 for (int c = nzc; c < C0; c++)
371 dst_block[i * C0 + c] = 0;
372 }
373 }
374 });
375 }

377 void forward(InputArrayOfArrays input_arrs,
378 OutputArrayOfArrays output_arrs,
379 OutputArrayOfArrays) CV_OVERRIDE
380 {
381 int ninputs = (int)input_arrs.total();
382 CV_Assert(ninputs >= 1);
383
384 Mat inp_storage;
385 Mat inp_ref = input_arrs.getMat(0);
386
387 if (float_input) {
388 CV_Assert(inp_ref.type() == CV_32F);
389 int C0 = getNetImpl(this)->defaultC0;
390 quantizeInterleaveBlock(inp_ref, inp_storage, C0);
391 }
392
393 const Mat& inp = float_input ? inp_storage : inp_ref;
394 MatShape inpshape = inp.shape();
395 CV_Assert(inpshape.layout == DATA_LAYOUT_BLOCK);
396 CV_Assert(inp.type() == CV_8SC1 || inp.type() == CV_8UC1);
397
398 int C0 = inpshape.back();
399
400 if (weights.empty() && !blobs.empty()) {
401 repackConvWeightsInt8(blobs[0], weights, ngroups, C0);
402 }
403
404 if (weightsVNNI.empty() && !weights.empty()) {
405 int K = wshape0[0];
406 int Cg = wshape0[1];
407 int Kg = K / ngroups;
408 repackWeightsForVNNI(weights, ngroups, Kg, Cg,
409 biasInt32, weightsVNNI, biasVNNI);
410
411 if (inputIsU8) {
412 biasU8.create({K}, CV_32SC1);
413 biasBaselineU8.create({K}, CV_32SC1);
414 const int32_t* bI = biasInt32.ptr<int32_t>();
415 const int32_t* bV = biasVNNI.ptr<int32_t>();
416 int32_t* bU = biasU8.ptr<int32_t>();
417 int32_t* bB = biasBaselineU8.ptr<int32_t>();
418 for (int k = 0; k < K; k++) {
419 int32_t wsum = (bI[k] - bV[k]) / 128;
420 bU[k] = bI[k] - input_zp * wsum;
421 bB[k] = bI[k] + (128 - input_zp) * wsum;
422 }
423 }
424 }
425
426 Mat residual;
427 const void* resptr = nullptr;
428 if (addResidual) {
429 residual = input_arrs.getMat(ninputs - 1);
430 resptr = residual.data;
431 ninputs--;
432 }
433
434 std::vector<int> emptyKernelShape;
435 MatShape outshape = convInferShape(inpshape, wshape0, emptyKernelShape,
436 ngroups, strides, dilations,
437 pads, auto_pad, ceil_mode);
438 int outtype = inp.type();
439
440 int outkind = output_arrs.kind();
441 Mat out;
442
443 if (outkind == _InputArray::STD_VECTOR_MAT) {
444 std::vector<Mat>& outs = output_arrs.getMatVecRef();
445 outs.resize(1);
446 outs[0].fit(outshape, outtype);
447 out = outs[0];
448 } else {
449 std::vector<UMat>& outs = output_arrs.getUMatVecRef();
450 outs.resize(1);
451 outs[0].fit(outshape, outtype);
452 out.fit(outshape, outtype);
453 }
454
455 if (inpshape != prevInpshape) {
456 cs.initConv(inpshape, wshape0, outshape, ngroups,
457 strides, dilations, pads, auto_pad, ceil_mode,
458 FAST_ACTIV_NONE, nullptr, {});
459
460 if (cs.depthwise) {
461 cs.wshape = getWpackShapeInt8(wshape0, ngroups, C0);
462 }
463 prevInpshape = inpshape;
464 }
465
466 const int8_t* lutptr = !activationLUT.empty() ? activationLUT.ptr<int8_t>() : nullptr;
467
468 const int* effectiveBiasVNNI = inputIsU8
469 ? (biasU8.empty() ? nullptr : biasU8.ptr<int>())
470 : (biasVNNI.empty() ? nullptr : biasVNNI.ptr<int>());
471
472 const int* baselineBias = (inputIsU8 && !biasBaselineU8.empty())
473 ? biasBaselineU8.ptr<int>()
474 : biasInt32.ptr<int>();
475
476 convInt8Block(inp.data, resptr, out.data, cs,
477 weights.data,
478 weightsVNNI.empty() ? nullptr : weightsVNNI.data,
479 baselineBias,
480 effectiveBiasVNNI,
481 outMultiplier.ptr<float>(),
482 input_zp, output_zp,
483 lutptr, inputIsU8);
484
485 if (outkind == _InputArray::STD_VECTOR_UMAT) {
486 std::vector<UMat>& outs = output_arrs.getUMatVecRef();
487 out.copyTo(outs[0]);
488 }
489 }
