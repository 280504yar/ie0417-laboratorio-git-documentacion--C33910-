27int simd_binop_f32_(const float* a, const float* b, float* out, int n, int op) {
28 int i = 0;
29#if (CV_SIMD || CV_SIMD_SCALABLE)
30 const int lanes = VTraits<v_float32>::vlanes();
31 if (op == SIMD_BIN_ADD) {
32 for (; i <= n - lanes * 4; i += lanes * 4) {
33 v_float32 a0 = vx_load(a + i), b0 = vx_load(b + i);
34 v_float32 a1 = vx_load(a + i + lanes), b1 = vx_load(b + i + lanes);
35 v_float32 a2 = vx_load(a + i + lanes * 2), b2 = vx_load(b + i + lanes * 2);
36 v_float32 a3 = vx_load(a + i + lanes * 3), b3 = vx_load(b + i + lanes * 3);
37 v_float32 r0 = v_add(a0, b0), r1 = v_add(a1, b1);
38 v_float32 r2 = v_add(a2, b2), r3 = v_add(a3, b3);
39 vx_store(out + i, r0); vx_store(out + i + lanes, r1);
40 vx_store(out + i + lanes * 2, r2); vx_store(out + i + lanes * 3, r3);
41 }
42 for (; i <= n - lanes; i += lanes)
43 vx_store(out + i, v_add(vx_load(a + i), vx_load(b + i)));
44 if (i < n && n >= lanes) {
45 i = n - lanes;
46 vx_store(out + i, v_add(vx_load(a + i), vx_load(b + i)));
47 i = n;
48 }
49 } else if (op == SIMD_BIN_MUL) {
50 for (; i <= n - lanes * 4; i += lanes * 4) {
51 v_float32 a0 = vx_load(a + i), b0 = vx_load(b + i);
52 v_float32 a1 = vx_load(a + i + lanes), b1 = vx_load(b + i + lanes);
53 v_float32 a2 = vx_load(a + i + lanes * 2), b2 = vx_load(b + i + lanes * 2);
54 v_float32 a3 = vx_load(a + i + lanes * 3), b3 = vx_load(b + i + lanes * 3);
55 v_float32 r0 = v_mul(a0, b0), r1 = v_mul(a1, b1);
56 v_float32 r2 = v_mul(a2, b2), r3 = v_mul(a3, b3);
57 vx_store(out + i, r0); vx_store(out + i + lanes, r1);
58 vx_store(out + i + lanes * 2, r2); vx_store(out + i + lanes * 3, r3);
59 }
60 for (; i <= n - lanes; i += lanes)
61 vx_store(out + i, v_mul(vx_load(a + i), vx_load(b + i)));
62 if (i < n && n >= lanes) {
63 i = n - lanes;

65 i = n;
66 }
67 } else if (op == SIMD_BIN_SUB) {
68 for (; i <= n - lanes * 4; i += lanes * 4) {
69 v_float32 a0 = vx_load(a + i), b0 = vx_load(b + i);

88 v_float32 a1 = vx_load(a + i + lanes), b1 = vx_load(b + i + lanes);

98 if (i < n && n >= lanes) {
99 i = n - lanes;
100 vx_store(out + i, v_div(vx_load(a + i), vx_load(b + i)));
101 i = n;
102 }
103 }
104 vx_cleanup();
105#endif
106 if (op == SIMD_BIN_ADD) {
107 for (; i < n; i++) out[i] = a[i] + b[i];
108 } else if (op == SIMD_BIN_MUL) {
109 for (; i < n; i++) out[i] = a[i] * b[i];
110 } else if (op == SIMD_BIN_SUB) {
111 for (; i < n; i++) out[i] = a[i] - b[i];
112 } else {
113 for (; i < n; i++) out[i] = a[i] / b[i];
114 }
115 return n;
